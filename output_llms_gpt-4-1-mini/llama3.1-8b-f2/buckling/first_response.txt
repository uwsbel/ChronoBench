import pychrono as chrono
import pychrono.irrlicht as irr
import pychrono.cascade as cascade
import math as m

# The path to the Chrono data directory containing various assets (meshes, textures, data files)
# is automatically set, relative to the default location of this demo.
# If running from a different directory, you must change the path to the data directory with: 
#chrono.SetChronoDataPath('path/to/data')

# If true, use provided callback functions for motor controls
use_callbacks = False

# =============================================================================

print ("Example: Create a beam with multiple finite elements, and use it to build a buckling demo.")

# Create the beam, with given length, cross section, material, and number of elements
beam = cascade.Beams(chrono.ChVector3d(0,0,0), chrono.ChVector3d(0,1.2,0), 
                     cascade.GetSCSBox(0.02,0.02,0.3), 
                     cascade.GetMaterialElastic(2e7,0.2), 
                     10)

# Set visualization type for the beam elements
beam.SetVisualType(cascade.Beams.VisualType_PRIMITIVES)

# Create the mechanical system
sys = chrono.ChSystemNSC()

# Add a fixed point
fixedPoint = chrono.ChBody()
fixedPoint.SetFixed(True)
sys.Add(fixedPoint)

# Add the beam to the system, with the specified root body (which must be a part of the system)
# and the specified orientation and position.
# The beam is connected to the fixed point via a rotational constraint.
beam_body = chrono.ChBody()
sys.Add(beam_body)
beam.Initialize(fixedPoint, chrono.QuatFromAngleAxis(.2, chrono.ChVector3d(1,0,0)), chrono.ChVector3d(0,0.2,0))
beam.SetColor(chrono.ChColor(0.8,0.8,0.5))
beam.SetName("my beam")

# Create a cascade demo driver to update the beam's geometry based on the finite element analysis
driver = cascade.ChCascadeDemoDriver(beam)

# Run the simulation
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024,768)
vis.SetWindowTitle('Test')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0,1.5,0), chrono.ChVector3d(0,0,0))
vis.AddTypicalLights()

# Mechanism driver (motor)
# When running this demo, the beam will buckle due to gravity. 
# This demo uses a callback function to control the rotation of the beam's root body.
# The callback function sets the desired angular velocity, and also adjusts the color of the beam 
# to indicate the actual velocity.

# Note that the callback function is not used if the demo is run without visualization.

def motor_callback(t):
    v = 0 if abs(beam.GetAngle() - 0.2) < 0.01 else -0.02
    beam_body.SetAngVelParent(chrono.ChVector3d(0,0,0))
    return v

# Create the driver for the mechanism (motor)
driver_motor = cascade.ChCascadeDriverMotor(beam_body, driver, motor_callback)

# Run the simulation
vis.Run()
