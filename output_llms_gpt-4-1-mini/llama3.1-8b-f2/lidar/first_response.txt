import pychrono as chrono
import pychrono.sensor as sens

import numpy as np
import time
import math


def main():
    # -----------------
    # Create the system
    # -----------------
    mphysicalSystem = chrono.ChSystemNSC()
    mphysicalSystem.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))

    # ----------------------------------------
    # add a mesh to be sensed (a cube in this case)
    # ----------------------------------------
    mmesh = chrono.ChTriangleMeshConnected()
    mmesh.LoadWavefrontMesh(chrono.GetChronoDataFile(
        "models/obj/porsche911/porsche911.obj"), False, True)
    mmesh.Transform(chrono.ChVector3d(0, 0, 0), chrono.ChMatrix33d(1))

    trimesh_shape = chrono.ChVisualShapeTriangleMesh()
    trimesh_shape.SetMesh(mmesh)
    trimesh_shape.SetName("Porsche 911 Mesh")
    trimesh_shape.SetMutable(False)

    mesh_body = chrono.ChBody()
    mesh_body.SetPos(chrono.ChVector3d(0, 0, 0))
    mesh_body.AddVisualShape(trimesh_shape)
    mesh_body.SetFixed(True)
    mphysicalSystem.Add(mesh_body)

    # -----------------------
    # Create a sensor manager
    # -----------------------
    manager = sens.ChSensorManager(mphysicalSystem)

    # ------------------------------------------------
    # Create a lidar and add it to the sensor manager
    # ------------------------------------------------
    offset_pose = chrono.ChFramed(
        chrono.ChVector3d(-8, 0, 1), chrono.QuatFromAngleAxis(0, chrono.ChVector3d(0, 1, 0)))
    lidar = sens.ChLidarSensor(
        mesh_body,              # body lidar is attached to
        update_rate,            # scanning rate in Hz
        offset_pose,            # offset pose
        horizontal_samples,     # number of horizontal samples
        vertical_samples,       # number of vertical channels
        horizontal_fov,         # horizontal field of view
        max_vert_angle,         # vertical field of view
        min_vert_angle,
        100.0,  # max lidar range
        sens.LidarBeamShape_RECTANGULAR,
        sample_radius,          # sample radius
        divergence_angle,       # divergence angle
        divergence_angle,       # divergence angle
        return_loss            # return lidar beam shape loss data?
    )
    lidar.SetName("Lidar Sensor")
    lidar.SetLag(lag)
    lidar.SetCollectionWindow(collection_time)

    # ------------------------------------------------------------------
    # Create a filter graph for post-processing the data from the lidar sensor
    # ------------------------------------------------------------------
    if noise_model == "CONST_NORMAL_XYZI":
        lidar.PushFilter(sens.ChFilterLidarNoiseXYZI(0.01, 0.001, 0.001, 0.01))
    elif noise_model == "NONE":
        # Don't add any noise models
        pass

    if vis:
        lidar.PushFilter(sens.ChFilterVisualize(
            horizontal_samples, vertical_samples, "Lidar Depth Data"))

    # Save lidar data to a png file at the specified path
    if save:
        lidar.PushFilter(sens.ChFilterSave(out_dir + "lidar/"))

    # Convert the lidar data to a point cloud and visualize it using PCL
    if vis_pcl:
        lidar.PushFilter(sens.ChFilterToPointCloud(
            point_cloud_radius,  # radius of point cloud samples to include
            100  # number of samples along beam
        ))
        lidar.PushFilter(sens.ChFilterPCLVisualize(100, 100, "Lidar Point Cloud"))

    # Merge the filter graph into the sensor
    lidar.ComposeFilterGraph()

    # ----------------------------------------
    # add the lidar sensor to the manager
    # ----------------------------------------
    manager.AddSensor(lidar)

    # -----------------------
    # Simulate the system
    # -----------------------
    orbit_radius = 5
    orbit_rate = 0.2
    ch_time = 0.0

    t1 = time.time()

    while (ch_time < end_time):
        lidar.SetOffsetPose(chrono.ChFramed(
            chrono.ChVector3d(-orbit_radius * math.cos(ch_time * orbit_rate), -
                             orbit_radius * math.sin(ch_time * orbit_rate), 1),
            chrono.QuatFromAngleAxis(ch_time * orbit_rate, chrono.ChVector3d(0, 0, 1))))

        # Access the lidar buffer and print the data
        lidar_buffer = lidar.GetMostRecentBuffer()
        if (lidar_buffer.HasData()):
            data = lidar_buffer.GetData()
            print('Lidar buffer, max: ', np.max(data), ' min: ', np.min(data))
        else:
            print('Lidar buffer is empty\n')

        # Update sensor manager
        # Will render/save/filter automatically
        manager.Update()

        # Perform step of dynamics
        mphysicalSystem.DoStepDynamics(step_size)

        # Get the current time of the simulation
        ch_time = mphysicalSystem.GetChTime()

    print("Sim time:", end_time, "Wall time:", time.time() - t1)

# -----------------
# Lidar parameters
# -----------------


# Noise model attached to the sensor
# TODO: Noise models haven't been implemented in python
# noise_model="CONST_NORMAL_XYZI"   # Gaussian noise with constant mean and standard deviation
noise_model = "NONE"                  # No noise model

# Update rate in Hz
update_rate = 5

# Number of horizontal and vertical samples
horizontal_samples = 1000
vertical_samples = 20

# Horizontal and vertical field of view (radians)
horizontal_fov =  chrono.CH_PI / 9  # 20 degrees
max_vert_angle = chrono.CH_PI / 12
min_vert_angle = -chrono.CH_PI / 6

# Lag time
lag = 0

# Collection window for the lidar
collection_time = 1. / update_rate  # typically 1/update rate

# Radius of samples to use, set to 0 for all
sample_radius = 0

# 3mm radius (as cited by velodyne)
divergence_angle = 0.003

# Return lidar beam shape loss data?
return_loss = False

# ---------------------
# Simulation parameters
# ---------------------

# Simulation step size
step_size = 1e-3

# Simulation end time
end_time = 20.0

# Save point cloud data
save = False

# Visualize point cloud data with PCL
vis_pcl = False

# Render camera images
vis = True

# Output directory
out_dir = "SENSOR_OUTPUT/"

# The path to the Chrono data directory containing various assets (meshes, textures, data files)
# is automatically set, relative to the default location of this demo.
# If running from a different directory, you must change the path to the data directory with:
# chrono.SetChronoDataPath('path/to/data')

main()
