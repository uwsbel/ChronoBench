# PyChrono Particle Gravitational Attraction Simulation

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import numpy as np
import random
import math

# Initialize ChronoEngine
chrono.SetChronoDataPath("../data/")
print("PyChrono version:", chrono.PYCHRONO_VERSION)

# Create the physical system
system = chrono.ChSystemNSC()

# Set gravity to zero (we'll apply our own gravitational forces)
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))

# Set solver parameters
system.SetSolverType(chrono.ChSolver.Type_PSOR)
system.SetSolverMaxIterations(100)
system.SetMaxItersSolverSpeed(100)
system.SetMaxItersSolverStab(100)
system.SetSolverForceTolerance(1e-10)

# Create ground
ground = chrono.ChBodyEasyBox(20, 1, 20, 1000, True, True)
ground.SetBodyFixed(True)
ground.SetPos(chrono.ChVectorD(0, -5, 0))
ground.SetCollide(True)
ground_mat = chrono.ChMaterialSurfaceNSC()
ground_mat.SetFriction(0.2)
ground.GetMaterialSurfaceNSC().SetFriction(0.2)
system.Add(ground)

# Materials for particles
particle_mat = chrono.ChMaterialSurfaceNSC()
particle_mat.SetFriction(0.4)
particle_mat.SetRestitution(0.6)

# Visual materials for particles
particle_textures = []
vis_mat1 = chrono.ChVisualMaterial()
vis_mat1.SetDiffuseColor(chrono.ChColor(0.8, 0.3, 0.3))
vis_mat2 = chrono.ChVisualMaterial()
vis_mat2.SetDiffuseColor(chrono.ChColor(0.3, 0.8, 0.3))
vis_mat3 = chrono.ChVisualMaterial()
vis_mat3.SetDiffuseColor(chrono.ChColor(0.3, 0.3, 0.8))
vis_mat4 = chrono.ChVisualMaterial()
vis_mat4.SetDiffuseColor(chrono.ChColor(0.8, 0.8, 0.3))

# List to store all particles
particles = []

# Gravitational constant (exaggerated for visual effect)
G = 6.67430e-2  # Much larger than the real G=6.67430e-11 m^3 kg^-1 s^-2

# Particle emitter class
class ParticleEmitter:
    def __init__(self, system, emit_position, emit_rate, particle_mass_range):
        self.system = system
        self.emit_position = emit_position
        self.emit_rate = emit_rate  # particles per second
        self.time_since_last_emit = 0
        self.emit_interval = 1.0 / emit_rate
        self.particle_mass_range = particle_mass_range  # (min_mass, max_mass)
        self.emit_radius = 5.0  # radius of emission area
        
    def update(self, dt):
        self.time_since_last_emit += dt
        if self.time_since_last_emit >= self.emit_interval:
            self.emit_particle()
            self.time_since_last_emit = 0
            
    def emit_particle(self):
        # Random position within emit radius
        angle = random.uniform(0, 2 * math.pi)
        radius = random.uniform(0, self.emit_radius)
        x = self.emit_position.x + radius * math.cos(angle)
        z = self.emit_position.z + radius * math.sin(angle)
        y = self.emit_position.y + random.uniform(-1, 1)
        
        # Random velocity
        vel_mag = random.uniform(0.5, 2.0)
        vel_angle = random.uniform(0, 2 * math.pi)
        vel_y = random.uniform(-0.5, 0.5)
        vel_x = vel_mag * math.cos(vel_angle)
        vel_z = vel_mag * math.sin(vel_angle)
        
        # Random shape
        shape_type = random.choice(['sphere', 'box', 'cylinder'])
        
        # Random mass
        mass = random.uniform(self.particle_mass_range[0], self.particle_mass_range[1])
        
        # Create particle based on shape type
        if shape_type == 'sphere':
            radius = random.uniform(0.1, 0.3)
            body = chrono.ChBodyEasySphere(radius, mass/volume_sphere(radius), True, True)
            # Add visual material
            body.GetVisualShape(0).SetMaterial(0, vis_mat1)
            
        elif shape_type == 'box':
            sx = random.uniform(0.1, 0.3)
            sy = random.uniform(0.1, 0.3)
            sz = random.uniform(0.1, 0.3)
            body = chrono.ChBodyEasyBox(sx, sy, sz, mass/volume_box(sx, sy, sz), True, True)
            # Add visual material
            body.GetVisualShape(0).SetMaterial(0, vis_mat2)
            
        else:  # cylinder
            radius = random.uniform(0.1, 0.2)
            height = random.uniform(0.2, 0.4)
            body = chrono.ChBodyEasyCylinder(radius, height, mass/volume_cylinder(radius, height), True, True)
            # Add visual material
            body.GetVisualShape(0).SetMaterial(0, vis_mat3)
        
        # Set properties
        body.SetPos(chrono.ChVectorD(x, y, z))
        body.SetPos_dt(chrono.ChVectorD(vel_x, vel_y, vel_z))
        
        # Random orientation
        rot_x = random.uniform(0, 2 * math.pi)
        rot_y = random.uniform(0, 2 * math.pi)
        rot_z = random.uniform(0, 2 * math.pi)
        rotation = chrono.Q_from_Euler123(chrono.ChVectorD(rot_x, rot_y, rot_z))
        body.SetRot(rotation)
        
        # Set material properties
        body.GetMaterialSurfaceNSC().SetFriction(0.4)
        body.GetMaterialSurfaceNSC().SetRestitution(0.6)
        
        # Add to system and particles list
        self.system.Add(body)
        particles.append(body)

# Helper functions to calculate volume
def volume_sphere(radius):
    return (4/3) * math.pi * radius**3

def volume_box(x, y, z):
    return x * y * z

def volume_cylinder(radius, height):
    return math.pi * radius**2 * height

# Create particle emitter
emitter = ParticleEmitter(
    system,
    emit_position=chrono.ChVectorD(0, 5, 0),
    emit_rate=3.0,  # particles per second
    particle_mass_range=(1.0, 5.0)  # kg
)

# Function to apply gravitational forces between particles
def apply_gravitational_forces():
    # Apply forces between each pair of particles
    for i in range(len(particles)):
        for j in range(i+1, len(particles)):
            # Get positions and masses
            pos_i = particles[i].GetPos()
            pos_j = particles[j].GetPos()
            
            # Calculate distance and direction
            dx = pos_j.x - pos_i.x
            dy = pos_j.y - pos_i.y
            dz = pos_j.z - pos_i.z
            distance = math.sqrt(dx*dx + dy*dy + dz*dz)
            
            # Skip if particles are too close (to avoid extreme forces)
            if distance < 0.1:
                continue
                
            # Calculate gravitational force magnitude: F = G * m1 * m2 / r^2
            mass_i = particles[i].GetMass()
            mass_j = particles[j].GetMass()
            force_mag = G * mass_i * mass_j / (distance * distance)
            
            # Calculate force components
            force_x = force_mag * dx / distance
            force_y = force_mag * dy / distance
            force_z = force_mag * dz / distance
            
            # Apply equal and opposite forces to both particles
            particles[i].Accumulate_force(chrono.ChVectorD(force_x, force_y, force_z), 
                                         pos_i, False)
            particles[j].Accumulate_force(chrono.ChVectorD(-force_x, -force_y, -force_z), 
                                         pos_j, False)

# Set up Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Particle Gravity Simulation")
vis.Initialize()
vis.AddCamera(chrono.ChVectorD(0, 10, -10), chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()

# Simulation loop
time_step = 0.01
time = 0
max_particles = 100  # Limit to prevent simulation slowdown

while vis.Run():
    # Update the emitter
    emitter.update(time_step)
    
    # Apply gravitational forces
    apply_gravitational_forces()
    
    # Remove particles that are too far from origin to manage memory
    particles_to_remove = []
    for i, particle in enumerate(particles):
        pos = particle.GetPos()
        dist_from_origin = math.sqrt(pos.x**2 + pos.y**2 + pos.z**2)
        if dist_from_origin > 50:
            particles_to_remove.append(i)
            system.Remove(particle)
    
    # Remove from list in reverse order to maintain correct indices
    for i in sorted(particles_to_remove, reverse=True):
        particles.pop(i)
    
    # Limit particle count
    while len(particles) > max_particles:
        system.Remove(particles[0])
        particles.pop(0)
    
    # Update visualization and system
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    system.DoStepDynamics(time_step)
    
    time += time_step
```

This script creates a particle gravitational attraction simulation with the following features:

1. A particle emitter that generates particles with:
   - Random shapes (spheres, boxes, and cylinders)
   - Random positions within a defined emission area
   - Random velocities and orientations
   - Random masses within a specified range
   - Different colors for different particle shapes

2. Custom gravitational forces:
   - Particles attract each other using a simplified gravitational model
   - Force magnitude is proportional to the product of masses and inversely proportional to the square of distance
   - The gravitational constant is exaggerated for visual effect

3. Resource management:
   - Particles that move too far from the origin are removed
   - The total number of particles is limited to prevent simulation slowdown

4. Visualization using Irrlicht:
   - 3D rendering of the particles and their interactions
   - Camera positioned to view the particle system
   - Typical lighting for better visibility

The simulation creates an interesting gravitational system where particles naturally form clusters and orbit around each other based on their masses and initial conditions.
