```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Set the path to Chrono data directory
chrono.SetChronoDataPath('/usr/local/share/chrono/data/')

# Create the physical system using NSC contact method
sys = chrono.ChSystemNSC()

# Set default collision envelope and margin for better collision detection
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.01)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.005)

# Create a material for contact properties
material = chrono.ChMaterialSurfaceNSC()
material.SetFriction(0.5)
material.SetDampingF(0.1)

# Create a fixed ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
ground.SetPos(chrono.ChVectorD(0, 0, 0))
ground.SetMaterialSurface(material)
ground.EnableCollision(True)

# Add a box shape to the ground for visualization and collision
ground_shape = chrono.ChBoxShape()
ground_shape.GetBoxGeometry().Size = chrono.ChVectorD(5, 0.1, 5)
ground.AddVisualShape(ground_shape)
ground.GetCollisionModel().ClearModel()
ground.GetCollisionModel().AddBox(5, 0.1, 5)
ground.GetCollisionModel().BuildModel()

sys.Add(ground)

# Create a dynamic sphere body
sphere = chrono.ChBodyEasySphere(0.5, 1000, True, True, material)
sphere.SetPos(chrono.ChVectorD(0, 2, 0))
sys.Add(sphere)

# Initialize the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('PyChrono Sphere Drop Simulation')
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 3, 6))
vis.AddTypicalLights()

# Simulation loop parameters
time_step = 0.01
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(time_step)
```
