import pychrono.irrlicht as irr

system = chrono.ChSystemNSC()

application.SetLights()
application.SetCamera(irr.vector3df(0, 1, 3))
application.AssetBindAll()
application.AssetUpdateAll()

ground = chrono.ChBodyEasyBox(2, 0.1, 2, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -0.05, 0))
ground.SetBodyFixed(True)
system.Add(ground)

radius = 0.1  # meters
inertia = chrono.ChMatrix33D()
# For a solid sphere, inertia can be approximated as (2/5)*m*r^2

pendulum.SetPos(chrono.ChVectorD(0, -1, 0))

pendulum.SetRot(chrono.Q_from_AngX(initial_angle_rad))
# Optional: give an initial angular velocity

# 7. Configure the simulation parameters
system.SetSolverType(chrono.ChSolver.SolverType_SOR)
system.SetMaxItersSolverSpeed(100)
system.SetTimestep(0.001)

application.AssetBindAll()
application.AssetUpdateAll()

# 9. Simulation loop with periodic logging
import math
import time

print("Starting simulation...")
simulation_time = 0
end_time = 10  # seconds
log_interval = 0.1  # seconds
next_log_time = 0

while application.GetDevice().run() and simulation_time < end_time:
    # Advance the simulation
    system.DoStepDynamics(system.GetTimestep())
    simulation_time += system.GetTimestep()

    if simulation_time >= next_log_time:

        pos = pendulum.GetPos()

    application.BeginScene()
    application.DrawAll()
    application.EndScene()
