```python
import pychrono.core as chrono
import pychrono.irrlicht as irr

# Initialize Chrono system
system = chrono.ChSystemNSC()

# Create the Irrlicht application for visualization
application = irr.ChIrrApp(system, "Mass-Spring-Damper System", irr.dimension2d(1024, 768))
application.AddTypicalLights()
application.AddTypicalCamera(irr.vector3df(0, 0.5, 1.5))
application.AssetBindAll()
application.AssetUpdateAll()

# Create ground (fixed body)
ground = chrono.ChBodyEasyBox(0.2, 0.02, 0.2, 1000, True, True)
ground.SetBodyFixed(True)
system.Add(ground)

# Create the mass (dynamic body)
mass_radius = 0.05
mass = chrono.ChBodyEasySphere(mass_radius, 1000, True, True)
mass.SetPos(chrono.ChVectorD(0, 0.2, 0))
system.Add(mass)

# Add visual shape for the mass
mass_visual = chrono.ChSphereShape(mass_radius)
mass.GetVisualShape().AddVisualShape(mass_visual)

# Create the spring-damper link (ChLinkTSDA)
spring_length = 0.2  # Rest length of the spring
spring = chrono.ChLinkTSDA()
spring.Initialize(ground, mass, False, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 0, 0))
system.Add(spring)

# Set default linear spring-damper parameters
spring.SetSpringCoefficient(100.0)  # Spring stiffness
spring.SetDampingCoefficient(1.0)   # Damping coefficient
spring.SetRestLength(spring_length)

# Visualize the spring as a cylinder
spring_visual = chrono.ChCylinderShape()
spring_visual.GetCylinderGeometry().p1 = chrono.ChVectorD(0, 0, 0)
spring_visual.GetCylinderGeometry().p2 = chrono.ChVectorD(0, -spring_length, 0)
spring_visual.GetCylinderGeometry().radius = 0.005

# Attach the visual shape to the link
spring.AddVisualShape(spring_visual)

# Set up the simulation parameters
application.AssetBindAll()
application.AssetUpdateAll()

# Simulation loop
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.EndScene()
    system.DoStepDynamics(1e-3)
```
