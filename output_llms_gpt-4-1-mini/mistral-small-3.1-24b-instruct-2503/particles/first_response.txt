Sure! Below is a Python script that uses PyChrono to create a simulation modeling the gravitational attraction between particles. The script initializes the PyChrono environment, sets up a particle emitter to generate particles with random shapes, positions, velocities, and orientations, and visualizes the simulation using Irrlicht.

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.physics as chronophys
import random

# Initialize the Chrono::Engine system
chrono.SetChronoDataPath('path/to/chrono/data')  # Set the path to Chrono data

# Create a Chrono::Engine physical system
system = chrono.ChSystemNSC()

# Create an Irrlicht visualization (renderer)
visualization = chronoirr.ChIrrApp()
visualization.AddTypicalSky()
visualization.AddTypicalLogo()
visualization.AddTypicalLights()
visualization.AddLightWithShadow(chrono.ChVectorD(1, 1, 1), chrono.ChVectorD(0, 0, 0), 10, 5, 50, 40, 50)
visualization.SetSymbolScale(1.0)
visualization.SetContactDrawSize(0.01)
visualization.SetContactForceDrawSize(0.001)
visualization.SetFrameSpeed(60)
visualization.SetTryRealTimeStep(True)
visualization.AssetBindAll()
visualization.AssetUpdateAll()

# Set up the system
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))  # Set gravity

# Function to create a particle with random properties
def create_random_particle():
    particle = chrono.ChParticle()
    particle.SetPos(chrono.ChVectorD(random.uniform(-5, 5), random.uniform(-5, 5), random.uniform(5, 10)))
    particle.SetMass(1)
    particle.SetVel(chrono.ChVectorD(random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1)))
    particle.SetCollide(True)
    particle.SetBodyFixed(False)
    particle.SetName("Particle")
    return particle

# Function to add gravitational force between particles
def add_gravitational_force(particle1, particle2):
    G = 6.67430e-11  # Gravitational constant
    r = particle1.GetPos() - particle2.GetPos()
    distance = r.Length()
    if distance == 0:
        return
    force_magnitude = G * (particle1.GetMass() * particle2.GetMass()) / (distance ** 2)
    force_direction = r.Normalize()
    force = force_direction * force_magnitude
    particle1.AddForce(force)
    particle2.AddForce(-force)

# Particle emitter
particles = []
emission_rate = 1  # Number of particles to emit per second
emission_interval = 1 / emission_rate

# Main simulation loop
def main():
    visualization.SetWindowSize(800, 600)
    visualization.SetWindowTitle('Particle Gravity Simulation')
    visualization.Initialize()
    visualization.AddSystem(system)

    # Create a timer for particle emission
    emission_timer = 0

    while visualization.Run():
        current_time = visualization.GetTime()

        # Emit particles at the specified rate
        if current_time - emission_timer >= emission_interval:
            new_particle = create_random_particle()
            system.Add(new_particle)
            particles.append(new_particle)
            emission_timer = current_time

        # Update the system
        system.DoStepDynamics(0.01)

        # Apply gravitational forces between all pairs of particles
        for i in range(len(particles)):
            for j in range(i + 1, len(particles)):
                add_gravitational_force(particles[i], particles[j])

        # Update visualization
        visualization.BeginScene()
        visualization.Render()
        visualization.EndScene()

if __name__ == "__main__":
    main()
```

### Explanation:
1. **Initialization**: The script initializes the Chrono::Engine system and sets up Irrlicht for visualization.
2. **Particle Creation**: The `create_random_particle` function generates particles with random positions, velocities, and masses.
3. **Gravitational Force**: The `add_gravitational_force` function calculates and applies gravitational forces between pairs of particles.
4. **Particle Emission**: Particles are emitted at a specified rate using a timer.
5. **Simulation Loop**: The main loop runs the simulation, updates the system dynamics, applies gravitational forces, and renders the visualization.
