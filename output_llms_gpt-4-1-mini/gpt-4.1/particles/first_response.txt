Certainly! Below is a **PyChrono** simulation script that fulfills your requirements:

- **Initializes PyChrono and Irrlicht visualization**
- **Implements a particle emitter** that generates particles with random shapes (sphere, box, cylinder), positions, velocities, and orientations
- **Applies custom gravitational forces** between all particles (Newtonian attraction)
- **Visualizes the simulation** in real time

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import random
import math

# --- 1. Initialize PyChrono system and visualization ---
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
system = chrono.ChSystemNSC()

# Irrlicht visualization
application = chronoirr.ChIrrApp(system, "Particle Gravity Simulation", chronoirr.dimension2du(1024,768))
application.AddTypicalSky()
application.AddTypicalCamera(chronoirr.vector3df(0, 2, -8))
application.AddTypicalLights()

# --- 2. Particle Emitter ---
particle_list = []

def random_shape():
    shape_type = random.choice(['sphere', 'box', 'cylinder'])
    if shape_type == 'sphere':
        radius = random.uniform(0.05, 0.15)
        body = chrono.ChBodyEasySphere(radius, 1000, True, True)
        body.SetMass(0.1)
    elif shape_type == 'box':
        size = chrono.ChVectorD(random.uniform(0.05,0.15), random.uniform(0.05,0.15), random.uniform(0.05,0.15))
        body = chrono.ChBodyEasyBox(size.x, size.y, size.z, 1000, True, True)
        body.SetMass(0.1)
    else: # cylinder
        radius = random.uniform(0.05, 0.12)
        height = random.uniform(0.05, 0.15)
        body = chrono.ChBodyEasyCylinder(radius, height, 1000, True, True)
        body.SetMass(0.1)
    return body

def emit_particle():
    body = random_shape()
    # Random position in a cube around the origin
    pos = chrono.ChVectorD(random.uniform(-1,1), random.uniform(0.5,2), random.uniform(-1,1))
    body.SetPos(pos)
    # Random orientation
    q = chrono.Q_from_AngAxis(random.uniform(0,math.pi), chrono.ChVectorD(random.uniform(-1,1), random.uniform(-1,1), random.uniform(-1,1)).GetNormalized())
    body.SetRot(q)
    # Random velocity
    vel = chrono.ChVectorD(random.uniform(-1,1), random.uniform(-1,1), random.uniform(-1,1))
    body.SetPos_dt(vel)
    # Make it dynamic
    body.SetBodyFixed(False)
    # Add to system and list
    system.Add(body)
    particle_list.append(body)

# --- 3. Custom gravitational attraction ---
def apply_gravitational_forces():
    G = 0.05  # Gravitational constant (tunable for simulation scale)
    n = len(particle_list)
    for i in range(n):
        for j in range(i+1, n):
            a = particle_list[i]
            b = particle_list[j]
            # Compute vector from a to b
            r_vec = b.GetPos() - a.GetPos()
            r = r_vec.Length()
            if r < 1e-4:
                continue  # avoid singularity
            force_mag = G * a.GetMass() * b.GetMass() / (r*r)
            force = r_vec.GetNormalized() * force_mag
            a.Empty_forces_accumulators()
            b.Empty_forces_accumulators()
            a.Accumulate_force(force, a.GetPos(), False)
            b.Accumulate_force(-force, b.GetPos(), False)

# --- 4. Main simulation loop ---
application.AssetBindAll()
application.AssetUpdateAll()

emit_interval = 0.5  # seconds
emit_timer = 0.0
max_particles = 30

while(application.GetDevice().run()):
    application.BeginScene()
    application.DrawAll()
    
    # Emit new particles at intervals
    if len(particle_list) < max_particles:
        emit_timer += application.GetStep()
        if emit_timer >= emit_interval:
            emit_particle()
            emit_timer = 0.0

    # Apply custom gravitational forces
    apply_gravitational_forces()
    
    application.DoStep()
    application.EndScene()
```

---

**Notes:**
- The gravitational constant `G` is set for simulation stability; adjust as needed.
- The emitter creates up to `max_particles` with random properties.
- Each simulation step, all particle pairs attract each other via Newton's law.
- Visualization uses Irrlicht; close the window to stop the simulation.

**Dependencies:**  
Make sure you have `pychrono` installed and working with Irrlicht bindings.
