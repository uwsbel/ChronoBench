```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as robot
from pychrono import irrlicht as chronoirr

# 1. Import the sensor module
import pychrono.sensor as sens

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))  # Corrected: Set_G_acc, ChVectorD

chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create ground body with contact material and add it to the system
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVectorD(0, 0, -0.5))  # Corrected: ChVectorD
ground.SetBodyFixed(True)  # Corrected: SetBodyFixed
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# create a long box for rover to cross
box = chrono.ChBodyEasyBox(0.25, 5, 0.25, 1000, True, True, ground_mat)
box.SetPos(chrono.ChVectorD(0, 0, 0.0))
box.SetBodyFixed(True)
box.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
system.Add(box)

# Create Curiosity rover and add it to the system
rover = robot.Curiosity(system)

# Create driver for rover
driver = robot.CuriosityDCMotorControl()
rover.SetDriver(driver)

# Initialize rover position and orientation
init_pos = chrono.ChVectorD(-5, 0.0, 0)
init_rot = chrono.ChQuaternionD(1, 0, 0, 0)
rover.Initialize(chrono.ChFrameD(init_pos, init_rot))  # Corrected: ChFrameD

# 2. Create a sensor manager
manager = sens.ChSensorManager(system)

# 3. Add a lidar sensor to the rover
# Get the chassis body of the rover
chassis = rover.GetChassisBody()

# Lidar parameters
lidar_update_rate = 10.0  # Hz
lidar_h_samples = 512     # horizontal samples
lidar_v_samples = 32      # vertical samples
lidar_h_fov = 2 * math.pi # 360 deg
lidar_v_fov = math.radians(30)  # 30 deg vertical FOV
lidar_max_dist = 20.0     # meters

# Lidar relative pose (positioned above the chassis)
lidar_offset = chrono.ChFrameD(chrono.ChVectorD(0.0, 0.0, 0.5), chrono.QUNIT)  # 0.5m above chassis

# Create the lidar sensor
lidar = sens.ChLidarSensor(
    parent=chassis,
    updateRate=lidar_update_rate,
    offsetPose=lidar_offset,
    horizontal_samples=lidar_h_samples,
    vertical_samples=lidar_v_samples,
    horizontal_fov=lidar_h_fov,
    vertical_fov=lidar_v_fov,
    max_distance=lidar_max_dist
)

# Add filters to the lidar sensor
lidar.PushFilter(sens.ChFilterLidarAccess())  # Access filter for retrieving data

# Add the lidar sensor to the manager
manager.AddSensor(lidar)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Curiosity rover - Rigid terrain')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 3, 3), chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVectorD(1.5, -2.5, 5.5), chrono.ChVectorD(0, 0, 0), 3, 4, 10, 40, 512)

# Enable shadows (commented out to improve performance)
# vis.EnableShadows()

# Set the simulation time step
time_step = 1e-3

# Simulation loop
time = 0
while vis.Run():
    time += time_step

    # ask rover to move forward
    driver.SetSteering(0.0)

    # Update rover dynamics
    rover.Update()

    # 4. Update the sensor manager
    manager.Update()

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance simulation by one time step
    system.DoStepDynamics(time_step)
```
