camera.setPosition(chrono.Vec3(0, 0, 10))

)

vehicle_data = vehicle.VehicleData(

    width=2.0,
    length=5.0,
    height=1.0,
    mass=1000.0,
    contact_method=vehicle.ContactMethod.GROUND,
    tire_model=tires.TireModel.RUBBER,
)
vehicle = vehicle.Vehicle(vehicle_data, irr_env)

# Set initial vehicle parameters
vehicle.setPosition(chrono.Vec3(0, 0, 0))
vehicle.setOrientation(chrono.Vec3(0, 0, 0))
vehicle.setContactMethod(vehicle.ContactMethod.GROUND)
vehicle.setTireModel(tires.TireModel.RUBBER)

# Set camera follow
camera.setPosition(chrono.Vec3(0, 0, 10))

# Set simulation parameters
