```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Ensure data path is set (optional)

# Create the physical system
system = chrono.ChSystemNSC()

# 2. Add the required physical systems and objects

# Create the ground body (fixed)
ground = chrono.ChBody()
ground.SetBodyFixed(True)
system.Add(ground)

# Create the mass body
mass = chrono.ChBody()
mass.SetMass(1.0)  # 1 kg mass
mass.SetInertiaXX(chrono.ChVectorD(0.1, 0.1, 0.1))  # Some small inertia
mass.SetPos(chrono.ChVectorD(0, 0, 1))  # Start 1 meter above ground
system.Add(mass)

# Create a spring-damper link (ChLinkTSDA) between ground and mass
spring = chrono.ChLinkTSDA()
spring.Initialize(ground, mass,
                  False,
                  chrono.ChVectorD(0, 0, 0),  # attachment point on ground
                  chrono.ChVectorD(0, 0, 1))  # attachment point on mass

# 3. Set necessary default parameters such as positions, forces, and interactions

# Default linear spring-damper parameters
spring.SetSpringCoefficient(1000)  # N/m
spring.SetDampingCoefficient(10)   # Ns/m
spring.SetRestLength(1.0)           # meters

system.AddLink(spring)

# Visualization setup

# Create the Irrlicht application
application = chronoirr.ChIrrApp(system, "Mass-Spring-Damper System", chronoirr.dimension2du(800, 600))

# Add typical elements
application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(chronoirr.vector3df(2, 2, 2))

# Add visual shape for the mass (a box)
mass_shape = chrono.ChBoxShape()
mass_shape.GetBoxGeometry().Size = chrono.ChVectorD(0.1, 0.1, 0.1)
mass.GetAssets().push_back(mass_shape)

# Add visual shape for the spring (a cylinder between the two attachment points)
spring_shape = chrono.ChCylinderShape()
spring_shape.GetCylinderGeometry().p1 = chrono.ChVectorD(0, 0, 0)
spring_shape.GetCylinderGeometry().p2 = chrono.ChVectorD(0, 0, 1)
spring_shape.GetCylinderGeometry().rad = 0.02
spring.GetAssets().push_back(spring_shape)

# Bind all assets and update
application.AssetBindAll()
application.AssetUpdateAll()

# Simulation loop
application.SetTimestep(0.01)

while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```
