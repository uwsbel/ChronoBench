```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Set data path for assets if needed

# Create the physical system
system = chrono.ChSystemNSC()

# Set gravity (optional, gears usually work in zero gravity or normal gravity)
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# 2. Add the required physical systems and objects

# Create a fixed truss (a fixed frame to hold gears)
truss = chrono.ChBody()
truss.SetBodyFixed(True)
system.Add(truss)

# Create a rotating bar attached to the truss by a revolute joint
bar = chrono.ChBody()
bar.SetMass(2)
bar.SetInertiaXX(chrono.ChVectorD(0.1, 0.1, 0.1))
bar.SetPos(chrono.ChVectorD(0, 0, 0))
system.Add(bar)

# Revolute joint between truss and bar (rotation around Z axis)
joint_bar = chrono.ChLinkLockRevolute()
joint_bar.Initialize(bar, truss, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 0, 1))))
system.Add(joint_bar)

# Create two gears: sun gear and planet gear
# For simplicity, model gears as cylinders with appropriate radii

# Sun gear (attached to bar)
sun_gear = chrono.ChBody()
sun_gear.SetMass(1)
sun_gear.SetInertiaXX(chrono.ChVectorD(0.05, 0.05, 0.05))
sun_gear_radius = 0.2
sun_gear.SetPos(chrono.ChVectorD(0.5, 0, 0))
system.Add(sun_gear)

# Revolute joint between sun gear and bar (gear rotates relative to bar)
joint_sun = chrono.ChLinkLockRevolute()
joint_sun.Initialize(sun_gear, bar, chrono.ChCoordsysD(sun_gear.GetPos(), chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 0, 1))))
system.Add(joint_sun)

# Planet gear (meshing with sun gear, fixed to truss but can rotate)
planet_gear = chrono.ChBody()
planet_gear.SetMass(1)
planet_gear.SetInertiaXX(chrono.ChVectorD(0.05, 0.05, 0.05))
planet_gear_radius = 0.1
# Position planet gear so it meshes with sun gear (distance = sum of radii)
planet_gear.SetPos(chrono.ChVectorD(0.5 + sun_gear_radius + planet_gear_radius, 0, 0))
system.Add(planet_gear)

# Revolute joint between planet gear and truss (planet gear rotates around its own axis)
joint_planet = chrono.ChLinkLockRevolute()
joint_planet.Initialize(planet_gear, truss, chrono.ChCoordsysD(planet_gear.GetPos(), chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 0, 1))))
system.Add(joint_planet)

# Create a gear constraint between sun gear and planet gear to enforce gear ratio
gear_ratio = sun_gear_radius / planet_gear_radius  # ratio of radii

gear_constraint = chrono.ChLinkGear()
gear_constraint.Initialize(sun_gear, planet_gear, chrono.ChFrameD(), chrono.ChFrameD())
gear_constraint.SetRatio(gear_ratio)
system.Add(gear_constraint)

# Create a gear constraint between bar and sun gear to enforce their relative rotation (optional)
# This is not strictly necessary if sun gear is revolute jointed to bar, but can be used for more complex constraints

# 3. Add one gear motor to enforce a constant rotation speed on the bar

motor = chrono.ChLinkMotorRotationSpeed()
motor_speed = chrono.ChFunction_Const(chrono.CH_C_PI / 2)  # 90 deg/s = pi/2 rad/s

motor.Initialize(bar, truss, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 0, 1))))
motor.SetSpeedFunction(motor_speed)
system.Add(motor)

# 4. Visualization setup using Irrlicht

application = chronoirr.ChIrrApp(system, "Epicyclic Gear System", chronoirr.dimension2du(1024, 768))

application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(chronoirr.vector3df(1.5, 1.5, 1.5))

# Create materials for visualization

# Truss material (fixed frame)
mat_truss = chrono.ChVisualMaterial()
mat_truss.SetDiffuseColor(chrono.ChColor(0.3, 0.3, 0.3))
mat_truss.SetFresnelR0(chrono.ChVectorD(0.1, 0.1, 0.1))
mat_truss.SetRoughness(0.5)

# Bar material
mat_bar = chrono.ChVisualMaterial()
mat_bar.SetDiffuseColor(chrono.ChColor(0.8, 0.1, 0.1))
mat_bar.SetFresnelR0(chrono.ChVectorD(0.7, 0.1, 0.1))
mat_bar.SetRoughness(0.3)

# Gear material
mat_gear = chrono.ChVisualMaterial()
mat_gear.SetDiffuseColor(chrono.ChColor(0.1, 0.1, 0.8))
mat_gear.SetFresnelR0(chrono.ChVectorD(0.1, 0.1, 0.7))
mat_gear.SetRoughness(0.2)

# Add visual shapes

# Truss: simple box to represent fixed frame
truss_shape = chrono.ChBoxShape()
truss_shape.GetBoxGeometry().Size = chrono.ChVectorD(0.05, 0.5, 0.05)
truss.AddVisualShape(truss_shape, chrono.ChFrameD(chrono.ChVectorD(0.5 + sun_gear_radius + planet_gear_radius, 0, 0)))

# Bar: box shape
bar_shape = chrono.ChBoxShape()
bar_shape.GetBoxGeometry().Size = chrono.ChVectorD(0.5, 0.05, 0.05)
bar.AddVisualShape(bar_shape, chrono.ChFrameD(chrono.ChVectorD(0.25, 0, 0)))

# Sun gear: cylinder shape
sun_gear_shape = chrono.ChCylinderShape()
sun_gear_shape.GetCylinderGeometry().p1 = chrono.ChVectorD(0, 0, -0.05)
sun_gear_shape.GetCylinderGeometry().p2 = chrono.ChVectorD(0, 0, 0.05)
sun_gear_shape.GetCylinderGeometry().rad = sun_gear_radius
sun_gear.AddVisualShape(sun_gear_shape)

# Planet gear: cylinder shape
planet_gear_shape = chrono.ChCylinderShape()
planet_gear_shape.GetCylinderGeometry().p1 = chrono.ChVectorD(0, 0, -0.05)
planet_gear_shape.GetCylinderGeometry().p2 = chrono.ChVectorD(0, 0, 0.05)
planet_gear_shape.GetCylinderGeometry().rad = planet_gear_radius
planet_gear.AddVisualShape(planet_gear_shape)

# Assign materials to visual shapes
for body, mat in [(truss, mat_truss), (bar, mat_bar), (sun_gear, mat_gear), (planet_gear, mat_gear)]:
    for vis_shape in body.GetVisualShapeList():
        vis_shape.SetMaterial(mat)

# Lighting: add a directional light to illuminate the scene nicely
light = chronoirr.IrrlichtLight(application.GetDevice(), chronoirr.irrlicht.ELightType.ELT_DIRECTIONAL)
light.SetDirection(chronoirr.vector3df(-1, -1, -1))
light.SetDiffuseColor(chronoirr.SColorf(1, 1, 1))
light.SetSpecularColor(chronoirr.SColorf(1, 1, 1))
application.AddLight(light)

# 5. Simulation loop

application.AssetBindAll()
application.AssetUpdateAll()

application.SetTimestep(0.01)

while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```
