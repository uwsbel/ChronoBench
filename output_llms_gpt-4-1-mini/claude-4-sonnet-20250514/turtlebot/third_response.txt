```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as turtlebot
import pychrono.sensor as sens
from pychrono import irrlicht as chronoirr

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))  # Set gravity in negative Z direction
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create ground body as terrain plane that robot will drive on
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVector3d(0, 0, -0.6))  # Position the ground as specified
ground.SetFixed(True)  # Fix the ground in place
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# Create randomly placed boxes
box_mat = chrono.ChContactMaterialNSC()
np.random.seed(42)  # For reproducible results
for i in range(5):
    box_size = chrono.ChVector3d(0.3, 0.3, 0.3)
    box_pos = chrono.ChVector3d(
        np.random.uniform(-8, 8),
        np.random.uniform(-8, 8),
        0.15
    )
    box = chrono.ChBodyEasyBox(box_size.x, box_size.y, box_size.z, 100, True, True, box_mat)
    box.SetPos(box_pos)
    box.SetFixed(False)
    system.Add(box)

# Create Turtlebot Robot
init_pos = chrono.ChVector3d(0, 0.2, 0)  # Initial position of the robot
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)  # Initial orientation of the robot
robot = turtlebot.TurtleBot(system, init_pos, init_rot)  # Create Turtlebot instance
robot.Initialize()  # Initialize the robot

# Create and configure sensor manager
manager = sens.ChSensorManager(system)
manager.scene.AddPointLight(chrono.ChVector3f(2, 2.5, 100), chrono.ChColor(1, 1, 1), 500.0)

# Add lidar sensor
offset_pose = chrono.ChFramed(chrono.ChVector3d(0.0, 0, 0.2), chrono.QuatFromAngleAxis(0, chrono.ChVector3d(0, 1, 0)))
lidar = sens.ChLidarSensor(
    robot.GetChassisBody(),              # body lidar is attached to
    30,                                   # scanning rate in Hz
    offset_pose,                         # offset pose
    800,                                 # number of horizontal samples
    300,                                 # number of vertical samples
    2*chrono.CH_PI,                      # horizontal field of view
    chrono.CH_PI/12,                     # max vertical field of view
    -chrono.CH_PI/6,                     # min vertical field of view
    100.0,                               # max lidar range
    sens.LidarBeamShape_RECTANGULAR,     # beam shape
    2,                                   # sample radius
    0.003,                               # divergence angle
    0.003,                               # divergence angle (typically same as above)
    sens.LidarReturnMode_STRONGEST_RETURN # return mode
)
lidar.SetName("Lidar Sensor")
lidar.SetLag(0)
lidar.SetCollectionWindow(0)

# Add noise filter
noise_model = sens.ChNoiseNone()
lidar.PushFilter(sens.ChFilterLidarNoiseXYZI(noise_model, noise_model, noise_model, noise_model))

# Add visualization filter
lidar.PushFilter(sens.ChFilterVisualize(800, 300, "Lidar"))

# Add the lidar to the sensor manager
manager.AddSensor(lidar)

# Motion control function
def move(mode):
    if mode == 'straight':
        robot.SetMotorSpeed(math.pi, turtlebot.LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(math.pi, turtlebot.RIGHT_DRIVE_WHEEL)
    elif mode == 'left':
        robot.SetMotorSpeed(0, turtlebot.LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(math.pi, turtlebot.RIGHT_DRIVE_WHEEL)
    elif mode == 'right':
        robot.SetMotorSpeed(math.pi, turtlebot.LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0, turtlebot.RIGHT_DRIVE_WHEEL)
    else:
        robot.SetMotorSpeed(0, turtlebot.LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0, turtlebot.RIGHT_DRIVE_WHEEL)

# Create run-time visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Turtlebot Robot - Rigid terrain')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 1.5, 0.2), chrono.ChVector3d(0, 0, 0.2))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0.5), 3, 4, 10, 40, 512)

# Enable shadows (commented out to improve performance)
# vis.EnableShadows()

# Set the simulation time step
time_step = 2e-3

# Simulation loop
time = 0
while vis.Run():
    # Use move function for straight movement
    move('straight')

    # Update sensor manager
    manager.Update()

    # Increment time counter
    time += time_step

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance the simulation by one time step
    system.DoStepDynamics(time_step)
```
