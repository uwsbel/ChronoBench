```python
import os
import math
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.sensor as sens

# Set the Chrono data path
chrono.SetChronoDataPath(os.environ.get('CHRONO_DATA_DIR', '/path/to/chrono/data'))

# Initialize the system
sys = chrono.ChSystemSMC()

# Load the triangular mesh from a Wavefront .obj file
mesh_body = chrono.ChBody()
sys.Add(mesh_body)
mesh_body.SetFixed(True)  # Fixed in space
mesh_body.SetPos(chrono.ChVector3d(0, 0, 0))

# Load the mesh geometry
mesh = chrono.ChTriangleMeshConnected()
mesh.LoadWavefrontMesh(chrono.GetChronoDataFile("meshes/your_mesh.obj"), False, True)

# Create a visual shape for the mesh
vis_shape = chrono.ChVisualShapeTriangleMesh()
vis_shape.SetMesh(mesh)
mesh_body.AddVisualShape(vis_shape)

# Create a body for the camera's position/orientation
camera_body = chrono.ChBody()
sys.Add(camera_body)
camera_body.SetFixed(False)  # Will be manually controlled
camera_body.EnableCollision(False)  # No collision needed

# Initialize the sensor manager
manager = sens.ChSensorManager(sys)

# Camera sensor parameters
update_rate = 60  # Hz
width, height = 1280, 720
horizontal_fov = 60 * chrono.CH_PI / 180  # Convert degrees to radians
offset_pose = chrono.ChFramed(chrono.ChVector3d(0, 0, 0), chrono.Quaterniond(1, 0, 0, 0))

# Create the camera sensor
camera = sens.ChCameraSensor(
    camera_body,  # Parent body
    update_rate,
    offset_pose,
    width,
    height,
    horizontal_fov
)
camera.SetName("Orbiting Camera")
camera.SetLag(0.0)  # No lag
camera.SetCollectionWindow(0.01)  # Short collection window

# Add noise filter (Gaussian noise with mean 0, std 0.01)
camera.AddFilter(sens.ChFilterNoiseGaussian(0.0, 0.01))

# Add visualization filter for real-time display
camera.AddFilter(sens.ChFilterVisualize(width, height, "Camera Feed"))

# Add RGBA8 access filter to retrieve pixel data
camera.AddFilter(sens.ChFilterRGBA8Access())

# Register the camera with the manager
manager.AddSensor(camera)

# Setup visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Triangular Mesh with Orbiting Camera")
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 6))
vis.AddTypicalLights()

# Simulation loop parameters
time_step = 0.01
total_time = 10
theta = 0.0  # Initial angle for orbit
orbit_radius = 5.0  # Distance from mesh
orbit_height = 2.0  # Elevation above mesh
angular_velocity = 0.5  # Radians per second

# Simulation loop
while vis.Run():
    # Calculate new camera position (circular orbit in X-Z plane)
    theta += angular_velocity * time_step
    cam_x = orbit_radius * math.cos(theta)
    cam_z = orbit_radius * math.sin(theta)
    cam_pos = chrono.ChVector3d(cam_x, orbit_height, cam_z)
    camera_body.SetPos(cam_pos)

    # Compute rotation to point camera at the mesh origin
    dir_to_origin = chrono.ChVector3d(0, 0, 0) - cam_pos
    dir_to_origin.Normalize()
    z_axis = chrono.ChVector3d(0, 0, 1)
    rotation = chrono.QuatFromTwoVectors(z_axis, dir_to_origin)
    camera_body.SetRot(rotation)

    # Advance the simulation
    sys.DoStepDynamics(time_step)
    
    # Update sensors
    manager.Update()

    # Access and print camera buffer data
    buffer = camera.GetMostRecentBuffer()
    if buffer.HasData():
        rgba_data = buffer.GetAsRGBA8()
        if rgba_data:
            # Print timestamp and first pixel value (simplified)
            print(f"Time: {sys.GetChTime():.2f}s | First pixel: {rgba_data[0]}")

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
```

### Key Features:
1. **Mesh Loading**: Loads a `.obj` file as a fixed body using `ChTriangleMeshConnected`.
2. **Orbiting Camera**: 
   - Camera orbits the mesh using polar coordinates (θ updated each step).
   - Automatically reorients to face the mesh origin using `QuatFromTwoVectors`.
3. **Sensor Pipeline**:
   - Applies Gaussian noise to images.
   - Visualizes the camera feed in real-time.
   - Exposes raw RGBA8 pixel data for analysis.
4. **Dynamic Updates**: Camera position/orientation updated every simulation step.

### Requirements:
- Replace `"meshes/your_mesh.obj"` with a valid `.obj` file path.
- Ensure Chrono's data directory is correctly set (e.g., via `CHRONO_DATA_DIR` environment variable).

### Output:
- Real-time 3D visualization with camera feed.
- Printed timestamps and pixel data in the console.
