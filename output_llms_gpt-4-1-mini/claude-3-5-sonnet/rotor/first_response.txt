```python
import pychrono as chrono
import pychrono.fea as fea
import pychrono.irrlicht as chronoirr
import numpy as np

# Initialize the ChSystem
system = chrono.ChSystemNSC()

# Set gravity
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create a mesh
mesh = fea.ChMesh()

# Create IGA beam
beam_length = 1.0
beam_radius = 0.01
beam_density = 7800  # kg/m^3
beam_Young = 210e9  # Pa
beam_Poisson = 0.3
beam_damping = 0.01

# Create the beam section
beam_section = fea.ChBeamSectionIGA()
beam_section.SetAsCircularSection(beam_radius)
beam_section.SetYoungModulus(beam_Young)
beam_section.SetGshearModulus(beam_Young / (2 * (1 + beam_Poisson)))
beam_section.SetBeamRaleyghDamping(beam_damping)
beam_section.SetDensity(beam_density)

# Create the beam
builder = fea.ChBuilderBeamIGA()
builder.BuildBeam(mesh, beam_section, 10, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(beam_length, 0, 0))

# Add the mesh to the system
system.Add(mesh)

# Create a flywheel
flywheel_mass = 1.0
flywheel_radius = 0.1
flywheel = chrono.ChBodyEasyCylinder(flywheel_radius, 0.05, flywheel_mass)
flywheel.SetPos(chrono.ChVectorD(beam_length/2, 0, 0))
system.Add(flywheel)

# Connect flywheel to the beam
beam_center_node = mesh.GetNode(5)  # Assuming 10 nodes, the center is at index 5
constraint = fea.ChLinkPointFrame()
constraint.Initialize(beam_center_node, flywheel)
system.Add(constraint)

# Create a motor at one end of the beam
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(mesh.GetNode(0), system.GetGround(), chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))
motor_speed = chrono.ChFunction_Const(5.0)  # 5 rad/s
motor.SetSpeedFunction(motor_speed)
system.Add(motor)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Jeffcott Rotor IGA Beam Simulation")
vis.Initialize()
vis.AddCamera(chrono.ChVectorD(0, 0.5, -1))
vis.AddTypicalLights()

# Create a FEM visualization
beam_vis = fea.ChVisualizationFEAmesh(mesh)
beam_vis.SetFEMdataType(fea.ChVisualizationFEAmesh.E_PLOT_ELEM_BEAM_MZ)
beam_vis.SetColorscaleMinMax(-0.4, 0.4)
beam_vis.SetSmoothFaces(True)
beam_vis.SetWireframe(False)
mesh.AddAsset(beam_vis)

# Main simulation loop
time_step = 0.001
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    system.DoStepDynamics(time_step)

vis.GetDevice().closeDevice()
```

This script does the following:

1. Initializes the PyChrono environment and creates a `ChSystem`.
2. Sets up gravity in the system.
3. Creates an IGA beam using `ChMesh` and `ChBuilderBeamIGA`.
4. Adds a flywheel at the center of the beam.
5. Connects the flywheel to the beam using a constraint.
6. Adds a rotational motor at one end of the beam.
7. Sets up Irrlicht visualization for the simulation.
8. Adds FEM visualization for the beam.
9. Runs the main simulation loop.

This script creates a Jeffcott rotor model with an IGA beam, a flywheel at its center, and a motor driving one end. The Irrlicht window allows you to observe the dynamics of the system, and the FEM visualization helps in understanding the beam's behavior under rotation.

You can adjust parameters like beam dimensions, material properties, flywheel mass, motor speed, etc., to experiment with different configurations of the Jeffcott rotor system.
