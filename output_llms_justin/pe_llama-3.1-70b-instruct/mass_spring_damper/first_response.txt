Here's a PyChrono script that meets the provided specifications:
```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Create the simulation system
sys = chrono.ChSystemNSC()

# Set the path to the Chrono data directory
chrono.SetChronoDataPath('/path/to/data')

# Create the visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Mass-Spring-Damper Simulation')
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 6))
vis.AddTypicalLights()

# Create the ground body
ground = chrono.ChBody()
ground.SetPos(chrono.ChVector3d(0, 0, 0))
ground.SetBodyFixed(True)
sys.Add(ground)

# Create the mass body
mass = chrono.ChBodyEasySphere(1.0, 1000, True, True)
mass.SetPos(chrono.ChVector3d(0, 1, 0))
sys.Add(mass)

# Create the mass-spring-damper system
link = chrono.ChLinkTSDA()
link.Initialize(ground, mass, chrono.ChCoordsysD(chrono.ChVector3d(0, 1, 0), chrono.QUNIT))
link.SetSpringCoefficient(1000)
link.SetDampingCoefficient(10)
link.SetRestLength(1.0)
sys.Add(link)

# Add visual shapes
mass_shape = chrono.ChVisualShapeSphere(1.0)
mass_shape.SetColor(chrono.ChColor(1, 0, 0))  # Red color
mass.AddVisualShape(mass_shape)

spring_shape = chrono.ChVisualShapeCylinder(0.1, 1.0)
spring_shape.SetColor(chrono.ChColor(0, 1, 0))  # Green color
link.AddVisualShape(spring_shape)

# Set up the simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(0.01)

# Wait for the simulation to finish
vis.WaitForExit()
```
This script initializes a PyChrono simulation environment with a single mass-spring-damper system connected to the ground. It uses Irrlicht visualization with a camera, lighting, and visual shapes for the mass and spring. The simulation loop advances the simulation by a fixed time step, rendering the scene at each step.

Note that you should replace `'/path/to/data'` with the actual path to the Chrono data directory on your system.
