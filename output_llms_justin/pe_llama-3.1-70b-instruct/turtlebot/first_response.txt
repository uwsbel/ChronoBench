Here's a Python script that implements the specified simulation:

```python
import os
import math
import numpy as np
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import pychrono.robot as robosimian

# Set the path to the Chrono data directory
chrono.SetChronoDataPath(os.path.join(os.path.dirname(__file__), "data"))

# Create the simulation system and set the gravity
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the ground body
ground = chrono.ChBodyEasyBox(system, 10, 10, 1, 1000, True, True, chrono.ChColor(0.4, 0.4, 0.5))
ground.SetPos(chrono.ChVectorD(0, -5, 0))
ground.SetBodyFixed(True)

# Create the Turtlebot robot
turtlebot = robosimian.Turtlebot(system, True, True)
turtlebot.Initialize(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngX(chrono.CH_PI)))

# Set the driver for the Turtlebot
driver = robosimian.Turtlebot_Driver(turtlebot, True)
turtlebot.SetDriver(driver)

# Create the visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Turtlebot Simulation")
vis.AttachSystem(system)

# Set the camera position and orientation
vis.AddCamera(chrono.ChVectorD(0, 1, 2))
vis.AddLightWithShadow(chrono.ChVectorD(1, 2, 3), chrono.ChColor(1, 1, 1), True)

# Add a skybox to the visualization
vis.AddSkyBox()

# Define the simulation loop
def simulate(turtlebot, driver, vis, system):
    # Simulation time and time step
    time = 0
    dt = 0.01

    # Set the initial motor speeds
    driver.SetSpeed(0, 0)

    # Simulation loop
    while vis.Run():
        # Update the simulation
        system.DoStepDynamics(dt)

        # Update the visualization
        vis.BeginScene()
        vis.DrawAll()
        vis.EndScene()

        # Update the time
        time += dt

        # Control the Turtlebot's motors
        if time < 5:
            driver.SetSpeed(0, 1)  # Turn left
        elif time < 10:
            driver.SetSpeed(1, 0)  # Turn right
        else:
            driver.SetSpeed(0, 0)  # Stop

# Run the simulation
simulate(turtlebot, driver, vis, system)
```

This script initializes the PyChrono environment, creates a Turtlebot robot on a rigid terrain, and sets up a real-time visualization using Irrlicht. It then implements a simulation loop where the Turtlebot's motors control left and right turns at specified times, updating and rendering the robot at each timestep.
