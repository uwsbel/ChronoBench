Here's a Python script that implements the simulation as described in the instructions:

```python
import os
import math
import numpy as np
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the PyChrono environment and core components
chrono.SetChronoDataPath('/path/to/data')

# Create the simulation system and set the solver
mysystem = chrono.ChSystemSMC()
mysystem.SetSolverType(chrono.ChSolver.Type_PSSOR)
mysystem.SetSolverMaxIterations(100)

# Set the gravity and ground body
mysystem.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))
ground = chrono.ChBodyEasyBox(mysystem, 10, 1, 10, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -1, 0))
ground.SetBodyFixed(True)

# Initialize the Viper rover and its driver
viper = veh.Viper(mysystem, True, True)
viper.SetContactMethod(chrono.ChContactMethod_NSC)
viper.SetChassisCollisionType(chrono.ChCollisionType_AABB)
viper.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT))
viper.Initialize()

driver = veh.VehicleDriverViper()
driver.SetVehicle(viper)
driver.SetSteeringFunction(chrono.ChFunction_Const(0.2))
driver.SetThrottleFunction(chrono.ChFunction_Const(0.5))
viper.SetDriver(driver)

# Create a real-time visualization using Irrlicht
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Viper Rover Simulation')
vis.AttachSystem(mysystem)
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 3, 6))
vis.AddTypicalLights()

# Set up the simulation loop
steering_time = 10  # seconds
steering_angle = 0.2
timestep = 0.01
time = 0

while time < steering_time:
    # Update the steering angle
    driver.SetSteeringFunction(chrono.ChFunction_Const(steering_angle))
    steering_angle += 0.01

    # Update and render the rover
    mysystem.DoStepDynamics(timestep)
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()

    # Increment the time
    time += timestep

    # Check for keyboard events
    if vis.IsEventHandled():
        break
```

This script initializes the PyChrono environment, sets up the simulation system with gravity and a ground body, and creates a Viper rover with its driver. It then creates a real-time visualization using Irrlicht and implements a simulation loop where the rover's steering gradually changes over a specified time period, updating and rendering the rover at each timestep.
