msection.SetShearModulus(0.01e9 * 0.3)
msection.SetRayleighDamping(0.000)
msection.SetCentroid(0, 0.02)
msection.SetShearCenter(0, 0.1)
msection.SetSectionRotation(45 * chrono.CH_DEG_TO_RAD)

# Define the length of the beam elements
beam_L = 0.1

# Create nodes for the positions that will be used for beams
hnode1 = fea.ChNodeFEAxyzrot()
hnode1.SetPos(chrono.ChVector3d(0, 0, 0))
hnode2 = fea.ChNodeFEAxyzrot()
hnode2.SetPos(chrono.ChVector3d(beam_L, 0, 0))
hnode3 = fea.ChNodeFEAxyzrot()
hnode3.SetPos(chrono.ChVector3d(beam_L * 2, 0, 0))

mesh.AddNode(hnode1)

mesh.AddNode(hnode3)

# Create the first beam element and set its properties

belement1.SetSection(msection)
mesh.AddElement(belement1)

# Create the second beam element and set its properties

belement2.SetSection(msection)

hnode3.SetTorque(chrono.ChVector3d(0, -0.04, 0))

mtruss.SetFixed(True)
sys.Add(mtruss)

constr_bc = chrono.ChLinkMateGeneric()
constr_bc.Initialize(hnode3, mtruss, False, hnode3.GetFrame(), mtruss.GetFrame())
sys.Add(constr_bc)

# Create and initialize a constraint that connects node 1 to the fixed truss
constr_d = chrono.ChLinkMateGeneric()
