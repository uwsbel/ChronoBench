

```python
import pychrono as chrono
import pychrono.ros as chros
import rospy
import std_msgs.msg as std_msgs

# Initialize the Chrono system with gravity
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create a default material for contact
mat = chrono.ChMaterialSurfaceNSC()
mat.SetFriction(0.2)

# Add a fixed floor
floor = chrono.ChBody()
floor.SetBodyFixed(True)
floor.SetPos(chrono.ChVectorD(0, -1, 0))
floor.GetCollisionModel().ClearModel()
floor.GetCollisionModel().AddBox(10, 1, 10)
floor.SetCollide(True)
floor.SetMaterialSurface(mat)
system.Add(floor)

# Add a movable box
box = chrono.ChBody()
box.SetMass(10)
box.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
box.SetPos(chrono.ChVectorD(0, 0.5, 0))
box.GetCollisionModel().ClearModel()
box.GetCollisionModel().AddBox(0.5, 0.5, 0.5)
box.SetCollide(True)
box.SetMaterialSurface(mat)
system.Add(box)

# Custom ROS handler to publish integer messages
class CustomROSHandler(chros.ChROSHandler):
    def __init__(self, topic_name):
        super().__init__()
        self.topic_name = topic_name
        self.publisher = None

    def OnInitialize(self):
        self.publisher = rospy.Publisher(self.topic_name, std_msgs.Int32, queue_size=10)
        return True

    def OnUpdate(self, time, step):
        msg = std_msgs.Int32()
        msg.data = step
        self.publisher.publish(msg)
        return True

# Configure ROS manager with necessary handlers
ros_manager = chros.ChROSManager()
ros_manager.RegisterHandler(chros.ChROSClockHandler())  # Clock handler for /clock
ros_manager.RegisterHandler(chros.ChROSBodyHandler(box, "box_state"))  # Box state publisher
ros_manager.RegisterHandler(chros.ChROSTransformHandler(floor, "floor_transform"))  # Floor transform publisher
ros_manager.RegisterHandler(CustomROSHandler("custom_topic"))  # Custom integer publisher

ros_manager.Initialize()

# Simulation loop with real-time synchronization
step_timer = chrono.ChRealtimeStepTimer()
time = 0.0
step = 0
timestep = 0.01  # 100 Hz simulation step

while True:
    step_timer.Spin(time)  # Maintain real-time execution
    system.DoStepDynamics(timestep)  # Advance Chrono simulation
    time += timestep
    ros_manager.Update(time, step)  # Update ROS communication
    step += 1
```
