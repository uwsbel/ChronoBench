# Camera settings
camera = vis.GetCamera()
camera.SetPosition(chrono.ChVectorD(10, 5, 5))

vis.AddLight(chrono.ChLight(chrono.ChVectorD(5, 5, 5), chrono.ChColor(1, 1, 1)))

# 5. Simulation loop
while vis.Run():

    # Advance the simulation
