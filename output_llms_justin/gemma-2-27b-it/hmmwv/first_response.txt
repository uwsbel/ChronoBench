```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as chronovehicle

# ---------------------------------------------------------------------
#
#  Create the simulation system and add items
#
# ---------------------------------------------------------------------

# Create the physical system
system = chrono.ChSystemNSC()

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle('HMMWV Simulation')
vis.Initialize()
vis.AddCamera(chrono.ChCamera(chrono.ChVectorD(0, 2, -5)))

# Create a ground body
ground = chrono.ChBodyEasyBox(100, 100, 1, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -0.1, 0))
ground.SetBodyFixed(True)
system.AddBody(ground)

# Create the HMMWV vehicle
vehicle = chronovehicle.ChVehicle("HMMWV", system)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChVectorD(0, 0.5, 0))
vehicle.SetInitOrientation(chrono.ChQuaternionD(1, 0, 0, 0))

# Add the HMMWV chassis
vehicle.AddChassisRedundant(chrono.ChChassis.TYPE_SIMPLE, 
                            chrono.ChVectorD(0, 0, 0),
                            chrono.ChVectorD(0, 0, 0),
                            chrono.ChVectorD(1, 1, 1))

# Add wheels
vehicle.AddWheel(chrono.ChWheelInfo.TYPE_TMEASY,
                 chrono.ChVectorD(1.5, 0.5, 0),
                 chrono.ChVectorD(0, -1, 0),
                 chrono.ChVectorD(0, 0, 1))
vehicle.AddWheel(chrono.ChWheelInfo.TYPE_TMEASY,
                 chrono.ChVectorD(-1.5, 0.5, 0),
                 chrono.ChVectorD(0, -1, 0),
                 chrono.ChVectorD(0, 0, 1))
vehicle.AddWheel(chrono.ChWheelInfo.TYPE_TMEASY,
                 chrono.ChVectorD(1.5, -0.5, 0),
                 chrono.ChVectorD(0, -1, 0),
                 chrono.ChVectorD(0, 0, 1))
vehicle.AddWheel(chrono.ChWheelInfo.TYPE_TMEASY,
                 chrono.ChVectorD(-1.5, -0.5, 0),
                 chrono.ChVectorD(0, -1, 0),
                 chrono.ChVectorD(0, 0, 1))

# Set contact method
vehicle.SetContactMethod(chrono.ChContactMethod.NSC)

# Set visualization
vehicle.SetChassisVisualizationType(chrono.ChVisualizationType.PRIMITIVES)
vehicle.SetWheelVisualizationType(chrono.ChVisualizationType.PRIMITIVES)

# ---------------------------------------------------------------------
#
#  Set up the interactive driver system
#
# ---------------------------------------------------------------------

driver = chronovehicle.ChDriver(vehicle)
driver.SetSteeringGain(0.5)
driver.SetThrottleGain(0.5)
driver.SetBrakingGain(0.5)

# ---------------------------------------------------------------------
#
#  Simulation loop
#
# ---------------------------------------------------------------------

# Simulation settings
timestep = 0.01
simulation_time = 10

while vis.Run() and system.GetChTime() < simulation_time:
    # Get driver input
    driver.Update(vis.GetKeyboardInput())

    # Advance the simulation
    system.DoStepDynamics(timestep)

    # Render the scene
    vis.Render()

    # Update the frame rate
    chrono.ChSystem.SetTargetTime(system, timestep)

vis.Deinitialize()
```

**Explanation:**

1. **Initialization:** The script initializes the PyChrono environment, creates a visualization system using Irrlicht, and defines a ground body.

2. **Vehicle Creation:** It then creates an HMMWV vehicle using the `chronovehicle` module, adds a chassis, four wheels with the TMEASY tire model, and sets the contact method to NSC.

3. **Visualization:** The script sets the visualization type for the chassis and wheels to primitives for simplicity.

4. **Driver System:** An interactive driver system is created using `chronovehicle.ChDriver`, allowing for steering, throttle, and braking control.

5. **Simulation Loop:** The main loop runs until the simulation time is reached or the user closes the window. It updates the driver input, advances the simulation dynamics, renders the scene, and controls the frame rate.
