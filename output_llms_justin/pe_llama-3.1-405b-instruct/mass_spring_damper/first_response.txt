```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr

# Initialize the PyChrono environment and core components
sys = chrono.ChSystemNSC()
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Mass-Spring-Damper System')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 2, 4))
vis.AddTypicalLights()

# Create the mass and ground bodies
mass_body = chrono.ChBody()
mass_body.SetPos(chrono.ChVectorD(0, 1, 0))
mass_body.SetMass(10)
mass_body.SetInertiaXX(chrono.ChVectorD(0.1, 0.1, 0.1))

ground_body = chrono.ChBody()
ground_body.SetBodyFixed(True)

# Create the ChLinkTSDA connection between the mass and ground
link = chrono.ChLinkTSDA()
link.Initialize(mass_body, ground_body, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))
link.SetSpringCoefficient(100)
link.SetDampingCoefficient(10)

# Add visual shapes for the mass and spring
mass_shape = chrono.ChVisualShapeSphere(0.2)
mass_body.AddVisualShape(mass_shape)

spring_shape = chrono.ChVisualShapeCylinder(0.05, 1)
link.AddVisualShape(spring_shape)

# Add the mass, ground, and link to the system
sys.Add(mass_body)
sys.Add(ground_body)
sys.Add(link)

# Simulation loop
while vis.Run():
    sys.DoStepDynamics(0.01)
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()
```

This script creates a simple mass-spring-damper system using PyChrono, with Irrlicht visualization and default linear spring-damper parameters for the `ChLinkTSDA` connection. The simulation includes visual shapes for the mass and spring, and the camera and lighting settings are adjusted for better visibility. The script runs the simulation loop, updating the system dynamics and rendering the scene at each time step.
