sys.Add(mjointB)

mjointC = chrono.ChLinkLockPrismatic()
mjointC.Initialize(mpiston, mfloor, chrono.ChFrameD(crank_center + chrono.ChVectorD(crank_rad + rod_length, 0, 0), chrono.Q_ROTATE_Z_TO_X))
sys.Add(mjointC)

vis = chronoirr.ChVisualSystemIrrlicht()

vis.SetWindowTitle('Crank demo')

vis.AddTypicalLights()

# Arrays for plotting
array_time = []
array_angle = []
array_pos = []
array_speed = []

while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(1e-3)
    
    current_time = sys.GetChTime()
    current_angle = my_motor.GetMotorRot()
    current_pos = mpiston.GetPos().x
    current_speed = mpiston.GetPos_dt().x
    
    array_time.append(current_time)
    array_angle.append(current_angle)
    array_pos.append(current_pos)
    array_speed.append(current_speed)
    
    if current_time >= 20:
        break
