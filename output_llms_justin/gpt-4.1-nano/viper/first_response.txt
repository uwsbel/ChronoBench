Certainly! Here's a comprehensive PyChrono script that follows your instructions. It initializes the simulation environment with gravity, sets up a rigid terrain, adds a Viper rover, initializes a driver, configures real-time visualization with Irrlicht, and implements a simulation loop where the rover's steering gradually changes over time.

```python
import pychrono.core as chol
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# 1. Initialize the Chrono system with gravity
system = chol.ChSystemNSC()
gravity = chol.ChVectorD(0, -9.81, 0)
system.Set_G_acc(gravity)

# 2. Create a rigid terrain (ground body)
ground = chol.ChBodyEasyBox(50, 1, 50, 1000, True, True)
ground.SetPosition(chol.ChVectorD(0, -0.5, 0))
ground.SetBodyFixed(True)
system.Add(ground)

# Add contact surface properties
groundSurface = chol.ChMaterialSurfaceNSC()
groundSurface.SetFriction(0.8)
ground.GetCollisionModel().ClearModel()
ground.GetCollisionModel().AddBox(50, 1, 50, chol.ChVectorD(0, 0, 0), chol.QUNIT)
ground.GetCollisionModel().BuildModel()
ground.SetMaterialSurface(groundSurface)

# 3. Initialize the Viper rover
# Using the Vehicle model for Viper
viper = veh.Viper(vehicle_step=1e-3)

# Set initial position for the rover
initial_pos = chol.ChCoordsysD(chol.ChVectorD(0, 1, 0), chol.Q_from_AngY(0))
viper.GetChassisBody().SetPos(initial_pos)

# Add the rover to the system
viper.Initialize(system)

# Set the visualization to follow the rover
viper.GetVehicle().GetChassis().SetFixed(False)

# 4. Initialize the driver (apply controls)
driver = veh.ChIrrGuiDriver(viper.GetVehicle())

# 5. Set up Irrlicht visualization
application = irr.ChIrrApp(system, 'Viper Rover Simulation', irr.dimension2du(800, 600))
application.GetSceneManager().AddSkyBox()
application.GetVideoDriver().SetLightPosition(irr.vector3df(30, 45, 30))
application.GetSceneManager().AddCameraSceneNode(irr.E_SCENE_NODE_TYPE.ENU_CAMERA,
                                                 irr.vector3df(0, 5, -15),
                                                 irr.vector3df(0, 2, 0))
application.AssetBindAll()
application.AssetUpdateAll()

# Enable shadows for better visualization
application.SetShadows473()
application.AssetBindAll()
application.AssetUpdateAll()

# 6. Simulation parameters
time_step = 1e-3  # 1 ms timestep
simulation_time = 20.0  # total simulation time in seconds
steering_change_duration = 10.0  # seconds over which steering increases
max_steering_angle = 0.5  # radians (~28.6 degrees)

# Initialize steering control
initial_steering = 0.0
target_steering = max_steering_angle

# 7. Main simulation loop
t = 0.0
while t < simulation_time:
    # Update driver inputs: gradually increase steering over first 'steering_change_duration' seconds
    if t <= steering_change_duration:
        steering_ratio = t / steering_change_duration
        current_steering = initial_steering + steering_ratio * (target_steering - initial_steering)
    else:
        current_steering = target_steering

    # Set driver inputs
    driver.SetSteering(current_steering)
    driver.SetThrottle(0.8)  # constant throttle
    driver.SetBraking(0.0)

    # Integrate system
    system.DoStepDynamics(time_step)

    # Begin scene rendering
    application.BeginScene()
    application.DrawAll()
    application.EndScene()

    # Advance in time
    t += time_step

    # Keep the application responsive
    if application.GetDevice().run():
        continue
    else:
        break

# 8. Cleanup
application.GetDevice().closeDevice()
```

**Notes:**

- This script sets up a large ground plane as the terrain with friction.
- The Viper rover is initialized at position `(0,1,0)` and starts from rest.
- The driver's steering gradually increases from 0 to 0.5 radians over 10 seconds.
- The simulation runs for 20 seconds, with real-time visualization enabled.
- Adjust parameters such as `time_step`, `simulation_time`, and `max_steering_angle` as needed for your specific scenario.
