Here's a PyChrono simulation script that meets your requirements. This script loads a triangular mesh from a Wavefront .obj file, creates a fixed body with it, adds a lidar sensor with noise filters and visualization, and simulates the system while dynamically updating the lidar's position in an orbit around the mesh.

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import numpy as np
import os

# Set the path to the Chrono data directory
chrono.SetChronoDataPath(os.path.join(os.path.dirname(__file__), '..', '..', 'data'))

# Create a Chrono physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle("Lidar Simulation with Triangular Mesh")
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 1, 5))
vis.AddTypicalLights()
vis.SetCameraVertical(chrono.ChCameraVerticalDir::Z)

# Load the triangular mesh from a Wavefront .obj file
mesh_filename = "mesh.obj"  # Replace with your actual .obj file path
mesh = chrono.ChTriangleMeshConnected()
mesh.LoadWavefrontMesh(mesh_filename, False, True)

# Create a body with the mesh
body = chrono.ChBody()
body.SetBodyFixed(True)  # Make it a fixed body
body.GetVisualModel().AddTriangleMesh(mesh)
body.GetCollisionModel().AddTriangleMesh(mesh)
system.Add(body)

# Create a sensor manager
sensor_manager = chrono.sensor.ChSensorManager(system)

# Create a lidar sensor
lidar = chrono.sensor.ChLidarSensor()
lidar.SetName("LidarSensor")
lidar.SetUpdateRate(100)  # Hz
lidar.SetCollectInterval(0.01)  # seconds

# Set lidar parameters
lidar.SetRayCount(100, 100)  # 100x100 grid
lidar.SetRayLength(10.0)  # meters
lidar.SetRaySpreadAngle(chrono.ChVector2d(60, 60))  # degrees

# Add noise filters to the lidar
noise_model = chrono.sensor.ChGaussianNoiseModel()
noise_model.SetNoiseAmount(0.01)  # 1% noise
lidar.AddNoiseModel(noise_model)

# Create a visualization for the lidar
lidar_visualization = chrono.sensor.ChLidarVisualization(system)
lidar_visualization.SetLidar(lidar)
lidar_visualization.SetPointSize(0.01)  # Size of points in visualization
lidar_visualization.SetColor(chrono.ChColor(1, 0, 0))  # Red color

# Add the lidar to the sensor manager
sensor_manager.AddSensor(lidar)

# Initialize the lidar position (orbit parameters)
orbit_radius = 2.0
orbit_height = 1.0
angular_speed = 0.5  # radians per second
current_angle = 0.0

# Simulation parameters
time_step = 0.01
end_time = 10.0

# Simulation loop
while vis.Run():
    current_time = system.GetChTime()

    # Update the lidar position in an orbit around the mesh
    current_angle += angular_speed * time_step
    lidar_pos = chrono.ChVectorD(
        orbit_radius * np.cos(current_angle),
        orbit_height,
        orbit_radius * np.sin(current_angle)
    )

    # Set the lidar position and orientation (pointing toward the mesh)
    lidar.SetOffsetPos(lidar_pos)
    lidar.SetOffsetRot(chrono.ChQuaternionD().Q_from_AngAxis(-current_angle, chrono.ChVectorD(0, 1, 0)))

    # Update the sensor manager
    sensor_manager.Update()

    # Print lidar buffer data at each simulation step
    if lidar.GetBufferSize() > 0:
        buffer = lidar.GetMostRecentBuffer()
        print(f"Time: {current_time:.2f}s - Lidar points: {buffer.size()}")

    # Advance the simulation
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    system.DoStepDynamics(time_step)

    if current_time >= end_time:
        break

# Save the lidar data to a file (optional)
lidar.SaveBufferToFile("lidar_data.csv")
```

### Key Features:
1. **Initialization**: Sets up the Chrono physical system and Irrlicht visualization.
2. **Mesh Loading**: Loads a triangular mesh from a Wavefront .obj file and creates a fixed body with it.
3. **Lidar Sensor**: Adds a lidar sensor with noise filters and visualization.
4. **Dynamic Orbit**: Updates the lidar's position in an orbit around the mesh during simulation.
5. **Data Output**: Prints lidar buffer data at each simulation step and optionally saves it to a file.

### Notes:
- Replace `"mesh.obj"` with the path to your actual Wavefront .obj file.
- Adjust the lidar parameters (ray count, spread angle, etc.) as needed for your application.
- The simulation runs for 10 seconds by default, but you can change `end_time` to suit your needs.
