Here's a complete PyChrono simulation script that demonstrates the use of beam finite elements with visualization using the Irrlicht engine:

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath('PATH_TO_CHRONO_DATA')  # Set path to Chrono data directory
sys = chrono.ChSystemNSC()  # Create a non-smooth (NSC) system
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# 2. Create a beam finite element mesh
# Create a simple cantilever beam with 5 elements
num_elements = 5
beam_length = 0.5  # meters
beam_width = 0.05  # meters
beam_height = 0.05  # meters
density = 7800     # kg/m^3 (steel)

# Create nodes
nodes = []
for i in range(num_elements + 1):
    x_pos = i * beam_length / num_elements
    node = chrono.ChNodeFEAxyzrot()
    node.SetPos(chrono.ChVectorD(x_pos, 0, 0))
    node.SetMass(beam_width * beam_height * beam_length/num_elements * density / (num_elements + 1))
    node.SetInertiaXX(chrono.ChVectorD(
        (beam_width**2 + beam_height**2) * beam_length/num_elements * density / 12,
        beam_width**2 * beam_length/num_elements * density / 12,
        beam_height**2 * beam_length/num_elements * density / 12
    ))
    sys.Add(node)
    nodes.append(node)

# Create beam elements
for i in range(num_elements):
    beam = chrono.ChElementBeamEuler()
    beam.SetNodes(nodes[i], nodes[i+1])

    # Set beam properties
    beam.SetBeamSection(chrono.ChBeamSectionAdvanced())
    beam.GetBeamSection().SetAsRectangularSection(beam_width, beam_height)
    beam.GetBeamSection().SetYoungModulus(2.1e11)  # Steel Young's modulus
    beam.GetBeamSection().SetShearModulus(7.9e10)  # Steel shear modulus
    beam.GetBeamSection().SetPoissonRatio(0.3)     # Steel Poisson ratio
    beam.GetBeamSection().SetDensity(density)

    sys.Add(beam)

# 3. Fix the first node to create a cantilever beam
fix_constraint = chrono.ChLinkLockLock()
fix_constraint.Initialize(nodes[0], chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))
sys.Add(fix_constraint)

# 4. Add visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle('Beam FEA Simulation')
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(1, 0.5, 1), chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVectorD(1.5, 5, 1.5), chrono.ChVectorD(0, 0, 0), 3, 0.5, 5, 30, 512)

# 5. Simulation loop
time_step = 0.001
vis.SetStep(0.01)  # For smoother visualization
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    sys.DoStepDynamics(time_step)
```

Notes about the script:
1. The script creates a cantilever beam with 5 Euler-Bernoulli beam elements.
2. The beam has rectangular cross-section with properties similar to steel.
3. The first node is fixed to create a cantilever boundary condition.
4. The Irrlicht visualization system is used for real-time rendering.
5. The simulation uses a time step of 0.001 seconds for numerical stability.

To run this script, you'll need to:
1. Have PyChrono properly installed
2. Set the correct path to the Chrono data directory
3. Have the Irrlicht visualization module available

You can modify the beam properties (length, width, height, material properties) or add external forces to see different behaviors.
