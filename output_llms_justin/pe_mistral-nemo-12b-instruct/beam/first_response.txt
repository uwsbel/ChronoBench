```python
# Import necessary libraries
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.fea as fea

# Set Chrono's data path
chrono.SetChronoDataPath(os.path.join(os.getcwd(), '..', 'data'))

# Initialize the system with Signorini-Moreau Contact (SMC)
sys = chrono.ChSystemSMC()

# Define beam properties
beam_L = 5.0
beam_wy = 0.1
beam_wz = 0.05

# Create a mesh
mesh = fea.ChMesh()

# Define beam section properties
msection = fea.ChBeamSectionEulerAdvanced()
msection.SetAsRectangularSection(beam_wy, beam_wz)
msection.SetYoungModulus(1.0e9)
msection.SetShearModulus(0.3 * 1.0e9)
msection.SetRayleighDamping(0.000)

# Add nodes to the mesh
hnode1 = fea.ChNodeFEAxyzrot(chrono.ChFramed(chrono.ChVector3d(0, 0, 0)))
hnode2 = fea.ChNodeFEAxyzrot(chrono.ChFramed(chrono.ChVector3d(beam_L, 0, 0)))
mesh.AddNode(hnode1)
mesh.AddNode(hnode2)

# Add beam element to the mesh
belement1 = fea.ChElementBeamEuler()
belement1.SetNodes(hnode1, hnode2)
belement1.SetSection(msection)
mesh.AddElement(belement1)

# Add the mesh to the system
sys.Add(mesh)

# Initialize the visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Beam FEA Simulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 6))
vis.AddTypicalLights()

# Simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    sys.DoStepDynamics(0.01)
    vis.EndScene()
```

This script initializes a PyChrono environment with a beam finite element simulation. It creates a mesh composed of beams and nodes with specified properties, adds the mesh to the physical system, and sets up visualization using the Irrlicht engine. The simulation loop runs with a time step of 0.01 seconds, updating the system and rendering each frame.
