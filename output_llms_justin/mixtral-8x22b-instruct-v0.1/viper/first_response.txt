```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import math
import time

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath("/path/to/chrono/data")  # Replace with your Chrono data path
chrono.SetChronoDataPath("/path/to/irrlicht/data")  # Replace with your Irrlicht data path

my_system = chrono.ChSystemNSC()
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# 2. Add the required physical systems and objects

# Create ground body
ground = chrono.ChBodyEasyBox(20, 1, 20, 1000, True, True)
ground.SetBodyFixed(True)
my_system.Add(ground)

# Create Viper rover
viper = chrono.ChBodyEasyCylinder(0.3, 0.1, 1000, True, True)
viper.SetPos(chrono.ChVectorD(0, 0.1, 0))
my_system.Add(viper)

# Create four wheels for the Viper rover
wheel_radius = 0.1
wheel_width = 0.1
wheel_mass = 1
wheel_positions = [chrono.ChVectorD(-0.2, 0.05, 0.3), chrono.ChVectorD(-0.2, 0.05, -0.3),
                   chrono.ChVectorD(0.2, 0.05, 0.3), chrono.ChVectorD(0.2, 0.05, -0.3)]

for pos in wheel_positions:
    wheel = chrono.ChBodyEasyCylinder(wheel_radius, wheel_width, wheel_mass, True, True)
    wheel.SetPos(pos)
    my_system.Add(wheel)

# 3. Set necessary default parameters such as positions, forces, and interactions

# Create a driver for the Viper rover
driver = chrono.ChIrrGuiDriver(my_system, chronoirr.ChIrrApp(my_system, "Viper Rover", chronoirr.dimension2du(800, 600)))

# Set camera settings and lighting
driver.GetView()->SetPos(chrono.ChVectorD(0, 1, -3))
driver.GetView()->SetFar(100)
driver.GetLight()->SetPos(chrono.ChVectorD(5, 5, -5))
driver.GetLight()->SetDirection(chrono.ChVectorD(-1, -1, -1))

# Set simulation time parameters
total_time = 10  # Simulation duration (seconds)
timestep = 0.01  # Timestep (seconds)

# Implement a simulation loop
start_time = time.time()
current_time = 0
steering_change_duration = 5  # Time for steering to change (seconds)
steering_angle_change = 30 * math.pi / 180  # Steering angle change (radians)

while current_time < total_time:
    # Update the steering angle over time
    steering_angle = steering_angle_change * min(current_time / steering_change_duration, 1)

    # Update and render the rover at each timestep
    my_system.DoStepDynamics(timestep)
    driver.BeginScene()
    driver.DrawAll()
    driver.EndScene()

    current_time = time.time() - start_time

driver.Close()
```
