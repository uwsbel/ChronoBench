```python
import os
import math
import numpy as np
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import rospy
from std_msgs.msg import Int32

# Initialize the PyChrono environment
chrono.SetChronoDataPath('/path/to/data')

# Create the Chrono system
sys = chrono.ChSystemNSC()

# Set the gravitational acceleration
sys.Set_G(chrono.ChVectorD(0, -9.81, 0))

# Create the contact material
material = chrono.ChMaterialSurfaceNSC()
material.SetFriction(0.5)
material.SetDampingF(0.1)
material.SetCompliance(0.01)

# Create the floor
floor = chrono.ChBodyEasyBox(sys, 10, 1, 10, 1000, True, material)
floor.SetPos(chrono.ChVectorD(0, -5, 0))
floor.SetBodyFixed(True)
sys.Add(floor)

# Create the box
box = chrono.ChBodyEasyBox(sys, 1, 1, 1, 1000, True, material)
box.SetPos(chrono.ChVectorD(0, 5, 0))
sys.Add(box)

# Create a custom ROS handler
class CustomROSHnadler:
    def __init__(self):
        self.pub = rospy.Publisher('custom_topic', Int32, queue_size=10)

    def Update(self):
        msg = Int32()
        msg.data = 1
        self.pub.publish(msg)

# Initialize the ROS node
rospy.init_node('pychrono_ros')

# Create the ROS manager
ros_manager = veh.ROSManager()

# Create the clock handler
clock_handler = veh.ROSClockHandler()

# Create the body handler
body_handler = veh.ROSBodyHandler()

# Create the transform handler
transform_handler = veh.ROSTransformHandler()

# Register the handlers
ros_manager.RegisterHandler(clock_handler)
ros_manager.RegisterHandler(body_handler)
ros_manager.RegisterHandler(transform_handler)
ros_manager.RegisterHandler(CustomROSHnadler())

# Run the simulation loop
while not rospy.is_shutdown():
    sys.DoStepDynamics(0.01)
    ros_manager.Update()
    rospy.sleep(0.01)
```

This code initializes the PyChrono environment, sets up a Chrono system with gravitational acceleration, and adds a fixed floor and a movable box to the system. It also creates a custom ROS handler to publish integer messages to a specified topic and configures the ROS manager with clock, body, and transform handlers. The simulation loop advances the system and updates ROS communication at each timestep, maintaining real-time execution.
