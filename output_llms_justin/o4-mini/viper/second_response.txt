Below is a self‐contained, corrected, and enhanced PyChrono script that

 1. Fixes the API‐name and syntax issues in the original (e.g. uses  
    Set_G_acc, ChMaterialSurfaceNSC, SetBodyFixed, ChFrameD, etc.).  
 2. Adds the sensor imports and manager.  
 3. Adds a point light to the sensor scene.  
 4. Creates a third‐person camera sensor mounted on the rover.  
 5. Advances the sensor manager each step.  
 6. Only renders the Irrlicht window at 25 FPS.  

```python
import os
import math
import numpy as np

import pychrono as chrono
import pychrono.robot as viper
import pychrono.sensor as sens
from pychrono import irrlicht as chronoirr

# ------------------------------------------------------------------
# Create Chrono system
# ------------------------------------------------------------------
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
# correct gravity setter
system.Set_G_acc(chrono.ChVector3d(0, 0, -9.81))

chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# ------------------------------------------------------------------
# Create ground body (rigid box)
# ------------------------------------------------------------------
ground_mat = chrono.ChMaterialSurfaceNSC()
ground = chrono.ChBodyEasyBox(
    20.0,   # x-length
    20.0,   # y-length
    1.0,    # z-thickness
    1000,   # density
    True,   # visualization?
    True,   # collision?
    ground_mat
)
ground.SetPos(chrono.ChVector3d(0, 0, -1.0))
ground.SetBodyFixed(True)
# apply a concrete texture
ground.GetVisualShape(0).SetTexture(
    chrono.GetChronoDataFile("textures/concrete.jpg")
)
system.Add(ground)

# ------------------------------------------------------------------
# Create Viper Rover + driver
# ------------------------------------------------------------------
rover = viper.Viper(system)
driver = viper.ViperDCMotorControl()
rover.SetDriver(driver)

# initial pose: small offset in Y, identity rotation
init_pos = chrono.ChVector3d(0, 0.2, 0)
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)
# ChFrameD takes (quat, pos)
initial_frame = chrono.ChFrameD(init_rot, init_pos)
rover.Initialize(initial_frame)

# ------------------------------------------------------------------
# Set up Irrlicht visualization
# ------------------------------------------------------------------
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle("Viper rover - Rigid terrain")
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.AddSkyBox()
# a fixed “free” camera looking at the scene
vis.AddCamera(
    chrono.ChVector3d(0, 2.5, 1.5),
    chrono.ChVector3d(0, 0, 1)
)
vis.AddTypicalLights()
vis.AddLightWithShadow(
    chrono.ChVector3d(1.5, -2.5, 5.5),
    chrono.ChVector3d(0, 0, 0.5),
    3, 4, 10, 40, 512
)

# ------------------------------------------------------------------
# Create a Sensor Manager and add a point light for the sensors
# ------------------------------------------------------------------
manager = sens.ChSensorManager(system)
intensity = 1.0
# note: in some versions use manager.GetScene().AddPointLight(...)
manager.scene.AddPointLight(
    chrono.ChVector3f(2, 2.5, 100),
    chrono.ChColor(intensity, intensity, intensity),
    500.0
)

# ------------------------------------------------------------------
# Create a third‐person camera sensor on the rover chassis
# ------------------------------------------------------------------
# define the camera frame relative to the chassis:
# translate (1,0,1.45), pitch down by 0.2 rad about Y
q_pitch = chrono.QuatFromAngleAxis(0.2, chrono.ChVector3d(0, 1, 0))
offset_pose = chrono.ChFrameD(q_pitch, chrono.ChVector3d(1.0, 0, 1.45))

cam = sens.ChCameraSensor(
    rover.GetChassis().GetBody(),  # parent body
    15,                            # update rate [Hz]
    offset_pose,                   # pose in the body frame
    720,                           # image width
    480,                           # image height
    1.408                          # vertical FOV [rad]
)
cam.SetName("Third Person POV")
# push a built‐in OpenGL visualizer for debugging
cam.PushFilter(sens.ChFilterVisualize(720, 480, "Viper Front Camera"))
manager.AddSensor(cam)

# ------------------------------------------------------------------
# Simulation parameters
# ------------------------------------------------------------------
time_step = 1e-3
step_number = 0

# we only want to render at ~25 FPS
render_step_size = 1.0 / 25.0
render_steps = math.ceil(render_step_size / time_step)

# ------------------------------------------------------------------
# Main simulation loop
# ------------------------------------------------------------------
while vis.Run():
    # read the Chrono system time
    t = system.GetChTime()

    # simple steering profile
    max_steer = math.pi / 6.0
    steer = 0.0
    if 2.0 < t < 7.0:
        steer = max_steer * (t - 2.0) / 5.0
    elif 7.0 <= t < 12.0:
        steer = max_steer * (12.0 - t) / 5.0

    driver.SetSteering(steer)
    rover.Update()

    # advance the physics
    system.DoStepDynamics(time_step)

    # advance the sensors
    manager.Update()

    # render only every N steps to achieve ~25 FPS
    if (step_number % render_steps) == 0:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()

    step_number += 1
```

Explanation of the key fixes and additions:

  • Replaced `SetGravitationalAcceleration` with the correct `Set_G_acc`.  
  • Used `ChMaterialSurfaceNSC` instead of the non‐existent `ChContactMaterialNSC`.  
  • Called `SetBodyFixed(True)` on the ground (the correct method name).  
  • Initialized the rover with a `ChFrameD(quat, pos)`, not the undefined `ChFramed`.  
  • Imported `pychrono.sensor` as `sens`, created a `ChSensorManager`,  
    and added a point light to `manager.scene`.  
  • Built and attached a `ChCameraSensor` with a small pitch angle.  
  • Introduced `step_number`, `render_step_size`, and `render_steps` so that  
    the Irrlicht visualization updates at ~25 FPS, while the physics and  
    sensors run at 1 kHz.
