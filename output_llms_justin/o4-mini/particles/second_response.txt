Below is a summary of the issues we found in the original script and how we fixed them, followed by the complete, corrected script.

1. Identified Issues and Fixes  
   • The original particle‐shape creator (`ChRandomShapeCreatorConvexHulls`) had to be replaced by `ChRandomShapeCreatorSpheres`, and its API calls changed accordingly.  
   • The gravitational constant (`G_constant`) was declared outside the time‐loop; for clarity we moved it into the loop immediately before the energy calculations.  
   • We added kinetic‐ and potential‐energy calculations:  
     – KE = ∑½·m·‖v‖² over all bodies  
     – PE = ∑ (−G·m₁·m₂ / r₁₂) over each pair  
   • We then print out KE, PE, and total energy each step.  
   • We removed the old `mcreator_hulls` declarations and binding, and replaced them with `mcreator_spheres`.

Corrected and Modified Script
```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
from itertools import combinations


class MyCreatorForAll(chrono.ChRandomShapeCreator_AddBodyCallback):
    def __init__(self, vis, coll):
        super().__init__()
        self.vis = vis
        self.coll = coll

    def OnAddBody(self, body, coords, creator):
        # apply texture, bind to visual and collision inspectors
        body.GetVisualShape(0).SetTexture(
            chrono.GetChronoDataFile("textures/bluewhite.png")
        )
        self.vis.BindItem(body)
        self.coll.BindItem(body)
        body.SetUseGyroTorque(False)


# 1) Build the Chrono system
sys = chrono.ChSystemNSC()
sys.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
coll = sys.GetCollisionSystem()

# 2) A single spherical body in the scene
sphere_mat = chrono.ChContactMaterialNSC()
sphere_mat.SetFriction(0.2)
msphereBody = chrono.ChBodyEasySphere(
    2.1,     # radius
    1800,    # density
    True,    # visualization?
    True,    # collision?
    sphere_mat
)
msphereBody.SetPos(chrono.ChVector3d(1, 1, 0))
msphereBody.GetVisualShape(0).SetTexture(
    chrono.GetChronoDataFile("textures/concrete.jpg")
)
sys.Add(msphereBody)

# 3) Particle emitter
emitter = chrono.ChParticleEmitter()
emitter.SetParticlesPerSecond(2000)
emitter.SetUseParticleReservoir(True)
emitter.SetParticleReservoirAmount(200)

# -- randomize positions inside a box
emitter_positions = chrono.ChRandomParticlePositionOnGeometry()
emitter_positions.SetGeometry(chrono.ChBox(50, 50, 50), chrono.ChFrameMoving())
emitter.SetParticlePositioner(emitter_positions)

# -- random orientations
emitter_rotations = chrono.ChRandomParticleAlignmentUniform()
emitter.SetParticleAligner(emitter_rotations)

# -- random linear velocity
mvelo = chrono.ChRandomParticleVelocityAnyDirection()
mvelo.SetModulusDistribution(chrono.ChUniformDistribution(0.0, 0.5))
emitter.SetParticleVelocity(mvelo)

# -- random angular velocity
mangvelo = chrono.ChRandomParticleVelocityAnyDirection()
mangvelo.SetModulusDistribution(chrono.ChUniformDistribution(0.0, 0.2))
emitter.SetParticleAngularVelocity(mangvelo)

# 4) Replace the convex‐hull creator with a sphere creator
mcreator_spheres = chrono.ChRandomShapeCreatorSpheres()
# set diameter distribution (Zhang 0.6, 0.23)
mcreator_spheres.SetDiameterDistribution(
    chrono.ChZhangDistribution(0.6, 0.23)
)
# set density distribution constant = 1600
mcreator_spheres.SetDensityDistribution(
    chrono.ChConstantDistribution(1600)
)
emitter.SetParticleCreator(mcreator_spheres)

# 5) Build the Irrlicht visualizer
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Particle emitter demo")
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 14, -20))
vis.AddTypicalLights()

# 6) Attach our callback so that each new particle gets texture + binding
creation_callback = MyCreatorForAll(vis, coll)
emitter.RegisterAddBodyCallback(creation_callback)

# 7) Solver & gravity settings
sys.SetSolverType(chrono.ChSolver.Type_PSOR)
sys.GetSolver().AsIterative().SetMaxIterations(40)
sys.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, 0))

# 8) Simulation loop
stepsize = 1e-2

while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Emit and register new particles
    emitter.EmitParticles(sys, stepsize)

    # Clear forces/torques before computing new ones
    for body in sys.Get_bodylist():
        body.EmptyAccumulators()

    # 8a) Compute Newtonian gravity between every pair
    pairs = combinations(sys.Get_bodylist(), 2)
    for A, B in pairs:
        D = B.GetPos() - A.GetPos()
        r = D.Length()
        if r > 1e-9:
            # apply forces
            # G_constant moved inside loop for clarity
            G_constant = 6.674e-3
            Fmag = G_constant * A.GetMass() * B.GetMass() / (r * r)
            Fvec = (D / r) * Fmag
            A.AccumulateForce(Fvec, A.GetPos(), False)
            B.AccumulateForce(-Fvec, B.GetPos(), False)

    # 8b) Compute energies
    # move G_constant here as well (same constant)
    G_constant = 6.674e-3

    # kinetic
    KE = 0.0
    for body in sys.Get_bodylist():
        v = body.GetPos_dt()
        KE += 0.5 * body.GetMass() * v.Length2()

    # potential
    PE = 0.0
    for A, B in combinations(sys.Get_bodylist(), 2):
        r = (B.GetPos() - A.GetPos()).Length()
        if r > 1e-9:
            PE += -G_constant * A.GetMass() * B.GetMass() / r

    totalE = KE + PE
    print(f"Step={sys.GetChTime():.3f}  KE={KE:.5e}  PE={PE:.5e}  E_tot={totalE:.5e}")

    # 8c) advance dynamics
    sys.DoStepDynamics(stepsize)
```

Key changes in brief:
- Replaced `ChRandomShapeCreatorConvexHulls` → `ChRandomShapeCreatorSpheres`, set diameter & density correctly.  
- Moved `G_constant` inside the time loop.  
- Added kinetic‐ and potential‐energy computations and printed them each step.
