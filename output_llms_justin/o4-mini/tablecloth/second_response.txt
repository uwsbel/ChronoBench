```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.fea as fea
import pychrono.pardisomkl as mkl
import errno
import os

# ------------------------------------------------------------------------
# Output directory setup
# ------------------------------------------------------------------------
out_dir = os.path.join(chrono.GetChronoOutputPath(), "FEA_SHELLS_BST")
try:
    os.mkdir(out_dir)
except OSError as exc:
    if exc.errno != errno.EEXIST:
        print("Error creating output directory")

# ------------------------------------------------------------------------
# Create Chrono physical system
# ------------------------------------------------------------------------
sys = chrono.ChSystemSMC()

# ------------------------------------------------------------------------
# Create and add mesh to the system
# ------------------------------------------------------------------------
mesh = fea.ChMesh()
sys.Add(mesh)

# ------------------------------------------------------------------------
# Material properties
# ------------------------------------------------------------------------
density   = 100
E         = 6e4
nu        = 0.0
thickness = 0.01

# Create material
melasticity = fea.ChElasticityKirchhoffIsothropic(E, nu)
material    = fea.ChMaterialShellKirchhoff(melasticity)
material.SetDensity(density)

# ------------------------------------------------------------------------
# Mesh dimensions
# ------------------------------------------------------------------------
L_x, L_z       = 1.0, 1.0
nsections_x = 40
nsections_z = 40

# ------------------------------------------------------------------------
# Create nodes
# ------------------------------------------------------------------------
mynodes = []
for iz in range(nsections_z + 1):
    for ix in range(nsections_x + 1):
        p = chrono.ChVector3d(
            ix * (L_x / nsections_x),
            0,
            iz * (L_z / nsections_z),
        )
        mnode = fea.ChNodeFEAxyz(p)
        mesh.AddNode(mnode)
        mynodes.append(mnode)

# ------------------------------------------------------------------------
# Node monitoring and loading setup
# ------------------------------------------------------------------------
# pick two nodes for later plotting
nodePlotA = mynodes[0]       # corner
nodePlotB = mynodes[-1]      # opposite corner

# gather nodes where we want to apply loads
nodesLoad = [nodePlotB]

# interpolation functions (for reference tracking, e.g. plotting)
ref_X = chrono.ChFunction_Recorder()
ref_Y = chrono.ChFunction_Recorder()

# load force vector
load_force = chrono.ChVectorD(0, -100, 0)

# apply loads to selected nodes
for node in nodesLoad:
    # Note: using ChLoadFEAxyz to apply a point load to an FEA node
    load = fea.ChLoadFEAxyz(node, load_force)
    mesh.AddLoad(load)

# ------------------------------------------------------------------------
# Create elements (triangle BST shells) and assign a monitor element
# ------------------------------------------------------------------------
mnodemonitor     = None
melementmonitor  = None

for iz in range(nsections_z):
    for ix in range(nsections_x):
        # -- first triangle element A --
        melementA = fea.ChElementShellBST()

        # boundary nodes with checks
        b1 = mynodes[(iz+1)*(nsections_x+1) + (ix+1)]
        b2 = mynodes[(iz+1)*(nsections_x+1) + (ix-1)] if ix > 0 else None
        b3 = mynodes[(iz-1)*(nsections_x+1) + (ix+1)] if iz > 0 else None

        n1 = mynodes[ iz   *(nsections_x+1) + ix    ]
        n2 = mynodes[ iz   *(nsections_x+1) + ix+1  ]
        n3 = mynodes[(iz+1)*(nsections_x+1) + ix    ]

        melementA.SetNodes(n1, n2, n3, b1, b2, b3)
        melementA.AddLayer(thickness, 0, material)
        mesh.AddElement(melementA)

        # monitor this element if at iz==0, ix==1
        if iz == 0 and ix == 1:
            mnodemonitor    = n1
            melementmonitor = melementA

        # -- second triangle element B --
        melementB = fea.ChElementShellBST()

        b1 = mynodes[ iz   *(nsections_x+1) + ix    ]
        b2 = mynodes[ iz   *(nsections_x+1) + ix+2  ] if ix < nsections_x-1 else None
        b3 = mynodes[(iz+2)*(nsections_x+1) + ix    ] if iz < nsections_z-1 else None

        n1b = mynodes[(iz+1)*(nsections_x+1) + ix+1]
        n2b = mynodes[(iz+1)*(nsections_x+1) + ix  ]
        n3b = mynodes[ iz   *(nsections_x+1) + ix+1]

        melementB.SetNodes(n1b, n2b, n3b, b1, b2, b3)
        melementB.AddLayer(thickness, 0, material)
        mesh.AddElement(melementB)

# ------------------------------------------------------------------------
# Fix upper nodes (make them clamped)
# ------------------------------------------------------------------------
for j in range(30):
    for k in range(30):
        idx = j * (nsections_x+1) + k
        mynodes[idx].SetFixed(True)

# ------------------------------------------------------------------------
# Create visualizations for shell elements
# ------------------------------------------------------------------------
# Visualization A: smooth faces, wireframe, optional backface culling
mvisualizeshellA = chrono.ChVisualShapeFEA(mesh)
mvisualizeshellA.SetShellResolution(2)
mvisualizeshellA.SetSmoothFaces(True)
mvisualizeshellA.SetWireframe(True)
# mvisualizeshellA.SetBackfaceCulling(True)   # optional
mesh.AddVisualShapeFEA(mvisualizeshellA)

# Visualization B: glyphs, no FEM data coloring
mvisualizeshellB = chrono.ChVisualShapeFEA(mesh)
mvisualizeshellB.SetFEMglyphType(chrono.ChVisualShapeFEA.GlyphType_NODE_DOT_POS)
mvisualizeshellB.SetSymbolsThickness(0.006)
mvisualizeshellB.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_NONE)
mesh.AddVisualShapeFEA(mvisualizeshellB)

# ------------------------------------------------------------------------
# Irrlicht visualization system setup
# ------------------------------------------------------------------------
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Shells FEA test: triangle BST elements')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(1, 0.3, 1.3), chrono.ChVector3d(0.5, -0.3, 0.5))
vis.AddTypicalLights()

# ------------------------------------------------------------------------
# Change solver to PardisoMKL (lock sparsity pattern for speed)
# ------------------------------------------------------------------------
mkl_solver = mkl.ChSolverPardisoMKL()
mkl_solver.LockSparsityPattern(True)
sys.SetSolver(mkl_solver)

# ------------------------------------------------------------------------
# Time step for simulation (adjusted)
# ------------------------------------------------------------------------
timestep = 0.005

# ------------------------------------------------------------------------
# Final setup
# ------------------------------------------------------------------------
sys.Setup()
sys.Update()

# ------------------------------------------------------------------------
# Simulation loop
# ------------------------------------------------------------------------
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(timestep)
```
