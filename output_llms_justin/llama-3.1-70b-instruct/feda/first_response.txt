```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import numpy as np

# Create the simulation
mysim = chrono.ChSystemNSC()

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht(mysim)

# Set up the path to the Chrono data directory
chrono.GetLog().SetPathToChronoData("/path/to/data")

# Create the vehicle
feda = veh.FE_Vehicle(mysim)

# Set vehicle location and orientation
feda.SetChassisPosition(chrono.ChVectorD(0, 0.5, 0))
feda.SetChassisOrientation(chrono.Q_from_AngX(chrono.CH_C_PI / 4))

# Set contact method and tire model
feda.SetContactMethod(chrono.ContactMethod_NSC)
feda.SetTireModel(veh.TireModel_RigidRing)

# Create the terrain
terrain = veh.RigidTerrain(mysim)
terrain.SetTexture(chrono.GetChronoDataFile("terrain/textures/tile4.jpg"))
terrain.SetFriction(0.9)

# Create the vehicle visualization
feda.AddVisualization(chrono.ChVisualizationType_MESH)
feda.SetVisualizationType(chrono.ChVisualizationType_MESH)

# Create the driver system
driver = veh.ChIrrlichtDriver(feda, vis)
driver.SetSteering(0)
driver.SetThrottle(0)
driver.SetBraking(0)

# Set camera to follow the vehicle
vis.SetCamera(chrono.ChVectorD(0, 1, -3), chrono.ChVectorD(0, 1, 0), chrono.ChVectorD(0, 1, 0))

# Simulation loop
while vis.Run():
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()
    mysim.DoStepDynamics(0.02)
    driver.Synchronize()
    vis.DrawAll()
    vis.EndScene()
    chrono.ChUtils::Sleep(0.02)
```

This script initializes the PyChrono environment, creates a FEDA vehicle and a rigid terrain, and sets up the visualization using Irrlicht. The vehicle is initialized with specific parameters such as location, orientation, contact method, and tire model. The camera is set to follow the vehicle from a designated position. The simulation loop runs at 50 frames per second, updating the vehicle dynamics and rendering the scene in real time.
