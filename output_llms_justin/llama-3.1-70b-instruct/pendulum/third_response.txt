```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import math

# Initialize the Chrono simulation system
sys = chrono.ChSystemNSC()

# Set the gravitational acceleration for the system (in m/s^2)
sys.SetGravitationalAcceleration(chrono.ChVector3d(0, -9.81, 0))  # g = 9.81 m/s^2

# Create the ground body and add it to the simulation
ground = chrono.ChBodyEasyBox(sys, 4, 0.5, 4, 1000, True, chrono.ChVector3d(0, -2, 0))
sys.Add(ground)

# Create the first pendulum body and add it to the simulation
pend_1 = chrono.ChBodyEasyCylinder(sys, 0.2, 2, 1000, True, chrono.ChVector3d(1, 0, 1))
sys.Add(pend_1)

# Create a revolute joint to connect the first pendulum to the ground
rev_1 = chrono.ChLinkLockRevolute()
rev_1.Initialize(ground, pend_1, chrono.ChFrameD(chrono.ChVector3d(0, 0, 1)))
sys.AddLink(rev_1)

# Create the second pendulum body and add it to the simulation
pend_2 = chrono.ChBodyEasyCylinder(sys, 0.2, 2, 1000, True, chrono.ChVector3d(2, 0, 1))
sys.Add(pend_2)

# Create a revolute joint to connect the second pendulum to the first pendulum
rev_2 = chrono.ChLinkLockRevolute()
rev_2.Initialize(pend_1, pend_2, chrono.ChFrameD(chrono.ChVector3d(0, 0, 1)))
sys.AddLink(rev_2)

# Add visualization to the bodies
vis_ground = chrono.ChVisualShapeBox(chrono.ChVector3d(4, 0.5, 4))
vis_ground.SetColor(chrono.ChColor(0.5, 0.5, 0.5))
ground.AddVisualShape(vis_ground)

vis_pend_1 = chrono.ChVisualShapeCylinder(0.2, 2)
vis_pend_1.SetColor(chrono.ChColor(0.6, 0, 0))
pend_1.AddVisualShape(vis_pend_1)

vis_pend_2 = chrono.ChVisualShapeCylinder(0.2, 2)
vis_pend_2.SetColor(chrono.ChColor(0, 0.6, 0))
pend_2.AddVisualShape(vis_pend_2)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)  # Attach the Chrono system to the visualization
vis.SetWindowSize(1024, 768)  # Set the window size for the visualization
vis.SetWindowTitle('Double Pendulum demo')  # Set the window title
vis.Initialize()  # Initialize the visualization system
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))  # Add a logo to the visualization
vis.AddSkyBox()  # Add a skybox for better visual appearance
vis.AddCamera(chrono.ChVector3d(0, 3, 6))  # Add a camera to the visualization
vis.AddTypicalLights()  # Add typical lights for better visualization

# Simulation loop
log_info = True  # Flag to control logging
while vis.Run():
    vis.BeginScene()  # Begin the scene for rendering
    vis.Render()  # Render the scene
    vis.EndScene()  # End the scene
    sys.DoStepDynamics(1e-3)  # Advance the simulation by a time step of 0.001 seconds

    # Log information after 1 second of simulation time
    if log_info and sys.GetChTime() > 1:
        pos_1 = pend_1.GetPos()  # Get the position of the first pendulum
        print("t = ", sys.GetChTime())
        print("     ", pos_1.x, "  ", pos_1.y)
        lin_vel_1 = pend_1.GetPosDt()  # Get the linear velocity of the first pendulum
        print("     ", lin_vel_1.x, "  ", lin_vel_1.y)

        pos_2 = pend_2.GetPos()  # Get the position of the second pendulum
        print("     ", pos_2.x, "  ", pos_2.y)
        lin_vel_2 = pend_2.GetPosDt()  # Get the linear velocity of the second pendulum
        print("     ", lin_vel_2.x, "  ", lin_vel_2.y)

        log_info = False  # Disable further logging
```

The original script had several issues:

*   The ground body was created without a shape or mass, which would cause issues with the simulation. I replaced it with a `ChBodyEasyBox` to create a box-shaped ground with a specified mass.
*   The pendulum bodies were created without a shape or mass. I replaced them with `ChBodyEasyCylinder` to create cylinder-shaped pendulums with a specified mass.
*   The revolute joints were created without specifying the axis of rotation. I added the axis of rotation to the `Initialize` method of the revolute joints.
*   The visualization shapes were not added to the bodies. I added visualization shapes to the ground and pendulum bodies.
*   The simulation loop was not logging information about the second pendulum. I added logging statements to print the position and linear velocity of the second pendulum.

With these changes, the script should now simulate a double pendulum system with two pendulums attached to each other and the ground, and log information about the position and linear velocity of both pendulums.
