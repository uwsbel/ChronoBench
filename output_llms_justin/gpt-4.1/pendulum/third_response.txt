### **Corrections and Modifications**

#### **Identified Issues in the Original Script**
1. **Incorrect method names and usage:**
   - `SetGravitationalAcceleration` should be `Set_G_acc`.
   - `EnableCollision` should be `SetCollide`.
   - `AddBody` should be `Add`.
   - `ChFramed` should be `ChFrame`.
   - `ChVisualShapeCylinder` expects (radius, height), but the orientation and position must be set via `ChFrame`.
   - `chrono.VNULL` does not exist; should use `chrono.ChVector3d(0,0,0)`.
   - `chrono.QuatFromAngleY` is not a valid function; use `chrono.Q_from_AngY`.
   - `chrono.CH_PI_2` should be `chrono.CH_C_PI_2`.
2. **Visualization shapes are being reused for different bodies.**
3. **Inertia tensor should be realistic for a cylinder.**
4. **The revolute joint should be `ChLinkRevolute` (not `ChLinkLockRevolute` for modern PyChrono).**
5. **The initial position and orientation of the pendulum should be set so that the joint aligns with the end of the previous body.**
6. **For a double pendulum, a second pendulum and a second revolute joint are needed.**

### **Final Corrected and Modified Script**

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import math

# Initialize the Chrono simulation system
sys = chrono.ChSystemNSC()

# Set the gravitational acceleration for the system (in m/s^2)
sys.Set_G_acc(chrono.ChVector3d(0, -9.81, 0))  # g = 9.81 m/s^2

# Create the ground body and add it to the simulation
ground = chrono.ChBody()
ground.SetBodyFixed(True)  # The ground is fixed and does not move
ground.SetCollide(False)   # Disable collision detection for the ground
sys.Add(ground)

# Add a visualization cylinder to the ground (vertical axis at (0,0,1))
cyl_ground = chrono.ChVisualShapeCylinder(0.2, 0.4)  # radius, height
cyl_ground.SetColor(chrono.ChColor(0.4, 0.4, 0.4))
ground.AddVisualShape(cyl_ground, chrono.ChFrame(chrono.ChVector3d(0, 0, 1)))

# Pendulum 1 parameters
length1 = 2.0
radius1 = 0.2
mass1 = 1.0

# Create first pendulum body
pend_1 = chrono.ChBody()
pend_1.SetBodyFixed(False)
pend_1.SetCollide(False)
pend_1.SetMass(mass1)
# Inertia for a solid cylinder about center: Ixx = Izz = (1/12)m(3r^2 + h^2), Iyy = (1/2)m*r^2
Ixx1 = (1/12) * mass1 * (3*radius1**2 + length1**2)
Iyy1 = 0.5 * mass1 * radius1**2
pend_1.SetInertiaXX(chrono.ChVector3d(Ixx1, Iyy1, Ixx1))
sys.Add(pend_1)

# Add visualization to pendulum 1 (horizontal, along X axis)
cyl_pend1 = chrono.ChVisualShapeCylinder(radius1, length1)
cyl_pend1.SetColor(chrono.ChColor(0.6, 0, 0))
# The cylinder is centered at the CoM, so to make it horizontal along X, rotate by 90 deg about Y
pend_1.AddVisualShape(cyl_pend1, chrono.ChFrame(chrono.ChVector3d(0, 0, 0), chrono.Q_from_AngY(chrono.CH_C_PI_2)))

# Set the initial position of pendulum 1 (center of mass)
# The joint will be at (0,0,1), so the CoM is at (length1/2, 0, 1)
pend_1.SetPos(chrono.ChVector3d(length1/2, 0, 1))

# Create a revolute joint to connect pendulum 1 to the ground at (0,0,1), axis Z
rev_1 = chrono.ChLinkRevolute()
frame_rev1 = chrono.ChFrame(chrono.ChVector3d(0, 0, 1), chrono.QUNIT)  # Identity rotation
rev_1.Initialize(ground, pend_1, frame_rev1)
sys.AddLink(rev_1)

# Pendulum 2 parameters
length2 = 1.5
radius2 = 0.15
mass2 = 0.8

# Create second pendulum body
pend_2 = chrono.ChBody()
pend_2.SetBodyFixed(False)
pend_2.SetCollide(False)
pend_2.SetMass(mass2)
Ixx2 = (1/12) * mass2 * (3*radius2**2 + length2**2)
Iyy2 = 0.5 * mass2 * radius2**2
pend_2.SetInertiaXX(chrono.ChVector3d(Ixx2, Iyy2, Ixx2))
sys.Add(pend_2)

# Add visualization to pendulum 2 (horizontal, along X axis)
cyl_pend2 = chrono.ChVisualShapeCylinder(radius2, length2)
cyl_pend2.SetColor(chrono.ChColor(0, 0, 0.6))
pend_2.AddVisualShape(cyl_pend2, chrono.ChFrame(chrono.ChVector3d(0, 0, 0), chrono.Q_from_AngY(chrono.CH_C_PI_2)))

# Set the initial position of pendulum 2 (center of mass)
# The joint will be at the tip of pendulum 1: (length1, 0, 1)
# So CoM is at (length1 + length2/2, 0, 1)
pend_2.SetPos(chrono.ChVector3d(length1 + length2/2, 0, 1))

# Create a revolute joint to connect pendulum 2 to pendulum 1 at (length1, 0, 1), axis Z
rev_2 = chrono.ChLinkRevolute()
frame_rev2 = chrono.ChFrame(chrono.ChVector3d(length1, 0, 1), chrono.QUNIT)
rev_2.Initialize(pend_1, pend_2, frame_rev2)
sys.AddLink(rev_2)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Double Pendulum Demo')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 8))
vis.AddTypicalLights()

# Simulation loop
log_info = True
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(1e-3)

    # Log information after 1 second of simulation time
    if log_info and sys.GetChTime() > 1:
        pos_1 = pend_1.GetPos()
        pos_2 = pend_2.GetPos()
        print("t = ", sys.GetChTime())
        print("Pendulum 1: ", pos_1.x, "  ", pos_1.y, "  ", pos_1.z)
        lin_vel_1 = pend_1.GetPos_dt()
        print("Pendulum 1 vel: ", lin_vel_1.x, "  ", lin_vel_1.y, "  ", lin_vel_1.z)
        print("Pendulum 2: ", pos_2.x, "  ", pos_2.y, "  ", pos_2.z)
        lin_vel_2 = pend_2.GetPos_dt()
        print("Pendulum 2 vel: ", lin_vel_2.x, "  ", lin_vel_2.y, "  ", lin_vel_2.z)
        log_info = False
```
