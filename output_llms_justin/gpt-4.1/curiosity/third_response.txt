```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as robot
import pychrono.sensor as sens  # 1. Added sensor module import
from pychrono import irrlicht as chronoirr

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))  # Corrected: SetGravitationalAcceleration -> Set_G_acc, ChVector3d -> ChVectorD
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create ground body with contact material and add it to the system
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVectorD(0, 0, -0.5))  # Corrected: ChVector3d -> ChVectorD
ground.SetBodyFixed(True)  # Corrected: SetFixed -> SetBodyFixed
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# Create a long box for rover to cross
box = chrono.ChBodyEasyBox(0.25, 5, 0.25, 1000, True, True, ground_mat)
box.SetPos(chrono.ChVectorD(0, 0, 0.0))  # Corrected: ChVector3d -> ChVectorD
box.SetBodyFixed(True)  # Corrected: SetFixed -> SetBodyFixed
box.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
system.Add(box)

# Create Curiosity rover and add it to the system
rover = robot.Curiosity(system)

# Create driver for rover
driver = robot.CuriosityDCMotorControl()
rover.SetDriver(driver)

# Initialize rover position and orientation
init_pos = chrono.ChVectorD(-5, 0.0, 0)  # Corrected: ChVector3d -> ChVectorD
init_rot = chrono.ChQuaternionD(1, 0, 0, 0)  # Corrected: ChQuaterniond -> ChQuaternionD
rover.Initialize(chrono.ChFrameD(init_pos, init_rot))  # Corrected: ChFramed -> ChFrameD

# 2. Create the sensor manager
manager = sens.ChSensorManager(system)

# 3. Add a lidar sensor to the rover
# Lidar parameters
lidar_update_rate = 10.0  # Hz
lidar_horz_samples = 512
lidar_vert_samples = 32
lidar_horz_fov = 2 * math.pi  # 360 deg
lidar_vert_fov = math.radians(30)  # 30 deg
lidar_max_range = 20.0  # meters

# Attach lidar to rover's chassis
chassis = rover.GetChassis()  # Get the main body of the rover

# Lidar relative pose (positioned above the chassis)
lidar_offset = chrono.ChFrameD(chrono.ChVectorD(0.5, 0, 0.5), chrono.QUNIT)  # 0.5m forward, 0.5m up

# Create lidar sensor
lidar = sens.ChLidarSensor(
    chassis,                # parent body
    lidar_update_rate,      # update rate (Hz)
    lidar_offset,           # offset pose
    lidar_horz_samples,     # horizontal samples
    lidar_vert_samples,     # vertical samples
    lidar_horz_fov,         # horizontal FOV (radians)
    lidar_vert_fov,         # vertical FOV (radians)
    lidar_max_range         # max range (meters)
)

# Set lidar name
lidar.SetName("Lidar Sensor")

# Add filters for lidar data
lidar.PushFilter(sens.ChFilterLidarAccess())  # Access filter for retrieving data
lidar.PushFilter(sens.ChFilterSavePtCloud("lidar_points/"))  # Save point cloud to disk

# Add lidar sensor to the manager
manager.AddSensor(lidar)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Curiosity rover - Rigid terrain')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 3, 3), chrono.ChVectorD(0, 0, 0))  # Corrected: ChVector3d -> ChVectorD
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVectorD(1.5, -2.5, 5.5), chrono.ChVectorD(0, 0, 0), 3, 4, 10, 40, 512)  # Corrected: ChVector3d -> ChVectorD

# Enable shadows (commented out to improve performance)
# vis.EnableShadows()

# Set the simulation time step
time_step = 1e-3

# Simulation loop
time = 0
while vis.Run():
    time += time_step

    # ask rover to move forward
    driver.SetSteering(0.0)

    # Update rover dynamics
    rover.Update()

    # 4. Update the sensor manager
    manager.Update()

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance simulation by one time step
    system.DoStepDynamics(time_step)
```
