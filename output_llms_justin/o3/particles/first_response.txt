Python script using PyChrono 8.0+ (and its Irrlicht bindings)  
Save as, e.g., particle_gravity.py and run with a Python build of Chrono that was compiled with Irrlicht support.

```python
# ---------------------------------------------------------------------
# particle_gravity.py
#
# A minimal-yet-complete demonstration of particle–particle gravity in
# PyChrono.  A light “emitter” injects bodies of random shape, size,
# orientation, position, and initial velocity.  At every step we compute
# the mutual Newtonian attraction between all living particles.
# ---------------------------------------------------------------------

import random, math, time
import pychrono as chrono

try:
    import pychrono.irrlicht as irr
except ImportError:
    raise RuntimeError("This script needs the Irrlicht module in your Chrono build.")

# ---------------------------------------------------------------------
# 0.  Global parameters
# ---------------------------------------------------------------------
G_CONST          = 6.674e-2        # Artificially large to see something!
TIME_STEP        = 1.0/250.0
EMISSION_RATE    = 4.0             # [particles / real-second]
MAX_PARTICLES    = 250             # Stop emitting after this many
EMIT_RADIUS      = 0.2             # Size of (spherical) emission region
INIT_VEL_MAG     = 0.4             # Max magnitude of initial velocity
COLORS           = [ irr.SColorf(1,0,0), irr.SColorf(0,1,0),
                     irr.SColorf(0,0,1), irr.SColorf(1,1,0) ]

# ---------------------------------------------------------------------
# 1.  Chrono physical system
# ---------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
sys             = chrono.ChSystemSMC()           # soft-contact system
sys.Set_G_acc(chrono.ChVectorD(0,0,0))          # Disable global gravity

# ---------------------------------------------------------------------
# 2.  Irrlicht application
# ---------------------------------------------------------------------
app = irr.ChIrrApp(sys, "Particle Gravity Demo", irr.dimension2du(1280,720))
app.AddTypicalLogo()
app.AddTypicalSky()
app.AddTypicalLights()
app.AddCamera(irr.vector3df(2,2,2), irr.vector3df(0,0,0))

# Floor only for a visual frame of reference (fixed, no collision)
floor = chrono.ChBodyEasyBox(4, 0.02, 4, 1000, True, True)
floor.SetBodyFixed(True)
sys.Add(floor)

# ---------------------------------------------------------------------
# 3.  Particle emitter helper
# ---------------------------------------------------------------------
particles   = []                   # Keep a python list of living bodies
time_buffer = 0.0                  # Accumulates simulation time for rate

def random_vec_in_sphere(radius):
    while True:
        v = chrono.ChVectorD(random.uniform(-1,1),
                             random.uniform(-1,1),
                             random.uniform(-1,1))
        if v.Length2() <= 1:       # inside unit sphere
            return v * radius

def add_particle():
    shape_selector = random.choice(['sphere','box','cyl'])
    col = random.choice(COLORS)

    if shape_selector == 'sphere':
        r    = random.uniform(0.04,0.08)
        body = chrono.ChBodyEasySphere(r,          # radius
                                       1000,       # density ρ
                                       True, True) # collide, visual
    elif shape_selector == 'box':
        sx,sy,sz  = [random.uniform(0.05,0.12) for _ in range(3)]
        body = chrono.ChBodyEasyBox(sx,sy,sz, 1000, True, True)
    else:                                  # cylinder
        r  = random.uniform(0.04,0.08)
        h  = random.uniform(0.05,0.12)
        body = chrono.ChBodyEasyCylinder(r,h, 1000, True, True)

    # Random pose inside emission volume
    pos = random_vec_in_sphere(EMIT_RADIUS) + chrono.ChVectorD(0,0.5,0)
    rot = chrono.ChQuaternionD()
    rot.Q_from_AngAxis(random.uniform(0,2*math.pi), chrono.ChVectorD(1,0,0))
    body.SetPos(pos)
    body.SetRot(rot)

    # Random initial velocity
    vdir = random_vec_in_sphere(1.0)
    if vdir.Length() > 0:
        vdir = vdir.GetNormalized()*random.uniform(0,INIT_VEL_MAG)
    body.SetPos_dt(vdir)

    # Nice colour
    for asset in body.GetAssets():
        if isinstance(asset, irr.ChColorAsset):
            asset.SetColor(col)

    sys.Add(body)
    particles.append(body)

# ---------------------------------------------------------------------
# 4.  Custom Newtonian gravity routine
# ---------------------------------------------------------------------
def apply_mutual_gravity():
    n = len(particles)
    eps = 1e-9                       # softening to avoid blow-up

    # 4a. wipe forces already present on the bodies
    for p in particles:
        p.Empty_forces_accumulators()

    # 4b. pairwise interactions  (naïve O(N²) for clarity)
    for i in range(n):
        bi = particles[i]
        mi = bi.GetMass()
        pi = bi.GetPos()

        for j in range(i+1, n):
            bj = particles[j]
            mj = bj.GetMass()
            pj = bj.GetPos()

            rij  = pj - pi
            d2   = rij.Length2() + eps
            invR = 1.0/math.sqrt(d2)
            Fmag = G_CONST * mi * mj * invR*invR    # because 1/r²
            force_ij = rij * (Fmag*invR)            # normalize then scale

            # Apply equal and opposite forces
            bi.Accumulate_force(force_ij , pi, False)
            bj.Accumulate_force(-force_ij, pj, False)

# ---------------------------------------------------------------------
# 5.  Main loop
# ---------------------------------------------------------------------
print("Simulation running...  ESC to quit.")
app.SetTimestep(TIME_STEP)

while app.GetDevice().run():
    # 5a. Emit new particles if requested
    if len(particles) < MAX_PARTICLES:
        time_buffer += TIME_STEP
        while time_buffer >= 1.0/EMISSION_RATE and len(particles) < MAX_PARTICLES:
            add_particle()
            time_buffer -= 1.0/EMISSION_RATE

    # 5b. physics & custom forces
    apply_mutual_gravity()
    sys.DoStepDynamics(TIME_STEP)

    # 5c. Render
    app.BeginScene()
    app.DrawAll()
    app.EndScene()

    if app.GetDevice().isWindowActive() == False:
        time.sleep(0.002)           # polite when window not active
```

How it works
1. System uses `ChSystemSMC` and its standard contact model; global gravity is disabled—only the custom Newtonian gravity acts.
2. An inexpensive “emitter” periodically calls `add_particle()`, choosing a random solid (sphere, box, cylinder), random colour, size, pose, and velocity, then inserts it into the system and a local `particles` list.
3. Every time step `apply_mutual_gravity()` clears any existing forces on every particle and loops over all unique pairs, computing  
      F = G m₁ m₂ / r²  
   and adding / subtracting the forces to each body.
4. The usual Chrono integrator, collision detection, etc. run via `DoStepDynamics`.
5. An Irrlicht window shows the evolving particle cloud while you can orbit the camera with mouse buttons. Press ESC or close the window to exit.

Feel free to tune `G_CONST`, particle density, the emission region, or implement smarter N-body techniques (Barnes–Hut, OpenMP, etc.) for large particle counts—this simple O(N²) example is intentionally short and educational.
