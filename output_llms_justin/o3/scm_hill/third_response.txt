Corrected & updated PyChrono script  
(changed from an SCM deformable terrain to a rigid-terrain patch and switched the contact method to NSC)

```python
# =============================================================================
# PyChrono vehicle demo – HMMWV on a rigid height-map terrain
# =============================================================================
# – Contact method:  NSC
# – Terrain:         one RigidTerrain patch generated from a height-map
# – Texture:         dirt.jpg
# – All SCM specific calls (soil parameters, moving patch, false-color plots)
#   have been removed.
# =============================================================================

import pychrono.core as chrono
import pychrono.irrlicht as irr
import pychrono.vehicle as veh
import math
import os

# -----------------------------------------------------------------------------
# Chrono data directories
# -----------------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())                       # change if necessary
veh.SetDataPath   (chrono.GetChronoDataPath() + 'vehicle/')

# -----------------------------------------------------------------------------
# Initial vehicle state
# -----------------------------------------------------------------------------
initLoc = chrono.ChVector3d(-15, 0, 1.2)
initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# -----------------------------------------------------------------------------
# Vehicle visualisation & collision settings
# -----------------------------------------------------------------------------
vis_type               = veh.VisualizationType_MESH
chassis_collision_type = veh.CollisionType_NONE
tire_model             = veh.TireModelType_RIGID

# -----------------------------------------------------------------------------
# Terrain parameters
# -----------------------------------------------------------------------------
terrainLength = 100.0      # X size
terrainWidth  = 100.0      # Y size

# -----------------------------------------------------------------------------
# Contact / solver
# -----------------------------------------------------------------------------
contact_method = chrono.ChContactMethod_NSC        # *** switched to NSC ***

# -----------------------------------------------------------------------------
# Simulation settings
# -----------------------------------------------------------------------------
step_size        = 1e-3
tire_step_size   = step_size
render_step_size = 1.0 / 20.0                      # 20 FPS

# -----------------------------------------------------------------------------
# Create the HMMWV vehicle
# -----------------------------------------------------------------------------
vehicle = veh.HMMWV_Full()
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
vehicle.SetTireType(tire_model)
vehicle.SetTireStepSize(tire_step_size)
vehicle.Initialize()

# Visualisation types ---------------------------------------------------------
vehicle.SetChassisVisualizationType    (vis_type)
vehicle.SetSuspensionVisualizationType (vis_type)
vehicle.SetSteeringVisualizationType   (vis_type)
vehicle.SetWheelVisualizationType      (vis_type)
vehicle.SetTireVisualizationType       (vis_type)

# Use the default Bullet collision system
vehicle.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# -----------------------------------------------------------------------------
# Create a rigid height-map terrain (single patch)
# -----------------------------------------------------------------------------
system      = vehicle.GetSystem()
terrain     = veh.RigidTerrain(system)

# Choose the appropriate surface material for the selected contact method
if contact_method == chrono.ChContactMethod_NSC:
    patch_mat = chrono.ChMaterialSurfaceNSC()
else:
    patch_mat = chrono.ChMaterialSurfaceSMC()

heightmap_file = veh.GetDataFile("terrain/height_maps/bump64.bmp")
hMin, hMax     = -1.0, 1.0                              # lowest & highest point in the height-map

patch = terrain.AddPatch(
        patch_mat,                                      # contact material
        heightmap_file,                                 # height-map image
        hMin, hMax,                                     # height range
        chrono.ChVector3d(0, 0, 0),                     # patch position
        chrono.ChVector3d(0, 0, 1),                     # patch normal
        terrainLength, terrainWidth)                    # size X, Y

patch.SetTexture(veh.GetDataFile("terrain/textures/dirt.jpg"), 6.0, 6.0)

terrain.Initialize()

# -----------------------------------------------------------------------------
# Create Irrlicht visualization
# -----------------------------------------------------------------------------
trackPoint = chrono.ChVector3d(0.0, 0.0, 1.71)          # point tracked by camera

vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle('HMMWV – rigid terrain demo')
vis.SetWindowSize(1280, 1024)
vis.SetChaseCamera(trackPoint, 6.0, 0.5)
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetVehicle())

# -----------------------------------------------------------------------------
# Driver (keyboard)
# -----------------------------------------------------------------------------
driver = veh.ChInteractiveDriverIRR(vis)

steering_time = 1.0
throttle_time = 1.0
braking_time  = 0.3
driver.SetSteeringDelta(render_step_size / steering_time)
driver.SetThrottleDelta(render_step_size / throttle_time)
driver.SetBrakingDelta(render_step_size / braking_time)
driver.Initialize()

# -----------------------------------------------------------------------------
# Simulation loop
# -----------------------------------------------------------------------------
print("VEHICLE MASS:", vehicle.GetVehicle().GetMass())

render_steps   = math.ceil(render_step_size / step_size)
realtime_timer = chrono.ChRealtimeStepTimer()
step_number    = 0
render_frame   = 0

while vis.Run():
    time = system.GetChTime()

    # -----------------
    # Render
    # -----------------
    if step_number % render_steps == 0:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        render_frame += 1

    # -----------------
    # Input and sync
    # -----------------
    driver_inputs = driver.GetInputs()

    driver.Synchronize(time)
    terrain.Synchronize(time)                    # (stub – does nothing for rigid terrain, but harmless)
    vehicle.Synchronize(time, driver_inputs, terrain)
    vis.Synchronize(time, driver_inputs)

    # -----------------
    # Advance dynamics
    # -----------------
    driver.Advance(step_size)
    terrain.Advance(step_size)                   # (stub for rigid terrain)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    # -----------------
    # Book-keeping
    # -----------------
    step_number += 1
    realtime_timer.Spin(step_size)
```

Key changes & fixes made
1. Switched contact method from `SMC` to `NSC`.
2. Replaced the `SCMTerrain` block with a `RigidTerrain` block that:
   • adds a single patch created from the height-map `bump64.bmp`,  
   • applies the texture `dirt.jpg`,  
   • eliminates soil-parameter, moving-patch, and false-color plot calls.
3. Removed all SCM-specific API calls that would otherwise raise run-time errors.
4. Added a proper surface material instance (`ChMaterialSurfaceNSC` or `SMC`).
5. Retained calls to `terrain.Synchronize/Advance` (they are no-ops for rigid terrain but keep the loop generic).
