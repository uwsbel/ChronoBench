```python
import pychrono as chrono
import pychrono.irrlicht as irr
import math

# Create the Chrono simulation system
sys = chrono.ChSystemNSC()
sys.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# Create a fixed truss
truss = chrono.ChBodyEasyBox(4, 0.1, 4, 1000, True, True)
truss.SetPos(chrono.ChVector3d(0, 0, 0))
truss.SetFixed(True)
truss.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
sys.Add(truss)

# Create a rotating bar
bar = chrono.ChBodyEasyBox(0.1, 0.5, 0.1, 1000, True, True)
bar.SetPos(chrono.ChVector3d(0, 0.5, 0))
bar.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
sys.Add(bar)

# Create a revolute joint to connect the bar to the truss
joint = chrono.ChLinkLockRevolute()
joint.Initialize(bar, truss, chrono.ChFramed(chrono.ChVector3d(0, 0, 0)))
sys.Add(joint)

# Create the first gear
gear1 = chrono.ChBodyEasyCylinder(0.1, 0.5, 100, 1000, True, True)
gear1.SetPos(chrono.ChVector3d(0, 0.5, 1))
gear1.SetRot(chrono.ChQuaterniond(1, 0, 0, 0))
gear1.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
sys.Add(gear1)

# Create the second gear
gear2 = chrono.ChBodyEasyCylinder(0.2, 0.5, 100, 1000, True, True)
gear2.SetPos(chrono.ChVector3d(0, 0.5, -1))
gear2.SetRot(chrono.ChQuaterniond(1, 0, 0, 0))
gear2.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
sys.Add(gear2)

# Create a prismatic joint to connect the first gear to the bar
joint1 = chrono.ChLinkLockPrismatic()
joint1.Initialize(gear1, bar, chrono.ChFramed(chrono.ChVector3d(0, 0, 0)))
sys.Add(joint1)

# Create a prismatic joint to connect the second gear to the bar
joint2 = chrono.ChLinkLockPrismatic()
joint2.Initialize(gear2, bar, chrono.ChFramed(chrono.ChVector3d(0, 0, 0)))
sys.Add(joint2)

# Create a gear constraint to connect the first gear to the second gear
constraint = chrono.ChLinkGear()
constraint.Initialize(gear1, gear2, chrono.ChFramed(chrono.ChVector3d(0, 0, 0)))
constraint.SetGearRatio(1)
sys.Add(constraint)

# Create a motor to rotate the first gear
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(gear1, truss, chrono.ChFramed(chrono.ChVector3d(0, 0, 0)))
motor.SetMotorFunction(chrono.ChFunctionConst(10 * chrono.CH_PI / 180))
sys.Add(motor)

# Create the Irrlicht visualization system
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(chrono.ChVector2i(1280, 720))
vis.SetWindowTitle('Epicyclic Gears Simulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 2, -6), chrono.ChVector3d(0, 1, 0))
vis.AddTypicalLights()

# Simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(1e-3)
```
