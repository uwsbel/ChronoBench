```python
import pychrono as chrono
import pychrono.robot as turtlebot
import math

# Create the Chrono system
system = chrono.ChSystemNSC()
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))

# Create the ground body
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(40, 40, 2, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVector3d(0, 0, -1))
ground.SetFixed(True)
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# Initialize the Turtlebot robot
robot = turtlebot.TurtleBot(system)
robot.SetInitPosition(chrono.ChCoordsysd(chrono.ChVector3d(0, 0, 0.5), chrono.ChQuaterniond(1, 0, 0, 0)))
robot.SetChassisFixed(False)
robot.SetContactMethod(chrono.ChContactMethod_NSC)
robot.SetTireType(turtlebot.TireModelType_TMEASY)
robot.SetTreadType(turtlebot.TireTreadType_RUBBER)
robot.SetTrackWidth(0.3)
robot.SetTireStepSize(1e-3)
robot.SetMaxTireForce(1000)
robot.SetInitFwdVel(0)
robot.SetSteeringGearRatio(1)
robot.SetPowertrainType(turtlebot.PowertrainModelType_SIMPLE)
robot.SetPowertrainEfficiency(1)
robot.SetMotorTorqueCurve(turtlebot.MOTOR_TORQUE_CURVE_TYPE_CONSTANT)
robot.Initialize()

robot.SetChassisVisualizationType(chrono.VisualizationType_MESH)
robot.SetSuspensionVisualizationType(chrono.VisualizationType_MESH)
robot.SetSteeringVisualizationType(chrono.VisualizationType_MESH)
robot.SetWheelVisualizationType(chrono.VisualizationType_MESH)
robot.SetTireVisualizationType(chrono.VisualizationType_MESH)

robot.GetChassis().GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/turtlebot_chassis.png"))
robot.GetWheelBody(turtlebot.LEFT_WHEEL).GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/turtlebot_wheel.png"))
robot.GetWheelBody(turtlebot.RIGHT_WHEEL).GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/turtlebot_wheel.png"))

# Create the Irrlicht visualization
vis = chrono.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Turtlebot Robot')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 2.5, 1.5), chrono.ChVector3d(0, 0, 1))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0.5), 3, 4, 10, 40, 512)

# Simulation loop
time = 0
robot_time = 0
while vis.Run():
    time = system.GetChTime()
    robot_time += robot.GetTimeStep()

    if time < 2:
        robot.SetMotorTorque(turtlebot.LEFT_MOTOR, 0)
        robot.SetMotorTorque(turtlebot.RIGHT_MOTOR, 30)
    elif time < 4:
        robot.SetMotorTorque(turtlebot.LEFT_MOTOR, 30)
        robot.SetMotorTorque(turtlebot.RIGHT_MOTOR, 30)
    elif time < 6:
        robot.SetMotorTorque(turtlebot.LEFT_MOTOR, 30)
        robot.SetMotorTorque(turtlebot.RIGHT_MOTOR, 0)
    else:
        robot.SetMotorTorque(turtlebot.LEFT_MOTOR, 0)
        robot.SetMotorTorque(turtlebot.RIGHT_MOTOR, 0)

    robot.Update()

    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    system.DoStepDynamics(robot.GetTimeStep())
```
