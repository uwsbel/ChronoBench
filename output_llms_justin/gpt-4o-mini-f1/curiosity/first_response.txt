```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import math

# Create the Chrono simulation system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))

# Create the ground body with collision properties
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(10, 10, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVector3d(0, 0, -1))
ground.SetFixed(True)
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# Create the rover vehicle
rover = chrono.ChWheeledVehicle()
rover.SetChassisBody(chrono.ChBody())
rover.GetChassisBody().SetPos(chrono.ChVector3d(0, 0, 0))
rover.GetChassisBody().SetFixed(True)

# Create the rover's wheels
for i in range(4):
    wheel = chrono.ChBody()
    wheel.SetPos(chrono.ChVector3d(i * 1.5, 0, 0))
    rover.AddWheel(wheel)

# Create the rover's steering controller
steering_controller = chrono.ChSteeringController()
steering_controller.SetVehicle(rover)

# Create the motor control driver for real-time steering inputs
motor_control = chrono.ChMotorControlDriver()
motor_control.SetVehicle(rover)
motor_control.SetSteering(0.0)
motor_control.SetThrottle(0.0)
motor_control.SetBraking(0.0)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(chrono.ChVector2i(1280, 720))
vis.SetWindowTitle('Rover Simulation with Motor Control')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 0, 2), chrono.ChVector3d(0, 0, 0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 4.5), chrono.ChVector3d(0, 0, 0.5), 3, 4, 10, 40, 512)

# Set the time step for the simulation
time_step = 1e-3

# Run the simulation loop
while vis.Run():
    # Update the steering inputs based on user interactions
    steering_controller.Update()
    motor_control.Update()

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance the simulation by one time step
    system.DoStepDynamics(time_step)
```
