# PyChrono Gator Vehicle Simulation with Sensors and Interactive Driver

```python
import math
import numpy as np
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.sensor as sens
import pychrono.irrlicht as irr

# ---------------------------------------------------------------------
# Initialize the Chrono system and set simulation parameters
# ---------------------------------------------------------------------

chrono.SetChronoDataPath("../data/")
chrono.SetVerbose(False)

# Create the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))
system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)
system.SetSolverMaxIterations(150)
system.SetMaxPenetrationRecoverySpeed(4.0)

# ---------------------------------------------------------------------
# Create the Gator vehicle
# ---------------------------------------------------------------------

# Set vehicle and powertrain JSON file paths
vehicle_file = "vehicle/Gator.json"
powertrain_file = "vehicle/Gator_SimplePowertrain.json"
tire_file = "vehicle/Gator_TMeasyTire.json"

# Initialize vehicle location and orientation
init_loc = chrono.ChVectorD(0, 0, 0.5)
init_rot = chrono.ChQuaternionD(1, 0, 0, 0)

# Create and initialize the Gator vehicle
gator = veh.WheeledVehicle(system, vehicle_file)
gator.Initialize(chrono.ChCoordsysD(init_loc, init_rot))
gator.SetChassisVisualizationType(veh.VisualizationType_MESH)
gator.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
gator.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
gator.SetWheelVisualizationType(veh.VisualizationType_MESH)

# Create and initialize the powertrain
powertrain = veh.SimplePowertrain(powertrain_file)
gator.InitializePowertrain(powertrain)

# Create and attach the tires
for axle in range(gator.GetNumberAxles()):
    for side in range(2):
        tire = veh.TMeasyTire(tire_file)
        gator.InitializeTire(tire, axle, veh.VehicleSide(side))

# ---------------------------------------------------------------------
# Create the terrain
# ---------------------------------------------------------------------

terrain = veh.RigidTerrain(system, "vehicle/terrain/RigidPlane.json")

# ---------------------------------------------------------------------
# Create the interactive driver system
# ---------------------------------------------------------------------

driver = veh.ChIrrGuiDriver(gator.GetVehicle())
driver.SetSteeringDelta(0.05)
driver.SetThrottleDelta(0.05)
driver.SetBrakingDelta(0.1)
driver.Initialize()

# ---------------------------------------------------------------------
# Create the Irrlicht visualization
# ---------------------------------------------------------------------

vis_width = 1280
vis_height = 720
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(vis_width, vis_height)
vis.SetWindowTitle("Gator Vehicle Simulation with Sensors")
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 4, 1.5), chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVectorD(10, 20, 30), chrono.ChVectorD(0, 0, 0), 150, 10, 40, 60, 512)

# ---------------------------------------------------------------------
# Create a sensor manager and add sensors
# ---------------------------------------------------------------------

# Initialize sensor manager with the chrono system
manager = sens.ChSensorManager(system)
manager.scene.AddPointLight(chrono.ChVectorD(0, 0, 100), chrono.ChVectorD(1, 1, 1), 1000)

# Get vehicle's chassis body
chassis_body = gator.GetChassisBody()

# Add a camera attached to the chassis
offset_pose = chrono.ChFrameD(
    chrono.ChVectorD(0, 0, 2),  # position relative to chassis
    chrono.Q_from_AngZ(0)  # orientation (forward-facing)
)
camera = sens.ChCameraSensor(
    chassis_body,          # body camera is attached to
    30.0,                  # update rate (Hz)
    offset_pose,           # offset pose from body
    vis_width,             # image width
    vis_height,            # image height
    chrono.CH_C_PI / 3     # field of view
)
camera.SetName("Camera Sensor")
camera.SetLag(0.1)         # lag time
camera.SetCollectionWindow(0.05)  # collection window

# Set camera properties
camera.PushFilter(sens.ChFilterRGBA8Access())
camera.PushFilter(sens.ChFilterSave("camera/image_"))

# Add the camera to the sensor manager
manager.AddSensor(camera)

# ---------------------------------------------------------------------
# Simulation loop
# ---------------------------------------------------------------------

# Initialize simulation time and step size
step_size = 1e-3
time_end = 20.0

# Setup vehicle output manager
output_step_size = 0.1
gator.GetVehicle().SetChassisOutput(True)
gator.GetVehicle().SetSuspensionOutput(0, True)
gator.GetVehicle().SetSteeringOutput(0, True)

# Initialize real-time simulation
realtime_timer = chrono.ChRealtimeStepTimer()
frame_count = 0

while vis.Run() and system.GetChTime() < time_end:
    # Get driver inputs
    driver_inputs = driver.GetInputs()
    
    # Update modules (driver, terrain, vehicle, sensors)
    driver.Synchronize(system.GetChTime())
    terrain.Synchronize(system.GetChTime())
    gator.Synchronize(system.GetChTime(), driver_inputs, terrain)
    manager.Update()
    
    # Advance simulation
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    
    system.DoStepDynamics(step_size)
    
    # Limit to real-time speed
    realtime_timer.Spin(step_size)
    
    # Print vehicle speed
    if frame_count % 100 == 0:
        speed_kmh = gator.GetVehicle().GetSpeed() * 3.6
        print(f"Time: {system.GetChTime():.1f} s, Speed: {speed_kmh:.1f} km/h")
    
    frame_count += 1

# Clean up
vis.GetDevice().closeDevice()
```

This script creates a complete PyChrono simulation featuring:

1. A Gator vehicle with customized visualization types for different components
2. A rigid terrain for the vehicle to drive on
3. An interactive driver system to control the vehicle
4. Irrlicht visualization for the simulation
5. A sensor manager with:
   - Point lights for illumination
   - A camera sensor attached to the vehicle's chassis
   - Image rendering and saving functionality
6. A simulation loop that:
   - Updates the driver, terrain, vehicle, and sensor manager
   - Synchronizes all components
   - Advances the simulation with proper time steps
   - Maintains real-time performance

The simulation allows interactive control of the vehicle while rendering and saving camera images from the vehicle's perspective.
