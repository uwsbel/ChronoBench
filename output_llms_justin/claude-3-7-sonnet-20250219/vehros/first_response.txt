# PyChrono HMMWV Vehicle Simulation with ROS Integration

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import pychrono.ros as ros
import os
import math
import numpy as np

# -----------------------------------------------------------------------------
# Initialize the ChSystem and set solver parameters
# -----------------------------------------------------------------------------

def main():
    # Create the chrono system
    system = chrono.ChSystemSMC()
    
    # Set gravity
    system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))
    
    # Set solver parameters
    system.SetSolverMaxIterations(150)
    system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)
    system.SetTimestepperType(chrono.ChTimestepper.Type_HHT)
    
    # -----------------------------------------------------------------------------
    # Terrain setup
    # -----------------------------------------------------------------------------
    
    # Create a rigid terrain
    terrain = veh.RigidTerrain(system)
    
    # Set terrain parameters
    terrain_params = veh.RigidTerrainParams()
    terrain_params.SetDefaultMaterial(0.9,  # static friction
                                      0.7,  # kinetic friction
                                      0.3)  # restitution
    terrain_params.SetDefaultColor(chrono.ChColor(0.8, 0.8, 1.0))
    terrain_params.SetDefaultTexture("textures/grass.jpg")
    
    # Set terrain dimensions
    terrain_length = 200.0
    terrain_width = 200.0
    patch_params = veh.RigidTerrainPatchParams(
        chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT),
        chrono.ChVectorD(terrain_length, terrain_width, 2)
    )
    
    # Create the terrain patch
    terrain.AddPatch(patch_params)
    terrain.Initialize()
    
    # -----------------------------------------------------------------------------
    # Vehicle setup
    # -----------------------------------------------------------------------------
    
    # Create the HMMWV vehicle, set parameters, specify contact method
    vehicle_file = veh.GetDataFile("hmmwv/vehicle/HMMWV_Vehicle.json")
    powertrain_file = veh.GetDataFile("hmmwv/powertrain/HMMWV_ShaftsPowertrain.json")
    tire_file = veh.GetDataFile("hmmwv/tire/HMMWV_TMeasyTire.json")
    
    # Create the vehicle and initialize
    vehicle = veh.HMMWV_Full(system)
    vehicle.SetContactMethod(chrono.ChContactMethod_SMC)
    vehicle.SetChassisFixed(False)
    vehicle.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0.5), chrono.QUNIT))
    vehicle.SetPowertrainType(veh.PowertrainModelType_SHAFTS)
    vehicle.SetDriveType(veh.DrivelineTypeWV_AWD)
    vehicle.SetTireType(veh.TireModelType_TMEASY)
    vehicle.SetTireStepSize(1e-3)
    
    # Initialize the vehicle
    vehicle.Initialize()
    
    # Set up the vehicle tires
    for i in range(4):
        vehicle.GetTire(i).SetVisualizationType(veh.VisualizationType_MESH)
    
    # -----------------------------------------------------------------------------
    # Driver system setup
    # -----------------------------------------------------------------------------
    
    # Create the driver system
    driver_data = veh.PathFollowerDriverInputs(vehicle.GetVehicle())
    driver = veh.ChPathFollowerDriver(vehicle.GetVehicle())
    
    # Create a path to follow
    terrain_center = chrono.ChVectorD(0, 0, 0.5)
    target_speed = 10  # m/s
    
    # Define a simple path
    path_points = []
    path_points.append(chrono.ChVectorD(0, 0, 0.5))
    path_points.append(chrono.ChVectorD(50, 0, 0.5))
    path_points.append(chrono.ChVectorD(100, 50, 0.5))
    path_points.append(chrono.ChVectorD(150, 0, 0.5))
    path_points.append(chrono.ChVectorD(200, 0, 0.5))
    
    # Create the path from the points
    path = chrono.ChBezierCurve(path_points)
    
    # Configure the driver
    driver.SetDesiredSpeed(target_speed)
    driver.SetLookAheadDistance(10)
    driver.SetGains(0.8, 0.0, 0.0)  # Proportional, Integral, Derivative gains
    driver.Initialize()
    
    # Set the path to follow
    driver.SetPath(path)
    
    # -----------------------------------------------------------------------------
    # ROS integration
    # -----------------------------------------------------------------------------
    
    # Create a ROS node manager
    ros_manager = ros.ChROSNodeManager()
    
    # Register a handler for the clock
    ros_manager.RegisterHandler(
        ros.ChROSClockHandler(),
        "/clock"
    )
    
    # Register a handler for driver inputs
    ros_manager.RegisterHandler(
        ros.ChROSDriverInputsHandler(driver_data),
        "/vehicle/driver_inputs"
    )
    
    # Register a handler for vehicle state
    ros_manager.RegisterHandler(
        ros.ChROSVehicleStateHandler(vehicle.GetVehicle()),
        "/vehicle/state"
    )
    
    # Initialize ROS node
    ros_manager.InitializeNode("pychrono_hmmwv_node")
    
    # -----------------------------------------------------------------------------
    # Visualization
    # -----------------------------------------------------------------------------
    
    # Create the Irrlicht visualization
    vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
    vis.SetWindowTitle("HMMWV Vehicle Simulation with ROS")
    vis.SetWindowSize(1280, 720)
    vis.Initialize()
    vis.AddTypicalLights()
    vis.AddSkyBox()
    vis.AddCamera(chrono.ChVectorD(0, -10, 2), chrono.ChVectorD(0, 0, 0))
    vis.AttachVehicle(vehicle.GetVehicle())
    
    # -----------------------------------------------------------------------------
    # Simulation loop
    # -----------------------------------------------------------------------------
    
    # Set simulation end time and step size
    end_time = 100  # seconds
    step_size = 1e-3  # seconds
    
    # Initialize simulation time and output counter
    time = 0
    step_number = 0
    output_freq = 100  # output every 100 steps
    
    # Run the simulation loop
    while time < end_time:
        # Synchronize the ROS manager
        ros_manager.Synchronize(time)
        
        # Get driver inputs
        throttle, steering, braking = driver.GetInputs()
        
        # Update the vehicle
        driver.Advance(step_size)
        vehicle.Synchronize(time, driver_data, terrain)
        terrain.Synchronize(time)
        vis.Synchronize(time, driver_data)
        
        # Advance the simulation
        system.DoStepDynamics(step_size)
        
        # Update the visualization
        vis.Advance(step_size)
        
        # Output the vehicle state data periodically
        if step_number % output_freq == 0:
            vehicle_pos = vehicle.GetVehicle().GetChassisBody().GetPos()
            vehicle_vel = vehicle.GetVehicle().GetChassisBody().GetPos_dt()
            vehicle_speed = vehicle_vel.Length()
            print(f"Time: {time:.3f} | Position: ({vehicle_pos.x:.2f}, {vehicle_pos.y:.2f}, {vehicle_pos.z:.2f}) | Speed: {vehicle_speed:.2f} m/s")
        
        # Increment time and step counter
        time += step_size
        step_number += 1
        
        # Process any user input for visualization
        if not vis.Run():
            break
    
    # Finalize ROS node
    ros_manager.FinalizeNode()
    
    print("Simulation completed!")
    return 0

if __name__ == "__main__":
    main()
```

This script creates a complete PyChrono simulation with an HMMWV vehicle, a terrain, a driver system, and ROS integration. Here's a breakdown of the main components:

1. **System Setup**: Initializes the ChSystem with SMC contact method and appropriate solver parameters.

2. **Terrain**: Creates a rigid terrain with defined friction and restitution properties.

3. **Vehicle**: Sets up an HMMWV vehicle with:
   - SMC contact method
   - Shaft-based powertrain
   - All-wheel drive
   - TMeasy tire model

4. **Driver System**: Creates a path follower driver that guides the vehicle along a defined Bezier curve.

5. **ROS Integration**: 
   - Creates a ROS node manager
   - Registers handlers for clock synchronization
   - Handles driver inputs
   - Publishes vehicle state data

6. **Visualization**: Sets up Irrlicht visualization for the simulation.

7. **Simulation Loop**: Advances the simulation step by step, ensuring synchronization between:
   - Vehicle system
   - Terrain system
   - Driver system
   - ROS manager for data publishing

The script ensures proper integration between all components and maintains the proper simulation timestep throughout execution.
