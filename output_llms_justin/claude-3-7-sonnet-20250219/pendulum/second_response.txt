import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import math

# Initialize the Chrono simulation system
sys = chrono.ChSystemNSC()

# Set the gravitational acceleration for the system (in m/s^2)
# Modified: Changed to moon gravity (1.62 m/s^2)
sys.SetGravitationalAcceleration(chrono.ChVector(0, -1.62, 0))

# Create the ground body and add it to the simulation
ground = chrono.ChBody()
sys.Add(ground)
ground.SetBodyFixed(True)  # Corrected: SetFixed -> SetBodyFixed
ground.SetCollide(False)  # Corrected: EnableCollision -> SetCollide

# Add a visualization sphere to the ground for the joint
# Modified: Changed cylinder to sphere with radius 2
sphere_ground = chrono.ChVisualShapeSphere(2)
sphere_ground.SetColor(chrono.ChColor(0.2, 0.2, 0.8))
ground.AddVisualShape(sphere_ground, chrono.ChFrame(chrono.ChVector(0, 0, 1)))  # Corrected: ChFramed -> ChFrame

# Create a pendulum body and add it to the simulation
pend_1 = chrono.ChBody()
sys.Add(pend_1)  # Corrected: AddBody -> Add
pend_1.SetBodyFixed(False)  # Corrected: SetFixed -> SetBodyFixed
pend_1.SetCollide(False)  # Corrected: EnableCollision -> SetCollide
# Modified: Changed mass to 2 kg
pend_1.SetMass(2)
# Modified: Changed inertia tensor
pend_1.SetInertiaXX(chrono.ChVector(0.4, 1.5, 1.5))

# Add a visualization cylinder to the pendulum
# Modified: Changed dimensions to radius 0.1 and height 1.5
cyl_1 = chrono.ChVisualShapeCylinder(0.1, 1.5)
cyl_1.SetColor(chrono.ChColor(0.6, 0, 0))
pend_1.AddVisualShape(cyl_1, chrono.ChFrame(chrono.VNULL, chrono.Q_from_AngY(chrono.CH_C_PI_2)))  # Corrected: QuatFromAngleY -> Q_from_AngY, ChFramed -> ChFrame

# Set the initial position of the pendulum
pend_1.SetPos(chrono.ChVector(1, 0, 1))

# Modified: Set an initial angular velocity for the pendulum (around z-axis)
pend_1.SetWvel_par(chrono.ChVector(0, 0, 2.0))  # 2 rad/s

# Modified: Create a spherical joint instead of revolute joint
sph_1 = chrono.ChLinkSpherical()
sph_1.Initialize(ground, pend_1, chrono.ChFrame(chrono.ChVector(0, 0, 1)))
sys.Add(sph_1)  # Corrected: AddLink -> Add

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Modified Pendulum Demo')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector(0, 3, 6))
vis.AddTypicalLights()

# Simulation loop
log_info = True
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(1e-3)

    # Log information after 1 second of simulation time
    if log_info and sys.GetChTime() > 1:
        pos_1 = pend_1.GetPos()
        print("t = ", sys.GetChTime())
        print("     ", pos_1.x, "  ", pos_1.y, "  ", pos_1.z)
        lin_vel_1 = pend_1.GetPos_dt()  # Corrected: GetPosDt -> GetPos_dt
        print("     ", lin_vel_1.x, "  ", lin_vel_1.y, "  ", lin_vel_1.z)
        ang_vel_1 = pend_1.GetWvel_loc()
        print("Angular velocity: ", ang_vel_1.x, "  ", ang_vel_1.y, "  ", ang_vel_1.z)
        log_info = False
