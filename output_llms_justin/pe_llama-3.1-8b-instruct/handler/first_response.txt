```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as robot
import pychrono.irrlicht as irrlicht
import pychrono.vehicle as vehicle
import rospy
import tf
import message_filters

# Library imports
os = os
math = math
np = numpy
chrono = pychrono
robot = pychrono.robot
irrlicht = pychrono.irrlicht
vehicle = pychrono.vehicle
rospy = rospy
tf = tf
message_filters = message_filters

# Contact and collision settings
my_system = chrono.ChSystemNSC()
chrono.SetChronoDataPath('/path/to/data')
material_nsc = chrono.ChContactMaterialNSC()
material_nsc.SetFriction(0.5)
material_nsc.SetDampingF(0.1)
material_nsc.SetCompliance(0.01)
my_system.SetMaterialDefaultContact(material_nsc)

# Visualization settings
vis = irrlicht.ChVisualSystemIrrlicht()
vis.AttachSystem(my_system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Simulation Window')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 6))
vis.AddTypicalLights()

# Body initialization
body_floor = chrono.ChBody()
body_floor.SetPos(chrono.ChVector3d(0, 0, 0))
body_floor_shape = chrono.ChVisualShapeBox(10, 10, 0.1)
body_floor.AddVisualShape(body_floor_shape)
my_system.Add(body_floor)

body_box = chrono.ChBody()
body_box.SetPos(chrono.ChVector3d(0, 0, 1))
body_box_shape = chrono.ChVisualShapeBox(1, 1, 1)
body_box.AddVisualShape(body_box_shape)
my_system.Add(body_box)

# Joints and links
link_shaker = chrono.ChLinkLockLock()
link_shaker.Initialize(body_box, body_floor, chrono.ChFramed())
my_system.Add(link_shaker)

# Simulation loop
def simulation_step():
    my_system.DoStepDynamics(0.01)
    vis.Render()

# ROS handler
class CustomROSHandler:
    def __init__(self):
        self.pub = rospy.Publisher('/my_topic', Int32, queue_size=10)

    def on_message(self, msg):
        self.pub.publish(msg.data)

# ROS manager
rospy.init_node('pychrono_ros_node')
clock_handler = rospy.TimeHandler()
body_handler = rospy.BodyHandler()
transform_handler = rospy.TransformHandler()
custom_handler = CustomROSHandler()
ros_manager = rospy.Manager(clock_handler, body_handler, transform_handler, custom_handler)

# Register custom handler
ros_manager.RegisterHandler(custom_handler)

# Run simulation loop
while vis.Run():
    simulation_step()
    ros_manager.Update()
    ros_manager.ClockStep(0.01)
```

This code initializes the PyChrono environment, sets up the required physical systems and objects, and configures the ROS manager with the custom handler. The simulation loop advances the system and updates ROS communication at each timestep, maintaining real-time execution.
