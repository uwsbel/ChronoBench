```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.robot as robosimian
import numpy as np

# Initialize the PyChrono environment
chrono.SetChronoDataPath('/path/to/data')

# Create a new simulation system using the Signorini-Moreau Contact (SMC) model
sys = chrono.ChSystemSMC()

# Set up the beam's properties
beam_L = 5.0  # Beam length
beam_wy = 0.1  # Beam width in the y-direction
beam_wz = 0.1  # Beam width in the z-direction
beam_rho = 1000.0  # Beam density
beam_E = 2.0e9  # Beam Young's modulus
beam_G = 0.8e9  # Beam shear modulus
beam_R = 0.01  # Beam radius

# Create the beam mesh
mesh = veh.ChMesh()
msection = veh.ChBeamSectionEulerAdvanced()
msection.SetAsRectangularSection(beam_wy, beam_wz)
msection.SetYoungModulus(beam_E)
msection.SetShearModulus(beam_G * 0.3)
msection.SetRayleighDamping(0.000)
hnode1 = veh.ChNodeFEAxyzrot(chrono.ChFramed(chrono.ChVector3d(0, 0, 0)))
hnode2 = veh.ChNodeFEAxyzrot(chrono.ChFramed(chrono.ChVector3d(beam_L, 0, 0)))
mesh.AddNode(hnode1)
mesh.AddNode(hnode2)
belement1 = veh.ChElementBeamEuler()
belement1.SetNodes(hnode1, hnode2)
belement1.SetSection(msection)
mesh.AddElement(belement1)

# Add the beam to the simulation system
sys.Add(mesh)

# Create a hinge joint to fix the beam's end
joint = chrono.ChLinkLockLock()
joint.Initialize(mesh, chrono.ChFramed())
sys.Add(joint)

# Set up gravity
sys.AddForce(chrono.ChForceGravity())

# Create a visualization system
vis = chrono.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Beam Simulation')
vis.Initialize()

# Set up the simulation loop
while vis.Run():
    sys.DoStepDynamics(0.01)
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
```

This script initializes the PyChrono environment, sets up the beam's properties, creates the beam mesh, adds the beam to the simulation system, fixes the beam's end with a hinge joint, sets up gravity, and creates a visualization system. The simulation loop then runs, advancing the simulation by small time steps and rendering the beam's deformation and nodal positions using the Irrlicht engine.
