Here's a Python script that implements the simulation as described:

```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.robot as chronorobot
import pychrono.vehicle as chronovehicle

# Initialize PyChrono
chrono.SetChronoDataPath('/path/to/data')

# Create the system
sys = chrono.ChSystemNSC()

# Define the gravitational constant
g = 9.81

# Define the particle emitter
class ParticleEmitter:
    def __init__(self, system, emitter_radius, emitter_velocity, particle_radius, particle_mass):
        self.system = system
        self.emitter_radius = emitter_radius
        self.emitter_velocity = emitter_velocity
        self.particle_radius = particle_radius
        self.particle_mass = particle_mass
        self.particles = []

    def emit_particles(self, num_particles):
        for _ in range(num_particles):
            particle = chrono.ChBody()
            particle.SetPos(chrono.ChVector3d(np.random.uniform(-self.emitter_radius, self.emitter_radius), np.random.uniform(-self.emitter_radius, self.emitter_radius), np.random.uniform(-self.emitter_radius, self.emitter_radius)))
            particle.SetMass(self.particle_mass)
            particle_shape = chrono.ChVisualShapeSphere(self.particle_radius)
            particle.AddVisualShape(particle_shape)
            self.system.Add(particle)
            self.particles.append(particle)

    def apply_gravity(self):
        for particle in self.particles:
            particle.AddForce(chrono.ChVector3d(0, 0, -g * self.particle_mass))

# Create the particle emitter
emitter = ParticleEmitter(sys, 10, 1, 0.5, 1)
emitter.emit_particles(100)

# Create the visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Particle Simulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()

# Create the Irrlicht camera
camera = vis.AddCamera(chrono.ChVector3d(0, 3, 6))

# Main simulation loop
while vis.Run():
    sys.DoStepDynamics(0.01)
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
```

This script initializes the PyChrono environment, creates a particle emitter, and simulates the gravitational attraction between particles. It also sets up a visualization using Irrlicht and applies custom gravitational forces to simulate particle attraction dynamically.
