Here's a Python script that follows the instructions and sets up the simulation of a simple pendulum:

```python
# Import necessary libraries
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Initialize the PyChrono environment
chrono.SetChronoDataPath('/path/to/data')

# Create a system using Non-Smooth Contact (NSC) mechanics
sys = chrono.ChSystemNSC()

# Create a ground body
ground = chrono.ChBody()
ground.SetPos(chrono.ChVector3d(0, 0, 0))
ground.SetMass(1000)
ground.SetInertiaXX(chrono.ChVector3d(1000, 1000, 1000))
ground.SetFixed(True)
sys.Add(ground)

# Create a pendulum body
pendulum = chrono.ChBody()
pendulum.SetPos(chrono.ChVector3d(0, 1, 0))
pendulum.SetMass(1)
pendulum.SetInertiaXX(chrono.ChVector3d(0.1, 0.1, 0.1))
pendulum.SetFixed(False)
sys.Add(pendulum)

# Create a revolute joint to connect the pendulum to the ground
joint = chrono.ChLinkLockRevolute()
joint.Initialize(ground, pendulum, chrono.ChFramed())
sys.Add(joint)

# Create a visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Pendulum Simulation')
vis.Initialize()
vis.AttachSystem(sys)

# Add a camera to the visualization system
vis.AddCamera(chrono.ChVector3d(0, 3, 6))

# Set up the visualization
vis.AddSkyBox()
vis.AddTypicalLights()

# Create a visual shape for the pendulum
pendulum_shape = chrono.ChVisualShapeBox(0.1, 0.1, 0.1)
pendulum_shape.SetColor(chrono.ChColor(1, 0, 0))
pendulum.AddVisualShape(pendulum_shape)

# Set up the simulation loop
while vis.Run():
    # Advance the simulation by a time step
    sys.DoStepDynamics(0.01)
    
    # Log the pendulum's position and velocity
    pos = pendulum.GetPos()
    vel = pendulum.GetVelocity()
    print(f'Position: {pos}, Velocity: {vel}')
    
    # Render the simulation
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
```

This script initializes the PyChrono environment, sets up the simulation with a ground body and a pendulum, and connects them with a revolute joint. It also creates a visualization system using Irrlicht and sets up the simulation loop to advance the simulation by a time step and log the pendulum's position and velocity. The visualization is rendered periodically, showing the pendulum's motion.
