```python
# Import necessary PyChrono and PyIGA packages
from pychrono.core import *
from pychrono.vehicle_beam import *
from pychrono.vehicle_flywheel import *
from pychrono.vehicle_motor import *
from pychrono.vehicle_ig_beam import *
from pychrono.vehicle_irrlicht_viewer import *
from pychrono.vehicle_ig_beam import *
from pychrono.vehicle_motor_rotary import *

# Initialize the PyChrono environment
chrono_init()

# Create the IGA beam
iga_beam = IGA_Beam()
iga_beam.set_length(1.0)  # Set the length of the beam to 1 meter
iga_beam.set_mass(10.0)  # Set the mass of the beam to 10 kg
iga_beam.set_radius(0.05)  # Set the radius of the beam to 0.05 meters
iga_beam.set_elasticity_modulus(210e9)  # Set the elasticity modulus to 210 GPa
iga_beam.set_poisson_ratio(0.3)  # Set the Poisson's ratio to 0.3
iga_beam.set_density(7800)  # Set the density to 7800 kg/m^3
iga_beam.set_damping(0.01)  # Set the damping coefficient to 0.01
iga_beam.set_axis_orientation(CH_AXIS_X)  # Set the beam's axis of rotation to the X-axis

# Create the flywheel
flywheel = IGA_Flywheel()
flywheel.set_radius(0.1)  # Set the flywheel radius to 0.1 meters
flywheel.set_mass(5.0)  # Set the flywheel mass to 5 kg
flywheel.set_moment_of_inertia(0.0005)  # Set the moment of inertia to 0.0005 kg*m^2
flywheel.set_angular_velocity(0.0)  # Initial angular velocity of the flywheel
flywheel.set_angular_acceleration(0.0)  # Initial angular acceleration of the flywheel

# Create the rotary motor
rotary_motor = RotaryMotor()
rotary_motor.set_torque(10.0)  # Set the motor torque to 10 Nm
rotary_motor.set_position(0.5)  # Position the motor at the center of the beam (0.5 meters from one end)

# Attach the flywheel to the beam
beam_with_flywheel = BeamWithFlywheel()
beam_with_flywheel.set_beam(iga_beam)
beam_with_flywheel.add_flywheel(flywheel)
beam_with_flywheel.set_attachment_point(iga_beam.get_attachment_point(0.5))  # Attach the flywheel at the center of the beam

# Set up the simulation parameters
sim_params = SimulationParameters()
sim_params.set_max_time(10.0)  # Set the maximum simulation time to 10 seconds
sim_params.set_gravity(0.0, -9.81, 0.0)  # Set gravity to Earth's gravity
sim_params.set_time_step(0.01)  # Set the time step to 0.01 seconds

# Add the motor to the simulation
sim_params.add_rigid_body(rotary_motor)

# Add the beam with flywheel to the simulation
sim_params.add_rigid_body(beam_with_flywheel)

# Create the visualization using Irrlicht
viewer = IrrlichtViewer()
viewer.add_irrlicht_scene(iga_beam)
viewer.add_irrlicht_scene(beam_with_flywheel)

# Run the simulation
sim = Simulation(sim_params)
sim.add_rigid_body(rotary_motor)
sim.add_rigid_body(beam_with_flywheel)
sim.integrate(10000)  # Integrate for 10000 time steps

# Start the visualization
viewer.start("Jeffcott_rotor_simulation.avi")
```
