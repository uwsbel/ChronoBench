### Solution 1:

Certainly! Below is a simple C++ program that uses the `<random>` header to generate a random number between 1 and 1000 and prints it out.

```cpp
#include <iostream>
#include <random>

int main() {
    // Create a random device
    std::random_device rd;

    // Create a Mersenne Twister engine
    std::mt19937 rng(rd());

    // Create a uniform distribution between 1 and 1000
    std::uniform_int_distribution<int> dist(1, 1000);

    // Generate a random number
    int randomNumber = dist(rng);

    // Print the random number
    std::cout << "Random number between 1 and 1000: " << randomNumber << std::endl;

    return 0;
}
```
