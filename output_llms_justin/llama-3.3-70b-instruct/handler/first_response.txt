```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import rospy
from std_msgs.msg import Int32

# Initialize the PyChrono environment and core components
sys = chrono.ChSystemNSC()

# Set the gravitational acceleration
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create a material with default properties
material = chrono.ChMaterialSurfaceNSC()
material.SetFriction(0.4)
material.SetDampingF(0.2)
material.SetCompliance(0.0001)
material.SetComplianceT(0.0001)

# Create a fixed floor
floor = chrono.ChBodyEasyBox(sys, 10, 1, 10, 1000, True, material)
floor.SetPos(chrono.ChVectorD(0, -5, 0))
floor.SetBodyFixed(True)
sys.Add(floor)

# Create a movable box
box = chrono.ChBodyEasyBox(sys, 1, 1, 1, 10, True, material)
box.SetPos(chrono.ChVectorD(0, 5, 0))
sys.Add(box)

# Create a custom ROS handler to publish integer messages
class CustomROSHändler:
    def __init__(self):
        self.pub = rospy.Publisher('custom_topic', Int32, queue_size=10)

    def Update(self):
        msg = Int32()
        msg.data = 1  # Example data to publish
        self.pub.publish(msg)

# Initialize the ROS node
rospy.init_node('pychrono_simulation')

# Create a ROS manager
ros_manager = chronoirr.ChVisualSystemIrrlicht()

# Configure the ROS manager with clock, body, and transform handlers
ros_manager.AddTypicalClock()
ros_manager.AddTypicalBody(sys)
ros_manager.AddTypicalTransform(sys)

# Register the custom ROS handler
custom_handler = CustomROSHändler()
ros_manager.RegisterTypicalHandler(custom_handler)

# Run the simulation loop
while not rospy.is_shutdown():
    # Advance the system
    sys.DoStepDynamics(0.01)

    # Update ROS communication
    ros_manager.BeginScene()
    ros_manager.DrawAll()
    ros_manager.EndScene()

    # Maintain real-time execution
    rospy.sleep(0.01)
```

This script sets up a PyChrono simulation integrated with ROS, including a fixed floor and a movable box, and publishes integer messages to a specified topic using a custom ROS handler. The simulation loop advances the system and updates ROS communication at each timestep while maintaining real-time execution.
