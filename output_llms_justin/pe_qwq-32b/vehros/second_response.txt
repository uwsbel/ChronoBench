```python
import pychrono as ch
import pychrono.vehicle as veh
import pychrono.ros as chros
from pychrono import irrlicht as chronoirr  # Added for visualization

def main():
    # Set Chrono data path for vehicle assets
    veh.SetDataPath(ch.GetChronoDataPath() + 'vehicle/')  # Instruction 2

    # Create the HMMWV vehicle and set its parameters.
    hmmwv = veh.HMMWV_Full()
    hmmwv.SetContactMethod(ch.ChContactMethod_NSC)
    hmmwv.SetChassisCollisionType(veh.CollisionType_NONE)
    hmmwv.SetChassisFixed(False)
    hmmwv.SetInitPosition(ch.ChCoordsysd(ch.ChVectorD(0, 0, 1.6), ch.ChQuaternionD(1, 0, 0, 0)))
    hmmwv.SetEngineType(veh.EngineModelType_SHAFTS)
    hmmwv.SetTransmissionType(veh.TransmissionModelType_AUTOMATIC_SHAFTS)
    hmmwv.SetDriveType(veh.DrivelineTypeWV_AWD)
    hmmwv.SetSteeringType(veh.SteeringTypeWV_PITMAN_ARM)
    hmmwv.SetTireType(veh.TireModelType_TMEASY)
    hmmwv.SetTireStepSize(1e-3)

    # Enhanced visualization settings (Instruction 3)
    hmmwv.SetChassisVisualizationType(veh.VisualizationType_MESH)
    hmmwv.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
    hmmwv.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
    hmmwv.SetWheelVisualizationType(veh.VisualizationType_MESH)
    hmmwv.SetTireVisualizationType(veh.VisualizationType_NONE)
    hmmwv.Initialize()  # Initialize after setting visualization

    # Create terrain with texture (Instruction 4)
    terrain = veh.RigidTerrain(hmmwv.GetSystem())
    patch_mat = ch.ChContactMaterialNSC()
    patch_mat.SetFriction(0.9)
    patch_mat.SetRestitution(0.01)
    patch = terrain.AddPatch(patch_mat, ch.ChFrameD(), 100.0, 100.0)
    patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 100, 100)
    terrain.Initialize()

    # Driver setup
    driver = veh.ChDriver(hmmwv.GetVehicle())
    driver.Initialize()

    # ROS Manager setup
    ros_manager = chros.ChROSPythonManager()
    ros_manager.RegisterHandler(chros.ChROSClockHandler())
    ros_manager.RegisterHandler(chros.ChROSDriverInputsHandler(25, driver, "~/input/driver_inputs"))
    ros_manager.RegisterHandler(chros.ChROSBodyHandler(25, hmmwv.GetChassisBody(), "~/output/hmmwv/state"))
    ros_manager.Initialize()

    # Visualization system setup (Instructions 5 and 6)
    vis = chronoirr.ChVisualSystemIrrlicht()
    vis.AttachSystem(hmmwv.GetSystem())
    vis.SetWindowSize(1280, 1024)
    vis.SetWindowTitle('HMMWV Simulation')
    vis.Initialize()
    vis.AddLogo(chronoirr.GetDataFile('logo_pychrono_alpha.png'))
    vis.AddSkyBox()
    vis.SetCamera(chronoirr.ChVectorD(0, 3, 15))
    vis.AddTypicalLights()
    vis.SetCameraSpeed(2.0)
    vis.SetShadowFBO(True)

    # Simulation parameters
    time_step = 1e-3
    time_end = 30
    hmmwv.GetSystem().SetTolerance(1e-6)  # Added for numerical stability

    # Main simulation loop
    while vis.Run() and time < time_end:
        time = hmmwv.GetSystem().GetChTime()

        # Update driver inputs and synchronize systems
        driver_inputs = driver.GetInputs()
        driver.Synchronize(time)
        terrain.Synchronize(time)
        hmmwv.Synchronize(time, driver_inputs, terrain)

        # Advance simulation step
        driver.Advance(time_step)
        terrain.Advance(time_step)
        hmmwv.Advance(time_step)

        # Visualization rendering
        vis.BeginScene()
        vis.Render()
        vis.EndScene()

        # ROS communication update
        if not ros_manager.Update(time, time_step):
            break

    # Cleanup
    vis.Close()
    ros_manager.Shutdown()

if __name__ == "__main__":
    main()
```
