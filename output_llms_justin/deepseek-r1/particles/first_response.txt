
To model gravitational attraction between particles with a custom emitter in PyChrono, follow this solution. The script initializes the environment, creates particles with random shapes (spheres or boxes), applies pairwise gravitational forces, and visualizes the simulation using Irrlicht.

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import random
import math

# Constants
G = 0.1  # Gravitational constant (tuned for visible effects)
MAX_PARTICLES = 50
EMIT_INTERVAL = 0.2  # Seconds between particle emissions
EMIT_BOX_SIZE = chrono.ChVectorD(5, 1, 5)  # Emission area dimensions

# Custom action to compute gravitational forces
class GravitationalForce(chrono.ChAction):
    def __init__(self, particle_list):
        super().__init__()
        self.particle_list = particle_list
        self.G = G

    def ActionPerformed(self):
        n = len(self.particle_list)
        for i in range(n):
            body_i = self.particle_list[i]
            pos_i = body_i.GetPos()
            mass_i = body_i.GetMass()
            for j in range(i + 1, n):
                body_j = self.particle_list[j]
                pos_j = body_j.GetPos()
                mass_j = body_j.GetMass()
                
                # Calculate distance vector
                r_vect = pos_j - pos_i
                r_len = r_vect.Length()
                
                if r_len > 0.01:  # Avoid division by zero
                    # Calculate gravitational force (F = G * m1 * m2 / r^2)
                    force_mag = self.G * mass_i * mass_j / (r_len * r_len)
                    force_dir = r_vect / r_len
                    force = force_mag * force_dir
                    
                    # Apply equal and opposite forces
                    body_i.Accumulate_force(force, pos_i, False)
                    body_j.Accumulate_force(-force, pos_j, False)

# Create physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))  # Disable default gravity
particles = []  # Track particles for gravity calculation

# Add gravitational force action
grav_force = GravitationalForce(particles)
system.Add(grav_force)

# Create Irrlicht visualization
application = chronoirr.ChIrrApp(system, "Particle Gravity", chronoirr.dimension2du(800, 600))
application.AddLogo()
application.AddSkyBox()
application.AddCamera(chrono.ChVectorD(0, 14, -20), chrono.ChVectorD(0, 0, 0))
application.AddLight(chrono.ChVectorD(30, 30, -30), 100)

# Function to create random particles
def create_particle():
    # Random position in emission box
    pos = chrono.ChVectorD(
        random.uniform(-EMIT_BOX_SIZE.x/2, EMIT_BOX_SIZE.x/2),
        random.uniform(0, EMIT_BOX_SIZE.y),
        random.uniform(-EMIT_BOX_SIZE.z/2, EMIT_BOX_SIZE.z/2)
    )
    
    # Random linear velocity
    vel = chrono.ChVectorD(
        random.uniform(-0.5, 0.5),
        random.uniform(-0.5, 0.5),
        random.uniform(-0.5, 0.5)
    )
    
    # Random orientation
    quat = chrono.ChQuaternionD()
    quat.Q_from_AngAxis(random.uniform(0, 2 * math.pi), chrono.ChVectorD(0, 1, 0))
    
    # Random shape (sphere or box)
    if random.choice([True, False]):
        # Create sphere
        radius = random.uniform(0.3, 0.6)
        density = random.uniform(500, 1500)
        body = chrono.ChBodyEasySphere(radius, density)
    else:
        # Create box
        size = chrono.ChVectorD(
            random.uniform(0.3, 0.6),
            random.uniform(0.3, 0.6),
            random.uniform(0.3, 0.6)
        )
        density = random.uniform(500, 1500)
        body = chrono.ChBodyEasyBox(size.x, size.y, size.z, density)
    
    # Set particle properties
    body.SetPos(pos)
    body.SetPos_dt(vel)
    body.SetRot(quat)
    body.SetCollide(True)
    
    # Visual properties
    body.GetVisualShape(0).SetColor(chrono.ChColor(
        random.uniform(0.3, 1.0),
        random.uniform(0.3, 1.0),
        random.uniform(0.3, 1.0)
    ))
    
    system.Add(body)
    particles.append(body)

# Simulation loop setup
application.AssetBindAll()
application.AssetUpdateAll()
time_step = 0.002
next_emit_time = 0

# Main simulation loop
while application.GetDevice().run():
    current_time = system.GetChTime()
    
    # Emit new particles
    if current_time >= next_emit_time and len(particles) < MAX_PARTICLES:
        create_particle()
        next_emit_time = current_time + EMIT_INTERVAL
        application.AssetBindAll()
        application.AssetUpdateAll()
    
    # Simulation step
    application.BeginScene(True, True, chronoirr.SColor(255, 140, 161, 192))
    application.DrawAll()
    application.DoStepDynamics(time_step)
    application.EndScene()

```

### Key Features:
1. **Particle Emitter**: Creates particles at random positions within a defined box with:
   - Random shapes (spheres or boxes)
   - Random sizes, densities, and colors
   - Random initial velocities and orientations
   - Controlled emission rate (`EMIT_INTERVAL`)

2. **Custom Gravity**:
   - Disables default gravity (`Set_G_acc(0,0,0)`)
   - Implements n-body gravitational attraction via `GravitationalForce` action
   - Computes pairwise forces using Newton's law of gravitation

3. **Visualization**:
   - 3D view with camera positioned to observe particle interactions
   - Color-coded particles for visual distinction
   - Skybox and lighting for enhanced visualization

4. **Parameters**:
   - Adjust `G` to control attraction strength
   - Modify `EMIT_INTERVAL` and `MAX_PARTICLES` for emission frequency/count
   - Tune `EMIT_BOX_SIZE` to change emission area

Run the script to see particles emit from a box, interact through gravitational attraction, and form dynamic clusters!
