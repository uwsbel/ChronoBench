```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import math

def main():
    # Set the Chrono data path
    chrono.SetChronoDataPath("../../../data/")
    veh.SetDataPath("../../../data/vehicle/")

    # Initialize the ChronoENGINE physical system
    system = chrono.ChSystemNSC()
    system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))
    
    # Set solver settings
    system.SetSolverMaxIterations(150)
    system.SetSolverForceTolerance(1e-10)

    # Create the terrain
    terrain = veh.RigidTerrain(system)
    
    # Define terrain patch parameters
    patch_mat = chrono.ChMaterialSurfaceNSC()
    patch_mat.SetFriction(0.8)
    patch_mat.SetRestitution(0.01)
    patch_mat.SetYoungModulus(2e7)
    patch_mat.SetPoissonRatio(0.3)
    
    # Create terrain patch
    patch = terrain.AddPatch(patch_mat, 
                           chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT),
                           200.0, 200.0)  # 200m x 200m terrain
    patch.SetTexture(veh.GetDataFile("terrain/textures/grass.jpg"), 100, 100)
    patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
    terrain.Initialize()

    # Create the Kraz vehicle
    kraz = veh.Kraz()
    
    # Set initial vehicle position and orientation
    init_pos = chrono.ChVectorD(0, 0, 1.0)  # 1m above ground
    init_rot = chrono.ChQuaternionD(1, 0, 0, 0)  # No rotation
    
    # Initialize the vehicle
    kraz.SetContactMethod(chrono.ChContactMethod_NSC)
    kraz.SetChassisCollisionType(veh.CollisionType_NONE)
    kraz.SetChassisFixed(False)
    kraz.SetInitPosition(chrono.ChCoordsysD(init_pos, init_rot))
    kraz.SetTireType(veh.TireModelType_TMEASY)
    kraz.SetTireStepSize(1e-3)
    kraz.Initialize()

    # Enable visualization of vehicle components
    kraz.SetChassisVisualizationType(veh.VisualizationType_MESH)
    kraz.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
    kraz.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
    kraz.SetWheelVisualizationType(veh.VisualizationType_MESH)
    kraz.SetTireVisualizationType(veh.VisualizationType_MESH)

    # Create the driver system
    driver = veh.ChInteractiveDriverIRR()
    
    # Set the time response for steering and throttle inputs
    steering_time = 1.0  # time to go from 0 to max steering
    throttle_time = 1.0  # time to go from 0 to max throttle
    braking_time = 0.3   # time to go from 0 to max braking
    
    driver.SetSteeringDelta(0.02)
    driver.SetThrottleDelta(0.02)
    driver.SetBrakingDelta(0.02)
    
    # Initialize the driver
    driver.Initialize()

    # Create the Irrlicht visualization application
    vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
    vis.SetWindowTitle("Kraz Vehicle Simulation")
    vis.SetWindowSize(1280, 720)
    vis.SetChaseCamera(chrono.ChVectorD(0.0, 0.0, 1.75), 8.0, 0.5)
    vis.Initialize()
    vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
    vis.AddSkyBox()
    vis.AddCamera(chrono.ChVectorD(-8, 5, 3), chrono.ChVectorD(0, 0, 0))
    vis.AddTypicalLights()
    vis.AttachVehicle(kraz.GetVehicle())

    # Simulation parameters
    step_size = 1e-3  # 1ms timestep
    realtime_timer = chrono.ChRealtimeStepTimer()
    time = 0
    
    # Simulation loop
    while vis.Run():
        time = kraz.GetSystem().GetChTime()
        
        # Render scene and output POV-Ray data
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        
        # Get driver inputs
        driver_inputs = driver.GetInputs()
        
        # Synchronize subsystems
        driver.Synchronize(time)
        terrain.Synchronize(time)
        kraz.Synchronize(time, driver_inputs, terrain)
        vis.Synchronize(time, driver_inputs)
        
        # Advance simulation for all modules
        driver.Advance(step_size)
        terrain.Advance(step_size)
        kraz.Advance(step_size)
        vis.Advance(step_size)
        
        # Spin in place for real-time to catch up
        realtime_timer.Spin(step_size)

    return 0

if __name__ == "__main__":
    main()
```

3. **Kraz Vehicle**:

4. **Driver System**:
   - Interactive driver that responds to keyboard inputs

5. **Irrlicht Visualization**:
   - 1280x720 window with chase camera
   - Proper lighting and skybox
   - Real-time rendering capabilities

6. **Simulation Loop**:
   - 1ms timestep for accuracy
   - Synchronizes all subsystems
   - Maintains real-time execution
   - Handles user input for vehicle control
