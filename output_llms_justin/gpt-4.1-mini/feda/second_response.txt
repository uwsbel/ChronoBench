```python
import pychrono.core as chrono
import pychrono.irrlicht as irr
import pychrono.vehicle as veh
import math

chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

# Changed initial vehicle location per instruction #2 (-50 instead of 0 in X)
initLoc = chrono.ChVector3d(-50, 0, 0.5)
initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# Visualization type for vehicle parts (PRIMITIVES, MESH, or NONE)
vis_type = veh.VisualizationType_MESH

# Collision type for chassis (PRIMITIVES, MESH, or NONE)
chassis_collision_type = veh.CollisionType_NONE

# Type of tire model (RIGID, TMEASY)
tire_model = veh.TireModelType_TMEASY

# Terrain length changed per instruction #3 (200.0 instead of 100.0)
terrainHeight = 0      # terrain height
terrainLength = 200.0  # size in X direction, increased from 100.0
terrainWidth = 100.0   # size in Y direction

# Point tracked by the camera on the chassis
trackPoint = chrono.ChVector3d(-53.0, 0.0, 1.1)  # shifted to follow vehicle start position

# Contact method
contact_method = chrono.ChContactMethod_NSC
contact_vis = False

# Simulation step sizes
step_size = 1e-3
tire_step_size = step_size

# Time interval between two render frames
render_step_size = 1.0 / 50  # FPS = 50


# Create the FEDA vehicle, set parameters, and initialize
vehicle = veh.FEDA()
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
vehicle.SetTireType(tire_model)
vehicle.SetTireStepSize(tire_step_size)

vehicle.Initialize()

vehicle.SetChassisVisualizationType(vis_type)
vehicle.SetSuspensionVisualizationType(vis_type)
vehicle.SetSteeringVisualizationType(vis_type)
vehicle.SetWheelVisualizationType(vis_type)
vehicle.SetTireVisualizationType(vis_type)

vehicle.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# Create the terrain
patch_mat = chrono.ChContactMaterialNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)
terrain = veh.RigidTerrain(vehicle.GetSystem())
patch = terrain.AddPatch(patch_mat,
                        chrono.ChCoordsysd(chrono.ChVector3d(0, 0, 0), chrono.QUNIT),
                        terrainLength, terrainWidth)

patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)
patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
terrain.Initialize()


# ===================================================================
# Path-Follower with Cruise control driver system setup per instruction #1:
# ===================================================================

# Create the double lane change maneuver ISO filepath using PyChrono API

# Set double lane change maneuver parameters (typical ISO 3888-2 double lane change)
length = 100.0  # total maneuver length (approximate, we adjust accordingly)
lane_width = 3.5

# Create the path points following the ISO double lane change standard:
# The path consists of 7 points:
# Start (straight)
# First lane change left (offset +lanewidth)
# Return to center
# right lane change offset (-lanewidth)
# Return to center (final)
# etc.

# Construct a smooth path for double lane change:
# We'll use four sine curves portions to simulate the lane changes:
# But veh.Path expects a vector of ChVector points,
# but the vehicle::real and ChVectord are compatible and in meters.

# Constructing waypoints manually:

# The following points are chosen as reference for the lane changes along X and lateral Y shifts

path_points = []

# Start point, X = 0
path_points.append(chrono.ChVectorD(0.0, 0.0, 0.0))

# First lane change start and end points - approximate lateral shifts at X coords
# According to ISO 3888-2, total maneuver length is approx 98 m
# We'll approximate 4 lateral offsets with length approximations

# First lane change left at about X=20 (offset +lane_width)
path_points.append(chrono.ChVectorD(17.5, +lane_width, 0.0))

# Return to center at about X=40
path_points.append(chrono.ChVectorD(37.5, 0.0, 0.0))

# Right lane change start at about X=62
path_points.append(chrono.ChVectorD(62.5, -lane_width, 0.0))

# Return to center at about X=82
path_points.append(chrono.ChVectorD(82.5, 0.0, 0.0))

# End point at X=100 (final straight)
path_points.append(chrono.ChVectorD(100.0, 0.0, 0.0))

# Because the initial vehicle X position is -50, the path will start at 0,
# which is 50 m ahead of vehicle start. This is okay since path-follower can handle that.

# Create a ChLineSegment by interpolating line segments between points for the path
# For more smoothness, an actual VSpline or Bezier could be used, but PyChrono's vehicle module
# does not provide such by default. We'll use ChLineSegments and specify paths between waypoints.

# Create a ChLinePath
path = veh.ChBezierCurve()
# PyChrono vehicle module supports Bezier curves, let's use that for smoother path.

# The ChBezierCurve expects control points; approx the above waypoints can be control points.
for pt in path_points:
    path.AddPoint(pt)

# Set vehicle initial position to align better with path start:
# It is currently at X=-50 and path starts at 0.
# To follow the path we can have the path points shifted backward by 50 units in X.
# Alternatively, we move the vehicle start to (0, 0, 0.5) and extend terrain forward.
# But instructions want initial loc at -50.
# So we'll shift path points -50 on X so vehicle starts roughly on path start.
#
# To do this, subtract 50 from all path points X coordinate:
for i in range(path.GetNumPoints()):
    pt = path.GetPoint(i)
    path.SetPoint(i, chrono.ChVectorD(pt.x - 50.0, pt.y, pt.z))

# Now path start is at X=-50 matching vehicle start

# Create the path-follower driver system
driver = veh.ChPathFollowerDriver(vehicle, path, "my_path_follower_driver")

# Set target speed (m/s)
target_speed = 10.0
driver.SetTargetSpeed(target_speed)

# Configure steering controller:
# Look-ahead distance (recommended ~ 5 m per instructions)
driver.GetSteeringController().SetLookAheadDistance(5.0)
# Set steering control gains (typical values, can be tuned)
driver.GetSteeringController().SetGains(1.0, 0.05, 0.01)

# Configure speed controller gains (typical PID gains for speed, can tune as needed)
driver.GetSpeedController().SetGains(0.5, 0.01, 0)

driver.Initialize()

# output vehicle mass
print("VEHICLE MASS: ", vehicle.GetVehicle().GetMass())

# Number of simulation steps between miscellaneous events
render_steps = math.ceil(render_step_size / step_size)

# Initialize simulation frame counter s
realtime_timer = chrono.ChRealtimeStepTimer()
step_number = 0
render_frame = 0

# Create the vehicle Irrlicht interface after driver initialization (so vis tracks vehicle position properly)
vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle('FEDA vehicle - Double Lane Change Path Follower')
vis.SetWindowSize(1280, 1024)
vis.SetChaseCamera(trackPoint, 6.0, 0.5)
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetVehicle())


while vis.Run():
    time = vehicle.GetSystem().GetChTime()

    # Render scene and output POV-Ray data
    if (step_number % render_steps == 0):
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        render_frame += 1

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Update modules (process inputs from other modules)
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    vis.Synchronize(time, driver_inputs)

    # Advance simulation for one timestep for all modules
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    # Increment frame number
    step_number += 1

    # Spin in place for real time to catch up
    realtime_timer.Spin(step_size)
```
