```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment and core components

# Create a Chrono physical system
system = chrono.ChSystemNSC()

# 2. Add the required physical systems and objects as specified

# Create the ground body - fixed in space
ground = chrono.ChBody()
ground.SetBodyFixed(True)
ground.SetName("Ground")
system.Add(ground)

# Create the pendulum body
pendulum = chrono.ChBodyEasyBox(0.2, 1.0, 0.05,  # Size: thickness, length, width (m)
                                1000,           # Density (kg/m³)
                                True,           # Visualization on
                                True)           # Collision enabled (not critical here but good practice)
pendulum.SetName("Pendulum")

# Position the pendulum initially hanging downward from the pivot at origin
# We'll set the pendulum's center roughly at (0, -0.5, 0)
pendulum.SetPos(chrono.ChVectorD(0, -0.5, 0))

# Set some reasonable mass and inertia explicitly (optional, as SetBodyEasyBox handles it)
# pendulum.SetMass(1.0)
# pendulum.SetInertiaXX(chrono.ChVectorD(0.01, 0.01, 0.01))

system.Add(pendulum)

# Create a revolute joint (a hinge) connecting pendulum to ground at origin, allowing rotation around Z axis
joint = chrono.ChLinkLockRevolute()
joint.Initialize(pendulum, ground, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 0, 1))))
system.AddLink(joint)

# 3. Setup the Irrlicht visualization system
vis = chronoirr.ChIrrApp(system, "Simple Pendulum", chronoirr.dimension2du(800, 600))

# Add typical elements to the Irrlicht scene
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(1.0, 0.0, 1.5))
vis.AddLightWithShadow(chrono.ChVectorD(2, 4, 2), chrono.ChVectorD(0, 0, 0), 9, 1, 9, 30)
vis.AddShadowAll()

vis.AssetBindAll()
vis.AssetUpdateAll()

# Run simulation with logging pendulum position and velocity
time_step = 1e-3
next_log_time = 0
log_interval = 0.1

vis.SetTimestep(time_step)

while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    system.DoStepDynamics(time_step)

    # Periodically log pendulum position and velocity
    current_time = system.GetChTime()
    if current_time >= next_log_time:
        pos = pendulum.GetPos()
        vel = pendulum.GetPos_dt()
        print(f"Time: {current_time:.2f} s, Position: ({pos.x:.3f}, {pos.y:.3f}, {pos.z:.3f}), "
              f"Velocity: ({vel.x:.3f}, {vel.y:.3f}, {vel.z:.3f})")
        next_log_time += log_interval
```
