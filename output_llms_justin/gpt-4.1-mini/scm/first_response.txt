```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.vehicle.suspension as veh_susp
import pychrono.vehicle.wheeled_vehicle as veh_whl
import pychrono.vehicle.driver as veh_drv
import pychrono.vehicle.terrain as veh_terrain
import pychrono.irrlicht as chronoirr
import os

# -----------------------------------------------------------------------------
# Initialize everything
# -----------------------------------------------------------------------------

chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # make sure data path is set properly

# Create system (with default collision system)
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

# -----------------------------------------------------------------------------
# Create the HMMWV vehicle system
# -----------------------------------------------------------------------------

# Vehicle initial position and orientation
initLoc = chrono.ChVectorD(0, 0, 1.0)  # 1m above ground to avoid initial penetration
initRot = chrono.ChQuaternionD(1, 0, 0, 0)  # identity quaternion (no rotation)

vehicle = veh.HMMWV_Full()
vehicle.SetContactMethod(chrono.ChContactMethod_NSC)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysD(initLoc, initRot))
vehicle.SetTireType(veh.VehicleTireType.RIGID)  # rigid tires
vehicle.SetTireCollision(False)  # disable tire collisions (default: False)
vehicle.SetInitWheelRotations(True)  # initialize wheel rotation speed to zero
vehicle.SetUseMeshVisualization(True)  # use mesh visualization for all vehicle parts

vehicle.Initialize(system)
vehicle.SetChassisVisualizationType(veh.VisualizationType_MESH)
vehicle.SetSuspensionVisualizationType(veh.VisualizationType_MESH)
vehicle.SetSteeringVisualizationType(veh.VisualizationType_MESH)
vehicle.SetWheelVisualizationType(veh.VisualizationType_MESH)
vehicle.SetTireVisualizationType(veh.VisualizationType_MESH)

# -----------------------------------------------------------------------------
# Create the SCM deformable terrain
# -----------------------------------------------------------------------------

terrain = veh_terrain.SCMTerrain(system)
terrain.SetPlane(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT))

# Custom soil parameters (example values, can be adjusted)
soil_params = {
    "Bekker_Kphi": 150000,     # cohesion modulus [Pa/m^(n+1)]
    "Bekker_Kc": 20000,        # friction modulus [Pa/m^(n+2)]
    "Bekker_n": 1.1,           # exponent
    "Mohr_cohesion": 0,        # cohesion [Pa]
    "Mohr_friction": 30,       # friction angle [deg]
    "Janosi_shear": 56000,     # shear coefficient [Pa]
    "Elastic_K": 20000000,     # elastic stiffness [Pa/m]
    "Damping_R": 1500,         # damping coefficient [Pa*s/m]
}

terrain.SetSoilParameters(
    soil_params["Bekker_Kphi"],
    soil_params["Bekker_Kc"],
    soil_params["Bekker_n"],
    soil_params["Mohr_cohesion"],
    soil_params["Mohr_friction"],
    soil_params["Janosi_shear"],
    soil_params["Elastic_K"],
    soil_params["Damping_R"],
)

# Set plot type to visualize sinkage with false color
terrain.SetPlotType(veh_terrain.SCMTerrain.PlotType_SINKAGE)

# Enable moving patch (to reduce computational cost and keep terrain under the vehicle active)
terrain.EnableMovingPatch(True)
terrain.SetMovingPatchVehicle(vehicle.GetChassis().GetBody())
terrain.SetPatchDimensions(6, 6)  # patch dimensions in meters

terrain.Initialize()

# -----------------------------------------------------------------------------
# Create the interactive Irrlicht visualization system
# -----------------------------------------------------------------------------

vis = chronoirr.ChIrrApp(vehicle.GetVehicle(), "HMMWV Full Vehicle on SCM Terrain", chronoirr.dimension2du(1280, 720))
vis.AddTypicalLights()
vis.AddTypicalCamera(chronoirr.vector3df(0, -10, 2), chronoirr.vector3df(0, 0, 1))
vis.AssetBindAll()
vis.AssetUpdateAll()
vis.SetTimestep(0.02)  # Irrlicht update timestep (50Hz)

# -----------------------------------------------------------------------------
# Create the interactive driver system
# -----------------------------------------------------------------------------

driver = veh_drv.ChInteractiveDriverIrrlicht(vis)
driver.SetInputDelay(0.0)  # no input delay
driver.SetThrottleDelta(0.05)  # throttle change per keyboard press
driver.SetSteeringDelta(0.05)  # steering change per keyboard press
driver.SetBrakingDelta(0.1)  # brake change per keyboard press

# -----------------------------------------------------------------------------
# Simulation parameters
# -----------------------------------------------------------------------------

step_size = 0.02  # 50 Hz simulation update
render_step_size = 1.0 / 50  # 50 FPS rendering

# Initialize timer for syncing simulation and rendering
time = 0
render_frame = 0

# -----------------------------------------------------------------------------
# Simulation loop
# -----------------------------------------------------------------------------

vis.SetEasyCamera(True)

while vis.Run():
    # Synchronize driver inputs
    driver.Synchronize(time)
    # Synchronize vehicle with driver inputs and terrain forces
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver.GetInputs(), terrain)
    vis.BeginScene()
    vis.DrawAll()

    # Advance simulation at fixed step size
    vehicle.Advance(step_size)
    terrain.Advance(step_size)
    driver.Advance(step_size)
    system.DoStepDynamics(step_size)

    vis.EndScene()

    time += step_size

# -----------------------------------------------------------------------------
# End of simulation
# -----------------------------------------------------------------------------
```
