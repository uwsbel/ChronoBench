```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as turtlebot
from pychrono import irrlicht as chronoirr
import pychrono.sensor as sens  # Added sensor import as per instruction

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))  # Gravity in negative Z direction
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create ground body as terrain plane that robot will drive on
ground_mat = chrono.ChMaterialSurfaceNSC()  # Corrected: ChContactMaterialNSC -> ChMaterialSurfaceNSC
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVector3d(0, 0, -0.6))  # Position changed from -1 to -0.6 per instruction
ground.SetFixed(True)
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# Add 5 boxes with random positions for interaction (instruction #3)
np.random.seed(42)  # fixed seed for reproducibility
for i in range(5):
    box_size = [0.3, 0.3, 0.3]
    box_pos = chrono.ChVector3d(np.random.uniform(-5, 5),
                                np.random.uniform(-5, 5),
                                box_size[2] / 2 - 0.6)  # place on ground, ground at z=-0.6
    box_mat = chrono.ChMaterialSurfaceNSC()
    box = chrono.ChBodyEasyBox(box_size[0], box_size[1], box_size[2], 1000, True, True, box_mat)
    box.SetPos(box_pos)
    box.SetMass(5)
    box.SetInertiaXX(chrono.ChVectorD(0.1, 0.1, 0.1))
    box.SetCollide(True)
    system.Add(box)

# Create Turtlebot Robot
init_pos = chrono.ChVector3d(0, 0.2, 0)  # Initial position of the robot
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)  # Initial orientation of the robot
robot = turtlebot.TurtleBot(system, init_pos, init_rot)
robot.Initialize()

# Create sensor manager and add lidar sensor (instruction #2)
# Attach the sensor manager to the Chrono system and to the robot body
manager = sens.ChSensorManager(system)
manager.SetVerbose(False)
manager.Initialize()

# Set update rate of sensor (Hz)
sensor_update_rate = 30  # 30 Hz

# Sensor pose relative to robot chassis frame
# Place lidar sensor about 0.15 m above base frame and facing forward
lidar_pose = chrono.ChFrameD()
lidar_pose.SetPos(chrono.ChVectorD(0, 0, 0.15))

# Configure lidar sensor parameters
lidar = sens.ChLidarSensor(
    robot.GetBody(),          # sensor parent body
    sensor_update_rate,       # update rate in Hz
    lidar_pose,               # offset pose relative to parent body
    25.0,                    # scanning rotation frequency (Hz)
    120,                     # horizontal samples per scan
    math.radians(40),        # horizontal field of view (fov) in radians (40 deg)
    3,                       # number of vertical channels
    math.radians(20),        # vertical field of view in radians (20 deg)
    20,                      # max distance (m)
    20000                    # points per sweep (max)
)

# Add filter chain to lidar sensor
# 1. point cloud to XYZRGB
filter_xyz = sens.ChFilterPCfromXYZI()
lidar.PushFilter(filter_xyz)

# 2. Add color depending on range (indigo colorize)
filter_indigo = sens.ChFilterPCtoColorIndigo(0, 20)
filter_xyz.PushFilter(filter_indigo)

# 3. Publish point cloud as visualization cloud
filter_vis = sens.ChFilterVisualizePointCloud()
filter_indigo.PushFilter(filter_vis)
vis.AddVisualModel(filter_vis.GetVisualModel())

# Add lidar sensor to manager
manager.AddSensor(lidar)

# Create run-time visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Turtlebot Robot - Rigid terrain with sensors')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 1.5, 0.2), chrono.ChVector3d(0, 0, 0.2))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0.5), 3, 4, 10, 40, 512)

# Enable shadows (commented out for performance)
# vis.EnableShadows()

# Set the simulation time step
time_step = 2e-3

# Define WheelID enum values for motor control for clarity outside loop
LEFT_DRIVE_WHEEL = 0
RIGHT_DRIVE_WHEEL = 1

# Motion control function for Turtlebot (instruction #4)
def move(mode):
    if mode == 'straight':
        # Both wheels same forward speed
        speed = math.pi / 2  # moderate forward speed
        robot.SetMotorSpeed(speed, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(speed, RIGHT_DRIVE_WHEEL)
    elif mode == 'left':
        # Turn left by rotating right wheel forward, left wheel stopped
        robot.SetMotorSpeed(0, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(math.pi / 2, RIGHT_DRIVE_WHEEL)
    elif mode == 'right':
        # Turn right by rotating left wheel forward, right wheel stopped
        robot.SetMotorSpeed(math.pi / 2, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0, RIGHT_DRIVE_WHEEL)
    elif mode == 'stop':
        # Stop
        robot.SetMotorSpeed(0, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0, RIGHT_DRIVE_WHEEL)
    else:
        # Default stop
        robot.SetMotorSpeed(0, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0, RIGHT_DRIVE_WHEEL)

# Simulation loop (instruction #5)
time = 0.0
while vis.Run():
    # Move turtlebot straight continuously
    move('straight')

    # Update sensor manager (important!)
    manager.Update()

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance the simulation by one time step
    system.DoStepDynamics(time_step)

    # Increment time counter
    time += time_step
```
