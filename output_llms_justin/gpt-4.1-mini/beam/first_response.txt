```python
import pychrono as chrono
import pychrono.fea as fea
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # ensure data path is set

# Create a physical system (ChSystemSMC recommended for FEA with beams)
system = chrono.ChSystemSMC()

# 2. Add the required physical systems and objects as specified:
# Create a mesh container to hold beam elements and nodes
mesh = fea.ChMesh()

# Define beam section properties (e.g., rectangular beam section)
beam_section = fea.ChBeamSectionEulerAdvanced()
beam_section.SetDiameter(0.02)  # circular beam diameter (m)
beam_section.SetYoungModulus(2.1e11)  # steel approx. Young's modulus (Pa)
beam_section.SetGwithPoissonRatio(8.1e10, 0.3)  # Shear modulus G and Poisson ratio v
beam_section.SetBeamRaleyghDamping(0.02)  # some structural damping

# Create nodes
# For example, create a simple cantilever beam with 5 nodes connected by 4 beam elements

num_nodes = 5
length = 1.0  # total beam length in meters
node_spacing = length / (num_nodes - 1)

nodes = []
for i in range(num_nodes):
    pos = chrono.ChVectorD(i * node_spacing, 0, 0)
    node = fea.ChNodeFEAxyzrot(pos)
    node.SetMass(0.1)  # assign some mass (kg)
    # Fix first node (cantilever root)
    if i == 0:
        node.SetFixed(True)
    mesh.AddNode(node)
    nodes.append(node)

# Create beam elements (ChElementBeamEuler)
for i in range(num_nodes - 1):
    beam_element = fea.ChElementBeamEuler()
    beam_element.SetNodes(nodes[i], nodes[i + 1])
    beam_element.SetSection(beam_section)
    beam_element.SetDensity(7850)  # steel density kg/m^3
    mesh.AddElement(beam_element)

# Add the mesh to the physical system
system.Add(mesh)

# 3. Add visualization
# Visualization of beam elements and nodes
mesh.AddVisualization(fea.ChVisualization.Type_BEAMS)
mesh.AddVisualization(fea.ChVisualization.Type_NODE_CSYS)

# Optionally add external forces or gravity
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # gravity

# 4. Create Irrlicht application for real-time rendering and run the simulation loop
app = chronoirr.ChIrrApp(system, "Beam Finite Element Demo", chronoirr.dimension2du(800, 600))
app.AddTypicalSky()
app.AddTypicalLights()
app.AddTypicalCamera(chronoirr.vector3df(1.5, 1.5, 1.5))

app.AssetBindAll()
app.AssetUpdateAll()

app.SetTimestep(1e-3)
app.SetTryRealtime(True)

# Simulation loop
while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()
    app.DoStep()
    app.EndScene()
```
