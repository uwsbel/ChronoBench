```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr

# -----------------------------------------------------------------------------
# 1. Initialize PyChrono environment and core components
# -----------------------------------------------------------------------------

chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # ensure data path is set

time_step = 1.0 / 1000  # simulation timestep

# Create the physical system
system = chrono.ChSystemSMC()

# -----------------------------------------------------------------------------
# 2. Add the required physical systems and objects
# -----------------------------------------------------------------------------

# Create terrain (rigid terrain)
terrain = veh.RigidTerrain(system)

# Define a rigid patch for ground with specific friction and restitution
patch = terrain.AddPatch(
    chrono.ChVectorD(0, 0, 0),                 # center location
    chrono.ChVectorD(0, 0, 1),                 # normal direction (z-up)
    100, 100                                   # size X, Y (100 m x 100 m)
)
# friction and restitution
patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)
patch.SetMaterialSurface(
    chrono.ChMaterialSurfaceSMC()
)
patch.GetMaterialSurface().SetFriction(0.9)
patch.GetMaterialSurface().SetRestitution(0.01)

terrain.Initialize()

# Create the M113 vehicle system
m113 = veh.M113Vehicle(system)
m113.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0.5), chrono.QUNIT))
m113.SetChassisVisualizationType(veh.VisualizationType_MESH)
m113.SetTrackShoeVisualizationType(veh.VisualizationType_MESH)
m113.Initialize()

# -----------------------------------------------------------------------------
# Initialize a driver system for vehicle control
# -----------------------------------------------------------------------------

# Create a driver to generate inputs for the vehicle.
# Using a simple interactive driver with Irrlicht GUI
driver = veh.ChIrrGuiDriver(m113.GetVehicle())

# Set driver time response (for speed, steering, braking inputs).
driver.SetInputTimeResponse(1.0, 1.0, 0.5)

# Set initial driver inputs: neutral steering and throttle
driver.Initialize()

# -----------------------------------------------------------------------------
# 3. Setup Irrlicht visualization system
# -----------------------------------------------------------------------------

vis = veh.ChWheeledVehicleIrrApp(m113.GetVehicle(), "M113 Vehicle Simulation")

# Configure the irrlicht application window and camera
vis.AddTypicalLights()
vis.AddSkyBox()
vis.AddLogo()
vis.AddHelp()

# Customize camera position and target (focus on vehicle)
vis.SetChaseCamera(trackPoint=chrono.ChVectorD(0, 0, 1.75),
                   chaseDist=7.0,
                   chaseHeight=1.75,
                   chasePitch=0.3)

vis.GetDevice().GetVideoDriver().getMaterialRenderer(chrono.material.E_MATERIAL_FLAG::EMF_WIREFRAME)

# Initialize the visualization system
vis.AssetBindAll()
vis.AssetUpdateAll()
vis.SetTimestep(time_step)
vis.Initialize()

# -----------------------------------------------------------------------------
# Simulation loop
# -----------------------------------------------------------------------------

while vis.GetDevice().run():
    vis.BeginScene()
    vis.DrawAll()

    # Get driver inputs (steering, throttle, braking)
    steering = driver.GetSteering()
    throttle = driver.GetThrottle()
    braking = driver.GetBraking()

    # Synchronize modules at current time
    time = system.GetChTime()
    driver.Synchronize(time)
    m113.Synchronize(time, steering, throttle, braking, terrain)
    terrain.Synchronize(time)
    vis.Synchronize(time, steering, throttle, braking)

    # Advance all modules by the simulation timestep
    driver.Advance(time_step)
    m113.Advance(time_step)
    terrain.Advance(time_step)
    vis.Advance(time_step)

    # Advance the physical system
    system.DoStepDynamics(time_step)

    vis.EndScene()
```
