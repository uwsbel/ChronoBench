```python
# Import necessary PyChrono modules
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Initialize PyChrono environment
chrono.Ch纪元Initialize()

# Create a physical system
system = chrono.ChSystemNSC()

# Add the ground
ground = chrono.ChBodyEasyBox(100, 100, 1, 0, 0, 0)
ground.SetPos(chrono.ChVectorD(-50, -50, 0))
ground.SetBodyFixed(True)
system.AddBody(ground)

# Define terrain dimensions and create RigidTerrain
terrain_dimensions = chrono.ChVectorD(100, 100, 0.1)
terrain = chrono.ChRigidTerrain(terrain_dimensions.x, terrain_dimensions.y, terrain_dimensions.z)
terrain.SetContactMaterial(chrono.ChMaterialSurfaceNSC())
terrain.SetFriction(0.8)
terrain.SetRestitution(0.0)
system.Add(terrain)

# Load custom terrain texture
terrain_texture = chrono.ChTexture()
terrain_texture.SetTextureFilename(chrono.GetChronoDataFile("terrain/terrain.jpg"))
terrain.SetTexture(terrain_texture)

# Add the ARTcar vehicle
artcar_mass = 1000
artcar_inertia = chrono.ChVectorD(100, 100, 100)
artcar = chrono.ChBodyEasyCylinder(2, artcar_mass, artcar_inertia, 0, 0, 0)
artcar.SetPos(chrono.ChVectorD(0, 0, 0.05))
artcar.SetBodyFixed(False)
system.Add(artcar)

# Define the driver system
driver = chrono.ChDriverSteeringThrottleBrake()
driver.SetVehicle(artcar)
driver.SetMaxSteeringAngle(chrono.CH_C_PI_2)
driver.SetMaxThrottle(1.0)
driver.SetMaxBrake(1.0)

# Add Irrlicht visualization
vis_system = chronoirr.ChIrrApp(system, "ARTcar Simulation", chronoirr.dimension2du(800, 600))
vis_system.AddTypicalCameraSetup()
vis_system.AddTypicalLightsSetup()
vis_system.AddTypicalDriverSetup(driver)
vis_system.SetTimestep(1.0/50)
vis_system.AddTypicalPostProcessing()
vis_system.AddTypicalVehicleDrawSettings()
vis_system.AssetBindAll()
vis_system.AssetUpdateAll()
vis_system.AddLogStream(chrono.IRRLOG_STANDARD)

# Run the simulation loop
while vis_system.GetDevice().run():
    vis_system.BeginScene()
    vis_system.DrawAll()
    system.DoStepDynamics(1.0/50)
    vis_system.EndScene()

# Finalize PyChrono environment
chrono.Ch纪元Finalize()
```
