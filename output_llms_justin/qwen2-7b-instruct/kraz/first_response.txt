```python
# Import necessary PyChrono modules
from pychrono.core import ChSystem, ChVector
from pychrono.vehicle import ChKraz, ChDriverSystem
from pychrono.irrlicht import ChIrrApp, irr
from pychrono.geometry import ChTerrain, ChTerrainMaterial

# Initialize PyChrono environment
chrono = ChSystem()
chrono.Set_G_acc(ChVector<>(0, -9.81, 0))

# Define physical parameters
vehicle_mass = 1000.0
terrain_height = 100.0
terrain_width = 100.0
terrain_length = 100.0
terrain_friction = 0.8
terrain_restitution = 0.1

# Create a Kraz vehicle
kraz = ChKraz()
kraz.SetMass(vehicle_mass)
kraz.Initialize(chrono, ChVector<>(0, 0, 0), ChVector<>(0, 0, 0))

# Set initial conditions for the Kraz vehicle
kraz.SetPos(ChVector<>(0, terrain_height / 2, 0))
kraz.SetRot(ChQuaternion<>(1, 0, 0, 0))
kraz.SetVel(ChVector<>(0, 0, 0))
kraz.SetOmega(ChVector<>(0, 0, 0))

# Create terrain
terrain_material = ChTerrainMaterial()
terrain_material.SetFriction(terrain_friction)
terrain_material.SetRestitution(terrain_restitution)
terrain = ChTerrain()
terrain.Initialize(chrono, terrain_material, terrain_height, terrain_width, terrain_length)

# Create a driver system for vehicle control
driver = ChDriverSystem()
driver.Initialize(chrono, kraz)

# Initialize Irrlicht visualization
app = ChIrrApp(chrono, "PyChrono Kraz Vehicle Simulation", irr::core::dimension2d<irr::u32>(800, 600))
app.AddTypicalSky()
app.AddTypicalLogo("logo.pch")
app.AddTypicalCamera(ChVector<>(0, 10, 5))
app.AddTypicalLights(irr::video::E_LIGHT_TYPE::ETS_LIGHT_TYPE_DIRECTIONAL)
app.SetSceneShadowMode(irr::video::ESHADEMODE_SHADOWMAPS_POINT)

# Simulation loop parameters
dt = 0.01  # Time step
max_steps = 10000  # Maximum simulation steps

# Simulation loop
for step in range(max_steps):
    # Advance vehicle, terrain, driver, and visualization systems
    driver.Advance(dt)
    terrain.Advance(dt)
    app.SetTimestep(dt)
    app.DoStep()
    app.RenderStep()

# Finalize PyChrono environment
chrono.Finalize()
```

This script initializes a PyChrono simulation featuring a Kraz vehicle with specified initial conditions and visualization settings. It sets up a rigid terrain with defined friction and restitution, creates a driver system for vehicle control, and initializes a real-time visualization using Irrlicht with camera settings and lighting. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.
