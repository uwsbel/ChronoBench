```python
# Import necessary PyChrono components
from pychrono importchrono as ch
from pychrono.core import ChVector_, ChMatrixSp_
from pychrono.geometric import ChColorAsset
from pychrono.physics import ChSystemParallel
from pychrono.inputoutput import ChVisualization, ChIrrApp
from pychrono.physics import ChSystemChRONO

# Initialize the PyChrono system
chrono_system = ChSystemChRONO()

# Add the Irrlicht visualization system
vis_sys = ChVisualization(chrono_system)
vis_sys.SetWindowSize(800, 600)
vis_sys.SetCameraPos(ChVector_(0, 0, 20))
vis_sys.SetCameraTarget(ChVector_(0, 0, 0))
vis_sys.SetCameraUpVector(ChVector_(0, 1, 0))
vis_sys.Initialize()

# Enable the Irrlicht visualization
vis_sys.EnableWindow()

# Add the truss (fixed support)
truss = ch.ChLinkTrapezoidBeam()
truss.SetMaterialSurface(ch.ChMaterialSurfaceStandard())
truss.SetBeamSection(ch.ChVector<>(0.1, 0.05, 0.05))
truss.SetBeamMaterial(ch.ChMaterialSurfaceSteel())
truss.SetBeamColor(ChColorAsset(0.8, 0.8, 0.8, 1))
truss.SetBeamDensity(7850)
truss.SetBeamElasticity(210e9)
truss.SetBeamPoissonRatio(0.3)
truss.SetBeamCrossSection(ch.ChVector<>(0.1, 0.05, 0.05))
truss.SetBeamLength(10)
truss.SetBeamMassPerMeter(500)
truss.SetBeamMassCenter(ChVector_(0, 0, 0))
truss.SetBeamSupportEnd(ch.ChLinkTrapezoidBeam.EndType.Fixed)
truss.SetBeamSupportStart(ch.ChLinkTrapezoidBeam.EndType.Fixed)
chrono_system.AddLink(truss)

# Add the rotating bar
bar = ch.ChLinkBar()
bar.SetMaterialSurface(ch.ChMaterialSurfaceStandard())
bar.SetBarSection(ch.ChVector<>(0.1, 0.01, 0.01))
bar.SetBarMaterial(ch.ChMaterialSurfaceSteel())
bar.SetBarColor(ChColorAsset(0.6, 0.6, 0.6, 1))
bar.SetBarDensity(7850)
bar.SetBarElasticity(210e9)
bar.SetBarPoissonRatio(0.3)
bar.SetBarCrossSection(ch.ChVector<>(0.1, 0.01, 0.01))
bar.SetBarLength(5)
bar.SetBarMassPerMeter(100)
bar.SetBarMassCenter(ChVector_(0, 0, 0))
bar.SetBarSupportEnd(ch.ChLinkBar.EndType.Fixed)
bar.SetBarSupportStart(ch.ChLinkBar.EndType.Free)
chrono_system.AddLink(bar)

# Add the first gear
gear1 = ch.ChLinkGearTrain()
gear1.SetMaterialSurface(ch.ChMaterialSurfaceStandard())
gear1.SetGearType(ch.ChLinkGearTrain.GearType.Circular)
gear1.SetGearRatio(1)
gear1.SetGearRadius(0.1)
gear1.SetGearMass(10)
gear1.SetGearColor(ChColorAsset(0.5, 0.5, 1, 1))
gear1.SetGearMaterial(ch.ChMaterialSurfaceSteel())
gear1.SetGearDensity(7850)
gear1.SetGearElasticity(210e9)
gear1.SetGearPoissonRatio(0.3)
gear1.SetGearCrossSection(ch.ChVector<>(0.01, 0.01, 0.01))
gear1.SetGearLength(0.2)
gear1.SetGearCenter(ChVector_(0, 0, 0))
gear1.SetGearSupportEnd(ch.ChLinkGearTrain.EndType.Fixed)
gear1.SetGearSupportStart(ch.ChLinkGearTrain.EndType.Free)
chrono_system.AddLink(gear1)

# Add the second gear
gear2 = ch.ChLinkGearTrain()
gear2.SetMaterialSurface(ch.ChMaterialSurfaceStandard())
gear2.SetGearType(ch.ChLinkGearTrain.GearType.Circular)
gear2.SetGearRatio(1)
gear2.SetGearRadius(0.1)
gear2.SetGearMass(10)
gear2.SetGearColor(ChColorAsset(1, 0.5, 0.5, 1))
gear2.SetGearMaterial(ch.ChMaterialSurfaceSteel())
gear2.SetGearDensity(7850)
gear2.SetGearElasticity(210e9)
gear2.SetGearPoissonRatio(0.3)
gear2.SetGearCrossSection(ch.ChVector<>(0.01, 0.01, 0.01))
gear2.SetGearLength(0.2)
gear2.SetGearCenter(ChVector_(0, 0, 0))
gear2.SetGearSupportEnd(ch.ChLinkGearTrain.EndType.Free)
gear2.SetGearSupportStart(ch.ChLinkGearTrain.EndType.Fixed)
chrono_system.AddLink(gear2)

# Add the gear motor to enforce constant rotation speed
motor = ch.ChLinkMotorTorque()
motor.SetMotorType(ch.ChLinkMotorTorque.MotorType.Gear)
motor.SetMotorTorque(100)
motor.SetMotorMass(1)
motor.SetMotorColor(ChColorAsset(1, 1, 1, 1))
motor.SetMotorMaterial(ch.ChMaterialSurfaceSteel())
motor.SetMotorDensity(7850)
motor.SetMotorElasticity(210e9)
motor.SetMotorPoissonRatio(0.3)
motor.SetMotorCrossSection(ch.ChVector<>(0.01, 0.01, 0.01))
motor.SetMotorLength(0.2)
motor.SetMotorCenter(ChVector_(0, 0, 0))
motor.SetMotorSupportEnd(ch.ChLinkMotorTorque.EndType.Fixed)
motor.SetMotorSupportStart(ch.ChLinkMotorTorque.EndType.Free)
chrono_system.AddLink(motor)

# Connect gears and motor to the bar
gear1.ConnectTo(bar.GetFrame1(), bar.GetFrame1().GetPos())
gear2.ConnectTo(bar.GetFrame2(), bar.GetFrame2().GetPos())
motor.ConnectTo(gear1.GetFrame1(), gear1.GetFrame1().GetPos())

# Set the initial conditions
chrono_system.SetGravity(ChVector_(0, -9.81, 0))
chrono_system.SetTimestepping(chrono.ChTimestep(0.001))

# Start the simulation
vis_sys.StartSimulation(chrono_system)
```
