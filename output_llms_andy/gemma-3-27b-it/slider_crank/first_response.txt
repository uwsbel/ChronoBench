```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import math

# ---
#  Setup the simulation
# ---

# Create a Chrono physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# ---
# Create the floor (a simple truss)
# ---

body_floor = chrono.ChBodyEasy()
body_floor.SetBodyFixed(True)
body_floor.SetPos(chrono.ChVectorD(0, -0.5, 0))
body_floor.SetShapeBoxes(0.5, 0.1, 1)  # Dimensions: x, y, z
body_floor.SetCollide(True)
system.Add(body_floor)

# ---
# Create the crankshaft
# ---

crank_radius = 0.2
crank_length = 0.3
crank_mass = 0.1
crank_inertia = 0.01

body_crank = chrono.ChBodyEasy()
body_crank.SetMass(crank_mass)
body_crank.SetPos(chrono.ChVectorD(0, 0.5, 0))
body_crank.SetShapeCylinder(crank_radius, crank_length)
body_crank.SetCollide(True)
system.Add(body_crank)

# ---
# Create the connecting rod
# ---

rod_length = 0.4
rod_mass = 0.1
rod_inertia = 0.01

body_rod = chrono.ChBodyEasy()
body_rod.SetMass(rod_mass)
body_rod.SetPos(chrono.ChVectorD(0, 0.5, 0))
body_rod.SetShapeBox(rod_length / 2, 0.05, 0.05)
body_rod.SetCollide(True)
system.Add(body_rod)

# ---
# Create the piston
# ---

piston_radius = 0.05
piston_length = 0.1
piston_mass = 0.1
piston_inertia = 0.01

body_piston = chrono.ChBodyEasy()
body_piston.SetMass(piston_mass)
body_piston.SetPos(chrono.ChVectorD(0, 0.5, 0))
body_piston.SetShapeCylinder(piston_radius, piston_length)
body_piston.SetCollide(True)
system.Add(body_piston)

# ---
# Create the joints
# ---

# Crank-Rod joint (Revolute)
joint_crank_rod = chrono.ChLinkRevolute()
joint_crank_rod.Initialize(body_crank, body_rod, chrono.ChCoordsys(chrono.ChVectorD(crank_length / 2, 0, 0)))
system.Add(joint_crank_rod)

# Rod-Piston joint (Revolute)
joint_rod_piston = chrono.ChLinkRevolute()
joint_rod_piston.Initialize(body_rod, body_piston, chrono.ChCoordsys(chrono.ChVectorD(rod_length / 2, 0, 0)))
system.Add(joint_rod_piston)

# Piston-Floor joint (Prismatic)
joint_piston_floor = chrono.ChLinkPrismatic()
joint_piston_floor.Initialize(body_piston, body_floor, chrono.ChCoordsys(chrono.ChVectorD(0, -piston_length / 2, 0)))
joint_piston_floor.SetDirection(chrono.ChVectorD(0, 1, 0))
joint_piston_floor.SetLimitLower(0)
joint_piston_floor.SetLimitUpper(1)
system.Add(joint_piston_floor)

# ---
# Add a motor to the crankshaft
# ---

motor = chrono.ChFunction_Const()
motor.Set_y(10)  # Angular velocity (rad/s)
motor.Set_min_y(-10)
motor.Set_max_y(10)

driver = chrono.ChDriver(motor)
joint_crank_rod.Set_motor_function(driver)
joint_crank_rod.Enable_motor()

# ---
# Visualization
# ---

vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Crank-Slider Mechanism')
vis.Initialize()
vis.AddCamera(chrono.ChVectorD(2, 2, -2))
vis.AddTypicalLights()

# Add a logo
# vis.AddLogo()

# ---
# Simulation loop
# ---

time_step = 0.005
simulation_time = 10

while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    system.DoStepDynamics(time_step)

    if system.GetChTime() > simulation_time:
        break
```
