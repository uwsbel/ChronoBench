    update_rate,
    horizontal_samples,
    horizontal_fov,
    vertical_samples,
    vertical_fov,
    100.0,

    chrono.ChVector3d(0, 0, 0),

lidar.PushFilter(sens.ChFilterVisualizePointCloud(640, 480, 1.0, "Lidar Point Cloud"))
manager.AddSensor(lidar)

    # Update the sensor manager
    manager.Update()

    # Render the scene

    vis.Render()
