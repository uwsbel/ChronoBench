    Create a PyChrono simulation showcasing a Curiosity rover navigating on a rigid terrain. The system includes a fixed ground with collision properties and a rover model equipped with a motor control driver for real-time steering inputs. Visualize the simulation using the Irrlicht visualization system with customizable camera settings, lighting, shadows, and additional visual elements like textures and logos.
     ```python
    import pychrono.core as chrono
    import pychrono.irrlicht as irr
    import pychrono.robot as robot
    import pychrono.vehicle as veh
    import math

    # Initialize the Chrono simulation system
    sys = chrono.ChSystemNSC()

    # Create a contact material shared among all bodies for collision interactions
    contact_mat = chrono.ChContactMaterialNSC()
    contact_mat.SetFriction(0.9)
    contact_mat.SetRestitution(0.01)

    # Create the ground body as a box, fix it, and set collision properties
    ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, contact_mat)
    ground.SetPos(chrono.ChVector3d(0, 0, -0.5))  # Position the ground slightly below the origin
    ground.SetFixed(True)  # Fix the ground in place
    ground.GetVisualShape(0).SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)
    ground.SetName("ground")
    sys.Add(ground)  # Add the ground to the simulation system

    # Create a rigid terrain patch and add it to the ground body
    patch_mat = chrono.ChContactMaterialNSC()
    patch_mat.SetFriction(0.9)
    patch_mat.SetRestitution(0.01)
    terrain = veh.RigidTerrain(sys, ground)
    patch = terrain.AddPatch(patch_mat, chrono.ChCoordsysd(chrono.ChVector3d(0, 0, 0), chrono.ChQuaterniond(1, 0, 0, 0)), 20, 20)
    patch.SetTexture(veh.GetDataFile("terrain/textures/dirt.jpg"), 200, 200)
    patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
    terrain.Build()  # Build the terrain

    # Create the Curiosity rover and add it to the simulation system
    rover = robot.Curiosity(sys, contact_mat)
    rover.SetPos(chrono.ChVector3d(0, 0.2, 0))  # Position the rover slightly above the ground
    rover.SetDriver(robot.ChWheeledRobotDCMotorControl())
    rover.GetDriver().SetSteeringDelta(0.0)  # Initialize steering to zero
    sys.Add(rover)  # Add the rover to the simulation system

    # Create the Irrlicht visualization system
    vis = irr.ChVisualSystemIrrlicht()
    vis.AttachSystem(sys)  # Attach the Chrono system to the visualization
    vis.SetWindowSize(1280, 1024)  # Set the window size for the visualization
    vis.SetWindowTitle('Curiosity rover - Rigid terrain')  # Set the window title
    vis.Initialize()  # Initialize the visualization system
    vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))  # Add a logo to the visualization
    vis.AddSkyBox()  # Add a skybox for better visual appearance
    vis.AddCamera(chrono.ChVector3d(0, 2.5, 0.2), chrono.ChVector3d(0, 0, 0.2))  # Add a camera to the visualization
    vis.AddTypicalLights()  # Add typical lights for better illumination
    vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0.5), 3, 4, 10, 40, 512)  # Add a light with shadow

    # Set the simulation time step
    time_step = 1e-3

    # Simulation loop
    time = 0
    while vis.Run():
        # Define the steering behavior over time
        if time < 1:
            rover.GetDriver().SetSteeringDelta(0.2)
        elif time < 3:
            rover.GetDriver().SetSteeringDelta(0.0)
        else:
            rover.GetDriver().SetSteeringDelta(-0.2)
        
        # Update rover dynamics
        rover.Update()

        # Render the scene
        vis.BeginScene()
        vis.Render()
        vis.EndScene()

        # Advance the simulation by one time step
        sys.DoStepDynamics(time_step)

        # Increment the simulation time
        time += time_step

    ```
