```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import math as m

# Create a Chrono::Engine physical system
sys = chrono.ChSystemNSC()

# Create contact material shared among all bodies
mat = chrono.ChContactMaterialNSC()

# Create all rigid bodies with specific dimensions
radA = 1.5
radB = 3.5

# Create the truss
mbody_truss = chrono.ChBodyEasyBox(15, 8, 2, 1000, True, False, mat)
sys.Add(mbody_truss)
mbody_truss.SetFixed(True)
mbody_truss.SetPos(chrono.ChVector3d(0, 0, 3))

# Shared visualization material for enhanced aesthetics
vis_mat = chrono.ChVisualMaterial()
vis_mat.SetKdTexture(chrono.GetChronoDataFile('textures/pinkwhite.png'))

# Create the rotating bar support for the two epicycloidal wheels
mbody_train = chrono.ChBodyEasyBox(8, 1.5, 1.0, 1000, True, False, mat)
sys.Add(mbody_train)
mbody_train.SetPos(chrono.ChVector3d(3, 0, 0))

# Create a revolute joint between truss and rotating bar, allowing rotation along the Z-axis
link_revoluteTT = chrono.ChLinkLockRevolute()
link_revoluteTT.Initialize(mbody_truss, mbody_train,
                           chrono.ChFrameD(chrono.ChVector3d(0, 0, 0), chrono.QUNIT))
sys.AddLink(link_revoluteTT)

# Create the first gear
mbody_gearA = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, radA, 0.5, 1000, True, False, mat)
sys.Add(mbody_gearA)
mbody_gearA.SetPos(chrono.ChVector3d(0, 0, -1))
mbody_gearA.SetRot(chrono.Q_from_AngX(m.pi / 2))
mbody_gearA.GetVisualShape(0).SetMaterial(0, vis_mat)

# Adding a thin cylinder only for visualization purpose
mshaft_shape = chrono.ChVisualShapeCylinder(radA * 0.3, 10)
mbody_gearA.AddVisualShape(mshaft_shape, chrono.ChFrameD(chrono.ChVector3d(0, 3.5, 0), chrono.Q_from_AngX(chrono.CH_PI_2)))

# Impose rotation speed on the first gear relative to the fixed truss
link_motor = chrono.ChLinkMotorRotationSpeed()
link_motor.Initialize(mbody_gearA, mbody_truss,
                      chrono.ChFrameD(chrono.ChVector3d(0, 0, 0), chrono.QUNIT))
link_motor.SetSpeedFunction(chrono.ChFunctionConst(3))
sys.AddLink(link_motor)

# Create the second gear
interaxis12 = radA + radB
mbody_gearB = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, radB, 0.4, 1000, True, False, mat)
sys.Add(mbody_gearB)
mbody_gearB.SetPos(chrono.ChVector3d(interaxis12, 0, -2))
mbody_gearB.SetRot(chrono.Q_from_AngX(m.pi / 2))
mbody_gearB.GetVisualShape(0).SetMaterial(0, vis_mat)

# Fix second gear to the rotating bar with a revolute joint
link_revolute = chrono.ChLinkLockRevolute()
link_revolute.Initialize(mbody_gearB, mbody_train,
                         chrono.ChFrameD(chrono.ChVector3d(interaxis12, 0, 0), chrono.QUNIT))
sys.AddLink(link_revolute)

# Create the gear constraint between the two gears, A and B
link_gearAB = chrono.ChLinkGear()
link_gearAB.Initialize(mbody_gearA, mbody_gearB, False,  # False = external gears
                       chrono.ChFrameD(chrono.ChVector3d(0, 0, -1), chrono.Q_from_AngX(-m.pi / 2)),
                       chrono.ChFrameD(chrono.ChVector3d(interaxis12, 0, -2), chrono.Q_from_AngX(-m.pi / 2)))
link_gearAB.SetTransmissionRatio(radA / radB)
sys.AddLink(link_gearAB)

# Create the gear constraint between second gear B and a large wheel C with inner teeth
radC = 2 * radB + radA
link_gearBC = chrono.ChLinkGear()
link_gearBC.Initialize(mbody_gearB, mbody_truss, True,  # True = internal gear
                       chrono.ChFrameD(chrono.ChVector3d(interaxis12, 0, -2), chrono.Q_from_AngX(-m.pi / 2)),
                       chrono.ChFrameD(chrono.ChVector3d(0, 0, -4), chrono.QUNIT))
link_gearBC.SetTransmissionRatio(radB / radC)
sys.AddLink(link_gearBC)

# ------------------- NEW: Add Bevel Gear D -------------------
radD = 5
mbody_gearD = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, radD, 0.6, 1000, True, False, mat)
sys.Add(mbody_gearD)
mbody_gearD.SetPos(chrono.ChVector3d(-10, 0, -9))
# Rotate by 90 deg around Z: Q = Q_from_AngZ(pi/2)
mbody_gearD.SetRot(chrono.Q_from_AngZ(m.pi / 2))
# Visual material
vis_matD = chrono.ChVisualMaterial()
vis_matD.SetDiffuseColor(chrono.ChColor(0.2, 0.7, 0.2))
mbody_gearD.GetVisualShape(0).SetMaterial(0, vis_matD)

# Revolute joint for gear D (horizontal axis: X)
link_revoluteD = chrono.ChLinkLockRevolute()
link_revoluteD.Initialize(mbody_gearD, mbody_truss,
                          chrono.ChFrameD(chrono.ChVector3d(-10, 0, -9), chrono.Q_from_AngY(m.pi / 2)))
sys.AddLink(link_revoluteD)

# Gear constraint between A and D (1:1, bevel gear)
# For bevel gears, use ChLinkGear with axes at 90 deg
link_gearAD = chrono.ChLinkGear()
# Frame1: on gearA, axis Y (vertical), at (0,0,-1)
# Frame2: on gearD, axis X (horizontal), at (-10,0,-9)
frameA = chrono.ChFrameD(chrono.ChVector3d(0, 0, -1), chrono.Q_from_AngX(-m.pi / 2))
frameD = chrono.ChFrameD(chrono.ChVector3d(-10, 0, -9), chrono.Q_from_AngY(m.pi / 2))
link_gearAD.Initialize(mbody_gearA, mbody_gearD, False, frameA, frameD)
link_gearAD.SetTransmissionRatio(1.0)
sys.AddLink(link_gearAD)

# ------------------- NEW: Add Pulley E -------------------
radE = 2
mbody_pulleyE = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, radE, 0.3, 500, True, False, mat)
sys.Add(mbody_pulleyE)
mbody_pulleyE.SetPos(chrono.ChVector3d(-10, -11, -9))
# Rotate by 90 deg around Z
mbody_pulleyE.SetRot(chrono.Q_from_AngZ(m.pi / 2))
# Visual material
vis_matE = chrono.ChVisualMaterial()
vis_matE.SetDiffuseColor(chrono.ChColor(0.7, 0.2, 0.7))
mbody_pulleyE.GetVisualShape(0).SetMaterial(0, vis_matE)

# Revolute joint for pulley E (horizontal axis: X)
link_revoluteE = chrono.ChLinkLockRevolute()
link_revoluteE.Initialize(mbody_pulleyE, mbody_truss,
                          chrono.ChFrameD(chrono.ChVector3d(-10, -11, -9), chrono.Q_from_AngY(m.pi / 2)))
sys.AddLink(link_revoluteE)

# ------------------- NEW: Pulley constraint (synchro belt) between D and E -------------------
# Use ChLinkPulley for a synchronous belt
link_pulley = chrono.ChLinkPulley()
# Pulley1: gear D, at (-10, 0, -9), axis X
# Pulley2: pulley E, at (-10, -11, -9), axis X
frameD_pulley = chrono.ChFrameD(chrono.ChVector3d(-10, 0, -9), chrono.Q_from_AngY(m.pi / 2))
frameE_pulley = chrono.ChFrameD(chrono.ChVector3d(-10, -11, -9), chrono.Q_from_AngY(m.pi / 2))
link_pulley.Initialize(mbody_gearD, mbody_pulleyE, frameD_pulley, frameE_pulley)
link_pulley.SetR1(radD)
link_pulley.SetR2(radE)
link_pulley.SetTransmissionRatio(-radD / radE)  # Negative for crossed belt, positive for open
sys.AddLink(link_pulley)

# ------------------- Visualization: Draw a simple belt between D and E -------------------
# We'll add a visual shape (thin cylinder) between the two pulley centers
belt_radius = 0.15
belt_length = abs(mbody_pulleyE.GetPos().y - mbody_gearD.GetPos().y)
belt_center = chrono.ChVector3d(-10, -5.5, -9)
belt_shape = chrono.ChVisualShapeCylinder(belt_radius, belt_length)
belt_shape.SetColor(chrono.ChColor(0.2, 0.2, 0.8))
# The belt is along Y axis, so no rotation needed
mbody_truss.AddVisualShape(belt_shape, chrono.ChFrameD(belt_center, chrono.QUNIT))

# ------------------- Visualization System -------------------
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Gears and pulleys')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(12, 15, -20))
vis.AddTypicalLights()

# Set integrator type
sys.SetTimestepperType(chrono.ChTimestepper.Type_EULER_IMPLICIT_PROJECTED)

# Simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(1e-3)
```

- **Error Fixes:**

    - `chrono.QuatFromAngleX` and similar are not valid; replaced with `chrono.Q_from_AngX`, `Q_from_AngY`, `Q_from_AngZ`.
    - `ChLinkLockGear` is deprecated; use `ChLinkGear` for gear constraints.
    - `SetFrameShaft1`, `SetFrameShaft2`, `SetEnforcePhase` are not valid for `ChLinkGear`.
    - `ChVisualShapeCylinder` expects (radius, length).
- **Bevel Gear D:**
    - Added as a cylinder, rotated by 90° around Z.
    - Linked to truss with a revolute joint along X.
    - Gear constraint with A, 1:1 ratio, axes at 90°.
- **Pulley E:**
    - Added as a cylinder, rotated by 90° around Z.
    - Linked to truss with a revolute joint along X.
    - Synchronous belt constraint with D using `ChLinkPulley`.
- **Belt Visualization:**
    - Simple cylinder between D and E.
- **Materials:**
    - Added distinct colors for new parts.
