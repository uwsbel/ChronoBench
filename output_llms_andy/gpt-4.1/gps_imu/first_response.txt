Certainly! Below is a **PyChrono** simulation script that sets up an HMMWV vehicle with terrain, driver, Irrlicht visualization, and attaches both IMU and GPS sensors to the chassis. The script processes sensor data at each step, outputs the vehicle mass, and renders the scene.

**Note:**  
- This script assumes you have PyChrono and its modules (`chrono`, `chrono_vehicle`, `chrono_sensor`, `chrono_irrlicht`) installed.
- Adjust paths and parameters as needed for your environment.

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import pychrono.sensor as sens
import math

# 1. Initialize PyChrono and set up the simulation
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

# 2. Create the physical system
system = chrono.ChSystemSMC()

# 3. Create the HMMWV vehicle
initLoc = chrono.ChVectorD(0, 0, 1.0)
initRot = chrono.QUNIT
vehicle = veh.HMMWV_Full(system)
vehicle.SetContactMethod(chrono.ChContactMethod_SMC)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysD(initLoc, initRot))
vehicle.Initialize()

# 4. Create the terrain
terrain = veh.RigidTerrain(system)
patch = terrain.AddPatch(chrono.CSYSNORM, 200, 200)
patch.SetContactFrictionCoefficient(0.9)
patch.SetRestitutionCoefficient(0.01)
patch.SetMaterialSurface(chrono.ChMaterialSurfaceSMC())
patch.SetTexture(chrono.GetChronoDataFile("terrain/textures/grass.jpg"), 200, 200)
patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
terrain.Initialize()

# 5. Create the driver system (simple path-following driver)
path = veh.StraightLinePath(chrono.ChVectorD(0, 0, 1.0), chrono.ChVectorD(100, 0, 1.0), 1)
target_speed = 10.0  # m/s
driver = veh.ChPathFollowerDriver(vehicle, path, "my_path", target_speed)
driver.Initialize()

# 6. Create the Irrlicht visualization
vis = veh.ChWheeledVehicleIrrApp(vehicle, 'HMMWV Sensor Demo', irr.dimension2du(1024,768))
vis.SetSkyBox()
vis.AddTypicalLights()
vis.AddTypicalLogo()
vis.AddTypicalCamera(chrono.ChVectorD(0, 2, 1.75), chrono.ChVectorD(0, 0, 1.0))
vis.SetChaseCamera(chrono.ChVectorD(0.0, 0.0, 1.75), 6.0, 0.5)
vis.AssetBindAll()
vis.AssetUpdateAll()

# 7. Set up the sensor manager
manager = sens.ChSensorManager(system)
manager.scene.AddPointLight(chrono.ChVectorF(100, 100, 100), chrono.ChVectorF(1, 1, 1), 500)

# 8. Add IMU sensor to the chassis
imu_offset = chrono.ChFrameD(chrono.ChVectorD(0, 0, 1.2))
imu = sens.ChIMUSensor(
    vehicle.GetChassisBody(),  # body to which the sensor is attached
    imu_offset,                # offset pose
    100.0,                     # update rate [Hz]
    chrono.ChVectorD(0),       # noise stddev (no noise)
    sens.IMUNoiseModel_NONE
)
imu.PushFilter(sens.ChFilterIMUUpdate())
manager.AddSensor(imu)

# 9. Add GPS sensor to the chassis
gps_offset = chrono.ChFrameD(chrono.ChVectorD(0, 0, 1.5))
gps = sens.ChGPSSensor(
    vehicle.GetChassisBody(),
    gps_offset,
    10.0,                      # update rate [Hz]
    chrono.ChVectorD(0),       # noise stddev (no noise)
    sens.GPSNoiseModel_NONE
)
gps.PushFilter(sens.ChFilterGPSUpdate())
manager.AddSensor(gps)

# 10. Simulation loop parameters
step_size = 2e-3
render_steps = int(1.0 / (60 * step_size))  # render at ~60 FPS
sim_time = 0
frame = 0

print("Vehicle mass:", vehicle.GetVehicleMass(), "kg")

while vis.Run():
    # 1. Get driver inputs
    driver_inputs = driver.GetInputs()

    # 2. Synchronize modules
    driver.Synchronize(sim_time)
    terrain.Synchronize(sim_time)
    vehicle.Synchronize(sim_time, driver_inputs, terrain)
    vis.Synchronize("HMMWV Demo", driver_inputs)

    # 3. Advance simulation for all modules
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    # 4. Update sensors
    manager.Update()

    # 5. Process IMU data
    imu_data = imu.GetMostRecentBuffer()
    if imu_data:
        acc = imu_data.Acc
        ang_vel = imu_data.AngVel
        print(f"[{sim_time:.2f}s] IMU Acc: {acc.x:.2f}, {acc.y:.2f}, {acc.z:.2f} | AngVel: {ang_vel.x:.2f}, {ang_vel.y:.2f}, {ang_vel.z:.2f}")

    # 6. Process GPS data
    gps_data = gps.GetMostRecentBuffer()
    if gps_data:
        pos = gps_data.Position
        print(f"[{sim_time:.2f}s] GPS Position: {pos.x:.2f}, {pos.y:.2f}, {pos.z:.2f}")

    # 7. Render scene
    if frame % render_steps == 0:
        vis.BeginScene()
        vis.DrawAll()
        vis.EndScene()

    # 8. Advance time
    sim_time += step_size
    frame += 1

    # Optional: stop after some time
    if sim_time > 10.0:
        break

# Output vehicle mass at the end
print("Final vehicle mass:", vehicle.GetVehicleMass(), "kg")
```

**Key Points:**
- The script initializes PyChrono, creates the vehicle, terrain, driver, and visualization.
- IMU and GPS sensors are attached to the chassis and updated each step.
- Sensor data is printed to the console.
- The simulation loop synchronizes and advances all modules, renders the scene, and outputs vehicle mass.
