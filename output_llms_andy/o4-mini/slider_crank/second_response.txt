Below is a single, self‐contained PyChrono–Python script that

1. fixes the few API‐usage errors in your original (wrong constructor arguments, wrong function names, missing quaternion utilities, etc.),  
2. adds the 4 arrays and appends data on every step,  
3. breaks the loop after 20 s,  
4. and finally produces two Matplotlib subplots (position vs. crank angle and speed vs. crank angle) with π‐based x‐ticks.

You should be able to run this as‐is (assuming you have PyChrono, Irrlicht and Matplotlib installed).

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import matplotlib.pyplot as plt
import numpy as np

# -----------------------------------------------------------------------------
# 1) Set up Chrono system
# -----------------------------------------------------------------------------
sys = chrono.ChSystemNSC()

# geometry & mass parameters
crank_center = chrono.ChVectorD(-1, 0.5, 0)   # hinge location of crank
crank_rad    = 0.4
crank_thick  = 0.1
rod_length   = 1.5

# -- Floor (fixed support) -----------------------------------------------
mfloor = chrono.ChBodyEasyBox(3, 1, 3,            # x,y,z size
                              1000,               # density
                              True, True)         # visualize / collide
mfloor.SetPos(chrono.ChVectorD(0, -0.5, 0))
mfloor.SetBodyFixed(True)
sys.Add(mfloor)

# -- Crank (cylinder along Z by default) --------------------------------
# ChBodyEasyCylinder(radius, length, density, visualize, collide)
mcrank = chrono.ChBodyEasyCylinder(crank_rad,
                                   crank_thick,
                                   1000,
                                   True, True)
# place it so its axis of rotation goes through crank_center
mcrank.SetPos(crank_center)
# no extra SetRot() needed: default cylinder axis is body‐Z, our motor
# rotates about Z so everything lines up.
sys.Add(mcrank)

# -- Connecting rod (box along X) ----------------------------------------
mrod = chrono.ChBodyEasyBox(rod_length, 0.1, 0.1,
                            1000, True, True)
# put center of rod at (crank_center.x + crank_rad + rod_length/2, y,z)
mrod.SetPos(crank_center + chrono.ChVectorD(crank_rad + rod_length/2, 0, 0))
sys.Add(mrod)

# -- Piston (also cylinder along Z, sliding in X) ------------------------
mpiston = chrono.ChBodyEasyCylinder(0.2,
                                    0.3,
                                    1000,
                                    True, True)
mpiston.SetPos(crank_center + chrono.ChVectorD(crank_rad + rod_length, 0, 0))
sys.Add(mpiston)

# -----------------------------------------------------------------------------
# 2) Add motor + joints
# -----------------------------------------------------------------------------
# -- Motor to spin the crank about Z at constant speed π rad/s -----------
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(mcrank,            # crank is "body A"
                 mfloor,            # floor is "body B"
                 chrono.ChFrameD(crank_center))   # absolute frame of motor
# constant-speed function
motor.SetSpeedFunction(chrono.ChFunction_Const(chrono.CH_C_PI))
sys.Add(motor)

# -- Revolute joint crank--rod at the crank pin --------------------------
jointA = chrono.ChLinkLockRevolute()
jointA.Initialize(mcrank,
                  mrod,
                  chrono.ChFrameD(crank_center + chrono.ChVectorD(crank_rad, 0, 0)))
sys.Add(jointA)

# -- Revolute joint rod--piston at the small end -------------------------
jointB = chrono.ChLinkLockRevolute()
jointB.Initialize(mrod,
                  mpiston,
                  chrono.ChFrameD(crank_center + chrono.ChVectorD(crank_rad + rod_length, 0, 0)))
sys.Add(jointB)

# -- Prismatic joint piston--floor to slider along X --------------------
# default prismatic slides along the local Z axis of the joint frame,
# so we rotate that Z into global X by +90° about the Y axis.
q_z2x = chrono.Q_from_AngAxis(chrono.CH_C_PI/2, chrono.ChVectorD(0, 1, 0))
jointC = chrono.ChLinkLockPrismatic()
jointC.Initialize(mfloor,
                  mpiston,
                  chrono.ChFrameD(crank_center + chrono.ChVectorD(crank_rad + rod_length, 0, 0),
                                  q_z2x))
sys.Add(jointC)

# -----------------------------------------------------------------------------
# 3) Initialize Irrlicht visualization
# -----------------------------------------------------------------------------
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Crank–Slider Demo')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(1, 1, 3),
              chrono.ChVectorD(0, 1, 0))
vis.AddTypicalLights()

# -----------------------------------------------------------------------------
# 4) Prepare arrays for data collection
# -----------------------------------------------------------------------------
array_time  = []
array_angle = []
array_pos   = []
array_speed = []

t_end = 20.0   # seconds
dT    = 1e-3   # time step

# -----------------------------------------------------------------------------
# 5) Simulation loop: collect data, stop at t_end
# -----------------------------------------------------------------------------
while vis.Run():
    t = sys.GetChTime()
    if t > t_end:
        break

    # record
    array_time.append(t)
    # motor rotation angle about Z:
    angle = motor.GetMotorRot()    # [rad]
    array_angle.append(angle)
    # piston X position & speed:
    pos   = mpiston.GetPos().x
    speed = mpiston.GetPos_dt().x
    array_pos.append(pos)
    array_speed.append(speed)

    # do one step
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(dT)

# optional: close the Irrlicht window
vis.GetIrrlichtDevice().closeDevice()

# -----------------------------------------------------------------------------
# 6) Post‐processing & plotting with Matplotlib
# -----------------------------------------------------------------------------
array_angle = np.array(array_angle)
array_pos   = np.array(array_pos)
array_speed = np.array(array_speed)

fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(8, 6), tight_layout=True)

# -- 1st subplot: piston position vs. crank angle -----------
ax1.plot(array_angle, array_pos, 'b-')
ax1.set_xlabel('Crank angle [rad]')
ax1.set_ylabel('Piston pos [m]')
ax1.grid(True)

# -- 2nd subplot: piston speed vs. crank angle -------------
ax2.plot(array_angle, array_speed, 'r-')
ax2.set_xlabel('Crank angle [rad]')
ax2.set_ylabel('Piston speed [m/s]')
ax2.grid(True)

# -- customize x‐ticks in π/2 increments --------------------
# we'll show from 0 to 2π
ticks = np.arange(0, 2 * np.pi + 1e-6, np.pi/2)
labels = ['0', r'$\frac{\pi}{2}$', r'$\pi$', r'$\frac{3\pi}{2}$', r'$2\pi$']
ax1.set_xticks(ticks)
ax1.set_xticklabels(labels)
ax2.set_xticks(ticks)
ax2.set_xticklabels(labels)

plt.show()
```

Explanation of the main fixes and changes:

1. **Fixed constructors**:
   - `ChBodyEasyCylinder` takes `(radius, length, density, visualize, collide)`—no axis argument.
   - Provided the two boolean flags `True, True` for visualization and collision.

2. **Corrected joint and motor initialization**:
   - Used `chrono.ChFrameD(...)` instead of a non‐existent `ChFramed`.
   - Used `motor.SetSpeedFunction(...)` (not `SetMotorFunction`).

3. **Prismatic joint orientation**:
   - Built a quaternion `q_z2x = Q_from_AngAxis(pi/2, Y)` to rotate the joint’s local Z‐axis into the global X‐axis (so the piston truly slides in X).

4. **Data arrays & collection**:
   - Declared four Python lists before the loop.
   - In each step appended time, crank angle, piston X, and piston X‐velocity.

5. **Stop after 20 s**:
   - Checked `if t > t_end: break` inside the loop.

6. **Matplotlib plotting**:
   - Created two subplots.
   - Set x‐axis ticks at `[0, π/2, π, 3π/2, 2π]` with LaTeX‐style labels.

This script should run interactively, then at the end pop up the Matplotlib window with your two desired plots.
