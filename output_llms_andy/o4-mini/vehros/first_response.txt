```python
# PyChrono HMMWV + ROS Integration Example Script

import pychrono.core as chrono
import pychrono.vehicle as veh

# Assume PyChrono-ROS bindings are available under pychrono.ros2
import pychrono.ros2 as ros2

# -----------------------------------------------------------------------------
# 1. INITIALIZE CHRONO & SET DATA PATH
# -----------------------------------------------------------------------------
chrono.SetChronoDataPath("/path/to/chrono/data/")     # adjust to your installation

# -----------------------------------------------------------------------------
# 2. CREATE THE HMMWV VEHICLE
# -----------------------------------------------------------------------------
#  Choose contact method, engine, driveline, transmission, and tire models
contact_method = chrono.ChMaterialSurface.SMC

vehicle = veh.HMMWV_Vehicle(contact_method=contact_method,
                            engine_type=veh.EngineModelType_SimpleMap,
                            transmission_type=veh.TransmissionModelType_SimpleMap,
                            tire_type=veh.TireModelType_Fiala)

# Initialize at origin, 0° yaw, 1.0 m above ground
init_loc = chrono.ChVectorD(0, 0, 1.0)
init_rot = chrono.Q_from_AngX(0)  # no initial rotation
vehicle.Initialize(chrono.ChCoordsysD(init_loc, init_rot))

# Set some vehicle parameters (optional)
vehicle.SetChassisVisualizationType(veh.VisualizationType_Primitive)
vehicle.SetSuspensionVisualizationType(veh.VisualizationType_Primitive)
vehicle.SetSteeringVisualizationType(veh.VisualizationType_Primitive)
vehicle.SetWheelVisualizationType(veh.VisualizationType_Primitive)

# -----------------------------------------------------------------------------
# 3. CREATE RIGID TERRAIN
# -----------------------------------------------------------------------------
terrain = veh.RigidTerrain(vehicle.GetSystem())

# Add a flat patch
patch = terrain.AddPatch(chrono.ChCoordsysD(chrono.ChVectorD(0,0,0), chrono.QUNIT),
                         chrono.ChVectorD(20, 20, 0),  # patch half-sizes
                         0.8,                            # friction coefficient
                         0.1,                            # restitution
                         2e7,                            # Young's modulus
                         0.3)                            # Poisson ratio

patch.SetColor(chrono.ChColor(0.3, 0.3, 0.3))
patch.SetTexture(chrono.GetChronoDataFile("concrete.jpg"), 4, 4)
terrain.Initialize()

# -----------------------------------------------------------------------------
# 4. CREATE DRIVER (no GUI; inputs come from ROS topics)
# -----------------------------------------------------------------------------
driver = veh.ChDriver(vehicle)
driver.SetInputDelay(0.1)  # seconds

# -----------------------------------------------------------------------------
# 5. SET UP ROS2 MANAGER & HANDLERS
# -----------------------------------------------------------------------------
ros_node_name = "hmmwv_sim_node"
ros_manager = ros2.ChRosManager(ros_node_name)

# 5a) Clock synchronization publisher
def clock_publisher():
    msg = ros2.ClockMsg()
    msg.clock = vehicle.GetSystem().GetChTime()
    return msg

ros_manager.CreatePublisher("/clock", ros2.ClockMsg, clock_publisher,
                            topic_qos=ros2.QoSProfile(depth=1, reliable=True))

# 5b) Driver input subscriber
#    Expect a custom DriverInputMsg with fields: throttle, steering, braking
def driver_input_callback(msg):
    driver.SetThrottle(msg.throttle)
    driver.SetSteering(msg.steering)
    driver.SetBraking(msg.braking)

ros_manager.CreateSubscriber("/driver/input",
                             ros2.DriverInputMsg,
                             driver_input_callback,
                             topic_qos=ros2.QoSProfile(depth=1, reliable=True))

# 5c) Vehicle state publisher
def vehicle_state_publisher():
    state = vehicle.ExportStateJSON()  # assume convenience method
    msg = ros2.VehicleStateMsg()
    msg.json_state = state
    return msg

ros_manager.CreatePublisher("/vehicle/state",
                            ros2.VehicleStateMsg,
                            vehicle_state_publisher,
                            topic_qos=ros2.QoSProfile(depth=1, reliable=True))

# Start ROS spinning in background
ros_manager.Start()

# -----------------------------------------------------------------------------
# 6. SIMULATION LOOP
# -----------------------------------------------------------------------------
t_end      = 10.0        # simulation end time
step_size  = 1.0/100.0   # fixed integration step

system = vehicle.GetSystem()
time   = 0.0

while time < t_end:
    # 6a) Spin ROS once to process incoming inputs & publish clock
    ros_manager.SpinOnce()

    # 6b) Synchronize all modules to current time
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver.GetInputs(), terrain.GetForces())
    ros_manager.PublishAll()  # publish any registered topics

    # 6c) Advance all modules by one timestep
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)

    # 6d) Increment time
    time += step_size

# Stop ROS & clean up
ros_manager.Shutdown()
```

Explanation of key sections:

1. **Initialization**  
   - We set the Chrono data path for access to meshes/textures.  
   - We choose SMC contact, a simple‐map engine/transmission, and Fiala tires.

2. **Terrain**  
   - A rigid patch with specified friction, restitution, elasticity, and texture.

3. **Driver**  
   - A Chrono driver object whose inputs will be updated by ROS messages.

4. **ROS Integration**  
   - `ChRosManager` is used to create publishers/subscribers.  
   - We publish the simulation clock on `/clock`, subscribe to driver inputs on `/driver/input`, and publish vehicle state on `/vehicle/state`.  

5. **Simulation Loop**  
   - Each iteration: process ROS messages, synchronize Chrono modules to the current time, publish data, advance all systems by a fixed timestep, and increment simulation time.  

Ensure you have the PyChrono-ROS bindings installed and that your message definitions (`ClockMsg`, `DriverInputMsg`, `VehicleStateMsg`) match the fields used here.
