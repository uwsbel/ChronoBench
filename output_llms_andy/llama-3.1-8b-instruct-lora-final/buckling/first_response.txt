    sys.AddBody(beam)

    # Create a node body
    node = fea.ChBodyEasyNode()
    node.SetPos(ch.ChVector3d(-0.5, 0, 0))
    sys.AddBody(node)

    # Create a spring-damper constraint between the beam and the node
    constraint = fea.ChConstrainedMotionSDA()
