```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import math

# Initialize the Chrono simulation system
sys = chrono.ChSystemNSC()

# Set the gravitational acceleration for the system (in m/s^2)
sys.SetGravitationalAcceleration(chrono.ChVector3d(0, -9.81, 0))  # g = 9.81 m/s^2

# Create the ground body and add it to the simulation
ground = chrono.ChBody()
sys.Add(ground)
ground.SetFixed(True)  # The ground is fixed and does not move
ground.EnableCollision(False)  # Disable collision detection for the ground

# Add a visualization cylinder to the ground
cyl_ground_vis = chrono.ChVisualShapeCylinder(0.2, 0.4)  # Cylinder with radius 0.2 and height 0.4
ground.AddVisualShape(cyl_ground_vis, chrono.ChFrameD(chrono.ChVector3d(0, 0, 0)))

# Create the first pendulum body and add it to the simulation
pend_1 = chrono.ChBody()
sys.Add(pend_1)
pend_1.SetFixed(False)  # The pendulum can move
pend_1.EnableCollision(False)  # Disable collision detection for the pendulum
pend_1.SetMass(1)  # Set the mass of the pendulum (in kg)
pend_1.SetInertiaXX(chrono.ChVector3d(0.2, 0.2, 0.2))  # Set the inertia tensor (approximate)

# Add a visualization cylinder to the first pendulum
cyl_pend1_vis = chrono.ChVisualShapeCylinder(0.2, 2)  # Radius 0.2, height 2
cyl_pend1_vis.SetColor(chrono.ChColor(0.6, 0, 0))  # Set the color of the cylinder (RGB)
pend_1.AddVisualShape(cyl_pend1_vis, chrono.ChFrameD(chrono.ChVector3d(0, -1, 0)))

# Set the initial position of the first pendulum (center of mass) in the absolute frame
# The pendulum is initially hanging downward from the origin
pend_1.SetPos(chrono.ChVector3d(0, -1, 0))
# Also set initial orientation if needed (default is identity)

# Create the second pendulum body and add it to the simulation
pend_2 = chrono.ChBody()
sys.Add(pend_2)
pend_2.SetFixed(False)
pend_2.EnableCollision(False)
pend_2.SetMass(1)
pend_2.SetInertiaXX(chrono.ChVector3d(0.2, 0.2, 0.2))

# Add visualization for the second pendulum
cyl_pend2_vis = chrono.ChVisualShapeCylinder(0.2, 2)
cyl_pend2_vis.SetColor(chrono.ChColor(0, 0.6, 0))
pend_2.AddVisualShape(cyl_pend2_vis, chrono.ChFrameD(chrono.ChVector3d(0, -1, 0)))

# Position the second pendulum initially relative to the first pendulum
# It will be attached at the end of the first pendulum
# First, set the position of the second pendulum to be at the end of the first pendulum
pend_2.SetPos(chrono.ChVector3d(0, -3, 0))  # 2 units below the first pendulum's center

# Create a revolute joint to connect pend_1 to the ground at the pivot point
rev_ground_to_pend1 = chrono.ChLinkLockRevolute()
pivot_point_1 = chrono.ChVector3d(0, 0, 0)  # Pivot at origin
rev_ground_to_pend1.Initialize(ground, pend_1, chrono.ChFrameD(pivot_point_1, chrono.Q_from_AngY(0)))
sys.AddLink(rev_ground_to_pend1)

# Create a revolute joint to connect pend_2 to pend_1 at the joint point
rev_pend1_to_pend2 = chrono.ChLinkLockRevolute()
joint_point = chrono.ChVector3d(0, -1, 0)  # End of pend_1 (center of pend_1 + offset)
rev_pend1_to_pend2.Initialize(pend_1, pend_2, chrono.ChFrameD(joint_point, chrono.Q_from_AngY(0)))
sys.AddLink(rev_pend1_to_pend2)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)  # Attach the Chrono system to the visualization
vis.SetWindowSize(1024, 768)  # Set the window size for the visualization
vis.SetWindowTitle('Double Pendulum Demo')  # Set the window title
vis.Initialize()  # Initialize the visualization system
# Note: AddLogo may require a valid file path; comment out if not available
# vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()  # Add a skybox for better visual appearance
# Position the camera to view the system nicely
vis.AddCamera(chrono.ChVector3d(0, 3, 6))
vis.AddTypicalLights()  # Add typical lights for better visualization

# Simulation loop
log_info = True  # Flag to control logging
while vis.Run():
    vis.BeginScene()  # Begin the scene for rendering
    vis.Render()  # Render the scene
    vis.EndScene()  # End the scene
    sys.DoStepDynamics(1e-3)  # Advance the simulation by a time step of 0.001 seconds

    # Log information after 1 second of simulation time
    if log_info and sys.GetChTime() > 1:
        pos_1 = pend_1.GetPos()
        print("t = ", sys.GetChTime())
        print("Pendulum 1 position: ", pos_1.x, pos_1.y, pos_1.z)
        lin_vel_1 = pend_1.GetPosDt()
        print("Pendulum 1 velocity: ", lin_vel_1.x, lin_vel_1.y, lin_vel_1.z)
        pos_2 = pend_2.GetPos()
        print("Pendulum 2 position: ", pos_2.x, pos_2.y, pos_2.z)
        lin_vel_2 = pend_2.GetPosDt()
        print("Pendulum 2 velocity: ", lin_vel_2.x, lin_vel_2.y, lin_vel_2.z)
        log_info = False  # Disable further logging
```
