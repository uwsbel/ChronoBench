```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.viper as viper

# Initialize the PyChrono environment
chrono.Chrono.SetChronoDataPath('/path/to/chrono/data/')

# Create the Chrono system
sys = chrono.ChSystemNSC()
sys.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))  # Set gravity

# Create the ground body
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, False)
ground.SetBodyFixed(True)
ground.SetPos(chrono.ChVectorD(0, 0, -0.5))
sys.Add(ground)

# Initialize the Viper rover and its driver
rover = viper.Viper(sys)
rover.SetDriver(viper.ViperDCMotorControl(rover.GetMotor(0), rover.GetMotor(1), rover.GetMotor(2), rover.GetMotor(3)))

# Create a real-time visualization using Irrlicht
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle('Viper Rover Simulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(2, 2, 1), chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVectorD(1.5, 1.5, 5.5), chrono.ChVectorD(0, 0, 0), 10, 1, 10, 55)

# Simulation loop parameters
time_step = 0.01
time_end = 10
steering_time = 5

# Simulation loop
for time in chrono.ChFunction_Ramp(0, time_step).Get_y()[:int(time_end / time_step)]:
    if time > steering_time:
        rover.GetDriver().SetSteering(-0.6)
    else:
        rover.GetDriver().SetSteering(0.6 * time / steering_time)

    sys.DoStepDynamics(time_step)
    vis.Run()
    vis.Render()
    vis.BeginScene(True, True, chrono.ChColor(0.2, 0.2, 0.2))
    vis.EndScene()

vis.GetDevice().closeDevice()
```
