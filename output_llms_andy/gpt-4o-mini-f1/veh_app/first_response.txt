```python
import pychrono as ch
import pychrono.vehicle as veh
import pychrono.sensor as sens
import math

# Initial vehicle location and orientation
initLoc = ch.ChVector3d(0, 0, 0.5)
initRot = ch.ChQuaterniond(1, 0, 0, 0)

# Visualization type for vehicle parts (PRIMITIVES, MESH, or NONE)
vis_type = veh.VisualizationType_MESH

# Collision type for chassis (PRIMITIVES, MESH, or NONE)
chassis_collision_type = veh.CollisionType_NONE

# Type of tire model (RIGID, TMEASY)
tire_model = veh.TireModelType_TMEASY

# Rigid terrain
# terrain_model = veh.RigidTerrain.BOX
terrainHeight = 0      # terrain height
terrainLength = 100.0  # size in X direction
terrainWidth = 100.0   # size in Y direction

# Poon chassis tracked by the camera
trackPoint = ch.ChVector3d(-3.0, 0.0, 1.1)

# Contact method
contact_method = ch.ChContactMethod_NSC
contact_vis = False

# Simulation step sizes
step_size = 1e-3
tire_step_size = step_size

# Time interval between two render frames
render_step_size = 1.0 / 50  # FPS = 50

# ---------------
# Create systems
# ---------------

# Create the Gator vehicle, set parameters, and initialize
vehicle = veh.Gator()
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(ch.ChCoordsysd(initLoc, initRot))
vehicle.SetTireType(tire_model)
vehicle.SetTireStepSize(tire_step_size)
vehicle.SetMaxNumConsecutiveBadTireContacts(20)

vehicle.Initialize()

vehicle.SetChassisVisualizationType(vis_type)
vehicle.SetSuspensionVisualizationType(vis_type)
vehicle.SetSteeringVisualizationType(vis_type)
vehicle.SetWheelVisualizationType(vis_type)
vehicle.SetTireVisualizationType(vis_type)

vehicle.GetSystem().SetCollisionSystemType(ch.ChCollisionSystem.Type_BULLET)

# Create the terrain
patch_mat = ch.ChContactMaterialNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)
terrain = veh.RigidTerrain(vehicle.GetSystem())
patch = terrain.AddPatch(patch_mat, 
    ch.ChCoordsysd(ch.ChVector3d(0, 0, 0), ch.QUNIT), 
    terrainLength, terrainWidth)

patch.SetTexture(r'../../data/terrain/textures/tile4.jpg', 200, 200)
patch.SetColor(ch.ChColor(0.8, 0.8, 0.5))
terrain.Initialize()

# Create the driver system
driver = veh.ChInteractiveDriverIRB(vehicle.GetSystem())

# Create the sensor manager
manager = sens.ChSensorManager(vehicle.GetSystem())

# Create the point light source
light = sens.ChPointLightSource(pos=ch.ChVector3f(0, 0, 10), 
                                color=ch.ChColor(1, 1, 1), 
                                intensity=1.0)
light.SetAttenuation(1, 0, 0)  # linear, quadratic
light.SetRenderSize(512, 512)
manager.scene.AddPointLightSource(light)

# Create the camera
offset_pose = ch.ChFramed(ch.ChVector3d(0, 0, 2.5), ch.QuatFromAngleAxis(0, ch.ChVector3d(0, 1, 0)))
render_camera = sens.ChCameraSensor(vehicle.GetChassisBody(), 
                                    update_rate=30, 
                                    offset_pose=offset_pose)
render_camera.SetName("Camera")
render_camera.SetCollectionWindowSize(512, 512)
render_camera.SetFOV(math.radians(60))
render_camera.SetLag(0)
render_camera.PushFilter(sens.ChFilterVisualize())
manager.AddSensor(render_camera)

# Create the IMU
offset_pose = ch.ChFramed(ch.ChVector3d(0, 0, 0), ch.QuatFromAngleAxis(0, ch.ChVector3d(0, 1, 0)))
imu = sens.ChIMUSensor(vehicle.GetChassisBody(), 
                       update_rate=20, 
                       offset_pose=offset_pose)
imu.SetName("IMU")
imu.SetLag(0)
imu.PushFilter(sens.ChFilterDIAccess())
manager.AddSensor(imu)

# ---------------
# Simulation loop
# ---------------

# output vehicle mass
print( "VEHICLE MASS: ",  vehicle.GetVehicle().GetMass())

# Number of simulation steps between miscellaneous events
render_steps = math.ceil(render_step_size / step_size)

# Initialize simulation frame counter s
realtime_timer = ch.ChRealtimeStepTimer()
step_number = 0
render_frame = 0

while (True):
    time = vehicle.GetSystem().GetChTime()

    # Render scene and output POV-Ray data
    if (step_number % render_steps == 0):
        # manager.Render()
        pass

    # Get the latest camera image from the sensor manager
    # image = manager.GetLastImage(render_camera)
    # if image is not None:
    #     image.SavePNG("image_" + str(render_frame) + ".png")
    #     render_frame += 1

    # Update driver inputs
    driver_inputs = driver.GetInputs()
    driver_inputs = driver.GetInputs()
    driver.SetSteering(0.0)
    driver.SetThrottle(0.0)
    driver.SetBraking(0.0)
    driver.Update()

    # Update modules (process inputs from other modules)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    manager.Synchronize(time)

    # Advance simulation for one timestep for all modules
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    manager.Advance(step_size)

    # Increment frame number
    step_number += 1

    # Spin in place for real time to catch up
    realtime_timer.Spin(step_size)
```
