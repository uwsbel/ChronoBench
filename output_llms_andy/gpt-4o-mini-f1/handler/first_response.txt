```python
import pychrono as chrono
import pychrono.ros as chros
import rospy
from std_msgs.msg import Int32

class CustomROSHandler(chros.ChROSHandler):
    def __init__(self, topic_name):
        super().__init__()
        self.publisher = rospy.Publisher(topic_name, Int32, queue_size=10)
        self.message = Int32()

    def on_init(self):
        self.message.data = 0

    def on_step(self):
        self.message.data += 1
        self.publisher.publish(self.message)

def main():
    # Initialize the ROS node
    rospy.init_node('chrono_ros_example', anonymous=True)

    # Create the Chrono physical system
    system = chrono.ChSystemNSC()
    system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))
    mat = chrono.ChContactMaterialNSC()
    mat.SetFriction(0.5)
    mat.SetRestitution(0.01)
    system.GetCollisionSystem().SetCollisionEnvelope(0.0025)

    # Create a fixed floor
    floor = chrono.ChBodyEasyBox(10, 10, 1, 1000, True, True, mat)
    floor.SetPos(chrono.ChVector3d(0, 0, -1))
    floor.SetFixed(True)
    floor.SetName("floor")
    floor.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
    system.Add(floor)

    # Create a movable box
    box = chrono.ChBodyEasyBox(1, 1, 1, 1000, True, True, mat)
    box.SetPos(chrono.ChVector3d(0, 0, 2))
    box.SetName("box")
    box.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
    system.Add(box)

    # Create the ROS manager
    ros_manager = chros.ChROSEventManager()
    ros_manager.RegisterHandler(chros.ChROSTransformHandler())
    ros_manager.RegisterHandler(chros.ChROSClockHandler())
    ros_manager.RegisterHandler(chros.ChROSBulletBodyHandler(system))
    custom_handler = CustomROSHandler("chrono_example_topic")
    ros_manager.RegisterHandler(custom_handler)

    # Create the ROS manager
    ros_manager.Initialize()

    # Simulation loop
    time_step = 1e-3
    time = 0
    time_end = 10
    time_step = 1e-3
    time = 0
    time_end = 10
    while time < time_end:
        # Advance the simulation by one timestep
        system.DoStepDynamics(time_step)

        # Update ROS communication
        ros_manager.OnStep(time)

        # Maintain real-time execution
        rospy.Rate(1 / time_step).sleep()
        time += time_step

if __name__ == "__main__":
    main()
```
