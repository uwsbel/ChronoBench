import math

# Create the Chrono simulation system
sys = chrono.ChSystemNSC()
sys.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# Create a fixed body with a triangular mesh
mesh_body = chrono.ChBodyEasyMesh()

mesh_body.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
mesh_body.GetVisualShape(0).SetMesh(chrono.GetChronoDataFile("meshes/triangulated/triangulated.obj"))
sys.Add(mesh_body)
