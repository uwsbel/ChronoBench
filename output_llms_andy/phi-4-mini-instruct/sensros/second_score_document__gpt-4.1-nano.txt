```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Initialize the system
system = chrono.ChSystemNSC()

# Set data path for resources
chrono.SetChronoDataPath('/path/to/chrono/data')

# Create ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
system.Add(ground)

# Create a box body
box_body = chrono.ChBodyEasyBox(1, 1, 1, 1000, True, True, chrono.ChMaterialSurfaceNSC())
box_body.SetPos(chrono.ChVector3d(0, 0.5, 0))
system.Add(box_body)

# Create a sphere body
sphere_body = chrono.ChBodyEasySphere(0.5, 500, True, True, chrono.ChMaterialSurfaceNSC())
sphere_body.SetPos(chrono.ChVector3d(0, 2, 0))
system.Add(sphere_body)

# Create a revolute joint between ground and box
revolute_joint = chrono.ChLinkLockRevolute()
revolute_joint.Initialize(ground, box_body, chrono.ChCoordsysD(chrono.ChVector3d(0, 0.5, 0)))
system.Add(revolute_joint)

# Create a prismatic joint between ground and sphere
prismatic_joint = chrono.ChLinkLockPrismatic()
prismatic_joint.Initialize(ground, sphere_body, chrono.ChCoordsysD(chrono.ChVector3d(0, 2, 0)))
system.Add(prismatic_joint)

# Set up visualization
visualizer = chronoirr.ChVisualSystemIrrlicht()
visualizer.AttachSystem(system)
visualizer.SetWindowSize(800, 600)
visualizer.SetWindowTitle('PyChrono Simulation')
visualizer.Initialize()
visualizer.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
visualizer.AddSkyBox()
visualizer.AddCamera(chrono.ChVector3d(0, 3, 6))
visualizer.AddTypicalLights()

# Simulation loop
time_step = 0.01
while visualizer.Run():
    system.DoStepDynamics(time_step)
    visualizer.BeginScene()
    visualizer.Render()
    visualizer.EndScene()
```
