```
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Set the path to Chrono data directory
chrono.SetChronoDataPath('/usr/local/share/chrono/data/')

# Create the physical system using Non-Smooth Contact (NSC) method
sys = chrono.ChSystemNSC()

# Create a material for contact properties
material = chrono.ChContactMaterialNSC()
material.SetFriction(0.4)
material.SetDampingF(0.2)

# Create a fixed ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
ground.SetCollide(True)
ground.SetMaterialSurface(material)
ground_shape = chrono.ChVisualShapeBox(chrono.ChVectorD(10, 1, 10))
ground.AddVisualShape(ground_shape)
sys.Add(ground)

# Create a dynamic box body
box = chrono.ChBodyEasyBox(1, 1, 1, 1000, True, True, material)
box.SetPos(chrono.ChVectorD(0, 5, 0))
sys.Add(box)

# Create a revolute joint between ground and box at the box's bottom center
joint = chrono.ChLinkLockRevolute()
joint_frame = chrono.ChCoordsysD(chrono.ChVectorD(0, 4.5, 0))
joint.Initialize(ground, box, joint_frame)
sys.Add(joint)

# Setup the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('PyChrono Revolute Joint Example')
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 3, 8))
vis.AddTypicalLights()

# Simulation loop
time_step = 0.01
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(time_step)
```
