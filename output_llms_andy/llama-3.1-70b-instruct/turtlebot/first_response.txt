```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.engine as chronoengine
import numpy as np

# Create the Chrono engine
sys = chrono.ChSystemNSC()

# Set the gravity
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the ground body
ground = chrono.ChBodyEasyBox(sys, 10, 2, 10, 1000, True, chrono.ChVectorD(0, -2, 0), chrono.ChQuaternionD(1, 0, 0, 0))
sys.Add(ground)

# Create the Turtlebot robot
turtlebot = chrono.ChBodyEasyCylinder(sys, 0.3, 0.1, 1000, True, chrono.ChVectorD(0, 0.5, 0), chrono.ChQuaternionD(1, 0, 0, 0))
sys.Add(turtlebot)

# Create the left and right motors
left_motor = chrono.ChLinkMotorRotationSpeed()
left_motor.Initialize(turtlebot, ground, chrono.ChFrameD(chrono.ChVectorD(-0.15, 0.05, 0)))
sys.Add(left_motor)

right_motor = chrono.ChLinkMotorRotationSpeed()
right_motor.Initialize(turtlebot, ground, chrono.ChFrameD(chrono.ChVectorD(0.15, 0.05, 0)))
sys.Add(right_motor)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(800, 600)
vis.SetWindowTitle("Turtlebot Simulation")
vis.SetCamera(chrono.ChVectorD(0, 1, -3), chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 1, 0))
vis.SetTypicalLenght(1.0)
vis.SetCameraUp(chrono.ChVectorD(0, 1, 0))
vis.SetCameraSpeed(0.5)
vis.SetCameraRotSpeed(0.1)
vis.AddTypicalLights()

# Set the simulation parameters
tstep = 0.01
tend = 10
time = 0

# Simulation loop
while time < tend:
    # Control the motors
    if time < 2:
        left_motor.SetSpeed(-1)
        right_motor.SetSpeed(1)
    elif time < 4:
        left_motor.SetSpeed(1)
        right_motor.SetSpeed(-1)
    else:
        left_motor.SetSpeed(0)
        right_motor.SetSpeed(0)

    # Update the simulation
    sys.DoStepDynamics(tstep)

    # Update the visualization
    vis.Render()

    # Increment the time
    time += tstep

# Wait for the visualization to close
vis.Run()

# Remove the visualization
vis.Finalize()
```

This script creates a PyChrono simulation with a Turtlebot robot on a rigid terrain. The simulation includes gravity, a ground body, and the Turtlebot with specific position and orientation. The script also sets up a real-time visualization using Irrlicht with camera settings and lighting. The simulation loop controls the Turtlebot's motors to turn left and right at specified times, updating and rendering the robot at each timestep.
