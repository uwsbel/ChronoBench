```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.engine as chronoengine

# Initialize the PyChrono environment
chrono.SetChronoDataPath("path/to/data")

# Create the simulation system and add it to the physical system
mysystem = chrono.ChSystemNSC()
mysystem.SetSolverType(chrono.ChSolver.Type_PSSOR)

# Create the floor (truss)
floor = chrono.ChBodyEasyBox(mysystem, 10, 1, 10, 1000, True, True, chrono.ChVectorD(0, -3, 0))
floor.SetBodyFixed(True)

# Create the crankshaft
crankshaft = chrono.ChBodyEasyCylinder(mysystem, 0.5, 2, 1000, True, True, chrono.ChVectorD(0, 0, 0))
crankshaft.SetPos(chrono.ChVectorD(0, 2, 0))
crankshaft.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI / 4))

# Create the connecting rod
connecting_rod = chrono.ChBodyEasyBox(mysystem, 2, 0.5, 0.5, 1000, True, True, chrono.ChVectorD(0, 0, 0))
connecting_rod.SetPos(chrono.ChVectorD(0, 3, 0))

# Create the piston
piston = chrono.ChBodyEasyBox(mysystem, 1, 1, 1, 1000, True, True, chrono.ChVectorD(0, 0, 0))
piston.SetPos(chrono.ChVectorD(0, 4, 0))

# Create joints
revolute_joint = chrono.ChLinkRevolute()
revolute_joint.Initialize(crankshaft, connecting_rod, chrono.ChFrameD(chrono.ChVectorD(0, 2, 0)))
mysystem.Add(revolute_joint)

prismatic_joint = chrono.ChLinkPrismatic()
prismatic_joint.Initialize(connecting_rod, piston, chrono.ChFrameD(chrono.ChVectorD(0, 3, 0)))
mysystem.Add(prismatic_joint)

# Add motor to crankshaft
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(crankshaft, floor, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))
motor.SetSpeed(chrono.CH_C_PI)
mysystem.Add(motor)

# Set up visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(800, 600)
vis.SetWindowTitle("Crank-Slider Mechanism")
vis.SetCamera(chrono.ChVectorD(0, 0, 10), chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 1, 0))
vis.AddSkyBox()
vis.AddTypicalLights()
vis.AddLogo()
vis.AddCameraVerticalTrack()
vis.AddCameraHorizontalTrack()
vis.AddCameraZoom()
vis.SetShowFPS(True)

# Add visualization assets
vis.AddTypicalLigths()
vis.AddGrid(chrono.ChVectorD(0, -3, 0), 10, 10, 1, 1, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngX(chrono.CH_C_PI / 2)))
vis.AddCoordSystem(chrono.ChVectorD(0, 0, 0), 1, True)

# Simulate
while vis.Run():
    mysystem.DoStepDynamics(0.01)
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()
```

Note: You should replace `"path/to/data"` with the actual path to the PyChrono data directory.
