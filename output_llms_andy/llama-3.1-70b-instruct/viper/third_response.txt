```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as viper
import pychrono.vehicle as veh
from pychrono import irrlicht as chronoirr
from pychrono import vehicle as veh

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))  # Set gravity in the negative Z direction
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create SCM deformable terrain
terrain_length = 10.0
terrain_width = 10.0
terrain_mesh_res = 0.1
terrain_soil_type = 0  # 0 = rigid, 1 = elastic, 2 = plastic
terrain_soil_par = veh.SoilParameters()
terrain_soil_par.kC = 2e6  # cohesion
terrain_soil_par.kPhi = 0.5  # angle of internal friction
terrain_soil_par.kPsi = 0.5  # angle of dilatancy
terrain_soil_par.kE = 1e6  # elastic modulus
terrain_soil_par.knu = 0.3  # Poisson ratio
terrain_soil_par.kcohesion = 2e6  # cohesion
terrain_soil_par.kadhesion = 0.0  # adhesion
terrain_soil_par.kfriction = 0.5  # friction coefficient
terrain_soil_par.kdensity = 1500  # density
terrain_soil_par.kyoungs = 2e6  # Young's modulus
terrain_soil_par.kpoisson = 0.3  # Poisson ratio
terrain_soil_par.kadhesion = 0.0  # adhesion
terrain_soil_par.krolling = 0.0  # rolling resistance
terrain_soil_par.kspring = 1e6  # spring constant
terrain_soil_par.kdamping = 0.0  # damping coefficient
terrain_soil_par.kmu1 = 0.5  # friction coefficient
terrain_soil_par.kmu2 = 0.5  # friction coefficient
terrain_soil_par.kCte = 0.0  # cohesion
terrain_soil_par.kCteI = 0.0  # cohesion
terrain_soil_par.kCteV = 0.0  # cohesion
terrain_soil_par.kCteL = 0.0  # cohesion
terrain_soil_par.kCteR = 0.0  # cohesion
terrain_soil_par.kCteF = 0.0  # cohesion
terrain_soil_par.kCteB = 0.0  # cohesion
terrain_soil_par.kCteT = 0.0  # cohesion
terrain = veh.SCMDeformableTerrain(system, terrain_length, terrain_width, terrain_mesh_res, terrain_soil_type, terrain_soil_par)
terrain.SetName("terrain")
terrain.SetCollide(True)
system.Add(terrain)

# Create Viper rover
rover = viper.Viper(system)  # Instantiate the Viper rover
driver = viper.ViperDCMotorControl()  # Create a driver for the rover
rover.SetDriver(driver)  # Assign the driver to the rover

# Initialize rover position and orientation
init_pos = chrono.ChVector3d(0, 0.2, 0)
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)
rover.Initialize(chrono.ChFramed(init_pos, init_rot))

# Create run-time visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Viper rover - Deformable terrain')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 2.5, 1.5), chrono.ChVector3d(0, 0, 1))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0.5), 3, 4, 10, 40, 512)

# Enable shadows (commented out to improve performance)
# vis.EnableShadows()

# Set the simulation time step
time_step = 1e-3

# Simulation loop
time = 0
while vis.Run():
    time += time_step  # Increment the simulation time
    steering = 0.0  # Set steering to a constant value

    driver.SetSteering(steering)  # Set the steering for the rover

    rover.Update()  # Update the rover's state

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    
    # Advance the simulation by one time step
    system.DoStepDynamics(time_step)
```
