vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()

system.Add(vehicle.GetVehicle())

# Create a terrain with multiple patches
terrain = chrono.ChTerrain()

terrain.Initialize()

# Add the terrain to the system
system.Add(terrain)
