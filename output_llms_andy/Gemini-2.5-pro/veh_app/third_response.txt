```python
import pychrono.core as chrono
# import pychrono.irrlicht as irr # Removed as not used
import pychrono.vehicle as veh
import pychrono.sensor as sens
import math
import os

# Set the data path for Chrono assets
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

print(chrono.GetChronoDataPath() + 'vehicle/')

# Initial vehicle location and orientation
initLoc = chrono.ChVector3d(0, -5, 0.4)
initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# Visualization type for vehicle parts (PRIMITIVES, MESH, or NONE)
chassis_vis_type = veh.VisualizationType_MESH
suspension_vis_type = veh.VisualizationType_PRIMITIVES
steering_vis_type = veh.VisualizationType_PRIMITIVES
wheel_vis_type = veh.VisualizationType_NONE # Changed to NONE as per original, MESH is also common
tire_vis_type = veh.VisualizationType_MESH

# Simulation step sizes
step_size = 1e-3
tire_step_size = step_size # Tire step size can be different from system step size

# Simulation end time
# tend = 1000 # Original variable, end_time is used in the loop
end_time = 30 # Simulation duration

# Sensor parameters
update_rate = 10        # Update rate in Hz for the sensors
image_width = 1280      # Image width for cameras
image_height = 720      # Image height for cameras
fov = 1.408             # Camera's horizontal field of view (radians)
lag = 0                 # Lag (in seconds) between sensing and data access
# exposure_time = 0     # Unused in this setup, sensors have their own exposure if applicable

# Create the vehicle, set parameters, and initialize
gator = veh.Gator()
gator.SetContactMethod(chrono.ChContactMethod_NSC)
gator.SetChassisFixed(False)
gator.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
gator.SetBrakeType(veh.BrakeType_SHAFTS)
gator.SetTireType(veh.TireModelType_TMEASY) # Using TMEASY tires
gator.SetTireStepSize(tire_step_size)
gator.SetInitFwdVel(0.0)
gator.Initialize()

gator.SetChassisVisualizationType(chassis_vis_type)
gator.SetSuspensionVisualizationType(suspension_vis_type)
gator.SetSteeringVisualizationType(steering_vis_type)
gator.SetWheelVisualizationType(wheel_vis_type)
gator.SetTireVisualizationType(tire_vis_type)

# Print vehicle information
print("Vehicle mass:   " + str(gator.GetVehicle().GetMass()))
print("Driveline type: " + gator.GetVehicle().GetDriveline().GetTemplateName())
print("Brake type:     " + gator.GetVehicle().GetBrake(0, veh.LEFT).GetTemplateName()) # Gator has 2 axles (0 and 1)
print("Tire type:      " + gator.GetVehicle().GetTire(0, veh.LEFT).GetTemplateName()) # Gator has 2 axles (0 and 1)
print("\n")

# Set collision system type
gator.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# ------------------
# Create the terrain
# ------------------
terrain = veh.RigidTerrain(gator.GetSystem())
patch_mat = chrono.ChContactMaterialNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)
# Create a patch using a ChCoordsysd (position and orientation of the patch)
# chrono.CSYSNORM is chrono.ChCoordsysd(chrono.VNULL, chrono.QUNIT) -> centered at origin, XY plane
patch = terrain.AddPatch(patch_mat, chrono.CSYSNORM, 200, 200) # Increased size for more driving room
patch.SetColor(chrono.ChColor(0.8, 0.8, 1.0))
patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200) # Match texture scale to patch size
terrain.Initialize()

# Create a box obstacle
box = chrono.ChBodyEasyBox(1, 1, 1, 1000) # 1x1x1m box, density 1000
box.SetPos(chrono.ChVector3d(5, 0, 0.5)) # Positioned on the ground
box.SetFixed(True)
box.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.png"))
gator.GetSystem().AddBody(box)

# Create a cylinder obstacle
# ChBodyEasyCylinder(axis, radius, height, density)
# Original: chrono.ChAxis_Y means cylinder lies along Y axis
cylinder = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y ,0.5, 1, 1000)
cylinder.SetPos(chrono.ChVector3d(7, 2, 1.0)) # Positioned with its center at Z=1.0
cylinder.SetFixed(True)
cylinder.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.png"))
gator.GetSystem().AddBody(cylinder)


# Create the interactive driver system
driver = veh.ChDriver(gator.GetVehicle())
driver.Initialize()

# -----------------------
# Create a sensor manager
# -----------------------
manager = sens.ChSensorManager(gator.GetSystem())
intensity = 1.0
# Set the background scene color (R, G, B, Alpha), 0-1 range
manager.scene.GetBackground().color = chrono.ChColor4f(0.1, 0.2, 0.3, 1.0) # Example: a darker sky
manager.scene.AddPointLight(chrono.ChVector3f(20, 20, 100), chrono.ChColor3f(intensity, intensity, intensity), 500.0)
manager.scene.AddPointLight(chrono.ChVector3f(-20, 20, 100), chrono.ChColor3f(intensity, intensity, intensity), 500.0)


# Create RGB camera and add it to the sensor manager
cam_offset_pose = chrono.ChFramed(chrono.ChVector3d(0.5, 0, 1.45), chrono.QuatFromAngleAxis(0, chrono.ChVector3d(0, 1, 0)))
# Using slightly different pose for main camera to avoid direct view into the vehicle model for 3rd person
# cam_offset_pose = chrono.ChFramed(chrono.ChVector3d(-8.0, 0, 1.45), chrono.QuatFromAngleAxis(0.2, chrono.ChVector3d(0, 1, 0)))

rgb_cam = sens.ChCameraSensor(
    gator.GetChassisBody(), # Parent body
    update_rate,            # Update rate in Hz
    cam_offset_pose,        # Offset pose relative to parent
    image_width,            # Image width
    image_height,           # Image height
    fov                     # Horizontal FOV
    # lag,                  # Lag (sec)
    # exposure_time         # Exposure time (sec) - camera has its own default
)
rgb_cam.SetName("RGB Camera")
rgb_cam.SetLag(lag)
# rgb_cam.SetCollectionWindow(exposure_time) # Not collection window, exposure time
rgb_cam.PushFilter(sens.ChFilterVisualize(image_width, image_height, "RGB Camera Feed"))
manager.AddSensor(rgb_cam)

# Create Lidar sensor
lidar_offset_pose = chrono.ChFramed(
        chrono.ChVector3d(0.0, 0, 1.8), # Slightly higher to clear vehicle body
        chrono.QuatFromAngleAxis(0, chrono.ChVector3d(0, 1, 0)) # Forward facing
    )
lidar = sens.ChLidarSensor(
    gator.GetChassisBody(),              # Body lidar is attached to
    update_rate,            # Scanning rate in Hz
    lidar_offset_pose,      # Offset pose
    1800,                   # Number of horizontal samples (increased for better density)
    32,                     # Number of vertical channels (e.g., Velodyne VLP-32C)
    2 * chrono.CH_PI,       # Horizontal field of view
    chrono.CH_PI / 6,       # Maximum vertical FOV (e.g., +15 degrees)
    -chrono.CH_PI / 6,      # Minimum vertical FOV (e.g., -15 degrees)
    100.0,                  # Maximum lidar range
    sens.LidarBeamShape_RECTANGULAR, # Beam shape
    1,                      # Sample radius for rectangular beam (horizontal)
    1,                      # Sample radius for rectangular beam (vertical)
    0.003,                  # Horizontal divergence angle (radians)
    0.003,                  # Vertical divergence angle (radians)
    sens.LidarReturnMode_STRONGEST_RETURN, # Return mode
    0.01                    # Clipping radius (for close points)
)
lidar.SetName("Lidar Sensor")
lidar.SetLag(lag)
# lidar.SetCollectionWindow(exposure_time) # Lidar has its own collection mechanism related to update_rate
lidar.SetCollectionWindow(0) # Collect instantaneously at update moment
lidar.PushFilter(sens.ChFilterDIAccess()) # Accesses Depth and Intensity
lidar.PushFilter(sens.ChFilterPCfromDepth()) # Converts to Point Cloud
lidar.PushFilter(sens.ChFilterXYZIAccess()) # Accesses XYZI point cloud data
if True: # Optionally visualize Lidar
    lidar.PushFilter(sens.ChFilterVisualizePointCloud(640, 480, 0.5, "Lidar Point Cloud")) # Smaller point size
manager.AddSensor(lidar)


# ----------------------
# ADDED: Depth Camera
# ----------------------
depth_cam_offset_pose = chrono.ChFramed(
    chrono.ChVector3d(-5.0, 0, 2.0), # As specified: x=-5m, y=0, z=2m relative to chassis
    chrono.QUNIT # Standard orientation (camera X along chassis X, Y along Y, Z along Z)
                 # This means it looks "forward" from that offset point.
)
depth_cam = sens.ChCameraSensor(
    gator.GetChassisBody(),
    update_rate,
    depth_cam_offset_pose,
    image_width,            # As specified: 1280
    image_height,           # As specified: 720
    fov,                    # As specified: 1.408
    # lag and exposure_time can be set if needed, using defaults or global 'lag'
)
depth_cam.SetName("Depth Camera")
depth_cam.SetLag(lag)
# Set clipping planes for depth camera: near_dist, far_dist (max_depth)
depth_cam.SetClippingNearFar(0.1, 30.0) # Max depth of 30

# Add filters for depth camera
depth_cam.PushFilter(sens.ChFilterDepthAccess()) # To get depth data
depth_cam.PushFilter(sens.ChFilterVisualize(image_width, image_height, "Depth Map Visualization")) # To visualize it
manager.AddSensor(depth_cam)


# ---------------
# Simulation loop
# ---------------
realtime_timer = chrono.ChRealtimeStepTimer()
sim_time = 0 # Use a separate variable for simulation time tracking if needed

# Open a file for logging vehicle state (optional, otherwise prints to console)
# log_file_path = "vehicle_state_log.txt"
# log_file = open(log_file_path, "w")
# log_file.write("Time (s)\tPos_X (m)\tPos_Y (m)\tPos_Z (m)\tHeading (deg)\n")


print("Starting simulation...")
while sim_time < end_time:
    current_time_gator = gator.GetSystem().GetChTime()

    # Set driver inputs
    # Example: Simple sinusoidal steering
    steering_input = 0.5 * math.sin(2 * chrono.CH_PI * 0.1 * current_time_gator)
    throttle_input = 0.2
    braking_input = 0.0
    
    if current_time_gator > end_time / 2 : # Brake in the second half
        throttle_input = 0.0
        braking_input = 0.5

    driver.SetSteering(steering_input)
    driver.SetThrottle(throttle_input)
    driver.SetBraking(braking_input)
    
    # Collect output data from modules
    driver_inputs = driver.GetInputs()

    # Update modules (process inputs from other modules)
    driver.Synchronize(current_time_gator)
    terrain.Synchronize(current_time_gator)
    gator.Synchronize(current_time_gator, driver_inputs, terrain)

    # Advance simulation for one timestep for all modules
    driver.Advance(step_size)
    terrain.Advance(step_size)
    gator.Advance(step_size) # This advances the ChSystem time

    # Update sensor manager (after system state has been advanced)
    manager.Update()

    # Update sim_time based on the system's time after advancing
    sim_time = gator.GetSystem().GetChTime()

    # --------------------------
    # ADDED: Log Vehicle State
    # --------------------------
    # Get vehicle position and orientation after advancing
    vehicle_pos = gator.GetVehicle().GetPos()
    vehicle_rot_quat = gator.GetVehicle().GetRot()
    
    # Get heading (Yaw angle) from quaternion
    # For a typical vehicle setup (X forward, Y left, Z up for the chassis frame),
    # GetYaw() should give the rotation around the global Z axis.
    vehicle_heading_rad = vehicle_rot_quat.GetYaw()
    vehicle_heading_deg = math.degrees(vehicle_heading_rad)

    # Log to console
    log_string = (f"Time: {sim_time:.3f} s | "
                  f"Pos: ({vehicle_pos.x:.3f}, {vehicle_pos.y:.3f}, {vehicle_pos.z:.3f}) m | "
                  f"Heading: {vehicle_heading_deg:.2f} deg")
    print(log_string)

    # Log to file (if enabled)
    # log_file.write(f"{sim_time:.3f}\t{vehicle_pos.x:.3f}\t{vehicle_pos.y:.3f}\t{vehicle_pos.z:.3f}\t{vehicle_heading_deg:.2f}\n")

    # Spin in place for real time to catch up
    realtime_timer.Spin(step_size)

# Close log file (if enabled)
# log_file.close()
# print(f"Vehicle state log saved to {log_file_path}")

print("Simulation finished.")
```
