```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr
import os
import math

# ---------------------------------------------------------------------
# Set Chrono data directory
# ---------------------------------------------------------------------
chrono.SetChronoDataPath(os.environ.get('CHRONO_DATA_DIR', '../../../chrono_data/'))
veh.SetDataPath(os.path.join(chrono.GetChronoDataPath(), 'vehicle', ''))

# ---------------------------------------------------------------------
# Simulation parameters
# ---------------------------------------------------------------------
step_size = 0.005  # Time step for integration
simulation_time = 60 # Total simulation time in seconds

# Initial vehicle position and orientation
init_loc = chrono.ChVectorD(-70, 0, 0.5)
init_rot = chrono.ChQuaternionD(1, 0, 0, 0) # No rotation initially

# HMMWV parameters
engine_model = veh.EngineModelType_SHAFTS
drive_type = veh.DrivelineTypeWV_AWD
tire_model = veh.TireModelType_RIGID # Using rigid tires for rigid terrain

# Visualization type for vehicle parts
vis_type_chassis = veh.VisualizationType_MESH
vis_type_suspension = veh.VisualizationType_MESH # Can be PRIMITIVES for performance
vis_type_steering = veh.VisualizationType_MESH # Can be PRIMITIVES
vis_type_wheel = veh.VisualizationType_MESH
vis_type_tire = veh.VisualizationType_MESH

# Collision type for chassis (PRIMITIVES is usually sufficient and faster)
chassis_collision_type = veh.CollisionType_PRIMITIVES

# Point on chassis tracked by the camera
trackPoint = chrono.ChVectorD(0.0, 0.0, 1.75)

# ---------------------------------------------------------------------
# Create the Chrono system
# ---------------------------------------------------------------------
# Using NSC (Non-Smooth Contact) for this simulation
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))
system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN) # A fast solver
system.SetSolverMaxIterations(150)
system.SetMaxPenetrationRecoverySpeed(4.0)

# ---------------------------------------------------------------------
# Create the HMMWV vehicle
# ---------------------------------------------------------------------
hmmwv = veh.HMMWV_Full()
hmmwv.SetContactMethod(chrono.ChMaterialSurface.NSC)
hmmwv.SetChassisFixed(False)
hmmwv.SetInitPosition(chrono.ChCoordsysD(init_loc, init_rot))
hmmwv.SetEngineType(engine_model)
hmmwv.SetDrivelineType(drive_type)
hmmwv.SetTireType(tire_model)
hmmwv.SetChassisCollisionType(chassis_collision_type)

# Initialize the vehicle (constructor in HMMWV_Full calls Initialize)
hmmwv.Initialize()

# Set visualization types for vehicle components
hmmwv.SetChassisVisualizationType(vis_type_chassis)
hmmwv.SetSuspensionVisualizationType(vis_type_suspension)
hmmwv.SetSteeringVisualizationType(vis_type_steering)
hmmwv.SetWheelVisualizationType(vis_type_wheel)
hmmwv.SetTireVisualizationType(vis_type_tire)

# Get the vehicle (which is a ChWheeledVehicle)
vehicle = hmmwv.GetVehicle()
# Associate the ChSystem
vehicle.SetSystem(system)


# ---------------------------------------------------------------------
# Create the rigid terrain
# ---------------------------------------------------------------------
terrain = veh.RigidTerrain(system)
terrain.SetContactFrictionCoefficient(0.9)
terrain.SetContactRestitutionCoefficient(0.01)
terrain.SetContactMaterialProperties(2e7, 0.3) # Young's modulus and Poisson ratio

# Patch 1: Flat textured patch (e.g., asphalt)
patch_material = chrono.ChMaterialSurfaceNSC()
patch_material.SetFriction(0.8)
patch_material.SetRestitution(0.01)

texture_file_asphalt = chrono.GetChronoDataFile("vehicle/terrain/textures/tile4.jpg")
terrain.AddPatch(patch_material,
                 chrono.CSYSNORM, # Centered at origin, normal to Z
                 60.0, 20.0, # length (X), width (Y)
                 1.0, # thickness
                 True, # Tillable texture
                 texture_file_asphalt,
                 20, 20) # texture scale U, V

# Patch 2: Mesh-based bump
# Create a ChTriangleMeshConnected to define the bump geometry
# For simplicity, we'll use a pre-defined bump mesh from Chrono data
bump_mesh_file = chrono.GetChronoDataFile("vehicle/terrain/meshes/bump.obj")
bump_texture_file = chrono.GetChronoDataFile("vehicle/terrain/textures/concrete.jpg")
bump_trimesh = chrono.ChTriangleMeshConnected()
bump_trimesh.LoadWavefrontMesh(bump_mesh_file, False, True)

# Transform the bump mesh (position it after the first patch)
bump_transform = chrono.ChTransformD()
bump_transform.SetPos(chrono.ChVectorD(40, 0, 0)) # Position of the bump center
bump_transform.SetRot(chrono.Q_from_AngZ(0))      # Rotation

# Add the mesh patch
# Note: MeshDriver requires a TRImesh, not ChTriangleMeshConnected directly
# However, RigidTerrain's AddPatch with a ChTriangleMeshConnected handles this.
# The API for RigidTerrain AddPatch using ChTriangleMeshConnected is:
# AddPatch(material, trimesh, texture_file, texture_scale_UV, collision_mesh, sweep_sphere_radius)
# Or more commonly for simple meshes: AddPatch(material, mesh, transform, texture_file, texture_scale_U, texture_scale_V, is_tiled)
# The most direct way using a mesh *file* is:
# terrain.AddPatch(patch_material, chrono.ChVectorD(40,0,0), bump_mesh_file, 0.05, 1.0, 1.0, False)
# Let's try this one for simplicity
# Arguments: material, position, mesh_filename, sweep_sphere_radius, tex_scale_x, tex_scale_y, is_tiled_texture
# The texture on mesh for RigidTerrain is a bit tricky. Often the obj file has its own .mtl.
# If not, it might use a default color or apply a texture if specified.
# Let's use a simpler method by creating a BoxDriver-like definition but with a mesh:
# A more robust way if you have the mesh object:
mesh_body = chrono.ChBody()
mesh_body.SetPos(chrono.ChVectorD(40, 0, 0)) # Position of bump
mesh_body.SetBodyFixed(True)
mesh_body.GetCollisionModel().ClearModel()
mesh_body.GetCollisionModel().AddTriangleMesh(patch_material, bump_trimesh, False, False, chrono.ChVectorD(0,0,0), chrono.ChMatrix33D(1), 0.05)
mesh_body.GetCollisionModel().BuildModel()
system.Add(mesh_body)

# Add visualization for the bump mesh
bump_shape = chrono.ChTriangleMeshShape()
bump_shape.SetMesh(bump_trimesh)
bump_shape.SetTexture(bump_texture_file, 10, 10) # scale U, V
mesh_body.AddVisualShape(bump_shape)


# Patch 3: Heightmap-based patch (e.g., rough terrain or slope)
heightmap_file = chrono.GetChronoDataFile("vehicle/terrain/height_maps/slope.bmp")
texture_file_grass = chrono.GetChronoDataFile("vehicle/terrain/textures/grass.jpg")
# Parameters: hf_filename, mesh_name, sizeX, sizeZ, hMin, hMax, pos, texture_file, tex_scale_x, tex_scale_y
# Note: 'sizeZ' in HeightMapDriver means size along the Y-axis of the terrain patch.
# Heightmap texture scaling is often trial and error.
# hMin and hMax should match the black-to-white range of your BMP.
# Position is the center of the heightmap patch.
terrain.AddPatch(patch_material,
                 chrono.ChVectorD(90, 0, -0.5), # Position (center X, Y, Z - Z is height of hMin)
                 heightmap_file, "heightmap_patch",
                 40.0, 30.0, # Size X, Size Y (width)
                 0.0, 2.0, # Min height, Max height (relative to patch Z position)
                 True, # Tillable texture
                 texture_file_grass,
                 10, 10) # texture scale U, V

# Patch 4: Another flat textured patch (e.g., concrete)
texture_file_concrete = chrono.GetChronoDataFile("vehicle/terrain/textures/concrete.jpg")
terrain.AddPatch(patch_material,
                 chrono.ChVectorD(140, 0, 0), # Positioned after heightmap
                 60.0, 20.0,
                 1.0, # thickness
                 True,
                 texture_file_concrete,
                 15, 15)

# Initialize the terrain
terrain.Initialize()

# ---------------------------------------------------------------------
# Create the Irrlicht application for visualization
# ---------------------------------------------------------------------
app = chronoirr.ChIrrApp(system, "HMMWV on Complex Rigid Terrain", chronoirr.dimension2du(1280, 720))
app.SetTimestep(step_size)
app.AddTypicalSky()
app.AddTypicalLights()

# Set camera to follow the vehicle
app.SetChaseCamera(trackPoint, 6.0, 0.5) # Point on vehicle, distance, height
app.SetChaseCameraPosition(vehicle.GetPos() + chrono.ChVectorD(0,0,20)) # Initial camera pos

# Bind all assets for visualization
app.AssetBindAll()
app.AssetUpdateAll()

# ---------------------------------------------------------------------
# Create the interactive driver system
# ---------------------------------------------------------------------
driver = veh.ChIrrGuiDriver(app)

# Set the time response for steering and throttle keyboard inputs.
# NOTE: this is not exact, since keyboard inputs are processed events
# not sampledshield_layer time.
steering_time = 1.0  # time to go from 0 to +1 (or from 0 to -1)
throttle_time = 1.0  # time to go from 0 to +1
braking_time = 0.3   # time to go from 0 to +1

driver.SetSteeringDelta(step_size / steering_time)
driver.SetThrottleDelta(step_size / throttle_time)
driver.SetBrakingDelta(step_size / braking_time)

driver.Initialize()

# ---------------------------------------------------------------------
# Simulation loop
# ---------------------------------------------------------------------
realtime_timer = chrono.ChRealtimeStepTimer()
time = 0

while app.GetDevice().run():
    time = system.GetChTime()
    if time > simulation_time:
        break

    app.BeginScene(True, True, chronoirr.SColor(255, 140, 160, 190))
    app.DrawAll()
    
    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Update modules (process inputs from other modules)
    driver.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    terrain.Synchronize(time) # Important for moving patches, though not used here
    app.Synchronize(driver.GetInputModeAsString(), driver_inputs) # For on-screen messages

    # Advance simulation for one timestep
    driver.Advance(step_size)
    vehicle.Advance(step_size)
    terrain.Advance(step_size) # Important for moving patches
    app.Advance(step_size) # Update camera, etc.
    
    # Advance the system dynamics
    system.DoStepDynamics(step_size)

    app.EndScene()
    realtime_timer.Spin(step_size)

print("Simulation finished.")
```
