# -----------------------------------------------------------------------------
# SCM Terrain Parameter Encapsulation Class
# -----------------------------------------------------------------------------
class SCMTerrainParameters:
    """
    Manages SCM terrain soil parameters and allows configuration from predefined sets.
    """
    def __init__(self):
        # Default parameters (can be overridden by configure method)
        self.BekkerKphi = 0.2e6  # Bekker Kphi (Pa/m^n)
        self.BekkerKc = 0       # Bekker Kc (Pa/m^(n-1))
        self.BekkerNExponent = 1.1 # Bekker n exponent (-)
        self.MohrCohesiveLimit = 0 # Mohr-Coulomb cohesion (Pa)
        self.MohrFrictionLimit = 20 # Mohr-Coulomb friction angle (degrees)
        self.JanosiShearCoefficient = 0.01 # Janosi shear parameter J (m)
        self.ElasticStiffness = 2e7 # Elastic stiffness K (Pa/m) (or K_e)
        self.Damping = 3e4      # Damping C (Pa*s/m)

    def configure(self, config_type="soft"):
        """
        Sets soil parameters based on predefined configurations.
        Supported types: "soft", "mid", "hard".
        """
        if config_type == "soft":
            self.BekkerKphi = 2e5
            self.BekkerKc = 0
            self.BekkerNExponent = 1.1
            self.MohrCohesiveLimit = 0
            self.MohrFrictionLimit = 20
            self.JanosiShearCoefficient = 0.01
            self.ElasticStiffness = 2e7  # Lower stiffness for softer terrain
            self.Damping = 3e4
        elif config_type == "mid": # Matches original script parameters
            self.BekkerKphi = 2e6
            self.BekkerKc = 0
            self.BekkerNExponent = 1.1
            self.MohrCohesiveLimit = 0
            self.MohrFrictionLimit = 30
            self.JanosiShearCoefficient = 0.01
            self.ElasticStiffness = 2e8
            self.Damping = 3e4
        elif config_type == "hard":
            self.BekkerKphi = 5e6
            self.BekkerKc = 1e3 # Small cohesion for harder pack
            self.BekkerNExponent = 1.0
            self.MohrCohesiveLimit = 1e3
            self.MohrFrictionLimit = 35
            self.JanosiShearCoefficient = 0.02 # Slightly larger shear displacement
            self.ElasticStiffness = 5e8  # Higher stiffness
            self.Damping = 5e4
        else:
            print(f"Warning: SCM terrain configuration '{config_type}' not recognized. Using default 'soft' parameters.")
            self.configure("soft") # Default to soft if unknown configuration

    def apply_to(self, scm_terrain_object):
        """
        Applies the stored parameters to the given SCMTerrain object.
        """
        if not isinstance(scm_terrain_object, veh.SCMTerrain):
            raise TypeError("scm_terrain_object must be an instance of veh.SCMTerrain")
        
        scm_terrain_object.SetSoilParameters(
            self.BekkerKphi,
            self.BekkerKc,
            self.BekkerNExponent,
            self.MohrCohesiveLimit,
            self.MohrFrictionLimit,
            self.JanosiShearCoefficient,
            self.ElasticStiffness,
            self.Damping
        )
