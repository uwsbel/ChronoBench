```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import os
import math

# -----------------------------------------------------------------------------
# Set Chrono data directory
# -----------------------------------------------------------------------------
chrono.SetChronoDataPath(os.environ.get('CHRONO_DATA_DIR', '../../../chrono_data/'))
veh.SetDataPath(os.path.join(chrono.GetChronoDataPath(), 'vehicle/'))

# -----------------------------------------------------------------------------
# Script Parameters
# -----------------------------------------------------------------------------

# Simulation step size
step_size = 0.005  # For TMEASY, a smaller step size is often better

# Initial vehicle position and orientation
initLoc = chrono.ChVector3d(0, 0, 0.7)  # Slightly above ground
initRot = chrono.ChQuaterniond(1, 0, 0, 0) # No rotation

# Contact method (NSC is common for rigid terrain vehicle sims)
contact_method = chrono.ChContactMethod_NSC

# Vehicle JSON files
vehicle_file = veh.GetDataFile("MAN_10t/vehicle/MAN_10t_Vehicle.json")
# The MAN_10t_Vehicle.json should reference TMeasy tire parameter files by default
# e.g., "MAN_10t/tire/MAN_10t_TMeasyTire.json"
# If you want to explicitly set a TMeasy parameter file, you can use:
# tmeasy_tire_file = veh.GetDataFile("MAN_10t/tire/MAN_10t_TMeasyTire.json")
powertrain_file = veh.GetDataFile("MAN_10t/powertrain/MAN_10t_SimpleMapPowertrain.json")

# Tire model type
tire_model = veh.TireModelType_TMEASY

# Visualization settings
chassis_vis_type = veh.VisualizationType_MESH # or PRIMITIVES, NONE
suspension_vis_type = veh.VisualizationType_PRIMITIVES
steering_vis_type = veh.VisualizationType_PRIMITIVES
wheel_vis_type = veh.VisualizationType_MESH # NONE is also common for TMEASY if performance is an issue
tire_vis_type = veh.VisualizationType_MESH # Or PRIMITIVES, NONE

# Collision settings
chassis_collide = True
# For TMeasy, the tire itself handles contact, so wheel (rim) collision is optional
# It can be useful if the rim might hit curbs or large obstacles.
# For standard driving on a flat plane, False is often fine to avoid conflicts.
wheel_collide = False # Set to True if rim collision is desired

# Terrain parameters
terrain_height = 0.0  # z-level of the terrain
terrain_size_x = 200  # meters
terrain_size_y = 200  # meters
terrain_friction = 0.8
terrain_restitution = 0.01
terrain_texture_file = veh.GetDataFile("terrain/textures/Concrete.jpg")
terrain_texture_scale_x = 20 # How many times the texture repeats in X
terrain_texture_scale_y = 20 # How many times the texture repeats in Y

# Irrlicht visualization parameters
irr_window_title = "MAN 10t TMEASY on Rigid Terrain"
irr_window_size = irr.dimension2du(1280, 720)
irr_logo_file = chrono.GetChronoDataFile('logo_pychrono_alpha.png')
irr_chase_cam_pos = chrono.ChVector3d(0.0, 0.0, 1.75) # Point on chassis
irr_chase_cam_dist = 8.0 # Distance from point
irr_chase_cam_height = 1.5 # Height of camera relative to lookat point

# Driver control parameters
driver_steering_delta = 0.02
driver_throttle_delta = 0.02
driver_braking_delta = 0.05

# -----------------------------------------------------------------------------
# 1. Initialize PyChrono environment and core components
# -----------------------------------------------------------------------------
print("Creating Chrono system...")
sys = chrono.ChSystemNSC() if contact_method == chrono.ChContactMethod_NSC else chrono.ChSystemSMC()
sys.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))

# Solver settings (good defaults for vehicle simulations)
sys.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)
sys.GetSettings().solver.max_iteration_bilateral = 150
sys.GetSettings().solver.max_iteration_normal = 0 # Not used by BARZILAIBORWEIN
sys.GetSettings().solver.max_iteration_sliding = 150
sys.GetSettings().solver.max_iteration_spinning = 0 # Not used by BARZILAIBORWEIN
sys.GetSettings().solver.alpha = 0 # Not used by BARZILAIBORWEIN
sys.GetSettings().solver.contact_recovery_speed = 0.4
sys.SetMaxPenetrationRecoverySpeed(4.0)
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.03)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.02)

# -----------------------------------------------------------------------------
# 2. Add required physical systems and objects
# -----------------------------------------------------------------------------

# Create the MAN 10t truck
print(f"Loading vehicle from: {vehicle_file}")
truck = veh.WheeledVehicle(sys, vehicle_file)
truck.Initialize(chrono.ChCoordsysd(initLoc, initRot))

# Set visualization types
print("Setting vehicle visualization...")
truck.SetChassisVisualizationType(chassis_vis_type)
truck.SetSuspensionVisualizationType(suspension_vis_type)
truck.SetSteeringVisualizationType(steering_vis_type)
truck.SetWheelVisualizationType(wheel_vis_type)
truck.SetTireVisualizationType(tire_vis_type) # Important for TMEASY

# Set collision properties
print("Setting vehicle collision...")
truck.SetChassisCollide(chassis_collide)
truck.SetWheelCollide(wheel_collide) # Typically False for TMEASY as tire handles contact

# Set tire model (TMEASY)
print(f"Setting tire model to TMEASY...")
truck.SetTireType(tire_model)
# If you needed to explicitly set the TMEASY parameter file for ALL tires:
# truck.SetTireJSONFileForAll(tmeasy_tire_file) # This would override vehicle JSON defaults

# Create and initialize powertrain
print(f"Loading powertrain from: {powertrain_file}")
truck.CreatePowertrain(powertrain_file)

# Initialize the tires (this actually creates the TMEASY tire objects)
# It uses the tire model type set above and the tire JSON files specified
# in the main vehicle JSON or overridden by SetTireJSONFileForAll.
print("Initializing tires...")
truck.InitializeTires()


# Create the rigid terrain
print("Creating rigid terrain...")
terrain = veh.RigidTerrain(sys)
patch_mat = chrono.ChContactMaterialNSC() if contact_method == chrono.ChContactMethod_NSC else chrono.ChContactMaterialSMC()
patch_mat.SetFriction(terrain_friction)
patch_mat.SetRestitution(terrain_restitution)

# Create a large flat patch. CSYSNORM means Z-up, centered at origin.
# Texture coordinates will scale the texture image over the patch dimensions.
patch = terrain.AddPatch(
    patch_mat,
    chrono.CSYSNORM, # Plane at Z=0 normal to Z axis
    terrain_size_x, terrain_size_y,
    terrain_height # Thickness (not relevant for flat plane vis)
)
# Set texture for the patch
patch.SetTexture(terrain_texture_file, terrain_texture_scale_x, terrain_texture_scale_y)
patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5)) # Fallback color if texture fails

terrain.Initialize()

# -----------------------------------------------------------------------------
# 3. Initialize Irrlicht Visualization System
# -----------------------------------------------------------------------------
print("Creating Irrlicht visualization system...")
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowTitle(irr_window_title)
vis.SetWindowSize(irr_window_size)
vis.Initialize()

# Add logo
if os.path.exists(irr_logo_file):
    vis.AddLogo(irr_logo_file)
else:
    print(f"Warning: Logo file not found at {irr_logo_file}")

# Add skybox
vis.AddSkyBox()

# Add typical lights (directional)
# Parameters: e.g., vis.AddTypicalLights(irr.vector3df(30,100,30), irr.vector3df(30,100,50), 100, 30)
# Or use simpler defaults:
vis.AddTypicalLights(
    hemispheric = False, # Set to True for better ambient lighting
    # Default directional light parameters if hemispheric is False:
    # dir = irr.vector3df(2, 2, -2),
    # an_color = chrono.ChColor(0.2, 0.2, 0.2),
    # di_color = chrono.ChColor(0.8, 0.8, 0.8)
)
# Customize light direction if needed
# vis.GetSceneManager().getLight(0).setRotation(irr.vector3df(80, 120, 0)) # Example: (pitch, yaw, roll) degrees

# Configure chase camera (will be controlled by Irrlicht GUI driver)
# This sets the parameters for the default chase camera mode in the driver.
vis.SetChaseCamera(irr_chase_cam_pos, irr_chase_cam_dist, irr_chase_cam_height)
# To set camera target (usually default is fine, following vehicle CoG)
# vis.SetChaseCameraLookAt(chrono.ChVector3d(0,0,0)) # look at point, relative to chassis
# vis.SetChaseCameraAngle(-20 * chrono.CH_DEG_TO_RAD) # Set fixed camera viewing angle (pitch)

# Bind visualization assets (call AFTER all bodies are in the system)
vis.BindAll()


# -----------------------------------------------------------------------------
# 4. Create and initialize the driver system (GUI for real-time control)
# -----------------------------------------------------------------------------
print("Creating GUI driver...")
driver = veh.ChIrrGuiDriver(vis) # Pass the Irrlicht visualizer
driver.SetSteeringDelta(driver_steering_delta)
driver.SetThrottleDelta(driver_throttle_delta)
driver.SetBrakingDelta(driver_braking_delta)
driver.Initialize()
driver.AttachVehicle(truck.GetVehicle()) # GetVehicle() returns the ChWheeledVehicle

# -----------------------------------------------------------------------------
# 5. Simulation Loop
# -----------------------------------------------------------------------------
print("\nStarting simulation loop...")
print("------------------------------------")
print("Press T to increase throttle.")
print("Press G to decrease throttle.")
print("Press A to steer left.")
print("Press D to steer right.")
print("Press S to apply brakes.")
print("Press C to cycle camera modes (chase, fixed, inside).")
print("------------------------------------")

# Simulation time
time = 0.0

while vis.Run():
    time = sys.GetChTime()

    # Begin scene rendering
    vis.BeginScene(True, True, chrono.ChColor(0.2, 0.2, 0.2)) # Clear color

    # Render visualization elements
    vis.Render()

    # Optionally, draw a grid
    irr.ChIrrTools.drawGrid(vis, 1, 1, 50, 50,
                            chrono.ChCoordsysd(chrono.ChVector3d(0, 0, terrain_height + 0.01), chrono.QUNIT),
                            chrono.ChColor(0.1, 0.25, 0.25), True)

    # Get driver inputs
    driver.Synchronize(time)

    # Update vehicle, terrain, and other parts of the system
    truck.Synchronize(time, driver.GetSteering(), driver.GetThrottle(), driver.GetBraking())
    terrain.Synchronize(time) # Important for moving terrains, less so for static ones

    # Advance simulation
    driver.Advance(step_size)
    truck.Advance(step_size)
    terrain.Advance(step_size) # Same as above for static
    sys.DoStepDynamics(step_size)

    # End scene rendering
    vis.EndScene()

    if time > 300: # Max simulation time
        print("Simulation ended.")
        break

del sys # Explicitly delete system to ensure proper cleanup, though Python's GC usually handles it.
print("Simulation finished.")
```

**Explanation and Key Features:**

1.  **Chrono Data Path:** Crucial for PyChrono to find vehicle JSON files, textures, etc. It tries to get `CHRONO_DATA_DIR` from environment variables or defaults to a common relative path.
2.  **Parameters:** Most important settings are grouped at the top for easy modification.
3.  **ChSystem:** `ChSystemNSC` (Non-Smooth Contact) is chosen, suitable for vehicle dynamics on rigid surfaces. Solver settings are typical for such simulations.
4.  **MAN 10t Truck:**
    *   Loaded from `MAN_10t_Vehicle.json`. This file defines the chassis, suspension, steering, axles, and references to default tire and powertrain files.
    *   `Initialize()`: Sets its initial position and orientation.
    *   **Visualization Types:** You can choose between `MESH` (detailed visual models), `PRIMITIVES` (simpler shapes), or `NONE` for different components.
    *   **Collision Settings:** `SetChassisCollide` and `SetWheelCollide` enable/disable collision detection for these parts. For TMEASY tires, `SetWheelCollide(False)` is often preferred as the tire model itself handles interaction with the terrain. Wheel collision would be for the rim hitting objects.
    *   **TMEASY Tire Model:** `truck.SetTireType(veh.TireModelType_TMEASY)` selects the TMEASY model. The specific TMEASY *parameters* are typically loaded from JSON files referenced within the `MAN_10t_Vehicle.json` (e.g., `MAN_10t_TMeasyTire.json`).
    *   `truck.SetTireVisualizationType(veh.VisualizationType_MESH)` ensures the TMEASY tire meshes are visualized.
    *   **Powertrain:** A simple map-based powertrain is loaded from `MAN_10t_SimpleMapPowertrain.json`.
    *   `truck.InitializeTires()`: This is a critical step that actually creates and initializes the tire objects based on the selected `TireModelType` and associated parameter files.
5.  **Rigid Terrain:**
    *   `veh.RigidTerrain` creates a flat, rigid ground.
    *   `AddPatch()`: Defines a rectangular area with specified dimensions, material properties (friction, restitution), and a texture.
    *   The texture file (`Concrete.jpg`) is scaled using `terrain_texture_scale_x/y`.
6.  **Irrlicht Visualization:**
    *   `irr.ChVisualSystemIrrlicht` is the core visualization manager.
    *   `AttachSystem()`: Links the visualizer to the physics system.
    *   `AddLogo()`, `AddSkyBox()`, `AddTypicalLights()`: Standard scene enhancements. `AddTypicalLights()` creates directional lighting by default.
    *   `SetChaseCamera()`: Configures the parameters for one of the camera modes available through the `ChIrrGuiDriver`.
    *   `BindAll()`: Crucial step to create visualization shapes for all physical bodies in the system.
7.  **Real-Time Driver Controls:**
    *   `veh.ChIrrGuiDriver` provides an interactive GUI for controlling throttle, steering, and braking using keyboard inputs.
    *   It also handles camera mode switching (chase cam, fixed, inside view if available).
    *   `AttachVehicle()` connects the driver inputs to the truck.
8.  **Simulation Loop:**
    *   `vis.Run()`: Keeps the loop going as long as the Irrlicht window is open.
    *   `vis.BeginScene()` / `vis.EndScene()`: Standard Irrlicht rendering calls.
    *   `irr.ChIrrTools.drawGrid()`: Optional helper to visualize a grid on the terrain.
    *   **Synchronization:**
        *   `driver.Synchronize(time)`: Reads inputs from the GUI.
        *   `truck.Synchronize(time, steering, throttle, braking)`: Applies driver inputs to the vehicle model.
        *   `terrain.Synchronize(time)`: Updates terrain (mainly for moving terrains, but good practice).
    *   **Advancing State:**
        *   `driver.Advance(step_size)`
        *   `truck.Advance(step_size)`
        *   `terrain.Advance(step_size)`
        *   `sys.DoStepDynamics(step_size)`: The core physics integration step.

To run this script:
