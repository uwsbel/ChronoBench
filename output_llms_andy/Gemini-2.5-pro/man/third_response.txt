```python
import pychrono.core as chrono
import pychrono.irrlicht as irr
import pychrono.vehicle as veh
import math
# ------------- Added imports ------------
import pychrono.sensor as sens
import numpy as np
import random
# ----------------------------------------

# Set the path to the Chrono data files
# Ensure CHRONO_DATA_DIR is set in your environment variables or specify path directly
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

# Initial vehicle location and orientation
initLoc = chrono.ChVector3d(0, 0, 0.5)
initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# Visualization type for vehicle parts (PRIMITIVES, MESH, or NONE)
vis_type = veh.VisualizationType_MESH

# Collision type for chassis (PRIMITIVES, MESH, or NONE)
chassis_collision_type = veh.CollisionType_NONE # Kept as NONE as per original script

# Type of tire model (RIGID, TMEASY)
tire_model = veh.TireModelType_TMEASY

# Rigid terrain parameters
terrainHeight = 0      # Central height of the terrain plane
terrainLength = 100.0  # Size in X direction
terrainWidth = 100.0   # Size in Y direction
# Note: RigidTerrain creates a box, typically 0.1m thick by default.
# Top surface of terrain will be at terrainHeight + 0.05m if thickness is 0.1m and centered at terrainHeight.
terrain_top_surface_z = terrainHeight + 0.05 

# Point on chassis tracked by the camera
trackPoint = chrono.ChVector3d(-3.0, 0.0, 1.1) # Relative to chassis CoG

# Contact method
contact_method = chrono.ChContactMethod_NSC

# Simulation step sizes
step_size = 1e-3
tire_step_size = step_size

# Time interval between two render frames
render_step_size = 1.0 / 50  # FPS = 50

# Create the MAN vehicle, set parameters, and initialize
vehicle = veh.MAN_10t()
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
vehicle.SetTireType(tire_model)
vehicle.SetTireStepSize(tire_step_size)
vehicle.Initialize()

vehicle.SetChassisVisualizationType(vis_type)
vehicle.SetSuspensionVisualizationType(vis_type)
vehicle.SetSteeringVisualizationType(vis_type)
vehicle.SetWheelVisualizationType(vis_type)
vehicle.SetTireVisualizationType(vis_type)

# Set the collision system type for the entire Chrono system
vehicle.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# Create the terrain
patch_mat = chrono.ChContactMaterialNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)
terrain = veh.RigidTerrain(vehicle.GetSystem())

# Add patch with new texture
# The patch is a box centered at (0,0,terrainHeight)
patch_coordsys = chrono.ChCoordsysd(chrono.ChVector3d(0, 0, terrainHeight), chrono.QUNIT)
patch = terrain.AddPatch(patch_mat, patch_coordsys, terrainLength, terrainWidth)

# ------------- Terrain texture changed ------------
patch.SetTexture(veh.GetDataFile("terrain/textures/grass.jpg"), 200, 200) # Texture change
# ----------------------------------------------------
patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
terrain.Initialize()


# ------------- Create and add random boxes ------------
num_boxes = 20 # Number of random boxes
box_density = 1000 # kg/m^3
box_texture_file = chrono.GetChronoDataFile("textures/concrete.jpg")

for i in range(num_boxes):
    s_x = random.uniform(0.5, 1.5) # Box size X
    s_y = random.uniform(0.5, 1.5) # Box size Y
    s_z = random.uniform(0.5, 2.0) # Box size Z
    
    # Position boxes randomly, but not too close to the vehicle's start (0,0)
    # Place boxes in an area, e.g. +/- terrainLength/3, +/- terrainWidth/3
    p_x = random.uniform(-terrainLength / 3, terrainLength / 3)
    p_y = random.uniform(-terrainWidth / 3, terrainWidth / 3)
    
    # Create a clear path ~10m wide around origin for vehicle start
    if abs(p_x) < 5.0 and abs(p_y) < 5.0:
        p_x = (np.sign(p_x) if p_x != 0 else 1.0) * (abs(p_x) + 5.0)
        p_y = (np.sign(p_y) if p_y != 0 else 1.0) * (abs(p_y) + 5.0)
        
    # Place box on top of the terrain surface
    p_z = terrain_top_surface_z + s_z / 2.0 + 0.02 # Small epsilon to avoid initial penetration

    box_body = chrono.ChBodyEasyBox(s_x, s_y, s_z,
                                    box_density,    # density
                                    True,           # enable visualization
                                    True)           # enable collision
    box_body.SetPos(chrono.ChVector3d(p_x, p_y, p_z))
    
    # Optional: Set random orientation for boxes
    # rand_q = chrono.ChQuaterniond(random.uniform(0,1), random.uniform(0,1), random.uniform(0,1), random.uniform(0,1))
    # rand_q.Normalize()
    # box_body.SetRot(rand_q)

    box_body.GetMaterialSurfaceNSC().SetFriction(0.7)
    box_body.GetMaterialSurfaceNSC().SetRestitution(0.1)
    
    # Add texture to boxes
    if box_texture_file:
        box_ch_texture = chrono.ChTexture()
        box_ch_texture.SetTextureFilename(box_texture_file)
        # ChBodyEasyBox creates one visual shape (asset) at index 0
        vis_shape = box_body.GetVisualShape(0)
        if vis_shape:
             vis_shape.SetTexture(box_ch_texture)

    vehicle.GetSystem().Add(box_body)
# ------------------------------------------------------

# ------------- Add sensor manager and lidar sensor ------------
sensor_manager = sens.ChSensorManager(vehicle.GetSystem())
# sensor_manager.SetRaycaster(...) # Optionally set a specific raycaster type if needed (e.g. OptiX, Embree)
# sensor_manager.SetRaycastingSource(num_samples_x, num_samples_y) # For some raycasters
sensor_manager.SetVerbose(False)

# Get chassis body to attach sensor
chassis_body = vehicle.GetChassisBody()

# Define lidar parameters
lidar_update_rate = 10.0  # Hz
# Lidar offset pose, relative to chassis body's CoG
# Example: on top of the cabin, facing forward
# MAN_10t chassis CoG is relatively low. Cabin top could be ~2m above chassis CoG.
# Front of vehicle is positive X in chassis frame for MAN_10t.
lidar_pos_chassis = chrono.ChVector3d(3.5, 0, 2.0) # (Forward, Left, Up) from chassis CoG
lidar_rot_chassis = chrono.Q_from_AngX(0)         # Standard orientation (X-forward, Y-left, Z-up)
lidar_offset_pose = chrono.ChFrameD(lidar_pos_chassis, lidar_rot_chassis)

horizontal_samples = 360      # Number of samples per horizontal scan line
vertical_samples = 32         # Number of vertical scan lines (rings)
horizontal_fov = 2 * math.pi  # 360 degrees
max_vert_angle = math.pi / 12.0  # E.g., +15 degrees from horizontal
min_vert_angle = -math.pi / 6.0 # E.g., -30 degrees from horizontal
max_distance = 100.0          # Max sensing distance

lidar = sens.ChLidarSensor(
    chassis_body,         # Parent body sensor is attached to
    lidar_update_rate,    # Update rate in Hz
    lidar_offset_pose,    # Offset pose relative to parent body
    horizontal_samples,   # Number of horizontal samples
    vertical_samples,     # Number of vertical samples
    horizontal_fov,       # Horizontal field of view
    min_vert_angle,       # Minimum vertical angle
    max_vert_angle,       # Maximum vertical angle
    max_distance,         # Maximum sensing distance
    show_debug_points=True # Visualize LiDAR points in Irrlicht
)
lidar.SetName("LidarSensor")
# Optional: Add filters for data access or processing, e.g.:
# pointcloud_filter = sens.ChFilterPCfromLidar("LidarToPointCloud")
# lidar.PushFilter(pointcloud_filter) 
# This would allow access to point cloud data via pointcloud_filter.GetData()

sensor_manager.AddSensor(lidar)
# -----------------------------------------------------------


# Create the vehicle Irrlicht interface
vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle('MAN 10t Lidar Demo')
vis.SetWindowSize(1280, 1024)
vis.SetChaseCamera(trackPoint, 8.0, 0.7) # Adjusted chase distance and height for better view
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetVehicle())


# Create the driver system
driver = veh.ChInteractiveDriverIRR(vis)

# Set the time response for steering and throttle keyboard inputs.
steering_time = 1.0  # time to go from 0 to +1 (or from 0 to -1)
throttle_time = 1.0  # time to go from 0 to +1
braking_time = 0.3   # time to go from 0 to +1
driver.SetSteeringDelta(render_step_size / steering_time)
driver.SetThrottleDelta(render_step_size / throttle_time)
driver.SetBrakingDelta(render_step_size / braking_time)
driver.Initialize()

# Output vehicle mass
print( "VEHICLE MASS: ",  vehicle.GetVehicle().GetMass())

# Number of simulation steps between miscellaneous events
render_steps = math.ceil(render_step_size / step_size)

# Initialize simulation frame counter and real-time timer
realtime_timer = chrono.ChRealtimeStepTimer()
step_number = 0

# Simulation loop
while vis.Run() :
    time = vehicle.GetSystem().GetChTime()

    # Render scene (if it's time)
    if (step_number % render_steps == 0) :
        vis.BeginScene()
        vis.Render()
        vis.EndScene()

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Update modules (process inputs from other modules)
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    
    # ------------- Update sensor manager ------------
    sensor_manager.Update(time) # Sensor data is updated based on current system state
    # ------------------------------------------------
    
    vis.Synchronize(time, driver_inputs) # Visualization synchronized with current state

    # Advance simulation for one timestep for all modules
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size) # This advances Irrlicht visualization time/state, not physics

    # Increment step number
    step_number += 1

    # Spin in place for real time to catch up (if simulation is faster than real-time)
    realtime_timer.Spin(step_size)

# End of simulation; no specific cleanup needed for these Python objects typically
```
