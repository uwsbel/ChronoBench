
Here's a PyChrono simulation script demonstrating an epicyclic gear system with visualization:

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

def main():
    # Initialize system
    system = chrono.ChSystemNSC()
    system.Set_G_acc(chrono.ChVectorD(0, 0, 0))  # Zero gravity

    # Create materials
    steel = chrono.ChMaterialSurfaceNSC()
    steel.SetFriction(0.1)
    steel.SetRestitution(0.1)

    # Create ground body (truss)
    ground = chrono.ChBody()
    ground.SetBodyFixed(True)
    ground.SetPos(chrono.ChVectorD(0, 0, 0))
    
    # Create sun gear (fixed to ground)
    sun_radius = 0.5
    sun = chrono.ChBodyEasyCylinder(sun_radius, 0.1, 1000, True, True, steel)
    sun.SetPos(chrono.ChVectorD(0, 0, 0))
    sun.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI_2))
    sun.GetVisualShape(0).SetColor(chrono.ChColor(0.8, 0.2, 0.2))
    system.Add(sun)
    
    # Connect sun to ground
    sun_revolute = chrono.ChLinkLockRevolute()
    sun_revolute.Initialize(sun, ground, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0)))
    system.Add(sun_revolute)

    # Create rotating carrier bar
    carrier = chrono.ChBodyEasyBox(0.1, 1.0, 0.1, 1000, True, True, steel)
    carrier.SetPos(chrono.ChVectorD(0, 0, 0))
    carrier.GetVisualShape(0).SetColor(chrono.ChColor(0.2, 0.8, 0.2))
    system.Add(carrier)
    
    # Connect carrier to ground
    carrier_revolute = chrono.ChLinkLockRevolute()
    carrier_revolute.Initialize(carrier, ground, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0)))
    system.Add(carrier_revolute)

    # Add motor to carrier
    motor = chrono.ChLinkMotorRotationSpeed()
    motor.Initialize(carrier, ground, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))
    motor.SetSpeedFunction(chrono.ChFunction_Const(chrono.CH_C_PI / 3))  # 60 deg/s
    system.Add(motor)

    # Create planet gear
    planet_radius = 0.2
    planet = chrono.ChBodyEasyCylinder(planet_radius, 0.1, 1000, True, True, steel)
    planet.SetPos(chrono.ChVectorD(0, 0.7, 0))
    planet.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI_2))
    planet.GetVisualShape(0).SetColor(chrono.ChColor(0.2, 0.2, 0.8))
    system.Add(planet)
    
    # Connect planet to carrier
    planet_revolute = chrono.ChLinkLockRevolute()
    planet_revolute.Initialize(planet, carrier, 
                              chrono.ChCoordsysD(chrono.ChVectorD(0, 0.7, 0)))
    system.Add(planet_revolute)

    # Create visualization
    vis = chronoirr.ChVisualSystemIrrlicht()
    vis.AttachSystem(system)
    vis.SetWindowSize(1024, 768)
    vis.SetWindowTitle('Epicyclic Gears')
    vis.Initialize()
    vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
    vis.AddSkyBox()
    vis.AddCamera(chrono.ChVectorD(1, 0.5, 1))
    vis.AddTypicalLights()

    # Simulation loop
    time_step = 0.001
    while vis.Run():
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        system.DoStepDynamics(time_step)

if __name__ == "__main__":
    main()
```

This script creates:
1. A fixed ground body (truss)
2. A red sun gear fixed at the origin
3. A green rotating carrier bar connected to ground via revolute joint
4. A blue planet gear attached to the carrier's end
5. A motor enforcing constant rotation (60°/s) on the carrier

Key features:
- Uses NSC (Non-Smooth Contact) system for mechanical simulation
- Implements proper gear positioning (planet orbits sun)
- Includes realistic friction parameters
- Provides 3D visualization with Irrlicht
- Features colored objects and lighting effects
- Sets camera position for optimal viewing

The epicyclic gear motion is achieved through:
1. Sun gear fixed to ground
2. Carrier rotating via motor
3. Planet gear attached to carrier via revolute joint
4. Gravity disabled to focus on rotational dynamics

To run this, ensure you have PyChrono installed. The simulation will show the planet gear orbiting the sun gear while rotating due to contact interactions.
