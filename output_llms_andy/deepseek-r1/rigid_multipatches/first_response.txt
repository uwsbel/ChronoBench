
```python
"""PyChrono HMMWV Simulation on Complex Terrain"""
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

def main():
    # =========================================================================
    # Initialize Chrono system and environment
    # =========================================================================
    chrono.SetChronoDataPath(chrono.GetChronoDataPath() + '\\vehicle\\')
    system = chrono.ChSystemNSC()
    system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)
    system.SetSolverMaxIterations(150)
    system.SetMaxPenetrationRecoverySpeed(4.0)
    
    # =========================================================================
    # Create and configure HMMWV vehicle
    # =========================================================================
    # Initialize vehicle parameters
    init_pos = chrono.ChVectorD(0, 0.5, 0)
    init_rot = chrono.ChQuaternionD(1, 0, 0, 0)
    
    # Create vehicle system
    hmmwv = veh.HMMWV_Full(system)
    hmmwv.SetInitPosition(chrono.ChCoordsysD(init_pos, init_rot))
    hmmwv.SetEngineType(veh.EngineModelType_SIMPLE_MAP)
    hmmwv.SetTransmissionType(veh.TransmissionModelType_AUTOMATIC_SIMPLE_MAP)
    hmmwv.SetDriveType(veh.DrivelineTypeWV_AWD)
    hmmwv.SetTireType(veh.TireModelType_RIGID)
    
    # Initialize vehicle systems
    hmmwv.Initialize()
    
    # Set visualization modes
    hmmwv.SetChassisVisualizationType(veh.VisualizationType_MESH)
    hmmwv.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
    hmmwv.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
    hmmwv.SetWheelVisualizationType(veh.VisualizationType_MESH)
    hmmwv.SetTireVisualizationType(veh.VisualizationType_MESH)
    
    # =========================================================================
    # Create complex terrain with multiple patches
    # =========================================================================
    terrain = veh.RigidTerrain(system)
    
    # Patch 1: Flat textured terrain
    patch1_mat = chrono.ChMaterialSurfaceNSC()
    patch1 = terrain.AddPatch(patch1_mat, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 1, 0), 50, 30)
    patch1.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 50, 30)
    patch1.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
    
    # Patch 2: Second flat terrain with different texture
    patch2 = terrain.AddPatch(patch1_mat, chrono.ChVectorD(50, 0, -20), chrono.ChVectorD(0, 1, 0), 30, 40)
    patch2.SetTexture(veh.GetDataFile("terrain/textures/concrete.jpg"), 30, 40)
    patch2.SetColor(chrono.ChColor(0.7, 0.7, 0.7))
    
    # Patch 3: Mesh-based bump obstacle
    patch3 = terrain.AddPatch(patch1_mat, chrono.ChVectorD(30, 0, 20), chrono.ChVectorD(0, 1, 0), 10, 10)
    mesh = chrono.ChTriangleMeshConnected()
    mesh.LoadWavefrontMesh(chrono.GetChronoDataFile("models/bump.obj"))
    patch3.SetMesh(mesh)
    patch3.SetTexture(veh.GetDataFile("terrain/textures/dirt.jpg"), 10, 10)
    
    # Patch 4: Heightmap terrain
    patch4_mat = chrono.ChMaterialSurfaceNSC()
    patch4_mat.SetFriction(0.9)
    patch4 = terrain.AddPatch(
        patch4_mat, chrono.CSYSNORM, 
        veh.GetDataFile("terrain/height_maps/bump64.bmp"), 
        "heightmap", 40, 40, 0, 5
    )
    patch4.SetTexture(veh.GetDataFile("terrain/textures/grass.jpg"), 40, 40)
    
    terrain.Initialize()
    
    # =========================================================================
    # Create Irrlicht visualization
    # =========================================================================
    vis = irr.ChVisualSystemIrrlicht()
    vis.AttachSystem(system)
    vis.SetWindowSize(1280, 720)
    vis.SetWindowTitle("HMMWV on Complex Terrain")
    vis.Initialize()
    vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
    vis.AddSkyBox()
    vis.AddCamera(chrono.ChVectorD(6, 3, 6))
    vis.AddTypicalLights()
    
    # Add visual assets
    vis.AddModel(chrono.GetChronoDataFile('models/trees.obj'), chrono.ChVectorD(20,0,10))
    vis.AddModel(chrono.GetChronoDataFile('models/trees.obj'), chrono.ChVectorD(40,0,-5))
    
    # =========================================================================
    # Create interactive driver system
    # =========================================================================
    driver = veh.ChInteractiveDriverIRR(vis)
    driver.SetSteeringDelta(0.02)
    driver.SetThrottleDelta(0.02)
    driver.SetBrakingDelta(0.06)
    driver.Initialize()
    
    # =========================================================================
    # Simulation loop parameters
    # =========================================================================
    step_size = 0.02
    realtime_timer = chrono.ChRealtimeStepTimer()
    
    while vis.Run():
        time = system.GetChTime()
        
        # Update visualization and driver inputs
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        
        # Get driver inputs
        driver_inputs = driver.GetInputs()
        
        # Update vehicle systems
        driver.Synchronize(time)
        terrain.Synchronize(time)
        hmmwv.Synchronize(time, driver_inputs, terrain)
        
        # Advance simulation
        driver.Advance(step_size)
        terrain.Advance(step_size)
        hmmwv.Advance(step_size)
        system.DoStepDynamics(step_size)
        
        # Maintain real-time simulation
        realtime_timer.Spin(step_size)

if __name__ == "__main__":
    main()
```

This script creates a PyChrono simulation featuring:

1. **HMMWV Vehicle**:
   - Initialized at position (0, 0.5, 0)
   - Configured with simple engine model, automatic transmission, and AWD drivetrain
   - Rigid tires and mesh visualization for chassis/wheels

2. **Complex Terrain**:
   - Flat textured patch (tile texture)
   - Secondary flat patch (concrete texture)
   - Mesh-based bump obstacle (dirt texture)
   - Heightmap terrain (grass texture with elevation variations)

3. **Interactive Driver**:
   - Steering/throttle/braking controls via Irrlicht GUI
   - Real-time input handling during simulation

4. **Visualization**:
   - Irrlicht rendering system with skybox and lighting
   - Additional environment models (trees)
   - Camera positioned for optimal viewing

The simulation runs with a 0.02-second timestep using the Barzilai-Borwein solver and maintains real-time performance. All vehicle components and terrain patches have appropriate visualization properties set.
