# Create Irrlicht visualization

vis_app.AddCamera(chrono.ChVectorD(5, 5, 5), chrono.ChVectorD(0, 0, 0))
vis_app.SetAmbientLight(chrono.ChColor(0.2, 0.2, 0.2))
