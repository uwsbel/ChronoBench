  ```python
      def __init__(self, topic):
          super().__init__(1)
          self.topic = topic
          self.publisher = None
          self.ticker = 0
          self.message = "Hello, world! At time: "
      
      def Initialize(self, interface):
          print(f"Creating publisher for topic {self.topic} ...")
          self.publisher = interface.GetNode().create_publisher(String, self.topic, 1)
          return True
      
      def Tick(self, time):
          print(self.message + str(self.ticker))
          msg = String()
          msg.data = self.message + str(self.ticker)
          self.publisher.publish(msg)
          self.ticker += 1
  ```

  - `Tick()` publishes a `String` message, concatenating a message with a counter.
