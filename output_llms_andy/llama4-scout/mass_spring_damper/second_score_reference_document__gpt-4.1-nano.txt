  class MySpringForce(chrono.ForceFunctor):

          self.damping_coef = damping_coef

      def evaluate(self, force, torque, time, body1, body2, pos1, pos2, vel1, vel2):
          dist = pos2 - pos1
          vel_dist = vel2 - vel1
          force_magnitude = -self.spring_coef * dist.Length() - self.damping_coef * vel_dist.Dot(dist) / dist.Length()
