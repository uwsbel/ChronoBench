### PyChrono Simulation: Folding Tablecloth with Shell Elements

```python
import chrono as ch
import chrono.irrlicht as chirr
import numpy as np

# 1. Initialize the PyChrono environment and core components
ch.set_default_G_acc(0, 0, -9.81)  # Set gravity

# 2. Add the required physical systems and objects
sys = ch.ChSystemSMC()

# Create the Irrlicht application
app = chirr.IrrlichtApplication(sys)
app.setTimestep(0.01)
app.set_window_size(ch.ChVector2(800, 600))
app.set_caption('Folding Tablecloth Simulation')
app.show_stats()

# 3. Set up the isotropic Kirchhoff material
mat = ch.ChMaterialShellKirchhoffSMC()
mat.SetDensity(1000)  # kg/m^3
mat.SetYoungModulus(100e3)  # Pa
mat.SetPoissonRatio(0.3)

# Generate a mesh grid for the tablecloth
grid_size = 10
num_nodes = (grid_size + 1) ** 2
num_elements = grid_size ** 2

nodes = []
elements = []

# Add nodes
for i in range(grid_size + 1):
    for j in range(grid_size + 1):
        node = ch.ChNodeFEAxyzP(
            ch.ChVector3(i - grid_size // 2, j - grid_size // 2, 0),
            ch.ChQuaternion(1, 0, 0, 0)
        )
        sys.AddNode(node)
        nodes.append(node)

# Add elements
for i in range(grid_size):
    for j in range(grid_size):
        n1 = nodes[i * (grid_size + 1) + j]
        n2 = nodes[(i + 1) * (grid_size + 1) + j]
        n3 = nodes[(i + 1) * (grid_size + 1) + (j + 1)]
        n4 = nodes[i * (grid_size + 1) + (j + 1)]

        elem = ch.ChElementShellReissner4(
            n1, n2, n3, n4,
            mat,
            ch.ChVector3(0, 0, 1)
        )
        sys.AddElement(elem)
        elements.append(elem)

# Set up PardisoMKL solver
sys.SetSolverType(ch.ChSolver.Type.PARDISO_MKL)

# Add a floor body to interact with the tablecloth
floor_body = ch.ChBodyEasyBody(
    ch.ChCollisionShapeBox(100, 100, 0.1),
    1000
)
floor_body.SetPos(ch.ChVector3(0, 0, -5))
sys.AddBody(floor_body)

# Set contact material for the floor and tablecloth
contact_mat = ch.ChContactMaterialSMC()
contact_mat.SetFriction(0.5)
contact_mat.SetRestitutionCoeff(0.5)

# Create a contact surface for the floor
floor_shape = ch.ChCollisionShapeBox(
    ch.ChVector3(100, 100, 0.1),
    ch.ChVector3(0, 0, -5)
)
floor_shape.SetContactMaterial(contact_mat)
floor_body.AddCollisionShape(floor_shape)

# Create a contact surface for the tablecloth
tablecloth_shape = ch.ChCollisionShapeTriangleMesh(
    ch.ChTriangleMeshConnected(),
    contact_mat
)

# Initialize the triangle mesh
mesh = ch.ChTriangleMeshConnected()
for i, elem in enumerate(elements):
    n1, n2, n3, n4 = elem.GetNodesXYZP()
    tri1 = ch.ChTriangle(n1, n2, n3)
    tri2 = ch.ChTriangle(n1, n3, n4)
    mesh.AddTriangle(tri1)
    mesh.AddTriangle(tri2)

tablecloth_shape.SetTriangleMesh(mesh)
for elem in elements:
    n1, n2, n3, n4 = elem.GetNodesXYZP()
    tablecloth_shape.AddTriangle(
        ch.ChTriangle(n1, n2, n3),
        ch.ChVector3(0, 0, 0)
    )
    tablecloth_shape.AddTriangle(
        ch.ChTriangle(n1, n3, n4),
        ch.ChVector3(0, 0, 0)
    )

# Add the contact surface to the tablecloth nodes
for node in nodes:
    node.AddCollisionShape(tablecloth_shape)

# Run the simulation
while app.run():
    sys.DoStepDynamics(0.01)
```

### Notes:
*   The simulation models a tablecloth using shell elements with an isotropic Kirchhoff material.
*   The mesh grid for the tablecloth is generated with a specified grid size.
*   The PardisoMKL solver is used for efficient dynamics handling.
*   An Irrlicht visualization system is set up to display the simulation.

### Usage:
*   Run the script using Python (e.g., `python tablecloth_simulation.py`).
*   The simulation will display in an Irrlicht window.
*   You can adjust parameters such as grid size, material properties, and simulation time step to explore different scenarios.
