Below is a complete, self-contained Python script that follows the requested guidelines.  
It brings-up a full Chrono simulation featuring an M113 tracked vehicle, a rigid terrain with user–defined surface properties, an interactive GUI driver, Irrlicht visualization (skybox, lights, chase-camera) and a real-time synchronized simulation loop.

```python
# =============================================================================
#  M113 tracked–vehicle demo
#  ------------------------------------------------
#  * Rigid–body contact              :  NSC (penalty)
#  * Terrain                         :  Rigid patch, user μ / e
#  * Driver                          :  Interactive (keyboard)
#  * Visualizer                      :  Irrlicht
#  * Real-time loop                  :  Chrono.RealtimeStepTimer
# =============================================================================
#
#  Controls inside the Irrlicht window
#  -----------------------------------
#    W / S    : throttle ↑ / ↓
#    A / D    : steer left / right
#    Space    : full brake
#    F1       : toggle driver GUI help
#
# =============================================================================

import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# ----------------------------------------------------------------------------- 
# 1. Chrono global settings and paths
# -----------------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(veh.GetDataPath())

# ----------------------------------------------------------------------------- 
# 2. Simulation parameters (feel free to tweak)
# -----------------------------------------------------------------------------
step_size      = 1.0e-3          # integrator step [s]
render_step    = 1.0/60.0        # draw at 60 fps
t_end          = 60.0            # stop after N seconds (press ESC to quit earlier)

# Terrain contact properties
terrain_mu     = 0.8             # friction coefficient
terrain_e      = 0.01            # restitution coefficient (bounciness)

# ----------------------------------------------------------------------------- 
# 3. Create the M113 vehicle
# -----------------------------------------------------------------------------
contact_method = chrono.ChContactMethod_NSC
shoe_type      = veh.TrackShoeType.BAND_BUSHING        # realistic rubber band
chassis_fixed  = False

vehicle = veh.M113(shoe_type)
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisFixed(chassis_fixed)
vehicle.SetTrackAssemblyType(veh.SuspensionType.TORSION_BAR)
vehicle.SetInitPosition(
    chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0.5), chrono.QUNIT)  # (x,y,z), quaternion
)
vehicle.Initialize()

# Shortcut to the Chrono system holding all bodies
system = vehicle.GetSystem()

# ----------------------------------------------------------------------------- 
# 4. Rigid terrain
# -----------------------------------------------------------------------------
terrain = veh.RigidTerrain(system)

patch = terrain.AddPatch(
    chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT),   # location/orientation
    chrono.ChVectorD(400, 400, 0.02),                              # size (length, width, thickness)
)
patch.SetContactFrictionCoefficient(terrain_mu)
patch.SetContactRestitutionCoefficient(terrain_e)
patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 40, 40)

terrain.Initialize()

# ----------------------------------------------------------------------------- 
# 5. Irrlicht application (visual interface)
# -----------------------------------------------------------------------------
app = veh.ChTrackedVehicleIrrApp(
    vehicle,                                   # the tracked vehicle we just built
    "M113 simulation demo",                    # window title
    irr.dimension2du(1280, 720)                # window size
)

app.SetSkyBox()
app.AddTypicalLights(                                     # two white lights
    chrono.ChVectorD(+120, +100, 200),                   # position 1
    chrono.ChVectorD(-120, -100, 200), 250, 130          # position 2, radius, attenuation
)
app.SetChaseCamera(chrono.ChVectorD(0, 0, 0.0),          # relative position on chassis
                  8.0,                                   # chase distance
                  0.5)                                   # min distance clamp
app.Initialize()
app.AddLogo()                                            # Chrono / Irr logo

# ----------------------------------------------------------------------------- 
# 6. Interactive GUI driver (keyboard)
# -----------------------------------------------------------------------------
driver = veh.ChIrrGuiDriver(app)
driver.SetTimestep(step_size)
driver.Initialize()

# Let the driver system produce a help overlay
print("\nControls:")
print("  W/S  : throttle up / down")
print("  A/D  : steer left / right")
print("  SPACE: full brake\n")

# ----------------------------------------------------------------------------- 
# 7. Real-time simulation loop
# -----------------------------------------------------------------------------
realtime_timer   = chrono.ChRealtimeStepTimer()
render_accum     = 0.0

while app.GetDevice().run():

    # ----- end condition ------------------------------------------------------
    if system.GetChTime() >= t_end:
        break

    # ----- Collect driver inputs ---------------------------------------------
    driver_inputs = driver.GetInputs()

    # ----- Simulation amount to advance this frame ---------------------------
    step = step_size

    # ----- Synchronize all modules -------------------------------------------
    time = system.GetChTime()
    driver.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    terrain.Synchronize(time)
    app.Synchronize("M113 demo", driver_inputs)

    # ----- Advance the state of each module ----------------------------------
    driver.Advance(step)
    vehicle.Advance(step)
    terrain.Advance(step)
    app.Advance(step)

    # ----- Render (at ~60 Hz) -------------------------------------------------
    render_accum += step
    if render_accum >= render_step:
        app.BeginScene(True, True, chrono.SColor(255, 140, 161, 192))
        app.DrawAll()
        app.EndScene()
        render_accum = 0.0

    # ----- Real-time pace enforcement ----------------------------------------
    if not realtime_timer.Spin(step):
        # (optional) If running slower than real-time you could break here
        pass

# ----------------------------------------------------------------------------- 
# 8. Clean exit
# -----------------------------------------------------------------------------
print("Simulation finished.")
```

How to run
----------
1. Install Project Chrono with Python bindings (`pychrono`) and its Irrlicht add-on.  
2. Save this script as `m113_demo.py`.  
3. Execute with: `python m113_demo.py`.

The window will open with the M113 resting on a large rigid patch.  
Use W/S for throttle, A/D for steering and Space for braking.  
Chrono’s `ChRealtimeStepTimer` keeps the loop in (or as close as possible to) real time.
