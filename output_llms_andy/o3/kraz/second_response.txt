Corrected & updated PyChrono script
-------------------------------------------------
The script below
  • fixes a few API-name problems found in the original code  
    – correct chassis collision enum  
    – correct calls that required only one parameter (tire visualization)  
  • applies all requested changes (start position, camera settings, …)  
  • replaces the interactive driver with a simple programmed driver that
    executes a double-lane-change manoeuvre.

```python
import math
import pychrono.core as chrono
import pychrono.irrlicht as irr
import pychrono.vehicle as veh

# ---------------------------------------------------------------------
# Chrono/Vehicle initialisation
# ---------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

# ---------------------------------------------------------------------
# Vehicle initial position and orientation
# ---------------------------------------------------------------------
initLoc = chrono.ChVector3d(-15, 0, 0.5)          # CHANGED      (-15,0,0.5)
initRot = chrono.ChQuaterniond(1, 0, 0, 0)        # identity (no rotation)

# ---------------------------------------------------------------------
# Visualisation and collision parameters
# ---------------------------------------------------------------------
vis_type               = veh.VisualizationType_MESH
chassis_collision_type = veh.ChassisCollisionType_NONE          # FIXED enum
tire_model             = veh.TireModelType_TMEASY

# ---------------------------------------------------------------------
# Terrain definition (rigid patch)
# ---------------------------------------------------------------------
terrainHeight = 0
terrainLength = 100.0
terrainWidth  = 100.0

# ---------------------------------------------------------------------
# Camera parameters
# ---------------------------------------------------------------------
trackPoint = chrono.ChVector3d(3, 0, 2.1)         # CHANGED
chase_dist = 25.0                                 # unchanged
chase_hgt  = 10.5                                 # CHANGED

# ---------------------------------------------------------------------
# Contact, step sizes, rendering
# ---------------------------------------------------------------------
contact_method     = chrono.ChContactMethod_NSC
step_size          = 1e-3
render_step_size   = 1.0 / 50.0                   # 50 FPS
render_steps       = math.ceil(render_step_size / step_size)

# ---------------------------------------------------------------------
# Create the Kraz vehicle
# ---------------------------------------------------------------------
vehicle = veh.Kraz()
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
vehicle.Initialize()

vehicle.SetChassisVisualizationType(vis_type, vis_type)
vehicle.SetSteeringVisualizationType(vis_type)
vehicle.SetSuspensionVisualizationType(vis_type, vis_type)
vehicle.SetWheelVisualizationType(vis_type, vis_type)
vehicle.SetTireVisualizationType(vis_type)        # FIXED (only one arg)

# Use Bullet collision system
vehicle.GetSystem().SetCollisionSystemType(
    chrono.ChCollisionSystem.Type_BULLET)

# ---------------------------------------------------------------------
# Rigid terrain
# ---------------------------------------------------------------------
patch_mat = chrono.ChContactMaterialNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)

terrain = veh.RigidTerrain(vehicle.GetSystem())
patch   = terrain.AddPatch(
            patch_mat,
            chrono.ChCoordsysd(chrono.ChVector3d(0, 0, 0), chrono.QUNIT),
            terrainLength, terrainWidth)
patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)
patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
terrain.Initialize()

# ---------------------------------------------------------------------
# Irrlicht visual system
# ---------------------------------------------------------------------
vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle('Kraz – Double Lane Change')
vis.SetWindowSize(1280, 1024)
vis.SetChaseCamera(trackPoint, chase_dist, chase_hgt)   # CHANGED
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetTractor())                 # follow the tractor

# ---------------------------------------------------------------------
# Driver: programmed double-lane-change
# ---------------------------------------------------------------------
class DoubleLaneChangeDriver(veh.ChDriver):
    """Simple driver that generates a double lane change manoeuvre"""
    def __init__(self, vehicle):
        super(DoubleLaneChangeDriver, self).__init__(vehicle)
        self.throttle = 0.6

    # Called every time step by the simulation loop
    def Synchronize(self, time):
        # Piece-wise linear steering profile
        steer = 0.0
        # First lane change to the left (positive steer) 1 – 4 s
        if 1.0 <= time < 2.5:
            steer = 0.35 * (time - 1.0) / (2.5 - 1.0)           # ramp up
        elif 2.5 <= time < 4.0:
            steer = 0.35 * (4.0 - time) / (4.0 - 2.5)           # ramp down
        # Second lane change to the right (negative steer) 4 – 7 s
        elif 4.0 <= time < 5.5:
            steer = -0.35 * (time - 4.0) / (5.5 - 4.0)          # ramp up
        elif 5.5 <= time < 7.0:
            steer = -0.35 * (7.0 - time) / (7.0 - 5.5)          # ramp down

        # Populate the protected member m_inputs
        self.m_inputs.m_steering = steer
        self.m_inputs.m_throttle = self.throttle
        self.m_inputs.m_braking  = 0.0

    def Advance(self, step):
        pass   # nothing fancy to do here

# Instantiate the driver
driver = DoubleLaneChangeDriver(vehicle.GetTractor())
driver.Initialize()

# Output some vehicle info
print("VEHICLE MASS:", vehicle.GetTractor().GetMass())

# ---------------------------------------------------------------------
# Simulation loop
# ---------------------------------------------------------------------
realtime_timer = chrono.ChRealtimeStepTimer()
step_number    = 0
render_frame   = 0

while vis.Run():
    time = vehicle.GetSystem().GetChTime()

    # Render scene
    if step_number % render_steps == 0:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        render_frame += 1

    # Driver inputs (created inside driver.Synchronize)
    driver.Synchronize(time)
    driver_inputs = driver.GetInputs()

    # Synchronise other modules
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    vis.Synchronize(time, driver_inputs)

    # Advance modules
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    # Bookkeeping
    step_number += 1
    realtime_timer.Spin(step_size)
```

Key changes & fixes
1. Initial vehicle location set to (-15, 0, 0.5).  
2. Camera track point moved to (3, 0, 2.1); chase height increased to 10.5 m.  
3. Correct enumeration name for chassis collision (`veh.ChassisCollisionType_NONE`).  
4. Corrected call to `SetTireVisualizationType` (only one argument).  
5. Replaced keyboard driver by a scripted `DoubleLaneChangeDriver` that automatically
   performs a double lane-change manoeuvre through a time-based steering profile while
   maintaining constant throttle.
