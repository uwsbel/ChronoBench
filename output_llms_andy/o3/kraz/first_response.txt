```python
# =============================================================================
#  PyChrono simulation script
#  Kraz 64431 6x6 truck on rigid terrain + Irrlicht real-time visualization
#
#  Requirements:
#     pip install pychrono
#     (Chrono must be compiled/installed with the VEHICLE and IRRLICHT modules)
# =============================================================================
import math
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr


# -----------------------------------------------------------------------------
# 1. Chrono initialisation
# -----------------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())          # core Chrono data
veh.SetDataPath(chrono.GetChronoDataPath() + "vehicle/")      # vehicle data

contact_method        = chrono.ChContactMethod_NSC            # NSC contact
system_track_time_step = 2e-3                                 # integrator step
render_fps             = 50                                   # Irrlicht FPS

# -----------------------------------------------------------------------------
# 2. Create the vehicle (Kraz 64431)
# -----------------------------------------------------------------------------
print("Initialising Kraz vehicle ...")
vehicle = veh.Kraz(contact_method,               # contact formulation
                   True,                         # chassis collision?
                   chrono.ChassisCollisionType_NONE)

init_pos = chrono.ChVectorD(0.0, 0.0, 0.5)       # (x,y,z)
init_rot = chrono.ChQuaternionD(1, 0, 0, 0)      # no yaw/pitch/roll
vehicle.Initialize(chrono.ChCoordsysD(init_pos, init_rot))

# Optional tweaks -------------------------------------------------------------
vehicle.SetChassisFixed(False)                   # make truck free to move
vehicle.SetTireType(veh.TireType_RIGID)          # rigid‐body tires

# -----------------------------------------------------------------------------
# 3. Create rigid terrain
# -----------------------------------------------------------------------------
print("Creating rigid terrain ...")
terrain = veh.RigidTerrain(vehicle.GetSystem())

# Material with desired contact properties
mat = chrono.ChMaterialSurfaceNSC()
mat.SetFriction(0.9)
mat.SetRestitution(0.01)

# An infinite horizontal patch (200 m × 200 m plane is plenty)
patch = terrain.AddPatch(mat,
                         chrono.ChVectorD(0, 0, 0),           # position
                         chrono.ChVectorD(0, 0, 1),           # normal
                         200, 200)                            # length, width
patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)
patch.SetColor(chrono.ChColor(0.6, 0.6, 0.6))
terrain.Initialize()

# -----------------------------------------------------------------------------
# 4. Irrlicht visualisation app
# -----------------------------------------------------------------------------
print("Starting Irrlicht application ...")
app = veh.ChWheeledVehicleIrrApp(vehicle,
                                 "Kraz Rigid-Terrain Demo",
                                 irr.dimension2du(1280, 720))

app.SetSkyBox()
app.AddTypicalLights(chrono.ChVectorD(30, 100, 30),
                     chrono.ChVectorD(-30, 100, 30),
                     250, 130)
app.AddLightWithShadow(chrono.ChVectorD(60, 120, 60),
                       chrono.ChVectorD(0, 0, 0),
                       100, 1, 80, 80)
app.SetChaseCamera(chrono.ChVectorD(0, 0.5, 0),   # relative position to cab
                   8.0,                           # chase distance
                   0.5)                           # elevation angle
app.SetStepsize(system_track_time_step)
app.Initialize()

# -----------------------------------------------------------------------------
# 5. Driver (keyboard GUI)
# -----------------------------------------------------------------------------
driver = veh.ChIrrGuiDriver(app)
driver.SetSteeringDelta(0.02)
driver.SetThrottleDelta(0.02)
driver.SetBrakingDelta(0.1)
driver.Initialize()

# -----------------------------------------------------------------------------
# 6. Real-time simulation loop
# -----------------------------------------------------------------------------
realtime_timer = chrono.ChRealtimeStepTimer()
render_interval = math.ceil((1.0 / render_fps) / system_track_time_step)
step_number = 0

print("Simulation loop started. Press W/A/S/D, Space (brake), and P (parking).")
while app.GetDevice().run():

    # Irrlicht: begin drawing
    if step_number % render_interval == 0:
        app.BeginScene()
        app.DrawAll()
        app.EndScene()

    # Current Chrono time
    time = vehicle.GetSystem().GetChTime()

    # ------------------------------------------------------------------
    # Synchronise modules (order is not strictly enforced but sensible)
    # ------------------------------------------------------------------
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver.GetInputs(), terrain)
    app.Synchronize("Kraz demo", driver.GetInputs())

    # Advance simulation state
    driver.Advance(system_track_time_step)
    terrain.Advance(system_track_time_step)
    vehicle.Advance(system_track_time_step)
    app.Advance(system_track_time_step)

    # Spin in real time
    realtime_timer.Spin(system_track_time_step)
    step_number += 1

print("Demo terminated.")
```

Explanation of the main sections
1. Chrono initialisation – loads the data path and selects the non-smooth contact (NSC) formulation.
2. Vehicle – instantiates the Kraz 6×6 truck, positions it 0.5 m above the ground, and lets all sub-systems (suspension, tires, driveline, etc.) initialise themselves.
3. Terrain – builds one large rigid patch with user-defined friction (0.9) and restitution (0.01).
4. Irrlicht – makes a window, adds a sky box, two omnidirectional lights and one shadow-casting light, and defines a chase camera that follows the truck.
5. Driver – a simple keyboard driver created by `ChIrrGuiDriver` (arrow/WASD keys, braking, etc.). Sensitivity deltas are tuned for smoother inputs.
6. Simulation loop – runs at the chosen step size (2 ms) while trying to maintain real-time execution through `ChRealtimeStepTimer`. Each iteration:
   • Obtains new driver inputs  
   • Synchronises vehicle, terrain, driver, and Irrlicht subsystems  
   • Advances them by one time step  
   • Draws the scene at the requested FPS.

The script is completely self-contained; simply execute it to watch the Kraz drive around on a flat rigid surface in real time.
