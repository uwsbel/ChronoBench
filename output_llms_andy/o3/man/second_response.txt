Corrected & updated PyChrono script
------------------------------------------------
```python
import math
import pychrono.core as chrono
import pychrono.irrlicht as irr          # keeps Irrlicht symbols in scope
import pychrono.vehicle as veh


# ---------------------------------------------------------------------
# Path to data files
# ---------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

# ---------------------------------------------------------------------
# Simulation parameters
# ---------------------------------------------------------------------
# 1.  Vehicle initial location  (   CHANGED   )
initLoc = chrono.ChVector3d(-20, 0, 1.5)          # [m]
initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# 2.  Visualization / collision / tire model
vis_type               = veh.VisualizationType_MESH
chassis_collision_type = veh.CollisionType_NONE
tire_model             = veh.TireModelType_TMEASY

# 3.  Terrain parameters
terrainLength = 100.0                              # [m] X-direction size
terrainWidth  = 100.0                              # [m] Y-direction size

# 4.  Other global settings
trackPoint      = chrono.ChVector3d(-3.0, 0.0, 1.1)
contact_method  = chrono.ChContactMethod_NSC
step_size       = 1e-3                             # [s]
render_step     = 1.0 / 50                         # 50 FPS
tire_step_size  = step_size

# ---------------------------------------------------------------------
# Create the MAN 5-ton vehicle                (   CHANGED   )
# ---------------------------------------------------------------------
vehicle = veh.MAN_5t()                                   # vehicle type changed
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
vehicle.SetTireType(tire_model)
vehicle.SetTireStepSize(tire_step_size)
vehicle.Initialize()

vehicle.SetChassisVisualizationType(vis_type)
vehicle.SetSuspensionVisualizationType(vis_type)
vehicle.SetSteeringVisualizationType(vis_type)
vehicle.SetWheelVisualizationType(vis_type)
vehicle.SetTireVisualizationType(vis_type)

# optional: switch to Bullet collision system (more robust)
vehicle.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# ---------------------------------------------------------------------
# Create a RIGID TERRAIN based on a HEIGHT MAP   (   CHANGED   )
# ---------------------------------------------------------------------
patch_mat = chrono.ChContactMaterialNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)

terrain = veh.RigidTerrain(vehicle.GetSystem())

# Height-map image (gray-scale).  A file that ships with Chrono data is used.
height_map_file = veh.GetDataFile("terrain/height_maps/bump64.png")

# Add the height-map patch:
#     h_min = 0   => black pixels
#     h_max = 3   => white pixels   (3-m tall “hills”)
patch = terrain.AddPatch(
    patch_mat,
    height_map_file,
    chrono.ChCoordsysd(chrono.ChVector3d(0, 0, 0), chrono.QUNIT),
    terrainLength,
    terrainWidth,
    0.0,        # h_min  [m]
    3.0)        # h_max  [m]

# Apply grass texture  (   CHANGED   )
patch.SetTexture(veh.GetDataFile("terrain/textures/grass.jpg"), 200, 200)

terrain.Initialize()

# ---------------------------------------------------------------------
# Create Irrlicht visualization
# ---------------------------------------------------------------------
vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle('MAN 5t – height-map terrain')
vis.SetWindowSize(1280, 1024)
vis.SetChaseCamera(trackPoint, 15.0, 0.5)
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetVehicle())

# ---------------------------------------------------------------------
# Interactive driver (keyboard)
# ---------------------------------------------------------------------
driver = veh.ChInteractiveDriverIRR(vis)

steering_time = 1.0     # time to go from 0 to ±1 steering
throttle_time = 1.0     # time to go from 0 to +1 throttle
braking_time  = 0.3     # time to go from 0 to +1 braking

driver.SetSteeringDelta(render_step / steering_time)
driver.SetThrottleDelta(render_step / throttle_time)
driver.SetBrakingDelta(render_step  / braking_time)
driver.Initialize()

# ---------------------------------------------------------------------
# Informational printout
# ---------------------------------------------------------------------
print("VEHICLE MASS :", vehicle.GetVehicle().GetMass(), "kg")

# ---------------------------------------------------------------------
# Simulation loop
# ---------------------------------------------------------------------
render_steps   = math.ceil(render_step / step_size)
step_number    = 0
render_frame   = 0
realtime_timer = chrono.ChRealtimeStepTimer()

while vis.Run():
    time = vehicle.GetSystem().GetChTime()

    # Render scene
    if step_number % render_steps == 0:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        render_frame += 1

    # Collect driver inputs
    driver_inputs = driver.GetInputs()

    # Synchronize all modules
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    vis.Synchronize(time, driver_inputs)

    # Advance simulation
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    # Increment step counter and maintain real time
    step_number += 1
    realtime_timer.Spin(step_size)
```

What was fixed / changed
1. Corrected enumeration & function names (all are consistent with current Chrono-Python API).
2. Moved the initial vehicle location to `(-20, 0, 1.5)`.
3. Switched the vehicle model from `veh.MAN_10t()` to `veh.MAN_5t()`.
4. Replaced the flat patch with a rigid height-map terrain (`bump64.png`) that creates 3-m-high hills.
5. Texture for the terrain changed from `"tile4.jpg"` to `"grass.jpg"`.
6. Added explanatory comments and grouped code for clarity; removed a few unused variables.
