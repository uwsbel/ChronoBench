```python
#
#  Corrected and modified PyChrono script
#  (see comments marked with  <<< FIX  >>>  or  <<< NEW  >>>)
#

import math
import pychrono.core as chrono
import pychrono.irrlicht as irr
import pychrono.vehicle as veh
import pychrono.sensor as sens

# -------------------------------------------------------------------------
# Misc. helper functions
# -------------------------------------------------------------------------

def q_from_angle_axis(angle, axis):
    """Utility wrapper because the python binding is called Q_from_AngAxis."""
    return chrono.Q_from_AngAxis(angle, axis)


# -------------------------------------------------------------------------
# Chrono & Vehicle initialisation
# -------------------------------------------------------------------------

chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

# Initial vehicle location and orientation
initLoc = chrono.ChVector3d(0, 0, 0.5)
initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# Visualisation / collision / tyre setup
vis_type               = veh.VisualizationType_MESH
chassis_collision_type = veh.CollisionType_NONE
tire_model             = veh.TireModelType_TMEASY

# Terrain parameters
terrainHeight = 0
terrainLength = 100.0
terrainWidth  = 100.0

# Camera track point
trackPoint = chrono.ChVector3d(-3.0, 0.0, 1.1)

# Contact + step sizes
contact_method      = chrono.ChContactMethod_NSC
contact_vis         = False
step_size           = 1e-3
tire_step_size      = step_size
render_step_size    = 1.0 / 50.0        # 50 FPS

# -------------------------------------------------------------------------
# Added data–logging parameters          <<< NEW >>>
# -------------------------------------------------------------------------
log_step_size = 0.10                     # log every 0.1 s
log_steps     = math.ceil(log_step_size/step_size)
gps_data      = []                       # list that will store GPS readings


# -------------------------------------------------------------------------
# Create the HMMWV model
# -------------------------------------------------------------------------

vehicle = veh.HMMWV_Full()
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
vehicle.SetTireType(tire_model)
vehicle.SetTireStepSize(tire_step_size)
vehicle.Initialize()

vehicle.SetChassisVisualizationType(vis_type)
vehicle.SetSuspensionVisualizationType(vis_type)
vehicle.SetSteeringVisualizationType(vis_type)
vehicle.SetWheelVisualizationType(vis_type)
vehicle.SetTireVisualizationType(vis_type)

vehicle.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# -------------------------------------------------------------------------
# Build rigid terrain
# -------------------------------------------------------------------------

patch_mat = chrono.ChContactMaterialNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)

terrain = veh.RigidTerrain(vehicle.GetSystem())
patch   = terrain.AddPatch(patch_mat,
                           chrono.ChCoordsysd(chrono.ChVector3d(0, 0, 0),
                                              chrono.QUNIT),
                           terrainLength, terrainWidth)
patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)
patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
terrain.Initialize()

# -------------------------------------------------------------------------
# Irrlicht visual system
# -------------------------------------------------------------------------

vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle('HMMWV Demo')
vis.SetWindowSize(1280, 1024)
vis.SetChaseCamera(trackPoint, 6.0, 0.5)
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetVehicle())

# -------------------------------------------------------------------------
# “Driver” – we still create one for convenience, but we will overwrite
#               its outputs according to time                     <<< NEW >>>
# -------------------------------------------------------------------------

driver = veh.ChInteractiveDriverIRR(vis)

steering_time = 1.0
throttle_time = 1.0
braking_time  = 0.3
driver.SetSteeringDelta(render_step_size / steering_time)
driver.SetThrottleDelta(render_step_size / throttle_time)
driver.SetBrakingDelta(render_step_size / braking_time)
driver.Initialize()

# -------------------------------------------------------------------------
# Sensor manager and sensors
# -------------------------------------------------------------------------

manager = sens.ChSensorManager(vehicle.GetSystem())

# -----------------  IMU  -----------------
offset_pose = chrono.ChFrame(chrono.ChVector3d(-8, 0, 1),
                             q_from_angle_axis(0.0, chrono.ChVector3d(0, 1, 0)))      # <<< FIX >>>

imu = sens.ChAccelerometerSensor(
        vehicle.GetChassisBody(),     # body to which it is attached
        10.0,                         # update rate (Hz)
        offset_pose,                  # relative pose
        sens.ChNoiseNone())           # noise model

imu.SetName("IMU Sensor")
imu.SetLag(0)
imu.SetCollectionWindow(0)
imu.PushFilter(sens.ChFilterAccelAccess())
manager.AddSensor(imu)

# -----------------  GPS  -----------------
gps = sens.ChGPSSensor(
        vehicle.GetChassisBody(),
        10.0,                         # update rate (Hz)
        offset_pose,
        chrono.ChVector3d(-89.400, 43.070, 260.0),     # (lon, lat, alt) reference
        sens.ChNoiseNone())

gps.SetName("GPS Sensor")
gps.SetLag(0)
gps.SetCollectionWindow(0)
gps.PushFilter(sens.ChFilterGPSAccess())
manager.AddSensor(gps)

# -------------------------------------------------------------------------
# Simulation loop
# -------------------------------------------------------------------------

print("VEHICLE MASS:", vehicle.GetVehicle().GetMass())

render_steps   = math.ceil(render_step_size / step_size)
realtime_timer = chrono.ChRealtimeStepTimer()
step_number    = 0
render_frame   = 0

while vis.Run():

    time = vehicle.GetSystem().GetChTime()

    # ----------------- Rendering -----------------
    if step_number % render_steps == 0:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        render_frame += 1

    # ----------------- Driver inputs (time-based)      <<< NEW >>> -----------------
    # Instead of using only keyboard interaction, we prescribe simple manoeuvres:
    #   0-2s  : accelerate straight
    #   2-4s  : accelerate with right steering
    #   4-6s  : accelerate with left steering
    #   >6s   : brake to stop
    di = veh.DriverInputs()   # blank struct

    if time < 2.0:
        di.m_throttle = 0.8
        di.m_steering = 0.0
        di.m_braking  = 0.0
    elif time < 4.0:
        di.m_throttle = 0.8
        di.m_steering = 0.25
        di.m_braking  = 0.0
    elif time < 6.0:
        di.m_throttle = 0.8
        di.m_steering = -0.25
        di.m_braking  = 0.0
    else:
        di.m_throttle = 0.0
        di.m_steering = 0.0
        di.m_braking  = 0.6

    # ----------------- Synchronise modules -----------------
    driver.Synchronize(time)                              # still update keyboard etc.
    terrain.Synchronize(time)
    vehicle.Synchronize(time, di, terrain)
    vis.Synchronize(time, di)

    # ----------------- Advance all modules -----------------
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    # ----------------- Sensor manager update --------------
    manager.Update()

    # ----------------- Log GPS data              <<< NEW >>> --------------
    if step_number % log_steps == 0:
        buff = gps.GetMostRecentGPSBuffer()
        if buff is not None and buff.HasData():
            gps_point = buff.GetGPSData()[0]         # take first entry in buffer
            # The gps_point contains easting/northing/alt etc.  Simply convert to tuple
            gps_data.append((gps_point.x, gps_point.y, gps_point.z))

    # ----------------- House-keeping -----------------
    step_number += 1
    realtime_timer.Spin(step_size)

# -------------------------------------------------------------------------
#  Output collected GPS data                                   <<< NEW >>>
# -------------------------------------------------------------------------
print("\nGPS Data:", gps_data)
```

Explanation of the main fixes and additions
------------------------------------------
1. `chrono.ChFramed` and `chrono.QuatFromAngleAxis` do not exist.  
   Replaced by `chrono.ChFrame` and helper `q_from_angle_axis`.

2. Instantiation of `DriverInputs` is done through `veh.DriverInputs()` so that
   we can prescribe throttle / steering / braking values according to the
   current simulation time (as required).

3. Added ``log_step_size``, ``log_steps`` and ``gps_data`` to control and store
   GPS logging.

4. Implemented logging: every `log_step_size` seconds the most recent GPS
   buffer is polled and, if valid data exist, the first GPS sample is appended
   to `gps_data`.

5. Printed the collected GPS data once the simulation window is closed.

The rest of the original demo (terrain, vehicle, sensors and visualisation)
remains unchanged except for the small corrections listed above.
