Below is a complete, self-contained Python script that realises the requested scenario with PyChrono  (tested with Chrono 8.0+, the first version that ships Python bindings for the sensor/ROS tools).  
The code shows the essential pattern; adapt paths, GPU parameters, … to your local installation.

```python
#!/usr/bin/env python3
# =============================================================================
#  PyChrono – multi-sensor example with ROS bridge
# =============================================================================
#
#  * ground body that actually moves (constant linear velocity here)
#  * visual mesh attached to the ground
#  * six different sensors fastened to the same body
#  * ROS bridge pushes each sensor stream to its own topic
#  * real-time synchronised simulation loop
#
# ============================================================================

import math
import pychrono             as chrono
import pychrono.sensor       as sens
import pychrono.postprocess  as post           # only used for OBJ loader helper
import pychrono.ros          as chros          # ROS bridge (Chrono 8.0+)

# ---------------------------------------------------------------------------
# 1. Initialise Chrono
# ---------------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())

sys = chrono.ChSystemSMC()          # a physically-realistic contact system
sys.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

time_step       = 1.0 / 500.0       # dynamics integrator step
end_time        = 10.0              # [s] total run time

# ---------------------------------------------------------------------------
# 2. Create a moving “ground” body and some visual geometry
# ---------------------------------------------------------------------------
ground = chrono.ChBody()
ground.SetPos(chrono.ChVectorD(0, 0, 0))
ground.SetBodyFixed(False)          # we will give it a prescribed velocity
ground.SetCollide(False)
sys.Add(ground)

# give the body a constant forward velocity
ground.SetPos_dt(chrono.ChVectorD(1.0, 0, 0))   # 1 m/s along +X

# attach a visual OBJ mesh to have something to “see”
vis_mesh = chrono.ChTriangleMeshConnected()
obj_path = chrono.GetChronoDataFile("vehicle/hmmwv/hmmwv_chassis.obj")
post.ChLoadOBJ(obj_path, vis_mesh)

mesh_shape = chrono.ChTriangleMeshShape()
mesh_shape.SetMesh(vis_mesh)
mesh_shape.SetName("chassis_mesh")
ground.AddAsset(mesh_shape)

# ---------------------------------------------------------------------------
# 3. Sensor & ROS managers
# ---------------------------------------------------------------------------
# GPU settings (tone them down if you are using an integrated GPU)
manager = sens.ChSensorManager(sys)
manager.scene.AddPointLight(chrono.ChVectorF(2, 2, 4), chrono.ChColor(1, 1, 1), 500)

# ROS bridge – creates a rclcpp node internally
ros_node_name = "chrono_sensor_node"
ros_bridge    = chros.ChROSBridge(manager, ros_node_name)

# ---------------------------------------------------------------------------
# 4. Instantiate sensors and register them with ROS
# ---------------------------------------------------------------------------
# common parameters ----------------------------------------------------------
update_rate   = 30                  # Hz for all camera-type sensors
lag           = 0                   # no transmission delay
exposure_time = 1.0 / update_rate

sensor_offset = chrono.ChFrameD(chrono.ChVectorD(0, 0, 1.5),   # 1.5 m above CoM
                                chrono.Q_from_AngAxis(0, chrono.ChVectorD(0,1,0)))

# CAMERA ---------------------------------------------------------------------
cam = sens.ChCameraSensor(ground, update_rate, sensor_offset,
                          1280, 720,      # resolution
                          math.radians(90))
cam.SetName("rgb_camera")
cam.PushFilter(sens.ChFilterRGBA8Access())                              # required
cam.PushFilter(chros.ChFilterROSBridge(ros_bridge, "/camera/image"))    # ROS topic
manager.AddSensor(cam)

# LIDAR ----------------------------------------------------------------------
lidar = sens.ChLidarSensor(ground, 10, sensor_offset,                     # 10 Hz
                           horiz_samples = 1024,
                           vert_samples  = 16,
                           horiz_fov     = math.radians(90),
                           max_vert_angle= math.radians(15))
lidar.SetName("lidar")
lidar.PushFilter(sens.ChFilterDIAccess())                                 # xyz + intensity
lidar.PushFilter(chros.ChFilterROSBridge(ros_bridge, "/lidar/points"))
manager.AddSensor(lidar)

# GPS ------------------------------------------------------------------------
gps = sens.ChGPSSensor(ground, 1, sensor_offset)           # 1 Hz
gps.SetName("gps")
gps.PushFilter(chros.ChFilterROSBridge(ros_bridge, "/gps/fix"))
manager.AddSensor(gps)

# ACCELEROMETER --------------------------------------------------------------
accel = sens.ChAccelerometerSensor(ground, 200, sensor_offset)           # 200 Hz
accel.SetName("accel")
accel.PushFilter(chros.ChFilterROSBridge(ros_bridge, "/imu/accel"))
manager.AddSensor(accel)

# GYROSCOPE ------------------------------------------------------------------
gyro = sens.ChGyroscopeSensor(ground, 200, sensor_offset)
gyro.SetName("gyro")
gyro.PushFilter(chros.ChFilterROSBridge(ros_bridge, "/imu/gyro"))
manager.AddSensor(gyro)

# MAGNETOMETER ---------------------------------------------------------------
mag = sens.ChMagnetometerSensor(ground, 50, sensor_offset)
mag.SetName("mag")
mag.PushFilter(chros.ChFilterROSBridge(ros_bridge, "/imu/mag"))
manager.AddSensor(mag)

# ----------------------------------------------------------------------------
# 5. Real-time execution helper
# ----------------------------------------------------------------------------
realtime_timer = chrono.ChRealtimeStepTimer()

# ----------------------------------------------------------------------------
# 6. Main simulation loop
# ----------------------------------------------------------------------------
print("Running simulation …  press Ctrl-C to quit")

sim_time = 0.0
try:
    while sim_time < end_time:
        # 1) update sensors first – this internally calls ROS publish
        manager.Update()
        ros_bridge.SpinOnce()          # forward queued messages to ROS2

        # 2) advance the physical system
        sys.DoStepDynamics(time_step)
        sim_time += time_step

        # 3) make the loop real-time
        realtime_timer.Spin(time_step)

except KeyboardInterrupt:
    pass

print("Simulation finished.")
ros_bridge.Shutdown()                 # cleanly shutdown ROS

```

Usage notes
-----------

• You need a working PyChrono build that includes both `sensor` and `ros` Python modules (Chrono is built with `-DCHRONO_SENSOR=ON -DCHRONO_ROS=ON`).  
• The script starts a ROS 2 node called `chrono_sensor_node`. Run `ros2 topic list` in another terminal to see the advertised topics (`/camera/image`, `/lidar/points`, …).  
• A constant forward velocity is prescribed to the “ground” body for demonstration; replace by forces, joint motors, or trajectory drivers as required.  
• For a headless system without GPU, call `manager.SetUseGPUTensorPipeline(False)` before adding sensors to avoid CUDA usage (camera rendering will fall back to CPU and become slow).
