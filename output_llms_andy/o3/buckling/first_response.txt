```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
#
# Beam-buckling demo with FEA, custom motor, constraints and Irrlicht
#
# Author : <your-name>
# ---------------------------------------------------------------------
#
# Prerequisites
#  • An up-to-date Chrono build that contains the modules:
#       chrono, chrono.irrlicht, chrono.fea
#  • Python 3.x
# ---------------------------------------------------------------------

import math
import pychrono as chrono
import pychrono.irrlicht as irr
import pychrono.fea    as fea

# ---------------------------------------------------------------------
# 1. Chrono system initialisation
# ---------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())   # textures, fonts …
sys = chrono.ChSystemSMC()                             # SMC = penalty contacts (unused here)

# basic environment
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))           # gravity (downwards, Y axis)

# ---------------------------------------------------------------------
# 2. Geometry, materials, mesh (FEA beam)
# ---------------------------------------------------------------------
# Simulation units :  [m , kg , s , N]
beam_len   = 1.0                                       # 1 m long beam
beam_w     = 0.02                                      # 2 cm square cross-section
E          = 210e9                                     # Young [Pa] (steel)
G          = 80e9                                      # Shear [Pa]
rho        = 7800                                      # density [kg/m³]

#
# 2.1 Beam section
#
section = fea.ChBeamSectionAdvanced()
section.SetYoungModulus(E)
section.SetGwithPoissonRatio(E, 0.3)
section.SetDensity(rho)
section.SetAsRectangularSection(beam_w, beam_w, True)

#
# 2.2 Mesh container
#
mesh = fea.ChMesh()

#
# 2.3 Nodes and elements (Euler-Bernoulli or Timoshenko, choose Euler here)
#
n_elem = 20
dz     = beam_len / n_elem

# first node (x=0) is coincident with ground body later
nodes = []

for i in range(n_elem+1):
    node = fea.ChNodeFEAxyzrot( chrono.ChFrameD( chrono.ChVectorD( dz*i, 0, 0 ) ) )
    node.SetMass( rho * beam_w * beam_w * dz )         # simple lumped mass per node
    mesh.AddNode(node)
    nodes.append(node)

for i in range(n_elem):
    element = fea.ChElementBeamEuler()
    element.SetNodes( nodes[i], nodes[i+1] )
    element.SetSection( section )
    mesh.AddElement( element )

sys.Add(mesh)

# ---------------------------------------------------------------------
# 3. Rigid bodies, constraints, custom motor
# ---------------------------------------------------------------------
# 3.1 Ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
sys.Add(ground)

# Visual aid for ground
ground_shape = chrono.ChBoxShape()
ground_shape.GetBoxGeometry().Size = chrono.ChVectorD(0.05, 0.05, 0.05)
ground.AddAsset(ground_shape)

# 3.2 Sliding body (acts as actuator platen compressing the beam)
platen = chrono.ChBody()
platen.SetMass(1.0)
platen.SetPos( chrono.ChVectorD( beam_len, 0, 0 ) )
sys.Add(platen)

# Cylinder visualisation
cyl = chrono.ChCylinderShape()
cyl.GetCylinderGeometry().p1 = chrono.ChVectorD(-0.02, -0.02, 0)
cyl.GetCylinderGeometry().p2 = chrono.ChVectorD( 0.02,  0.02, 0)
cyl.GetCylinderGeometry().rad = 0.03
platen.AddAsset(cyl)

# 3.3 Tie beam ends to rigid bodies
#     Node 0  -> ground;   Node n_elem -> platen
ground_constraint = fea.ChLinkNodeFrame()
ground_constraint.Initialize( nodes[0], ground )
sys.Add(ground_constraint)

platen_constraint = fea.ChLinkNodeFrame()
platen_constraint.Initialize( nodes[-1], platen )
sys.Add(platen_constraint)

# 3.4 Prismatic joint between ground and platen so platen moves along X only
prism = chrono.ChLinkLockPrismatic()
prism.Initialize( platen, ground,
                  chrono.ChCoordsysD( chrono.ChVectorD(beam_len, 0, 0),
                                      chrono.Q_from_AngAxis(math.pi/2, chrono.ChVectorD(0,0,1)) ) )
sys.Add(prism)

# 3.5 Linear motor (prescribed relative displacement = compression)
linmot = chrono.ChLinkMotorLinearPosition()
linmot.Initialize( platen, ground, chrono.ChFrameD( chrono.ChVectorD(0,0,0) ) )  # the same axis (X) is used

#
# Custom displacement function: start at t=0, move −delta over T seconds then hold
#
class CompressionFunction(chrono.ChFunction):
    def __init__(self, displacement=-0.02, ramp_time=0.5):
        chrono.ChFunction.__init__(self)
        self.disp = displacement
        self.tramp = ramp_time

    def Get_y(self, t):
        if t < self.tramp:
            return self.disp * (t/self.tramp)
        else:
            return self.disp

    def Get_y_dx(self, t):
        if t < self.tramp:
            return self.disp / self.tramp
        else:
            return 0.0

    def Get_y_dxdx(self, t):
        return 0.0

linmot.SetMotionFunction(CompressionFunction(displacement=-0.05, ramp_time=1.0))
sys.Add(linmot)

# ---------------------------------------------------------------------
# 4. Solver and integrator
# ---------------------------------------------------------------------
# 4.1 Solver (use MKL if present, fallback to MINRES)
try:
    mkl = chrono.ChSolverMKL()
    mkl.SetVerbose(False)
    sys.SetSolver(mkl)
except:
    print("[INFO] MKL not available, switching to MINRES")
    minres = chrono.ChSolverMINRES()
    minres.SetMaxIterations(200)
    minres.SetTolerance(1e-10)
    minres.EnableDiagonalPreconditioner(True)
    sys.SetSolver(minres)

# 4.2 Time-stepper : HHT (α = −0.2 for dissipation) better for stiff FEA
hht = chrono.ChTimestepperHHT()
hht.SetAlpha(-0.2)
hht.SetMaxiters(50)
hht.SetTolerance(1e-9)
sys.SetTimestepper(hht)

# ---------------------------------------------------------------------
# 5. Irrlicht visualisation
# ---------------------------------------------------------------------
application = irr.ChIrrApp(sys, "Beam buckling / FEA demo", irr.dimension2du(1280,720))
application.AddTypicalLogo()
application.AddTypicalSky()
application.AddTypicalCamera(irr.vector3df(0.2, 0.3, 1.0))
application.AddTypicalLights()
application.AssetBindAll()
application.AssetUpdateAll()

# ---------------------------------------------------------------------
# 6. Simulation loop
# ---------------------------------------------------------------------
step_size = 1e-3

while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()

    # (Optional) draw link forces, nodes, etc. :
    # fea.ChVisualizationFEAmesh(mesh, ...)

    sys.DoStepDynamics(step_size)

    application.EndScene()
```

Explanation of the main building blocks
• Finite-Element Beam  
  – An `ChMesh` container holds Euler beam elements (`ChElementBeamEuler`) built from `ChNodeFEAxyzrot` nodes.  
  – Section parameters (E, G, density, and geometry) are set through `ChBeamSectionAdvanced`.

• Boundary conditions / constraints  
  – Node 0 is rigidly fixed to the ground; the last node is rigidly fixed to the moving platen.  
  – A prismatic joint allows the platen to translate only along the beam axis.  
  – A linear motor (`ChLinkMotorLinearPosition`) is attached on the same axis to prescribe a controlled compression; a custom subclass of `ChFunction` generates a ramp-down displacement (negative = shortening) over 1 s and then keeps it constant.

• Solver & integrator  
  – MKL (direct) is preferred; if not available, MINRES (iterative) is configured.  
  – The HHT α-method provides good robustness for stiff FEA problems such as buckling.

• Visualisation  
  – Irrlicht renders the beam, ground, and platen in real time.  
  – `AssetBindAll()` and `AssetUpdateAll()` automatically attach visualization shapes to FEA items and rigid bodies.

Run the script: the beam will be gradually compressed by 5 cm; when the critical load is reached it will buckle laterally under gravity (you can switch gravity off to see pure Euler buckling).
