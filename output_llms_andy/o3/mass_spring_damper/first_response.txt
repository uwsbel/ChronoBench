```python
# =============================================================================
# Basic PyChrono example:
# A single mass connected to ground through a linear spring–damper (ChLinkTSDA)
# -----------------------------------------------------------------------------
# – includes Irrlicht 3-D visualization (sky, lights, camera)
# – shows a box for the moving mass and a coil-like graphic for the TSDA link
# =============================================================================
#
# Tested with ProjectChrono 7.*  /  Python ≥ 3.8
# ------------------------------------------------------------------------------

import pychrono as chrono
import pychrono.irrlicht as irr

# ---------------------------------------------------------------------
# 1. Create the physical system
# ---------------------------------------------------------------------
sys = chrono.ChSystemSMC()          # (or ChSystemNSC – either works here)
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# ---------------------------------------------------------------------
# 2. Create (fixed) ground
# ---------------------------------------------------------------------
ground = chrono.ChBody()
ground.SetBodyFixed(True)
sys.Add(ground)

# Optional: a thin box to see the ground in the 3-D window
ground_visual = chrono.ChBodyEasyBox(3.0, 0.1, 3.0,          # size (x,y,z)
                                     1000,                   # density  (irrelevant for fixed)
                                     True, True)             # collide?, visual?
ground_visual.SetBodyFixed(True)
ground_visual.SetPos(chrono.ChVectorD(0, -0.05, 0))
sys.Add(ground_visual)

# ---------------------------------------------------------------------
# 3. Create the moving mass (a small cube)
# ---------------------------------------------------------------------
mass          = 1.0                          # kg
size          = 0.2                          # box edge length (m)
initial_pos   = chrono.ChVectorD(0, 0.5, 0)  # starting location

cube = chrono.ChBodyEasyBox(size, size, size,
                            mass,
                            True,  True)      # collide?, visualize?
cube.SetPos(initial_pos)

# The default inertias from ChBodyEasy* are fine for this demo, but we can set
# them explicitly if desired:
Ixx = Iyy = Izz = (1/6) * mass * size**2
cube.SetInertiaXX(chrono.ChVectorD(Ixx, Iyy, Izz))

sys.Add(cube)

# ---------------------------------------------------------------------
# 4. Add the spring–damper (TSDA) between ground and cube
# ---------------------------------------------------------------------
k   = 100.0   # [N/m] spring stiffness
c   =   5.0   # [N·s/m] damping coefficient
rest_length = 0.40  # [m] spring free length

# End-points of the link (expressed in global coordinates)
pos_ground = chrono.ChVectorD(0, 0.00, 0)                 # on ground
pos_cube   = chrono.ChVectorD(0, +size/2, 0)              # top face of cube

tsda = chrono.ChLinkTSDA()
tsda.Initialize(ground, cube, False, pos_ground, pos_cube)
tsda.SetSpringCoefficient(k)
tsda.SetDampingCoefficient(c)
tsda.SetRestLength(rest_length)

# Ask Chrono to generate a simple coil-like visualization for the TSDA
# (radius, number_of_turns, color):
tsda.AddVisualShapeSpring(0.02, 20, chrono.ChColor(0.4, 0.4, 0.9))

sys.Add(tsda)

# ---------------------------------------------------------------------
# 5. Irrlicht application (visualization)
# ---------------------------------------------------------------------
app = irr.ChIrrApp(sys,
                   "Single mass–spring–damper (ChLinkTSDA) demo",
                   irr.dimension2du(1024, 768))

app.AddTypicalLogo()
app.AddTypicalSky()

# Two point lights
app.AddTypicalLights(chrono.ChVectorD(+2, 4, -2),
                     chrono.ChVectorD(-2, 4, +2), 9, 1)

# Camera: (position, target)
app.AddCamera(chrono.ChVectorD(1.5, 1.0, 1.5),
              chrono.ChVectorD(0, 0.25, 0))

# Tell Irrlicht to use the visual assets we attached
app.AssetBindAll()
app.AssetUpdateAll()

# ---------------------------------------------------------------------
# 6. Simulation main loop
# ---------------------------------------------------------------------
time_step = 1e-3
app.SetTimestep(time_step)

while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()
    app.DoStep()          # advances dynamics by 'time_step'
    app.EndScene()

# ---------------------------------------------------------------------
# End of file
# ---------------------------------------------------------------------
```

How it works
• A `ChBodyEasyBox` is used for the 1 kg mass; its top face is the upper attachment point for the TSDA.  
• A `ChLinkTSDA` connects that point to a point on the fixed ground.  Default linear behaviour is specified through `SetSpringCoefficient()`, `SetDampingCoefficient()`, and `SetRestLength()`.  
• `AddVisualShapeSpring()` (added in Chrono 6.0+) generates a helix between the two attachment nodes so you can see the spring stretching/contracting during the run.  
• The Irrlicht helper (`ChIrrApp`) provides a ready-to-use window with lights, sky, and camera, while `DoStep()` keeps physics and graphics synchronized.
