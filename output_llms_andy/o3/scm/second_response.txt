```python
# =============================================================================
#  PyChrono HMMWV demo
#
#  This version fixes the problems found in the original script and introduces
#  a dedicated helper-class that manages all SCM deformable-terrain parameters.
#  A user can now switch among three predefined soil configurations:
#     “soft”, “mid” (default) and “hard”.
# =============================================================================

import math
import pychrono.core     as chrono
import pychrono.irrlicht as irr          # needed by the interactive driver
import pychrono.vehicle  as veh


# -----------------------------------------------------------------------------
# 1.  Terrain parameter helper
# -----------------------------------------------------------------------------
class SCMTerrainParameters:
    """
    Store and apply SCM‐terrain parameters.

    preset (str):  'soft' | 'mid' (default) | 'hard'
    """

    def __init__(self, preset: str = 'mid'):
        preset = preset.lower()

        if preset == 'soft':
            self.Kphi   = 1.0e6      # Bekker Kphi
            self.Kc     = 0.0        # Bekker Kc
            self.n      = 1.1        # Bekker n exponent
            self.c      = 0.0        # Mohr cohesive limit (Pa)
            self.phi    = 25.0       # Mohr friction limit (deg)
            self.Janosi = 0.01       # Janosi shear coefficient (m)
            self.E      = 1.0e8      # Elastic stiffness (Pa/m)
            self.damp   = 2.0e4      # Damping (Pa·s/m)

        elif preset == 'hard':
            self.Kphi   = 4.0e6
            self.Kc     = 0.0
            self.n      = 1.1
            self.c      = 0.0
            self.phi    = 35.0
            self.Janosi = 0.01
            self.E      = 4.0e8
            self.damp   = 4.0e4

        else:                     # default: 'mid'
            self.Kphi   = 2.0e6
            self.Kc     = 0.0
            self.n      = 1.1
            self.c      = 0.0
            self.phi    = 30.0
            self.Janosi = 0.01
            self.E      = 2.0e8
            self.damp   = 3.0e4

    # -------------------------------------------------------------------------
    def apply_to(self, scm_terrain: veh.SCMTerrain):
        """Apply the stored parameters to the provided SCMTerrain instance."""
        scm_terrain.SetSoilParameters(self.Kphi,
                                      self.Kc,
                                      self.n,
                                      self.c,
                                      self.phi,
                                      self.Janosi,
                                      self.E,
                                      self.damp)


# -----------------------------------------------------------------------------
# 2.  Chrono initialisation & path setup
# -----------------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())           # make sure the
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')       # data path is set


# -----------------------------------------------------------------------------
# 3.  Vehicle configuration
# -----------------------------------------------------------------------------
initLoc  = chrono.ChVector3d(-8, 0, 0.6)
initRot  = chrono.ChQuaterniond(1, 0, 0, 0)

vis_type              = veh.VisualizationType.MESH
chassis_collision_type = veh.ChassisCollisionType.NONE
tire_model            = veh.TireModelType.RIGID

step_size        = 1e-3
tire_step_size   = step_size
render_step_size = 1.0 / 50.0                 #   50 FPS


# -----------------------------------------------------------------------------
# 4.  Create the vehicle
# -----------------------------------------------------------------------------
vehicle = veh.HMMWV_Full()
vehicle.SetContactMethod(chrono.ChContactMethod_SMC)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
vehicle.SetTireType(tire_model)
vehicle.SetTireStepSize(tire_step_size)
vehicle.Initialize()

vehicle.SetChassisVisualizationType(vis_type)
vehicle.SetSuspensionVisualizationType(vis_type)
vehicle.SetSteeringVisualizationType(vis_type)
vehicle.SetWheelVisualizationType(vis_type)
vehicle.SetTireVisualizationType(vis_type)

vehicle.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)


# -----------------------------------------------------------------------------
# 5.  Create and configure the SCM terrain
# -----------------------------------------------------------------------------
terrain = veh.SCMTerrain(vehicle.GetSystem())

# 5-A. Choose one of the available presets: 'soft', 'mid', 'hard'
soil = SCMTerrainParameters(preset='mid')     # <<< change here if desired
soil.apply_to(terrain)                        # apply it to the terrain object

# 5-B. Add a moving patch that follows the chassis (performance optimisation)
terrain.AddMovingPatch(vehicle.GetChassisBody(),
                       chrono.ChVector3d(0, 0, 0),
                       chrono.ChVector3d(5, 3, 1))

terrain.SetPlotType(veh.SCMTerrain.PLOT_SINKAGE, 0, 0.1)
terrain.Initialize(20, 20, 0.02)


# -----------------------------------------------------------------------------
# 6.  Visualisation and driver
# -----------------------------------------------------------------------------
trackPoint = chrono.ChVector3d(0.0, 0.0, 1.71)

vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle('HMMWV SCM-Terrain demo')
vis.SetWindowSize(1280, 1024)
vis.SetChaseCamera(trackPoint, 6.0, 0.5)
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetVehicle())

# keyboard driver
driver = veh.ChInteractiveDriverIRR(vis)
steering_time = 1.0
throttle_time = 1.0
braking_time  = 0.3
driver.SetSteeringDelta(render_step_size / steering_time)
driver.SetThrottleDelta(render_step_size / throttle_time)
driver.SetBrakingDelta(render_step_size  / braking_time)
driver.Initialize()


# -----------------------------------------------------------------------------
# 7.  Simulation loop
# -----------------------------------------------------------------------------
print("===================================================")
print("  VEHICLE MASS: ", vehicle.GetVehicle().GetMass())
print("  Using soil preset:", soil.__dict__)
print("===================================================")

render_steps     = math.ceil(render_step_size / step_size)
realtime_timer   = chrono.ChRealtimeStepTimer()
step_number      = 0
render_frame     = 0

while vis.Run():
    time = vehicle.GetSystem().GetChTime()

    # -----------------
    # Render every n-th step
    # -----------------
    if step_number % render_steps == 0:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        render_frame += 1

    # -----------------
    # Input & synchronisation
    # -----------------
    driver_inputs = driver.GetInputs()

    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    vis.Synchronize(time, driver_inputs)

    # -----------------
    # Advance the dynamics
    # -----------------
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    step_number += 1
    realtime_timer.Spin(step_size)
```

Key fixes and improvements
==========================
1. Enumeration names have been corrected to the current PyChrono API:
      • `veh.VisualizationType.MESH`
      • `veh.ChassisCollisionType.NONE`
      • `veh.TireModelType.RIGID`

2. A dedicated class `SCMTerrainParameters` encapsulates all soil parameters.
   Three presets (`soft`, `mid`, `hard`) are available and applied through
   the `apply_to()` method.

3. All former direct calls to `terrain.SetSoilParameters()` were removed in
   favor of the new class to satisfy the given requirements.

The script can now be executed unmodified except for changing the preset
string when a different soil condition is desired.
