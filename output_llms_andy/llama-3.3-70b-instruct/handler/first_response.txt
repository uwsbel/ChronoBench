```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import numpy as np
import rospy
from std_msgs.msg import Int32

# Initialize the PyChrono environment
sys = chrono.ChSystemNSC()

# Set the gravitational acceleration
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the floor
floor = chrono.ChBodyEasyBox(sys, 10, 1, 10, 1000, True, True)
floor.SetPos(chrono.ChVectorD(0, -5, 0))
floor.SetBodyFixed(True)
sys.Add(floor)

# Create the box
box = chrono.ChBodyEasyBox(sys, 1, 1, 1, 10, True, True)
box.SetPos(chrono.ChVectorD(0, 5, 0))
sys.Add(box)

# Create a custom ROS handler
class CustomROSHandler:
    def __init__(self):
        self.pub = rospy.Publisher('custom_topic', Int32, queue_size=10)
        self.count = 0

    def Update(self):
        self.count += 1
        self.pub.publish(self.count)

# Initialize the ROS node
rospy.init_node('pychrono_simulation')

# Create the ROS manager
ros_manager = chrono.ChROSManager(sys)

# Create the clock handler
clock_handler = chrono.ChROSClockHandler(ros_manager)

# Create the body handler
body_handler = chrono.ChROSBODYHandler(ros_manager)

# Create the transform handler
transform_handler = chrono.ChROSTransformHandler(ros_manager)

# Register the custom handler
custom_handler = CustomROSHandler()
ros_manager.RegisterHandler(custom_handler)

# Run the simulation loop
while not rospy.is_shutdown():
    sys.DoStepDynamics(0.01)
    ros_manager.Update()
    rospy.sleep(0.01)
```

This script initializes a PyChrono simulation with a fixed floor and a movable box, and sets up a custom ROS handler to publish integer messages to a specified topic. The simulation loop advances the system and updates ROS communication at each timestep, while maintaining real-time execution.
