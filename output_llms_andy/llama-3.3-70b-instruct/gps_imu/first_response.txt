```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the PyChrono environment and core components
sys = chrono.ChSystemNSC()

# Create the vehicle system
hmmwv = veh.HMMWV()
hmmwv.Initialize(sys)
hmmwv.SetContactMaterial(chrono.ChMaterialSurfaceNSC())
hmmwv.SetChassisFixed(False)

# Add the vehicle to the system
sys.Add(hmmwv.GetChassisBody())
sys.Add(hmmwv.GetBody steering)
sys.Add(hmmwv.GetBody suspension)

# Create the terrain system
terrain = veh.RigidTerrain(sys)
terrain.SetPlane(chrono.ChPlane(chrono.ChVectorD(0, 1, 0), 0))

# Add the terrain to the system
sys.Add(terrain)

# Create the driver system
driver = veh.ChIrrlichtDriver()
driver.SetVehicle(hmmwv)
driver.SetSteeringDelta(0.1)
driver.SetThrottleDelta(0.1)
driver.SetBrakingDelta(0.1)

# Create the visual system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AddTypicalLights()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 5, -10))
vis.AddTypicalLogo()
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("HMMWV Simulation")

# Add the vehicle and terrain to the visual system
vis.Add(hmmwv)
vis.Add(terrain)

# Create IMU and GPS sensors
imu = veh.ChSensorIMU(hmmwv.GetChassisBody())
gps = veh.ChSensorGPS(hmmwv.GetChassisBody())

# Add the sensors to the system
sys.Add(imu)
sys.Add(gps)

# Set the simulation parameters
step_size = 0.01
t_end = 10

# Run the simulation loop
t = 0
while t < t_end:
    # Update the driver inputs
    driver.Synchronize(t)
    
    # Update the vehicle
    hmmwv.Synchronize(t)
    
    # Update the terrain
    terrain.Synchronize(t)
    
    # Update the visual system
    vis.Synchronize(t)
    
    # Update the IMU and GPS sensors
    imu.Synchronize(t)
    gps.Synchronize(t)
    
    # Output the vehicle mass
    print("Vehicle mass: ", hmmwv.GetChassisBody().GetMass())
    
    # Render the scene
    vis.Render()
    
    # Advance the simulation
    sys.DoStepDynamics(step_size)
    
    # Increment the time
    t += step_size

# Clean up
del sys
del vis
```

This script creates a PyChrono simulation featuring an HMMWV vehicle with a full system setup, including terrain, driver inputs, and visual interface using Irrlicht. It adds and manages IMU and GPS sensors attached to the vehicle's chassis, processing and updating sensor data at each simulation step. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system modules while outputting vehicle mass and rendering scenes.
