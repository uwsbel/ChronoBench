Here's a Python script that models the gravitational attraction between particles using PyChrono. The simulation includes a particle emitter that generates particles with random shapes, positions, velocities, and orientations. The Irrlicht visualization is used to display the simulation.

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.particlefactory as chrono_particle
import random

# Initialize the PyChrono environment
chrono.SetChronoDataPath("/path/to/chrono/data/")

# Create the system
sys = chrono.ChSystemSMC()

# Set default collision and gravitational parameters
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))
sys.SetStep(1e-4)

# Create a particle factory
particle_factory = chrono_particle.ChParticleEmitterFactory()

# Create a particle emitter
emitter = chrono_particle.ChParticleEmitter(
    particle_factory,
    sys,
    chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.ChQuaternionD(1, 0, 0, 0)),
    100,  # number of particles emitted per second
    10,   # total number of particles emitted
    0.01, # emission radius
    chrono.ChVectorD(0, 0.1, 0),  # average velocity
    chrono.ChVectorD(0.1, 0.1, 0.1),  # velocity spread
)

# Create a custom gravitational force
def custom_gravity(body1, body2):
    G = 6.67430e-11  # gravitational constant
    pos1 = body1.GetPos()
    pos2 = body2.GetPos()
    distance = (pos2 - pos1).Length()
    if distance > 1e-6:
        direction = (pos2 - pos1) / distance
        force = G * body1.GetMass() * body2.GetMass() / (distance ** 2) * direction
        body1.AddForce(force)
        body2.AddForce(-force)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(800, 600)
vis.SetWindowTitle('Particle Emitter Simulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataPath() + 'logo_pychrono_alpha.png')
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 2, -4))
vis.AttachSystem(sys)

# Simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()

    # Emit particles
    emitter.Update(chrono.ChTime().GetStep())

    # Apply custom gravitational forces
    for body1 in sys.Get_bodylist():
        for body2 in sys.Get_bodylist():
            if body1 != body2:
                custom_gravity(body1, body2)

    # Step the simulation
    sys.DoStepDynamics(chrono.ChTime().GetStep())

    vis.EndScene()
```

To run the simulation, replace `"/path/to/chrono/data/"` with the actual path to the Chrono data directory.

```python
# Replace this line:
chrono.SetChronoDataPath("/path/to/chrono/data/")
# With the actual path:
chrono.SetChronoDataPath("/usr/local/share/chrono/data/")
```
