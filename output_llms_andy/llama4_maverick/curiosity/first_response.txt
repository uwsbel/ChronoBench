To create a PyChrono simulation that showcases a Curiosity rover navigating on a rigid terrain, we will follow the provided instructions step by step. Below is a Python script that accomplishes this task.

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import pychrono.sensor as sens

# Initialize the PyChrono environment
print("Copyright (c) 2023 Project Chrono")

# Create the system
sys = chrono.ChSystemNSC()

# Create the ground body
ground_mat = chrono.ChMaterialSurfaceNSC()
ground = chrono.ChBody()
ground.SetBodyFixed(True)
ground.SetCollide(True)
ground.GetCollisionModel().ClearModel()
ground.GetCollisionModel().AddBox(ground_mat, 100, 1, 100, chrono.ChFrame(chrono.ChVector3f(0, -1, 0)))
ground.GetCollisionModel().BuildModel()
ground.SetPos(chrono.ChVector3f(0, 0, 0))
sys.Add(ground)

# Create the Curiosity rover
init_loc = chrono.ChVector3f(0, 0.5, -5)
init_rot = chrono.ChQuaternionf(1, 0, 0, 0)
rover = veh.Curiosity(sys, init_loc, init_rot)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Curiosity Rover Simulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3f(5, 5, 5), chrono.ChVector3f(0, 0, 0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3f(10, 15, 10), chrono.ChVector3f(0, 2.5, 0), 15, 4, 15, 120)
vis.EnableShadows()

# Simulation loop
while vis.Run():
    time = sys.GetChTime()
    sys.DoStepDynamics(1e-3)
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Steering input (simple example: change direction every 2 seconds)
    if time % 4 < 2:
        rover.SetSteering(0.5)
    else:
        rover.SetSteering(-0.5)

    # Update the rover
    rover.Update()

    # Update the visualization
    vis.GetDevice().run()
```
