```python
import pychrono as chrono
import pychrono.fea as fea
import pychrono.irrlicht as irr
import math

# ---------------------------------------------------------------------
#
#  Create the simulation system
#
# ---------------------------------------------------------------------

mysystem = chrono.ChSystemNSC()
mysystem.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# ---------------------------------------------------------------------
#
# Create the Irrlicht visualization system
#
# ---------------------------------------------------------------------
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(mysystem)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle('Beam Buckling Simulation')
vis.Initialize()
vis.AddCamera(chrono.ChVectorD(10, 5, 5))
vis.AddLight(chrono.ChVectorD(5, 5, 5), chrono.ChColor(1, 1, 1))

# ---------------------------------------------------------------------
#
# Create the FEA beam
#
# ---------------------------------------------------------------------

beam_length = 2.0
beam_width = 0.1
beam_height = 0.1

beam_material = fea.ChMaterialShellANCF(100e9, 0.3, 7850)  # Young's modulus, Poisson ratio, density

beam_section = fea.ChBeamSectionRectangular(beam_width, beam_height)

beam = fea.ChBeamFEA(beam_section, beam_material)
beam.SetNumElements(20)

# Set initial position and orientation of the beam
beam.SetBodyFixed(True)
beam.SetPos(chrono.ChVectorD(0, 0, 0))
beam.SetRot(chrono.Q_from_AngAxis(math.pi / 2, chrono.ChVectorD(0, 0, 1)))

# Add the beam to the system
mysystem.Add(beam)

# ---------------------------------------------------------------------
#
# Create fixed constraints
#
# ---------------------------------------------------------------------

# Fix one end of the beam
fixed_constraint = chrono.ChLinkLockRevolute()
fixed_constraint.Initialize(beam.GetNode(0), mysystem.GetBody(0))
mysystem.Add(fixed_constraint)

# ---------------------------------------------------------------------
#
# Define custom motor function for the other end of the beam
#
# ---------------------------------------------------------------------

def motor_function(time):
    return 0.1 * math.sin(2 * math.pi * 0.5 * time)

motor_constraint = chrono.ChLinkMotorRotationSpeed()
motor_constraint.Initialize(beam.GetNode(beam.GetNumNodes() - 1), mysystem.GetBody(0))
motor_constraint.SetSpeedFunction(motor_function)
mysystem.Add(motor_constraint)

# ---------------------------------------------------------------------
#
# Set solver and timestepper
#
# ---------------------------------------------------------------------

mysystem.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)
mysystem.SetSolverMaxIterations(100)
mysystem.SetTimestepperType(chrono.ChTimestepper.Type_EULER_IMPLICIT_LINEARIZED)
mysystem.SetTimestep(0.001)

# ---------------------------------------------------------------------
#
# Simulation loop
#
# ---------------------------------------------------------------------

while vis.Run():
    mysystem.DoStepDynamics(0.01)
    vis.Render()

```

**Explanation:**

1. **Initialization:** The code initializes the PyChrono environment, sets gravity, and creates the Irrlicht visualization system.

2. **FEA Beam:** A beam object is created using the `fea` module, specifying material properties, cross-section, and number of elements. The beam is positioned and oriented in the scene.

3. **Constraints:** Fixed constraints are added to fix one end of the beam, and a custom motor function is defined to control the rotation of the other end.

4. **Solver and Timestepper:** The code sets up a suitable solver and timestepper for accurate dynamics simulation.

5. **Simulation Loop:** The simulation loop runs until the Irrlicht window is closed. In each iteration, the system dynamics are updated, and the scene is rendered.
