```python
import chrono
from chrono.collision import collision as col
from chrono.irrlicht import irrlicht as il
from chrono.vehicle import vehicle as veh

# Initialize the PyChrono environment and core components
chrono.SetChronoDataPath('path/to/chrono/data')
my_system = chrono.ChSystemSolverSH()

# Add the required physical systems and objects
ground = my_system.Add(col.ChCollisionModel())
ground_shape = ground.AddBox(100, 1, 100)
ground_shape.SetPos(chrono.ChVectorD(0, -1, 0))
ground.SetContactMethod(col.ChCollisionModel.ContactMethod.NSC)

rover = my_system.Add(veh.ChVehicle())
chassis = rover.GetChassis()
chassis.SetPos(chrono.ChVectorD(0, 1, 0))
chassis.AddBox(1, 1, 1)

motor = veh.ChVehicleMotorRT(100, 50)
rover.AddMotor(motor, veh.ChVehicleMotorRT.ID_FRONT)

# Set necessary default parameters such as positions, forces, and interactions
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Visualization settings
vis = il.ChVisualSystemIrrlicht()
vis.SetChronoSystem(my_system)
vis.Initialize()
vis.AddLogo('path/to/your/logo.png')
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 5, 5), chrono.ChVectorD(0, 0, 0))
vis.SetAmbientLight(chrono.ChVectorD(0.2, 0.2, 0.2))
vis.SetSpecularLight(chrono.ChVectorD(0.5, 0.5, 0.5))
vis.SetBackgroundColor2(chrono.ChVectorD(0.3, 0.3, 0.3))
vis.SetShadowQuality(il.ChVisualSystemIrrlicht.SHADOW_QUALITY_MEDIUM)

# Run the simulation
step = 0
while step < 1000:
    my_system.DoStepDynamics(1 / 60.0)
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()
    step += 1

vis.Finalize()
```

Please replace `'path/to/chrono/data'` and `'path/to/your/logo.png'` with the actual paths to the Chrono data and your custom logo on your system.
