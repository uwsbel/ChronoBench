# PyChrono HMMWV on SCM Deformable Terrain Simulation

```python
# Import required modules
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr
import os
import math
import numpy as np

# -----------------------------------------------------
# Initialize the simulation environment
# -----------------------------------------------------

# Set the path to Chrono data directory
chrono.SetChronoDataPath(os.path.join(os.path.dirname(os.path.abspath(__file__)), "data/"))

# Create the system
system = chrono.ChSystemSMC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# Set solver parameters
system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)
system.SetSolverMaxIterations(150)
system.SetMaxPenetrationRecoverySpeed(4.0)

# -----------------------------------------------------
# Create the vehicle
# -----------------------------------------------------

# Create the HMMWV vehicle
vehicle_file = veh.GetDataFile("hmmwv/vehicle/HMMWV_Vehicle.json")
tire_file = veh.GetDataFile("hmmwv/tire/HMMWV_RigidTire.json")
powertrain_file = veh.GetDataFile("hmmwv/powertrain/HMMWV_ShaftsPowertrain.json")

# Initial vehicle position and orientation
init_pos = chrono.ChVectorD(0, 1.0, 0)
init_rot = chrono.ChQuaternionD(1, 0, 0, 0)

# Create the vehicle, set visualization mode, tire model, and powertrain
hmmwv = veh.HMMWV_Full()
hmmwv.SetContactMethod(chrono.ChContactMethod_SMC)
hmmwv.SetChassisCollisionType(veh.ChassisCollisionType_NONE)
hmmwv.SetChassisFixed(False)
hmmwv.SetInitPosition(chrono.ChCoordsysD(init_pos, init_rot))
hmmwv.SetTireType(veh.TireModelType_RIGID)
hmmwv.SetTireStepSize(1e-3)
hmmwv.SetPowertrainType(veh.PowertrainModelType_SHAFTS)
hmmwv.Initialize()

vehicle = hmmwv.GetVehicle()
vehicle.SetChassisVisualizationType(veh.VisualizationType_MESH)
vehicle.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetWheelVisualizationType(veh.VisualizationType_MESH)

# -----------------------------------------------------
# Create the terrain
# -----------------------------------------------------

# SCM deformable soil parameters
SCM_params = chrono.SCMDeformableSoilParameters()
SCM_params.Bekker_Kphi = 0.2e6   # Pressure-sinkage coefficient (N/m^3)
SCM_params.Bekker_Kc = 0         # Cohesive pressure-sinkage coefficient (N/m^2)
SCM_params.Bekker_n = 1.1        # Exponent of the pressure-sinkage relationship
SCM_params.Mohr_cohesion = 0     # Cohesion (Pa)
SCM_params.Mohr_friction = 30    # Friction angle (degrees)
SCM_params.Janosi_shear = 0.01   # Shear deformation modulus (m)
SCM_params.elastic_K = 4e7       # Elastic stiffness (Pa/m)
SCM_params.damping_R = 3e4       # Damping (Pa.s/m)

# Create the SCM terrain
terrain = veh.SCMTerrain(system)
terrain.SetSoilParameters(SCM_params)

# Terrain grid parameters
terrain_length = 100.0  # Size in X direction
terrain_width = 100.0   # Size in Y direction
delta = 0.05            # Grid cell size

# Set the terrain height with a flat surface
terrain.Initialize(terrain_length, terrain_width, delta)

# Optional: Add a custom height map (example of adding a bump)
def height_function(x, y):
    # Example: create a small bump in the middle of the terrain
    bump_x = terrain_length / 2
    bump_y = terrain_width / 2
    dist = math.sqrt((x - bump_x)**2 + (y - bump_y)**2)
    if dist < 5.0:
        return 0.2 * (1 - dist/5.0)**2
    return 0.0

# Apply the height function to the terrain
for ix in range(terrain.GetNx()):
    for iy in range(terrain.GetNy()):
        x = terrain.GetMeshX(ix)
        y = terrain.GetMeshY(iy)
        terrain.SetHeight(ix, iy, height_function(x, y))

# Set visualization properties
terrain.SetPlotType(veh.SCMTerrain.PLOT_PRESSURE, 0, 30000)
terrain.SetMeshWireframe(True)

# -----------------------------------------------------
# Create the driver
# -----------------------------------------------------

# Create the driver system
driver_data = veh.ChDataDriver(vehicle_file)
driver = veh.ChDriver(vehicle.GetVehicle())
driver.Initialize()

# Set the driver inputs (initial throttle to get the vehicle moving)
driver_inputs = veh.DriverInputs()
driver_inputs.m_throttle = 0.3
driver_inputs.m_steering = 0.0
driver_inputs.m_braking = 0.0

# -----------------------------------------------------
# Create the visualization
# -----------------------------------------------------

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle("HMMWV on SCM Terrain")
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 5, -10), chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()

# -----------------------------------------------------
# Simulation loop
# -----------------------------------------------------

# Simulation settings
step_size = 1e-3
realtime_timer = chrono.ChRealtimeStepTimer()

print("Simulation started...")

while vis.Run():
    # Get current time
    time = system.GetChTime()
    
    # Get driver inputs
    driver.Synchronize(time)
    driver.SetInputs(driver_inputs)
    
    # Synchronize systems
    vehicle.Synchronize(time, driver_inputs, terrain)
    terrain.Synchronize(time)
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    
    # Advance simulation
    system.DoStepDynamics(step_size)
    
    # Try to keep real-time simulation
    realtime_timer.Spin(step_size)
    
    # Exit after a set time
    if time > 20:
        break

print("Simulation ended.")
```

This script creates a PyChrono simulation with an HMMWV vehicle on an SCM deformable terrain. The key components include:

1. **System Initialization**: Sets up the ChSystem with SMC contact method and appropriate solver parameters.

2. **Vehicle Creation**: Creates an HMMWV vehicle with specific visualization settings, tire model, and powertrain.

3. **SCM Terrain Setup**: 
   - Initializes an SCM deformable terrain with soil parameters
   - Creates a flat terrain with a small bump in the middle using a custom height function
   - Sets visualization properties for the terrain

4. **Driver System**: Initializes a driver system with initial inputs for throttle, steering, and braking.

5. **Visualization**: Sets up Irrlicht visualization with camera and lighting settings.

6. **Simulation Loop**: 
   - Synchronizes the vehicle, terrain, and driver systems
   - Renders the scene
   - Advances the simulation with a fixed timestep
   - Maintains real-time execution using a timer
   - Automatically exits after 20 seconds of simulation

The script provides a complete simulation environment for testing vehicle dynamics on deformable terrain.
