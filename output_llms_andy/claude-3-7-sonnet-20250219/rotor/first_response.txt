# PyChrono Jeffcott Rotor Simulation using IGA Beam

```python
# Import required modules
import pychrono as chrono
import pychrono.fea as fea
import pychrono.irrlicht as chronoirr
import numpy as np

# Global simulation parameters
simulation_time = 10.0  # Simulation duration in seconds
time_step = 0.001      # Time step for the solver
beam_length = 1.0      # Length of the beam in meters
beam_radius = 0.01     # Radius of the beam in meters
flywheel_mass = 2.0    # Mass of the flywheel in kg
flywheel_radius = 0.1  # Radius of the flywheel in meters
motor_speed = 30       # Motor speed in rad/s

# Create the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create a mesh for the IGA beam
mesh = fea.ChMesh()
mesh.SetAutomaticGravity(True)

# Create section for the IGA beam (circular cross-section)
beam_section = fea.ChBeamSectionCosserat(
    beam_area=np.pi * beam_radius**2,
    beam_Iyy=(np.pi/4) * beam_radius**4,
    beam_Izz=(np.pi/4) * beam_radius**4,
    beam_Jxx=(np.pi/2) * beam_radius**4,
    beam_E=210e9,  # Young's modulus for steel
    beam_G=80e9,   # Shear modulus for steel
)

# Create builder for IGA beam
builder = fea.ChBuilderBeamIGA()

# Define start and end points of the beam
start_point = chrono.ChVectorD(0, 0, 0)
end_point = chrono.ChVectorD(beam_length, 0, 0)

# Create the beam with a given section and number of control points
n_control_points = 10
beam_nodes = builder.BuildBeam(
    mesh=mesh,
    section=beam_section,
    n_control_points=n_control_points,
    start_point=start_point,
    end_point=end_point,
    fixed_A=True,   # Fixed at end A
    fixed_B=False   # Free at end B
)

# Attach the mesh to the system
system.Add(mesh)

# Create a visualization asset for the beam
for beam_element in builder.GetLastBeamElements():
    beam_element.AddVisualShape(chrono.ChColorAsset(0.0, 0.6, 0.2))

# Get the middle node for attaching the flywheel
middle_node_idx = n_control_points // 2
middle_node = beam_nodes[middle_node_idx]

# Create a flywheel (disk) as a rigid body
flywheel = chrono.ChBodyEasyCylinder(
    flywheel_radius,  # radius
    0.05,             # height
    7800,             # density (kg/m^3)
    True,             # visualization
    True              # collision
)
flywheel.SetPos(chrono.ChVectorD(beam_length/2, 0, 0))
flywheel.SetRot(chrono.Q_from_AngAxis(chrono.CH_C_PI_2, chrono.ChVectorD(0, 0, 1)))
flywheel.SetMass(flywheel_mass)
flywheel.SetInertiaXX(chrono.ChVectorD(
    (1/4)*flywheel_mass*flywheel_radius**2 + (1/12)*flywheel_mass*0.05**2,
    (1/4)*flywheel_mass*flywheel_radius**2 + (1/12)*flywheel_mass*0.05**2,
    (1/2)*flywheel_mass*flywheel_radius**2
))
flywheel.AddAsset(chrono.ChColorAsset(0.6, 0.2, 0.2))
system.Add(flywheel)

# Create a constraint between the flywheel and the middle node of the beam
constraint = chrono.ChLinkMateFix()
constraint.Initialize(middle_node, flywheel)
system.Add(constraint)

# Create a rotational motor at the first node
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(
    beam_nodes[0],                    # First beam node
    chrono.ChFrameD(start_point),     # Ground frame
    chrono.ChAxisD(chrono.ChVectorD(1, 0, 0))  # Rotation axis along X
)
motor_fun = chrono.ChFunction_Ramp(0, motor_speed)  # Ramp up to target speed
motor.SetSpeedFunction(motor_fun)
system.Add(motor)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Jeffcott Rotor IGA Beam Simulation")
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0.5, 0.5, 1.0), chrono.ChVectorD(0.5, 0, 0))
vis.AddTypicalLights()

# Set solver parameters
system.SetSolverType(chrono.ChSolver.Type_MINRES)
system.SetSolverMaxIterations(100)
system.SetSolverTolerance(1e-5)
system.SetTimestepperType(chrono.ChTimestepper.Type_HHT)
integrator = system.GetTimestepper()
integrator.SetAlpha(-0.2)
integrator.SetMaxiters(5)
integrator.SetAbsTolerances(1e-5)
integrator.SetMode(chrono.ChTimestepperHHT.POSITION)
integrator.SetScaling(True)

# Run the simulation
time = 0
while time < simulation_time and vis.Run():
    vis.BeginScene()
    vis.Render()
    
    # Display simulation info
    info = "Time: {:.2f} s".format(time)
    vis.DrawAll()
    chronoirr.drawAllContactPoints(vis, 4, chrono.ChColor(0, 1, 0))
    chronoirr.drawAllLinks(vis, 0.2)
    chronoirr.drawAllCOGs(vis, 0.1)
    vis.EndScene()
    
    # Advance simulation
    system.DoStepDynamics(time_step)
    time += time_step

print("Simulation completed!")
```

This script simulates a Jeffcott rotor using a PyChrono Isogeometric Analysis (IGA) beam. The key components are:

1. An IGA beam with fixed-free boundary conditions
2. A flywheel (cylindrical disk) attached at the middle of the beam
3. A rotational motor at one end driving the system

The simulation models the dynamic behavior of this rotor system, including:
- The flexibility of the shaft (beam)
- The inertial effects of the flywheel
- The dynamics of rotation and any resulting vibrations

The visualization is done using Irrlicht, showing the beam deformation and flywheel rotation during operation. You can modify parameters like motor speed, flywheel mass, and beam properties to observe different dynamic behaviors characteristic of rotor systems.
