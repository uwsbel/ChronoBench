```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Set up the Chrono system
chrono.SetChronoDataPath("/path/to/chrono/data/")

# Create a Chrono system
sys = chrono.ChSystemNSC()

# Add the ground body
ground = chrono.ChBodyEasyBox(sys,  # Parent system
                              100, 100, 2,  # x, y, z dimensions
                              1000,  # density
                              True,  # visualization?
                              True,  # collide?
                              0.9)  # friction coefficient
ground.SetBodyFixed(True)
sys.Add(ground)

# Initialize the Viper rover
rover = chrono.ChViperRover(sys)
rover.SetInitialPosition(chrono.ChVectorD(0, 0, 2))
rover.SetInitialRotation(chrono.ChQuaternionD(1, 0, 0, 0))
sys.Add(rover)

# Initialize the driver
driver = chrono.ChViperRoverDriver(rover)
sys.Add(driver)

# Create an Irrlicht application
app = chronoirr.ChVisualSystemIrrlicht()
app.AttachSystem(sys)
app.SetWindowSize(800, 600)
app.SetWindowTitle("Viper Rover Simulation")

# Set up the camera
camera = chronoirr.ChIrrNodeCamera()
camera.SetPosition(chrono.ChVectorD(0, -15, 5))
camera.SetTarget(chrono.ChVectorD(0, 0, 2))
app.AddCamera(camera)

# Set up lighting
light = chronoirr.ChIrrNodeLight()
light.SetPosition(chrono.ChVectorD(0, -15, 10))
app.AddLight(light)

# Simulation loop
t_end = 10.0  # end time
t_step = 0.01  # time step
t = 0
while t < t_end:
    # Update the rover's steering
    steering_angle = np.sin(t * 0.1) * 0.5
    driver.SetSteering(steering_angle)

    # Update the system
    sys.DoStepDynamics(t_step)

    # Render the system
    app.BeginScene()
    app.DrawAll()
    app.EndScene()

    # Update the time
    t += t_step
```

This script creates a PyChrono simulation with a Viper rover on a rigid terrain, sets up the Chrono system with gravity and ground body, initializes the Viper rover and its driver, creates a real-time visualization using Irrlicht with camera settings and lighting, and implements a simulation loop where the rover's steering gradually changes over a specified time period, updating and rendering the rover at each timestep.
