

```python
import pychrono as chrono
import pychrono.robotics as turtlebot_module
import pychrono.irrlicht as irrlicht

# Initialize the Chrono system with gravity
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the ground body (rigid terrain)
ground = chrono.ChBody()
ground.SetBodyFixed(True)
ground.SetCollide(True)
ground.GetCollisionModel().ClearModel()
ground.GetCollisionModel().AddBox(100, 1, 100)  # x, y, z half-lengths
ground.SetPos(chrono.ChVectorD(0, 0, 0))
system.Add(ground)

# Initialize the Turtlebot with specific position and orientation
# Position: (0, 0.6, 0) to place it above the ground, orientation: no rotation around Y-axis
turtlebot = turtlebot_module.Turtlebot(
    system,
    chrono.ChVectorD(0, 0.6, 0),
    chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 1, 0))
)

# Set up Irrlicht visualization
application = irrlicht.ChIrrApp(system, 'Turtlebot Simulation', irrlicht.dimension2du(1024, 768))
application.AddTypicalLogo()
application.AddTypicalCamera(chrono.Vector3d(0, 3, -6), chrono.Vector3d(0, 0, 0))
application.AddTypicalLights()
application.AssetBindAll()
application.AssetUpdateAll()

# Simulation loop
while application.GetDevice().run():
    current_time = system.GetChTime()

    # Control logic for Turtlebot's motors based on simulation time
    if current_time < 1.0:
        left_torque, right_torque = 0.0, 0.0
    elif current_time < 2.0:
        left_torque, right_torque = -1.0, 1.0  # Turn left
    elif current_time < 3.0:
        left_torque, right_torque = 1.0, -1.0   # Turn right
    else:
        left_torque, right_torque = 0.0, 0.0

    # Apply torques to Turtlebot's motors
    turtlebot.SetMotorTorque(left_torque, right_torque)

    # Step the simulation forward
    system.DoStepDynamics(1e-3)

    # Render the scene
    application.BeginScene()
    application.DrawAll()
    application.EndScene()
```
