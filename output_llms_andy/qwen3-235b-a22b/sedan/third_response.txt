

```python
import pychrono.core as chrono
import pychrono.irrlicht as irr
import pychrono.vehicle as veh
import math

# Set data path (ensure this is correct for your system)
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

# Adjusted initial vehicle location and orientation
initLoc = chrono.ChVector3d(0, 0, 1.5)  # Increased height for better terrain clearance
initRot = chrono.ChQuaterniond(1, 0, 0, 0)  # No rotation (identity quaternion)

# Visualization and simulation parameters
vis_type = veh.VisualizationType_MESH
chassis_collision_type = veh.CollisionType_NONE
tire_model = veh.TireModelType_TMEASY

# Decreased step sizes for finer control
step_size = 5e-4  # Reduced from 1e-3
render_step_size = 1.0 / 100  # Increased FPS to 100

# Terrain dimensions
terrainLength = 200.0
terrainWidth = 100.0

# Camera tracking point
trackPoint = chrono.ChVector3d(-5.0, 0.0, 1.8)

# Contact method
contact_method = chrono.ChContactMethod_NSC

# Create vehicle system
vehicle = veh.BMW_E90()
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
vehicle.SetTireType(tire_model)
vehicle.SetTireStepSize(step_size)
vehicle.Initialize()

# Set visualization types
vehicle.SetChassisVisualizationType(vis_type)
vehicle.SetSuspensionVisualizationType(vis_type)
vehicle.SetSteeringVisualizationType(vis_type)
vehicle.SetWheelVisualizationType(vis_type)
vehicle.SetTireVisualizationType(vis_type)

# Enable Bullet collision system
vehicle.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# Create terrain with highway mesh
try:
    # Attempt to load a highway mesh (replace with actual path if available)
    mesh = chrono.ChTriangleMeshConnected()
    mesh.LoadWavefrontMesh(chrono.GetChronoDataFile("terrain/mesh/highway.obj"), False, True)
    
    terrain_body = chrono.ChBody()
    terrain_body.SetPos(chrono.ChVector3d(0, 0, 0))
    terrain_body.SetFixed(True)
    
    # Collision model
    terrain_body.GetCollisionModel().ClearModel()
    chrono.ChCollisionModel.AddTriangleMesh(
        terrain_body, mesh, False, False, chrono.ChVector3d(0,0,0), chrono.ChMatrix33d()
    )
    terrain_body.SetMesh(mesh)
    vehicle.GetSystem().Add(terrain_body)
except:
    # Fallback to flat terrain with highway-like texture if mesh fails
    patch_mat = chrono.ChContactMaterialNSC()
    patch_mat.SetFriction(0.9)
    patch_mat.SetRestitution(0.01)
    terrain = veh.RigidTerrain(vehicle.GetSystem())
    patch = terrain.AddPatch(patch_mat, 
        chrono.ChCoordsysd(chrono.ChVector3d(0, 0, 0), chrono.QUNIT), 
        terrainLength, terrainWidth)
    patch.SetTexture(veh.GetDataFile("terrain/textures/track1.jpg"), 200, 200)
    patch.SetColor(chrono.ChColor(0.2, 0.2, 0.2))  # Dark gray for highway appearance
    terrain.Initialize()

# Create visualization system
vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle('Sedan - PID Controlled Highway Simulation')
vis.SetWindowSize(1280, 1024)
vis.SetChaseCamera(trackPoint, 6.0, 0.5)
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetVehicle())

# Create driver system with increased steering response time
driver = veh.ChInteractiveDriverIRR(vis)
steering_time = 5.0  # Increased from 1.0
throttle_time = 1.0
braking_time = 0.3
driver.SetSteeringDelta(render_step_size / steering_time)
driver.SetThrottleDelta(render_step_size / throttle_time)
driver.SetBrakingDelta(render_step_size / braking_time)
driver.Initialize()

# PID controller parameters
kp = 0.12
ki = 0.015
kd = 0.002
ref_speed = 30.0  # Target speed in m/s
prev_error = integral = 0.0

# Simulation loop variables
render_steps = math.ceil(render_step_size / step_size)
realtime_timer = chrono.ChRealtimeStepTimer()
step_number = 0

# Output vehicle mass
print("VEHICLE MASS: ", vehicle.GetVehicle().GetMass())

while vis.Run():
    time = vehicle.GetSystem().GetChTime()

    # Render scene
    if step_number % render_steps == 0:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Handle reference speed adjustments
    if vis.KeyDown(irr.KEY_UP):
        ref_speed += 0.5
    if vis.KeyDown(irr.KEY_DOWN):
        ref_speed -= 0.5
    ref_speed = max(0.0, min(40.0, ref_speed))  # Clamp speed between 0-40 m/s

    # PID speed control
    current_speed = vehicle.GetVehicle().GetSpeed()
    error = ref_speed - current_speed
    integral += error * step_size
    derivative = (error - prev_error) / step_size
    throttle = kp * error + ki * integral + kd * derivative
    throttle = max(0.0, min(1.0, throttle))  # Clamp throttle
    driver_inputs.throttle = throttle
    prev_error = error

    # Update modules
    driver.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain if 'terrain' in locals() else terrain_body)
    vis.Synchronize(time, driver_inputs)

    # Advance simulation
    driver.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)
    step_number += 1

    realtime_timer.Spin(step_size)
```
