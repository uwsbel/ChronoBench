#usda 1.0
(
    defaultPrim = "World"
    metersPerUnit = 1.0
    kilogramsPerUnit = 1.0
    upAxis = "Z"
    doc = """Isaac Sim articulation: a velocity-driven crank carrying a damped free pendulum.
    Authored in Isaac's idiom: UsdPhysics rigid bodies + MassAPI inertials + PhysX DriveAPI.
    Angular drive targets follow the USD Physics schema convention (DEGREES, degrees/second);
    the intended crank rate is 1.5 rad/s = 85.9437 deg/s. The elbow drive has zero stiffness,
    zero target velocity, and only damping: a pure viscous rotary damper (0.05 N m s/rad,
    expressed below in the schema's per-degree units)."""
)

def Xform "World"
{
    def PhysicsScene "PhysicsScene"
    {
        vector3f physics:gravityDirection = (0, 0, -1)
        float physics:gravityMagnitude = 9.81
    }

    def Xform "CrankPendulum" (
        prepend apiSchemas = ["PhysicsArticulationRootAPI"]
    )
    {
        def Xform "Mount" (
            prepend apiSchemas = ["PhysicsRigidBodyAPI"]
        )
        {
            double3 xformOp:translate = (0, 0, 1.5)
            uniform token[] xformOpOrder = ["xformOp:translate"]
        }

        def PhysicsFixedJoint "WorldAnchor"
        {
            rel physics:body1 = </World/CrankPendulum/Mount>
        }

        def Xform "Crank" (
            prepend apiSchemas = ["PhysicsRigidBodyAPI", "PhysicsMassAPI"]
        )
        {
            double3 xformOp:translate = (0, 0, 1.5)
            uniform token[] xformOpOrder = ["xformOp:translate"]
            float physics:mass = 0.5
            point3f physics:centerOfMass = (0, 0, -0.2)
            float3 physics:diagonalInertia = (0.0066667, 0.0066667, 0.0001)

            def Capsule "Geom"
            {
                double height = 0.36
                double radius = 0.02
                double3 xformOp:translate = (0, 0, -0.2)
                uniform token[] xformOpOrder = ["xformOp:translate"]
            }
        }

        def PhysicsRevoluteJoint "CrankDrive" (
            prepend apiSchemas = ["PhysicsDriveAPI:angular"]
        )
        {
            rel physics:body0 = </World/CrankPendulum/Mount>
            rel physics:body1 = </World/CrankPendulum/Crank>
            uniform token physics:axis = "Y"
            point3f physics:localPos0 = (0, 0, 0)
            point3f physics:localPos1 = (0, 0, 0)
            float drive:angular:physics:stiffness = 0
            float drive:angular:physics:damping = 50
            float drive:angular:physics:targetVelocity = 85.9437
            float drive:angular:physics:maxForce = 500
        }

        def Xform "Pendulum" (
            prepend apiSchemas = ["PhysicsRigidBodyAPI", "PhysicsMassAPI"]
        )
        {
            double3 xformOp:translate = (0, 0, 1.1)
            uniform token[] xformOpOrder = ["xformOp:translate"]
            float physics:mass = 0.3
            point3f physics:centerOfMass = (0, 0, -0.25)
            float3 physics:diagonalInertia = (0.00625, 0.00625, 0.0001)

            def Capsule "Geom"
            {
                double height = 0.46
                double radius = 0.02
                double3 xformOp:translate = (0, 0, -0.25)
                uniform token[] xformOpOrder = ["xformOp:translate"]
            }
        }

        def PhysicsRevoluteJoint "Elbow" (
            prepend apiSchemas = ["PhysicsDriveAPI:angular"]
        )
        {
            rel physics:body0 = </World/CrankPendulum/Crank>
            rel physics:body1 = </World/CrankPendulum/Pendulum>
            uniform token physics:axis = "Y"
            point3f physics:localPos0 = (0, 0, -0.4)
            point3f physics:localPos1 = (0, 0, 0)
            float drive:angular:physics:stiffness = 0
            float drive:angular:physics:damping = 0.000873
            float drive:angular:physics:targetVelocity = 0
        }
    }
}
