import pychrono.core as chrono
import pychrono.fea as fea
import pychrono.irrlicht as chronoirr

# ----------------------------------------------------------------------------
# Model1: Multiple chains of beam elements with connected bodies
# This model demonstrates the use of the utility class ChBuilderCableANCF
# with multiple chains and connected rigid bodies.
# ----------------------------------------------------------------------------

class Model1:
    def __init__(self, system, mesh, n_chains=6):
        self.system = system
        self.mesh = mesh
        self.n_chains = n_chains
        self.end_bodies = []  # Store end bodies for position printing
        
        # Create multiple chains
        for i in range(n_chains):
            self.create_chain(i)
    
    def create_chain(self, chain_index):
        # Create a section, i.e. define thickness and material properties for the cable beam
        msection_cable = fea.ChBeamSectionCable()
        msection_cable.SetDiameter(0.015)  # Set the diameter of the cable section to 15 mm
        msection_cable.SetYoungModulus(0.01e9)  # Set the Young's modulus of the cable section (0.01 GPa)
        msection_cable.SetRayleighDamping(0.0001)  # Set Rayleigh damping
        
        # Create a ChBuilderCableANCF helper object to facilitate the creation of ANCF beams
        builder = fea.ChBuilderCableANCF()
        
        # Number of elements increases with each chain
        num_elements = 5 + chain_index * 2
        
        # Set relative positions for starting and ending points to avoid overlap
        start_x = chain_index * 0.3  # Spread chains horizontally
        start_y = 0
        start_z = -0.1
        end_x = start_x
        end_y = 0
        end_z = -0.1 - 0.5  # Hang downward
        
        # Use BuildBeam to create a beam structure consisting of ANCF elements:
        builder.BuildBeam(
            self.mesh,  # The mesh to which the created nodes and elements will be added
            msection_cable,  # The beam section properties to use
            num_elements,  # Number of ANCF elements to create along the beam
            chrono.ChVector3d(start_x, start_y, start_z),  # Starting point of the beam
            chrono.ChVector3d(end_x, end_y, end_z)  # Ending point of the beam
        )

        # Create a truss body (a fixed reference frame in the simulation)
        mtruss = chrono.ChBody()
        mtruss.SetFixed(True)  # Fix the truss body
        mtruss.SetPos(chrono.ChVector3d(start_x, start_y, start_z))
        self.system.Add(mtruss)

        # Create and initialize a hinge constraint to fix beam's start point to the truss
        constraint_hinge = fea.ChLinkNodeFrame()
        constraint_hinge.Initialize(builder.GetLastBeamNodes().back(), mtruss)
        self.system.Add(constraint_hinge)  # Add the constraint to the system

        # Create a box body at the end of the beam
        end_body = chrono.ChBody()
        end_body.SetMass(0.1)  # Set mass of the box
        end_body.SetPos(chrono.ChVector3d(end_x, end_y, end_z))
        end_body.SetInertiaXX(chrono.ChVector3d(0.001, 0.001, 0.001))
        
        # Add visual shape to the box
        box_shape = chrono.ChVisualShapeBox(0.02, 0.02, 0.02)  # Small box
        box_shape.SetColor(chrono.ChColor(0.8, 0.2, 0.2))  # Red color
        end_body.AddVisualShape(box_shape)
        
        # Add collision shape to the box
        end_body.EnableCollision(True)
        collision_shape = chrono.ChCollisionShapeBox(None, 0.02, 0.02, 0.02)
        end_body.AddCollisionShape(collision_shape)
        
        self.system.Add(end_body)
        self.end_bodies.append(end_body)  # Store for position printing

        # Create constraint between beam end and box
        constraint_beam_box = fea.ChLinkNodeFrame()
        constraint_beam_box.Initialize(builder.GetLastBeamNodes().front(), end_body)
        self.system.Add(constraint_beam_box)

        # Apply forces to the end body
        end_body.SetForce(chrono.ChVector3d(0, -0.5, 0))  # Apply downward force
    
    def PrintBodyPositions(self):
        """Print positions of end bodies of each chain"""
        print("End body positions:")
        for i, body in enumerate(self.end_bodies):
            pos = body.GetPos()
            print(f"Chain {i}: ({pos.x:.3f}, {pos.y:.3f}, {pos.z:.3f})")
        print("---")
