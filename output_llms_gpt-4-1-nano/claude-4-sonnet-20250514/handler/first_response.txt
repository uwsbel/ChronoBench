```python
#!/usr/bin/env python3

import pychrono as chrono
import pychrono.ros as chros
from std_msgs.msg import Int32
import time

class CustomROSHandler(chros.ChROSHandler):
    """Custom ROS handler to publish integer messages"""
    
    def __init__(self, topic_name, publish_rate):
        super().__init__(publish_rate)
        self.topic_name = topic_name
        self.publisher = None
        self.counter = 0
        
    def Initialize(self, node):
        """Initialize the ROS publisher"""
        print(f"Creating publisher for topic {self.topic_name}")
        self.publisher = node.create_publisher(Int32, self.topic_name, 1)
        return True
        
    def Tick(self, time):
        """Publish integer message at each tick"""
        if self.publisher:
            msg = Int32()
            msg.data = self.counter
            self.publisher.publish(msg)
            print(f"Published: {self.counter} to topic {self.topic_name}")
            self.counter += 1

def main():
    print("Creating Chrono system...")
    
    # Initialize the PyChrono environment and core components
    system = chrono.ChSystemNSC()
    system.SetGravitationalAcceleration(chrono.ChVector3d(0, -9.81, 0))
    
    # Set necessary default parameters - collision system
    system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
    
    # Create physical material properties
    material = chrono.ChContactMaterialNSC()
    material.SetFriction(0.4)
    material.SetRestitution(0.1)
    
    # Add a fixed floor
    print("Adding fixed floor...")
    floor_body = chrono.ChBodyEasyBox(10, 0.2, 10, 1000, True, True, material)
    floor_body.SetPos(chrono.ChVector3d(0, -0.1, 0))
    floor_body.SetFixed(True)
    floor_body.GetVisualShape(0).SetColor(chrono.ChColor(0.2, 0.2, 0.8))  # Blue floor
    system.Add(floor_body)
    
    # Add a movable box
    print("Adding movable box...")
    box_body = chrono.ChBodyEasyBox(1, 1, 1, 1000, True, True, material)
    box_body.SetPos(chrono.ChVector3d(0, 5, 0))  # Start 5 meters above ground
    box_body.SetPosDt(chrono.ChVector3d(1, 0, 0))  # Initial velocity in x direction
    box_body.GetVisualShape(0).SetColor(chrono.ChColor(0.8, 0.2, 0.2))  # Red box
    system.Add(box_body)
    
    # Create ROS manager
    print("Setting up ROS manager...")
    ros_manager = chros.ChROSPythonManager()
    ros_manager.Initialize()
    
    # Configure ROS manager with handlers
    
    # 1. Clock handler - publishes simulation time
    clock_handler = chros.ChROSClockHandler()
    ros_manager.RegisterHandler(clock_handler)
    
    # 2. Body handler - publishes body states
    body_handler = chros.ChROSBodyHandler(25.0, box_body, "~/box_state")  # 25 Hz
    ros_manager.RegisterHandler(body_handler)
    
    # 3. Transform handler - publishes TF transforms
    tf_handler = chros.ChROSTFHandler(30.0)  # 30 Hz
    tf_handler.AddTransform(box_body, box_body.GetName(), "world")
    tf_handler.AddTransform(floor_body, floor_body.GetName(), "world")
    ros_manager.RegisterHandler(tf_handler)
    
    # 4. Register custom handler
    custom_handler = CustomROSHandler("~/custom_int_topic", 10.0)  # 10 Hz
    ros_manager.RegisterHandler(custom_handler)
    
    # Set up simulation parameters
    timestep = 0.01  # 10ms timestep
    simulation_time = 0
    max_simulation_time = 20.0  # Run for 20 seconds
    
    print("Starting simulation loop...")
    print(f"Timestep: {timestep} seconds")
    print(f"Max simulation time: {max_simulation_time} seconds")
    
    # Run simulation loop
    while simulation_time < max_simulation_time:
        # Store real time for real-time execution
        real_time_start = time.time()
        
        # Advance the physical system
        system.DoStepDynamics(timestep)
        
        # Update ROS communication
        if not ros_manager.Update(simulation_time, timestep):
            print("ROS manager update failed, breaking simulation loop")
            break
        
        # Update simulation time
        simulation_time += timestep
        
        # Print simulation progress every 2 seconds
        if int(simulation_time * 100) % 200 == 0:  # Every 2 seconds
            box_pos = box_body.GetPos()
            box_vel = box_body.GetPosDt()
            print(f"Time: {simulation_time:.2f}s, Box pos: ({box_pos.x:.2f}, {box_pos.y:.2f}, {box_pos.z:.2f}), "
                  f"Box vel: ({box_vel.x:.2f}, {box_vel.y:.2f}, {box_vel.z:.2f})")
        
        # Maintain real-time execution
        real_time_elapsed = time.time() - real_time_start
        sleep_time = timestep - real_time_elapsed
        if sleep_time > 0:
            time.sleep(sleep_time)
        elif sleep_time < -0.001:  # If we're more than 1ms behind
            print(f"Warning: Simulation running slower than real-time by {-sleep_time*1000:.1f}ms")
    
    print("Simulation completed!")
    print(f"Final simulation time: {simulation_time:.2f} seconds")
    
    # Clean up
    print("Cleaning up ROS manager...")
    ros_manager.Shutdown()

if __name__ == '__main__':
    try:
        main()
    except KeyboardInterrupt:
        print("\nSimulation interrupted by user")
    except Exception as e:
        print(f"An error occurred: {e}")
        import traceback
        traceback.print_exc()
```

This script creates a comprehensive PyChrono simulation integrated with ROS that includes:

## Key Features:

1. **PyChrono Environment Setup**:
   - Initializes a ChSystemNSC with gravitational acceleration
   - Sets up collision system and physical material properties

2. **Physical Objects**:
   - **Fixed Floor**: A large blue box positioned as the ground
   - **Movable Box**: A red box that starts 5 meters above the ground with initial velocity

3. **Custom ROS Handler**:
   - Publishes integer messages to a specified topic
   - Increments counter with each publication
   - Inherits from ChROSHandler base class

4. **ROS Manager Configuration**:
   - **Clock Handler**: Publishes simulation time
   - **Body Handler**: Publishes the box's state information
   - **Transform Handler**: Publishes TF transforms for both objects
   - **Custom Handler**: Publishes custom integer messages

5. **Simulation Loop**:
   - Advances the physical system at each timestep
   - Updates ROS communication
   - Maintains real-time execution with proper timing
   - Provides progress feedback every 2 seconds

## Usage:

1. Make sure you have PyChrono with ROS support installed
2. Ensure ROS2 is running (`ros2 daemon start`)
3. Run the script: `python3 chrono_ros_simulation.py`
4. Monitor ROS topics: `ros2 topic list` and `ros2 topic echo <topic_name>`

The simulation will run for 20 seconds, showing the box falling and moving under gravity and physics, while simultaneously publishing various ROS messages including the custom integer topic.
