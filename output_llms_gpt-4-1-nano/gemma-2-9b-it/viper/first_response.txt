# 4. Real-time visualization using Irrlicht
vis_app = vis.ChIrrApp(system, "Viper Rover Simulation")

vis_app.SetSunPosition(chrono.ChVectorD(1, 1, 1))
vis_app.SetCamera(vis_app.GetActiveCamera())

# 5. Simulation loop
dt = 0.01  # Timestep
steering_target = 30  # Target steering angle
start_time = chrono.ChTime()

while True:
    system.DoStepDynamics(dt)
    
    # Update steering angle gradually
    current_time = chrono.ChTime()
    elapsed_time = current_time - start_time
    steering_angle = driver.Get_SteeringAngle() + (steering_target - steering_angle) * (elapsed_time / 10)  # Adjust the time for desired change rate
    driver.Set_SteeringAngle(steering_angle)
    
    vis_app.Render()
    vis_app.Pump()
