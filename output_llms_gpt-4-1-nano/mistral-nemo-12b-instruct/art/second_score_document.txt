```python
import pychrono as chrono
import pychrono.vehicle as veh

# 1. Initialize the system
sys = chrono.ChSystemNSC()

# 2. Create and setup terrain
terrain_length = 50
terrain_width = 20
terrain = veh.RigidTerrain(sys)
# Add a textured patch
patch_mat = veh.ChVisualMaterial()
patch_mat.SetDiffuseColor(0.3, 0.3, 0.3)
patch = terrain.AddPatch(patch_mat, chrono.ChCoordsysD(chrono.ChVectorD(0,0,0), chrono.QUNIT), terrain_length, terrain_width)
# Add a custom texture (assuming the texture file exists)
texture_path = chrono.GetChronoDataFile('textures/concrete.jpg')
patch.GetMaterial().SetTexture(chrono.GetChronoDataFile('textures/concrete.jpg'))
terrain.Initialize()

# 3. Create the ARTcar vehicle with specified parameters
initial_location = chrono.ChVectorD(1, 0, 0.5)  # Changed from (0,0,0.5) to (1,0,0.5)
initial_rotation = chrono.QUNIT

# Instantiate vehicle
vehicle = veh.ARTcar()

# Set contact method
vehicle.SetContactMethod(veh.ContactMethod_NSC)

# Set initial position and orientation
vehicle.SetInitPosition(chrono.ChCoordsysD(initial_location, initial_rotation))

# Set visualization type to PRIMITIVES
vehicle.SetVisualizationType(veh.VisualizationType_PRIMITIVES)

# Set chassis collision type to MESH
vehicle.SetChassisCollisionType(veh.CollisionType_MESH)

# Set tire model type to FIALA
vehicle.SetTireModelType(veh.TireModelType_FIALA)

# Initialize the vehicle
vehicle.Initialize()

# 4. Set up Irrlicht visualization
vis = veh.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('ARTcar on Rigid Terrain')
vis.Initialize()

# Add lighting and camera
vis.AddTypicalLights()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(1, 2, 3))
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))

# 5. Add an interactive driver
driver = veh.ChIrrGuiDriver(vis)
driver.SetSteeringDelta(0.02)
driver.SetThrottleDelta(0.05)
driver.SetBrakingDelta(0.05)
vehicle.SetDriver(driver)

# 6. Simulation loop at 50 FPS (~0.02s time step)
time_step = 0.02
while vis.Run():
    # Driver inputs
    driver.Synchronize()
    driver.Advance(time_step)

    # Update vehicle controls
    vehicle.GetDrive().SetThrottle(driver.GetThrottle())
    vehicle.GetDrive().SetSteering(driver.GetSteering())
    vehicle.GetDrive().SetBraking(driver.GetBraking())

    # Advance simulation
    sys.DoStepDynamics(time_step)

    # Render scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
```
