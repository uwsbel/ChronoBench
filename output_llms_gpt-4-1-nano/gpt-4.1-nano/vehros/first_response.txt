vehicle = veh.HMMWV_Vehicle(system, vehicle_params)
vehicle.Initialize()

# Set initial position and orientation if needed
initial_pos = chrono.ChVectorD(0, 0, 0.5)  # example starting position
vehicle.SetChassisPosition(initial_pos)

# Create driver system
driver = veh.ChDriver(vehicle.GetVehicle())
driver.SetInputMode(veh.ChDriver.InputMode_Pedal)
driver.SetThrottle(0.0)
driver.SetSteering(0.0)
driver.SetBraking(0.0)
