```python
import pychrono as chrono
import pychrono.sensor as sens
import math
import numpy as np

# Initialize the ChronoEngine
chrono.SetChronoDataPath('/path/to/chrono/data/')
system = chrono.ChSystemNSC()

# Create the ground
ground = chrono.ChBodyEasyBox(20, 1, 20, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -0.5, 0))
ground.SetBodyFixed(True)
system.Add(ground)

# Load the triangular mesh
mesh = chrono.ChTriangleMeshConnected()
mesh.LoadWavefrontMesh('path/to/your/mesh.obj', False, True)

# Create a rigid body and add the mesh to it
mesh_body = chrono.ChBodyEasyMesh(
    mesh,
    1000,  # density
    True,  # visualization
    True,  # collision
    chrono.ChMaterialSurfaceNSC()  # contact material
)
mesh_body.SetPos(chrono.ChVectorD(0, 1, 0))
mesh_body.SetBodyFixed(True)
system.Add(mesh_body)

# Create a sensor manager
manager = sens.ChSensorManager(system)

# Create a lidar sensor
offset_pose = chrono.ChFrameD(chrono.ChVectorD(0, 0, 0))
lidar = sens.ChLidarSensor(
    mesh_body,           # body lidar is attached to
    100.0,               # scanning rate in Hz
    offset_pose,         # offset pose
    1000,                # number of horizontal samples
    100,                 # number of vertical channels
    chrono.CH_C_PI,      # horizontal field of view
    chrono.CH_C_PI / 6,  # vertical field of view
    100                  # max range
)

# Add noise models to the lidar
lidar.SetLag(0.1)
lidar.SetCollectionWindow(0.001)
noise_model = sens.ChLidarNoiseNormal(0.01, 0.01)
lidar.SetNoiseModel(noise_model)

# Add filters to the lidar
lidar.PushFilter(sens.ChFilterDeltaPose(0.05, 0.05))
lidar.PushFilter(sens.ChFilterPCfromDepth())

# Add the lidar to the sensor manager
manager.AddSensor(lidar)

# Create a visualization window
vis = sens.ChSensorVisualization(1280, 720)
lidar.PushFilter(vis)

# Set up data saving
lidar.PushFilter(sens.ChFilterSave("lidar_data/"))

# Simulation loop
time_step = 0.001
sim_time = 0
while sim_time < 10:  # Run for 10 seconds
    system.DoStepDynamics(time_step)
    manager.Update()

    # Update lidar position to orbit around the mesh
    orbit_radius = 5.0
    orbit_height = 3.0
    orbit_speed = 0.5
    x = orbit_radius * math.cos(orbit_speed * sim_time)
    z = orbit_radius * math.sin(orbit_speed * sim_time)
    new_pos = chrono.ChVectorD(x, orbit_height, z)
    new_rot = chrono.Q_from_AngAxis(-orbit_speed * sim_time, chrono.ChVectorD(0, 1, 0))
    new_frame = chrono.ChFrameD(new_pos, new_rot)
    lidar.SetOffsetPose(new_frame)

    # Print lidar buffer data
    if lidar.GetMostRecentBuffer():
        lidar_data = lidar.GetMostRecentBuffer().GetPointCloud()
        print(f"Simulation time: {sim_time:.3f}")
        print(f"Number of lidar points: {len(lidar_data)}")
        if len(lidar_data) > 0:
            print(f"First point: {lidar_data[0]}")
        print("---")

    sim_time += time_step

# Cleanup
del manager
```

This script does the following:

1. Initializes the PyChrono environment and creates a ground plane.
2. Loads a triangular mesh from a Wavefront .obj file and creates a fixed body with this mesh.
3. Sets up a sensor manager and creates a lidar sensor attached to the mesh body.
4. Adds noise models, filters, and visualization to the lidar sensor.
5. Sets up data saving for the lidar data.
6. Runs a simulation loop for 10 seconds, updating the lidar's position to orbit around the mesh.
7. Prints lidar buffer data at each simulation step.
