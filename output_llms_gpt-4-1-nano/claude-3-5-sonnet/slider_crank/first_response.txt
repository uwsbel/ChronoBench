Here's a PyChrono script that creates a simulation of a crank-slider mechanism based on your instructions:

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Initialize the ChSystem
system = chrono.ChSystemNSC()

# Set gravity
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the floor (truss)
floor = chrono.ChBodyEasyBox(2, 0.1, 1, 1000, True)
floor.SetBodyFixed(True)
floor.SetPos(chrono.ChVectorD(0, -0.05, 0))
system.Add(floor)

# Create the crankshaft
crankshaft = chrono.ChBodyEasyCylinder(0.05, 0.1, 7800)
crankshaft.SetPos(chrono.ChVectorD(0, 0.2, 0))
crankshaft.SetRot(chrono.Q_ROTATE_X_TO_Y)
system.Add(crankshaft)

# Create the connecting rod
rod = chrono.ChBodyEasyBox(0.5, 0.05, 0.05, 7800)
rod.SetPos(chrono.ChVectorD(0.25, 0.2, 0))
system.Add(rod)

# Create the piston
piston = chrono.ChBodyEasyBox(0.1, 0.1, 0.1, 7800)
piston.SetPos(chrono.ChVectorD(0.5, 0.2, 0))
system.Add(piston)

# Create joints

# Revolute joint for crankshaft
rev_crank = chrono.ChLinkLockRevolute()
rev_crank.Initialize(crankshaft, floor, chrono.ChCoordsysD(chrono.ChVectorD(0, 0.2, 0)))
system.AddLink(rev_crank)

# Revolute joint between crankshaft and connecting rod
rev_crank_rod = chrono.ChLinkLockRevolute()
rev_crank_rod.Initialize(crankshaft, rod, chrono.ChCoordsysD(chrono.ChVectorD(0.05, 0.2, 0)))
system.AddLink(rev_crank_rod)

# Revolute joint between connecting rod and piston
rev_rod_piston = chrono.ChLinkLockRevolute()
rev_rod_piston.Initialize(rod, piston, chrono.ChCoordsysD(chrono.ChVectorD(0.5, 0.2, 0)))
system.AddLink(rev_rod_piston)

# Prismatic joint for piston
prismatic = chrono.ChLinkLockPrismatic()
prismatic.Initialize(piston, floor, chrono.ChCoordsysD(chrono.ChVectorD(0.5, 0.2, 0), chrono.Q_ROTATE_Y_TO_Z))
system.AddLink(prismatic)

# Add motor to drive the crankshaft
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(crankshaft, floor, chrono.ChFrameD(chrono.ChVectorD(0, 0.2, 0)))
motor.SetSpeedFunction(chrono.ChFunction_Const(2.0))  # 2 rad/s
system.AddLink(motor)

# Create the Irrlicht visualization
application = chronoirr.ChIrrApp(system, "Crank-Slider Mechanism", chronoirr.dimension2du(1024, 768))
application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(chronoirr.vector3df(1, 1, 1))
application.AssetBindAll()
application.AssetUpdateAll()

# Add logo (optional)
logo = chrono.ChTexture()
logo.SetTextureFilename(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
floor.AddAsset(logo)

# Simulation loop
application.SetTimestep(0.01)
application.SetTryRealtime(True)

while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```

This script does the following:

1. Initializes the PyChrono environment and core components.
2. Creates the physical objects: floor (truss), crankshaft, connecting rod, and piston.
3. Sets up the joints: revolute joints for the crankshaft and connections, and a prismatic joint for the piston.
4. Adds a motor to drive the crankshaft at a constant angular speed.
5. Sets up the Irrlicht visualization system with customizable camera settings and lighting.
6. Adds a texture (logo) to the floor as an additional visual element.
7. Runs the simulation loop, updating and rendering the scene at each step.

You can run this script to see the crank-slider mechanism in action. The crankshaft will rotate at a constant speed, driving the connecting rod and piston in a typical slider-crank motion.
