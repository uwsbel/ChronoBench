```python
import pychrono as chrono
import pychrono.sensor as sens
import os
import math

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Ensure data path is set (optional if you have your own path)
system = chrono.ChSystemNSC()

# Set gravity (optional, since the mesh is fixed and lidar is sensor only)
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

# 2. Add the required physical systems and objects as specified.

# Load a triangular mesh from a Wavefront .obj file
# For demonstration, let's assume you have a mesh file "mesh.obj" in the current directory.
mesh_obj_path = os.path.join(os.getcwd(), "mesh.obj")

# Create a fixed body to hold the mesh
mesh_body = chrono.ChBody()
mesh_body.SetBodyFixed(True)

# Load the mesh
mesh = chrono.ChTriangleMeshConnected()
if not mesh.LoadWavefrontMesh(mesh_obj_path, True, True):
    raise RuntimeError(f"Failed to load mesh from {mesh_obj_path}")

# Add the mesh as a visualization asset to the body
mesh_asset = chrono.ChTriangleMeshShape()
mesh_asset.SetMesh(mesh)
mesh_asset.SetName("mesh")
mesh_asset.SetStatic(True)
mesh_body.AddAsset(mesh_asset)

# Add the body to the system
system.Add(mesh_body)

# 3. Set necessary default parameters such as positions, forces, and interactions.

# Create the sensor manager and attach it to the system
sensor_manager = sens.ChSensorManager(system)

# Create a Lidar sensor attached to the mesh_body
lidar = sens.ChLidarSensor(mesh_body)
lidar.SetName("LidarSensor")

# Set lidar parameters
lidar.SetOffsetPose(chrono.ChFrameD())  # Initial offset pose relative to mesh_body

# Lidar parameters: (example values)
lidar.SetScanWindow(0.1)  # seconds per scan
lidar.SetHorizontalResolution(400)  # number of horizontal points per scan line
lidar.SetVerticalResolution(16)  # number of vertical scan lines
lidar.SetHorizontalFOV(math.radians(360))  # 360 degrees horizontal FOV
lidar.SetVerticalFOV(math.radians(30))  # 30 degrees vertical FOV
lidar.SetMaxRange(20)  # max range in meters
lidar.SetRotationFrequency(10)  # rotations per second

# Add noise filter to lidar data
noise_filter = sens.ChFilterLidarNoise()
noise_filter.SetNoiseStdDev(0.01)  # standard deviation of noise in meters
lidar.AddFilter(noise_filter)

# Add visualization filter to lidar data (point cloud visualization)
visual_filter = sens.ChFilterVisualizePointCloud()
visual_filter.SetPointSize(3)
lidar.AddFilter(visual_filter)

# Add filter to save lidar data to a file (e.g., PCD format)
save_filter = sens.ChFilterSavePCD()
save_filter.SetDirectory("lidar_output")
save_filter.SetBaseFilename("scan")
save_filter.SetSaveOnUpdate(True)
lidar.AddFilter(save_filter)

# Add the lidar sensor to the sensor manager
sensor_manager.AddSensor(lidar)

# Simulation parameters
step_size = 0.01  # simulation step size in seconds
simulation_time = 10  # total simulation time in seconds

# Orbit parameters for lidar around the mesh
orbit_radius = 5.0  # meters
orbit_height = 2.0  # meters above mesh origin
orbit_angular_speed = 2 * math.pi / 5  # radians per second (one orbit every 5 seconds)

# Run the simulation
time = 0.0
while time < simulation_time:
    # Update lidar position in an orbit around the mesh
    angle = orbit_angular_speed * time
    x = orbit_radius * math.cos(angle)
    y = orbit_radius * math.sin(angle)
    z = orbit_height

    # Update lidar offset pose relative to mesh_body
    # Position the lidar at (x, y, z) relative to mesh origin
    # Point lidar towards the mesh center (origin)
    position = chrono.ChVectorD(x, y, z)

    # Compute rotation so lidar looks at origin
    direction = chrono.ChVectorD(0, 0, 0) - position
    direction.Normalize()

    # Create a rotation quaternion that aligns lidar's forward vector (X axis) with direction vector
    # Chrono's forward axis for lidar is X axis
    forward = chrono.ChVectorD(1, 0, 0)
    rot = chrono.Q_from_AngAxis(0, chrono.VECT_Z)  # default identity rotation

    # Compute rotation quaternion from forward to direction
    # Use chrono's function to get quaternion between two vectors
    def quat_from_two_vectors(v1, v2):
        v1 = v1.GetNormalized()
        v2 = v2.GetNormalized()
        c = v1.Cross(v2)
        d = v1.Dot(v2)
        if d < -0.999999:
            # 180 degree rotation around any orthogonal vector
            ortho = chrono.VECT_Y
            if abs(v1.Dot(ortho)) > 0.999:
                ortho = chrono.VECT_Z
            axis = v1.Cross(ortho)
            axis.Normalize()
            return chrono.Q_from_AngAxis(math.pi, axis)
        s = math.sqrt((1 + d) * 2)
        invs = 1 / s
        return chrono.ChQuaternionD(s * 0.5, c.x * invs, c.y * invs, c.z * invs)

    rot = quat_from_two_vectors(forward, direction)

    lidar.SetOffsetPose(chrono.ChFrameD(position, rot))

    # Advance simulation
    system.DoStepDynamics(step_size)
    sensor_manager.Update()

    # Access lidar buffer data and print summary info
    # The lidar buffer contains distance measurements in a 2D array (vertical x horizontal)
    lidar_buffer = lidar.GetLidarBuffer()
    # Print some info: e.g., min, max distance and number of points
    distances = lidar_buffer.GetDistances()
    if distances.size() > 0:
        min_dist = min(distances)
        max_dist = max(distances)
        print(f"Time {time:.2f}s: Lidar points = {distances.size()}, Min dist = {min_dist:.2f} m, Max dist = {max_dist:.2f} m")
    else:
        print(f"Time {time:.2f}s: Lidar buffer empty")

    time += step_size

print("Simulation complete.")
```
