```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Ensure data path is set (optional if environment variable is set)

# Create the physical system
system = chrono.ChSystemNSC()

# 2. Add the required physical systems and objects

# Create the ground body (fixed)
ground = chrono.ChBody()
ground.SetBodyFixed(True)
system.Add(ground)

# Create the pendulum body
pendulum = chrono.ChBody()
pendulum.SetMass(1.0)  # 1 kg mass
pendulum.SetInertiaXX(chrono.ChVectorD(0.01, 0.01, 0.01))  # Small inertia for a simple rod-like pendulum

# Set initial position of pendulum (pivot at origin, pendulum hanging down along -Y)
pendulum_length = 1.0  # 1 meter length pendulum
pendulum_pos = chrono.ChVectorD(0, -pendulum_length, 0)
pendulum.SetPos(pendulum_pos)

# Add a visualization shape (a simple cylinder to represent the pendulum rod)
cyl = chrono.ChCylinderShape()
cyl.GetCylinderGeometry().p1 = chrono.ChVectorD(0, 0, 0)
cyl.GetCylinderGeometry().p2 = chrono.ChVectorD(0, -pendulum_length, 0)
cyl.GetCylinderGeometry().rad = 0.05
pendulum.AddVisualShape(cyl)

system.Add(pendulum)

# 3. Connect pendulum to ground with a revolute joint at origin
joint = chrono.ChLinkLockRevolute()
joint.Initialize(pendulum, ground, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngAxis(chrono.CH_C_PI_2, chrono.VECT_X)))
system.AddLink(joint)

# 4. Setup Irrlicht visualization
application = chronoirr.ChIrrApp(system, "Simple Pendulum Simulation", chronoirr.dimension2du(800, 600))

application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(chronoirr.vector3df(2, 1, 2))

application.AssetBindAll()
application.AssetUpdateAll()

# 5. Simulation loop with periodic logging
time_step = 0.01
application.SetTimestep(time_step)

print("Time\tPosX\tPosY\tPosZ\tVelX\tVelY\tVelZ")
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()

    # Log pendulum position and velocity
    pos = pendulum.GetPos()
    vel = pendulum.GetPos_dt()
    print(f"{system.GetChTime():.3f}\t{pos.x:.4f}\t{pos.y:.4f}\t{pos.z:.4f}\t{vel.x:.4f}\t{vel.y:.4f}\t{vel.z:.4f}")

    application.DoStep()
    application.EndScene()
```
