Below is a **PyChrono** simulation script that creates an ARTcar vehicle on a rigid terrain, uses Irrlicht for visualization, sets up the vehicle and terrain with specified parameters, adds a custom texture, and incorporates an interactive driver. The simulation loop runs at 50 FPS, updating vehicle dynamics and visualization in real time.

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr
import os

# -----------------------------------------------------------------------------
# 1. Initialize PyChrono and set up the main simulation system
# -----------------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

# -----------------------------------------------------------------------------
# 2. Create the vehicle (ARTcar) with specific parameters
# -----------------------------------------------------------------------------
contact_method = chrono.ChContactMethod_NSC  # or ChContactMethod_SMC
system = chrono.ChSystemNSC() if contact_method == chrono.ChContactMethod_NSC else chrono.ChSystemSMC()

# Vehicle initial location and orientation
initLoc = chrono.ChVectorD(0, 0, 1.0)  # x, y, z
initYaw = chrono.CH_C_PI / 12           # 15 degrees

# Create the ARTcar vehicle
vehicle = veh.Artcar(system)
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysD(initLoc, chrono.Q_from_AngZ(initYaw)))
vehicle.SetPowertrainType(veh.PowertrainModelType_SHAFTS)
vehicle.SetDriveType(veh.DrivelineTypeWV_AWD)
vehicle.SetTireType(veh.TireModelType_RIGID)
vehicle.SetTireStepSize(1e-3)
vehicle.SetChassisVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetWheelVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.Initialize()

# -----------------------------------------------------------------------------
# 3. Create the rigid terrain with custom texture
# -----------------------------------------------------------------------------
terrain = veh.RigidTerrain(system)
patch = terrain.AddPatch(
    chrono.CSYSNORM,  # default orientation
    chrono.ChVectorD(0, 0, 0),  # center
    200,  # length (x)
    100,  # width (y)
    0.1   # thickness
)
patch.SetContactFrictionCoefficient(0.9)
patch.SetRestitutionCoefficient(0.01)
patch.SetMaterialSurface(chrono.material.CreateMaterial(contact_method))
# Add a custom texture (replace with your own texture file if desired)
patch.SetTexture(chrono.GetChronoDataFile("terrain/textures/grass.jpg"), 200, 100)
terrain.Initialize()

# -----------------------------------------------------------------------------
# 4. Create the Irrlicht visualization app
# -----------------------------------------------------------------------------
app = veh.ChWheeledVehicleIrrApp(vehicle, 'ARTcar on Rigid Terrain', chronoirr.dimension2du(1024,768))
app.SetSkyBox()
app.AddTypicalLights()
app.AddTypicalLogo()
app.SetChaseCamera(chrono.ChVectorD(0.0, 0.0, 1.75), 6.0, 0.5)
app.SetTimestep(0.01)

# -----------------------------------------------------------------------------
# 5. Create the interactive driver system
# -----------------------------------------------------------------------------
steering_time = 1.0   # time to go from 0 to max steering
throttle_time = 1.0   # time to go from 0 to max throttle
braking_time = 0.3    # time to go from 0 to max braking
driver = veh.ChIrrGuiDriver(app)
driver.SetSteeringDelta(chrono.CH_C_DEG_TO_RAD * 1.5)  # radians per keypress
driver.SetThrottleDelta(0.02)
driver.SetBrakingDelta(0.1)
driver.Initialize()

# -----------------------------------------------------------------------------
# 6. Simulation loop
# -----------------------------------------------------------------------------
step_size = 0.01
render_steps = int(1.0 / (step_size * 50))  # 50 FPS

while app.GetDevice().run():
    time = system.GetChTime()

    # Render scene
    if app.GetDevice().run():
        app.BeginScene()
        app.DrawAll()
        app.EndScene()

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Update modules
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    app.Synchronize("ARTcar Simulation", driver_inputs)

    # Advance simulation for one timestep
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    app.Advance(step_size)

    # Limit to 50 FPS
    chrono.ChRealtimeStepTimer().Spin(step_size)

    # Optionally, break after some time
    # if time > 60:
    #     break
```

**Notes:**
- Make sure you have the ARTcar vehicle files in your Chrono data path. If not, replace `veh.Artcar` with another vehicle class available in your installation.
- The texture file path can be changed to any image you prefer.
- The simulation loop is set for real-time at 50 FPS.
- The interactive driver uses keyboard controls (arrows for steering/throttle/brake).
