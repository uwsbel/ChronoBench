```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import matplotlib.pyplot as plt
import numpy as np

# ------------------- 1. Chrono System Setup -------------------
sys = chrono.ChSystemNSC()

# Parameters
crank_center = chrono.ChVectorD(-1, 0.5, 0)  # Correct: ChVectorD, not ChVector3d
crank_rad = 0.4
crank_thick = 0.1
rod_length = 1.5

# Floor (truss)
mfloor = chrono.ChBodyEasyBox(3, 1, 3, 1000)
mfloor.SetPos(chrono.ChVectorD(0, -0.5, 0))
mfloor.SetFixed(True)
sys.Add(mfloor)

# Crank
mcrank = chrono.ChBodyEasyCylinder(chrono.ChAxis_Z, crank_rad, crank_thick, 1000)  # Correct: ChAxis_Z for Z-axis
mcrank.SetPos(crank_center + chrono.ChVectorD(0, 0, 0))  # Place at crank_center
# No need to rotate if created along Z
sys.Add(mcrank)

# Connecting rod
mrod = chrono.ChBodyEasyBox(rod_length, 0.1, 0.1, 1000)
mrod.SetPos(crank_center + chrono.ChVectorD(crank_rad + rod_length / 2, 0, 0))
sys.Add(mrod)

# Piston
mpiston = chrono.ChBodyEasyCylinder(chrono.ChAxis_X, 0.2, 0.3, 1000)  # Correct: ChAxis_X for X-axis
mpiston.SetPos(crank_center + chrono.ChVectorD(crank_rad + rod_length, 0, 0))
# No need to rotate if created along X
sys.Add(mpiston)

# ------------------- 2. Joints and Motor -------------------
# Motor: rotates crank about Z at crank_center
my_motor = chrono.ChLinkMotorRotationSpeed()
my_motor.Initialize(mcrank, mfloor, chrono.ChFrameD(crank_center, chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 0, 1))))
my_angularspeed = chrono.ChFunction_Const(np.pi)
my_motor.SetSpeedFunction(my_angularspeed)
sys.Add(my_motor)

# Revolute joint: crank to rod (at crank pin)
crank_pin = crank_center + chrono.ChVectorD(crank_rad, 0, 0)
mjointA = chrono.ChLinkLockRevolute()
mjointA.Initialize(mrod, mcrank, chrono.ChCoordsysD(crank_pin, chrono.QUNIT))
sys.Add(mjointA)

# Revolute joint: rod to piston (at piston pin)
piston_pin = crank_center + chrono.ChVectorD(crank_rad + rod_length, 0, 0)
mjointB = chrono.ChLinkLockRevolute()
mjointB.Initialize(mpiston, mrod, chrono.ChCoordsysD(piston_pin, chrono.QUNIT))
sys.Add(mjointB)

# Prismatic joint: piston to floor (allow X translation only)
# The prismatic axis must be X, so rotate Z to X
prism_rot = chrono.Q_from_AngAxis(-np.pi/2, chrono.ChVectorD(0,1,0))  # Rotates Z to X
mjointC = chrono.ChLinkLockPrismatic()
mjointC.Initialize(mpiston, mfloor, chrono.ChCoordsysD(piston_pin, prism_rot))
sys.Add(mjointC)

# ------------------- 3. Visualization -------------------
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Crank demo')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(1, 1, 3), chrono.ChVectorD(0, 1, 0))
vis.AddTypicalLights()

# ------------------- 4. Data Arrays for Plotting -------------------
array_time = []
array_angle = []
array_pos = []
array_speed = []

# ------------------- 5. Simulation Loop with Data Collection -------------------
t_end = 20.0  # seconds
timestep = 1e-3
while vis.Run():
    t = sys.GetChTime()
    # Data collection
    # Crank angle: get relative rotation of crank w.r.t. ground (mfloor)
    # The motor keeps the crank at a constant angular speed, so angle = omega * t
    angle = np.pi * t  # since speed is pi rad/s

    # Piston position and speed along X
    piston_pos = mpiston.GetPos().x
    piston_speed = mpiston.GetPos_dt().x

    array_time.append(t)
    array_angle.append(angle)
    array_pos.append(piston_pos)
    array_speed.append(piston_speed)

    # Visualization and step
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(timestep)

    # Stop after t_end seconds
    if t >= t_end:
        break

# ------------------- 6. Matplotlib Plotting -------------------
array_angle = np.array(array_angle)
array_pos = np.array(array_pos)
array_speed = np.array(array_speed)

fig, axs = plt.subplots(2, 1, figsize=(8, 8), sharex=True)

# X-axis ticks at multiples of pi
xticks = np.arange(0, 2*np.pi*(t_end*np.pi//(2*np.pi)+1), np.pi/2)
xtick_labels = [r"$0$"] + [r"${}\pi/2$".format(int(i/2)) if i%2 else r"${}\pi$".format(i//2) for i in range(1, len(xticks))]
xtick_labels = []
for i, val in enumerate(xticks):
    if np.isclose(val, 0):
        xtick_labels.append(r"$0$")
    elif np.isclose(val % (2*np.pi), 0):
        n = int(val // (np.pi))
        xtick_labels.append(r"${}\pi$".format(n))
    elif np.isclose(val % np.pi, 0):
        n = int(val // np.pi)
        xtick_labels.append(r"${}\pi$".format(n))
    else:
        n = int(val // np.pi)
        xtick_labels.append(r"${}\pi/2$".format(n*2+1))

axs[0].plot(array_angle, array_pos, label="Piston position")
axs[0].set_ylabel("Position [m]")
axs[0].set_title("Piston Position vs Crank Angle")
axs[0].grid(True)

axs[1].plot(array_angle, array_speed, label="Piston speed", color='orange')
axs[1].set_xlabel("Crank Angle [rad]")
axs[1].set_ylabel("Speed [m/s]")
axs[1].set_title("Piston Speed vs Crank Angle")
axs[1].grid(True)

axs[1].set_xticks(xticks)
axs[1].set_xticklabels([r"$0$", r"$\frac{\pi}{2}$", r"$\pi$", r"$\frac{3\pi}{2}$", r"$2\pi$"])

plt.tight_layout()
plt.show()
```

## **Key Corrections and Explanations**

1. **Vector Types**:  
   - Use `ChVectorD` and `ChCoordsysD` (not `ChVector3d` or `ChFramed`).
2. **Cylinder Axis**:  
   - Use `ChAxis_Z` for crank (so no rotation needed), and `ChAxis_X` for piston.
3. **Motor Initialization**:  
   - Use `ChFrameD` for the frame, and `SetSpeedFunction` (not `SetMotorFunction`).
   - Use `ChFunction_Const` (not `ChFunctionConst`).
4. **Joint Initialization**:  
   - Use `ChCoordsysD` for revolute and prismatic joints.
   - For prismatic, rotate Z to X using `Q_from_AngAxis(-np.pi/2, ChVectorD(0,1,0))`.
5. **Data Collection**:  
   - Collect time, crank angle, piston position (X), and piston speed (X).
6. **Simulation Stop Condition**:  
   - Stop after 20 seconds.
7. **Matplotlib Plotting**:  
   - Two subplots: position vs angle, speed vs angle.
   - X-ticks at multiples of π.
