Below is a **PyChrono** simulation script that sets up a Curiosity rover navigating on a rigid terrain, with a fixed ground (including collision), a rover model, motor control for steering, and visualization using Irrlicht. The script includes camera, lighting, shadows, textures, and a logo.

**Requirements:**  
- PyChrono (with vehicle and Irrlicht modules)
- A Curiosity rover model (provided by PyChrono)
- Textures and logo images (from PyChrono data)

---

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr
import os

# -----------------------------------------------------------------------------
# 1. Initialize PyChrono and set data paths
# -----------------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(veh.GetDataPath())

# -----------------------------------------------------------------------------
# 2. Create the Chrono physical system
# -----------------------------------------------------------------------------
system = chrono.ChSystemNSC()

# -----------------------------------------------------------------------------
# 3. Create the rigid terrain (fixed ground with collision)
# -----------------------------------------------------------------------------
terrain = veh.RigidTerrain(system)
patch = terrain.AddPatch(
    chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT),
    chrono.ChVectorD(0, 0, 0),
    100, 100
)
patch.SetContactFrictionCoefficient(0.9)
patch.SetContactRestitutionCoefficient(0.01)
patch.SetContactMaterialProperties(2e7, 0.3)
patch.SetTexture(veh.GetDataFile("terrain/textures/grass.jpg"), 200, 200)
patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
patch.SetLayerThickness(0.02)
patch.SetCollisionEnabled(True)
terrain.Initialize()

# -----------------------------------------------------------------------------
# 4. Create the Curiosity rover model
# -----------------------------------------------------------------------------
init_pos = chrono.ChVectorD(0, 0, 0.5)
init_rot = chrono.Q_from_AngZ(chrono.CH_C_DEG_TO_RAD * 0)
rover = veh.Curiosity(system)
rover.Initialize(chrono.ChCoordsysD(init_pos, init_rot))

# -----------------------------------------------------------------------------
# 5. Create a motor control driver for real-time steering
# -----------------------------------------------------------------------------
driver = veh.ChDriverInputs()
driver.steering = 0
driver.throttle = 0.5
driver.braking = 0

# Optionally, use a custom driver for real-time keyboard input
class MyDriver(veh.ChDriver):
    def __init__(self):
        super().__init__()
        self.steering = 0
        self.throttle = 0.5
        self.braking = 0

    def Synchronize(self, time):
        # Simple keyboard control (requires Irrlicht event receiver)
        pass

    def GetInputs(self):
        return veh.ChDriverInputs(self.steering, self.throttle, self.braking)

my_driver = MyDriver()

# -----------------------------------------------------------------------------
# 6. Set up Irrlicht visualization
# -----------------------------------------------------------------------------
vis = veh.ChWheeledVehicleIrrApp(rover.GetVehicle(), 'Curiosity Rover on Rigid Terrain', chronoirr.dimension2du(1280, 720))
vis.SetSkyBox()
vis.AddTypicalLights(chrono.ChVectorD(30, 100, 100), chrono.ChVectorD(-30, -80, 100), 250, 130)
vis.AddTypicalLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.SetChaseCamera(chrono.ChVectorD(0.0, 0.0, 1.2), 6.0, 0.5)
vis.SetChaseCameraPosition(chrono.ChVectorD(-8, 0, 2))
vis.SetChaseCameraMultipliers(0.5, 2.0)
vis.EnableShadows(True)
vis.SetSymbolscale(1.0)
vis.SetContactsDrawMode(veh.IrrContactsDrawMode.CONTACT_DISTANCES)
vis.SetPOV(chrono.ChVectorD(0, 0, 2), chrono.ChVectorD(0, 0, 0))

# Add terrain and rover to visualization
vis.AssetBindAll()
vis.AssetUpdateAll()

# -----------------------------------------------------------------------------
# 7. Simulation loop
# -----------------------------------------------------------------------------
step_size = 1e-3
render_step_size = 1.0 / 50  # 50 FPS
realtime_timer = chrono.ChRealtimeStepTimer()

while vis.Run():
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()

    # Get driver inputs (could be replaced with real-time keyboard input)
    rover.Synchronize(system.GetChTime(), my_driver.GetInputs(), terrain)
    terrain.Synchronize(system.GetChTime())
    vis.Synchronize('Curiosity Rover on Rigid Terrain', my_driver.GetInputs())

    # Advance simulation
    system.DoStepDynamics(step_size)
    realtime_timer.Spin(step_size)

# -----------------------------------------------------------------------------
# End of script
# -----------------------------------------------------------------------------
```
