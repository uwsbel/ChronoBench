class ChFunctionMyFun(chrono.ChFunction):
    def __init__(self, A1, A2, T1, T2, T3, w):
        super(ChFunctionMyFun, self).__init__()
        self.A1 = A1  # First amplitude
        self.A2 = A2  # Second amplitude
        self.T1 = T1  # First time threshold
        self.T2 = T2  # Second time threshold
        self.T3 = T3  # Third time threshold
        self.w = w    # Angular frequency
        
    def GetVal(self, x, x_dt=None):
        # Piecewise function based on time x
        if x <= self.T1:
            return self.A1 * m.sin(self.w * x)
        elif x <= self.T2:
            return self.A2
        elif x <= self.T3:
            return self.A2 * (1 - (x - self.T2) / (self.T3 - self.T2))
        else:
            return 0.0
            
    def Clone(self):
        return ChFunctionMyFun(self.A1, self.A2, self.T1, self.T2, self.T3, self.w)
