```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Function to generate particles with random properties
def emit_particle(sys):
    # Random position in a circular area
    angle = np.random.uniform(0, 2*np.pi)
    radius = np.random.uniform(0, 2)
    x = radius * np.cos(angle)
    z = radius * np.sin(angle)
    y = 5  # Emit particles from above
    pos = chrono.ChVector3d(x, y, z)
    
    # Random velocity
    speed = np.random.uniform(0.1, 0.5)
    vel_dir = chrono.ChVector3d(np.random.uniform(-1,1), np.random.uniform(-1,1), np.random.uniform(-1,1))
    vel_dir.Normalize()
    vel = vel_dir * speed
    
    # Random shape: sphere or box
    if np.random.rand() > 0.5:
        radius = np.random.uniform(0.1, 0.3)
        body = chrono.ChBodyEasySphere(radius, 1000, True, True)
    else:
        size = chrono.ChVector3d(np.random.uniform(0.2, 0.5), np.random.uniform(0.2, 0.5), np.random.uniform(0.2, 0.5))
        body = chrono.ChBodyEasyBox(size.x, size.y, size.z, 1000, True, True)
    
    # Set position, velocity, and orientation
    body.SetPos(pos)
    body.SetPos_dt(vel)
    angle_rot = np.random.uniform(0, 2*np.pi)
    axis = chrono.ChVector3d(np.random.rand(), np.random.rand(), np.random.rand())
    axis.Normalize()
    quat = chrono.ChQuaterniond()
    quat.Q_from_AngAxis(angle_rot, axis)
    body.SetRot(quat)
    
    # Assign random color
    color = chrono.ChColor(np.random.rand(), np.random.rand(), np.random.rand())
    body.GetVisualShape(0).SetColor(color)
    
    # Add to the system
    sys.Add(body)

# Initialize the physical system
sys = chrono.ChSystemNSC()
sys.Set_G_value(chrono.ChVector3d(0, 0, 0))  # Disable default gravity
sys.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# Set default contact material
material = chrono.ChContactMaterialNSC()
material.SetFriction(0.2)
sys.SetDefaultMaterial(material)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Particle Gravity Simulation")
vis.Initialize()
vis.AddCamera(chrono.ChVector3d(0, 3, 5))
vis.AddTypicalLights()

# Simulation loop
step = 0
while vis.Run():
    # Emit new particles every 50 steps
    if step % 50 == 0:
        emit_particle(sys)
    
    # Apply custom gravitational forces between all particle pairs
    G = 1e-3  # Gravitational constant (scaled for visual effect)
    for body in sys.Get_bodylist():
        total_force = chrono.ChVector3d(0, 0, 0)
        for other in sys.Get_bodylist():
            if body == other:
                continue
            r_vec = other.GetPos() - body.GetPos()
            distance = r_vec.Length()
            if distance < 1e-3:  # Avoid division by zero
                continue
            direction = r_vec.GetNormalized()
            F = G * body.GetMass() * other.GetMass() / (distance ** 2)
            total_force += F * direction
        # Apply the computed force to the body
        body.Empty_forces()
        body.AddForce(total_force)
    
    # Advance the simulation
    sys.DoStepDynamics(0.01)
    
    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    step += 1
```

### Explanation

1. **Initialization**:
   - The system is initialized using `ChSystemNSC` for non-smooth contact handling.
   - Gravity is disabled to allow custom gravitational forces.

2. **Particle Emission**:
   - The `emit_particle` function creates particles with random shapes (spheres or boxes), positions, velocities, and orientations.
   - Particles are emitted from a circular area above the scene to allow natural descent.

3. **Custom Gravitational Force**:
   - In each simulation step, gravitational forces are calculated between all particle pairs using Newton's law of universal gravitation.
   - Forces are scaled with a custom gravitational constant `G` to ensure visible interactions.

4. **Visualization**:
   - The Irrlicht engine renders the simulation with a camera positioned to view the particle interactions.
   - Particles are colored randomly for visual distinction.

This script dynamically generates particles and simulates mutual gravitational attraction, providing a visually engaging demonstration of PyChrono's capabilities in handling complex, interactive physics scenarios.
