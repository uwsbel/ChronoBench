        super(MySpringForce, self).__init__()

        self.damping_coef = damping_coef

        distance = (point1 - point2).Length()

        relative_velocity = (body1.GetPos_dt() - body2.GetPos_dt()).Length()
        damping_force = self.damping_coef * relative_velocity

        # Apply the force
        force = (point1 - point2).Normalize() * (spring_force + damping_force)
        return force
