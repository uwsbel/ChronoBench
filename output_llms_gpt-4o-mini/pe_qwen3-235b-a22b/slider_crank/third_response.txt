To address the requirements and ensure the simulation runs correctly, we have made the following key modifications and corrections to the original PyChrono script:

### ✅ **Corrections Identified in the Original Script**

1. **Incorrect Function Class for Motor Speed**  
   The original script used `chrono.ChFunctionConst(...)`, which is not a valid class in PyChrono.  
   ✅ **Corrected to:** `chrono.ChFunction_Const(...)`.

2. **Joint Type Modifications**  
   - The crank-rod and rod-piston joints were originally revolute (hinge) joints.  
     ✅ **Changed to:** `chrono.ChLinkLockSpherical()` for ball-and-socket behavior.
   - The piston-floor joint was a prismatic joint.  
     ✅ **Changed to:** `chrono.ChLinkLockPlanar()` to allow motion and rotation in the X-Y plane.

3. **Frame Orientation for Planar Joint**  
   The original prismatic joint used `chrono.Q_ROTATE_Z_TO_X` to align the motion axis.  
   ✅ **Corrected to:** `chrono.QUNIT` (identity rotation) to align the planar joint's X-Y plane with the world X-Y plane.

### ✅ **Final Corrected and Modified Script**

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import matplotlib.pyplot as plt
import numpy as np

# Set the Chrono data path (uncomment and set if needed)
# chrono.SetChronoDataPath('path/to/data')

# Create the physical system
sys = chrono.ChSystemNSC()

# Shared data
crank_center = chrono.ChVector3d(-1, 0.5, 0)
crank_rad = 0.4
crank_thick = 0.1
rod_length = 1.5

# Create the floor truss
mfloor = chrono.ChBodyEasyBox(3, 1, 3, 1000)
mfloor.SetPos(chrono.ChVector3d(0, -0.5, 0))
mfloor.SetFixed(True)
sys.Add(mfloor)

# Create the crank (flywheel)
mcrank = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, crank_rad, crank_thick, 1000)
mcrank.SetPos(crank_center + chrono.ChVector3d(0, 0, -0.1))
mcrank.SetRot(chrono.Q_ROTATE_Y_TO_Z)  # Rotate cylinder to align with Z-axis
sys.Add(mcrank)

# Create the rod
mrod = chrono.ChBodyEasyBox(rod_length, 0.1, 0.1, 1000)
mrod.SetPos(crank_center + chrono.ChVector3d(crank_rad + rod_length / 2, 0, 0))
sys.Add(mrod)

# Create the piston
mpiston = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, 0.2, 0.3, 1000)
mpiston.SetPos(crank_center + chrono.ChVector3d(crank_rad + rod_length, 0, 0))
mpiston.SetRot(chrono.Q_ROTATE_Y_TO_X)
sys.Add(mpiston)

# Create crank-truss motor
my_motor = chrono.ChLinkMotorRotationSpeed()
my_motor.Initialize(mcrank, mfloor, chrono.ChFramed(crank_center))
my_angularspeed = chrono.ChFunction_Const(chrono.CH_PI)  # 180°/s
my_motor.SetMotorFunction(my_angularspeed)
sys.Add(my_motor)

# Create crank-rod spherical joint
mjointA = chrono.ChLinkLockSpherical()
mjointA.Initialize(mrod, mcrank, chrono.ChFramed(crank_center + chrono.ChVector3d(crank_rad, 0, 0)))
sys.Add(mjointA)

# Create rod-piston spherical joint
mjointB = chrono.ChLinkLockSpherical()
mjointB.Initialize(mpiston, mrod, chrono.ChFramed(crank_center + chrono.ChVector3d(crank_rad + rod_length, 0, 0)))
sys.Add(mjointB)

# Create piston-floor planar joint
mjointC = chrono.ChLinkLockPlanar()
mjointC.Initialize(mpiston, mfloor, chrono.ChFramed(
    crank_center + chrono.ChVector3d(crank_rad + rod_length, 0, 0),
    chrono.QUNIT  # Identity rotation to align with X-Y plane
))
sys.Add(mjointC)

# Set up visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Crank Mechanism with Spherical and Planar Joints')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(1, 1, 3), chrono.ChVector3d(0, 1, 0))
vis.AddTypicalLights()

# Data collection for plotting
array_time = []
array_angle = []
array_pos = []
array_speed = []

# Simulation loop
while vis.Run():
    # Record data
    array_time.append(sys.GetChTime())
    array_angle.append(my_motor.GetMotorAngle())
    array_pos.append(mpiston.GetPos().x)
    array_speed.append(mpiston.GetPosDt().x)

    # Render and step simulation
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(1e-3)

    # Stop after 20 seconds
    if sys.GetChTime() > 20:
        vis.GetDevice().closeDevice()

# Plot results
fig, (ax1, ax2) = plt.subplots(2, sharex=True)

ax1.plot(array_angle, array_pos)
ax1.set(ylabel='Position [m]')
ax1.grid()

ax2.plot(array_angle, array_speed, 'r--')
ax2.set(ylabel='Speed [m/s]', xlabel='Angle [rad]')
ax2.grid()

# Format x-axis with pi ticks
plt.xticks(np.linspace(0, 2 * np.pi, 5), ['0', r'$\pi/2$', r'$\pi$', r'$3\pi/2$', r'$2\pi$'])

plt.show()
```
