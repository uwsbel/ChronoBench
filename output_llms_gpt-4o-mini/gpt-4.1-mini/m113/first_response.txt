```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Ensure data path is set

# Create the physical system (ChSystem)
system = chrono.ChSystemNSC()

# 2. Add the required physical systems and objects as specified

# Create the M113 vehicle system
m113 = veh.M113_Vehicle()
m113.SetContactMethod(chrono.ChContactMethod_NSC)
m113.SetChassisFixed(False)
m113.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 1.0), chrono.QUNIT))  # Initial position (x=0,y=0,z=1m)
m113.SetTireType(veh.M113_Tire.TireType_RIGID)
m113.Initialize(system)

# Create the terrain (rigid terrain)
terrain = veh.RigidTerrain(system)
patch = terrain.AddPatch(chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 0, 1), 100, 100)
patch.SetFriction(0.9)
patch.SetRestitution(0.01)
patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
terrain.Initialize()

# 3. Initialize a driver system for vehicle control
driver = veh.ChIrrGuiDriver(m113.GetVehicle())
driver.Initialize()

# Set driver input defaults (e.g., no throttle, no steering, no braking)
driver.SetThrottle(0)
driver.SetSteering(0)
driver.SetBraking(0)

# 4. Create a real-time visualization using Irrlicht with camera settings and lighting
application = veh.ChIrrApp(m113.GetVehicle(), "M113 Vehicle Simulation", chronoirr.dimension2du(1280, 720))
application.AddTypicalLights()
application.AddTypicalSky()
application.AddTypicalCamera(chronoirr.vector3df(0, -10, 3))
application.SetTimestep(0.01)
application.AssetBindAll()
application.AssetUpdateAll()

# Simulation parameters
timestep = 0.01
application.SetTimestep(timestep)

# 5. Implement a simulation loop that synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.

while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Synchronize modules
    time = system.GetChTime()
    driver.Synchronize(time)
    terrain.Synchronize(time)
    m113.Synchronize(time, driver_inputs, terrain)
    application.Synchronize(time, driver_inputs)

    # Advance simulation by timestep
    driver.Advance(timestep)
    terrain.Advance(timestep)
    m113.Advance(timestep)
    application.Advance(timestep)

    application.EndScene()
```
