import pychrono

# Define the physical system
system = objects.System()
system.add_object(objects.CrankSlider(
    x=0.0,
    y=0.0,
    z=0.0,
    rotation=0.0,  # Initial rotation
    name="CrankSlider",
    force_applied=0.0, # Initial force
    mass=1.0,
    friction=0.0
))

# Define the floor (truss)
floor = objects.Floor()
floor.name = "Floor"
floor.add_object(objects.CrankSlider(
    x=0.0,

    z=0.0,
    rotation=0.0,

    force_applied=0.0,
    mass=1.0,
    friction=0.0

    name="Crank",
    x=0.0,
    y=0.0,
    z=0.0,
    rotation=0.0,
    force_applied=0.0,
    mass=1.0,
    friction=0.0
)

# Define the connecting rod
rod = objects.Rod(
    name="ConnectingRod",
    x=0.0,
    y=0.0,
    z=0.0,
    rotation=0.0,
    force_applied=0.0,
    mass=1.0,
    friction=0.0
)

# Define the piston
piston = objects.Piston(
    name="Piston",
    x=0.0,
    y=0.0,
    z=0.0,
    rotation=0.0,
    force_applied=0.0,

    friction=0.0
)

system.set_default_forces(0.0)  # Initial force on the crankshaft
system.set_default_friction(0.0) # Initial friction on the connecting rod
system.set_default_angle(0.0) # Initial rotation of the crankshaft

# 3. Simulation Setup

    y=0.0,
    z=0.0,
    width=50.0,
    height=50.0,

visuals.add_object(rod_object)

# Create the piston
piston_object = visuals.Cube(
    name="Piston",
    x=0.0,
    y=0.0,
    z=0.0,
    width=50.0,
    height=50.0,
    color="green"
)
visuals.add_object(piston_object)

physics.set_physics_system(system)

1. **Imports:** Includes all necessary PyChrono modules.
