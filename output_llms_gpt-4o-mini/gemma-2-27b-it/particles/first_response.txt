```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import random

# 1. Initialize PyChrono environment and core components
my_system = chrono.ChSystemNSC()
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# Create Irrlicht visualization
myapplication = chronoirr.ChIrrApp(my_system, "Particle Attraction Simulation", chronoirr.dimension2du(800, 600))
myapplication.AddTypicalSky()
myapplication.AddTypicalCamera(chronoirr.vector3df(0, 5, -10))
myapplication.SetSymbolscale(0.1)
myapplication.SetShowInfos(True)

# 2. Add particle emitter
emitter_pos = chrono.ChVectorD(0, 5, 0)
emitter_rate = 10  # Particles per second

class ParticleEmitter:
    def __init__(self, system, position, rate):
        self.system = system
        self.position = position
        self.rate = rate
        self.timer = 0

    def Update(self, time_step):
        self.timer += time_step
        if self.timer >= 1.0 / self.rate:
            self.timer = 0
            self.EmitParticle()

    def EmitParticle(self):
        # Create particle with random shape, position, velocity, and orientation
        particle = chrono.ChBodyEasySphere(0.1, 1000, True, True)
        particle.SetPos(self.position + chrono.ChVectorD(random.uniform(-0.5, 0.5), random.uniform(-0.5, 0.5), random.uniform(-0.5, 0.5)))
        particle.SetPos_dt(chrono.ChVectorD(random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1)))
        particle.SetRot(chrono.ChQuaternionD(random.uniform(0, 1), random.uniform(0, 1), random.uniform(0, 1), random.uniform(0, 1)).Normalize())

        # Add particle to the system
        self.system.Add(particle)

emitter = ParticleEmitter(my_system, emitter_pos, emitter_rate)

# 3. Set gravitational forces
class GravitationalForce:
    def __init__(self, system, strength):
        self.system = system
        self.strength = strength

    def Update(self, time_step):
        for i in range(self.system.Get_bodylist().size()):
            for j in range(i + 1, self.system.Get_bodylist().size()):
                body1 = self.system.Get_bodylist()[i]
                body2 = self.system.Get_bodylist()[j]
                dist = (body1.GetPos() - body2.GetPos()).Length()
                force_dir = (body2.GetPos() - body1.GetPos()).Normalize()
                force_mag = self.strength / (dist * dist)
                force = force_dir * force_mag
                body1.AddForce(force)
                body2.AddForce(-force)

gravity_force = GravitationalForce(my_system, 0.1)

# Simulation loop
while myapplication.GetDevice().run():
    time_step = myapplication.GetSystem().GetTimerStep()
    emitter.Update(time_step)
    gravity_force.Update(time_step)

    myapplication.DoStep()
    myapplication.DrawAll()
    myapplication.SwapBuffers()

```

**Explanation:**

1. **Initialization:**
   - Creates a `ChSystemNSC` object for the physics simulation.
   - Sets gravity using `Set_G_acc`.
   - Initializes the Irrlicht visualization using `ChIrrApp`.

2. **Particle Emitter:**
   - Defines a `ParticleEmitter` class to generate particles with random properties.
   - The `EmitParticle` method creates a sphere (`ChBodyEasySphere`) with random position, velocity, and orientation within a specified range.

3. **Gravitational Force:**
   - Defines a `GravitationalForce` class to apply custom gravitational attraction between particles.
   - The `Update` method iterates through all particle pairs and calculates the force based on their distance and a predefined strength.

4. **Simulation Loop:**
   - Runs the Irrlicht application loop.
   - Updates the particle emitter and gravitational force every simulation step.
   - Performs the physics simulation step using `myapplication.DoStep()`.
   - Renders the scene using `myapplication.DrawAll()` and swaps buffers for display.

**Notes:**



* You can adjust the `emitter_rate` and `gravity_force.strength` parameters to control the particle generation and attraction strength.
* The particle shapes are currently spheres, but you can experiment with other shapes by changing the `ChBodyEasySphere` constructor.
* This is a basic example, and you can extend it by adding more complex particle interactions, collision detection, and other features.
