```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# 1. Initialize PyChrono environment and core components
my_system = chrono.ChSystemNSC()
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity
my_system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)

# 2. Add physical systems and objects
# 2.1. Terrain
terrain = chrono.ChBodyEasyBox(
    200, 200, 1, 1000, True, True, chrono.ChMaterialSurfaceNSC.Default
)
terrain.SetPos(chrono.ChVectorD(0, -1, 0))
terrain.SetBodyFixed(True)
my_system.Add(terrain)

# 2.2. M113 Vehicle
vehicle = chrono.ChWheeledVehicle()
vehicle.SetSystem(my_system)

# Load M113 chassis and wheel geometries (replace with your actual file paths)
chassis = chrono.ChBodyEasyBox(
    5, 2, 1, 1000, True, True, chrono.ChMaterialSurfaceNSC.Default
)
vehicle.AddChassis(chassis)

# Add wheels (replace with your actual wheel parameters)
wheel_radius = 0.5
wheel_width = 0.2
for i in range(6):
    wheel = chrono.ChWheel(
        chrono.ChQuaternionD(1, 0, 0, 0),
        chrono.ChVectorD(0, 0, 0),
        wheel_radius,
        wheel_width,
        20,
        chrono.ChMaterialSurfaceNSC.Default,
    )
    vehicle.AddWheel(wheel)

# 2.3. Driver System (placeholder - replace with your desired driver implementation)
driver = chrono.ChDriver()
vehicle.SetDriver(driver)

# 3. Set default parameters
# Terrain friction and restitution
terrain.GetMaterialSurfaceNSC().SetFriction(0.8)
terrain.GetMaterialSurfaceNSC().SetRestitution(0.2)

# Initial vehicle position and orientation
vehicle.SetPos(chrono.ChVectorD(0, 1, 0))
vehicle.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))

# 4. Real-time visualization using Irrlicht
vis = chronoirr.ChIrrApp(my_system, "M113 Simulation", chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()
vis.AddSkyBox()

# Camera settings
vis.SetCameraPosition(chrono.ChVectorD(10, 5, 5))
vis.SetCameraLookAt(chrono.ChVectorD(0, 1, 0))

# 5. Simulation loop
while vis.Run():
    # Driver control (implement your desired logic here)
    driver.SetSteeringAngle(0)  # Example: keep steering straight
    driver.SetThrottle(0.5)  # Example: apply constant throttle

    # Advance simulation
    my_system.DoStepDynamics(0.01)  # Adjust timestep as needed

    # Update visualization
    vis.BeginScene()
    vis.Render()
    vis.DoStep()
    vis.EndScene()
```
