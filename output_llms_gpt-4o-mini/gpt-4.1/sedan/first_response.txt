Below is a **PyChrono** simulation script that sets up a BMW E90 Sedan driving on rigid terrain, with TMeasy tires, interactive driver controls, and full Irrlicht visualization (chase camera, lighting, skybox, and terrain textures/logos). This script assumes you have **PyChrono** and its vehicle module installed.

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import os

# -----------------------------------------------------------------------------
# 1. Initialize PyChrono and set up the simulation
# -----------------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(veh.GetDataPath())

# -----------------------------------------------------------------------------
# 2. Create the physical system and terrain
# -----------------------------------------------------------------------------
system = chrono.ChSystemNSC()

# Terrain dimensions and properties
terrain_length = 200.0
terrain_width = 100.0
terrain_height = 2.0

# Create the rigid terrain
terrain = veh.RigidTerrain(system)
patch = terrain.AddPatch(
    chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT),
    terrain_length, terrain_width, terrain_height
)
patch.SetContactFrictionCoefficient(0.9)
patch.SetRestitutionCoefficient(0.01)
patch.SetMaterialSurface(veh.CreateMaterial(chrono.ChContactMethod_NSC))

# Set a texture and a logo
patch.SetTexture(chrono.GetChronoDataFile("vehicle/terrain/textures/grass.jpg"), 200, 100)
patch.SetTexture(chrono.GetChronoDataFile("vehicle/terrain/textures/logo_chronoengine_alpha.png"), 10, 10)
terrain.Initialize()

# -----------------------------------------------------------------------------
# 3. Create the BMW E90 Sedan vehicle
# -----------------------------------------------------------------------------
initLoc = chrono.ChVectorD(-90, 0, 0.2)
initRot = chrono.Q_from_AngZ(0)

vehicle = veh.BmwE90Sedan(system)
vehicle.SetContactMethod(chrono.ChContactMethod_NSC)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysD(initLoc, initRot))
vehicle.SetChassisVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetWheelVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetTireType(veh.TireModelType_TMEASY)
vehicle.SetTireStepSize(1e-3)
vehicle.SetAerodynamicDrag(0.3, 5.0, 1.2)  # Cd, area, air density
vehicle.Initialize()

# Enable collision for chassis and wheels
vehicle.GetChassis().SetCollide(True)
for axle in vehicle.GetAxles():
    for wheel in axle.GetWheels():
        wheel.SetCollide(True)

# -----------------------------------------------------------------------------
# 4. Create the interactive driver system
# -----------------------------------------------------------------------------
driver = veh.ChIrrGuiDriver(
    veh.GetDataFile("UI/driver/driver_inputs.json")
)
driver.SetSteeringDelta(0.04)
driver.SetThrottleDelta(0.02)
driver.SetBrakingDelta(0.1)
driver.Initialize()

# -----------------------------------------------------------------------------
# 5. Set up Irrlicht visualization
# -----------------------------------------------------------------------------
app = veh.ChWheeledVehicleIrrApp(
    vehicle, "BMW E90 Sedan Demo", irr.dimension2du(1280, 720)
)
app.SetSkyBox()
app.AddTypicalLights(chrono.ChVectorD(50, 100, 200), chrono.ChVectorD(-50, -100, 200), 250, 130)
app.AddLightDirectional(chrono.ChVectorD(0, 0, 1), chrono.ChVectorD(1, 1, -1), 0.8)
app.SetChaseCamera(chrono.ChVectorD(0.0, 0.0, 1.75), 6.0, 0.5)
app.SetTimestep(0.01)
app.AssetBindAll()
app.AssetUpdateAll()

# -----------------------------------------------------------------------------
# 6. Simulation loop
# -----------------------------------------------------------------------------
step_size = 0.01

while app.GetDevice().run():
    # Collect driver inputs
    driver_inputs = driver.GetInputs()

    # Update modules
    driver.Synchronize(app.GetSimulationTime())
    terrain.Synchronize(app.GetSimulationTime())
    vehicle.Synchronize(app.GetSimulationTime(), driver_inputs, terrain)
    app.Synchronize("BMW E90 Sedan Demo", driver_inputs)

    # Advance simulation
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    app.Advance(step_size)

    # Render scene
    app.BeginScene()
    app.DrawAll()
    app.EndScene()
```

**NOTES:**
- **Interactive Controls:** Use the arrow keys and spacebar to control steering, throttle, and braking in the Irrlicht window.
- **Textures and Logos:** You can replace the texture/logo file paths with your own images if desired.
- **Visualization:** The chase camera, skybox, and lighting are all enabled for a realistic scene.
- **Tires:** The TMeasy tire model is used for all wheels.
- **Collision:** Both chassis and wheels have collision enabled.
