Certainly! Below is a **PyChrono** simulation script that follows your instructions. This script:

- Initializes the Chrono system with gravity.
- Adds a rigid ground body.
- Creates a Viper rover and its driver.
- Sets up Irrlicht visualization with camera and lighting.
- Implements a simulation loop where the rover's steering is gradually changed over a specified period.

**Note:**  
- Make sure you have the `pychrono` and `pychrono.vehicle` modules installed.
- Adjust the `viper_data_dir` path if needed.

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import math

# -----------------------------------------------------------------------------
# 1. Initialize PyChrono system and set gravity
# -----------------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
system = chrono.ChSystemSMC()
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

# -----------------------------------------------------------------------------
# 2. Add rigid terrain (ground body)
# -----------------------------------------------------------------------------
terrain = veh.RigidTerrain(system)
patch = terrain.AddPatch(
    chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT),
    length=10.0, width=10.0
)
patch.SetContactFrictionCoefficient(0.9)
patch.SetRestitutionCoefficient(0.01)
patch.SetMaterialSurface(veh.CreateMaterial(veh.ChContactMethod_SMC))
patch.SetColor(chrono.ChColor(0.4, 0.5, 0.4))
patch.SetTexture(chrono.GetChronoDataFile("terrain/textures/grass.jpg"), 10, 10)
terrain.Initialize()

# -----------------------------------------------------------------------------
# 3. Initialize Viper rover and driver
# -----------------------------------------------------------------------------
viper_data_dir = chrono.GetChronoDataFile("viper/")
rover = veh.Viper(system, viper_data_dir)
rover.Initialize(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0.1), chrono.QUNIT))

# Create a simple driver that allows us to set steering and throttle
driver = veh.ViperSimpleDriver()
driver.SetThrottle(0.5)  # Constant throttle

# -----------------------------------------------------------------------------
# 4. Set up Irrlicht visualization
# -----------------------------------------------------------------------------
vis = veh.ChWheeledVehicleIrrApp(rover.GetSystem(), "Viper Rover Simulation", irr.dimension2du(1024, 768))
vis.SetChaseCamera(chrono.ChVectorD(0, 0, 0.5), 2.0, 0.5)
vis.AddTypicalLights()
vis.AddSkyBox()
vis.SetSymbolscale(1.0)
vis.SetHUDLocation(500, 20)
vis.AssetBindAll()
vis.AssetUpdateAll()

# -----------------------------------------------------------------------------
# 5. Simulation loop with gradual steering change
# -----------------------------------------------------------------------------
timestep = 1e-3
end_time = 10.0
steer_time = 5.0  # Time over which steering changes from 0 to max
max_steering = 0.5  # Maximum steering input

time = 0.0
while vis.Run() and time < end_time:
    # Calculate steering input: ramp from 0 to max_steering over steer_time seconds
    if time < steer_time:
        steering = max_steering * (time / steer_time)
    else:
        steering = max_steering

    driver.SetSteering(steering)
    rover.Update(driver.GetInputs(), terrain)
    terrain.Synchronize(time)
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    system.DoStepDynamics(timestep)
    time += timestep

vis.GetDevice().closeDevice()
```

---

### **Key Points:**
- **Gravity:** Set to -9.81 m/s² in Z.
- **Ground:** Rigid terrain patch, friction and restitution set.
- **Viper Rover:** Initialized at (0, 0, 0.1).
- **Driver:** Constant throttle, steering ramps up over 5 seconds.
- **Visualization:** Irrlicht app with camera, lights, and skybox.
- **Simulation Loop:** Updates steering, rover, terrain, and renders each timestep.
