```python
import pychrono.core as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# 1. Initialize the Chrono engine and create the physical system
system = chrono.ChSystemNSC()

# Set gravity
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# 2. Create the terrain (rigid terrain with friction and restitution)
terrain = veh.RigidTerrain(system)
terrain.Initialize()

# Set terrain material properties
material = terrain.GetMaterialSurface()
material.SetFriction(0.8)        # Example friction coefficient
material.SetRestitution(0.1)     # Example restitution coefficient

# Create a ground plane
ground = terrain.GetGround()
ground.SetPosition(chrono.ChVectorD(0, 0, 0))
ground.SetMaterialSurface(material)

# 3. Create the Kraz vehicle
# Define initial position and orientation
initial_pos = chrono.ChVectorD(0, 0.5, 0)  # Slightly above ground
initial_rot = chrono.Q_from_AngleAxis(0, chrono.ChVectorD(0, 1, 0))

# Instantiate the vehicle
kraz_vehicle = veh.Kraz(vehicle::ChContactMethod::NSC)
kraz_vehicle.Initialize(initial_pos, initial_rot, system)

# Set vehicle initial conditions if needed
kraz_vehicle.GetChassis().SetPos(initial_pos)
kraz_vehicle.GetChassis().SetRot(initial_rot)

# 4. Set up the driver system for vehicle control
driver = veh.ChIrrGuiDriver()
driver.Initialize(system, kraz_vehicle.GetVehicle())

# 5. Set up the Irrlicht visualization
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Kraz Vehicle Simulation')
vis.Initialize()

# Configure camera
camera_pos = chrono.ChVectorD(0, 3, -10)
look_at = chrono.ChVectorD(0, 0.5, 0)
vis.GetActiveCamera().SetPosition(camera_pos)
vis.GetActiveCamera().SetTarget(look_at)

# Add lights
vis.AddLightWithShadow(chrono.ChVectorD(0, 10, -10),  # Light position
                       chrono.ChVectorD(0, 0, 0),     # Target
                       100,                         # Radius
                       0.5, 0.5, 0.5,               # Diffuse color
                       0.3, 0.3, 0.3,               # Specular color
                       0.2)                         # Shadow intensity

# 6. Simulation loop parameters
time_step = 1e-3
max_time = 20  # seconds
current_time = 0

# 7. Run the simulation
while vis.Run():
    # Synchronize driver inputs
    driver.Synchronize(current_time)
    driver.Advance(time_step)

    # Apply driver inputs to vehicle
    kraz_vehicle.GetVehicle().GetPowertrain().SetThrottle(driver.GetThrottle());
    kraz_vehicle.GetVehicle().GetSteering().SetSteering(driver.GetSteering());

    # Synchronize vehicle
    kraz_vehicle.Synchronize(current_time)

    # Advance the system
    system.DoStepDynamics(time_step)

    # Update visualization
    vis.Render()

    # Increment time
    current_time += time_step

    # Optional: add a real-time check or limit
    # time.sleep(time_step)  # Uncomment if needed for real-time pacing
```
