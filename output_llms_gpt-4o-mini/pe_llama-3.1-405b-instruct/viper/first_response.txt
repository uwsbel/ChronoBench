```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr

# Initialize the Chrono system with gravity and ground body
sys = chrono.ChSystemSMC()
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
sys.Add(ground)

# Create the Viper rover and its driver
rover = veh.Viper()
rover.SetContactMethod(chrono.ChContactMethod_SMC)
rover.SetChassisCollisionType(chrono.ChCollisionSystem.Type_BULLET)
rover.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0.5), chrono.QUNIT))
rover.Initialize()
sys.Add(rover)

# Create the driver
driver = veh.ChDriver()
rover.SetDriver(driver)

# Create a real-time visualization using Irrlicht
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle("Viper Rover Simulation")
vis.Initialize()
vis.AddCamera(chrono.ChVectorD(0, 0, 2), chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()

# Simulation loop
time_step = 0.01
time_end = 10.0
time = 0.0

while vis.Run():
    # Update the rover's steering
    steering = chrono.ChFunction_Sine(0.1, 0.5)
    rover.SetSteering(steering.Get_y(time))

    # Update and render the rover
    sys.DoStepDynamics(time_step)
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()
    time += time_step

    # Exit the simulation loop if the time exceeds the end time
    if time > time_end:
        break
```

This script creates a PyChrono simulation with a Viper rover on a rigid terrain, sets up the Chrono system with gravity and ground body, initializes the Viper rover and its driver, creates a real-time visualization using Irrlicht with camera settings and lighting, and implements a simulation loop where the rover's steering gradually changes over a specified time period, updating and rendering the rover at each timestep.
