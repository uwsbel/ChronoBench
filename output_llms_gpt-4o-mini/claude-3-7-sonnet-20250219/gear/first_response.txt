# PyChrono Epicyclic Gear System Simulation

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Initialize the simulation system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Set solver settings
system.SetSolverType(chrono.ChSolver.Type_PSOR)
system.SetSolverMaxIterations(100)
system.SetMaxPenetrationRecoverySpeed(1.0)
system.SetMinBounceSpeed(0.1)

# Create a fixed ground body
ground = chrono.ChBodyEasyBox(1, 0.1, 1, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -0.5, 0))
ground.SetBodyFixed(True)
system.Add(ground)

# Material for visualization
vis_material = chrono.ChVisualMaterial()
vis_material.SetAmbientColor(chrono.ChColor(0.5, 0.5, 0.5))
vis_material.SetDiffuseColor(chrono.ChColor(0.7, 0.7, 0.7))
vis_material.SetSpecularColor(chrono.ChColor(0.1, 0.1, 0.1))

# Create a fixed truss
truss = chrono.ChBody()
truss.SetPos(chrono.ChVectorD(0, 0, 0))
truss.SetBodyFixed(True)
system.Add(truss)

# Dimensions
sun_gear_radius = 0.1
planet_gear_radius = 0.05
bar_length = sun_gear_radius + planet_gear_radius + 0.01  # Distance from center to planet gear center

# Create the sun gear (center gear)
sun_gear = chrono.ChBodyEasyCylinder(sun_gear_radius, 0.03, 1000, True, True)
sun_gear.SetPos(chrono.ChVectorD(0, 0, 0))
sun_gear.GetVisualShape(0).SetColor(chrono.ChColor(0.9, 0.4, 0.1))
system.Add(sun_gear)

# Create a revolute joint between sun gear and truss
sun_gear_joint = chrono.ChLinkLockRevolute()
sun_gear_joint.Initialize(sun_gear, truss, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngZ(0)))
system.AddLink(sun_gear_joint)

# Create a motor for the sun gear
sun_gear_motor = chrono.ChLinkMotorRotationSpeed()
sun_gear_motor.Initialize(sun_gear, truss, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))
sun_gear_motor.SetSpeedFunction(chrono.ChFunction_Const(1.0))  # Constant speed of 1 rad/s
system.AddLink(sun_gear_motor)

# Create the rotating bar (carrier)
bar = chrono.ChBodyEasyBox(bar_length*2, 0.02, 0.02, 1000, True, True)
bar.SetPos(chrono.ChVectorD(0, 0, 0))
bar.GetVisualShape(0).SetColor(chrono.ChColor(0.1, 0.5, 0.9))
system.Add(bar)

# Create a revolute joint between bar and truss
bar_joint = chrono.ChLinkLockRevolute()
bar_joint.Initialize(bar, truss, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngZ(0)))
system.AddLink(bar_joint)

# Create the planet gear
planet_gear = chrono.ChBodyEasyCylinder(planet_gear_radius, 0.03, 1000, True, True)
planet_gear.SetPos(chrono.ChVectorD(bar_length, 0, 0))
planet_gear.GetVisualShape(0).SetColor(chrono.ChColor(0.1, 0.9, 0.3))
system.Add(planet_gear)

# Create a revolute joint between planet gear and bar
planet_gear_joint = chrono.ChLinkLockRevolute()
planet_gear_joint.Initialize(planet_gear, bar, chrono.ChCoordsysD(chrono.ChVectorD(bar_length, 0, 0), chrono.Q_from_AngZ(0)))
system.AddLink(planet_gear_joint)

# Create gear constraints between sun gear and planet gear
gear_constraint = chrono.ChLinkGear()
gear_constraint.Initialize(sun_gear, planet_gear, 
                          chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)),
                          chrono.ChFrameD(chrono.ChVectorD(bar_length, 0, 0)))
gear_constraint.SetTransmissionRatio(-sun_gear_radius / planet_gear_radius)
system.AddLink(gear_constraint)

# Create the visualization system
vis = chronoirr.ChIrrApp(system, "Epicyclic Gear System", chronoirr.dimension2du(1024, 768))
vis.AddTypicalLogo()
vis.AddTypicalSky()
vis.AddTypicalLights()
vis.AddTypicalCamera(chronoirr.vector3df(0, 0.5, -1), chronoirr.vector3df(0, 0, 0))
vis.SetSymbolScale(0.1)
vis.SetShowInfos(True)

# Initialize the visualization window
vis.AssetBindAll()
vis.AssetUpdateAll()

# Simulation time step
time_step = 0.001
vis.SetTimestep(time_step)

# Run the simulation
while vis.GetDevice().run():
    vis.BeginScene()
    vis.DrawAll()
    vis.DoStep()
    vis.EndScene()
```

This script creates a PyChrono simulation of an epicyclic gear system with:

1. A fixed ground and truss
2. A central sun gear driven by a constant speed motor
3. A rotating bar (carrier) connected to the fixed truss
4. A planet gear mounted on the carrier that meshes with the sun gear
5. Appropriate constraints and revolute joints
6. Irrlicht visualization with materials and lighting

The sun gear rotates at a constant speed of 1 rad/s, causing the planet gear to rotate as it orbits around the sun gear. The gear ratio is determined by the relative sizes of the sun and planet gears.
