import pychrono.core as chrono
import pychrono.fea as fea
import pychrono.irrlicht as chronoirr

# ----------------------------------------------------------------------------
# Model1: Multiple chains of ANCF cable elements, each with one end hinged to
# ground and the other end connected to a box, moving under gravity.
# This model demonstrates the use of the utility class ChBuilderCableANCF.
# ----------------------------------------------------------------------------

class Model1:
    def __init__(self, system, mesh, n_chains=6):
        self.system = system
        self.mesh = mesh
        self.n_chains = n_chains
        self.end_bodies = []  # Store the end bodies for position tracking
        
        # Define spacing between chains
        spacing = 0.2
        
        for i in range(n_chains):
            # Create a section with properties that vary slightly with each chain
            msection_cable = fea.ChBeamSectionCable()
            msection_cable.SetDiameter(0.015)  # 15 mm diameter
            msection_cable.SetYoungModulus((0.01 + i * 0.002) * 1e9)  # Increasing stiffness with each chain
            msection_cable.SetRayleighDamping(0.0001)
            
            # Create builder for ANCF cable elements
            builder = fea.ChBuilderCableANCF()
            
            # Calculate number of elements (increasing with each chain)
            n_elements = 10 + i * 2
            
            # Calculate start and end positions for this chain
            start_pos = chrono.ChVector3d(i * spacing, 0, -0.1)
            end_pos = chrono.ChVector3d(i * spacing + 0.5, 0, -0.1)
            
            # Build the beam
            builder.BuildBeam(
                mesh,
                msection_cable,
                n_elements,
                start_pos,
                end_pos
            )
            
            # Apply force to the front node
            front_node = builder.GetLastBeamNodes().front()
            front_node.SetForce(chrono.ChVector3d(0, -0.7 - i * 0.1, 0))
            
            # Create a truss body (fixed reference frame)
            mtruss = chrono.ChBody()
            mtruss.SetBodyFixed(True)  # Fix the truss body
            system.Add(mtruss)
            
            # Create a hinge constraint to fix beam's end point to the truss
            constraint_hinge = fea.ChLinkNodeFrame()
            constraint_hinge.Initialize(builder.GetLastBeamNodes().back(), mtruss)
            system.Add(constraint_hinge)
            
            # Create a box at the end of the beam
            end_box = chrono.ChBodyEasyBox(0.05, 0.05, 0.05, 1000)  # 5cm box with density 1000
            end_box.SetPos(chrono.ChVector3d(start_pos.x, start_pos.y - 0.05, start_pos.z))
            system.Add(end_box)
            self.end_bodies.append(end_box)  # Store for position tracking
            
            # Connect the box to the front node of the beam
            constraint_box = fea.ChLinkNodeBody()
            constraint_box.Initialize(front_node, end_box)
            system.Add(constraint_box)
    
    def PrintBodyPositions(self):
        """Print the positions of all end bodies"""
        print("End Body Positions:")
        for i, body in enumerate(self.end_bodies):
            pos = body.GetPos()
            print(f"Chain {i+1}: X={pos.x:.3f}, Y={pos.y:.3f}, Z={pos.z:.3f}")
        print("-------------------")

# Initialize the physical system and mesh container:
sys = chrono.ChSystemSMC()
mesh = fea.ChMesh()

# Create the model with 6 chains and add the mesh to the system
model = Model1(sys, mesh)
sys.Add(mesh)  # Add the mesh to the physical system

# Add visualization for the FEM mesh:
# Visualization for beam moments
visualizebeamA = chrono.ChVisualShapeFEA(mesh)
visualizebeamA.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_ELEM_BEAM_MZ)
visualizebeamA.SetColorscaleMinMax(-0.4, 0.4)
visualizebeamA.SetSmoothFaces(True)
visualizebeamA.SetWireframe(False)
mesh.AddVisualShapeFEA(visualizebeamA)

# Visualization for node positions
visualizebeamB = chrono.ChVisualShapeFEA(mesh)
visualizebeamB.SetFEMglyphType(chrono.ChVisualShapeFEA.GlyphType_NODE_DOT_POS)
visualizebeamB.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_NONE)
visualizebeamB.SetSymbolsThickness(0.006)
visualizebeamB.SetSymbolsScale(0.01)
visualizebeamB.SetZbufferHide(False)
mesh.AddVisualShapeFEA(visualizebeamB)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('FEA cables with multiple chains')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0.6, 0.6, -1.2))  # Adjusted to see all chains
vis.AddTypicalLights()

# Set solver type and settings
solver = chrono.ChSolverMINRES()
sys.SetSolver(solver)
solver.SetMaxIterations(200)
solver.SetTolerance(1e-10)
solver.EnableDiagonalPreconditioner(True)
solver.EnableWarmStart(True)
solver.SetVerbose(False)

# Set the timestepper for the simulation
ts = chrono.ChTimestepperEulerImplicitLinearized(sys)
sys.SetTimestepper(ts)

# Simulation counter for printing positions
step_counter = 0
print_interval = 20  # Print every 20 steps

# Simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(0.01)
    
    # Print body positions periodically
    step_counter += 1
    if step_counter % print_interval == 0:
        model.PrintBodyPositions()
