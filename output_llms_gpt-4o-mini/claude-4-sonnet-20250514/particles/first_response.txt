I'll create a PyChrono simulation that models gravitational attraction between particles with a particle emitter system. Here's the complete script:

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import random
import math

# Initialize the PyChrono environment
chrono.SetChronoDataPath('PYCHRONO_DATA_DIR')

# Create the physical system
system = chrono.ChSystemNSC()

# Set the global collision margins
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.005)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.003)

# Set gravitational acceleration (we'll override this with custom forces)
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))  # Disable default gravity

# Particle Emitter Class
class ParticleEmitter:
    def __init__(self, system, emission_rate=2.0):
        self.system = system
        self.emission_rate = emission_rate  # particles per second
        self.last_emission_time = 0
        self.particles = []
        self.max_particles = 50  # Limit number of particles for performance
        
        # Material for particles
        self.material = chrono.ChMaterialSurfaceNSC()
        self.material.SetFriction(0.3)
        self.material.SetRestitution(0.8)
    
    def create_particle(self):
        # Random shape selection
        shape_type = random.choice(['sphere', 'box', 'cylinder'])
        
        # Create particle body
        particle = chrono.ChBodyEasySphere(
            random.uniform(0.1, 0.3),  # Random radius
            1000,  # Density
            True,  # Enable collision
            True   # Enable visualization
        ) if shape_type == 'sphere' else (
            chrono.ChBodyEasyBox(
                random.uniform(0.1, 0.3), random.uniform(0.1, 0.3), random.uniform(0.1, 0.3),
                1000, True, True
            ) if shape_type == 'box' else
            chrono.ChBodyEasyCylinder(
                chrono.ChAxis_Y,
                random.uniform(0.1, 0.3), random.uniform(0.2, 0.4),
                1000, True, True
            )
        )
        
        # Set random position around emission area
        emission_radius = 5.0
        angle = random.uniform(0, 2 * math.pi)
        radius = random.uniform(0, emission_radius)
        
        pos_x = radius * math.cos(angle)
        pos_y = random.uniform(-2, 2)
        pos_z = radius * math.sin(angle)
        
        particle.SetPos(chrono.ChVectorD(pos_x, pos_y, pos_z))
        
        # Set random orientation
        particle.SetRot(chrono.Q_from_AngAxis(
            random.uniform(0, 2 * math.pi),
            chrono.ChVectorD(random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1)).GetNormalized()
        ))
        
        # Set random initial velocity
        particle.SetPos_dt(chrono.ChVectorD(
            random.uniform(-2, 2),
            random.uniform(-1, 1),
            random.uniform(-2, 2)
        ))
        
        # Set random angular velocity
        particle.SetWvel_par(chrono.ChVectorD(
            random.uniform(-5, 5),
            random.uniform(-5, 5),
            random.uniform(-5, 5)
        ))
        
        # Set material
        particle.GetMaterialSurface().SetFriction(self.material.GetSfriction())
        particle.GetMaterialSurface().SetRestitution(self.material.GetRestitution())
        
        # Set random color for visualization
        particle.GetVisualShape(0).SetColor(chrono.ChColor(
            random.uniform(0.2, 1.0),
            random.uniform(0.2, 1.0),
            random.uniform(0.2, 1.0)
        ))
        
        # Add particle to system
        self.system.Add(particle)
        self.particles.append(particle)
        
        return particle
    
    def update(self, current_time):
        # Emit particles at specified rate
        if current_time - self.last_emission_time >= 1.0 / self.emission_rate:
            if len(self.particles) < self.max_particles:
                self.create_particle()
                self.last_emission_time = current_time
    
    def remove_distant_particles(self):
        # Remove particles that are too far from origin
        max_distance = 20.0
        particles_to_remove = []
        
        for particle in self.particles[:]:
            pos = particle.GetPos()
            distance = math.sqrt(pos.x**2 + pos.y**2 + pos.z**2)
            
            if distance > max_distance:
                particles_to_remove.append(particle)
        
        for particle in particles_to_remove:
            self.system.Remove(particle)
            self.particles.remove(particle)

# Custom Force Class for Gravitational Attraction
class GravitationalForce(chrono.ChForce):
    def __init__(self, particle_system, gravitational_constant=10.0):
        super().__init__()
        self.particle_system = particle_system
        self.G = gravitational_constant
    
    def apply_gravitational_forces(self):
        particles = self.particle_system.particles
        
        # Clear previous forces
        for particle in particles:
            particle.Empty_forces_accumulators()
        
        # Calculate gravitational forces between all particle pairs
        for i in range(len(particles)):
            for j in range(i + 1, len(particles)):
                p1 = particles[i]
                p2 = particles[j]
                
                # Get positions
                pos1 = p1.GetPos()
                pos2 = p2.GetPos()
                
                # Calculate distance vector
                r_vec = pos2 - pos1
                r_magnitude = r_vec.Length()
                
                # Avoid division by zero and very small distances
                if r_magnitude < 0.1:
                    continue
                
                # Calculate gravitational force magnitude
                # F = G * m1 * m2 / r^2
                m1 = p1.GetMass()
                m2 = p2.GetMass()
                
                force_magnitude = self.G * m1 * m2 / (r_magnitude * r_magnitude)
                
                # Calculate unit vector
                r_unit = r_vec.GetNormalized()
                
                # Apply forces (Newton's third law)
                force_vec = r_unit * force_magnitude
                
                # Apply force to particle 1 (towards particle 2)
                p1.Accumulate_force(force_vec, pos1, False)
                
                # Apply equal and opposite force to particle 2
                p2.Accumulate_force(force_vec * -1, pos2, False)

# Create particle emitter
emitter = ParticleEmitter(system, emission_rate=1.5)

# Create gravitational force system
grav_force = GravitationalForce(emitter, gravitational_constant=15.0)

# Create some initial particles
for _ in range(5):
    emitter.create_particle()

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Gravitational Particle Attraction Simulation")
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 5, -15))
vis.AddTypicalLights()

# Simulation parameters
timestep = 0.01
simulation_time = 0

# Add a ground plane for reference
ground = chrono.ChBodyEasyBox(20, 0.2, 20, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -5, 0))
ground.SetBodyFixed(True)
ground.GetVisualShape(0).SetColor(chrono.ChColor(0.3, 0.3, 0.3))
system.Add(ground)

# Simulation loop
print("Starting gravitational particle simulation...")
print("Controls:")
print("- Mouse: Rotate view")
print("- Arrow keys: Move camera")
print("- Particles are emitted continuously and attract each other")

while vis.Run():
    # Update particle emitter
    emitter.update(simulation_time)
    
    # Remove distant particles to maintain performance
    if int(simulation_time * 10) % 50 == 0:  # Every 5 seconds
        emitter.remove_distant_particles()
    
    # Apply gravitational forces
    grav_force.apply_gravitational_forces()
    
    # Update visualization
    vis.BeginScene()
    vis.Render()
    
    # Add some text information
    vis.GetGUIEnvironment().getBuiltInFont().draw(
        f"Time: {simulation_time:.1f}s",
        chronoirr.rect(10, 10, 200, 30),
        chronoirr.SColor(255, 255, 255, 255)
    )
    
    vis.GetGUIEnvironment().getBuiltInFont().draw(
        f"Particles: {len(emitter.particles)}",
        chronoirr.rect(10, 35, 200, 55),
        chronoirr.SColor(255, 255, 255, 255)
    )
    
    vis.GetGUIEnvironment().getBuiltInFont().draw(
        "Gravitational Attraction Active",
        chronoirr.rect(10, 60, 300, 80),
        chronoirr.SColor(255, 255, 255, 0)
    )
    
    vis.EndScene()
    
    # Advance the simulation
    system.DoStepDynamics(timestep)
    simulation_time += timestep
    
    # Optional: Limit simulation time for demo purposes
    if simulation_time > 120:  # Run for 2 minutes
        break

print("Simulation completed!")
```

This simulation script creates a comprehensive gravitational particle system with the following features:

## Key Components:

1. **PyChrono Environment Setup**: 
   - Initializes the physics system with NSC (Non-Smooth Contact) solver
   - Disables default gravity to implement custom gravitational forces

2. **Particle Emitter Class**:
   - Generates particles with random shapes (spheres, boxes, cylinders)
   - Random positions, velocities, orientations, and colors
   - Manages particle lifecycle and removes distant particles

3. **Custom Gravitational Force System**:
   - Implements Newton's law of universal gravitation (F = G×m1×m2/r²)
   - Calculates forces between all particle pairs
   - Applies forces dynamically during simulation

4. **Irrlicht Visualization**:
   - Real-time 3D visualization with camera controls
   - Displays simulation statistics
   - Custom lighting and ground plane for reference

## Simulation Features:

- **Dynamic Particle Generation**: Continuous emission of particles with random properties
- **Realistic Physics**: Gravitational attraction causes particles to cluster and orbit
- **Performance Management**: Limits particle count and removes distant objects
- **Interactive Visualization**: Mouse and keyboard controls for camera movement
- **Visual Feedback**: Real-time display of particle count and simulation time

The simulation demonstrates complex emergent behavior as particles attract each other, form clusters, and create dynamic orbital patterns based on their masses and initial conditions.
