I'll create a comprehensive PyChrono simulation featuring the Curiosity rover on rigid terrain. Here's the complete script:

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import pychrono.robot as robot
import math
import numpy as np

# Initialize the Chrono system
chrono.SetChronoDataPath("path/to/chrono/data/")  # Set your Chrono data path
system = chrono.ChSystemNSC()

# Set gravitational acceleration
system.Set_G_acc(chrono.ChVectorD(0, 0, -3.71))  # Mars gravity (3.71 m/s²)

# Set solver parameters for better stability
system.SetSolverMaxIterations(50)
system.SetSolverTolerance(1e-3)

# Create contact material for terrain
terrain_material = chrono.ChMaterialSurfaceNSC()
terrain_material.SetFriction(0.8)
terrain_material.SetRestitution(0.1)
terrain_material.SetRollingFriction(0.05)
terrain_material.SetSpinningFriction(0.02)

# Create the rigid terrain (ground)
ground = chrono.ChBodyEasyBox(200, 200, 2, 1000, True, True, terrain_material)
ground.SetPos(chrono.ChVectorD(0, 0, -1))
ground.SetBodyFixed(True)
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# Add some rocky obstacles to make terrain more interesting
obstacle_material = chrono.ChMaterialSurfaceNSC()
obstacle_material.SetFriction(0.7)
obstacle_material.SetRestitution(0.2)

# Create several rock obstacles
rock_positions = [
    chrono.ChVectorD(10, 5, 1),
    chrono.ChVectorD(-8, 12, 0.8),
    chrono.ChVectorD(15, -10, 1.2),
    chrono.ChVectorD(-5, -8, 0.9),
    chrono.ChVectorD(20, 0, 1.1)
]

for i, pos in enumerate(rock_positions):
    size = 0.5 + 0.5 * (i % 3)  # Varying sizes
    rock = chrono.ChBodyEasyBox(size*2, size*1.5, size, 500, True, True, obstacle_material)
    rock.SetPos(pos)
    rock.SetBodyFixed(True)
    rock.GetVisualShape(0).SetColor(chrono.ChColor(0.4, 0.3, 0.2))
    system.Add(rock)

# Initialize the Curiosity rover
class CuriosityRover:
    def __init__(self, system, init_pos):
        self.system = system
        
        # Rover body dimensions (approximate Curiosity dimensions)
        self.body_length = 2.5
        self.body_width = 2.65
        self.body_height = 0.5
        self.body_mass = 899  # kg (Curiosity's mass)
        
        # Create rover body
        self.chassis = chrono.ChBodyEasyBox(
            self.body_length, self.body_width, self.body_height,
            self.body_mass, True, True, terrain_material
        )
        self.chassis.SetPos(init_pos)
        self.chassis.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
        self.chassis.GetVisualShape(0).SetColor(chrono.ChColor(0.8, 0.8, 0.9))
        system.Add(self.chassis)
        
        # Wheel parameters
        self.wheel_radius = 0.25
        self.wheel_width = 0.2
        self.wheel_mass = 5.0
        
        # Create wheels
        self.wheels = []
        self.wheel_motors = []
        self.wheel_positions = [
            chrono.ChVectorD(1.0, 1.1, -0.3),   # Front left
            chrono.ChVectorD(1.0, -1.1, -0.3),  # Front right
            chrono.ChVectorD(0.0, 1.1, -0.3),   # Middle left
            chrono.ChVectorD(0.0, -1.1, -0.3),  # Middle right
            chrono.ChVectorD(-1.0, 1.1, -0.3),  # Rear left
            chrono.ChVectorD(-1.0, -1.1, -0.3)  # Rear right
        ]
        
        wheel_material = chrono.ChMaterialSurfaceNSC()
        wheel_material.SetFriction(1.2)
        wheel_material.SetRestitution(0.1)
        
        for i, wheel_pos in enumerate(self.wheel_positions):
            # Create wheel
            wheel = chrono.ChBodyEasyCylinder(
                chrono.ChAxis_Y, self.wheel_radius, self.wheel_width,
                self.wheel_mass, True, True, wheel_material
            )
            abs_pos = init_pos + wheel_pos
            wheel.SetPos(abs_pos)
            wheel.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
            wheel.GetVisualShape(0).SetColor(chrono.ChColor(0.2, 0.2, 0.2))
            system.Add(wheel)
            self.wheels.append(wheel)
            
            # Create motor for wheel
            motor = chrono.ChLinkMotorRotationSpeed()
            motor.Initialize(wheel, self.chassis, chrono.ChFrameD(abs_pos, chrono.QUNIT))
            motor.SetSpeedFunction(chrono.ChFunction_Const(0))
            system.Add(motor)
            self.wheel_motors.append(motor)
        
        # Steering variables
        self.target_speed = 0.0
        self.target_steering = 0.0
        self.max_speed = 2.0  # m/s
        self.max_steering_angle = 0.5  # radians
        
    def update_controls(self, throttle, steering):
        """Update rover controls based on input"""
        self.target_speed = throttle * self.max_speed
        self.target_steering = steering * self.max_steering_angle
        
        # Differential steering (6-wheel skid steering)
        left_speed = self.target_speed - self.target_steering
        right_speed = self.target_speed + self.target_steering
        
        # Convert linear speed to angular speed
        left_angular_speed = left_speed / self.wheel_radius
        right_angular_speed = right_speed / self.wheel_radius
        
        # Apply to wheels (left side: 0, 2, 4; right side: 1, 3, 5)
        for i in range(6):
            if i % 2 == 0:  # Left wheels
                speed_func = chrono.ChFunction_Const(left_angular_speed)
            else:  # Right wheels
                speed_func = chrono.ChFunction_Const(-right_angular_speed)
            self.wheel_motors[i].SetSpeedFunction(speed_func)

# Create the rover
rover_start_pos = chrono.ChVectorD(0, 0, 2)
curiosity = CuriosityRover(system, rover_start_pos)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Curiosity Rover on Mars Terrain')
vis.Initialize()

# Set up camera
vis.AddCamera(chrono.ChVectorD(5, -10, 5), chrono.ChVectorD(0, 0, 1))
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)

# Lighting setup
vis.AddLight(chrono.ChVectorD(30, 50, 100), 200, chrono.ChColor(1.0, 0.9, 0.8))
vis.AddLight(chrono.ChVectorD(-30, 30, 50), 150, chrono.ChColor(0.8, 0.8, 1.0))

# Enable shadows
vis.EnableShadows()

# Add sky and atmosphere effects
vis.AddSkyBox()
vis.SetAmbientLight(chrono.ChColor(0.3, 0.2, 0.15))  # Mars-like ambient lighting

# Add logo and visual elements
vis.AddLogo()

# Control variables
throttle = 0.0
steering = 0.0
time_step = 0.01
simulation_time = 0.0
max_simulation_time = 60.0  # seconds

# Control parameters for autonomous movement pattern
control_mode = 0  # 0: manual, 1: autonomous pattern
autonomous_time = 0.0

print("Curiosity Rover Simulation")
print("Controls:")
print("  W/S: Forward/Backward")
print("  A/D: Turn Left/Right")
print("  Space: Stop")
print("  M: Toggle autonomous mode")
print("  ESC: Exit simulation")

# Main simulation loop
while vis.Run() and simulation_time < max_simulation_time:
    vis.BeginScene()
    vis.Render()
    
    # Handle user input
    if vis.GetDevice().isKeyPressed(ord('W')):
        throttle = min(1.0, throttle + 0.02)
    elif vis.GetDevice().isKeyPressed(ord('S')):
        throttle = max(-1.0, throttle - 0.02)
    else:
        throttle *= 0.95  # Gradual deceleration
    
    if vis.GetDevice().isKeyPressed(ord('A')):
        steering = max(-1.0, steering - 0.03)
    elif vis.GetDevice().isKeyPressed(ord('D')):
        steering = min(1.0, steering + 0.03)
    else:
        steering *= 0.9  # Return to center
    
    if vis.GetDevice().isKeyPressed(ord(' ')):
        throttle = 0.0
        steering = 0.0
    
    # Toggle autonomous mode
    if vis.GetDevice().isKeyPressed(ord('M')):
        control_mode = 1 - control_mode
        autonomous_time = 0.0
        print(f"Control mode: {'Autonomous' if control_mode else 'Manual'}")
    
    # Autonomous control pattern
    if control_mode == 1:
        autonomous_time += time_step
        # Create a figure-8 pattern
        throttle = 0.3
        steering = 0.5 * math.sin(autonomous_time * 0.5)
        
        # Add some stops and direction changes
        if int(autonomous_time) % 15 == 0 and autonomous_time % 1.0 < 0.1:
            throttle = 0.0
    
    # Update rover controls
    curiosity.update_controls(throttle, steering)
    
    # Update camera to follow rover
    rover_pos = curiosity.chassis.GetPos()
    rover_rot = curiosity.chassis.GetRot()
    
    # Camera follows behind the rover
    camera_offset = chrono.ChVectorD(-8, 0, 4)
    camera_pos = rover_pos + rover_rot.Rotate(camera_offset)
    vis.GetActiveCamera().setPosition(chronoirr.vector3df(
        float(camera_pos.x), float(camera_pos.y), float(camera_pos.z)
    ))
    vis.GetActiveCamera().setTarget(chronoirr.vector3df(
        float(rover_pos.x), float(rover_pos.y), float(rover_pos.z + 1)
    ))
    
    # Display simulation info
    info_text = f"Time: {simulation_time:.1f}s | Speed: {throttle*100:.0f}% | Steering: {steering*100:.0f}%"
    info_text += f" | Mode: {'Auto' if control_mode else 'Manual'}"
    
    # Advance the simulation
    system.DoStepDynamics(time_step)
    simulation_time += time_step
    
    vis.EndScene()

print("Simulation completed!")
```
