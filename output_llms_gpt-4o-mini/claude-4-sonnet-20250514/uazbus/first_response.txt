Here's a complete PyChrono simulation script based on your requirements:

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import math

def main():
    # Initialize the PyChrono environment
    chrono.SetChronoDataPath("path/to/chrono/data/")  # Set your Chrono data path
    
    # Create the Chrono system
    system = chrono.ChSystemNSC()
    system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))
    
    # Set solver parameters for better stability
    system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)
    system.SetSolverMaxIterations(150)
    system.SetMaxPenetrationRecoverySpeed(4.0)
    
    # Create and configure the terrain
    terrain = veh.RigidTerrain(system)
    
    # Create a patch for the terrain
    patch_mat = chrono.ChMaterialSurfaceNSC()
    patch_mat.SetFriction(0.9)        # Set friction coefficient
    patch_mat.SetRestitution(0.01)    # Set restitution coefficient
    patch_mat.SetYoungModulus(2e7)
    patch_mat.SetPoissonRatio(0.3)
    
    # Add terrain patch (large flat ground)
    patch = terrain.AddPatch(patch_mat, 
                           chrono.CSYSR(chrono.ChVectorD(0, 0, 0), chrono.QUNIT),
                           200.0, 200.0)  # 200m x 200m terrain
    patch.SetTexture(veh.GetDataFile("terrain/textures/grass.jpg"), 200, 200)
    patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
    
    # Initialize the terrain
    terrain.Initialize()
    
    # Create the UAZBUS vehicle
    vehicle = veh.UAZBUS()
    vehicle.SetContactMethod(chrono.ChContactMethod_NSC)
    vehicle.SetChassisCollisionType(veh.CollisionType_NONE)
    vehicle.SetChassisFixed(False)
    vehicle.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 1.0), 
                                             chrono.ChQuaternionD(1, 0, 0, 0)))
    vehicle.SetInitFwdVel(0.0)
    vehicle.SetTireType(veh.TireModelType_TMEASY)
    vehicle.SetTireStepSize(1e-3)
    
    # Initialize the vehicle
    vehicle.Initialize()
    
    # Set visualization type for vehicle parts
    vehicle.SetChassisVisualizationType(veh.VisualizationType_MESH)
    vehicle.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
    vehicle.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
    vehicle.SetWheelVisualizationType(veh.VisualizationType_MESH)
    vehicle.SetTireVisualizationType(veh.VisualizationType_MESH)
    
    # Create the driver system
    driver = veh.ChInteractiveDriverIRR()
    
    # Set driver input modes
    driver.SetSteeringDelta(0.02)
    driver.SetThrottleDelta(0.02)
    driver.SetBrakingDelta(0.02)
    
    # Initialize driver system
    driver.Initialize()
    
    # Create the Irrlicht visualization system
    vis = chronoirr.ChVisualSystemIrrlicht()
    vis.AttachSystem(system)
    vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
    vis.SetWindowSize(1280, 720)
    vis.SetWindowTitle('UAZBUS Vehicle Simulation')
    vis.Initialize()
    vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
    vis.AddSkyBox()
    vis.AddCamera(chrono.ChVectorD(0, -8, 3), chrono.ChVectorD(0, 0, 1))
    vis.AddTypicalLights()
    
    # Create a chase camera that follows the vehicle
    chase_cam = chrono.ChVectorD(-8.0, 0.0, 2.5)  # Camera offset from vehicle
    
    # Simulation parameters
    step_size = 1e-3
    render_frame_rate = 50  # FPS for rendering
    render_steps = int(1.0 / (render_frame_rate * step_size))
    
    # Simulation variables
    time = 0
    step_number = 0
    render_frame = 0
    
    # Real-time timer
    realtime_timer = chrono.ChRealtimeStepTimer()
    
    # Main simulation loop
    while vis.Run():
        # Get driver inputs
        driver_inputs = driver.GetInputs()
        
        # Current vehicle position for camera tracking
        vehicle_pos = vehicle.GetPos()
        vehicle_rot = vehicle.GetRot()
        
        # Update camera position to follow vehicle
        cam_pos = vehicle_pos + vehicle_rot.Rotate(chase_cam)
        look_at = vehicle_pos + vehicle_rot.Rotate(chrono.ChVectorD(10, 0, 0))
        
        # Synchronize systems
        time = system.GetChTime()
        
        # Synchronize driver
        driver.Synchronize(time)
        
        # Synchronize vehicle
        vehicle.Synchronize(time, driver_inputs, terrain)
        
        # Synchronize terrain
        terrain.Synchronize(time)
        
        # Advance system dynamics
        driver.Advance(step_size)
        vehicle.Advance(step_size)
        terrain.Advance(step_size)
        system.DoStepDynamics(step_size)
        
        # Render scene
        if step_number % render_steps == 0:
            # Update camera
            vis.UpdateCamera(cam_pos, look_at)
            
            # Begin scene
            vis.BeginScene()
            vis.Render()
            
            # Display vehicle information
            vehicle_speed = vehicle.GetSpeed()
            engine_rpm = vehicle.GetEngine().GetMotorSpeed() * 60 / (2 * math.pi)
            
            # Draw debug information
            vis.GetGUIEnvironment().getBuiltInFont().draw(
                f"Time: {time:.2f} s", 
                chronoirr.recti(10, 10, 200, 30), 
                chronoirr.SColor(255, 255, 255, 255)
            )
            vis.GetGUIEnvironment().getBuiltInFont().draw(
                f"Speed: {vehicle_speed:.2f} m/s", 
                chronoirr.recti(10, 35, 200, 55), 
                chronoirr.SColor(255, 255, 255, 255)
            )
            vis.GetGUIEnvironment().getBuiltInFont().draw(
                f"Engine RPM: {engine_rpm:.0f}", 
                chronoirr.recti(10, 60, 200, 80), 
                chronoirr.SColor(255, 255, 255, 255)
            )
            
            # End scene
            vis.EndScene()
            render_frame += 1
        
        # Spin in place for real time to catch up
        realtime_timer.Spin(step_size)
        step_number += 1
        
        # Optional: Exit condition (remove if you want infinite simulation)
        if time > 30.0:  # Run for 30 seconds
            break
    
    print(f"Simulation completed. Final time: {time:.2f} seconds")
    return 0

if __name__ == "__main__":
    main()
```

This script creates a comprehensive PyChrono simulation with the following features:

1. **PyChrono Environment Initialization**: Sets up the core system with NSC contact method and appropriate solver settings.

2. **UAZBUS Vehicle**: Creates a UAZBUS vehicle with specified initial conditions (position at 1m height, zero initial velocity) and proper visualization settings.

3. **Rigid Terrain**: Implements a large (200m x 200m) flat terrain with specified friction (0.9) and restitution (0.01) coefficients.

4. **Driver System**: Sets up an interactive driver system that can be controlled via keyboard inputs.

5. **Irrlicht Visualization**: Creates a real-time visualization system with:
   - Chase camera that follows the vehicle
   - Proper lighting and skybox
   - Vehicle information display (time, speed, engine RPM)
   - 1280x720 window resolution

6. **Simulation Loop**: Implements a synchronized simulation loop that:
   - Maintains real-time execution using `ChRealtimeStepTimer`
   - Updates all systems (vehicle, terrain, driver, visualization) at each timestep
   - Uses 1ms timestep for physics and 50 FPS for rendering
   - Provides real-time vehicle information display

**Controls**:
- **W/S**: Throttle/Brake
- **A/D**: Steering left/right
- **Arrow keys**: Alternative steering and throttle control

The simulation will run for 30 seconds by default, but you can modify or remove the exit condition for continuous operation.
