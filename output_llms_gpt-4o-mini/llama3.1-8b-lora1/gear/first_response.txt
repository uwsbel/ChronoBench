import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.cascade as cascade
from pychrono import irrlicht as irr

# The path to the Chrono data directory containing various assets (meshes, textures, data files)
# is automatically set, relative to the default location of this demo.
# If running from a different directory, you must change the path to the data directory with: 
#chrono.SetChronoDataPath('path/to/data')

#  Create the simulation system
system = chrono.ChSystemNSC()

# Set collision system to use a collision buffer
system.SetCollisionSystem(chrono.ChCollisionSystem.Type_BULLET)

# Set solver
system.SetSolver(chrono.ChSolver.Type_PSOR)

# Create the vehicle, set parameters, and initialize
vehicle = cascade.CascadeLagocha()
vehicle.SetContactMethod(chrono.ChContactMethod_NSC)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(chrono.ChVector3d(-11, -2, 1.5), chrono.QUNIT))
vehicle.SetEngineType(cascade.CEngineModelType_SIMPLE)
vehicle.SetTransmissionType(cascade.CTransmissionModelType_AUTOMATIC_SHAFTS)
vehicle.SetDriveType(cascade.CDriveTypeWV_AWD)
vehicle.SetThrottle(0.5)  # 0..1
vehicle.SetBrake(0.0)     # 0..1
vehicle.Initialize()
vehicle.SetChassisFixed(True)

# Create the ground, set parameters
patch_mat = chrono.ChContactMaterialNSC()
patch_mat.SetFriction(0.9)
patch = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, patch_mat)
patch.SetPos(chrono.ChVector3d(0, 0, -1))
patch.SetFixed(True)
patch.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile('textures/blue.png'))
system.Add(patch)

# Create the visualization system and attach it to the vehicle
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Cascade Lagocha')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(2.5,2.2,-2.4), chrono.ChVector3d(0.0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(3.5, 1.1, 4.5), chrono.ChVector3d(0, 0, 0), 1, 10, 10, 10, 25, 256)

#  Create the interactive driver
driver = chronoirr.ChInteractiveDriverIRR(vis)

#  Set the initial vehicle position
init_pos = vehicle.GetInitPos()

#  Run the simulation
while vis.Run():
    time = vehicle.GetChassisBody().GetPos().z

    # Draw scene
    vis.BeginScene()
    vis.GetSceneManager().getActiveCamera().setTargetPoint(chrono.ChVector3d(init_pos.x, init_pos.y, init_pos.z + 2.5))
    vis.GetSceneManager().getActiveCamera().setVerticalFOV(0.4)
    vis.Render()
    vis.EndScene()

    # Get user input
    driver_inputs = driver.GetInputs()

    # Update modules (process inputs from other modules)
    driver.Synchronize(time)
    vis.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, patch, vis)

    driver.Advance(step_size)
    vis.Advance(step_size)
    vehicle.Advance(step_size)
