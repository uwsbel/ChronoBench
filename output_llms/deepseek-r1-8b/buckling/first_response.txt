### PyChrono Simulation Script for Beam Buckling Scenario

```python
import pychrono as py
import numpy as np
import time

# Initialize PyChrono environment and core components
py.init()

# Set up the simulation
sim = py.Simulation()
sim.set_solver(py.SolverType.NEIGHBOR_KARKOVSKY)
sim.set_timestep(1e-4)  # 0.0001 seconds

# Define beam properties
beam_length = 4.0  # meters
beam_cross_section = py.CrossSectionType rectangles
beam_cross_section.length = 0.1  # meters
beam_cross_section.width = 0.05  # meters
beam_density = 8000  # kg/m^3
young_modulus = 210e9  # Pa

# Beam material properties
beam_material = py.MaterialType()
beam_material.density = beam_density
beam_material.young_modulus = young_modulus
beam_material.poisson_ratio = 0.0

# Create beam with FEA
beam = py.RigidMultiBody()
beam.set_name("Beam")
beam.set_cross_section(beam_cross_section)
beam.set_length(beam_length)
beam.set_material(beam_material)
beam.add_node(py.Node(py.NodeType.FIXED, [0, 0, 0]))  # Left support
beam.add_node(py.Node(py.NodeType.FIXED, [beam_length, 0, 0]))  # Right support

# Add nodes and elements
nodes = []
elements = []
for i in range(10):
    x = i * (beam_length / 9)
    nodes.append(py.Node(py.NodeType.RIGID, [x, 0, 0]))
    elements.append(py.BeamElementType.EULER_BERNOLLI)
    beam.add_node(nodes[i])
    beam.add_element(elements[i], nodes[i], nodes[i+1])

# Add gravity
sim.add_floor(py.FloorType.INDOOR, [0, -9.81, 0])

# Create motor
motor = py.RigidMultiBody()
motor.set_name("Motor")
motor.set_mass(10)  # kg
motor.set_size([0.1, 0.1, 0.1])  # meters
motor.set_color([1, 0, 0])  # Red color

# Motor initial position and orientation
motor.set_position([beam_length / 2, 0, 0])
motor.set_orientation([0, 0, 1])

# Add motor to simulation
sim.add_body(motor)
sim.add_constraint(py.ConstraintType.FIXED, motor, [0, 0, 0])

# Create visualization
viz = py.Visualization()
viz.set_camera([5, 5, 5], [0, 0, 0], [0, 0, 1])
viz.set renderer(py.RendererType.IRRlicht)
viz.add_body(motor, 0.5)
viz.add_body(beam, 0.5)

# Motor control functions
def motor_control(sim, body, time):
    # Simple motor control function that rotates the motor
    angle = time * 0.1  # degrees
    motor_pos = motor.get_position()
    motor_rot = py.Transform3D()
    motor_rot.set_rotation([0, 0, 1], angle * np.pi / 180)
    motor.set_position(motor_pos + motor_rot.to_global())
    # Apply force to the beam
    force_point = nodes[5]  # Midpoint of the beam
    force = [0, 0, 0]
    force_point.add_force(motor, [0, 0, -50], [force_point.get_position()])

def get_motor_position(sim):
    return motor.get_position()

def get_beam_nodes(sim):
    return nodes

# Add motor control function
sim.add_control_function(motor, motor_control)

# Add visualization callback
def visualize(sim):
    viz.update()
    # Draw grid
    py.draw_grid([10, 10, 10])
    # Draw beam nodes
    for node in nodes:
        py.draw_sphere(node, 0.01, [1, 0, 0])
    # Draw motor
    py.draw_box(motor, 0.05, [1, 0, 0])
    # Draw constraints
    py.draw_lines([nodes[0], nodes[1]], [0, 0, 0])
    py.draw_lines([nodes[-1], nodes[-2]], [0, 0, 0])

# Set visualization callback
viz.set_visualization_callback(visualize)

# Start simulation
sim.start()
```
