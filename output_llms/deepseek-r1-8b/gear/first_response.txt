### 1. **PyChrono Initialization**
```python
import pychrono as py
import pychrono.gear as gear
import pychrono.visualization.irrlight as vis
import numpy as np

# Initialize PyChrono
py.init()

# Create simulation
sim = py.Simulation()
sim.set_time_unit(py.TimeUnit.SECONDS)
sim.set_gravity(np.array([0, 0, -9.81]))
sim.set_damping(0.1)

# Create scene
scene = vis.Scene()
scene.set_lighting(vis.LightSetup(
    ambient_light=(1, 1, 1),
    directional_light=(1, 1, 1),
    directional_light_position=(5, 5, 5)
))

# Create camera
camera = vis.Camera()
camera.set_position(np.array([5, 5, 5]))
camera.set_lookat(np.array([0, 0, 0]))
camera.set_up(np.array([0, 1, 0]))
```

### 2. **Add Fixed Truss**
```python
# Create fixed truss using boxes
truss = []
truss.append(py.Box(np.array([0.5, 0.5, 0.5]), color=(0.5, 0.5, 0.5), material=py.Material.GRAY_BRICK))
truss.append(py.Box(np.array([0.5, 0.5, 0.5]), color=(0.5, 0.5, 0.5), material=py.Material.GRAY_BRICK))
truss.append(py.Box(np.array([0.5, 0.5, 0.5]), color=(0.5, 0.5, 0.5), material=py.Material.GRAY_BRICK))
truss.append(py.Box(np.array([0.5, 0.5, 0.5]), color=(0.5, 0.5, 0.5), material=py.Material.GRAY_BRICK))

# Position the truss
for box in truss:
    box.set_position(np.array([0, 0, 0]))
    scene.add_object(box)

# Add truss to the simulation
sim.add_object(truss[0])
sim.add_object(truss[1])
sim.add_object(truss[2])

```

### 3. **Add Rotating Bar**
```python
# Create rotating bar using cylinder
bar = py.Cylinder(np.array([0.1, 0.1, 0.1]), color=(0.8, 0.8, 0.8), material=py.Material.SILVER)
bar.set_position(np.array([1, 0, 0]))
bar.set_rotation(np.array([0, 0, 1]))
bar.set_linear_damping(0.1)

# Add forces to the bar
force = py.Forces()
force.add_linear_force(np.array([0, 0, 0]), np.array([0, 0, 5]))
bar.add_force(force)

# Add bar to the simulation
sim.add_object(bar)
```

### 4. **Add Epicyclic Gears**
```python
# Create fixed gear
fixed_gear = gear.EpicyclicGear()
fixed_gear.set_radius(0.5)
fixed_gear.set_number_of_teeth(20)
fixed_gear.set_position(np.array([1, 0, 0]))
fixed_gear.set_rotation(np.array([0, 0, 0]))
fixed_gear.set_moment_of_inertia(1e-4)

# Create moving gear
moving_gear = gear.EpicyclicGear()
moving_gear.set_radius(0.5)
moving_gear.set_number_of_teeth(20)
moving_gear.set_position(np.array([1, 1, 0]))
moving_gear.set_rotation(np.array([0, 0, 0]))
moving_gear.set_moment_of_inertia(1e-4)

# Add gears to the simulation
sim.add_object(fixed_gear)
sim.add_object(moving_gear)
```

### 5. **Add Gear Motor**
```python
# Create gear motor
motor = py.Motor()
motor.set_type(py.MotorType.ROTARY)
motor.set_position(np.array([1, 1, 0]))
motor.set_rotation(np.array([0, 0, 1]))
motor.set_torque_limit(10)
motor.set_force(np.array([0, 0, 5]))

# Add motor to the simulation
sim.add_object(motor)
```
