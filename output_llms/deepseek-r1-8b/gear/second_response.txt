import pychrono.irrlicht as chronoirr

# Create a Chrono::Engine physical system
sys = chrono.ChSystemNSC()

mat = chrono.ChContactMaterialNSC()
sys.AddContactMaterial(mat)

# Create the truss
radA = 1.5  # Modified gear radius
radB = 3.5  # Modified gear radius
truss_dims = chrono.ChBodyEasyBox(15, 8, 2, 1000, True, False, mat)  # Modified truss dimensions
sys.Add(truss_body)
truss_body.SetFixed(True)
truss_body.SetPos(chrono.ChVector3d(0, 0, 3))

# Visualization setup
vis_mat = chrono.ChVisualMaterial()
vis_mat.SetKdTexture(chrono.GetChronoDataFile('textures/pinkwhite.png'))
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Gears and pulleys')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(12, 15, -20))
vis.AddTypicalLights()

# Create rotating bar support
train_dims = chrono.ChBodyEasyBox(8, 1.5, 1.0, 1000, True, False, mat)
sys.Add(train_body)
train_body.SetPos(chrono.ChVector3d(3, 0, 0))

# Create revolute joint between truss and rotating bar
link_revolute = chrono.ChLinkLockRevolute()
link_revolute.Initialize(truss_body, train_body, chrono.ChFramed(chrono.ChVector3d(0, 0, 0), chrono.QUNIT))
sys.AddLink(link_revolute)

# Create first gear
gearA = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, radA, 0.5, 1000, True, False, mat)

gearA.SetPos(chrono.ChVector3d(0, 0, -1))
gearA.SetRot(chrono.QuatFromAngleX(m.pi / 2))
gearA.GetVisualShape(0).SetMaterial(0, vis_mat)

# Visualization shaft for gear A
mshaft_shape = chrono.ChVisualShapeCylinder(radA * 0.3, 10)  # Modified shaft size
gearA.AddVisualShape(mshaft_shape, chrono.ChFramed(chrono.ChVector3d(0, 3.5, 0), chrono.QuatFromAngleX(m.pi / 2)))

# Create motor link for gear rotation
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(gearA, truss_body, chrono.ChFramed(chrono.ChVector3d(0, 0, 0), chrono.QUNIT))
motor.SetSpeedFunction(chrono.ChFunctionConst(3))  # Modified speed

# Create second gear
interaxis12 = radA + radB  # Modified inter-axis distance
gearB = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, radB, 0.4, 1000, True, False, mat)
sys.Add(gearB)
gearB.SetPos(chrono.ChVector3d(interaxis12, 0, -2))  # Modified position
gearB.SetRot(chrono.QuatFromAngleX(m.pi / 2))
gearB.GetVisualShape(0).SetMaterial(0, vis_mat)

link_revolute = chrono.ChLinkLockRevolute()
