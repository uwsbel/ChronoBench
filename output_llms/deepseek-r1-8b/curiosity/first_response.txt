To create a PyChrono simulation of the Curiosity rover navigating on a rigid terrain, follow these steps:

### 1. **Import Required Libraries**
```python
import pychrono as pc
import numpy as np
import matplotlib.pyplot as plt
```

### 2. **Initialize the PyChrono Environment**
```python
# Create a new scene
scene = pc.Scene()
# Set the gravity vector
g = pc.Vec3(0, -9.81, 0)
scene.setGravity(g)
```

### 3. **Create the Ground**
```python
# Define the ground plane
ground = pc.Plane()
# Load a ground texture (replace with your texture file path)
ground.setTexture(pc.Texture("ground_texture.png"))
# Position the ground at the origin
ground.setPos(pc.Vec3(0, 0, 0))
# Enable collision for the ground
ground.setCollision(True)
# Set realistic ground properties
ground.setFriction(2.0)
ground.setNormal(0, 0, 1)
```

### 4. **Create the Rover**
```python
# Define the rover dimensions
rover_size = 1.5  # 1.5 meters for each side
# Create a rover body
rover_body = pc.Box(
    np.array([rover_size, 1.5, 1.0]),  # length, width, height
    pc.MESH_BOX,
    pc.CollisionObject()
)
# Position the rover
rover_body.setPos(pc.Vec3(0, 0, 0.5))
# Add a box model for visualization
rover_visual = pc.ModelBox(rover_size)
rover_visual.setPos(pc.Vec3(0, 0, 0.5))

# Add a motor control driver
motor_control = pc.MotorControlDriver()
# Set motor properties
motor_control.setMaxTorque(100)
motor_control.setSpeed(120)
motor_control.setGearRatio(1.0)

# Create wheels
wheel_radius = 0.5
wheel_body = pc.Cylinder(
    np.array([1.0, 0.5, 0.5]),  # radius, height, segments
    pc.MESH_CYLINDER,
    pc.CollisionObject()
)
wheel_body.setPos(pc.Vec3(0, 0.5, -0.5))
wheel_body.setCollision(True)
wheel_body.setFriction(1.0)

# Attach wheels to the motor
motor_control.addActuator(wheel_radius, wheel_body, pc.ROTATION_Z)
```

### 5. **Set Up the Camera**
```python
# Create a camera
camera = pc.PerspectiveCamera()
# Position the camera
camera.setPos(pc.Vec3(2, 2, 2))
camera.lookAt(pc.Vec3(0, 0, 0))
camera.setUp(pc.Vec3(0, 1, 0))
```

### 6. **Add Lighting and Visualization**
```python
# Add a directional light
light = pc.DirectionalLight()
light.setPos(pc.Vec3(5, 5, 5))
light.setDiffuse(pc.Vec3(1, 1, 1))
light.setSpecular(pc.Vec3(1, 1, 1))
light.setEnabled(True)

# Enable shadows
scene.setShadowEnabled(True)
scene.setShadowIntensity(0.5)
```

### 7. **Set Up the Simulation Loop**
```python
# Add the ground and rover to the scene
scene.addBody(ground)
scene.addBody(rover_body)
scene.addModel(rover_visual)
scene.addModel(wheel_body)
scene.addActuator(motor_control)

# Visualization setup
renderer = pc.IrrlichtRenderer()
renderer.setSize(800, 600)
renderer.setHMD(True)
renderer.setCamera(camera)
renderer.setLight(light)

# Add title and text display
renderer.addAnnotation("Curiosity Rover Simulation", pc.ANNOTATION_TOP_LEFT)
renderer.addAnnotation("Speed: 0 m/s", pc.ANNOTATION_TOP_RIGHT)
renderer.addAnnotation("Torque: 0 Nm", pc.ANNOTATION_BOTTOM_RIGHT)
```

### 8. **Simulate the Rover Navigation**
```python
# Set simulation time step
dt = 0.01

# Steer controls
steering_speed = 100  # Maximum steering speed
current_torque = 0

# Simulation loop
while True:
    # Handle user input for steering
    key = pc.Input().getKeyDown()
    if key['a']:
        current_torque = -steering_speed
    if key['d']:
        current_torque = steering_speed

    # Apply torque to the motor
    motor_control.setTorque(current_torque)

    # Update the simulation
    pc.simulate(scene, dt)

    # Visualize the simulation
    renderer.update()

    # Display current speed
    speed = scene.getSpeed()
    plt.figure(figsize=(6, 3))
    plt.plot(speed)
    plt.title("Rover Speed Over Time")
    plt.xlabel("Time (s)")
    plt.ylabel("Speed (m/s)")
    plt.show()
```

### 9. **Add Visualization Textures**
```python
# Add NASA logo to the ground
nasa_logo = pc.Texture("nasa_logo.png")
ground.setTexture(nasa_logo, pc.Texture.ABSTRACT, 0, 0, 0, 0, 0, 0, 0)

# Add rover logo
rover_logo = pc.Texture("rover_logo.png")
rover_visual.setTexture(rover_logo, pc.Texture.ABSTRACT, 0.5, 0.5, 0.5, 0.5, 0, 0, 0)
```

### 10. **Run the Simulation**
```python
# Start the simulation
pc.startSimulation(scene)

# Visualization
plt.show()
```

import pychrono as pc

import matplotlib.pyplot as plt

# Create a new scene

# Set simulation time step

# Steer controls

# Simulation loop

    key = pc.Input().getKeyDown()
    if key['a']:
        current_torque = -steering_speed

        current_torque = steering_speed

    # Visualize the simulation
