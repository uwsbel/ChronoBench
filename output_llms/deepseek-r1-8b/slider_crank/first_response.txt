To create a PyChrono simulation of a simple crank-slider mechanism, follow these steps:

### 1. Initialize the PyChrono Environment
```python
import pychrono as py
import pychrono visualize as py_vis
import pychrono.objects as objects
import pychrono.materials as materials
import pychrono.mesh as mesh
import pychrono.textures as textures

# Initialize PyChrono
py.initialize()

# Create a scene
scene = py.scene.Scene()

# Set up the camera
cam = scene.add("perspective_camera")
cam.set_position([5, 5, 5])
cam.set_lookat([0, 0, 0])
cam.set_up([0, 1, 0])
cam.set_fov(45)

# Add lighting
light = scene.add("directional_light")
light.set_position([10, 10, 10])
light.set_color([1, 1, 1])

# Add ambient light
ambient_light = scene.add("ambient_light")
ambient_light.set_position([10, 10, 10])
ambient_light.set_color([0.5, 0.5, 0.5])
```

### 2. Define the Floor (Truss)
```python
# Create a truss structure as the floor
floor_body = objects.Box2D(
    name="floor",
    size=[50, 5, 0],
    position=[0, 0, 0],
    material=materials.DefaultMaterial(),
    color=[1, 0, 0]
)
scene.add(floor_body)
```

### 3. Define the Crankshaft
```python
# Define crankshaft parameters
crankshaft_length = 10
crankshaft_radius = 5
crankshaft_height = 1

# Create the crankshaft body
crankshaft_body = objects.RigidBody(
    name="crankshaft",
    mass=0,
    size=[crankshaft_length, crankshaft_radius, crankshaft_height],
    position=[0, 0, 0],
    material=materials.DefaultMaterial(),
    color=[0.5, 0.5, 0.5]
)
scene.add(crankshaft_body)

# Create a revolute joint for the motor
crankshaft_joint = py.Joint.Revolving2D(
    name="crankshaft_motor_joint",
    axis=[0, 0, 1],
    rotation_range=[0, 2 * py.M_PI],
    torque_limit=100,
    rotational_inertia=1e-5,
    position=[0, 0, 0]
)
scene.add(crankshaft_joint)
```

### 4. Define the Connecting Rod
```python
# Define connecting rod parameters
connecting_rod_length = 15
connecting_rod_radius = 2
connecting_rod_height = 1

# Create the connecting rod body
connecting_rod_body = objects.RigidBody(
    name="connecting_rod",
    mass=0,
    size=[connecting_rod_length, connecting_rod_radius, connecting_rod_height],
    position=[0, 0, 0],
    material=materials.DefaultMaterial(),
    color=[0, 0, 1]
)
scene.add(connecting_rod_body)

# Create pin joint between crankshaft and connecting rod
connecting_rod_pin_joint = py.Joint.Pin2D(
    name="connecting_rod_pin_joint",
    position=[0, 0, 0]
)
scene.add(connecting_rod_pin_joint)

# Create prismatic joint between connecting rod and piston
connecting_rod_piston_joint = py.Joint.Prismatic2D(
    name="connecting_rod_piston_joint",
    axis=[0, 0, 1],
    position=[0, 0, 0]
)
scene.add(connecting_rod_piston_joint)
```

### 5. Define the Piston
```python
# Define piston parameters
piston_length = 5
piston_radius = 3
piston_height = 1

# Create the piston body
piston_body = objects.RigidBody(
    name="piston",
    mass=0,
    size=[piston_length, piston_radius, piston_height],
    position=[0, 0, 0],
    material=materials.DefaultMaterial(),
    color=[1, 1, 0]
)
scene.add(piston_body)

# Create prismatic joint between piston and floor
piston_floor_joint = py.Joint.Prismatic2D(
    name="piston_floor_joint",
    axis=[0, 0, 1],
    position=[0, 0, 0]
)
scene.add(piston_floor_joint)
```

### 6. Define the Motor
```python
# Create a rotational motor
motor = py.Motor.Rotational(
    name="crankshaft_motor",
    axis=[0, 0, 1],
    angular_velocity=1,
    torque_limit=100,
    rotational_inertia=1e-5
)
scene.add(motor)

# Attach the motor to the crankshaft joint
motor.attach(crankshaft_joint)
```

### 7. Define Visualization Elements
```python
# Add textures
textures.add("crankshaft_texture", "crankshaft.png")
textures.add("connecting_rod_texture", "connecting_rod.png")

# Apply textures to bodies
crankshaft_body.set_texture("crankshaft_texture", "front")
connecting_rod_body.set_texture("connecting_rod_texture", "front")

# Add logo
logo_body = objects.Text(
    name="logo",
    text="Crank-Slider Mechanism",
    font_size=2,
    position=[0, 0, 0],
    color=[0, 0, 0]
)
scene.add(logo_body)
```

### 8. Finalize the Scene
```python
# Set the simulation time step
py.set_time_step(0.001)

# Start the simulation
py.start()

# Visualize the scene
py_vis.add_scene(scene)
```

### 9. Run the Simulation
```python
# Keep the simulation running
while py.get_event() != "quit":
    py.step()
```

### 10. Close the Simulation
```python
py.close()
```

This script initializes a crank-slider mechanism with a floor, crankshaft, connecting rod, and piston, driven by a motor. The visualization uses Irrlicht with customizable camera settings, lighting, and textures. The crankshaft rotates at a constant angular speed, and the piston moves up and down as the connecting rod translates.
