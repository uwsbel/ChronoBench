import pychrono.core as chrono
import pychrono.sensor as sens

import numpy as np
import time
import math

def main():
    # Create the system
    mphysicalSystem = chrono.ChSystemNSC()

    # Create a box object
    box_body = chrono.ChBodyEasyBox(side=1, side=1, side=1, mass=1000)
    box_body.SetPos(chrono.ChVector3d(0, 0, 0))
    box_body.SetFixed(True)
    mphysicalSystem.Add(box_body)

    # Create a visual shape for the box
    trimesh_shape = chrono.ChVisualShapeBox()
    trimesh_shape.SetName("Box")
    box_body.AddVisualShape(trimesh_shape)

    # Create a sensor manager
    manager = sens.ChSensorManager(mphysicalSystem)

    # Create 3D lidar
    offset_pose = chrono.ChFramed(
        chrono.ChVector3d(-10, 0, 1), 
        chrono.QuatFromAngleAxis(0, chrono.ChVector3d(0, 1, 0))
    )
    lidar_3d = sens.ChLidarSensor(
        box_body,                   # Body to attach lidar
        10.0,                      # Update rate (Hz)
        offset_pose,               # Offset pose
        800,                       # Horizontal samples
        300,                       # Vertical samples
        2 * math.pi,               # Horizontal FOV
        math.pi / 12,              # Max vertical FOV
        -math.pi / 6,              # Min vertical FOV
        100.0,                    # Max range
        sens.LidarBeamShape_RECTANGULAR,  # Beam shape
        2,                         # Sample radius
        0.003,                    # Divergence angle
        0.003,                    # Divergence angle
        sens.LidarReturnMode_STRONGEST_RETURN  # Return mode
    )
    lidar_3d.SetName("3D Lidar")
    lidar_3d.SetLag(0)
    lidar_3d.SetCollectionWindow(1. / 10.0)  # Collection time

    # Add filters for 3D lidar
    lidar_3d.PushFilter(sens.ChFilterDIAccess())
    lidar_3d.PushFilter(sens.ChFilterPCfromDepth())

    if True:  # Visualization enabled
        lidar_3d.PushFilter(sens.ChFilterVisualizePointCloud(640, 480, 1.0, "3D Lidar Point Cloud"))

    # Create 2D lidar
    offset_pose_2d = chrono.ChFramed(
        chrono.ChVector2d(-5, 0),
        chrono.QuatFromAngleAxis(0, chrono.ChVector3d(0, 1, 0))
    )
    lidar_2d = sens.ChLidarSensor2D(
        box_body,                   # Body to attach lidar
        5.0,                      # Update rate (Hz)
        offset_pose_2d,            # Offset pose
        1,                        # Horizontal samples
        1,                        # Vertical samples (single channel)
        2 * math.pi,               # Horizontal FOV
        None,                      # Max vertical FOV (not applicable for 2D)
        None,                      # Min vertical FOV (not applicable for 2D)
        5.0,                    # Max range
        sens.LidarBeamShape_LINE,    # Beam shape (line for 2D)
        1.0,                      # Sample radius
        0.01,                    # Divergence angle
        sens.LidarReturnMode_STRONGEST_RETURN  # Return mode
    )
    lidar_2d.SetName("2D Lidar")
    lidar_2d.SetLag(0)
    lidar_2d.SetCollectionWindow(1. / 5.0)  # Collection time

    # Add filters for 2D lidar
    lidar_2d.PushFilter(sens.ChFilterDIAccess())
    if True:  # Visualization enabled
        lidar_2d.PushFilter(sens.ChFilterVisualizePointCloud2D(640, 480, 1.0, "2D Lidar Point Cloud"))

    # Add sensors to manager
    manager.AddSensor(lidar_3d)
    manager.AddSensor(lidar_2d)

    # Simulate system
    orbit_radius = 10
    orbit_rate = 0.1
    ch_time = 0.0

    render_time = 0
    t1 = time.time()

    while ch_time < 40.0:
        # Set lidar_3d to orbit around the box
        lidar_3d.SetOffsetPose(
            chrono.ChFramed(
                chrono.ChVector3d(
                    -orbit_radius * math.cos(ch_time * orbit_rate),
                    -orbit_radius * math.sin(ch_time * orbit_rate),
                    1
                ),
                chrono.QuatFromAngleAxis(ch_time * orbit_rate, chrono.ChVector3d(0, 0, 1))
            )
        )

        # Access the XYZI buffer from lidar_3d
        xyzi_buffer = lidar_3d.GetMostRecentXYZIBuffer()
        if xyzi_buffer.HasData():
            xyzi_data = xyzi_buffer.GetXYZIData()
            print('3D Lidar buffer received. Resolution: {0}x{1}'.format(xyzi_buffer.Width, xyzi_buffer.Height))
            print('Max Value: {0}'.format(np.max(xyzi_data)))

        # Access the XY buffer from lidar_2d
        xy_buffer = lidar_2d.GetMostRecentXYBuffer()
        if xy_buffer.HasData():
            xy_data = xy_buffer.GetXYData()
            print('2D Lidar buffer received. Resolution: {0}x{1}'.format(xy_buffer.Width, xy_buffer.Height))
            print('Max Value: {0}'.format(np.max(xy_data)))

        # Update sensor manager
        manager.Update()

        # Perform step of dynamics
        mphysicalSystem.DoStepDynamics(1e-3)

        # Get the current time of the simulation
        ch_time = mphysicalSystem.GetChTime()

    print("Sim time:", 40.0, "Wall time:", time.time() - t1)

# Define parameters

horizontal_samples = 800
