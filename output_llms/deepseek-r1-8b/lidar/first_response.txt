```python
import pychrono as pyc
from pychrono.duality import *
from pychrono import core, gpu
import numpy as np

# Initialize PyChrono environment
pyc.init()
scene = pyc.Scene()
physics = pyc.PhysicsManager()
renderer = pyc.Renderer()
renderer.set_camera(position=(5, 5, 5), look_at=(0, 0, 0))

# Load the triangular mesh from the .obj file
mesh = Duality().mesh_obj("triangle_mesh.obj")
body = mesh.body()
body.set_mass(0)  # Make the mesh a fixed body
body.set_position(core.Vector3(0, 0, 0))  # Position the mesh at origin

# Set up the lidar sensor
lidar = pyc.LidarSensor(body, renderer)
lidar.set_noise_filter(pyc.LidarSensor.NOISE_FILTER_LOW_PASS, 0.1)  # Apply low-pass filter
lidar.set_visualization(True)  # Enable visualization of the lidar
lidar.set_save_data(True)  # Save lidar data
lidar.set_data_buffer_size(1000)  # Buffer size for lidar data

# Define the orbit path
orbit_radius = 2.0
orbit_height = 1.0
lidar_position = core.Vector3(0, 0, orbit_height)
lidar_rotation = 0.0

# Add physics systems
physics.add_system(core.Force())
physics.add_system(core.Torque())
physics.add_system(core.Spring())
physics.add_system(core.Velocity())
physics.add_system(core.Acceleration())

# Link the mesh to a fixed body with a stiff spring
fixed_body = pyc.Body()
fixed_body.set_mass(0)  # Fixed body
fixed_body.set_position(core.Vector3(0, 0, 0))
fixed_body.set_size(core.Vector3(1, 1, 1))  # Set size of the fixed body
fixed_body.set_link(core.Link(fixed_body, body, 0, 0, 0, 0, 0, 0, 0, 1000000.0))  # Stiff spring connection

# Start simulation
simulation = pyc.Simulation(scene, physics, renderer)
simulation.start()

# Define the orbit movement
current_time = 0.0
orbit_speed = 0.5  # Rotation speed in degrees per second

def update_lidar():
    global current_time, lidar_rotation
    current_time += simulation.time_step
    lidar_rotation += orbit_speed * simulation.time_step
    lidar_position.y() = -orbit_radius * np.sin(lidar_rotation)  # Rotate around the X-axis

# Simulation loop
while simulation.running:
    if lidar.data_buffer_size() >= 1000:
        print("Lidar Data:", lidar.get_data_buffer())
    update_lidar()
    simulation.step()
    renderer.look_at(lidar_position)  # Rotate camera to follow lidar

# End simulation
simulation.stop()
pyc.done()
```
