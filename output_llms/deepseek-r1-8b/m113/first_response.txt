body = pychronoshapes.Box(4, 2, 1)
body.set_mass(1500)
body.set_inertia(12, 0, 0)
body.set_com(2, 0.5, 0)

# Create vehicle wheels

wheels = []

    wheel_body = pychrono.Body()

    wheel_body.set_position([1.5, -0.5, 0.5 + 0.3*i])
