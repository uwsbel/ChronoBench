```python
import pychrono as pychrono
import pychrono visualize as pyvis
import pychrono input as pyinput
import pychrono physics as physics

# Initialize PyChrono environment
pychrono.init()

# Create a vehicle model
vehicle = pychrono.objects.RigidBody("Vehicle")
vehicle.set_mass(2000)  # Mass of the truck
vehicle.set_size(pychrono.vector3(4, 6, 2))  # Dimensions of the truck
vehicle.set_position(pychrono.vector3(0, 0, 0))  # Initial position of the truck

# Create TMEASY tire models for each wheel
wheels = []
for i in range(4):
    wheel = pychrono.objects.Tire("TMEASY", "tire_0" + str(i))
    wheel.set_position(pychrono.vector3(1, -1, 0.5))  # Position of each wheel
    wheels.append(wheel)

# Create the terrain (rigid ground)
ground = pychrono.objects.RigidBody("Ground")
ground.set_mass(0)  # Mass is zero for a static object
ground.set_size(pychrono.vector3(100, 10, 10))  # Large enough area
ground.set_friction(1.5)  # High friction for realistic movement
ground.set_position(pychrono.vector3(0, -5, 0))  # Position of the ground

# Add lighting
light_dir = pychrono.objects.DirectionalLight("Sunlight")
light_dir.set_position(pychrono.vector3(10, 10, 10))
light_dir.set_color(pychrono.vector3(1, 1, 1))  # White light
light_amb = pychrono.objects.AmbientLight("Ambient")
light_amb.set_color(pychrono.vector3(0.5, 0.5, 0.5))  # Soft ambient light

# Create skybox
sky = pychrono.objects.Skybox("sky")
sky.set_size(pychrono.vector3(100, 100, 100))
sky.set_position(pychrono.vector3(-50, -50, -50))

# Set up camera
camera = pychrono.objects.PyChronoCamera("ChaseCam")
camera.set_position(pychrono.vector3(0, 0, 20))  # Initial camera position
camera.set_follow(pychrono.objects.find("Vehicle"))  # Follow the vehicle
camera.set_field_of_view(60)  # Field of view for the camera

# Visualization settings
pyvis.settings.set_shadows(True)  # Enable shadows
pyvis.settings.set_antialiasing(True)  # Enable antialiasing for smoother edges

# Set up input controls
pyinput.set_control_mode(pyinput.CONTROL_MODE_KEY_ONLY)
acceleration = pyinput.axis("W", "S")
steering = pyinput.axis("A", "D")
braking = pyinput.button("Space")

# Collision settings
pychrono.objects.set_collision_enabled(True)  # Enable collision detection
pychrono.objects.set_collision_response("Vehicle", True)  # Enable collision response for the vehicle
ground.set_collision_response("Ground", True)  # Enable collision response for the ground

# Set default parameters
vehicle.set_friction(0.3)  # Friction coefficient for the vehicle
for wheel in wheels:
    wheel.set_pressure(350000)  # Tire pressure
    wheel.set_radius(0.4)  # Tire radius
wheel[0].set_initial_angle(0)  # Initial angle for the front wheels
wheel[1].set_initial_angle(-45)  # Initial angle for the front right wheel
wheel[2].set_initial_angle(45)  # Initial angle for the rear right wheel
wheel[3].set_initial_angle(0)  # Initial angle for the rear wheels

# Start simulation
simulation = pychrono.simulation.Simulation()
simulation.add_object(vehicle)
simulation.add_object(ground)
simulation.add_object(light_dir)
simulation.add_object(light_amb)
simulation.add_object(sky)
for wheel in wheels:
    simulation.add_object(wheel)
simulation.add_camera(camera)

# Define the simulation loop
while True:
    # Get input
    speed = pyinput.axis("W", "S")
    angle = pyinput.axis("A", "D")
    brake = pyinput.button("Space")
    
    # Update vehicle controls
    if brake:
        vehicle.set_velocity((0, 0, -5))
    else:
        vehicle.set_velocity((speed * 2, angle * 1, 0))
    
    # Update simulation
    simulation.update(0.1)
    
    # Visualize
    pyvis.render()
```
