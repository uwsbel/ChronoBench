import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Custom Force Functor implementation
class MySpringForce(chrono.ForceFunctor):
    def __init__(self, spring_coef, damping_coef):
        super().__init__()
        self.spring_coef = spring_coef
        self.damping_coef = damping_coef

    def Evaluate(self, body1, body2, frame1, frame2, time):
        # Calculate displacement vector
        pos1 = body1.GetPos(frame1)
        pos2 = body2.GetPos(frame2)
        displacement = pos2 - pos1

        # Calculate velocity vector
        vel1 = body1.GetVelocity(frame1)
        vel2 = body2.GetVelocity(frame2)
        relative_velocity = vel2 - vel1

        # Spring force calculation
        spring_force = self.spring_coef * (displacement.Length() - rest_length)
        spring_force = spring_force * displacement.Normalize()

        # Damping force calculation
        damping_force = -self.damping_coef * relative_velocity.Length()
        damping_force = damping_force * relative_velocity.Normalize()

        # Combine forces
        total_force = spring_force + damping_force

        return total_force

# Simulation parameters
rest_length = 1.5
spring_coef = 50
damping_coef = 1

# Initialize system
sys = chrono.ChSystemNSC()
sys.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, 0))

# Ground setup

ground.SetFixed(True)
ground.EnableCollision(False)

# Visual elements for ground
sph_1 = chrono.ChVisualShapeSphere(0.1)
ground.AddVisualShape(sph_1, chrono.ChFramed(chrono.ChVector3d(-1, 0, 0)))

ground.AddVisualShape(sph_2, chrono.ChFramed(chrono.ChVector3d(1, 0, 0)))

# Body 1 setup
body_1 = chrono.ChBody()
sys.AddBody(body_1)

body_1.SetMass(1)

# Body 2 setup (mirror of body_1)
body_2 = chrono.ChBody()
sys.AddBody(body_2)

body_2.AddVisualShape(sphere_2)

# Create spring_1 with direct coefficients
spring_1 = chrono.ChLinkTSDA()
spring_1.Initialize(body_1, ground, True, chrono.ChVector3d(0, 0, 0), chrono.ChVector3d(-1, 0, 0))
spring_1.SetRestLength(rest_length)
spring_1.SetSpringCoefficient(spring_coef)
spring_1.SetDampingCoefficient(damping_coef)
sys.AddLink(spring_1)

# Create spring_2 with custom force functor
spring_2 = chrono.ChLinkTSDA()
