import pychrono.core as chrono
import pychrono.fea as fea
import pychrono.irrlicht as chronoirr
from pychrono.core import ChLinkNodeFrame

class Model1:
    def __init__(self, system, mesh):
        self.n_chains = 6  # Number of chains, default to 6
        self.chains = []
        
        # Create section properties for the cable
        msection_cable = fea.ChBeamSectionCable()
        msection_cable.SetDiameter(0.015)  # 15 mm diameter
        msection_cable.SetYoungModulus(0.01e9)  # 0.01 GPa Young's modulus
        msection_cable.SetRayleighDamping(0.0001)  # Rayleigh damping
        
        # Create a builder for ANCF cable elements
        builder = fea.ChBuilderCableANCF()
        
        # Generate multiple chains of ANCF beams
        for i in range(self.n_chains):
            # Create a truss body for each chain
            mtruss = chrono.ChBody()
            mtruss.SetFixed(True)  # Fix the truss body
            
            # Create the beam for the current chain
            num_elements = 10 + 5 * i  # Increase number of elements per chain
            beam_start = chrono.ChVector3d(-(i+1)*0.5, 0, -0.1)  # Offset each chain horizontally
            beam_end = chrono.ChVector3d((i+1)*0.5, 0, -0.1)
            
            # Create and initialize the beam
            beam = builder.BuildBeam(
                mesh,
                msection_cable,
                num_elements,
                beam_start,
                beam_end
            )
            
            # Add the beam to the system
            system.Add(beam)
            
            # Apply forces to the front node of the beam
            front_node = beam.GetLastBeamNodes().front()
            front_node.SetForce(chrono.ChVector3d(0, -0.7, 0))
            
            # Create and initialize hinge constraint
            constraint = fea.ChLinkNodeFrame()
            constraint.Initialize(front_node, mtruss)
            system.Add(constraint)
            
            # Add the truss to the system
            system.Add(mtruss)
            
            # Add visualization for the current chain
            self.chains.append(beam)
        
        # Add visualization for all nodes and beams
        self.visualize_beams = []
        self.visualize_nodes = []
        
        for chain in self.chains:
            # Visualize the beam
            visualize_beam = chrono.ChVisualShapeFEA(mesh)
            visualize_beam.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_ELEM_BEAM_MZ)
            visualize_beam.SetColorscaleMinMax(-0.4, 0.4)
            visualize_beam.SetSmoothFaces(True)
            visualize_beam.SetWireframe(False)
            mesh.AddVisualShapeFEA(visualize_beam)
            self.visualize_beams.append(visualize_beam)
            
            # Visualize the nodes
            visualize_nodes = chrono.ChVisualShapeFEA(mesh)
            visualize_nodes.SetFEMglyphType(chrono.ChVisualShapeFEA.GlyphType_NODE_DOT_POS)
            visualize_nodes.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_NONE)
            visualize_nodes.SetSymbolsThickness(0.006)
            visualize_nodes.SetSymbolsScale(0.01)
            visualize_nodes.SetZbufferHide(False)
            mesh.AddVisualShapeFEA(visualize_nodes)
            self.visualize_nodes.append(visualize_nodes)
        
        # Implement PrintBodyPositions function
        def PrintBodyPositions():
            for chain in self.chains:
                end_body = chain.GetLastBeamNodes().back().GetNode()
                pos = end_body.GetPosition()
                print(f"Chain {chain.GetIndex()} end body position: {pos}")
        
    def __del__(self):
        pass

# Initialize the physical system and mesh container
sys = chrono.ChSystemSMC()
mesh = fea.ChMesh()

# Create the model and add the mesh to the system
model = Model1(sys, mesh)
sys.Add(mesh)

# Add visualization for the FEM mesh
visualize = chrono.ChVisualShapeFEA(mesh)

    print("Using MINRES solver")

solver.EnableWarmStart(True)
