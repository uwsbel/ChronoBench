### Full Code

```python
import pychrono as pychrono
from pychrono.visualization import PyVisual
import pychrono.physics
from pychrono.render import *
import numpy as np

# Initialize PyChrono
pychrono.init()

# Set up physics and rendering
simulator = pychrono.Simulator()
physics = pychrono.Physics(simulator)
renderer = pychrono.Renderer(simulator)
renderer.setRenderingEngine(IrrlichtRenderer)
renderer.setWindow(800, 600, "Beam Simulation")

# Define beam element
class BeamElement(pychrono.ForceElement):
    def __init__(self, name, E, A, rho):
        super().__init__(name, pychrono.MATERIAL_CABLE)
        self.E = E
        self.A = A
        self.rho = rho

# Define node class
class Node:
    def __init__(self, x, y, z, mass, E, A, rho):
        self.position = pychrono.Vector3(x, y, z)
        self.mass = mass
        self.E = E
        self.A = A
        self.rho = rho

# Create beam
beam = pychrono.objects.Beam()
node1 = Node(0, 0, 0, 0.1, 210e9, 0.001, 7850)
node2 = Node(5, 0, 0, 0.1, 210e9, 0.001, 7850)
beam.add_node(node1)
beam.add_node(node2)
beam.add_element(BeamElement("e1", 210e9, 0.001, 7850))
beam.add_element(BeamElement("e2", 210e9, 0.001, 7850))

# Add hinge constraint at node1
physics.add_constraint(node1, pychrono.RigidBodyFixed())

# Add gravity forces
g = 9.81
for node in beam.nodes:
    force = pychrono.ForceVector(0, -node.mass * g, 0)
    physics.add_force(node, force)

# Visualization setup
scene = renderer.beginScene()
camera = scene.camera
light = scene.addLight(0, 0, 0, 1, 1, 1)
text = scene.addText("Beam Simulation", 50, 50, pychrono.Color(1, 1, 1))

def step():
    physics.step()
    return True

simulation_running = True
while simulation_running:
    step()
    renderer.update()
    
    scene.clear()
    scene.addLight(light)
    
    # Draw beam
    for i in range(len(beam.elements)):
        e = beam.elements[i]
        n1 = beam.nodes[i]
        n2 = beam.nodes[i+1]
        renderer.drawLine(n1.position, n2.position, pychrono.Color(1, 0, 0))
    
    # Draw nodes
    node_positions = [node.position for node in beam.nodes]
    for pos in node_positions:
        renderer.drawSphere(pos, 0.05, pychrono.Color(0, 1, 0))
    
    # Draw forces
    forces = [physics.get_forces(node)[0] for node in beam.nodes]
    for force in forces:
        if force:
            start_pos = beam.nodes[0].position + force/10
            end_pos = start_pos + force/10
            renderer.drawArrow(start_pos, end_pos, pychrono.Color(0, 0, 1))
    
    # Draw labels
    text.position = (50, 50)
    scene.addText("Beam", 25, 25, pychrono.Color(1, 1, 1))
    scene.addText("Nodes", 100, 100, pychrono.Color(1, 1, 1))
    scene.addText("Forces", 200, 100, pychrono.Color(1, 1, 1))
    
    renderer.endScene()

# Add damping for smoother simulation
physics.set_damping(0.1, 0.1)

# Run the simulation
while True:
    step()
    renderer.beginFrame()
    # Visualization code as above
    renderer.endFrame()
```

### Explanation

This code creates a flexible beam with two ANCF cable elements, hinged at the bottom node. Gravity is applied to each node, causing the beam to deform. The simulation runs in a loop, updating physics and rendering the scene using Irrlicht. The beam's deformation and node positions are visualized, along with the forces acting on the nodes. Adjust parameters like gravity, damping, and beam properties to observe different behaviors.
