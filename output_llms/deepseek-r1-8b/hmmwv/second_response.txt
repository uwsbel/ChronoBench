import pychrono.core as chrono
import pychrono.irrlicht as irr
import pychrono.vehicle as veh
import math

chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

# Initial vehicle location and orientation
initLoc = chrono.ChVector3d(0, 0, 0.5)
initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# Visualization type for vehicle parts (PRIMITIVES, MESH, or NONE)
vis_type = veh.VisualizationType_PRIMITIVES

# Collision type for chassis (PRIMITIVES, MESH, or NONE)
chassis_collision_type = veh.CollisionType_NONE

# Type of tire model (RIGID, TMEASY)
tire_model = veh.TireModelType_TMEASY

# Rigid terrain
terrainHeight = 0      # terrain height
terrainLength = 200.0  # increased from 100.0 to ensure path fits
terrainWidth = 100.0   # size in Y direction

trackPoint = chrono.ChVector3d(-3.0, 0.0, 1.1)

contact_method = chrono.ChContactMethod_NSC

step_size = 1e-3

render_step_size = 1.0 / 50  # FPS = 50

# Create the HMMWV vehicle, set parameters, and initialize
vehicle = veh.HMMWV_Full()
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)

vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetVehicle())

# Path follower implementation
class PathFollower:
    def __init__(self, vehicle):
        self.vehicle = vehicle
        self.path = None
        self.current_pos = self.vehicle.GetInitPosition()
        self.current_rot = self.vehicle.GetInitOrientation()
        self.path_length = 0.0
        self.total_length = 0.0
        self.radius = 5.0  # Circular path radius
        self.front_offset = 2.0  # Offset from center for path following
        
    def SetPath(self, start_pos, end_pos, radius=5.0):
        self.path = []
        self.current_pos = start_pos
        self.current_rot = self.vehicle.GetInitOrientation()
        self.path_length = 0.0
        self.total_length = 0.0
        
        # Generate path points
        for angle in range(0, 360, 1):
            x = self.radius * math.cos(angle * math.pi / 180)
            y = self.radius * math.sin(angle * math.pi / 180)
            pos = chrono.ChCoordsysd(chrono.ChVector3d(x, y, 0), self.current_rot)
            self.path.append(pos)
        self.total_length = self.radius * 2 * math.pi
        
        # Set initial path progress
        self.path_length = self.front_offset
    
    def Update(self, step_size):
        # Calculate progress along the path
        self.path_length += step_size
        
        # Clamp progress within path length
        if self.path_length > self.total_length:
            self.path_length = self.total_length
        elif self.path_length < 0:
            self.path_length = 0
        
        # Get current position and orientation from path
        if self.path_length > 0 and self.path_length < self.total_length:
            self.current_pos = self.path[int((self.path_length / self.total_length) * len(self.path))]
            self.current_rot = self.path[int((self.path_length / self.total_length) * len(self.path))].GetRotation()
        
        # Update vehicle position and orientation
        self.vehicle.SetInitPosition(self.current_pos)
        self.vehicle.SetInitOrientation(self.current_rot)

        vis.EndScene()

    vis.Advance(step_size)
