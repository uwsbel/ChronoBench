import pychrono as chrono
import numpy as np


def create_scm_terrain(system, length, width, resolution, soil_density, shear_modulus, friction_coefficient):
    terrain = chrono.ChSystemSMC()
    terrain.SetGravitationalAcceleration(chrono.ChVector3d(0, -9.81, 0))
    terrain.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

    patch = chrono.ChContinuumSMCSoilPatch()
    patch.SetSoilDensity(soil_density)
    patch.SetSoilShearModulus(shear_modulus)
    patch.SetSoilFrictionCoefficient(friction_coefficient)
    patch.SetSoilRestitutionCoefficient(0.2)
    patch.SetSoilDampingCoefficient(0.05)
    patch.SetSoilThickness(0.1)
    patch.SetSoilViscosityCoefficient(0.05)

    for i in range(resolution):
        for j in range(resolution):
            x = i * length / (resolution - 1)
            y = j * width / (resolution - 1)
            z = 0.0
            patch.AddPoint(chrono.ChVector3d(x, y, z))

    terrain.Add(patch)
    system.Add(terrain)
    return terrain
