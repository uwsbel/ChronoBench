import pychrono as chrono
import numpy as np


def create_ancf_cable_beam(system, num_segments=10, length=2.0, thickness=0.04, density=1000.0, young_modulus=1e6):
    # Create the ANCF cable beam
    beam = chrono.ChBody()
    beam.SetName("ANCF Cable Beam")
    beam.SetFixed(True)
    beam.EnableCollision(False)
    system.Add(beam)

    # Create the ANCF cable elements
    elements = []
    nodes = [chrono.ChNodeACNCable()]
    nodes[0].SetFixed(True)
    nodes[0].SetPos(chrono.ChVector3d(0, 0, 0))
    nodes[0].SetDir(chrono.ChVector3d(1, 0, 0))
    beam.AddNode(nodes[0])

    for i in range(1, num_segments):
        node = chrono.ChNodeACNCable()
        node.SetFixed(False)
        node.SetPos(chrono.ChVector3d(i * length / num_segments, 0, 0))
        node.SetDir(chrono.ChVector3d(1, 0, 0))
        beam.AddNode(node)
        elements.append(chrono.ChElementACNCable(chrono.ChVector3d(0, 0, 0), node.GetPos(), node.GetDir()))
        beam.AddElement(elements[-1])

    # Set material properties
    material = chrono.ChMaterialCompositeACNCFiber()
    material.SetDensity(density)
    material.SetYoungModulus(young_modulus)
    material.SetPoissonRatio(0.3)
    material.SetThickness(thickness)
    for element in elements:
        element.SetMaterial(material)

    return beam, elements, nodes


def add_ground_hinge(system):
    ground = chrono.ChBody()
    ground.SetName("Ground")
    ground.SetFixed(True)
    ground.EnableCollision(False)
    system.Add(ground)
    return ground


def add_gravity(system):
    system.SetGravitationalAcceleration(chrono.ChVector3d(0, -9.81, 0))


def visualize_beam(beam, elements, nodes, system):
    while system.GetChTime() < 2.0:
        system.DoStepDynamics(0.001)
        beam.Update()
        for element in elements:
            element.Update()
        print(f"Time: {system.GetChTime():.3f} | Nodes: {len(nodes)} | Elements: {len(elements)}")
        print(f"Node positions: {[node.GetPos() for node in nodes]}")
        print(f"Element forces: {[element.GetForce() for element in elements]}")
        system.DoStepDynamics(0.001)


def main():
    system = chrono.ChSystemNSC()
    add_ground_hinge(system)
    beam, elements, nodes = create_ancf_cable_beam(system, num_segments=10, length=2.0, thickness=0.04, density=1000.0, young_modulus=1e6)
    add_gravity(system)
    visualize_beam(beam, elements, nodes, system)


if __name__ == "__main__":
    main()
