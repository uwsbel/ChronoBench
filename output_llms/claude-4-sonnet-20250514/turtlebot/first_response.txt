Here's a complete PyChrono simulation script that creates a Turtlebot robot on rigid terrain with the specifications you requested:

```python
import pychrono as chrono
import pychrono.robot as turtlebot
import pychrono.irrlicht as chronoirr
import math
import time

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))  # Set gravity

# Set solver settings for better stability
system.SetSolverMaxIterations(150)
system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)

# Create ground body (rigid terrain)
ground = chrono.ChBodyEasyBox(20, 20, 0.2, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, 0, -0.1))
ground.SetBodyFixed(True)
ground.SetName("ground")

# Set ground material properties
ground_material = chrono.ChMaterialSurfaceNSC()
ground_material.SetFriction(0.7)
ground_material.SetRestitution(0.1)
ground.SetMaterialSurface(ground_material)

# Add ground texture/color
ground.GetVisualShape(0).SetColor(chrono.ChColor(0.5, 0.5, 0.5))
system.Add(ground)

# Initialize Turtlebot robot
# Set initial position and orientation
init_pos = chrono.ChVectorD(0, 0, 0.1)  # Slightly above ground
init_rot = chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 0, 1))  # No initial rotation

# Create Turtlebot robot
robot = turtlebot.TurtleBot(system, init_pos, init_rot)

# Set Turtlebot parameters
robot.Initialize()

# Create real-time visualization with Irrlicht
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Turtlebot Simulation")
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(3, 3, 1.5), chrono.ChVectorD(0, 0, 0.5))
vis.AddTypicalLights()

# Set camera to follow the robot
camera = vis.GetActiveCamera()

# Simulation parameters
time_step = 0.01
simulation_time = 0.0
total_time = 20.0  # Total simulation time in seconds

# Motor control parameters
left_motor_speed = 0.0
right_motor_speed = 0.0

# Control timing parameters
turn_duration = 2.0  # Duration of each turn in seconds
straight_duration = 3.0  # Duration of straight movement

print("Starting Turtlebot simulation...")
print("Controls: Robot will perform automated movements")

# Main simulation loop
while vis.Run() and simulation_time < total_time:
    # Motor control logic based on time
    current_phase = simulation_time % (turn_duration + straight_duration)
    
    if simulation_time < 3.0:
        # Move straight forward
        left_motor_speed = 2.0
        right_motor_speed = 2.0
    elif simulation_time < 5.0:
        # Turn left
        left_motor_speed = 0.5
        right_motor_speed = 2.5
    elif simulation_time < 8.0:
        # Move straight
        left_motor_speed = 2.0
        right_motor_speed = 2.0
    elif simulation_time < 10.0:
        # Turn right
        left_motor_speed = 2.5
        right_motor_speed = 0.5
    elif simulation_time < 13.0:
        # Move straight
        left_motor_speed = 2.0
        right_motor_speed = 2.0
    elif simulation_time < 15.0:
        # Turn left again
        left_motor_speed = 0.5
        right_motor_speed = 2.5
    else:
        # Final straight movement
        left_motor_speed = 2.0
        right_motor_speed = 2.0
    
    # Apply motor speeds to the Turtlebot
    robot.SetMotorSpeed(left_motor_speed, turtlebot.LEFT)
    robot.SetMotorSpeed(right_motor_speed, turtlebot.RIGHT)
    
    # Update camera to follow robot
    robot_pos = robot.GetChassis().GetPos()
    robot_rot = robot.GetChassis().GetRot()
    
    # Set camera position relative to robot
    camera_offset = chrono.ChVectorD(2, 2, 1.5)
    camera_pos = robot_pos + camera_offset
    vis.GetActiveCamera().setPosition(chronoirr.vector3df(
        camera_pos.x, camera_pos.y, camera_pos.z))
    vis.GetActiveCamera().setTarget(chronoirr.vector3df(
        robot_pos.x, robot_pos.y, robot_pos.z + 0.3))
    
    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    
    # Advance the simulation
    system.DoStepDynamics(time_step)
    simulation_time += time_step
    
    # Optional: Print robot status every second
    if int(simulation_time * 10) % 10 == 0:
        print(f"Time: {simulation_time:.1f}s, Robot position: ({robot_pos.x:.2f}, {robot_pos.y:.2f}, {robot_pos.z:.2f})")

print("Simulation completed!")
```

This simulation script includes all the requested features:

1. **PyChrono Environment Initialization**: 
   - Sets up the Chrono system with NSC (Non-Smooth Contact) method
   - Configures gravity and solver settings

2. **Physical Systems and Objects**:
   - Creates a rigid ground body (terrain) with appropriate material properties
   - Initializes a Turtlebot robot at a specified position and orientation

3. **Default Parameters**:
   - Sets initial position slightly above ground to prevent collision
   - Configures material properties for friction and restitution
   - Sets up motor control parameters

4. **Real-time Visualization with Irrlicht**:
   - Creates a visualization window with appropriate size and title
   - Sets up camera positioning and lighting
   - Implements camera following behavior

5. **Simulation Loop with Motor Control**:
   - Implements time-based motor control for left and right turns
   - Updates robot position and camera at each timestep
   - Renders the scene continuously

The robot will perform a sequence of movements including straight lines, left turns, and right turns over a 20-second simulation period. The camera follows the robot throughout the simulation, and status information is printed periodically.

To run this script, make sure you have PyChrono installed with the robot and Irrlicht modules enabled.
