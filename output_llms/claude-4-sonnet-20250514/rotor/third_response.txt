class ChFunctionMyFun(chrono.ChFunction):
    def __init__(self):
        super(ChFunctionMyFun, self).__init__()
        # Define parameters for the complex motor function
        self.A1 = 10.0      # Amplitude 1
        self.A2 = 5.0       # Amplitude 2
        self.T1 = 2.0       # Time threshold 1
        self.T2 = 5.0       # Time threshold 2
        self.T3 = 8.0       # Time threshold 3
        self.w = 2.0        # Angular frequency parameter
    
    def GetVal(self, x):
        # Piecewise function based on time x
        if x < self.T1:
            return self.A1 * m.sin(self.w * x)
        elif x < self.T2:
            return self.A2 * m.cos(self.w * x) + self.A1 * 0.5
        elif x < self.T3:
            return self.A1 * m.exp(-0.1 * (x - self.T2)) * m.sin(self.w * x)
        else:
            return self.A2 * m.sin(self.w * x * 0.5) + self.A1 * 0.3
