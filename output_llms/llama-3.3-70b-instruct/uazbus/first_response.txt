```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the PyChrono environment and core components
sys = chrono.ChSystemNSC()

# Create the UAZBUS vehicle
ua_zbus = veh.UAZBUS()
ua_zbus.Initialize(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngX(chrono.CH_C_PI_2)))
ua_zbus.Suspend(0.1)
ua_zbus.AddDriver()
sys.Add(ua_zbus)

# Create the rigid terrain
terrain = chrono.ChBodyEasyBox(sys, 10, 1, 10, 1000, True, True)
terrain.SetPos(chrono.ChVectorD(0, -1, 0))
terrain.SetMaterial(chrono.ChMaterialSurfaceNSC())
terrain.GetMaterial().SetFriction(0.9)
terrain.GetMaterial().SetRestitution(0.1)
sys.Add(terrain)

# Set default parameters
ua_zbus.SetChassisFixed(False)
ua_zbus.SetSuspensionHeight(0.5)
ua_zbus.SetTireFriction(0.8)
ua_zbus.SetTireRestitution(0.1)

# Create the driver system
driver = veh.ChIrrVehicleDriver(ua_zbus.GetPart(0))
driver.Set SteeringGain(2)
driver.SetThrottleGain(2)
sys.Add(driver)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AddSkyBox()
vis.AddTypicalLights()
vis.AddCamera(chrono.ChVectorD(0, 5, 10))
vis.AddTypicalLights()
vis.SetWindowSize(1024, 768)
vis.SetCameraDistance(10)

# Simulation loop
time = 0
dt = 0.01
while time < 10:
    # Advance the simulation
    sys.DoStepDynamics(dt)
    
    # Update the visualization
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()
    
    # Update the driver system
    driver.Synchronize(time)
    
    # Advance time
    time += dt

    # Synchronize with real-time
    chrono.ChRealtimeStep(sys)

# Clean up
sys.Delete()
```

This script generates a PyChrono simulation featuring a UAZBUS vehicle with specified initial conditions and visualization settings. It sets up a rigid terrain with defined friction and restitution, initializes a driver system for vehicle control, and creates a real-time visualization using Irrlicht with camera settings and lighting. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.
