Here's a Python script that creates a PyChrono simulation of an epicyclic gear system with the specified components and visualizes it using the Irrlicht visualization system:

```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.engine as chronoengine

# Initialize the PyChrono environment
sys = chrono.ChSystemNSC()

# Create the ground body (fixed truss)
ground = chrono.ChBodyEasyBox(sys, 10, 1, 10, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -5, 0))
ground.SetBodyFixed(True)

# Create the rotating bar
bar = chrono.ChBodyEasyBox(sys, 5, 1, 1, 1000, True, True)
bar.SetPos(chrono.ChVectorD(0, 0, 0))
bar.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI_2))

# Create the gear motor
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(bar, ground, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 0, 1), chrono.ChVectorD(0, 0, 1))
motor.SetSpeed(1)  # Set the rotation speed

# Create the first gear
gear1 = chrono.ChBodyEasyCylinder(sys, 2, 1, 1000, True, True)
gear1.SetPos(chrono.ChVectorD(-3, 0, 0))
gear1.SetRot(chrono.Q_from_AngZ(chrono.CH_C_PI_2))

# Create the second gear
gear2 = chrono.ChBodyEasyCylinder(sys, 1, 1, 1000, True, True)
gear2.SetPos(chrono.ChVectorD(3, 0, 0))
gear2.SetRot(chrono.Q_from_AngZ(-chrono.CH_C_PI_2))

# Create the contact material
mat = chrono.ChMaterialSurfaceNSC()
mat.SetFriction(0.5)
mat.SetDampingF(0.5)
mat.SetCompliance(0.0001)
mat.SetCoulombFriction(0.5)

# Add the contact material to the gears
gear1.AddAsset(mat)
gear2.AddAsset(mat)

# Create the contact between the gears
contact = chrono.ChLinkMateGeneric()
contact.Initialize(gear1, gear2, False, False, False)
contact.SetMaterial(mat)

# Add the bodies to the system
sys.Add(ground)
sys.Add(bar)
sys.Add(gear1)
sys.Add(gear2)
sys.Add(motor)
sys.Add(contact)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AddSkyBox()
vis.AddTypicalLights()

# Add the bodies to the visualization system
vis.AddCamera(chrono.ChVectorD(0, 5, -10))
vis.AddTypicalCameraMovements()

# Set the visualization parameters
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Epicyclic Gear System")
vis.SetLogo("")

# Create the simulation loop
while vis.Run():
    sys.DoStepDynamics(0.01)
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()
```

This script creates a PyChrono simulation of an epicyclic gear system with a fixed truss, a rotating bar, and two gears. The gear motor enforces a constant rotation speed. The simulation is visualized using the Irrlicht visualization system with a skybox, typical lights, and a camera. The simulation loop runs until the visualization system is closed.
