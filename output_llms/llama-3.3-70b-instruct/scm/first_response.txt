```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the PyChrono environment
sys = chrono.ChSystemNSC()

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AddSkyBox()
vis.AddTypicalLights()

# Create the vehicle
hmmwv = veh.HMMWV()
hmmwv.SetChassisFixed(False)
hmmwv.SetChassisPosition(chrono.ChVectorD(0, 1, 0))
hmmwv.SetChassisOrientation(chrono.Q_from_AngX(chrono.CH_C_PI_2))
hmmwv.AddRigidTire(chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(1, 0, 0))
hmmwv.AddRigidTire(chrono.ChVectorD(2, 0, 0), chrono.ChVectorD(1, 0, 0))
hmmwv.AddRigidTire(chrono.ChVectorD(0, 0, 2), chrono.ChVectorD(1, 0, 0))
hmmwv.AddRigidTire(chrono.ChVectorD(2, 0, 2), chrono.ChVectorD(1, 0, 0))

# Add mesh visualization to all vehicle components
for part in hmmwv.GetParts():
    vis.AddMesh(part)

# Create the SCM terrain
terrain = veh.SCMDeformableTerrain()
terrain.SetSoilParameters(1000, 0.1, 0.5)
terrain.EnableMovingPatch(True)
terrain.SetMovingPatchSize(10, 10)
terrain.SetSinkageVisualization(True, False)

# Add the vehicle and terrain to the system
sys.Add(hmmwv)
sys.Add(terrain)

# Set up the interactive driver system
driver = veh.InteractiveDriver()
driver.SetVehicle(hmmwv)
driver.SetSteeringRange(-chrono.CH_C_PI_4, chrono.CH_C_PI_4)
driver.SetThrottleRange(0, 1000)
driver.SetBrakingRange(0, 1000)

# Set up the simulation
sim = chrono.ChSimulation()
sim.SetSystem(sys)
sim.SetTimestep(0.02)
sim.SetMaxIters(10)

# Run the simulation
while sim.GetTime() < 10:
    sim.DoStepDynamics(0.02)
    vis.DrawAll()
    vis.PollEvents()
    vis.RunSimulation()

    # Update vehicle dynamics
    driver.Update()

    # Render the scene
    vis.Render()

    # Update the simulation time
    sim.Advance(0.02)

    # Limit the frame rate to 50 FPS
    chrono.ChSleep(0.02)
```

This script initializes the PyChrono environment, creates a full HMMWV vehicle with specific parameters, and sets up an interactive driver system. It also creates an SCM deformable terrain with custom soil parameters and enables a moving patch feature that dynamically follows the vehicle chassis. The simulation runs in real time, updating vehicle dynamics and rendering the scene at 50 frames per second. The terrain visualizes sinkage with false color plotting.
