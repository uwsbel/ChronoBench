```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the PyChrono environment and core components
sys = chrono.ChSystemNSC()

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(1024, 768)
vis.AddSkyBox()
vis.AddTypicalLights()

# Create the vehicle
vehicle = veh.ChVehicle()

# Set vehicle parameters
vehicle.SetContactMaterial(chrono.ChMaterialSurfaceNSC())
vehicle.SetChassisFixed(False)
vehicle.SetChassisMass(5000)
vehicle.SetChassisInertia(chrono.ChVectorD(1000, 1000, 1000))

# Set suspension parameters
vehicle.SetSuspensionSpringStiffness(50000)
vehicle.SetSuspensionDamping(5000)

# Set steering parameters
vehicle.SetSteeringMaxAngle(30 * chrono.CH_C_PI / 180)

# Set tire parameters
tire = veh.ChTireFiala()
tire.SetRimRadius(0.3)
tire.SetTreadWidth(0.2)
vehicle.SetTire(tire)

# Add the vehicle to the system
sys.Add(vehicle.GetChassisBody())

# Set the vehicle's initial position and orientation
vehicle.GetChassisBody().SetPos(chrono.ChVectorD(0, 0, 1))
vehicle.GetChassisBody().SetRot(chrono.Q_from_AngX(chrono.CH_C_PI / 4))

# Create the rigid terrain
terrain = chrono.ChBodyEasyBox(sys, 100, 100, 1, 1000, True, True)
terrain.SetPos(chrono.ChVectorD(0, -1, 0))
terrain.SetMaterial(chrono.ChMaterialSurfaceNSC())

# Add the terrain to the system
sys.Add(terrain)

# Set the terrain texture
vis.AddTexture(terrain, "terrain_texture.png")

# Create the interactive driver system
driver = veh.ChDriver()
driver.SetVehicle(vehicle)
driver.SetSteeringMaxAngle(30 * chrono.CH_C_PI / 180)
driver.SetThrottleMaxValue(1000)
driver.SetBrakingMaxValue(1000)

# Set the camera to follow the vehicle
vis.SetCamera(chrono.ChVectorD(0, 0, 5), chrono.ChVectorD(0, 0, 0), 1000)

# Simulation loop
while vis.Run():
    # Update the vehicle dynamics
    sys.DoStepDynamics(0.02)

    # Update the visualization
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()

    # Limit the simulation to 50 frames per second
    chrono.ChSleep(0.02)
```

This script simulates a CityBus vehicle on a rigid terrain using PyChrono and Irrlicht for visualization. The vehicle is initialized with parameters including location, orientation, and tire model. A combination of mesh and primitive visualization types is used for different vehicle parts, and the camera is set to follow the vehicle from a specified position. The simulation loop runs at 50 frames per second, updating the vehicle dynamics and rendering the scene in real time. The interactive driver system allows for control of the vehicle's steering, throttle, and braking.
