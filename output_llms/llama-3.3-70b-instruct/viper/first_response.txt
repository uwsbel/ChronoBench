```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the PyChrono environment and core components
sys = chrono.ChSystemNSC()

# Set the gravity
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the ground body
ground = chrono.ChBodyEasyBox(sys, 10, 1, 10, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -1, 0))
ground.SetBodyFixed(True)

# Initialize the Viper rover
viper = veh.Viper()
viper.Initialize(sys)

# Set the initial position of the rover
viper.SetChassisPos(chrono.ChVectorD(0, 0.5, 0))
viper.SetChassisRot(chrono.Q_from_AngX(0))

# Create the driver system
driver = veh.ChIrrlichtDriver()
driver.SetVehicle(viper)
driver.SetTrack(chrono.ChVectorD(0, 0, 0))
driver.SetCamera(chrono.ChVectorD(0, 2, -5), chrono.ChVectorD(0, 1, 0))

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AddTypicalLights()
vis.AddSkyBox()
vis.AddCamera(driver.GetCamera())
vis.AddTypicalLights()

# Set the simulation parameters
time_step = 0.01
steering_time = 10
steering_angle = 0.5

# Simulation loop
time = 0
while time < 20:
    # Update the steering angle
    if time < steering_time:
        viper.SetSteering(chrono.ChVectorD(steering_angle * time / steering_time, 0, 0))
    else:
        viper.SetSteering(chrono.ChVectorD(steering_angle, 0, 0))

    # Update the simulation
    sys.DoStepDynamics(time_step)

    # Render the visualization
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()

    # Increment the time
    time += time_step

    # Print the current time
    print(f"Time: {time:.2f} s")

# Finalize the simulation
vis.GetDevice().run()
vis.GetDevice().drop()
```

This script initializes the PyChrono environment, sets up the Viper rover on a rigid terrain, and creates a real-time visualization using Irrlicht. The simulation loop updates the rover's steering angle over a specified time period and renders the visualization at each timestep. The simulation runs for 20 seconds, and the current time is printed to the console at each timestep.
