```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the PyChrono environment and core components
system = chrono.ChSystemNSC()

# Set the default contact material
contact_material = chrono.ChMaterialSurfaceNSC()
contact_material.SetFriction(0.9)
contact_material.SetRestitution(0.8)
contact_material.SetYoungModulus(1e8)
system.SetDefaultContactMaterial(contact_material)

# Create the M113 vehicle
vehicle = veh.M113()
vehicle.SetChassisFixed(False)
vehicle.SetPosition(chrono.ChVectorD(0, 0, 1.5))
vehicle.SetRotation(chrono.Q_from_AngX(chrono.CH_C_PI_2))
system.Add(vehicle.GetChassisBody())

# Create the rigid terrain
terrain = veh.RigidTerrain(system)
terrain.SetPlane(chrono.ChPlane(chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 0, 1)))
system.Add(terrain.GetGroundBody())

# Set the vehicle's initial conditions
vehicle.Initialize(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 1.5), chrono.Q_from_AngX(chrono.CH_C_PI_2)))

# Create the driver system
driver = veh.ChIrrlichtDriver()
driver.SetVehicle(vehicle)
driver.SetTerrain(terrain)
driver.SetSteering(0)
driver.SetThrottle(0)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(1024, 768)
vis.AddSkyBox()
vis.AddTypicalLights()

# Create the camera
camera = vis.AddCamera(chrono.ChVectorD(0, 0, 5))
camera.SetLookAt(chrono.ChVectorD(0, 0, 0))

# Set the visualization settings
vis.SetLogo("PyChrono")
vis.SetWindowTitle("M113 Vehicle Simulation")

# Simulation loop
time_step = 0.01
end_time = 10
while system.GetChTime() < end_time:
    # Synchronize the vehicle, terrain, driver, and visual system
    vehicle.Synchronize(time_step)
    terrain.Synchronize(time_step)
    driver.Synchronize(time_step)
    vis.Synchronize(time_step)

    # Advance the simulation
    system.DoStepDynamics(time_step)

    # Update the visualization
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()

    # Check for keyboard input
    if vis.IsKeyboardHit():
        key = vis.GetKeyboardChar()
        if key == 'q':
            break

# Clean up
vis.GetDevice().run()
vis.GetDevice().drop()
```

This script initializes the PyChrono environment and core components, sets up the M113 vehicle and rigid terrain, and creates a driver system and Irrlicht visualization. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution. The simulation can be controlled using keyboard input, and the visualization can be customized using various settings and options.
