```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the PyChrono environment
sys = chrono.ChSystem()

# Create the Kraz vehicle
vehicle = veh.ChVehicleKraz()
vehicle.SetContactMaterial(chrono.ChMaterialSurfaceNSC())
vehicle.SetChassisFixed(False)
vehicle.SetSuspensionEnabled(True)
vehicle.SetSteeringEnabled(True)
vehicle.SetEngineEnabled(True)

# Set initial conditions for the vehicle
vehicle.SetChassisPosition(chrono.ChVectorD(0, 1, 0))
vehicle.SetChassisOrientation(chrono.Q_from_AngX(chrono.CH_C_PI_2))

# Add the vehicle to the system
sys.Add(vehicle)

# Create a rigid terrain
terrain = chrono.ChBodyEasyBox(sys, 10, 1, 10, 1000, True, True)
terrain.SetPosition(chrono.ChVectorD(0, -1, 0))
terrain.SetMaterial(chrono.ChMaterialSurfaceNSC())

# Set friction and restitution for the terrain
terrain.GetMaterial().SetFriction(0.8)
terrain.GetMaterial().SetRestitution(0.5)

# Add the terrain to the system
sys.Add(terrain)

# Create a driver system for vehicle control
driver = veh.ChIrrlichtDriver()
driver.SetVehicle(vehicle)
driver.SetSteeringGain(1.0)
driver.SetThrottleGain(1.0)
driver.SetBrakingGain(1.0)

# Create a real-time visualization using Irrlicht
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AddSkyBox()
vis.AddTypicalLights()
vis.AddCamera(chrono.ChVectorD(0, 5, -10))
vis.AddCameraTerrain(chrono.ChVectorD(0, 1, 0))
vis.SetWindowSize(1024, 768)
vis.SetLogo("")

# Create a simulation loop
time = 0
dt = 0.01
while time < 10:
    # Synchronize and advance the vehicle, terrain, driver, and visual system
    vehicle.Synchronize(dt)
    terrain.Synchronize(dt)
    driver.Synchronize(dt)
    vis.Synchronize(dt)
    
    # Advance the simulation
    sys.DoStepDynamics(dt)
    
    # Update the visualization
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()
    
    # Increment time
    time += dt

# Clean up
sys = None
vehicle = None
terrain = None
driver = None
vis = None
```

This code sets up a PyChrono simulation with a Kraz vehicle, a rigid terrain, and a driver system for vehicle control. The simulation features real-time visualization using Irrlicht with camera settings and lighting. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.
