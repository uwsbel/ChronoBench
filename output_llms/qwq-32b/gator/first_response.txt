```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# =============================================================================
# SYSTEM SETUP
# =============================================================================
my_system = chrono.ChSystem()
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Gravity

# =============================================================================
# VEHICLE SETUP
# =============================================================================
my_gator = veh.Gator()

# Vehicle position and orientation
my_gator.SetChassisPos(chrono.ChVectorD(0, 0.1, 0))  # Slightly above terrain
my_gator.SetChassisRot(chrono.Q_from_Euler_123(0, 0, 0))  # Default orientation

# Configure TMEasy tires for all wheels
for axle in my_gator.GetAxles():
    for wheel in axle.GetWheels():
        tire = veh.tire.TMEasy()
        wheel.SetTire(tire)

# Initialize vehicle with mesh visualization
my_gator.Initialize(my_system,               # System
                   False,                   # Disable chassis collision
                   veh.VisualizationType.MESH)

# =============================================================================
# TERRAIN SETUP
# =============================================================================
terrain = veh.RigidTerrain(my_system)

# Terrain material properties
patch_material = chrono.ChMaterialSurfaceNSC()
patch_material.SetFriction(0.9)  # Adjust as needed

# Create flat terrain patch
patch = terrain.AddPatch(patch_material,
                        chrono.ChVectorD(0, 1, 0),  # Normal vector pointing up
                        chrono.ChVectorD(0, 0, 0),  # Center position
                        100, 100)                   # Length and width (meters)

# Apply custom texture (replace with actual path)
texture_path = 'custom_terrain_texture.jpg'
patch.SetTexture(texture_path, 200, 200)  # Texture scale in U/V direction

terrain.Initialize()

# =============================================================================
# VISUALIZATION SETUP
# =============================================================================
my_app = irr.ChWheeledVehicleIrrApp(my_gator.GetVehicle(), 
                                   'Gator Simulation', 
                                   True)  # Enable camera smooth tracking

# Configure camera
my_app.SetChaseCamera(my_gator.GetChassis(), 6.0, 0.5)  # Distance and height
my_app.SetTimestep(0.02)  # 50 FPS (1/50 = 0.02 seconds)

# Initialize visualization
my_app.Initialize()
my_app.AddLightDirectional()

# Setup driver controls
driver = veh.ChIrrGuiDriver(my_app)
my_gator.SetDriver(driver)

# =============================================================================
# SIMULATION LOOP
# =============================================================================
while my_app.GetDevice().run():
    my_app.BeginScene()
    my_app.Render()
    my_app.EndScene()
    
    # Update simulation
    my_system.DoStepDynamics(my_app.GetTimestep())
    
    # Check for exit condition
    if my_app.GetDevice().hasQuitted():
        break

my_app.GetDevice().closeGraphicalWindow()
```
