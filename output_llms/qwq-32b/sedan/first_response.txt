Here's a PyChrono simulation script implementing the requested features:

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# Create Chrono system
my_system = chrono.ChSystem()

# Create Irrlicht visualization
app = irr.ChIrrApp(my_system, 'BMW E90 Sedan Simulation', irr.CHIRRFLAGS_USE_JOYSTICK | irr.CHIRRFLAGS_ALLOW_QUIT)
app.SetGpuShaderMode(irr.ChIrrApp.GPU_SHADER_LIT)
app.AddTypicalLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
app.AddTypicalSky()
app.AddTypicalLights()
app.SetChaseCamera(chrono.ChVectorD(0, 0, 1), 6.0)  # Initial camera settings

# Create terrain
terrain = veh.ChTerrain(my_system)
terrain.SetTexture(chrono.GetChronoDataFile('textures/concrete.jpg'), 20, 20)
terrain.SetColor([0.8, 0.8, 0.8])
terrain.Initialize()
app.AssetBind(terrain)
app.AssetUpdate(terrain)

# Create vehicle
vehicle = veh.CreateVehicle(my_system, veh.ChTireModelType.TMEASY)
vehicle.SetChassisVisualizationType(veh.VisualizationType_TEXTURED_MESH)
vehicle.SetTireVisualizationType(veh.VisualizationType_MESH)
vehicle.SetChassisCollisionType(veh.CollisionType_CONVEX_HULL)
vehicle.Initialize(chrono.ChVectorD(0, 0, 1), chrono.QUNIT, 1500)

# Create driver
driver = veh.ChDriver()
vehicle.SetDriver(driver)

# Setup chase camera
chase_cam = chrono.ChChaseCamera(my_system, app.GetSceneManager().getActiveCamera())
chase_cam.SetChaseVehicle(vehicle.GetChassis())
chase_cam.SetChaseDistance(10)
chase_cam.SetChaseHeight(1.5)
app.SetChaseCamera(chase_cam)

# Setup lighting and skybox
sky_texture = chrono.GetChronoDataFile('skybox/skybox_posx.jpg')
app.AddSkyBox(sky_texture)
light = irr.ChLight()
light.SetDirectional(True)
light.SetDirection(chrono.ChVectorD(0, 0, -1))
light.SetColor(chrono.ChColor(1, 1, 1))
light.SetIntensity(2)
app.GetSceneManager().addLight(light)

# Simulation parameters
step_size = 0.01

# Main simulation loop
while app.GetDevice().run():
    # Handle driver inputs
    key = app.GetKeyboard()
    
    steering = 0
    if key.isKeyDown(irr.KEY_KEY_LEFT):
        steering = -1
    elif key.isKeyDown(irr.KEY_KEY_RIGHT):
        steering = 1
    driver.SetSteering(steering)
    
    throttle = 0
    if key.isKeyDown(irr.KEY_KEY_UP):
        throttle = 1
    driver.SetThrottle(throttle)
    
    brake = 0
    if key.isKeyDown(irr.KEY_KEY_DOWN):
        brake = 1
    driver.SetBraking(brake)
    
    # Update scene
    app.BeginScene()
    app.DrawAll()
    app.EndScene()
    
    # Advance simulation
    my_system.DoStepDynamics(step_size)
    
    # Update camera
    chase_cam.Synchronize()
    
app.GetDevice().closeGraphicalWindow()
```
