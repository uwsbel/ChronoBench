Here's a PyChrono simulation script implementing the requested features:

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# Initialize core simulation system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))  # Gravity vector

# Create HMMWV vehicle with initial conditions
vehicle = veh.CreateHMMWV(system)
vehicle.SetInitPosition(chrono.ChCoords(chrono.ChVectorD(0, 0, 1), chrono.QUNIT))
vehicle.SetInitSpeed(0)

# Configure deformable terrain with SCM parameters
terrain = veh.SCMDeformableTerrain(system)
terrain.SetSize(20, 20)  # Terrain dimensions (meters)
terrain.SetHeightMapResolution(100, 100)  # Resolution of height map grid
terrain.SetHeightMapPlane(0)  # Flat terrain at z=0
terrain.SetSoilParameters(
    density=1500,          # kg/m³
    shear_modulus=1e6,     # Pa
    cohesion=1e5,          # Pa
    friction_angle=30 * chrono.CH_C_DEG_TO_RAD  # Friction angle in radians
)
vehicle.SetTerrain(terrain)  # Link vehicle to terrain

# Configure driver system for vehicle control
driver = veh.ChDriver()
driver.SetAutomaticTransmission(False)
driver.SetThrottle(0.5)  # Constant throttle for acceleration
driver.SetSteering(0)    # Straight driving
vehicle.SetDriver(driver)

# Initialize visualization with Irrlicht
vis = irr.ChIrrApp(system, 
                  'HMMWV on Deformable Terrain',
                  irr.dimension2du(1280, 720))
vis.AddTypicalLights()  # Add default lighting
vis.AddAllTypicalCamera(True)  # Add camera with FPS controls

# Configure camera settings
vis.SetCameraPosition(chrono.ChVectorD(0, -20, 2))  # Start position
vis.SetCameraDirection(chrono.ChVectorD(0, 1, 0))   # Look direction
camera = vis.GetSceneManager().getActiveCamera()
camera.setFollowMode(True)  # Follow vehicle chassis
camera.setFollowTarget(vehicle.GetChassis())

# Setup visualization for vehicle and terrain
vehicle.SetupIrrlicht(vis, terrain)

# Simulation parameters
dt = 0.01  # Time step (seconds)
max_time = 60.0  # Simulation duration

# Initialize simulation timing
vis.SetTimestep(dt)
vis.SetTryRealTime(True)  # Enable real-time simulation

# Launch visualization window
vis.LaunchApplication()

# Main simulation loop
while vis.Run():
    current_time = system.GetChTime()
    
    if current_time >= max_time:
        break
        
    # Update driver and terrain states
    driver.Synchronize(current_time)
    terrain.Synchronize(current_time)
    
    # Advance simulation step
    system.DoStepDynamics(dt)
    
    # Render visualization
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
```
