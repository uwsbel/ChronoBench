```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
from pychrono.vehicle import FEDAVehicle, RigidTerrain
from pychrono.tire import PacejkaDELTATire

# Simulation settings
gravity = chrono.ChVectorD(0, 0, -9.81)
time_step = 1 / 50.0  # 50 FPS
camera_offset = chrono.ChVectorD(0, -10, 2)  # Follow position behind vehicle

def main():
    # Initialize Chrono system
    my_system = chrono.ChSystemNSC()
    my_system.Set_G_acc(gravity)

    # Create Irrlicht visualization
    application = chronoirr.ChIrrApp(my_system, "FEDA Vehicle Simulation", chronoirr.dimension2du(1024, 768))
    application.AddTypicalLights()
    application.SetSymbolscale(0.01)
    application.SetCameraPosition(camera_offset)
    application.SetFollowMode(True)
    
    # Initialize terrain
    terrain = RigidTerrain(my_system)
    terrain.SetColor([0.8, 0.8, 0.8])
    terrain.SetTexture("path/to/texture.jpg", 5, 5)  # Replace with actual texture path
    terrain.Initialize()

    # Create FEDA vehicle
    vehicle = FEDAVehicle()
    vehicle.SetChassisVisualizationType(chrono.VisualizationType_MESH)
    vehicle.SetWheelVisualizationType(chrono.VisualizationType_MESH)
    vehicle.SetTireType(PacejkaDELTATire())  # Use Pacejka tire model
    
    # Set vehicle initial position and orientation
    init_position = chrono.ChVectorD(0, 0, 1)  # Start above terrain
    init_orientation = chrono.Q_from_Euler123(chrono.ChVectorD(0, 0, 0))
    vehicle.SetInitPosition(init_position)
    vehicle.SetInitOrientation(init_orientation)
    
    # Initialize vehicle with terrain
    vehicle.Initialize(terrain)

    # Create interactive driver
    driver = chrono.ChInteractiveDriver()
    driver.SetMaxSteering(0.5)  # Max steering angle (radians)
    driver.SetMaxThrottle(1.0)
    driver.SetMaxBraking(1.0)
    vehicle.SetDriver(driver)

    # Add visualization assets
    application.Add(vehicle.GetVehicle())
    application.Add(terrain.GetGround())

    # Main simulation loop
    application.SetTimestep(time_step)
    application.Initialize()
    
    while application.GetDevice().run():
        # Update driver inputs
        key = application.GetDevice().getKey()
        
        driver_inputs = driver.GetInputs()
        if key == chronoirr.KEY_KEY_W:
            driver_inputs.m_throttle = 1.0
        elif key == chronoirr.KEY_KEY_S:
            driver_inputs.m_braking = 1.0
        if key == chronoirr.KEY_KEY_LEFT:
            driver_inputs.m_steering = -1.0
        elif key == chronoirr.KEY_KEY_RIGHT:
            driver_inputs.m_steering = 1.0
        else:
            driver_inputs.m_steering = 0.0
            driver_inputs.m_throttle = 0.0
            driver_inputs.m_braking = 0.0
            
        driver.SetInputs(driver_inputs)
        
        # Advance simulation
        application.DoStep()
        
        # Render scene
        application.BeginScene()
        application.DrawAll()
        application.EndScene()

if __name__ == '__main__':
    main()
```
