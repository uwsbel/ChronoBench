import pychrono as chrono
import pychrono.irrlicht as chronoirr

rest_length = 1.5
spring_coef = 50
damping_coef = 1

sys = chrono.ChSystemNSC()
sys.SetGravitationalAcceleration(chrono.ChVectorD(0, 0, 0))

# Ground setup
ground = chrono.ChBody()
sys.AddBody(ground)
ground.SetFixed(True)
ground.EnableCollision(False)

# Add sphere to ground at (-1,0,0)
sph_1 = chrono.ChVisualShapeSphere(0.1)
ground.AddVisualShape(sph_1, chrono.ChFrameD(chrono.ChVectorD(-1, 0, 0)))

# Body 1 setup
body_1 = chrono.ChBody()
sys.AddBody(body_1)
body_1.SetPos(chrono.ChVectorD(-1, -3, 0))
body_1.SetMass(1)
body_1.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
body_1.EnableCollision(False)
body_1.SetFixed(False)

box_1 = chrono.ChVisualShapeBox(1, 1, 1)
box_1.SetColor(chrono.ChColor(0.6, 0, 0))
body_1.AddVisualShape(box_1)

# Body 2 setup
body_2 = chrono.ChBody()
sys.AddBody(body_2)
body_2.SetPos(chrono.ChVectorD(-1, -6, 0))
body_2.SetMass(1)
body_2.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
body_2.EnableCollision(False)
body_2.SetFixed(False)

box_2 = chrono.ChVisualShapeBox(1, 1, 1)
box_2.SetColor(chrono.ChColor(0, 0.6, 0))
body_2.AddVisualShape(box_2)

# Body 3 setup
body_3 = chrono.ChBody()
sys.AddBody(body_3)
body_3.SetPos(chrono.ChVectorD(-1, -9, 0))
body_3.SetMass(1)
body_3.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
body_3.EnableCollision(False)
body_3.SetFixed(False)

box_3 = chrono.ChVisualShapeBox(1, 1, 1)
box_3.SetColor(chrono.ChColor(0, 0, 0.6))
body_3.AddVisualShape(box_3)

# Springs setup
# Between body_1 and body_2
spring_1 = chrono.ChLinkTSDA()
spring_1.Initialize(body_1, body_2, True, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 0, 0))
spring_1.SetRestLength(rest_length)
spring_1.SetSpringCoefficient(spring_coef)
spring_1.SetDampingCoefficient(damping_coef)
sys.AddLink(spring_1)
spring_1.AddVisualShape(chrono.ChVisualShapeSpring(0.05, 80, 15))

# Between body_2 and body_3
spring_2 = chrono.ChLinkTSDA()
spring_2.Initialize(body_2, body_3, True, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 0, 0))
spring_2.SetRestLength(rest_length)
spring_2.SetSpringCoefficient(spring_coef)
spring_2.SetDampingCoefficient(damping_coef)

# Original spring between body_1 and ground (retained)

# Visualization setup
vis = chronoirr.ChVisualSystemIrrlicht()

    vis.BeginScene()
