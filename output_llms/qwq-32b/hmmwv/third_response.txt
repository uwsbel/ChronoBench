# Custom driver class
class MyDriver(veh.ChDriver):
    def __init__(self, delay=0.5):
        veh.ChDriver.__init__(self)
        self.delay = delay

    def Synchronize(self, time):
        t_delayed = time - self.delay

        # Throttle control
        throttle = 0.0
        if t_delayed >= 0.2:
            if t_delayed <= 0.2 + 0.5:
                throttle = (t_delayed - 0.2) / 0.5 * 0.7
            else:
                throttle = 0.7

        # Steering control (sinusoidal starting at real time 2.0)
        start_steering = 2.0 - self.delay
        if t_delayed >= start_steering:
            amplitude = 0.3  # radians
            freq = 1.0  # Hz
            steering = amplitude * math.sin(2 * math.pi * freq * (t_delayed - start_steering))
        else:
            steering = 0.0

        # Braking remains zero
        braking = 0.0

        self.SetThrottle(throttle)
        self.SetSteering(steering)
        self.SetBraking(braking)
