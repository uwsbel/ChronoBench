```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# Create the Chrono system
my_system = chrono.ChSystem()
my_system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

# Create the rigid terrain
terrain = veh.RigidTerrain(my_system)

# Define terrain material properties
material = chrono.ChMaterialSurfaceNSC()
material.SetFriction(0.9)

# Initialize the terrain with dimensions, texture, and position
terrain.Initialize(
    chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT),  # Placement
    material,
    False,  # has_concrete
    20, 20, 0.1,  # length, width, height
    chrono.GetChronoDataFile('textures/concrete.jpg'),  # texture filename
    1.0,  # texture scale
    False,  # grass
    chrono.ChColor(0.4, 0.4, 0.4)  # color
)

# Create the HMMWV vehicle
vehicle = veh.HMMWV(my_system)
vehicle.SetTireModel(veh.TireModelType.TMEASY)

# Initialize the vehicle at position (0,0,1) with default orientation
vehicle.Initialize(
    chrono.ChVectorD(0, 0, 1),  # chassis position
    chrono.ChQuaternionD(1, 0, 0, 0),  # chassis orientation
    terrain.GetCollisionModel()  # terrain collision model
)

# Set visualization to use primitives for all components
vehicle.SetChassisVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetWheelVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetTireVisualizationType(veh.VisualizationType_PRIMITIVES)

# Create the Irrlicht application
my_application = irr.ChIrrApp(my_system, 'HMMWV on Rigid Terrain', irr.dimension2du(1024, 768))
my_application.AddTypicalLights()
my_application.AddTypicalLogo()
my_application.AddTypicalSky()

# Add vehicle and terrain visualization
my_application.AddAsset(vehicle.GetVisualization())
my_application.AddAsset(terrain.GetVisualization())

# Configure simulation loop parameters
my_application.SetTimestep(0.02)  # 50 FPS
my_application.SetRealTimeFactor(1.0)  # real-time simulation

# Initialize interactive driver
driver = veh.InteractiveDriver()
vehicle.SetDriver(driver)

# Run the simulation
my_application.MainLoop()
```
