```python
import pychrono as chrono
import pychrono.vehicle as vehicle
import pychrono.irrlicht as irr

# =============================================================================
# Simulation setup
# =============================================================================
def main():
    # Create the Irrlicht application
    app = irr.ChIrrApp(chrono.GetChronoSystemD(), 
                      "HMMWV on SCM Deformable Terrain", 
                      irr.dimension2du(1280, 720))
    app.SetScheme(chronoIRR::IRR_DEFAULT_SCHEME)
    app.AddTypicalLights(chrono.ChVectorD(3, -3, 3), chrono.ChVectorD(3, 3, 3))
    
    # Set camera position
    app.SetCamera(chrono.ChVectorD(5, -15, 2), chrono.ChVectorD(0, 0, 1))

    # Create the HMMWV vehicle with rigid tires
    hmmwv = vehicle.HMMWV_Vehicle()
    hmmwv.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 1.0)))
    hmmwv.SetInitFwdVector(chrono.ChVectorD(1, 0, 0))
    hmmwv.SetTireType(vehicle.TireModelType.RIGID)
    
    # Create visualization system for the vehicle
    vis_system = vehicle.ChWheeledVehicleVisualSystemIrrlicht()
    vis_system.SetVehicle(hmmwv.GetVehicle())
    vis_system.SetChassisVisualizationType(vehicle.VisualizationType_MESH)
    vis_system.SetWheelVisualizationType(vehicle.VisualizationType_MESH)
    vis_system.SetSuspensionVisualizationType(vehicle.VisualizationType_MESH)
    hmmwv.SetCustomVisualization(vis_system)
    
    # Initialize vehicle systems
    hmmwv.Initialize()
    
    # Create SCM deformable terrain
    terrain = vehicle.ChDeformableTerrainIrrlicht()
    terrain.SetDimensions(50, 50, 0.2)  # X/Y size 50m, Z height 0.2m
    terrain.SetPosition(chrono.ChVectorD(0, 0, 0))
    
    # Configure soil parameters (SCM model)
    soil = vehicle.ChSoil()
    soil.SetShearModulus(5e6)          # G = 5 MPa
    soil.SetCohesion(1e5)              # Cohesion (Pa)
    soil.SetDensity(1500)              # kg/m³
    soil.SetFrictionAngle(chrono.CH_C_PI / 6)  # 30 degrees
    soil.SetDilationAngle(chrono.CH_C_PI / 12) # 15 degrees
    terrain.SetSoilModel(soil)
    
    # Enable moving patch following chassis
    terrain.SetMovingPatch(True)
    terrain.SetMovingPatchSize(10, 10)  # 10m x 10m patch
    
    # Enable false color visualization for sinkage
    terrain.SetVisualizationType(vehicle.TerrainVisualizationType.SINKAGE_COLOR)
    
    # Initialize terrain
    terrain.Initialize()
    
    # Add terrain to Irrlicht scene
    app.GetSceneManager().addCameraSceneNode()
    terrain.AddVisualizationAssets()
    
    # Create driver system
    driver = vehicle.ChIrrGuiDriver(app)
    driver.SetSteeringDelta(0.01)
    driver.SetThrottleDelta(0.01)
    driver.SetBrakingDelta(0.01)
    
    hmmwv.SetDriver(driver)
    
    # Simulation parameters
    time_step = 0.02  # 50 Hz
    app.SetTimestep(time_step)
    
    # Simulation loop
    while app.GetDevice().run():
        # Update driver inputs
        driver.Synchronize(time_step)
        
        # Update terrain moving patch position
        chassis_pos = hmmwv.GetVehicle().GetChassisFrame().GetPos()
        terrain.SetMovingPatchPosition(chassis_pos)
        
        # Advance simulation
        app.DoStep()
        
        # Update visualization
        terrain.Update()
        app.GetSceneManager().setAmbientLight(chrono.ChColor(0.5, 0.5, 0.5))
        app.Render()
        
        # Enforce real-time simulation
        app.GetSystem()->DoStepDynamics(time_step)
        app.EndScene()
        
        # Sleep to maintain frame rate
        app.GetDevice().getTimer().sleep(time_step * 1000)

if __name__ == '__main__':
    main()
```
