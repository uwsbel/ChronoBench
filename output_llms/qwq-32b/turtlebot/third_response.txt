import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as turtlebot
from pychrono import irrlicht as chronoirr
import pychrono.sensor as sens  # Added for sensor manager

# Define wheel IDs as constants
LEFT_DRIVE_WHEEL = 0
RIGHT_DRIVE_WHEEL = 1

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create ground body
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVector3d(0, 0, -0.6))  # Adjusted position
ground.SetFixed(True)
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# Add 5 random boxes
for _ in range(5):
    x = np.random.uniform(-4, 4)
    y = np.random.uniform(-4, 4)
    z = -0.6 + 0.25  # Box height 0.5, placed on ground
    box = chrono.ChBodyEasyBox(0.5, 0.5, 0.5, 1000, True, True, ground_mat)
    box.SetPos(chrono.ChVectorD(x, y, z))
    box.SetFixed(True)
    system.Add(box)

# Create Turtlebot Robot
init_pos = chrono.ChVector3d(0, 0.2, 0)
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)
robot = turtlebot.TurtleBot(system, init_pos, init_rot)
robot.Initialize()

# Create sensor manager and lidar
sensor_mgr = sens.ChSensorManager(system)
robot.GetBody().AddAsset(sensor_mgr)  # Attach to robot's body

lidar = sens.ChLidar()
lidar.SetName("lidar")
lidar.SetRelativePose(chrono.ChFrameD(
    chrono.ChVectorD(0, 0, 0.2),  # Position above robot
    chrono.Q_from_Euler123(chrono.ChVectorD(0, 0, 0))  # Orientation
))
lidar.SetFOVHorizontal(120 * chrono.CH_C_DEG_TO_RAD)
lidar.SetRange(0.1, 10.0)
lidar.SetSampling(0.05, 0.05)
lidar.SetResolution(64, 1024)
sensor_mgr.AddSensor(lidar)

# Add a point cloud filter
filter = sens.ChFilterPointCloud()
lidar.AddFilter(filter)

# Create visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Turtlebot with Sensors and Obstacles')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 3), chrono.ChVector3d(0, 0, 0.2))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0.5), 3, 4, 10, 40, 512)

# Simulation parameters
time_step = 2e-3
time = 0

def move(robot, mode):
    """Control robot movement based on mode."""
    if mode == 'straight':
        robot.SetMotorSpeed(math.pi, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(math.pi, RIGHT_DRIVE_WHEEL)
    elif mode == 'left':
        robot.SetMotorSpeed(0, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(math.pi, RIGHT_DRIVE_WHEEL)
    elif mode == 'right':
        robot.SetMotorSpeed(math.pi, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0, RIGHT_DRIVE_WHEEL)

# Simulation loop
while vis.Run():
    # Control robot movement
    move(robot, 'straight')  # Can change mode based on logic

    # Advance simulation step
    system.DoStepDynamics(time_step)

    # Update sensor manager (after physics step)
    sensor_mgr.Update()

    # Update time
    time += time_step

    # Render scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
