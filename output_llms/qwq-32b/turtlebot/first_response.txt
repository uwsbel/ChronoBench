Here's the PyChrono simulation script implementing the Turtlebot on rigid terrain with real-time visualization and motor control:

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

def main():
    # Create the Chrono system with gravity
    my_system = chrono.ChSystemNSC()
    my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

    # Create ground plane
    ground = chrono.ChBodyEasyBox(20, 1, 20, 1000, True, True)
    ground.SetPos(chrono.ChVectorD(0, -1, 0))
    ground.SetBodyFixed(True)
    my_system.Add(ground)

    # Initialize Turtlebot
    turtlebot = veh.Turtlebot(my_system)
    turtlebot.Initialize(chrono.ChVectorD(0, 0.1, 0), chrono.ChQuaternionD(1,0,0,0), 1)

    # Setup visualization
    vis = irr.ChIrrApp(my_system, 'Turtlebot Simulation', irr.dimension2du(1280, 720))
    vis.AddLightWithShadow(chrono.ChVectorD(0, 5, 5), chrono.ChVectorD(0, 0, 0), 2, 6, 120, 80)
    vis.AddCamera(chrono.ChVectorD(0, 3, 5))
    vis.SetCameraOrientation(chrono.Q_from_Euler_123(chrono.ChVectorD(0, 0, 0)))
    vis.AddAllTypicalHandlers()
    vis.Add(turtlebot.GetVisualization())

    # Simulation parameters
    vis.SetTimestep(0.01)  # Time step for physics simulation

    # Simulation loop
    while vis.Run():
        current_time = my_system.GetChTime()
        
        # Motor control logic
        if 2 <= current_time < 3:
            turtlebot.SetWheelSpeeds(-1, 1)  # Left turn (left wheel backward, right forward)
        elif 4 <= current_time < 5:
            turtlebot.SetWheelSpeeds(1, -1)  # Right turn (left forward, right backward)
        else:
            turtlebot.SetWheelSpeeds(0, 0)  # Stop
        
        # Advance simulation step
        my_system.DoStepDynamics(vis.GetDt())
        
        # Render the scene
        vis.BeginScene()
        vis.Render()
        vis.EndScene()

if __name__ == '__main__':
    main()
```
