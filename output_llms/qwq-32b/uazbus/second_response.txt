vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle('UAZBUS Double Lane Change')
vis.SetWindowSize(1280, 1024)
vis.SetChaseCamera(chrono.ChVectorD(-3.0, 0.0, 1.1), 6.0, 0.5)
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle)

# Create scripted driver
driver = veh.ChWheeledVehicleDriver(vehicle.GetVehicle())

# Simulation loop variables
step_number = 0
render_frame = 0
realtime_timer = chrono.ChRealtimeStepTimer()

while vis.Run():
    time = vehicle.GetSystem().GetChTime()

    # Update driver inputs based on time for double lane change
    steering = 0.0
    throttle = 0.5
    braking = 0.0

    if time < 2:
        steering = 0.0
    elif 2 <= time < 3:
        steering = 0.3  # Left turn
    elif 3 <= time < 4:
        steering = 0.0
    elif 4 <= time < 5:
        steering = -0.3  # Right turn
    elif 5 <= time < 6:
        steering = 0.0
    elif time >= 8:
        throttle = 0.0
        braking = 1.0  # Braking

    driver.SetSteering(steering)
    driver.SetThrottle(throttle)
    driver.SetBraking(braking)
    driver_inputs = driver.GetInputs()

    # Render scene
    if step_number % math.ceil(render_step_size / step_size) == 0:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        render_frame += 1

    # Synchronize and advance modules
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    vis.Synchronize(time, driver_inputs)

    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    step_number += 1
    realtime_timer.Spin(step_size)

print("Simulation completed.")
