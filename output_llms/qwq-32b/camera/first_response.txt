```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import math

# Initialize core system and visualization
my_system = chrono.ChSystemNSC()
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Gravity

vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('PyChrono Camera Orbit Simulation')
vis.SetSymbolsScale(0.01)
vis.SetShadows(True)
vis.SetLightDirection(chrono.ChVectorD(0.5, 0.5, 0.5))
vis.SetLightIntensity(1.5)
vis.SetCameraPosition(chrono.ChVectorD(0, 0, 10))  # Initial main camera position

my_system.SetVisualSystem(vis)

# Create fixed triangular mesh body
mesh_body = chrono.ChBodyEasyStatic()
mesh_body.SetPos(chrono.ChVectorD(0, 0, 0))

# Load mesh (replace with your .obj file path)
mesh = chrono.ChTriangleMeshShape()
mesh.SetName('my_mesh')
mesh.SetMesh(chrono.ChTriangleMesh())
mesh.GetMesh().LoadWavefrontMesh("path/to/your/mesh.obj", False, True)
mesh.GetMesh().FlipNormals()  # Adjust as needed

mesh_body.AddAsset(mesh)
mesh_body.AddAsset(chrono.ChColorAsset(chrono.ChColor(0.7, 0.7, 0.7)))  # Gray color
my_system.Add(mesh_body)

# Setup camera sensor
sensor_mgr = vis.GetSensorManager()

cam = chronoirr.ChCameraSensor()
cam.SetName('orbit_cam')

# Camera parameters
radius = 5.0
height = 2.0
angular_velocity = 0.1  # radians per second
theta = 0.0

cam.SetImageWidth(640)
cam.SetImageHeight(480)
cam.SetFov(60)
cam.SetNearPlane(0.1)
cam.SetFarPlane(100)
cam.SetUp(chrono.ChVectorD(0, 0, 1))

# Add noise filters
cam.AddGaussianNoise(0, 0.01)  # Mean 0, variance 0.01

# Enable visualization
cam.SetVisualizationType(chronoirr.ChVisualizationType.VISUALIZATION_RENDERING)
sensor_mgr.AddSensor(cam)

# Simulation loop
device = vis.GetDevice()
my_system.SetTimestep(0.01)  # Simulation time step

while device.Run():
    # Update camera position and orientation
    theta += angular_velocity * my_system.GetChTimeStep()
    x = radius * math.cos(theta)
    y = radius * math.sin(theta)
    
    cam.SetPosition(chrono.ChVectorD(x, y, height))
    cam.SetLookAtPoint(chrono.ChVectorD(0, 0, 0))  # Always look at origin
    
    # Step simulation
    my_system.DoStepDynamics()
    
    # Update sensors
    sensor_mgr.Update()
    
    # Get and print camera buffer data
    buffer = cam.GetLatestBuffer()
    if buffer:
        print(f"\nFrame: {sensor_mgr.GetFrameCount()}")
        print(f"Buffer dimensions: {buffer.GetWidth()}x{buffer.GetHeight()}")
        
        # Print first pixel RGB values (assuming RGB format)
        if buffer.GetPixelFormat() == chronoirr.ChPixelFormat.RGB:
            pixel = buffer.GetPixel(0, 0)
            print(f"First pixel RGB: {pixel.r}, {pixel.g}, {pixel.b}")
    
    # Render scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
```
