import pychrono as chrono
import pychrono.sensor as sens
import pychrono.irrlicht as chronoirr
import rospy
from std_msgs.msg import Float32MultiArray
from sensor_msgs.msg import Image, PointCloud2
from geometry_msgs.msg import Vector3

def main():
    # Initialize Chrono system
    system = chrono.ChSystemNSC()
    system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Gravity
    
    # Visualization setup
    vis = chronoirr.ChVisualSystemIrrlicht()
    vis.SetWindowSize(1024, 768)
    vis.SetWindowTitle('PyChrono Sensor Simulation')
    vis.SetCameraPosition(chrono.ChVectorD(0, 3, 10))
    vis.SetCameraRotation(chrono.Q_from_AngAxis(chrono.CH_C_PI_2, chrono.VECT_Y))
    system.SetVisualSystem(vis)

    # Create moving ground body
    ground_body = chrono.ChBody()
    system.Add(ground_body)
    ground_body.SetPos(chrono.ChVectorD(0, 0, 0))
    ground_body.SetMass(100)
    ground_body.SetInertiaXX(chrono.ChMatrix33D(1, 0, 0, 0, 1, 0, 0, 0, 1))
    ground_body.SetBodyFixed(False)
    ground_body.SetVelocity(chrono.ChVectorD(1, 0, 0))  # Initial velocity

    # Add visualization mesh to ground body
    box = chrono.ChBoxShape()
    box.GetBoxGeometry().SetLengths(chrono.ChVectorD(10, 10, 1))
    ground_body.AddAsset(box)

    # Initialize sensor manager
    sensor_mgr = sens.ChSensorManager(system, vis)
    sensor_mgr.SetRenderTarget(vis)

    # Create sensors
    # Camera
    camera = sens.ChCameraSensor()
    camera.SetParent(ground_body)
    camera.SetPosition(chrono.ChVectorD(0, 0, 1))
    camera.SetRotation(chrono.Q_from_Euler_123(0, 0, 0))
    camera.SetImageWidth(640)
    camera.SetImageHeight(480)
    camera.SetFov(60)
    sensor_mgr.AddSensor(camera)

    # Lidar
    lidar = sens.ChLidarSensor()
    lidar.SetParent(ground_body)
    lidar.SetPosition(chrono.ChVectorD(0, 0, 1))
    lidar.SetRotation(chrono.Q_from_Euler_123(0, 0, 0))
    lidar.SetHorizontalResolution(0.5)
    lidar.SetVerticalResolution(0.5)
    lidar.SetRange(100)
    sensor_mgr.AddSensor(lidar)

    # Initialize ROS node and publishers
    rospy.init_node('chronosensor_node', anonymous=True)
    pub_camera = rospy.Publisher('/camera/image', Image, queue_size=10)
    pub_lidar = rospy.Publisher('/lidar/points', PointCloud2, queue_size=10)
    pub_gps = rospy.Publisher('/gps/position', Float32MultiArray, queue_size=10)
    pub_accel = rospy.Publisher('/imu/accel', Vector3, queue_size=10)

    time_step = 0.001
    real_time_factor = 1.0
    end_time = 10.0

    while system.GetChTime() < end_time:
        # Advance simulation step
        system.DoStepDynamics(time_step)
        
        # Update sensors
        sensor_mgr.Update()
        
        # Get sensor data and publish via ROS
        # Camera
        cam_data = camera.GetImageData()
        # TODO: Convert ChImage to ROS Image message
        # pub_camera.publish(cam_data)
        
        # Lidar
        lidar_data = lidar.GetPoints()
        # TODO: Convert point cloud to ROS PointCloud2
        # pub_lidar.publish(lidar_data)
        
        # GPS
        pos = ground_body.GetPos()
        gps_msg = Float32MultiArray(data=[pos.x, pos.y, pos.z])
        pub_gps.publish(gps_msg)
        
        # Accelerometer
        acc = ground_body.GetAcc()
        acc_msg = Vector3(x=acc.x, y=acc.y, z=acc.z)
        pub_accel.publish(acc_msg)
        
        # Gyroscope
        ang_vel = ground_body.GetWvel_par()
        gyro_msg = Vector3(x=ang_vel.x, y=ang_vel.y, z=ang_vel.z)
        pub_gyro.publish(gyro_msg)
        
        # Magnetometer (orientation as Euler angles)
        rot = ground_body.GetRot()
        euler = rot.GetEulerAngles(chrono.ChFrameAngles.EULER_XYZ)
        mag_msg = Vector3(x=euler.eul1, y=euler.eul2, z=euler.eul3)
        pub_mag.publish(mag_msg)

        # Visualization update
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        
        # Maintain real-time
        time.sleep(time_step / real_time_factor)

    vis.Close()

if __name__ == '__main__':
