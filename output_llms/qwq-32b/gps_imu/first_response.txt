```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.sensor as sens
import pychrono.irrlicht as irr

# Initialize simulation system
system = chrono.ChSystemShared()

# Create terrain
terrain = veh.RigidTerrain(
    system,
    chrono.ChMaterialSurfaceNSC(),
    veh.ChTerrain()
)
ground = terrain.GetGround()
system.Add(ground)

# Create HMMWV vehicle
my_vehicle = veh.HMMWV(system)
my_vehicle.SetInitPosition(chrono.ChVectorD(0, 0, 1))  # Position above ground
my_vehicle.SetChassisVisualizationType(veh.VisualizationType_MESH)
my_vehicle.SetWheelVisualizationType(veh.VisualizationType_MESH)
my_vehicle.Initialize(terrain.GetCollisionModel(), chrono.ChVectorD(0, 0, 1), chrono.QUNIT)

# Setup driver with constant throttle
driver = my_vehicle.GetDriver()
driver.SetThrottle(0.5)  # Constant forward throttle
driver.SetSteering(0)     # No steering

# Initialize Irrlicht visualizer
vis = irr.ChIrrApp(system, 'HMMWV Simulation', irr.dimension2du(800, 600))
vis.AddTypicalLights()
vis.AddTypicalCamera(chrono.ChVectorD(0, 0, 5), chrono.ChVectorD(0, 0, 0))
vis.Add(ground)
my_vehicle.InitializeGraphics(system)
vis.AssetBind()
vis.AssetUpdate()
vis.SetTimestep(0.01)  # Simulation step size (100Hz)

# Add IMU and GPS sensors to chassis
chassis = my_vehicle.GetChassis()

# IMU Sensor
imu = sens.ChImu(chassis)
imu.SetName("imu")
imu.SetPosition(chrono.ChVectorD(0, 0, 0))
imu.SetUpdateInterval(0)  # Update every step
system.AddSensor(imu)

# GPS Sensor
gps = sens.ChGps(chassis)
gps.SetName("gps")
gps.SetPosition(chrono.ChVectorD(0, 0, 0))
gps.SetUpdateInterval(0)  # Update every step
system.AddSensor(gps)

# Simulation loop
time = 0.0
step_size = vis.GetTimestep()

while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    
    # Update driver and vehicle
    driver.Synchronize(time)
    driver_inputs = driver.GetInputs()
    my_vehicle.Synchronize(time, driver_inputs)
    my_vehicle.Update(time)
    
    # Advance simulation
    system.DoStepDynamics(step_size)
    time += step_size
    
    # Process sensor data
    imu_data = imu.GetReading()
    lin_acc = imu_data.GetAcceleration()
    ang_vel = imu_data.GetAngularVelocity()
    orientation = imu_data.GetOrientation()
    
    gps_data = gps.GetReading()
    position = gps_data.GetPos()
    velocity = gps_data.GetPos_dt()
    
    # Output vehicle mass once
    if time == 0:
        print(f"Vehicle mass: {my_vehicle.GetChassis().GetMass()} kg")
```
