```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono_ros as ros
import rospy
from std_msgs.msg import Float32MultiArray

def main():
    # Initialize Chrono system
    system = chrono.ChSystemNSC()
    system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

    # Create terrain
    terrain = veh.ChTerrain(system)
    material = chrono.ChMaterialSurface()
    material.SetFriction(0.9)
    material.SetRestitution(0.1)
    terrain.AddPatch(chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 1, 0), 100, 100, material)
    terrain.Build()

    # Create HMMWV vehicle
    vehicle = veh.HMMWV.HMMWVVehicle(system)
    vehicle.SetInitPosition(chrono.ChVectorD(0, 0.5, 0))
    vehicle.SetInitFwdVelocity(0)
    vehicle.Initialize(terrain, veh.ChTireContactMethod.SLIP)  # SLIP tire model
    vehicle.SetChassisVisualizationType(veh.VisualizationType.MESH)

    # Configure engine and tires
    engine = vehicle.GetEngine()
    engine.SetMaxPower(300000)  # 300 kW engine power
    
    for axle in vehicle.GetAxles():
        for wheel in axle.wheels:
            tire = wheel.GetTire()
            tire.SetType('pacejka')  # Pacejka tire model

    # Create driver
    driver = chrono.ChDriver()
    vehicle.SetDriver(driver)

    # ROS initialization
    node = ros.ChNodeROS()
    node.Initialize(system, 'hmmwv_simulation', 0.01)  # Node name and simulation step time

    # Register clock synchronization
    node.RegisterClockSync('/clock')

    # ROS Publishers and Subscribers
    def driver_inputs_callback(msg):
        throttle = msg.data[0]
        steering = msg.data[1]
        driver.SetThrottle(throttle)
        driver.SetSteering(steering)

    node.Subscribe('/vehicle/inputs', driver_inputs_callback, Float32MultiArray)

    def publish_vehicle_state():
        chassis = vehicle.GetChassis()
        pos = chassis.GetPos()
        lin_vel = chassis.GetPos_dt()
        ang_vel = chassis.GetWvel_par()
        state_msg = Float32MultiArray(data=[pos.x, pos.y, pos.z,
                                           lin_vel.x, lin_vel.y, lin_vel.z,
                                           ang_vel.x, ang_vel.y, ang_vel.z])
        node.Publish('/vehicle/state', state_msg)

    node.RegisterPublisher('/vehicle/state', publish_vehicle_state, 10)  # 10 Hz

    # Simulation loop
    while not rospy.is_shutdown():
        node.SpinOnce()  # Process incoming messages
        system.DoStepDynamics(node.GetTimeStep())  # Advance simulation
        node.Synchronize()  # Sync simulation time with ROS
        publish_vehicle_state()  # Update and publish state

if __name__ == '__main__':
    rospy.init_node('hmmwv_simulation_node')
    main()
```
