import pychrono as chrono
import pychrono.irrlicht as irr

# Initialize the Chrono system
sys = chrono.ChSystemMyChrono()
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Simulation parameters
beam_length = 2.0  # Length of the beam
num_elements = 10  # Number of elements in the beam
material = chrono.ChMaterialBeam()
material.SetYoungModulus(2e11)  # Young's modulus (steel)
material.SetPoissonRatio(0.3)
material.SetDensity(7850)  # Density of steel (kg/m³)

# Create the IGA FEM beam
beam = chrono.iga.ChIgaFEMBeam()
beam.SetLength(beam_length)
beam.SetWidth(0.1)  # Cross-sectional width
beam.SetHeight(0.1)  # Cross-sectional height
beam.SetNumElements(num_elements)
beam.SetMaterial(material)
sys.Add(beam)

# Add FEM visualization to the beam
beam_vis = chrono.ChVisualizationFEMMesh()
beam_vis.SetFEMMesh(beam.GetMesh())
beam.AddAsset(beam_vis)
beam.AddAsset(chrono.ChColorAsset(chrono.ChColor(0.8, 0.8, 0.8)))  # Grey color

# Create the flywheel at the beam's center
flywheel = chrono.ChBody()
flywheel.SetMass(10.0)  # Mass of the flywheel
flywheel.SetPos(chrono.ChVectorD(beam_length / 2, 0, 0))
flywheel_shape = chrono.ChSphereShape()
flywheel_shape.GetSphereGeometry().rad = 0.1  # Radius of the flywheel
flywheel.AddAsset(flywheel_shape)
flywheel.AddAsset(chrono.ChColorAsset(chrono.ChColor(1, 0, 0)))  # Red color
sys.Add(flywheel)

# Attach the flywheel to the beam's center with a spherical joint
joint = chrono.ChLinkLockSpherical()
joint.Initialize(beam, flywheel, chrono.ChFrameD(chrono.ChVectorD(beam_length / 2, 0, 0)))
sys.Add(joint)

# Create a rotational motor at the beam's end (right end)
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(
    beam,
    chrono.ChFrameD(chrono.ChVectorD(beam_length, 0, 0)),  # Position on the beam
    chrono.ChFrameD()  # Ground frame
)
motor.SetSpeedFunction(chrono.ChFunction.ConstantFunction(10.0))  # 10 rad/s rotational speed
sys.Add(motor)

# Setup Irrlicht visualization
vis = irr.ChIrrApp(sys, 'Jeffcott Rotor Simulation', irr.dimension2du(800, 600))
vis.AddTypicalLights()
vis.AddTypicalCamera(chrono.ChVectorD(beam_length, beam_length, beam_length), chrono.ChVectorD(beam_length / 2, 0, 0))
vis.SetCameraDistance(5)

# Run the simulation
vis.SetTimestep(0.001)
vis.Launch()

while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(vis.GetDt())
