```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# Create the Chrono system
system = chrono.ChSystemSMC()
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

# Create the HMMWV vehicle
vehicle = veh.HMMWV()
vehicle.SetChassisVisualizationType(veh.VisualizationType_MESH)
vehicle.SetWheelVisualizationType(veh.VisualizationType_MESH)
vehicle.SetSuspensionVisualizationType(veh.VisualizationType_MESH)
vehicle.SetTireVisualizationType(veh.VisualizationType_MESH)

# Initialize vehicle at specified location and orientation
init_location = chrono.ChVectorD(0, 0, 1)  # Above terrain
init_orientation = chrono.ChQuaternionD(1, 0, 0, 0)  # Default orientation
vehicle.Initialize(
    system,
    init_location,
    init_orientation,
    veh.ChassisCollisionType.MESH
)

# Configure TMEasy tires
for axle in vehicle.GetAxles():
    for tire in axle.GetTires():
        tm_tire = veh.TMEasyTire(vehicle.GetSystem())
        tm_tire.SetRadius(tire.GetRadius())
        tm_tire.SetWidth(tire.GetWidth())
        tm_tire.SetLongitudinalStiffness(1e5)
        tm_tire.SetLateralStiffness(1e5)
        tm_tire.SetMaxContactForce(5e4)
        tire.SetTire(tm_tire)

# Create custom terrain
terrain_body = chrono.ChBodyEasyStatic()
terrain_body.SetPos(chrono.ChVectorD(0, 0, 0))

# Load collision mesh
collision_mesh = chrono.ChTriangleMeshShape()
collision_mesh.SetMesh(chrono.ChTriangleMesh())
collision_mesh.GetMesh().LoadWavefrontMesh("Highway_col.obj")
terrain_body.AddAsset(collision_mesh)

# Load visual mesh
visual_shape = chrono.ChVisualShape()
visual_shape.SetMesh(chrono.ChTriangleMesh())
visual_shape.GetMesh().LoadWavefrontMesh("Highway_vis.obj")
terrain_body.AddAsset(visual_shape)

system.Add(terrain_body)

# Setup interactive driver
driver = veh.KeyboardDriver()
vehicle.SetDriver(driver)

# Initialize Irrlicht visualization
visualiz = irr.ChVisualSystemIrrlicht()
visualiz.SetWindowSize(1280, 1024)
visualiz.SetWindowTitle('HMMWV on Custom Terrain')
visualiz.SetCamera(chronoirrassets.CameraType.TRACKING, vehicle.GetChassis(), 3)
visualiz.SetCameraPosition(chrono.ChVectorD(0, 5, 5))  # Adjust camera position
system.SetVisualSystem(visualiz)
visualiz.Initialize()
visualiz.AddLogo()
visualiz.AddSkyBox()
visualiz.AddLightWithShadow(chrono.ChVectorD(0,0,10), chrono.ChVectorD(0,0,0), 200, 500, 250, 10, 80)

# Add drawable objects
visualiz.AddDrawable(vehicle.GetChassis())
visualiz.AddDrawable(terrain_body)

# Simulation parameters
step_size = 0.002  # 500 Hz physics step
realtime = True

# Main simulation loop
while visualiz.Run():
    time = system.GetChTime()
    
    # Update driver inputs
    driver.Synchronize(time)
    
    # Advance simulation step
    system.DoStepDynamics(step_size)
    
    # Update visualization
    visualiz.BeginScene()
    visualiz.DrawAll()
    visualiz.EndScene()
    
    # Real-time synchronization
    if realtime:
        irr.ChIrrTools.SpinWait(system.GetChTime() - time)
```
