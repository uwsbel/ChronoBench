```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

def main():
    # Initialize Chrono system
    my_system = chrono.ChSystemNSC()
    my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

    # Setup Irrlicht visualization
    vis = irr.ChVisualSystemIrrlicht()
    vis.SetWindowSize(1280, 1024)
    vis.SetWindowTitle('CityBus Simulation')
    vis.SetSymbolscale(0.01)
    vis.SetShadowAll(False)
    vis.SetCameraPosition(chrono.ChVectorD(0, 10, 30))
    vis.SetCameraRotation(chrono.Q_from_AngAxis(chrono.ChVectorD(1,0,0), chrono.CH_C_PI/4))
    my_system.SetVisualSystem(vis)

    # Create terrain with texture
    terrain_mat = chrono.ChMaterialSurface()
    terrain_mat.SetFriction(0.9)
    terrain = veh.RigidTerrain(my_system, terrain_mat, True)
    terrain.SetTexture(chrono.GetChronoDataFile('textures/concrete.jpg'), 200, 200)
    terrain.Initialize()

    # Create vehicle
    city_bus = veh.CityBus(my_system)
    init_pos = chrono.ChCoordsysD(chrono.ChVectorD(0, 0.1, 1), chrono.QUNIT)
    city_bus.SetInitPosition(init_pos)
    city_bus.SetDriverType(veh.DriverInputs())  # Interactive driver
    city_bus.Initialize()

    # Configure tire model
    city_bus.SetTireModel(veh.TireModelType.PACEJKA)
    tire_params = veh.PacejkaTireParams()
    tire_params.mu = 0.8  # Example parameter
    for i in range(city_bus.GetNumberofWheels()):
        city_bus.GetWheel(i).SetPacejkaParams(tire_params)

    # Visualization settings
    city_bus.SetChassisVisualizationType(veh.VisualizationType_MESH)
    city_bus.SetWheelVisualizationType(veh.VisualizationType_PRIMITIVES)
    city_bus.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
    city_bus.InitializeGraphics()

    # Get driver and inputs
    driver = city_bus.GetDriver()
    driver_inputs = driver.GetInputs()

    # Simulation loop parameters
    time_step = 1/50.0  # 50 FPS

    # Main simulation loop
    while vis.Run():
        # Update driver inputs from keyboard
        key = vis.GetKeyboard()
        driver_inputs.steering = 0.0
        driver_inputs.throttle = 0.0
        driver_inputs.brake = 0.0

        if key.isKeyDown(irr.KEY_RIGHT):
            driver_inputs.steering = 0.5
        elif key.isKeyDown(irr.KEY_LEFT):
            driver_inputs.steering = -0.5

        if key.isKeyDown(irr.KEY_UP):
            driver_inputs.throttle = 1.0
        elif key.isKeyDown(irr.KEY_DOWN):
            driver_inputs.brake = 1.0

        driver.SetInputs(driver_inputs)

        # Update simulation
        my_system.DoStepDynamics(time_step)

        # Update camera to follow vehicle
        chassis = city_bus.GetChassis()
        cam_pos = chassis.GetPos() + chrono.ChVectorD(0, 5, 10)
        vis.SetCameraPosition(cam_pos)
        vis.SetCameraLookAt(chassis.GetPos())

        # Render scene
        vis.BeginScene()
        vis.Render()
        vis.EndScene()

if __name__ == '__main__':
    main()
```
