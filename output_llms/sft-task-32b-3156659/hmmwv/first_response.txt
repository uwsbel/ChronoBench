

```python
import pychrono as chrono
import pychrono.vehicle as vehicle
import pychrono.irrlicht as chronoirr

# Initialize the Chrono system
sys = chrono.ChSystemNSC()
sys.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))

# Create and initialize the HMMWV vehicle
hmmwv = vehicle.HMMWV(sys)
hmmwv.SetContactMethod(vehicle.ChContactMethod.NSC)
hmmwv.SetTireType(vehicle.TireModelType.TMEASY)
hmmwv.Initialize(chrono.ChCoordsysd(chrono.ChVector3d(0, 0, 1), chrono.QUNIT))
hmmwv.SetChassisVisualizationType(vehicle.VisualizationType.PRIMITIVES)
hmmwv.SetSuspensionVisualizationType(vehicle.VisualizationType.PRIMITIVES)
hmmwv.SetSteeringVisualizationType(vehicle.VisualizationType.PRIMITIVES)
hmmwv.SetDrivelineVisualizationType(vehicle.VisualizationType.PRIMITIVES)

# Create the rigid terrain
terrain_mat = chrono.ChContactMaterialNSC()
terrain_mat.SetFriction(0.9)
terrain_mat.SetRestitution(0.01)
terrain = vehicle.RigidTerrain(sys)
terrain_patch = terrain.AddPatch(
    terrain_mat, 
    chrono.ChVector3d(0, 0, 0), 
    chrono.ChVector3d(100, 100, 0), 
    chrono.ChVector3d(100, 100, 1)
)
terrain_patch.SetTexture(chronoirr.video.ITexture.SOLID_COLOR, chrono.ChColor(0.8, 0.8, 0.8))
terrain.Initialize()

# Create the interactive driver
driver = vehicle.ChIrrGuiDriver(hmmwv.GetVehicle())
driver.SetSteeringDelta(0.02)
driver.SetThrottleDelta(0.02)
driver.SetBrakingDelta(0.02)

# Set up the Irrlicht visualization
app = vehicle.ChIrrApp(hmmwv, "HMMWV Simulation", chronoirr.dimension2du(800, 600))
app.SetSkyBox()
app.SetCamera(chronoirr.vector3df(0, -6, 2), chronoirr.vector3df(0, 0, 1.5))
app.SetTypicalLights()
app.AddLightWithShadow(chronoirr.vector3df(10, 10, 10), chronoirr.vector3df(0, 0, 0), 5, 20, 10, 50, 512)
app.SetShadowMapEnabled(True, True, 512, 512, 1.5, 5000, 45)
app.SetTimestep(0.02)

# Simulation loop
while app.GetDevice().run():
    app.BeginScene()
    driver.Update()
    hmmwv.Synchronize(app.GetTime())
    sys.DoStepDynamics(app.GetTimeStep())
    app.DrawAll()
    app.EndScene()
```
