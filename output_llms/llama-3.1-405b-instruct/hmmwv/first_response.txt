Here is a PyChrono simulation script based on the provided instructions:

```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Set the path to the Chrono data directory
chrono.SetChronoDataPath('C:/path/to/chrono/data/')

# Create a Chrono system
sys = chrono.ChSystemNSC()

# Create the ground body
ground = chrono.ChBodyEasyBox(sys, 100, 100, 1, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -1, 0))
sys.Add(ground)

# Create the vehicle
vehicle = veh.ChPart('HMMWV')
vehicle.SetChassisMass(2500)
vehicle.SetChassisDimensions(4.8, 1.8, 2.2)
vehicle.SetChassisPosition(chrono.ChVectorD(0, 1, 0))
vehicle.SetChassisOrientation(chrono.Q_from_AngY(0))

# Add the vehicle to the system
sys.Add(vehicle)

# Create the tires
tire_model = veh.ChTMEasy()
tire_model.SetTireMass(50)
tire_model.SetTireRadius(0.4)
tire_model.SetTireWidth(0.2)
tire_model.SetTireTreadDepth(0.05)
vehicle.AddTire(tire_model, chrono.ChVectorD(-1.5, -1, 0.5), chrono.Q_from_AngY(0))
vehicle.AddTire(tire_model, chrono.ChVectorD(1.5, -1, 0.5), chrono.Q_from_AngY(0))
vehicle.AddTire(tire_model, chrono.ChVectorD(-1.5, -1, -0.5), chrono.Q_from_AngY(0))
vehicle.AddTire(tire_model, chrono.ChVectorD(1.5, -1, -0.5), chrono.Q_from_AngY(0))

# Create the driver system
driver = veh.ChIrrNodeDriver(sys)
driver.Initialize()
driver.SetSteeringDelta(0.01)
driver.SetThrottleDelta(0.01)
driver.SetBrakingDelta(0.01)

# Create the terrain
terrain = veh.RigidTerrain(sys)
terrain.SetSize(100, 100)
terrain.SetTexture(chrono.GetChronoDataFile('textures/concrete.jpg'))
sys.Add(terrain)

# Create the Irrlicht application
app = chronoirr.ChIrrApp(sys, 'HMMWV Simulation', chronoirr.dimension2du(800, 600))
app.AddTypicalSky()
app.AddTypicalLights()
app.AddCamera(chrono.ChVectorD(0, 2, -5))
app.SetSymbolscale(0.02)
app.SetContactsDrawMode(chronoirr.ContactDrawMode::e_points)

# Simulation loop
while app.GetDevice().run():
    time = sys.GetChTime()
    driver.Synchronize(time)
    sys.DoStepDynamics(0.02)
    app.BeginScene(True, True, chronoirr.SColor(255, 140, 161, 192))
    app.DrawAll()
    app.EndScene()
```

This script creates a PyChrono simulation of a full HMMWV on a flat rigid terrain using Irrlicht for visualization. The vehicle is initialized with specific parameters such as location, orientation, contact method, and TMEASY tire model. The script also sets up a RigidTerrain with defined dimensions and texture, and configures an interactive driver system for steering, throttle, and braking. The simulation loop runs in real time at 50 frames per second, updating the vehicle dynamics and rendering the scene. 

Please note that you need to replace `'C:/path/to/chrono/data/'` with the actual path to the Chrono data directory on your system. Also, make sure that the `chrono` and `pychrono` modules are installed and imported correctly.
