Here is a PyChrono simulation script based on your instructions:
```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Initialize the Chrono system
chrono.SetChronoDataPath('/path/to/chrono/data')
system = chrono.ChSystemNSC()

# Add a ground body
ground = chrono.ChBodyEasyBox(system,  # parent
                              10, 10, 1,  # dimensions
                              1000,  # density
                              True,  # visualization
                              True,  # collision
                              chrono.ChMaterialSurfaceNSC())
ground.SetPos(chrono.ChVectorD(0, -1, 0))
system.Add(ground)

# Initialize the Turtlebot
turtlebot = chrono.ChBodyEasyBox(system,  # parent
                                 0.5, 0.5, 0.5,  # dimensions
                                 10,  # density
                                 True,  # visualization
                                 True,  # collision
                                 chrono.ChMaterialSurfaceNSC())
turtlebot.SetPos(chrono.ChVectorD(0, 0.5, 0))
turtlebot.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
system.Add(turtlebot)

# Create a motor for the Turtlebot's left wheel
left_motor = chrono.ChLinkMotorRotationSpeed(turtlebot,  # body
                                             chrono.ChFrameD(chrono.ChVectorD(-0.25, 0, 0)),  # frame
                                             chrono.ChFunction_Sine(0,  # amplitude
                                                                    1,  # frequency
                                                                    0))  # phase
system.Add(left_motor)

# Create a motor for the Turtlebot's right wheel
right_motor = chrono.ChLinkMotorRotationSpeed(turtlebot,  # body
                                              chrono.ChFrameD(chrono.ChVectorD(0.25, 0, 0)),  # frame
                                              chrono.ChFunction_Sine(0,  # amplitude
                                                                     1,  # frequency
                                                                     0))  # phase
system.Add(right_motor)

# Set up the Irrlicht visualization
application = chronoirr.ChIrrApp(system,  # system
                                  'Turtlebot Simulation',  # title
                                  chronoirr.dimension2du(800, 600))  # window size
application.AddTypicalCamera(chrono.ChVectorD(0, 0, 2))  # camera position
application.AddTypicalLights()  # lighting

# Simulation loop
while application.GetDevice().run():
    # Update the simulation
    system.DoStepDynamics(chrono.ChTime(1e-3))

    # Update the motors
    if system.GetChTime() < 5:
        left_motor.SetMotorFunction(chrono.ChFunction_Sine(0.5, 1, 0))
        right_motor.SetMotorFunction(chrono.ChFunction_Sine(0.5, 1, 0))
    elif system.GetChTime() < 10:
        left_motor.SetMotorFunction(chrono.ChFunction_Sine(-0.5, 1, 0))
        right_motor.SetMotorFunction(chrono.ChFunction_Sine(0.5, 1, 0))
    else:
        left_motor.SetMotorFunction(chrono.ChFunction_Sine(0, 1, 0))
        right_motor.SetMotorFunction(chrono.ChFunction_Sine(0, 1, 0))

    # Render the simulation
    application.BeginScene(True, True, chronoirr.SColor(255, 255, 255))
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```
This script creates a PyChrono simulation with a Turtlebot robot on a rigid terrain. The Turtlebot is controlled by two motors that turn the left and right wheels at specified times. The simulation is visualized in real-time using Irrlicht, with a camera and lighting. The simulation loop updates the motors and renders the simulation at each timestep.

Note that you will need to modify the script to match your specific use case, such as changing the dimensions and density of the Turtlebot and ground body, or adding additional objects to the simulation.
