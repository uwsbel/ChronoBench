Here's a Python script that simulates the specified PyChrono environment:

```python
import pychrono.core as chrono
import pychrono.sensor as sens
import numpy as np

# Initialize the PyChrono environment and core components
chrono.SetChronoDataPath('/path/to/chrono/data/')
my_system = chrono.ChSystemNSC()

# Create a triangular mesh from a Wavefront .obj file
mesh = chrono.ChTriangleMeshConnected()
mesh.LoadWavefrontMesh('./path/to/mesh.obj')

# Create a fixed body and add the mesh to it
body = chrono.ChBodyEasyMesh(mesh, 1000, True, True, 0.1, 0.1)
body.SetPos(chrono.ChVectorD(0, 0, 0))
body.SetBodyFixed(True)
my_system.Add(body)

# Create a sensor manager
manager = sens.ChSensorManager(my_system)

# Create a lidar sensor
lidar = sens.ChLidarSensor(
    body,  # body lidar is attached to
    100,   # scanning rate
    chrono.ChVectorD(0, 0, 1),  # direction
    0.02,  # beam length
    0.01,  # beam width
    100,   # horizontal samples
    100,   # vertical samples
    0.1,   # horizontal FOV
    0.1    # vertical FOV
)
manager.AddSensor(lidar)

# Add noise filter to the lidar
filter = sens.ChFilterNoise()
filter.SetStandardDeviation(0.01)
lidar.AddFilter(filter)

# Add visualization to the lidar
visualizer = sens.ChVisualizer()
visualizer.SetType(sens.ChVisualizer.Type.LIDAR)
visualizer.SetColor(chrono.ChColor(1, 0, 0))
lidar.AddVisualizer(visualizer)

# Set saving options for the lidar data
lidar.SetSaveData(True)
lidar.SetSaveFileName('./lidar_data/')

# Simulate the system
sim_time = 10
time_step = 0.01
orbit_radius = 1
angle = 0
angle_step = 0.1

while my_system.GetChTime() < sim_time:
    # Update lidar position in an orbit around the mesh
    lidar.SetOffsetPose(chrono.ChFrameD(
        chrono.ChVectorD(orbit_radius * np.cos(angle), orbit_radius * np.sin(angle), 0),
        chrono.ChQuaternionD(0, 0, 0, 1)
    ))
    angle += angle_step

    # Simulate the system for one time step
    my_system.DoStepDynamics(time_step)

    # Print lidar buffer data
    buffer = lidar.GetBuffer()
    print('Lidar buffer data:')
    for point in buffer:
        print(point)
```

Please replace `'/path/to/chrono/data/'` and `'./path/to/mesh.obj'` with the actual paths to your Chrono data directory and mesh file, respectively. Also, ensure that the directory `./lidar_data/` exists to save the lidar data.

This script creates a PyChrono simulation environment, adds a triangular mesh from a Wavefront .obj file as a fixed body, and attaches a lidar sensor to the body using a sensor manager. The lidar sensor is configured with noise filters, visualizations, and saving options. During the simulation, the lidar's position is updated in an orbit around the mesh, and the lidar buffer data is printed at each simulation step.
