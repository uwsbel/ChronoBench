```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Initialize PyChrono environment
chrono.SetChronoDataPath('/path/to/chrono/data/')

# Create a Chrono::Engine
my_system = chrono.ChSystemNSC()

# Create a rigid terrain
terrain = chrono.ChBodyEasyBox(
    my_system, 
    name='terrain', 
    pos=chrono.ChVectorD(0, -1, 0), 
    rot=chrono.ChQuaternionD(1, 0, 0, 0), 
    dim=chrono.ChVectorD(100, 1, 100), 
    density=1000
)

# Create a CityBus vehicle
vehicle = chrono.ChPart(
    my_system, 
    name='vehicle', 
    pos=chrono.ChVectorD(0, 1, 0), 
    rot=chrono.ChQuaternionD(1, 0, 0, 0)
)

# Initialize vehicle parameters
vehicle.SetMass(5000)
vehicle.SetInertiaXX(chrono.ChVectorD(1, 1, 1))

# Create tire models
tire_LF = chrono.ChPart(
    my_system, 
    name='tire_LF', 
    pos=chrono.ChVectorD(-1.5, 0.5, 1.5), 
    rot=chrono.ChQuaternionD(1, 0, 0, 0)
)
tire_RF = chrono.ChPart(
    my_system, 
    name='tire_RF', 
    pos=chrono.ChVectorD(1.5, 0.5, 1.5), 
    rot=chrono.ChQuaternionD(1, 0, 0, 0)
)
tire_LR = chrono.ChPart(
    my_system, 
    name='tire_LR', 
    pos=chrono.ChVectorD(-1.5, 0.5, -1.5), 
    rot=chrono.ChQuaternionD(1, 0, 0, 0)
)
tire_RR = chrono.ChPart(
    my_system, 
    name='tire_RR', 
    pos=chrono.ChVectorD(1.5, 0.5, -1.5), 
    rot=chrono.ChQuaternionD(1, 0, 0, 0)
)

# Create a revolute joint for each tire
revolute_joint_LF = chrono.ChLinkRevolute(
    vehicle, 
    tire_LF, 
    chrono.ChFrameD(chrono.ChVectorD(-1.5, 0.5, 1.5), chrono.ChQuaternionD(1, 0, 0, 0))
)
revolute_joint_RF = chrono.ChLinkRevolute(
    vehicle, 
    tire_RF, 
    chrono.ChFrameD(chrono.ChVectorD(1.5, 0.5, 1.5), chrono.ChQuaternionD(1, 0, 0, 0))
)
revolute_joint_LR = chrono.ChLinkRevolute(
    vehicle, 
    tire_LR, 
    chrono.ChFrameD(chrono.ChVectorD(-1.5, 0.5, -1.5), chrono.ChQuaternionD(1, 0, 0, 0))
)
revolute_joint_RR = chrono.ChLinkRevolute(
    vehicle, 
    tire_RR, 
    chrono.ChFrameD(chrono.ChVectorD(1.5, 0.5, -1.5), chrono.ChQuaternionD(1, 0, 0, 0))
)

# Create a driver system
driver = chrono.ChDriver(
    vehicle, 
    max_steering_angle=chrono.ChMath.deg2rad(30), 
    max_torque=1000, 
    max_braking_torque=500
)

# Create an Irrlicht application
myapplication = chronoirr.ChIrrApp(
    my_system, 
    'CityBus Simulation', 
    chronoirr.dimension2du(800, 600)
)

# Set the camera to follow the vehicle
myapplication.AddCamera(
    chronoirr.ChCamera(
        chronoirr.vector3df(0, 5, -10), 
        chronoirr.vector3df(0, 0, 0)
    )
)
myapplication.AddTypicalLights()

# Add the terrain and vehicle to the Irrlicht scene
myapplication.AddMesh(
    terrain, 
    chronoirr.ChMesh(
        chronoirr.vector3df(0, -1, 0), 
        chronoirr.vector3df(100, 1, 100)
    )
)
myapplication.AddMesh(
    vehicle, 
    chronoirr.ChMesh(
        chronoirr.vector3df(0, 1, 0), 
        chronoirr.vector3df(5, 2, 5)
    )
)

# Add the tires to the Irrlicht scene
myapplication.AddMesh(
    tire_LF, 
    chronoirr.ChMesh(
        chronoirr.vector3df(-1.5, 0.5, 1.5), 
        chronoirr.vector3df(1, 1, 1)
    )
)
myapplication.AddMesh(
    tire_RF, 
    chronoirr.ChMesh(
        chronoirr.vector3df(1.5, 0.5, 1.5), 
        chronoirr.vector3df(1, 1, 1)
    )
)
myapplication.AddMesh(
    tire_LR, 
    chronoirr.ChMesh(
        chronoirr.vector3df(-1.5, 0.5, -1.5), 
        chronoirr.vector3df(1, 1, 1)
    )
)
myapplication.AddMesh(
    tire_RR, 
    chronoirr.ChMesh(
        chronoirr.vector3df(1.5, 0.5, -1.5), 
        chronoirr.vector3df(1, 1, 1)
    )
)

# Set the simulation loop to run at 50 frames per second
myapplication.SetTimestep(0.02)

# Run the simulation loop
while myapplication.GetDevice().run():
    # Update the vehicle dynamics
    my_system.DoStepDynamics(0.02)

    # Render the scene
    myapplication.BeginScene()
    myapplication.DrawAll()
    myapplication.EndScene()

    # Update the driver system
    driver.Update()
```
This script initializes a PyChrono environment and creates a CityBus vehicle with parameters including location, orientation, and tire model. It also creates a rigid terrain with a custom texture and an interactive driver system that allows for control of the vehicle's steering, throttle, and braking. The simulation loop runs at 50 frames per second, updating the vehicle dynamics and rendering the scene in real time.

Note that this script assumes that you have the PyChrono and Irrlicht libraries installed and configured properly on your system. Additionally, you may need to modify the script to suit your specific needs and requirements.
