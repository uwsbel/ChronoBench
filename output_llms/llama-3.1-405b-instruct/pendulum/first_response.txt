Here's a Python script that simulates a simple pendulum using PyChrono with Irrlicht for visualization.

```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Initialize the Chrono::Engine environment
my_system = chrono.ChSystemNSC()

# Create a ground body
ground = chrono.ChBodyEasyBox(100, 1, 100,  # x, y, z dimensions
                               1000,          # density
                               True,          # visualization
                               True,          # collision
                               my_system.NewBody())
ground.SetPos(chrono.ChVectorD(0, -1, 0))
ground.SetBodyFixed(True)

# Create a pendulum body
pendulum = chrono.ChBodyEasyCylinder(0.1,  # radius
                                      1,    # height
                                      10,   # density
                                      True, # visualization
                                      True, # collision
                                      my_system.NewBody())
pendulum.SetPos(chrono.ChVectorD(0, 1, 0))
pendulum.SetMass(10)
pendulum.SetInertiaXX(chrono.ChVectorD(0.1, 0.1, 0.1))

# Create a revolute joint between the ground and pendulum
revolute_joint = chrono.ChLinkRevolute()
revolute_joint.Initialize(ground, pendulum, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.ChQuaternionD(1, 0, 0, 0)))
my_system.AddLink(revolute_joint)

# Initialize the Irrlicht visualization system
application = chronoirr.ChIrrApp(my_system, 'PyChrono Simple Pendulum Simulation', chronoirr.dimension2du(800, 600))
application.AddTypicalSky()
application.AddTypicalLights()
application.AddCamera(chronoirr.vector3df(0, 0, 3))
application.AssetBindAll()
application.AssetUpdateAll()

# Simulation time step
time_step = 0.01

# Simulation loop
while application.GetDevice().run():
    # Log the pendulum's position and velocity
    print(f'Pendulum position: {pendulum.GetPos().x}, {pendulum.GetPos().y}, {pendulum.GetPos().z}')
    print(f'Pendulum velocity: {pendulum.GetPos_dt().x}, {pendulum.GetPos_dt().y}, {pendulum.GetPos_dt().z}')

    # Advance the simulation by one time step
    my_system.DoStepDynamics(time_step)

    # Update the Irrlicht visualization system
    application.BeginScene()
    application.DrawAll()
    application.EndScene()
```

This script creates a simulation of a simple pendulum using PyChrono with Irrlicht for visualization. It sets up a ground body, a pendulum with appropriate mass and inertia, and connects them with a revolute joint. The visualization system is used to render the simulation, and the pendulum's position and velocity are periodically logged to the console.
