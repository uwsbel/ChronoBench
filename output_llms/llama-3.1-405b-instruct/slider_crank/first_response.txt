Here's a sample PyChrono script that simulates a simple crank-slider mechanism:

```python
# Import necessary PyChrono modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Set the simulation parameters
chrono.SetChronoDataPath("/path/to/chrono/data/")

# Create the simulation system
sys = chrono.ChSystemNSC()

# Create the floor (truss)
floor = chrono.ChBodyEasyBox(sys,  # parent system
                              10, 10, 0.1,  # dimensions (x, y, z)
                              1000,  # density
                              True,  # visualization
                              False,  # collision
                              0.5)  # friction coefficient
sys.Add(floor)

# Create the crankshaft
crankshaft = chrono.ChBodyEasyCylinder(sys,  # parent system
                                        0.5,  # radius
                                        2,  # height
                                        700,  # density
                                        True,  # visualization
                                        False,  # collision
                                        0.5)  # friction coefficient
crankshaft.SetPos(chrono.ChVectorD(0, 0, 1))
crankshaft.SetRot(chrono.ChQuaternionD(0, 0, 0, 1))
sys.Add(crankshaft)

# Create the connecting rod
rod = chrono.ChBodyEasyBox(sys,  # parent system
                            2, 0.1, 0.1,  # dimensions (x, y, z)
                            700,  # density
                            True,  # visualization
                            False,  # collision
                            0.5)  # friction coefficient
rod.SetPos(chrono.ChVectorD(1, 0, 1))
rod.SetRot(chrono.ChQuaternionD(0, 0, 0, 1))
sys.Add(rod)

# Create the piston
piston = chrono.ChBodyEasyCylinder(sys,  # parent system
                                    0.25,  # radius
                                    1,  # height
                                    700,  # density
                                    True,  # visualization
                                    False,  # collision
                                    0.5)  # friction coefficient
piston.SetPos(chrono.ChVectorD(2, 0, 1))
piston.SetRot(chrono.ChQuaternionD(0, 0, 0, 1))
sys.Add(piston)

# Create joints between the components
revolute_joint = chrono.ChLinkRevolute()
revolute_joint.Initialize(crankshaft, rod, chrono.ChVectorD(0, 0, 1))
sys.Add(revolute_joint)

revolute_joint2 = chrono.ChLinkRevolute()
revolute_joint2.Initialize(rod, piston, chrono.ChVectorD(2, 0, 1))
sys.Add(revolute_joint2)

# Create a motor to drive the crankshaft
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(crankshaft, floor, chrono.ChVectorD(0, 0, 1))
motor.SetMotorFunction(chrono.ChFunction_Const(chrono.CH_C_PI / 2))
sys.Add(motor)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(800, 600)
vis.SetWindowTitle("Crank-Slider Mechanism")
vis.Initialize()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 0, 5), chrono.ChVectorD(0, 0, 0))
vis.AddLightWithShadow(chrono.ChVectorD(2, 2, 2), chrono.ChVectorD(0, 0, 0), 3, 2, 2, 40, 512)

# Add visual elements
vis.AddLogo()
vis.AddTypicalLights()

# Simulate the system
while vis.Run():
    sys.DoStepDynamics(1e-3)
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()
```

This script creates a simple crank-slider mechanism with a floor, crankshaft, connecting rod, and piston. The crankshaft is driven by a motor to spin at a constant angular speed. The Irrlicht visualization system is used to visualize the simulation, with customizable camera settings, lighting, and additional visual elements like textures and logos.

Please note that you may need to adjust the path to the Chrono data directory and the simulation parameters to suit your specific needs. Additionally, this script assumes that you have the necessary dependencies installed, including PyChrono and Irrlicht.
