Here's a sample PyChrono simulation script that follows the given instructions:

```python
# Import necessary PyChrono modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr

# Initialize the PyChrono environment
chrono.SetChronoDataPath("C:/path/to/chrono/data/")

# Create a ChronoEngine physical system
my_system = chrono.ChSystemNSC()

# Create the Irrlicht visualization system
application = chronoirr.ChIrrApp(my_system, "PyChrono Crank-Slider Simulation", chronoirr.dimension2du(800, 600))

# Add a floor (truss) to the system
floor_body = chrono.ChBodyEasyBox(100, 0.5, 100,  # x, y, z dimensions
                                  1000,  # density
                                  True,  # collide
                                  True)  # visualization
floor_body.SetPos(chrono.ChVectorD(0, -5, 0))
my_system.Add(floor_body)

# Add a crankshaft to the system
crank_body = chrono.ChBodyEasyCylinder(2,  # radius
                                       10,  # height
                                       1000,  # density
                                       True,  # collide
                                       True)  # visualization
crank_body.SetPos(chrono.ChVectorD(0, 0, 0))
my_system.Add(crank_body)

# Add a connecting rod to the system
rod_body = chrono.ChBodyEasyBox(2, 10, 0.5,  # x, y, z dimensions
                                1000,  # density
                                True,  # collide
                                True)  # visualization
rod_body.SetPos(chrono.ChVectorD(10, 0, 0))
my_system.Add(rod_body)

# Add a piston to the system
piston_body = chrono.ChBodyEasyBox(2, 2, 2,  # x, y, z dimensions
                                   1000,  # density
                                   True,  # collide
                                   True)  # visualization
piston_body.SetPos(chrono.ChVectorD(20, 0, 0))
my_system.Add(piston_body)

# Create joints to connect the bodies
revolute_joint = chrono.ChLinkRevolute()
revolute_joint.Initialize(crank_body, rod_body, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))
my_system.AddLink(revolute_joint)

revolute_joint_2 = chrono.ChLinkRevolute()
revolute_joint_2.Initialize(rod_body, piston_body, chrono.ChFrameD(chrono.ChVectorD(10, 0, 0)))
my_system.AddLink(revolute_joint_2)

# Create a motor to drive the crankshaft
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(crank_body, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))
motor.SetMotorFunction(chrono.ChFunction_Const(chrono.CH_C_PI / 2))  # 90 degrees per second
my_system.AddLink(motor)

# Set up the Irrlicht visualization system
application.AddTypicalSky()
application.AddTypicalLights()
application.AddCamera(chronoirr.vector3df(0, 10, -20))
application.AssetBindAll()
application.AssetUpdateAll()

# Run the simulation
application.SetTimestep(0.01)
application.SetTryRealtime(True)

while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```

This script creates a PyChrono simulation of a simple crank-slider mechanism with a floor, crankshaft, connecting rod, and piston. The system is driven by a motor that spins the crankshaft at a constant angular speed. The simulation is visualized using the Irrlicht visualization system with customizable camera settings, lighting, and additional visual elements like textures and logos.

Note: You may need to adjust the file path in `chrono.SetChronoDataPath()` to match the location of your PyChrono data directory.
