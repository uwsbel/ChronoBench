Here's a sample PyChrono simulation script based on the given instructions:

```python
# Import necessary PyChrono modules
import pychrono.core as chrono
import pychrono.vehicle as veh
import pychrono.sensor as sens
import pychrono.irrlicht as irr

# Initialize the PyChrono environment
chrono.SetChronoDataPath('/path/to/chrono/data/')

# Create a Chrono::Engine physical system
my_system = chrono.ChSystemNSC()

# Set the gravitational acceleration
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the HMMWV vehicle
my_hmmwv = veh.ChHMMWV()
my_hmmwv.Initialize(my_system)
my_hmmwv.SetChassisVisualizationType(veh.VisualizationType_MESH)
my_hmmwv.SetChassisPosition(chrono.ChVectorD(0, 1.5, 0))

# Add the terrain
terrain = veh.RigidTerrain(my_system)
my_hmmwv.GetVehicle().SetTerrain(terrain)

# Create a driver system
my_driver = veh.ChIrrGuiDriver()
my_driver.Initialize(my_hmmwv.GetVehicle())

# Create an IMU sensor and attach it to the vehicle's chassis
my_imu = sens.ChIMUSensor()
my_imu.SetLocation(chrono.ChVectorD(0, 0.5, 0))
my_imu.SetNoiseParameters(0.01, 0.01, 0.01)
my_hmmwv.GetChassisBody().AddSensor(my_imu)

# Create a GPS sensor and attach it to the vehicle's chassis
my_gps = sens.ChGPSSensor()
my_gps.SetLocation(chrono.ChVectorD(0, 0.5, 0))
my_gps.SetNoiseParameters(0.1, 0.1, 0.1)
my_hmmwv.GetChassisBody().AddSensor(my_gps)

# Create an Irrlicht application
my_app = irr.ChIrrApp(my_system, "PyChrono HMMWV Simulation", irr.dimension2du(800, 600))

# Simulation loop
while my_app.GetDevice().run():
    # Output the vehicle mass
    print("Vehicle mass:", my_hmmwv.GetVehicle().GetTotalMass())

    # Advance the vehicle
    my_hmmwv.Advance(0.01)

    # Synchronize the vehicle and the terrain
    my_hmmwv.Synchronize(terrain, 0.01)

    # Update the driver inputs
    my_driver_inputs = my_driver.GetInputs()
    my_hmmwv.GetVehicle().SetThrottle(my_driver_inputs[0])
    my_hmmwv.GetVehicle().SetSteering(my_driver_inputs[1])
    my_hmmwv.GetVehicle().SetBraking(my_driver_inputs[2])

    # Update the sensor data
    my_imu.Update()
    my_gps.Update()

    # Render the scene
    my_app.BeginScene(True, True, irr.SColor(255, 140, 161, 192))
    my_app.DrawAll()
    my_app.EndScene()
```

This script sets up a PyChrono simulation with an HMMWV vehicle, including a full system setup with terrain, driver inputs, and a visual interface using Irrlicht. It also adds and manages IMU and GPS sensors attached to the vehicle's chassis, processing and updating sensor data at each simulation step. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system modules while outputting the vehicle mass and rendering the scenes.
