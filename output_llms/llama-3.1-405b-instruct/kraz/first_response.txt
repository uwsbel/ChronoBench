Here is a PyChrono simulation script based on the provided instructions:

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as chronoveh

# Initialize the PyChrono environment
chrono.SetChronoDataPath("path/to/chrono/data")

# Create a Chrono::Engine physical system
sys = chrono.ChSystemNSC()

# Create the ground body
ground = chrono.ChBodyEasyBox(sys,  # Parent system
                               100, 100, 2,  # Box dimensions
                               1000,  # Density
                               True,  # Visualization
                               True)  # Collision
ground.SetPos(chrono.ChVectorD(0, -2, 0))
ground.SetBodyFixed(True)
sys.Add(ground)

# Set the default material properties for the ground
material = chrono.ChMaterialSurfaceNSC()
material.SetFriction(0.5)
material.SetRestitution(0.5)
ground.SetMaterialSurface(material)

# Create the vehicle
vehicle = chronoveh.ChPartSys(sys)
vehicle.SetContactMethod(chronoveh.ChPartSys.NSC)
vehicle.SetChassisMass(1500)
vehicle.SetChassisDimension(chrono.ChVectorD(4.5, 1.8, 1.7))
vehicle.SetInitPos(chrono.ChVectorD(0, 1, 0))
vehicle.SetInitRot(chrono.Q_ROTATE_Y_TO_Z)
vehicle.SetPowertrainType(chronoveh.ChPartSys.SHAFT)
vehicle.SetTransmissionType(chronoveh.ChPartSys.MANUAL)
vehicle.SetGearRatios([3.5, 2.5, 1.5, 1.0, 0.75])
vehicle.SetDifferentialType(chronoveh.ChPartSys.OPEN)
vehicle.SetTireType(chronoveh.ChPartSys.RIGID)
vehicle.SetTireRadius(0.35)
vehicle.SetTireWidth(0.2)
vehicle.SetSuspensionType(chronoveh.ChPartSys.RIGID)
vehicle.SetSuspensionStiffness(10000)
vehicle.SetSuspensionDamping(500)
vehicle.SetSteeringType(chronoveh.ChPartSys.RIGID)
vehicle.SetSteeringRatio(15)
sys.Add(vehicle)

# Create a driver system
driver = chronoveh.ChDriver(vehicle)
driver.SetSteeringController(chronoveh.ChDriver.RIGID)
driver.SetThrottleController(chronoveh.ChDriver.RIGID)
driver.SetBrakingController(chronoveh.ChDriver.RIGID)
driver.SetGearController(chronoveh.ChDriver.MANUAL)
sys.Add(driver)

# Create an Irrlicht application
app = chronoirr.ChIrrApp(sys, "PyChrono Simulation", chronoirr.dimension2du(800, 600))
app.AddTypicalSky()
app.AddTypicalLights()
app.AddCamera(chronoirr.vector3df(0, 5, -10))
app.AddTypicalCamera()

# Simulation loop
while app.GetDevice().run():
    # Synchronize and advance the vehicle, terrain, driver, and visual system
    sys.DoStepDynamics(1e-3)
    app.BeginScene()
    app.DrawAll()
    app.EndScene()
```

This script initializes the PyChrono environment, sets up a rigid terrain with defined friction and restitution, creates a Kraz vehicle with specified initial conditions, and implements a driver system for vehicle control. It also creates a real-time visualization using Irrlicht with camera settings and lighting. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.
