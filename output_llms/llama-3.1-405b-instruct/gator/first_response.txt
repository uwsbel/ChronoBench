```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Initialize the PyChrono environment
chrono.SetChronoDataPath("/path/to/chrono/data")

# Create a ChronoEngine physical system
my_system = chrono.ChSystemNSC()

# Create the ground body
ground = chrono.ChBodyEasyBox(
    my_system,  # the parent system
    100, 100, 2,  # x, y, z dimensions
    1000,  # density
    True,  # contact geometry
    True  # enable visualization
)
ground.SetBodyFixed(True)
ground.SetCollide(True)

# Create the RigidTerrain
terrain = chrono.ChRigidTerrain(
    my_system,  # the parent system
    chrono.ChVector(0, 0, -1),  # location
    chrono.ChVector(0, 0, 0),  # rotation
    100, 100,  # x, y dimensions
    2  # height
)
terrain.SetTexture(chrono.ChTexture("terrain_texture.png"))

# Create the vehicle
vehicle = chrono.ChPart(
    my_system,  # the parent system
    "Gator_Vehicle",  # name
    chrono.ChVector(0, 0, 1),  # location
    chrono.ChQuaternion(1, 0, 0, 0)  # rotation
)
vehicle.SetMass(1500)
vehicle.SetInertiaXX(chrono.ChVector(500, 500, 500))

# Add vehicle components
chassis = chrono.ChPart(
    vehicle,  # the parent body
    "Chassis",  # name
    chrono.ChVector(0, 0, 0.5),  # location
    chrono.ChQuaternion(1, 0, 0, 0)  # rotation
)
chassis.SetMass(500)
chassis.SetInertiaXX(chrono.ChVector(100, 100, 100))

# Add wheels and tires
wheel_front_left = chrono.ChPart(
    vehicle,  # the parent body
    "Wheel_Front_Left",  # name
    chrono.ChVector(-0.5, 0.5, 0),  # location
    chrono.ChQuaternion(1, 0, 0, 0)  # rotation
)
wheel_front_left.SetMass(20)
wheel_front_left.SetInertiaXX(chrono.ChVector(5, 5, 5))

wheel_front_right = chrono.ChPart(
    vehicle,  # the parent body
    "Wheel_Front_Right",  # name
    chrono.ChVector(0.5, 0.5, 0),  # location
    chrono.ChQuaternion(1, 0, 0, 0)  # rotation
)
wheel_front_right.SetMass(20)
wheel_front_right.SetInertiaXX(chrono.ChVector(5, 5, 5))

wheel_rear_left = chrono.ChPart(
    vehicle,  # the parent body
    "Wheel_Rear_Left",  # name
    chrono.ChVector(-0.5, -0.5, 0),  # location
    chrono.ChQuaternion(1, 0, 0, 0)  # rotation
)
wheel_rear_left.SetMass(20)
wheel_rear_left.SetInertiaXX(chrono.ChVector(5, 5, 5))

wheel_rear_right = chrono.ChPart(
    vehicle,  # the parent body
    "Wheel_Rear_Right",  # name
    chrono.ChVector(0.5, -0.5, 0),  # location
    chrono.ChQuaternion(1, 0, 0, 0)  # rotation
)
wheel_rear_right.SetMass(20)
wheel_rear_right.SetInertiaXX(chrono.ChVector(5, 5, 5))

# Add TMEasy tire model
tire_model = chrono.ChTMEasy(
    my_system,  # the parent system
    "TMEasy_Tire_Model",  # name
    chrono.ChVector(0, 0, 0),  # location
    chrono.ChQuaternion(1, 0, 0, 0)  # rotation
)

# Add mesh visualization for vehicle components
chassis.AddAsset(chrono.ChSphereShape(0.5))
wheel_front_left.AddAsset(chrono.ChSphereShape(0.2))
wheel_front_right.AddAsset(chrono.ChSphereShape(0.2))
wheel_rear_left.AddAsset(chrono.ChSphereShape(0.2))
wheel_rear_right.AddAsset(chrono.ChSphereShape(0.2))

# Create an interactive driver system
driver = chrono.ChIrrNodeAsset(
    my_system,  # the parent system
    "Driver",  # name
    chrono.ChVector(0, 0, 1.5),  # location
    chrono.ChQuaternion(1, 0, 0, 0)  # rotation
)

# Create an Irrlicht application
myapplication = chronoirr.ChIrrApp(
    my_system,  # the parent system
    "PyChrono Simulation",  # title
    chronoirr.dimension2du(800, 600)  # window dimensions
)

# Set up the simulation loop
while myapplication.GetDevice().run():
    # Update vehicle dynamics
    my_system.DoStepDynamics(1 / 50.0)

    # Update rendering
    myapplication.BeginScene()
    myapplication.DrawAll()
    myapplication.EndScene()

    # Update driver input
    driver.SetSteering(chrono.ChFunction_Recorder(myapplication.GetDevice().getCursorPos().X))
    driver.SetThrottle(chrono.ChFunction_Recorder(myapplication.GetDevice().getCursorPos().Y))
    driver.SetBraking(chrono.ChFunction_Recorder(myapplication.GetDevice().getCursorPos().Z))
```

This script creates a PyChrono simulation of a Gator vehicle on a flat rigid terrain using Irrlicht for visualization. It initializes the vehicle with parameters such as location, orientation, contact method, and TMEASY tire model. The script also sets up a RigidTerrain with specified dimensions and a custom texture. An interactive driver system is implemented to allow control over the vehicle's steering, throttle, and braking. The simulation loop updates vehicle dynamics and rendering at 50 frames per second in real-time.
