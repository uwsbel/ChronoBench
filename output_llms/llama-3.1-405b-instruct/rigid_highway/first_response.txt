Here is a PyChrono simulation script based on the provided instructions:

```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the PyChrono environment
chrono.SetChronoDataPath('/path/to/chrono/data/')

# Create a Chrono::Engine physical system
sys = chrono.ChSystemNSC()

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
sys.Add(vis)

# Create the vehicle
vehicle = veh.ChPart()
vehicle.SetMass(2000)
vehicle.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
vehicle.SetPos(chrono.ChVectorD(0, 0.5, 0))
vehicle.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI / 2))

# Create the vehicle's chassis
chassis = veh.ChChassis()
chassis.SetLength(4.8)
chassis.SetWidth(2.1)
chassis.SetHeight(1.8)
chassis.SetMass(1500)
chassis.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
vehicle.AddAsset(chassis)

# Create the vehicle's wheels and add them to the vehicle
wheel_fl = veh.ChWheel()
wheel_fl.SetMass(50)
wheel_fl.SetInertiaXX(chrono.ChVectorD(0.5, 0.5, 0.5))
wheel_fl.SetPos(chrono.ChVectorD(-1.2, -0.5, 1.1))
wheel_fl.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI / 2))
vehicle.AddAsset(wheel_fl)

wheel_fr = veh.ChWheel()
wheel_fr.SetMass(50)
wheel_fr.SetInertiaXX(chrono.ChVectorD(0.5, 0.5, 0.5))
wheel_fr.SetPos(chrono.ChVectorD(1.2, -0.5, 1.1))
wheel_fr.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI / 2))
vehicle.AddAsset(wheel_fr)

wheel_rl = veh.ChWheel()
wheel_rl.SetMass(50)
wheel_rl.SetInertiaXX(chrono.ChVectorD(0.5, 0.5, 0.5))
wheel_rl.SetPos(chrono.ChVectorD(-1.2, -0.5, -1.1))
wheel_rl.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI / 2))
vehicle.AddAsset(wheel_rl)

wheel_rr = veh.ChWheel()
wheel_rr.SetMass(50)
wheel_rr.SetInertiaXX(chrono.ChVectorD(0.5, 0.5, 0.5))
wheel_rr.SetPos(chrono.ChVectorD(1.2, -0.5, -1.1))
wheel_rr.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI / 2))
vehicle.AddAsset(wheel_rr)

# Create the TMEasy tire model for the vehicle's wheels
tire_fl = veh.ChTMEasyTire()
tire_fl.SetRadius(0.35)
tire_fl.SetWidth(0.2)
wheel_fl.AddAsset(tire_fl)

tire_fr = veh.ChTMEasyTire()
tire_fr.SetRadius(0.35)
tire_fr.SetWidth(0.2)
wheel_fr.AddAsset(tire_fr)

tire_rl = veh.ChTMEasyTire()
tire_rl.SetRadius(0.35)
tire_rl.SetWidth(0.2)
wheel_rl.AddAsset(tire_rl)

tire_rr = veh.ChTMEasyTire()
tire_rr.SetRadius(0.35)
tire_rr.SetWidth(0.2)
wheel_rr.AddAsset(tire_rr)

# Create the custom terrain using collision and visual meshes
terrain = veh.ChTerrain()
terrain.SetMesh(chrono.ChVectorD(-10, -10, 0), chrono.ChVectorD(10, 10, 0), 10, 10, chrono.ChVectorD(0, 0, 1))
terrain.SetCollisionMesh('Highway_col.obj')
terrain.SetVisualMesh('Highway_vis.obj')
sys.Add(terrain)

# Create the driver system
driver = veh.ChIrrNodeDriver()
driver.Initialize()
driver.SetSteeringDelta(0.01)
driver.SetThrottleDelta(0.01)
driver.SetBrakingDelta(0.01)
sys.Add(driver)

# Add the vehicle to the system
sys.Add(vehicle)

# Set the simulation loop to run in real time at 50 frames per second
while vis.Run():
    # Update the vehicle dynamics
    sys.DoStepDynamics(1 / 50)

    # Render the scene
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()
```

This script creates a PyChrono simulation of a full HMMWV on a custom mesh terrain using Irrlicht for visualization. The vehicle is initialized with specific parameters such as location, orientation, contact method, and TMEASY tire model, utilizing mesh visualization for all vehicle components. The script also implements a custom terrain using collision and visual meshes (Highway_col.obj and Highway_vis.obj), and configures an interactive driver system for steering, throttle, and braking control. The simulation loop runs in real time at 50 frames per second, updating vehicle dynamics and rendering the scene.

Please note that you need to replace '/path/to/chrono/data/' with the actual path to the Chrono data directory on your system. Also, make sure that the Highway_col.obj and Highway_vis.obj files are in the correct location and format for the script to work properly.
