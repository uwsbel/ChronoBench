Here's a sample code that simulates the dynamic behavior of a Jeffcott rotor using an Isogeometric Analysis (IGA) beam in PyChrono:

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.fea as fea
import numpy as np

# Initialize the PyChrono environment
chrono.SetChronoDataPath('./chrono_data/')

# Create a ChronoEngine physical system
system = chrono.ChSystemNSC()

# Create a mesh for the IGA beam
beam_length = 1.0
beam_mesh = fea.ChMesh()
fea.ChBeamIGASectionAdvanced(beam_mesh, 
                             beam_length, 
                             0.1,  # thickness
                             0.2,  # width
                             0.1,  # radius
                             10,   # degree
                             10)   # num patches

# Create an IGA beam element
beam_element = fea.ChElementBeamIGA(beam_mesh)
beam_element.SetLayerZ(0.5)  # Set the beam layer thickness

# Create a flywheel body
flywheel_radius = 0.2
flywheel_body = chrono.ChBodyEasySphere(chrono.ChVectorD(0, 0, 0), 
                                       flywheel_radius, 
                                       1000,  # density
                                       True,  # collide
                                       True)  # visualize

# Create a motor to drive one end of the beam
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(beam_element.GetNode(0), 
                 chrono.ChFrameD(chrono.ChVectorD(0, 0, 0), 
                                 chrono.Q_ROTATE_X_TO_Y))
motor.SetMotorFunction(chrono.ChFunction_Const(10))  # Set the motor speed

# Add the beam element, flywheel body, and motor to the system
system.Add(beam_element)
system.Add(flywheel_body)
system.Add(motor)

# Add constraints to attach the flywheel to the beam center
constraint = fea.ChLinkPointFrame()
constraint.Initialize(beam_element.GetNode(beam_mesh.GetNumNodes() // 2), 
                      flywheel_body, 
                      chrono.ChFrameD(chrono.ChVectorD(0, 0, 0), 
                                      chrono.Q_ROTATE_X_TO_Y))
system.Add(constraint)

# Set up FEM visualization
visualization = fea.ChVisualizationFEAmesh(beam_mesh)
visualization.SetFEMdataType(fea.ChVisualizationFEAmesh.E_PLOT_NODE_SPEED)
visualization.SetColorscaleMinMax(-1, 1)
visualization.SetShrinkElements(True, 0.5)
system.Add(visualization)

# Set up the Irrlicht window
application = chronoirr.ChIrrApp(system, 
                                  'Jeffcott Rotor Simulation', 
                                  chronoirr.dimension2du(800, 600))
application.AddTypicalSky()
application.AddTypicalLights()
application.AddCamera(chronoirr.vector3df(0, 0, 2))
application.AssetBindAll()
application.AssetUpdateAll()

# Simulate the system
application.SetTimestep(0.01)
application.SetTryRealtime(True)
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```

This code initializes a PyChrono environment, creates an IGA beam element, a flywheel body, and a motor to drive one end of the beam. It adds these components to the system, sets up FEM visualization and an Irrlicht window, and simulates the system.
