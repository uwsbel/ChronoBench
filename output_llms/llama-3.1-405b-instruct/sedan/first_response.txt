Here is a PyChrono simulation script based on the provided instructions:

```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the PyChrono environment
chrono.SetChronoDataPath('/path/to/chrono/data/')

# Create a Chrono::Engine physical system
sys = chrono.ChSystemNSC()

# Create the ground body
ground = chrono.ChBodyEasyBox(sys,  # Parent system
                              100, 100, 2,  # Box dimensions (x, y, z)
                              1000,  # Density
                              True,  # Visualization
                              True,  # Collision
                              chrono.ChMaterialSurfaceNSC())  # Contact material
ground.SetBodyFixed(True)
sys.Add(ground)

# Create the vehicle
vehicle = veh.ChPart(sys,  # Parent system
                     veh.ChBoxShapeDirections.UpDown,  # Shape direction
                     chrono.ChVectorD(0, 0, 1.5))  # Position
vehicle.SetMass(1500)
vehicle.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
sys.Add(vehicle)

# Create the TMEasy tire model
tire = veh.ChTMEasy(sys,  # Parent system
                    vehicle,  # Vehicle body
                    chrono.ChVectorD(0, 0, -1.5),  # Position
                    chrono.ChVectorD(0, 0, 1))  # Direction
tire.SetMass(20)
tire.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
tire.SetLongitudinalStiffness(100000)
tire.SetLateralStiffness(50000)
sys.Add(tire)

# Create an interactive driver system
driver = veh.ChIrrNodeDriver(sys,  # Parent system
                             vehicle,  # Vehicle body
                             tire)  # Tire
driver.SetSteeringIncrement(0.1)
driver.SetThrottleIncrement(0.1)
driver.SetBrakingIncrement(0.1)
sys.Add(driver)

# Create the Irrlicht visualization system
application = chronoirr.ChIrrApp(sys,  # Physical system
                                 'BMW E90 Sedan Simulation',  # Window title
                                 chronoirr.dimension2du(800, 600))  # Window size

# Add a chase camera
camera = application.AddCamera(chronoirr.vector3df(0, 0, 5),  # Position
                               chronoirr.vector3df(0, 0, 0))  # Target
camera.SetChaseCamera(True)
camera.SetTarget(vehicle)

# Add directional lighting
light = application.AddLight(chronoirr.vector3df(0, 10, 0),  # Position
                             chronoirr.vector3df(0, 0, 0),  # Target
                             100,  # Range
                             0xffffffff)  # Color

# Add a skybox
application.AddSkyBox()

# Customize the terrain surface
ground_texture = application.AddTexture('ground_texture.jpg')
ground_logo = application.AddTexture('ground_logo.png')
application.AddMaterialSurface(ground, ground_texture, ground_logo)

# Initialize the simulation
application.AssetBindAll()
application.AssetUpdateAll()

# Run the simulation
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```

This script initializes the PyChrono environment, creates a physical system with a ground body, vehicle, TMEasy tire model, and interactive driver system, and sets up the Irrlicht visualization system with a chase camera, directional lighting, skybox, and customizable textures and logos on the terrain surface. The simulation can be run in real-time using the interactive driver system.

Note that you'll need to replace the `/path/to/chrono/data/` with the actual path to the Chrono data directory on your system. Additionally, you'll need to create the `ground_texture.jpg` and `ground_logo.png` files and place them in the same directory as the script.
