```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr

# Set the path to the Chrono data directory
chrono.SetChronoDataPath('/path/to/chrono/data/')

# Create a Chrono::Engine physical system
sys = chrono.ChSystemNSC()

# Create the ground body
ground = chrono.ChBodyEasyBox(sys,  # Parent system
                              100, 100, 2,  # Box dimensions (x, y, z)
                              1000,  # Density
                              True,  # Visualization
                              True,  # Collision
                              chrono.ChMaterialSurfaceNSC())  # Contact material
ground.SetBodyFixed(True)
sys.Add(ground)

# Create the vehicle
vehicle = veh.ChPart(sys,  # Parent system
                     veh.ChPartType_VEHICLE)  # Part type
vehicle.SetMass(1500)  # Mass
vehicle.SetInertiaXX(2500)  # Inertia
vehicle.SetPos(chrono.ChVectorD(0, 0, 1.5))  # Position

# Add visualization assets for the vehicle
vehicle.AddVisualizationAssets('BMW_E90_Sedan.obj',  # 3D model
                               'BMW_E90_Sedan_diffuse.png',  # Diffuse texture
                               'BMW_E90_Sedan_normal.png')  # Normal texture

# Add the TMEasy tire model
tire_model = veh.ChTMEasy(sys,  # Parent system
                          vehicle,  # Vehicle
                          veh.ChTMEasyModelType_RIGID)  # Tire model type
vehicle.AddTire(tire_model)

# Create the interactive driver system
driver = veh.ChIrrNodeDriver(sys,  # Parent system
                            vehicle,  # Vehicle
                            chronoirr.ChIrrNodeDriverType_DEFAULT)  # Driver type
sys.Add(driver)

# Create the Irrlicht application
app = chronoirr.ChIrrApp(sys,  # Physical system
                         'BMW E90 Sedan Simulation',  # Window title
                         chronoirr.dimension2du(800, 600))  # Window size

# Add a chase camera
camera = chronoirr.ChIrrNodeCameraChaseTarget(app.GetSceneManager(),  # Scene manager
                                              vehicle,  # Target
                                              chronoirr.vector3df(0, 0, 2))  # Offset
app.GetSceneManager().AddCamera(camera)

# Add directional lighting
light = chronoirr.ChIrrNodeLightDirectional(app.GetSceneManager(),  # Scene manager
                                           chronoirr.vector3df(0, 0, 1))  # Direction
app.GetSceneManager().AddLight(light)

# Add a skybox
skybox = chronoirr.ChIrrNodeSkyBox(app.GetSceneManager(),  # Scene manager
                                   'skybox.obj',  # 3D model
                                   'skybox_diffuse.png',  # Diffuse texture
                                   'skybox_normal.png')  # Normal texture
app.GetSceneManager().AddSkyBox(skybox)

# Add customizable textures and logos on the terrain surface
terrain_texture = chronoirr.ChIrrNodeTexture(app.GetSceneManager(),  # Scene manager
                                            'terrain_diffuse.png')  # Texture
app.GetSceneManager().AddTexture(terrain_texture)

logo = chronoirr.ChIrrNodeLogo(app.GetSceneManager(),  # Scene manager
                               'logo.png',  # Texture
                               chronoirr.vector3df(0, 0, 1))  # Position
app.GetSceneManager().AddLogo(logo)

# Set the simulation time step and simulation duration
app.SetTimestep(0.01)
app.SetSimulationDuration(100)

# Run the simulation
app.GetDevice().run()
```
This script creates a PyChrono simulation of a BMW E90 Sedan driving on a rigid terrain. The vehicle is modeled using a TMEasy tire model and is driven by an interactive driver system. The simulation is visualized using the Irrlicht system with a chase camera, directional lighting, a skybox, and customizable textures and logos on the terrain surface.
