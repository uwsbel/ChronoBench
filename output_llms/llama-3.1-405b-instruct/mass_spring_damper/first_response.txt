```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr

# Initialize the PyChrono environment
chrono.SetChronoDataPath("/path/to/chrono/data/")

# Create a Chrono::Engine physical system
sys = chrono.ChSystemNSC()

# Create the ground body
ground = chrono.ChBodyEasyBox(sys,  # parent system
                              1, 1, 1,  # dimensions
                              1000,  # density
                              True,  # visualization?
                              True,  # collision?
                              chrono.ChVectorD(0, -0.5, 0),  # initial position
                              chrono.ChQuaternionD(1, 0, 0, 0))  # initial rotation

# Create the mass body
mass = chrono.ChBodyEasySphere(sys,  # parent system
                               0.1,  # radius
                               1000,  # density
                               True,  # visualization?
                               True,  # collision?
                               chrono.ChVectorD(0, 1, 0),  # initial position
                               chrono.ChQuaternionD(1, 0, 0, 0))  # initial rotation

# Create a ChLinkTSDA (spring-damper) between ground and mass
spring = chrono.ChLinkTSDA()
spring.Initialize(ground, mass, True, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 0, 0))
sys.Add(spring)

# Set default linear spring-damper parameters
spring.SetSpringCoefficient(100)
spring.SetDampingCoefficient(10)

# Create an Irrlicht application
app = chronoirr.ChIrrApp(sys, "Mass-Spring-Damper Demo", chronoirr.dimension2du(800, 600))

# Add a camera
app.AddCamera(chronoirr.vector3df(0, 0, -2), chronoirr.vector3df(0, 0, 0))

# Add lighting
app.AddLightWithShadow(chronoirr.vector3df(2, 4, 2), chronoirr.vector3df(0, 0, 0), 3, 2, 2, 30, 512)

# Add visual shapes for mass and spring
mass_shape = chronoirr.ChIrrNodeAsset()
mass_shape.SetSphere(0.1)
mass.AddAsset(mass_shape)

spring_shape = chronoirr.ChIrrNodeAsset()
spring_shape.SetCylinder(0.05, 0.05, 1)
spring.AddAsset(spring_shape)

# Run the simulation
app.SetTimestep(0.01)
app.SetTryRealtime(True)
while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()
    app.EndScene()
```
Note: Make sure to replace `"/path/to/chrono/data/"` with the actual path to your Chrono::Engine data directory.
