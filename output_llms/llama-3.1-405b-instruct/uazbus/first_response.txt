Here's a sample PyChrono simulation script based on the given instructions:

```python
# Import necessary PyChrono modules
import pychrono.core as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# Set the simulation step size
step_size = 1e-3

# Set the simulation end time
end_time = 100

# Create a Chrono::Engine physical system
sys = chrono.ChSystemNSC()

# Create the ground body
ground = chrono.ChBodyEasyBox(sys,  # Parent system
                              100, 100, 2,  # Box dimensions
                              1000,  # Density
                              True,  # Visualization
                              True,  # Collision
                              chrono.ChMaterialSurfaceNSC()  # Material
                              )
ground.SetPos(chrono.ChVectorD(0, -2, 0))
ground.SetBodyFixed(True)
sys.Add(ground)

# Create the vehicle
vehicle = veh.ChUAZBUS()
vehicle.Initialize(sys, chrono.ChContactMethod_NSC)
vehicle.SetChassisPosition(chrono.ChVectorD(0, 1, 0))
vehicle.SetChassisVelocity(chrono.ChVectorD(5, 0, 0))
sys.Add(vehicle)

# Create the driver system
driver = veh.ChIrrNodeDriver(vehicle)
driver.Initialize()

# Create the Irrlicht application
app = irr.ChIrrApp(sys, "UAZBUS Simulation", irr.dimension2du(800, 600))
app.AddTypicalLights()
app.AddTypicalCamera(chrono.ChVectorD(0, 5, -10))
app.AssetBindAll()
app.AssetUpdateAll()

# Simulation loop
while app.GetDevice().run():
    time = sys.GetChTime()
    if time >= end_time:
        break

    # Synchronize and advance the vehicle, terrain, driver, and visual system
    driver.Synchronize(time)
    sys.DoStepDynamics(step_size)
    app.BeginScene(True, True, irr.SColor(255, 140, 161, 192))
    app.DrawAll()
    app.EndScene()

# Close the Irrlicht application
app.GetDevice().closeDevice()
```

This script creates a PyChrono simulation with a UAZBUS vehicle and a rigid terrain. It also initializes a driver system for vehicle control and sets up a real-time visualization using Irrlicht. The simulation loop advances the vehicle, terrain, driver, and visual system at each timestep while maintaining real-time execution.
