```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Initialize the system
system = chrono.ChSystemNSC()

# Set collision envelope and margin
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.01)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.005)

# Create ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
system.Add(ground)

# Add visual shape to ground
visual_shape_ground = chronoirr.ChVisualShapeBox(chrono.ChVector3d(10, 0.1, 10))
ground.AddVisualShape(visual_shape_ground)

# Create a dynamic body
body1 = chrono.ChBody()
body1.SetPos(chrono.ChVector3d(0, 1, 0))
body1.SetMass(10)
body1.SetInertiaXX(chrono.ChVector3d(1, 1, 1))
system.Add(body1)

# Add visual shape to body1
visual_shape_body1 = chronoirr.ChVisualShapeSphere(0.2)
visual_shape_body1.SetColor(chrono.ChColor(0, 0, 1))
body1.AddVisualShape(visual_shape_body1)

# Create a second body
body2 = chrono.ChBody()
body2.SetPos(chrono.ChVector3d(1, 1, 0))
body2.SetMass(10)
body2.SetInertiaXX(chrono.ChVector3d(1, 1, 1))
system.Add(body2)

# Add visual shape to body2
visual_shape_body2 = chronoirr.ChVisualShapeBox(chrono.ChVector3d(0.2, 0.2, 0.2))
visual_shape_body2.SetColor(chrono.ChColor(1, 0, 0))
body2.AddVisualShape(visual_shape_body2)

# Create a revolute joint between body1 and body2
revolute_joint = chrono.ChLinkLockRevolute()
revolute_joint.Initialize(body1, body2, chrono.ChCoordsysD(chrono.ChVector3d(0.5, 1, 0), chrono.QUNIT))
system.Add(revolute_joint)

# Create a visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle('PyChrono Simulation')
vis.Initialize()

# Add skybox and lights
vis.AddSkyBox()
vis.AddTypicalLights()

# Simulation loop
time_step = 0.01
while vis.Run():
    system.DoStepDynamics(time_step)
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
```
