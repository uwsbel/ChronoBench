Here's a Python script that implements the specified PyChrono simulation:

```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.sensor as sensor
import pychrono.irrlicht as chronoirr
import numpy as np

# Initialize the PyChrono environment
chrono.SetChronoDataPath('./data/')

# Create the simulation system and scene
mysystem = chrono.ChSystemNSC()
contact_material = chrono.ChMaterialSurfaceNSC()
contact_material.SetFriction(0.9)
contact_material.SetRestitution(0.1)
contact_material.SetFrictionV(0.9)
mysystem.SetContactMaterial(contact_material)

# Create the visualization ( Irrlicht ) module
application = chronoirr.ChVisualSystemIrrlicht()
application.SetWindowSize(800, 600)
application.SetWindowTitle('PyChrono Simulation')
application.SetTypicalLenght(1.0)
application.SetCameraDistance(2.0)

# Create the Irrlicht device
application.CreateDevice()

# Create the mesh from a Wavefront .obj file
mesh = chrono.ChTriangleMesh()
mesh.LoadWavefrontOBJ('mesh.obj')

# Create a ChBody from the mesh
body = chrono.ChBodyEasyMesh(mysystem, mesh, True, True)
body.SetPos(chrono.ChVectorD(0, 0, 0))
body.SetBodyFixed(True)

# Add the body to the system
mysystem.Add(body)

# Create a lidar sensor manager
lidar_manager = sensor.ChSensorManager(mysystem)

# Create a lidar sensor
lidar = sensor.ChLidarSensor()
lidar.SetName('LidarSensor')
lidar.SetNoiseType(sensor.ChLidarNoiseType.GAUSSIAN)
lidar.SetNoiseParams(0.01, 0.01, 0.01, 0.01, 0.01, 0.01)
lidar.SetVisualization(True)
lidar.SetSaveData(True)
lidar.SetSaveDataPath('./data/lidar_data.csv')

# Add the lidar sensor to the sensor manager
lidar_manager.AddSensor(lidar)

# Attach the lidar sensor to the body
lidar_manager.AttachSensorToBody(body, chrono.ChFrameD(chrono.ChVectorD(0, 0, 1)))

# Set the simulation step and time
step = 0.01
time = 0

# Simulate the system
while application.GetDevice().run():
    # Update the lidar's position in an orbit around the mesh
    angle = time * 0.1
    lidar_manager.SetSensorPosition(chrono.ChVectorD(np.cos(angle), np.sin(angle), 1))

    # Advance the simulation
    mysystem.DoStepDynamics(step)
    application.BeginScene()
    application.DrawAll()
    application.EndScene()

    # Print lidar buffer data
    print(lidar.GetBuffer())

    # Increment time
    time += step

# Delete the visualization module
application.DeleteDevice()
```

This script initializes the PyChrono environment, creates a simulation system, and adds a triangular mesh from a Wavefront .obj file as a fixed body in the scene. A lidar sensor is attached to the body using a sensor manager, applying noise filters, visualizations, and saving options to the lidar data. The simulation is run while dynamically updating the lidar's position in an orbit around the mesh, and the lidar buffer data is printed at each simulation step.
