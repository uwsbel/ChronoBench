Here is the Python code that implements the PyChrono simulation script based on the provided instructions:
```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the PyChrono environment and core components
sys = chrono.ChSystemNSC()

# Create the UAZBUS vehicle with specified initial conditions
UAZBUS = veh.ChUAZBUS()
UAZBUS.SetChassisFixed(True)
UAZBUS.SetChassisMass(1500)
UAZBUS.SetChassisInertia(chrono.ChVectorD(1000, 1000, 1000))
UAZBUS.SetChassisPosition(chrono.ChVectorD(0, 0, 1))
UAZBUS.SetChassisRotation(chrono.ChQuaternionD(1, 0, 0, 0))
UAZBUS.AddVisualization(chrono.ChVisualizationType.PRIMITIVES)
UAZBUS.SetVisualizationType(chrono.ChVisualizationType.MESH)
UAZBUS.SetVisualizationFile("UAZBUS.obj")

# Create the rigid terrain with defined friction and restitution
terrain = chrono.ChBodyEasyBox(sys, 10, 10, 1, 1000, True, True)
terrain.SetPos(chrono.ChVectorD(0, 0, -1))
terrain.SetMaterial(chrono.ChMaterialSurfaceNSC())
terrain.GetMaterial().SetFriction(0.5)
terrain.GetMaterial().SetRestitution(0.5)

# Add the vehicle and terrain to the system
sys.Add(UAZBUS.GetChassisBody())
sys.Add(terrain)

# Initialize the driver system for vehicle control
driver = veh.ChIrrlichtDriver(UAZBUS, sys)
driver.SetSteeringDelta(0.1)
driver.SetThrottleDelta(0.1)
driver.SetBrakingDelta(0.1)

# Create a real-time visualization using Irrlicht with camera settings and lighting
viz = chronoirr.ChVisualSystemIrrlicht()
viz.SetWindowSize(1024, 768)
viz.SetWindowTitle("UAZBUS Simulation")
viz.AddCamera(chrono.ChVectorD(0, 0, 5), chrono.ChVectorD(0, 0, 0))
viz.AddLight(chrono.ChVectorD(0, 0, 5), chrono.ChVectorD(1, 1, 1), 10)

# Add the vehicle and terrain to the visualization
viz.Add(UAZBUS.GetChassisBody())
viz.Add(terrain)

# Simulation loop
tstep = 0.01
tend = 10

while sys.GetChTime() < tend:
    # Synchronize and advance the vehicle, terrain, driver, and visual system at each timestep
    sys.DoStepDynamics(tstep)
    driver.Advance(tstep)
    viz.DoStep(tstep)

    # Maintain real-time execution
    chrono.ChUtils::Sleep(tstep)

# Finalize the simulation
sys.Reset()
viz.Finalize()
```
This code initializes the PyChrono environment and core components, creates the UAZBUS vehicle with specified initial conditions, sets up a rigid terrain with defined friction and restitution, initializes a driver system for vehicle control, and creates a real-time visualization using Irrlicht with camera settings and lighting. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.
