Here's a Python script that implements the specified PyChrono simulation:

```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import pychrono.tire as tire

# Initialize the PyChrono environment
chrono.SetChronoDataPath('./data/')

# Create the simulation system
mysystem = chrono.ChSystemNSC()

# Create the vehicle
vehicle = veh.ChWheeledVehicle(mysystem)
vehicle.SetChassisBodyShape(chrono.ChBox(1.5, 0.8, 1.2))
vehicle.SetChassisBodyPos(chrono.ChVectorD(0, 0.5, 0))
vehicle.SetChassisBodyMass(1500)

# Add visualization to the vehicle
vehicle.SetChassisVisualization(chrono.ChVisualizationType.SPRITE)
vehicle.SetChassisVisualizationColor(chrono.ChColor(0.8, 0.2, 0.2))

# Add collision to the vehicle
vehicle.SetChassisCollisionType(chrono.ChCollisionType.BULLET)

# Add wheels and tires to the vehicle
vehicle.AddWheeledVehicleWheel(tire.TMEASY(), chrono.ChVectorD(-1.1, -0.5, 1.1), chrono.ChVectorD(0, 1, 0))
vehicle.AddWheeledVehicleWheel(tire.TMEASY(), chrono.ChVectorD(1.1, -0.5, 1.1), chrono.ChVectorD(0, 1, 0))
vehicle.AddWheeledVehicleWheel(tire.TMEASY(), chrono.ChVectorD(-1.1, -0.5, -1.1), chrono.ChVectorD(0, 1, 0))
vehicle.AddWheeledVehicleWheel(tire.TMEASY(), chrono.ChVectorD(1.1, -0.5, -1.1), chrono.ChVectorD(0, 1, 0))

# Create the terrain
terrain = chrono.ChBodyEasyBox(mysystem, 10, 10, 0.2, 1000, 0.1, 0.1, chrono.ChVectorD(0, -0.2, 0))
terrain.SetBodyFixed(True)

# Add visualization to the terrain
terrain.SetBodyVisualization(chrono.ChVisualizationType.MESH)
terrain.SetBodyVisualizationColor(chrono.ChColor(0.2, 0.8, 0.2))

# Add collision to the terrain
terrain.SetBodyCollisionType(chrono.ChCollisionType.BULLET)

# Add a driver system to the vehicle
driver = veh.ChIrrlichtDriver()
driver.SetVehicle(vehicle)
driver.SetSteeringMode(veh.ChIrrlichtDriver.STEERING_MODE_CONTRAINED)

# Create the Irrlicht visualization
application = chronoirr.ChVisualSystemIrrlicht()
application.SetWindowSize(1024, 768)
application.SetWindowTitle("BMW E90 Sedan Simulation")
application.AddTypicalLights()
application.AddSkyBox()
application.AddLogo(chrono.GetChronoDataPath() + "/logo_pychrono_alpha.png")

# Create a camera and attach it to the vehicle
camera = application.AddCamera(chrono.ChVectorD(0, 1.5, -3))
camera.SetFollowBody(vehicle.GetChassisBody())

# Set the simulation time step and tolerance
mysystem.SetChTimeStep(0.01)
mysystem.SetChSolverMaxIterations(30)
mysystem.SetChSolverTolerance(1e-9)

# Start the simulation
application.AssetBindAll()
application.AssetUpdateAll()
while application.Run():
    application.BeginScene()
    application.DrawAll()
    application.EndScene()
    mysystem.DoStepDynamics(0.01)
    application.Synchronize(True)
```

This script creates a PyChrono simulation of a BMW E90 Sedan driving on a rigid terrain. The vehicle is equipped with a TMEASY tire model and an interactive driver system for real-time control of steering, throttle, and braking. The simulation is visualized using the Irrlicht system with a chase camera, directional lighting, a skybox, and customizable textures and logos on the terrain surface.
