Here's a Python script that implements the simulation as described:

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as chrono_vehicle

# Initialize the PyChrono environment and core components
system = chrono.ChSystemNSC()
contact_material = chrono.ChMaterialSurfaceNSC()
contact_material.SetFriction(0.9)
contact_material.SetRestitution(0.1)
contact_material.SetYoungModulus(1e8)
system.SetContactMaterial(contact_material)

# Create the vehicle
vehicle = chrono_vehicle.ChCityBus()
vehicle.SetChassisFixed(False)
vehicle.SetChassisMass(1500)
vehicle.SetChassisInertiaXX(chrono.ChVectorD(100, 100, 100))
vehicle.SetChassisInertiaYY(chrono.ChVectorD(100, 100, 100))
vehicle.SetChassisInertiaZZ(chrono.ChVectorD(100, 100, 100))
vehicle.SetChassisPosition(chrono.ChVectorD(0, 0.5, 0))
vehicle.SetChassisOrientation(chrono.ChQuaternionD(1, 0, 0, 0))

# Create the terrain
terrain = chrono.ChRigidTerrain()
terrain.SetTexture(chrono.ChTexture("terrain_texture.jpg"))
terrain.SetPlane(chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 1, 0))

# Create the visualization
irrApp = chronoirr.ChVisualSystemIrrlicht()
irrApp.SetWindowSize(1024, 768)
irrApp.SetWindowTitle("CityBus Simulation")
irrApp.SetTypicalLenght(1.0)
irrApp.SetCameraDistance(1.5)
irrApp.SetCameraAzimuth(45)
irrApp.SetCameraElevation(30)
irrApp.SetCameraPosition(chrono.ChVectorD(0, 0.5, 0))

# Add the vehicle and terrain to the visualization
irrApp.AddVehicle(vehicle)
irrApp.AddTerrain(terrain)

# Set up the driver system
driver = chrono_vehicle.ChIrrlichtDriver()
driver.SetVehicle(vehicle)
driver.SetCamera(irrApp.GetCamera())
driver.SetSteeringMin(-0.5)
driver.SetSteeringMax(0.5)
driver.SetThrottleMin(0)
driver.SetThrottleMax(100)
driver.SetBrakingMin(0)
driver.SetBrakingMax(100)

# Set up the simulation loop
step_size = 0.02
time_end = 10
while system.GetChTime() < time_end:
    # Update the vehicle dynamics
    vehicle.UpdateDynamics(step_size)
    terrain.UpdateDynamics(step_size)
    
    # Update the visualization
    irrApp.BeginScene(True, True, chrono.ChColor(0.5, 0.5, 0.5))
    irrApp.DrawAll()
    irrApp.EndScene()
    
    # Update the driver system
    driver.UpdateDriverInput()
    
    # Advance the simulation time
    system.DoStepDynamics(step_size)

# Clean up
del driver
del irrApp
del terrain
del vehicle
del system
```

This script creates a CityBus vehicle and a rigid terrain, and sets up a simulation loop that updates the vehicle dynamics and renders the scene in real time. The driver system is interactive, allowing the user to control the vehicle's steering, throttle, and braking. The simulation runs at 50 frames per second, and the camera follows the vehicle from a specified position.
