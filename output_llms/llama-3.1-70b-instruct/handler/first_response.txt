```
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono_ros as chrono_ros
import rospy
from std_msgs.msg import Int32

# Initialize PyChrono environment and core components
system = chrono.ChSystemNSC()
material = chrono.ChMaterialSurfaceNSC()
material.SetFriction(0.4)
material.SetRestitution(0.1)
material.SetFriction(0.4)
material.SetCompliance(0.00001)
material.SetDampingF(0.00001)

# Set up gravitational acceleration
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create a custom ROS handler
class MyRosHandler(chrono_ros.ChRosHandler):
    def __init__(self):
        super().__init__()
        self.pub = rospy.Publisher('my_topic', Int32, queue_size=10)

    def Advance(self, mstep):
        msg = Int32()
        msg.data = mstep
        self.pub.publish(msg)

# Create a ROS manager and add handlers
ros_manager = chrono_ros.ChRosManager()
ros_manager.AddHandler(chrono_ros.ChRosClockHandler(system))
ros_manager.AddHandler(chrono_ros.ChRosBodyHandler(system))
ros_manager.AddHandler(chrono_ros.ChRosTransformHandler(system))
ros_manager.AddHandler(MyRosHandler())

# Create a Chrono::Engine::irrlicht module
irrlicht_module = chronoirr.ChIrrlichtModule()
irrlicht_module.SetWindowSize(800, 600)
irrlicht_module.SetWindowTitle("PyChrono Simulation")
irrlicht_module.SetShowInfo(True)

# Create a fixed floor
floor = chrono.ChBodyEasyBox(system, 10, 1, 10, 1000, True, material)
floor.SetPos(chrono.ChVectorD(0, -1, 0))
floor.SetBodyFixed(True)
system.Add(floor)

# Create a movable box
box = chrono.ChBodyEasyBox(system, 1, 1, 1, 1000, True, material)
box.SetPos(chrono.ChVectorD(0, 2, 0))
box.SetMass(10)
box.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
system.Add(box)

# Set up the simulation loop
realtime_timer = chrono.ChRealtimeStepTimer()
while True:
    system.DoStepDynamics(0.01)
    ros_manager.Advance(system.GetChTime())
    irrlicht_module.DrawAll()
    rospy.sleep(realtime_timer.GetSleepTime())
```
This code initializes the PyChrono environment and core components, sets up a Chrono system with gravitational acceleration and physical material properties, adds a fixed floor and a movable box to the system, creates a custom ROS handler to publish integer messages to a specified topic, configures the ROS manager with clock, body, and transform handlers, and registers the custom handler. The simulation loop advances the system and updates ROS communication at each timestep, while maintaining real-time execution.
