Here's a Python script that implements the PyChrono simulation based on the provided instructions:

```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import numpy as np

# Initialize the PyChrono environment and core components
chrono.SetChronoDataPath("/path/to/data")

# Create the Chrono system
system = chrono.ChSystemNSC()

# Set gravity
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the ground body
ground = chrono.ChBodyEasyBox(system, 10, 1, 10, 1000, True, chrono.ChVectorD(0, -1, 0))
ground.SetBodyFixed(True)

# Create the Viper rover
viper = veh.VehicleViper(system)

# Initialize the Viper rover
viper.Initialize(chrono.ChCoordsysD(chrono.ChVectorD(0, 0.4, 0), chrono.Q_from_AngX(chrono.CH_C_PI_2)))

# Set the Viper rover's collision model
viper.SetCollide(True)

# Create the Viper rover's driver
driver = veh.ChIrrVehicleDriver(viper)

# Set the driver's input
driver.SetInput(chrono.ChVectorD(0, 0, 0))

# Create the Irrlicht visualization
app = chronoirr.ChVisualSystemIrrlicht()
app.SetWindowSize(1024, 768)
app.SetWindowTitle("Viper Rover Simulation")
app.SetTypicalLenght(1.0)
app.SetCameraRig(chrono.ChVectorD(0, 1, -3), chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 1, 0))
app.SetCameraLookAtPoint(chrono.ChVectorD(0, 0, 0))
app.SetCameraUpVector(chrono.ChVectorD(0, 1, 0))
app.AddSkyBox()
app.AddTypicalLights()

# Add the Viper rover to the visualization
app.Add(viper.GetChassisBody())
app.Add(ground)

# Set the simulation loop parameters
tstep = 0.01
tend = 10.0
t = 0.0

# Simulation loop
while app.Run() and t < tend:
    # Update the Viper rover's steering
    steering = np.sin(t)
    driver.SetInput(chrono.ChVectorD(steering, 0, 0))

    # Update the simulation
    system.DoStepDynamics(tstep)

    # Render the simulation
    app.BeginScene()
    app.DrawAll()
    app.EndScene()

    # Increment time
    t += tstep

# Close the Irrlicht visualization
app.Close()
```

This script initializes the PyChrono environment, creates a Viper rover on a rigid terrain, and sets up a real-time visualization using Irrlicht. The simulation loop updates the rover's steering over a specified time period, updating and rendering the rover at each timestep.
