```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.engine as chronoengine

# Initialize the PyChrono environment
chrono.SetChronoDataPath('./')

# Create the simulation system
mysystem = chrono.ChSystemNSC()

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht(mysystem)

# Set up the visualization window
vis.SetWindowSize(800, 600)
vis.SetWindowTitle("Epicyclic Gears Simulation")
vis.SetCamera(chrono.ChVectorD(0, 0, 1.5), chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 1, 0))

# Create a fixed truss
truss = chrono.ChBodyEasyBox(mysystem, 2, 0.1, 0.1, 1000, True, False)
truss.SetPos(chrono.ChVectorD(0, 0, 0))
truss.SetBodyFixed(True)
mysystem.Add(truss)

# Create a rotating bar
bar = chrono.ChBodyEasyCylinder(mysystem, 1, 0.1, 1000, True, False)
bar.SetPos(chrono.ChVectorD(0, 0, 0.5))
bar.SetRot(chrono.ChQuaternionD(chrono.ChVectorD(0, 1, 0), chrono.ChAngleD(chrono.CH_C_PI / 2)))
mysystem.Add(bar)

# Create a motor to enforce a constant rotation speed
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(bar, truss, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))
motor.SetSpeed(chrono.ChAngleD(chrono.CH_C_PI))
mysystem.Add(motor)

# Create two gears
gear1 = chrono.ChBodyEasyCylinder(mysystem, 0.5, 0.1, 1000, True, False)
gear1.SetPos(chrono.ChVectorD(1, 0, 0))
gear1.SetRot(chrono.ChQuaternionD(chrono.ChVectorD(0, 1, 0), chrono.ChAngleD(chrono.CH_C_PI / 2)))
mysystem.Add(gear1)

gear2 = chrono.ChBodyEasyCylinder(mysystem, 0.5, 0.1, 1000, True, False)
gear2.SetPos(chrono.ChVectorD(-1, 0, 0))
gear2.SetRot(chrono.ChQuaternionD(chrono.ChVectorD(0, 1, 0), chrono.ChAngleD(chrono.CH_C_PI / 2)))
mysystem.Add(gear2)

# Create contact materials and add them to the system
material = chrono.ChMaterialSurfaceNSC()
material.SetFriction(0.4)
material.SetDampingF(0.0)
material.SetCompliance(0.0001)
material.SetComplianceT(0.0001)

mysystem.SetDefaultMaterial(material)

# Set up the visualization materials and lighting
vis.AddSkyBox()
vis.AddTypicalLights()
vis.AddCameraVerticalAll()
vis.SetShowAxes(True)
vis.SetShowBodyCollisions(True)
vis.SetShowContactForces(True)
vis.SetShowContactWrenches(True)

# Simulate the system
while vis.Run():
    mysystem.DoStepDynamics(0.01)
    vis.BeginScene()
    vis.DrawAll()
    vis.EndScene()
```

This script initializes a PyChrono simulation with a fixed truss, a rotating bar, and two gears. It uses the Irrlicht visualization system to display the simulation. The script sets up the necessary physical systems, objects, and interactions, including a motor to enforce a constant rotation speed. The simulation is then run using the `DoStepDynamics` method, and the visualization is updated using the `BeginScene`, `DrawAll`, and `EndScene` methods.
