Here's a Python script that implements the specified PyChrono simulation:

```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import numpy as np

# Initialize the PyChrono environment
chrono.SetChronoDataPath('./data/')

# Create the simulation system and add it to the environment
mysystem = chrono.ChSystemNSC()
chrono.ChCollisionSystem.SetDefaultSuggestedEnvelope(0.001)

# Create the ground
ground = chrono.ChBodyEasyBox(mysystem, 10, 10, 1, 1000, True, False)
ground.SetPos(chrono.ChVectorD(0, -1, 0))
ground.SetBodyFixed(True)

# Create the vehicle
veh_hmmwv = veh.HMMWV(mysystem)

# Add the vehicle to the simulation system
mysystem.Add(veh_hmmwv.GetChassisBody())
mysystem.Add(veh_hmmwv.GetBody())
mysystem.Add(veh_hmmwv.GetSuspension())
mysystem.Add(veh_hmmwv.GetSteering())
mysystem.Add(veh_hmmwv.GetTires())
mysystem.Add(veh_hmmwv.GetWheels())

# Create the terrain
terrain = veh.RigidTerrain(mysystem)
terrain.SetPlane(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngX(chrono.CH_C_PI_2)))

# Create the driver
driver = veh.ChIrrGuiDriver(veh_hmmwv)

# Create the visual system
application = chronoirr.ChVisualSystemIrrlicht()
application.SetWindowSize(1024, 768)
application.AddTypicalLights()
application.AddSkyBox()
application.AddCamera(chrono.ChVectorD(0, 1, -3))
application.AddCamera(chrono.ChVectorD(0, 1, 0))
application.SetChaseCamera(chrono.ChVectorD(0, 1, 1.5), chrono.ChVectorD(0, 1, 1.8))
application.SetLogoTexture(chrono.GetChronoDataFile("logo.png"))
application.SetTypicalLegoCamera()
application.SetShowLogo(False)
application.SetUpAxis(chrono.VECT_Y)
application.SetShowFPS(True)

# Create the IMU and GPS sensors
imu = veh.ChSensorIMU(chrono.ChVectorD(0, 1, 0))
gps = veh.ChSensorGPS(chrono.ChVectorD(0, 1, 0))

# Add the sensors to the vehicle
veh_hmmwv.AddSensor(imu)
veh_hmmwv.AddSensor(gps)

# Set the simulation step and time
step_size = 1e-3
sim_time = 10

# Simulation loop
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()

    # Update the driver inputs
    driver.Synchronize(chrono.ChTime(0, 1))

    # Advance the simulation
    mysystem.DoStepDynamics(step_size)

    # Process the sensor data
    imu.Process(mysystem.GetChTime())
    gps.Process(mysystem.GetChTime())

    # Output the vehicle mass
    print("Vehicle mass:", veh_hmmwv.GetChassisBody().GetMass())

    # Advance the visual system
    application.DoStepVisual()
    application.EndScene()

    # Check for simulation end
    if mysystem.GetChTime() >= sim_time:
        break

# Finalize the simulation
application.GetDevice().closeDevice()
```

This script sets up a PyChrono simulation featuring an HMMWV vehicle with a full system setup, including terrain, driver inputs, and visual interface using Irrlicht. It also adds and manages IMU and GPS sensors attached to the vehicle's chassis, processing and updating sensor data at each simulation step. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system modules while outputting vehicle mass and rendering scenes.
