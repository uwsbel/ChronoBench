Here is the PyChrono simulation script based on the provided instructions:
```python
# Import necessary modules
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import numpy as np

# Initialize the PyChrono environment
chrono.SetChronoDataPath('./data/')

# Create the simulation system and add it to the environment
mysystem = chrono.ChSystemNSC()
chrono.ChCollisionSystem.SetCollisionEnvelope(0.01)

# Create the vehicle object and add it to the system
hmmwv = veh.HMMWV(mysystem)
hmmwv.SetChassisVisualization(chrono.ChVisualizationType.MESH)
hmmwv.SetSuspensionVisualization(chrono.ChVisualizationType.MESH)
hmmwv.SetSteeringVisualization(chrono.ChVisualizationType.MESH)
hmmwv.SetWheelVisualization(chrono.ChVisualizationType.MESH)

# Set the vehicle's initial location and orientation
hmmwv.SetChassisPosition(chrono.ChVectorD(0, 0.5, 0))
hmmwv.SetChassisOrientation(chrono.ChQuaternionD(1, 0, 0, 0))

# Set the vehicle's contact method and tire model
hmmwv.SetContactMethod(chrono.ChContactMethodType.SMC)
hmmwv.SetTireModel(veh.TireModelType.TMEASY)

# Create the custom terrain object and add it to the system
terrain = chrono.ChBodyEasyMesh(mysystem)
terrain.SetMesh(chrono.ChMesh(chrono.GetChronoDataFile('Highway_col.obj')))
terrain.SetCollisionModel(chrono.ChCollisionModelType.BRUTE_FORCE)
terrain.SetBodyFixed(True)

# Create the visual terrain object and add it to the system
terrain_vis = chrono.ChBodyEasyMesh(mysystem)
terrain_vis.SetMesh(chrono.ChMesh(chrono.GetChronoDataFile('Highway_vis.obj')))
terrain_vis.SetCollisionModel(chrono.ChCollisionModelType.NONE)
terrain_vis.SetBodyFixed(True)

# Create the interactive driver system
driver = veh.HMMWVDriver(hmmwv)
driver.SetSteeringMode(veh.SteeringModeType.INTERACTIVE)
driver.SetThrottleMode(veh.ThrottleModeType.INTERACTIVE)
driver.SetBrakingMode(veh.BrakingModeType.INTERACTIVE)

# Create the Irrlicht application
app = chronoirr.ChVisualSystemIrrlicht()
app.SetWindowSize(1024, 768)
app.SetWindowTitle('HMMWV Simulation')
app.AddTypicalLights()
app.AddSkyBox()
app.AddCamera(chrono.ChVectorD(0, 1, -3), chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 1, 0))
app.AddCamera(chrono.ChVectorD(0, 1, -3), chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 1, 0))

# Set up the simulation loop
while app.Run():
    # Update the simulation
    mysystem.DoStepDynamics(0.02)
    driver.Update(0.02)

    # Render the scene
    app.BeginScene()
    app.DrawAll()
    app.EndScene()

    # Limit the frame rate to 50 FPS
    app.GetDevice().sleep(20)
```
This script initializes the PyChrono environment and creates the necessary physical systems and objects, including the HMMWV vehicle, custom terrain, and interactive driver system. It also sets up the simulation loop to run in real-time at 50 frames per second, updating the vehicle dynamics and rendering the scene using Irrlicht.
