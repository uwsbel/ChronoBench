# Custom force functor for spring force
class MySpringForce(chrono.ForceFunctor):
    def __init__(self, spring_coef, damping_coef):
        super().__init__()
        self.spring_coef = spring_coef
        self.damping_coef = damping_coef

    def Evaluate(self, time, id, P, V, F):
        # Evaluate the spring force using custom parameters
        F = chrono.ChVector3D(0, 0, 0)
        F += self.spring_coef * (P - chrono.ChVector3D(0, 0, 0))
        F += self.damping_coef * V
        return F
