```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr
import os

def main():
    # 0. Set the path to the Chrono data files (important for models and textures)
    chrono.SetChronoDataPath(veh.GetDataFile(""))

    # 1. Initialize the PyChrono environment and core components
    # Create a Chrono physical system
    system = chrono.ChSystemNSC()
    system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0)) # Gravity
    # Set solver settings if needed (default is usually fine for demos)
    system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)
    system.SetSolverMaxIterations(150)
    system.SetMaxPenetrationRecoverySpeed(4.0)


    # 2. Add the required physical systems and objects
    # Create the Kraz vehicle
    kraz_vehicle = veh.Kraz(system)

    # Set initial position and orientation of the vehicle
    # ChCoordsysD(ChVectorD position, ChQuaternion rotation)
    # Position: x, y (height), z
    # Rotation: Q_from_AngZ(angle_in_radians) for yaw
    initial_pos = chrono.ChVectorD(0, 0.6, 0) # Start slightly above ground
    initial_rot = chrono.Q_from_AngZ(0) # No initial yaw
    kraz_vehicle.SetInitPosition(chrono.ChCoordsysD(initial_pos, initial_rot))

    kraz_vehicle.SetEngineType(veh.EngineModelType_SIMPLE_MAP)
    kraz_vehicle.SetTransmissionType(veh.TransmissionModelType_AUTOMATIC_SIMPLE_MAP)
    kraz_vehicle.SetDriveType(veh.DrivelineTypeWV_AWD) # Wheeled Vehicle All-Wheel Drive
    kraz_vehicle.SetTireType(veh.TireModelType_TMEASY) # Using TMeasy tire model

    # Initialize the vehicle (this creates the subsystems)
    kraz_vehicle.Initialize()

    # Set visualization type for vehicle components
    kraz_vehicle.SetChassisVisualizationType(veh.VisualizationType_MESH)
    kraz_vehicle.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
    kraz_vehicle.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
    kraz_vehicle.SetWheelVisualizationType(veh.VisualizationType_MESH)
    for axle in kraz_vehicle.GetAxles():
        for wheel in axle.GetWheels():
            wheel.SetTireVisualizationType(veh.VisualizationType_MESH)


    # Create the rigid terrain
    terrain = veh.RigidTerrain(system)
    # Set contact material properties for the terrain
    # Parameters: friction_coefficient, restitution_coefficient, Young's modulus, Poisson's ratio
    # Note: Young's modulus and Poisson's ratio are less critical for NSC than for SMC
    contact_material = chrono.ChMaterialSurfaceNSC()
    contact_material.SetFriction(0.9)       # Coefficient of friction
    contact_material.SetRestitution(0.01)   # Coefficient of restitution
    terrain.SetContactMaterial(contact_material)

    # Initialize the terrain as a flat patch
    # Parameters: height, length_x, width_y
    terrain_height = 0.0
    terrain_length = 200.0 # meters
    terrain_width = 200.0  # meters
    terrain.Initialize(terrain_height, terrain_length, terrain_width)
    
    # Optionally, set a texture for the terrain
    texture_file = veh.GetDataFile("terrain/textures/tile4.jpg")
    if os.path.exists(texture_file):
        terrain.SetTexture(texture_file, 200, 200) # Texture file, length_repeat, width_repeat
    else:
        print(f"Warning: Terrain texture file not found: {texture_file}")


    # 3. Initialize a driver system for vehicle control
    # Create an Irrlicht-based GUI driver
    # This driver is controlled by keyboard inputs in the Irrlicht window
    # First, we need to create the Irrlicht visualization system

    # Create the Irrlicht visualization system
    vis = chronoirr.ChVisualSystemIrrlicht()
    vis.SetWindowSize(1280, 720)
    vis.SetWindowTitle('Kraz Vehicle Simulation')
    vis.Initialize()
    vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
    vis.AddSkyBox()
    vis.AddCamera(chrono.ChVectorD(15, 10, 20), chrono.ChVectorD(0, 0, 0)) # Camera position and target
    vis.AddTypicalLights()
    
    # Attach the Chrono physical system to the Irrlicht visualization
    vis.AttachSystem(system) # This is crucial for Irrlicht to know what to render

    # Now create the driver system and link it to the vehicle and Irrlicht app
    driver = veh.ChIrrGuiDriver(vis)
    driver.SetSteeringDelta(0.04)  # Steering sensitivity
    driver.SetThrottleDelta(0.04)  # Throttle sensitivity
    driver.SetBrakingDelta(0.1)    # Braking sensitivity
    driver.Initialize()


    # 4. Implement a simulation loop
    # Simulation parameters
    timestep = 0.005  # s (Corresponds to 200 Hz) - TMeasy tires may need smaller steps
    system.SetTimestepperType(chrono.ChTimestepper.Type_EULER_IMPLICIT_PROJECTED) # Good for vehicle dynamics

    # Real-time synchronization
    realtime_timer = chrono.ChRealtimeStepTimer()

    # Simulation loop
    while vis.Run():
        time = system.GetChTime()

        # Get driver inputs
        driver_inputs = driver.GetInputs()
        steering_input = driver_inputs.m_steering
        throttle_input = driver_inputs.m_throttle
        braking_input = driver_inputs.m_braking

        # Update modules (process inputs from other modules)
        driver.Synchronize(time)
        kraz_vehicle.Synchronize(time, steering_input, braking_input, throttle_input, terrain)
        terrain.Synchronize(time) # Terrain might not do much in Synchronize for rigid terrain

        # Advance simulation for one timestep
        driver.Advance(timestep)
        kraz_vehicle.Advance(timestep)
        terrain.Advance(timestep)
        system.DoStepDynamics(timestep)

        # Update Irrlicht visualization
        vis.BeginScene()
        vis.Render()
        # Optional: Draw driver inputs on screen (HUD)
        driver.DrawAll(vis) # Renders the Irrlicht GUI elements for driver controls
        vis.EndScene()

        # Maintain real-time execution
        realtime_timer.Spin(timestep)

    print("Simulation ended.")

if __name__ == '__main__':
    main()
```

**Explanation:**

1.  **Initialization (SetChronoDataPath, ChSystemNSC):**
    *   `chrono.SetChronoDataPath(veh.GetDataFile(""))`: This is crucial. It tells PyChrono where to find mesh files, textures, and other data associated with vehicle models and terrain.
    *   `chrono.ChSystemNSC()`: Creates the main physics system. `NSC` stands for Non-Smooth Contact, suitable for many rigid body simulations including vehicles.
    *   `system.Set_G_acc()`: Sets the gravitational acceleration.
    *   Solver settings like `SetSolverType`, `SetSolverMaxIterations`, `SetMaxPenetrationRecoverySpeed` can be tuned for performance and stability.

2.  **Kraz Vehicle Setup:**
    *   `veh.Kraz(system)`: Instantiates the Kraz vehicle model and associates it with our physics system.
    *   `SetInitPosition()`: Defines the starting position (x, y, z) and orientation (as a quaternion) of the vehicle. `Q_from_AngZ(0)` means no initial yaw.
    *   `SetEngineType`, `SetTransmissionType`, `SetDriveType`, `SetTireType`: Configure the vehicle's powertrain and tire model. `TMEASY` is a relatively simple and fast tire model.
    *   `kraz_vehicle.Initialize()`: This is a critical step. It finalizes the vehicle setup, creating all its internal components (chassis, axles, wheels, driveline, etc.).
    *   `Set...VisualizationType()`: Controls how different parts of the vehicle are rendered (e.g., using detailed meshes or simpler geometric primitives).

3.  **Rigid Terrain Setup:**
    *   `veh.RigidTerrain(system)`: Creates a rigid, non-deformable terrain.
    *   `SetContactMaterial()`: Defines the frictional and restitution properties of the terrain surface. This uses a `ChMaterialSurfaceNSC` object for NSC systems.
    *   `terrain.Initialize(height, length, width)`: Creates a flat rectangular patch of terrain.
    *   `terrain.SetTexture()`: (Optional) Applies a visual texture to the terrain.

4.  **Irrlicht Visualization & Driver:**
    *   `chronoirr.ChVisualSystemIrrlicht()`: Creates the Irrlicht rendering window.
    *   `vis.SetWindowSize()`, `vis.SetWindowTitle()`, `vis.Initialize()`: Basic window setup.
    *   `vis.AddLogo()`, `vis.AddSkyBox()`, `vis.AddCamera()`, `vis.AddTypicalLights()`: Adds common visual elements.
    *   `vis.AttachSystem(system)`: **Very important!** This links the physics system to the visualization system, so Irrlicht knows what bodies and shapes to draw.
    *   `veh.ChIrrGuiDriver(vis)`: Creates a driver model that takes input from the keyboard via the Irrlicht window.
    *   `driver.SetSteeringDelta()`, etc.: Configure how sensitive the controls are.
    *   `driver.Initialize()`: Initializes the driver system.

5.  **Simulation Loop:**
    *   `timestep`: The duration of each simulation step. Smaller timesteps increase accuracy but also computation time. `0.005s` (200 Hz) is a reasonable starting point. For TMeasy tires, `0.001s` or `0.002s` might be more stable.
    *   `system.SetTimestepperType()`: Selects the integration scheme. `EULER_IMPLICIT_PROJECTED` is often a good choice.
    *   `chrono.ChRealtimeStepTimer()`: Helps to run the simulation at a speed close to real-time by introducing pauses if the simulation runs faster than wall-clock time.
    *   `while vis.Run()`: The main loop continues as long as the Irrlicht window is open.
    *   **Get Driver Inputs:** `driver.GetInputs()` retrieves steering, throttle, and braking commands.
    *   **Synchronization (`.Synchronize()`):**
        *   `driver.Synchronize(time)`: Updates the driver's internal state based on the current simulation time.
        *   `kraz_vehicle.Synchronize(time, steering, braking, throttle, terrain)`: Passes the driver inputs and terrain information to the vehicle. The vehicle uses terrain information for tire force calculations.
        *   `terrain.Synchronize(time)`: For rigid terrain, this usually doesn't do much, but it's good practice for consistency, especially if you switch to deformable terrain.
    *   **Advancement (`.Advance()`):**
        *   `driver.Advance(timestep)`: Advances the driver's internal logic.
        *   `kraz_vehicle.Advance(timestep)`: Calculates forces and updates the state of all vehicle components based on the synchronized inputs and internal dynamics.
        *   `terrain.Advance(timestep)`: For rigid terrain, this is minimal.
    *   **Physics Step (`system.DoStepDynamics(timestep)`):** This is where the Chrono solver integrates the equations of motion for all objects in the system.
    *   **Visualization Update:**
        *   `vis.BeginScene()`: Clears the rendering buffer.
        *   `vis.Render()`: Draws all visual assets attached to the bodies in the `system`.
        *   `driver.DrawAll(vis)`: Renders the on-screen display (HUD) for the `ChIrrGuiDriver` which shows current throttle/steering/braking levels.
        *   `vis.EndScene()`: Displays the rendered frame.
    *   **Real-time Pacing:** `realtime_timer.Spin(timestep)` ensures the loop doesn't run faster than real-time.

To run this script:
1.  Make sure you have PyChrono installed (`pip install pychrono`).
2.  Ensure the Chrono data files are accessible (PyChrono usually handles this if installed correctly, but `SetChronoDataPath` makes it explicit).
3.  Save the code as a `.py` file (e.g., `kraz_sim.py`) and run it from your terminal: `python kraz_sim.py`.
    You should see an Irrlicht window appear with the Kraz vehicle on a flat terrain. You can control it using:
    *   `A` / `D`: Steering left/right
    *   `W` / `S`: Throttle/Brake
    *   Use the mouse to move the camera.
