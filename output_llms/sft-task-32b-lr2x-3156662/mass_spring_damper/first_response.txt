

```python
import pychrono as chrono
import pychrono.irrlicht as irr
import numpy as np

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, 0))  # No gravity for 1D spring demo

# Create ground body with coordinate system for spring anchor
ground = chrono.ChBody()
ground.SetName("Ground")
ground.SetFixed(True)
ground.EnableCollision(False)
system.AddBody(ground)

# Create mass body (1 kg point mass represented as sphere)
mass = chrono.ChBodyEasySphere(0.1, 1000, True, True)
mass.SetName("Mass")
mass.SetMass(1.0)
mass.SetInertiaXX(chrono.ChVector3d(0.01, 0.01, 0.01))
mass.SetPos(chrono.ChVector3d(1, 0, 0))
system.AddBody(mass)

# Create TSDA spring-damper link between ground and mass
spring = chrono.ChLinkTSDA()
spring.SetName("Spring-Damper")
spring.Initialize(mass, ground, True, chrono.ChVector3d(0, 0, 0), chrono.ChVector3d(0, 0, 0))
spring.SetRestLength(1.0)
spring.SetSpringCoefficient(100.0)
spring.SetDampingCoefficient(1.0)
system.AddLink(spring)

# Add visual shapes
spring_shape = chrono.ChVisualShapeSpring(0.05, 50)
spring.AddVisualShape(spring_shape)

mass_shape = chrono.ChVisualShapeSphere(0.1)
mass.AddVisualShape(mass_shape)

# Set up Irrlicht visualization
app = irr.ChIrrApp(system, "Mass-Spring-Damper System", irr.dimension2du(1024, 768))
app.SetSkyBox()
app.SetCamera(chrono.ChVector3d(2, 2, 4), chrono.ChVector3d(0, 0, 0))
app.SetLightDirection(chrono.ChVector3d(0, 0, 1), True)
app.AssetBindAll()
app.AssetUpdateAll()

# Simulation loop parameters
TIME_STEP = 0.001
TOTAL_TIME = 5.0

app.SetTimestep(TIME_STEP)

while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()
    
    # Display spring parameters
    irr.GuiEngine.GetText().Add(20, 40, "Press ESC to exit")
    irr.GuiEngine.GetText().Add(20, 60, f"Mass position: {mass.GetPos().x:.4f} m")
    irr.GuiEngine.GetText().Add(20, 80, f"Spring length: {spring.GetLength():.4f} m")
    
    app.EndScene()
    app.DoStep()

    if system.GetChTime() >= TOTAL_TIME:
        app.GetDevice().closeDevice()
        break
```
