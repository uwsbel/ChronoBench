```python
#!/usr/bin/env python3
"""
Simple Pendulum Simulation using PyChrono
This script creates a pendulum connected to a fixed ground body via a revolute joint.
"""

# Import required PyChrono modules
import pychrono as chrono
import pychrono.irrlicht as chronoirr

def main():
    print("=== PyChrono Simple Pendulum Simulation ===\n")
    
    # 1. Initialize the PyChrono environment
    # Create a physical system
    system = chrono.ChSystemNSC()
    system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Gravity
    
    # Set simulation parameters
    system.SetSolverType(chrono.ChSolver.Type_SOR)
    system.SetMaxItersSolverSpeed(50)
    system.Set timestep(0.001)
    system.SetTimestepperType(chrono.ChTimestepper.Type_EULER_IMPLICIT)
    
    print("Physical system created with gravity: (0, -9.81, 0)")
    
    # 2. Create the ground body (fixed anchor)
    ground = chrono.ChBodyEasyBox(10.0, 1.0, 10.0, 1000, True, False)
    ground.SetPos(chrono.ChVectorD(0, -0.5, 0))
    ground.SetFixed(True)
    ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.png"))
    system.Add(ground)
    print("Ground body created: 10x1x10 box, fixed at (0, -0.5, 0)")
    
    # 3. Create the pendulum body (rod + bob)
    # Define pendulum parameters
    pendulum_length = 2.0  # meters
    bob_radius = 0.2       # meters
    bob_density = 1000      # kg/m³ (water-like)
    
    # Calculate mass and inertia
    bob_volume = (4/3) * 3.14159 * bob_radius**3
    bob_mass = bob_density * bob_volume
    bob_inertia = 0.4 * bob_mass * bob_radius**2  # Solid sphere inertia
    
    # Create pendulum body as a compound object
    pendulum = chrono.ChBody()
    
    # Add a visual sphere for the bob
    sphere_shape = chrono.ChSphereShape(bob_radius)
    pendulum.AddVisualShape(sphere_shape)
    
    # Set collision model (simplified)
    collision_model = chrono.ChCollisionModel()
    collision_model.SetSafeMargin(0.01)
    pendulum.AttachCollisionModel(collision_model)
    
    # Set mass and inertia
    pendulum.SetMass(bob_mass)
    pendulum.SetInertiaXX(chrono.ChVectorD(bob_inertia, bob_inertia, bob_inertia))
    
    # Set initial position (horizontal, so it will swing)
    initial_angle = 45.0  # degrees
    initial_angle_rad = chrono.CH_DEG_TO_RAD * initial_angle
    pendulum.SetPos(chrono.ChVectorD(
        pendulum_length * chrono.cos(initial_angle_rad),
        pendulum_length * chrono.sin(initial_angle_rad),
        0
    ))
    pendulum.SetFixed(False)
    
    # Add pendulum to system
    system.Add(pendulum)
    print(f" Pendulum bob created:")
    print(f"   - Mass: {bob_mass:.3f} kg")
    print(f"   - Radius: {bob_radius} m")
    print(f"   - Length from pivot: {pendulum_length} m")
    print(f"   - Initial angle: {initial_angle} degrees")
    
    # 4. Create the revolute joint (hinge)
    # Define pivot point (on the ground surface)
    pivot = chrono.ChVectorD(0, 0, 0)
    
    # Create revolute joint
    joint = chrono.ChLinkLockRevolute()
    joint.Initialize(ground, pendulum, 
                     chrono.ChCoordsysD(pivot, chrono.Q_ROTATE_Z_TO_X))
    system.AddLink(joint)
    print(f"\nRevolute joint created at pivot point: ({pivot.x}, {pivot.y}, {pivot.z})")
    
    # 5. Set up Irrlicht visualization
    print("\nInitializing Irrlicht visualization...")
    
    # Create the Irrlicht application
    vis = chronoirr.ChIrrApp(
        system,
        "PyChrono Pendulum Simulation",
        chronoirr.dimension2du(1280, 720)
    )
    
    # Setup the visualization environment
    vis.AddTypicalCamera(chronoirr.vector3d(4, 3, 5), chronoirr.vector3d(0, 1.5, 0))
    vis.AddTypicalLight()
    vis.AddTypicalSky()
    
    # Add a ground plane for reference
    ground_plane = chronoirr.ChIrrCableRobot(ground)
    
    # Setup the visual assets
    vis.AssetBindAll()
    vis.AssetUpdateAll()
    
    # 6. Simulation loop with logging
    print("\n" + "=" * 60)
    print("Starting simulation with logging...")
    print("=" * 60)
    print(f"{'Time (s)':<12} {'Angle (deg)':<15} {'Velocity (rad/s)':<18} {'Position (m)':<15}")
    print("-" * 60)
    
    # Simulation parameters
    total_time = 5.0  # Total simulation time
    output_interval = 0.05  # Log every 50ms
    next_log_time = 0.0
    
    # Custom timestep loop for logging
    simulation_step = 0
    while vis.Run():
        # Check if simulation should continue
        if system.GetChTime() >= total_time:
            print("\nSimulation complete!")
            break
        
        # Advance visualization
        vis.BeginScene()
        vis.DrawAll()
        
        # Draw info text
        vis.AddLogo(chronoirr.vector3d(1.5, 1.0, 0))
        
        # Advance physics
        vis.DoStep()
        
        # Log data at intervals
        if system.GetChTime() >= next_log_time:
            # Calculate angle from vertical
            pos = pendulum.GetPos()
            angle = chrono.CH_RAD_TO_DEG * chrono.atan2(pos.x, pos.y)
            angular_vel = joint.GetRelWvel().z
            pendulum_vel = pendulum.GetPos_dt()
            speed = pendulum_vel.Length()
            
            # Print log
            print(f"{system.GetChTime():<12.3f} {angle:<15.2f} {angular_vel:<18.4f} "
                  f"({pos.x:<8.3f}, {pos.y:<8.3f}, {pos.z:<8.3f})")
            
            next_log_time += output_interval
        
        vis.EndScene()
        simulation_step += 1
    
    # Final statistics
    print("\n" + "=" * 60)
    print("Simulation Statistics:")
    print("=" * 60)
    print(f"Total simulation time: {system.GetChTime():.3f} seconds")
    print(f"Total simulation steps: {simulation_step}")
    print(f"Average timestep: {system.GetChTime()/simulation_step:.6f} seconds")
    
    # Final pendulum state
    final_pos = pendulum.GetPos()
    final_angle = chrono.CH_RAD_TO_DEG * chrono.atan2(final_pos.x, final_pos.y)
    print(f"\nFinal pendulum angle: {final_angle:.2f} degrees")
    print(f"Final pendulum position: ({final_pos.x:.3f}, {final_pos.y:.3f}, {final_pos.z:.3f}) m")
    
    print("\nSimulation finished successfully!")

if __name__ == "__main__":
    main()
```

This script creates a complete PyChrono pendulum simulation with the following features:

## Key Components:

### 1. **Physical System Setup**
- Creates a `ChSystemNSC` (Non-Smooth Contact) system
- Sets gravitational acceleration to (0, -9.81, 0)
- Configures solver and timestepper parameters

### 2. **Ground Body**
- A fixed rectangular box (10 x 1 x 10 meters)
- Positioned at (0, -0.5, 0)
- Visualized with a concrete texture

### 3. **Pendulum Body**
- Created as a sphere (bob) with configurable:
  - **Length**: 2.0 meters from pivot
  - **Bob radius**: 0.2 meters
  - **Density**: 1000 kg/m³ (water-like)
  - **Mass**: ~33.5 kg (calculated from volume × density)
  - **Inertia**: Calculated for a solid sphere

### 4. **Revolute Joint**
- Connects ground to pendulum at the pivot point
- Allows rotation around the Z-axis only
- Constrains all other degrees of freedom

### 5. **Irrlicht Visualization**
- 3D rendering with camera and lighting
- Real-time visualization window
- Ground plane and skybox

### 6. **Data Logging**
- Logs data every 0.05 seconds during simulation
- Records:
  - Time
  - Pendulum angle from vertical
  - Angular velocity
  - 3D position

## To Run:
```bash
python pendulum_simulation.py
```

The pendulum will start at a 45-degree angle and swing under gravity, with real-time visualization and logged data showing the oscillation behavior.
