**Corrected & modified PyChrono script**

```python
import pychrono.core as chrono
import pychrono.irrlicht as irr
import pychrono.vehicle as veh
import pychrono.sensor as sens
import numpy as np
import math

# -----------------------------------------------------------------
# Set Chrono data paths
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + "vehicle/")

# -----------------------------------------------------------------
# Vehicle initial position / orientation
initLoc = chrono.ChVector3d(-15, 0, 1.2)
initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# Visualization / collision types
vis_type = veh.VisualizationType_MESH
chassis_collision_type = veh.CollisionType_NONE

# Tire model
tire_model = veh.TireModelType_RIGID

# Terrain (not used directly – we use SCM)
terrainHeight = 0
terrainLength = 100.0
terrainWidth = 100.0

# Camera‑track point on the chassis
trackPoint = chrono.ChVector3d(0.0, 0.0, 1.71)

# Contact method (SMC = smooth‑motion contacts)
contact_method = chrono.ChContactMethod_SMC
contact_vis = False

# Step sizes
step_size = 1e-3
tire_step_size = step_size
render_step_size = 1.0 / 20.0          # 20 fps rendering

# -----------------------------------------------------------------
# Create the HMMWV vehicle, set parameters and initialize
vehicle = veh.HMMWV_Full()
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
vehicle.SetTireType(tire_model)
vehicle.SetTireStepSize(tire_step_size)
vehicle.Initialize()

vehicle.SetChassisVisualizationType(vis_type)
vehicle.SetSuspensionVisualizationType(vis_type)
vehicle.SetSteeringVisualizationType(vis_type)
vehicle.SetWheelVisualizationType(vis_type)
vehicle.SetTireVisualizationType(vis_type)

vehicle.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# -----------------------------------------------------------------
# SCM deformable terrain
terrain = veh.SCMTerrain(vehicle.GetSystem())
terrain.SetSoilParameters(
    2e6,      # Bekker Kφ
    0.0,      # Bekker Kc
    1.1,      # Bekker n exponent
    0.0,      # Mohr cohesive limit (Pa)
    30.0,     # Mohr friction limit (deg)
    0.01,     # Janosi shear coefficient (m)
    2e8,      # Elastic stiffness (Pa/m)
    3e4       # Damping (Pa·s/m)
)

# Moving patch around the chassis
terrain.AddMovingPatch(vehicle.GetChassisBody(),
                      chrono.ChVector3d(0, 0, 0),
                      chrono.ChVector3d(5, 3, 1))

# False‑colour plotting of sinkage
terrain.SetPlotType(veh.SCMTerrain.PLOT_SINKAGE, 0, 0.1)

# Initialise terrain from a height‑map file
terrain.Initialize(veh.GetDataFile("terrain/height_maps/bump64.bmp"),
                    40, 40, -1, 1, 0.02)

terrain.SetTexture(veh.GetDataFile("terrain/textures/dirt.jpg"), 6.0, 6.0)

# -----------------------------------------------------------------
# Random box obstacles (5 boxes)
np.random.seed(12345)
for _ in range(5):
    sx = np.random.uniform(0.5, 1.5)
    sy = np.random.uniform(0.5, 1.5)
    sz = np.random.uniform(0.5, 1.5)
    pos = chrono.ChVector3d(np.random.uniform(-20, 20),
                            np.random.uniform(-20, 20),
                            sz * 0.5)
    mat_obs = chrono.ChMaterialSurfaceSMC()
    mat_obs.SetFriction(0.9)
    mat_obs.SetRestitution(0.0)
    body = chrono.ChBodyEasyBox(sx, sy, sz, 1000, mat_obs)
    body.SetPos(pos)
    body.SetBodyFixed(True)               # keep them static
    vehicle.GetSystem().AddBody(body)

# -----------------------------------------------------------------
# Sensor manager + Lidar
manager = sens.ChSensorManager(vehicle.GetSystem())

lidar_rate = 30.0                     # Hz
lidar_offset = chrono.ChVector3d(0, 0, 0.5)
lidar_rot = chrono.QUNIT               # identity quaternion

lidar = sens.ChLidarSensor(
    vehicle.GetChassisBody(),
    lidar_rate,
    lidar_offset,
    lidar_rot,
    1800,                # number of samples (rays)
    -math.pi,            # min azimuth angle
    math.pi,             # max azimuth angle
    0.2,                 # minimum range (m)
    100.0,               # maximum range (m)
    0.1,                 # angular resolution (rad)
    0.01                 # noise amplitude
)
lidar.SetName("Lidar")
lidar.PushFilter(sens.ChFilterVisualize())
lidar.PushFilter(sens.ChFilterDepth())
manager.AddSensor(lidar)

# -----------------------------------------------------------------
# Irrlicht visualisation
vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle("HMMWV Demo")
vis.SetWindowSize(1280, 1024)
vis.SetChaseCamera(trackPoint, 6.0, 0.5)
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetVehicle())

# -----------------------------------------------------------------
# Driver (keyboard / interactive)
driver = veh.ChInteractiveDriverIRR(vis)

steering_time = 1.0    # time to reach full steering
throttle_time = 1.0   # time to reach full throttle
braking_time = 0.3    # time to reach full brake
driver.SetSteeringDelta(render_step_size / steering_time)
driver.SetThrottleDelta(render_step_size / throttle_time)
driver.SetBrakingDelta(render_step_size / braking_time)
driver.Initialize()

# -----------------------------------------------------------------
print("VEHICLE MASS:", vehicle.GetVehicle().GetMass())

# -----------------------------------------------------------------
# Simulation loop
render_steps = math.ceil(render_step_size / step_size)
realtime_timer = chrono.ChRealtimeStepTimer()
step_number = 0
render_frame = 0

while vis.Run():
    t = vehicle.GetSystem().GetChTime()

    # ---- Rendering -------------------------------------------------
    if step_number % render_steps == 0:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        render_frame += 1

    # ---- Driver inputs --------------------------------------------
    driver_inputs = driver.GetInputs()

    # ---- Synchronise all modules -----------------------------------
    driver.Synchronize(t)
    terrain.Synchronize(t)
    vehicle.Synchronize(t, driver_inputs, terrain)
    vis.Synchronize(t, driver_inputs)

    # ---- Update sensor manager -------------------------------------
    manager.Update(step_size)

    # ---- Advance all modules ---------------------------------------
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    # ---- bookkeeping -----------------------------------------------
    step_number += 1
    realtime_timer.Spin(step_size)
```

| **Obstacles** | Five random boxes are created (`ChBodyEasyBox`) and added to the system before the main loop. |
| **Sensor manager** | A `ChSensorManager` is instantiated and a Lidar sensor (`ChLidarSensor`) is attached to the chassis with visualisation filters (`ChFilterVisualize`, `ChFilterDepth`). |
| **Loop update** | `manager.Update(step_size)` is called each iteration so the sensor data are refreshed. |
| **Minor corrections** | - Re‑placed the missing closing parenthesis in `terrain.SetSoilParameters(...)`.<br>- Used `math.pi` instead of the non‑existent `chrono.CH_C_PI` for the Lidar angular range.<br>- Made obstacle creation robust by giving each box a material (`ChMaterialSurfaceSMC`) and a positive density while keeping it fixed, avoiding possible zero‑mass issues. |
| **Other** | All original functionality (vehicle, terrain, driver, rendering) is preserved and the script follows the required “add X → do Y” pattern. |
