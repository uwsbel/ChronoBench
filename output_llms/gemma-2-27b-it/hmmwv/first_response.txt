```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import math

# ---------------------------------------------------------------------
#
#  Create the simulation system and add the terrain
#
# ---------------------------------------------------------------------

mysystem = chrono.ChSystemNSC()
mysystem.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Set simulation parameters
mysystem.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)
mysystem.SetSolverMaxIterations(100)
mysystem.SetStep(0.01)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(mysystem)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle('HMMWV Simulation')
vis.Initialize()
vis.AddCamera(chrono.ChVectorD(0, 2, -5))
vis.AddLight(chrono.ChVectorD(2, 5, 3), chrono.ChColor(1, 1, 1))

# Create the terrain
ground = chrono.ChRigidBody()
ground.SetBodyFixed(True)
ground_shape = chrono.ChBoxShape()
ground_shape.SetLengths(chrono.ChVectorD(100, 0.1, 100))
ground.AddShape(ground_shape)
ground.SetPos(chrono.ChVectorD(0, -0.1, 0))
mysystem.AddBody(ground)
ground_texture = chrono.ChTexture()
ground_texture.SetTextureFilename("textures/terrain.jpg")
ground.AddAsset(ground_texture)

# ---------------------------------------------------------------------
#
#  Create the HMMWV vehicle
#
# ---------------------------------------------------------------------

vehicle = chrono.ChVehicle()

# Chassis
chassis = chrono.ChBody()
chassis.SetMass(2500)
chassis.SetInertiaXX(chrono.ChVectorD(1000, 1500, 1000))
chassis.SetPos(chrono.ChVectorD(0, 1, 0))

# Add chassis visualization
chassis_shape = chrono.ChBoxShape()
chassis_shape.SetLengths(chrono.ChVectorD(4, 1.5, 2))
chassis.AddShape(chassis_shape)
chassis.SetCollide(True)

# Add chassis to vehicle and system
vehicle.AddChassis(chassis)
mysystem.AddBody(chassis)

# Wheels
wheel_radius = 0.4
wheel_width = 0.2
wheel_mass = 50
wheel_inertia = chrono.ChVectorD(0.5, 0.5, 0.5)

# Left front wheel
lf_wheel = chrono.ChWheel()
lf_wheel.SetRadius(wheel_radius)
lf_wheel.SetWidth(wheel_width)
lf_wheel.SetMass(wheel_mass)
lf_wheel.SetInertia(wheel_inertia)
lf_wheel.SetFriction(0.8)
lf_wheel.SetSuspensionForce(2000)
lf_wheel.SetSuspensionCompression(0.2)
lf_wheel.SetSuspensionDamping(100)
lf_wheel.SetSteering(True)
lf_wheel.SetPos(chrono.ChVectorD(1.5, -0.7, 1))
lf_wheel.SetDir(chrono.ChVectorD(0, -1, 0))
vehicle.AddWheel(lf_wheel)

# Right front wheel
rf_wheel = chrono.ChWheel()
rf_wheel.SetRadius(wheel_radius)
rf_wheel.SetWidth(wheel_width)
rf_wheel.SetMass(wheel_mass)
rf_wheel.SetInertia(wheel_inertia)
rf_wheel.SetFriction(0.8)
rf_wheel.SetSuspensionForce(2000)
rf_wheel.SetSuspensionCompression(0.2)
rf_wheel.SetSuspensionDamping(100)
rf_wheel.SetSteering(True)
rf_wheel.SetPos(chrono.ChVectorD(1.5, -0.7, -1))
rf_wheel.SetDir(chrono.ChVectorD(0, -1, 0))
vehicle.AddWheel(rf_wheel)

# Left rear wheel
lr_wheel = chrono.ChWheel()
lr_wheel.SetRadius(wheel_radius)
lr_wheel.SetWidth(wheel_width)
lr_wheel.SetMass(wheel_mass)
lr_wheel.SetInertia(wheel_inertia)
lr_wheel.SetFriction(0.8)
lr_wheel.SetSuspensionForce(2000)
lr_wheel.SetSuspensionCompression(0.2)
lr_wheel.SetSuspensionDamping(100)
lr_wheel.SetPos(chrono.ChVectorD(-1.5, -0.7, 1))
lr_wheel.SetDir(chrono.ChVectorD(0, -1, 0))
vehicle.AddWheel(lr_wheel)

# Right rear wheel
rr_wheel = chrono.ChWheel()
rr_wheel.SetRadius(wheel_radius)
rr_wheel.SetWidth(wheel_width)
rr_wheel.SetMass(wheel_mass)
rr_wheel.SetInertia(wheel_inertia)
rr_wheel.SetFriction(0.8)
rr_wheel.SetSuspensionForce(2000)
rr_wheel.SetSuspensionCompression(0.2)
rr_wheel.SetSuspensionDamping(100)
rr_wheel.SetPos(chrono.ChVectorD(-1.5, -0.7, -1))
rr_wheel.SetDir(chrono.ChVectorD(0, -1, 0))
vehicle.AddWheel(rr_wheel)

# Set tire model
vehicle.SetTireModel(chrono.ChTireModelTMEASY())

# Connect wheels to chassis
vehicle.ConnectWheelsToChassis()

# Add vehicle to the system
mysystem.Add(vehicle)

# ---------------------------------------------------------------------
#
#  Driver system
#
# ---------------------------------------------------------------------

driver = chronoirr.ChIrrGuiDriver(vis)
driver.SetSteeringSensitivity(0.01)
driver.SetThrottleSensitivity(0.1)
driver.SetBrakingSensitivity(0.1)

# ---------------------------------------------------------------------
#
#  Simulation loop
#
# ---------------------------------------------------------------------

while vis.Run():
    # Update driver inputs
    driver.Update()

    # Update the simulation
    mysystem.DoStepDynamics(0.02)

    # Render the scene
    vis.Render()

```

**Explanation:**

1. **Initialization:**
   - Imports necessary PyChrono and Irrlicht modules.
   - Creates a `ChSystemNSC` object for the simulation.
   - Sets gravity and solver parameters.
   - Initializes the Irrlicht visualization system and sets up the camera and lighting.

2. **Terrain:**
   - Creates a rigid body for the ground and defines its shape and position.
   - Adds a texture to the ground.

3. **HMMWV Vehicle:**
   - Creates a `ChVehicle` object to represent the HMMWV.
   - Creates the chassis body with mass, inertia, and shape.
   - Adds four wheels with specified parameters like radius, width, mass, inertia, friction, suspension properties, and steering capability.
   - Sets the tire model to `ChTireModelTMEASY`.
   - Connects the wheels to the chassis.
   - Adds the vehicle to the simulation system.

4. **Driver System:**
   - Creates an `ChIrrGuiDriver` object for interactive control using the Irrlicht GUI.
   - Sets steering, throttle, and braking sensitivities.

5. **Simulation Loop:**
   - Runs the Irrlicht visualization loop.
   - Updates driver inputs within each frame.
   - Performs a simulation step using `DoStepDynamics`.
   - Renders the scene using Irrlicht.

**Notes:**



* This script uses primitive shapes for visualization. You can replace them with more detailed 3D models for a more realistic appearance.
* The tire model and suspension parameters can be further customized for more accurate vehicle behavior.
* The driver system provides basic keyboard controls. You can extend it to include more advanced input methods or autonomous driving algorithms.
