def createVehicle(sys):
    # Create the vehicle system
    vehicle = veh.ChVehicleSystemNSC()
    sys.Add(vehicle)

    # Set vehicle parameters
    vehicle.SetChassisFixed(False)
    vehicle.SetChassisVisualizationType(veh.VisualizationType_PRIMITIVES)

    # Create the vehicle body
    vehicle.AddWheel(veh.ChWheelInfo(
        name="front_left",
        location=chrono.ChVector3d(1.5, 1.0, 0.0),
        radius=0.3,
        width=0.15,
        mass=10.0,
        inertia=chrono.ChVector3d(0.5, 0.5, 0.5),
        friction=0.8
    ))
    vehicle.AddWheel(veh.ChWheelInfo(
        name="front_right",
        location=chrono.ChVector3d(-1.5, 1.0, 0.0),
        radius=0.3,
        width=0.15,
        mass=10.0,
        inertia=chrono.ChVector3d(0.5, 0.5, 0.5),
        friction=0.8
    ))
    vehicle.AddWheel(veh.ChWheelInfo(
        name="rear_left",
        location=chrono.ChVector3d(1.5, -1.0, 0.0),
        radius=0.3,
        width=0.15,
        mass=10.0,
        inertia=chrono.ChVector3d(0.5, 0.5, 0.5),
        friction=0.8
    ))

        width=0.15,
        mass=10.0,
        inertia=chrono.ChVector3d(0.5, 0.5, 0.5),
        friction=0.8
    ))

    return vehicle

def createTerrain(sys):
    # Create the terrain
    ground = chrono.ChBodyEasyBox(
        100, 100, 0.1, 1000,
        True, True, chrono.ChVector3d(0, 0, -0.05)
    )
    ground.SetBodyFixed(True)
    ground.GetVisualModel().GetShape(0).SetTexture(chrono.GetChronoDataFile("textures/terrain.jpg"))
    ground.GetVisualModel().GetShape(0).SetColor(chrono.ChColor(0.8, 0.8, 0.8))
    sys.Add(ground)

    return ground
