```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as turtlebot
from pychrono import irrlicht as chronoirr
import pychrono.sensor as sens

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))  
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create ground body as terrain plane
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVector3d(0, 0, -0.6))  # Adjusted ground position
ground.SetFixed(True)
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# Create Turtlebot Robot
init_pos = chrono.ChVector3d(0, 0.2, 0) 
init_rot = chrono.ChQuaterniond(1, 0, 0, 0) 
robot = turtlebot.TurtleBot(system, init_pos, init_rot)
robot.Initialize()

# Create sensor manager and lidar sensor
sensor_manager = sens.ChSensorManager(system)
lidar = sens.ChLidarSensor()
lidar.SetName("my_lidar")
lidar.SetRobot(robot)
lidar.SetPos(chrono.ChVector3d(0.2, 0.1, 0.3))
lidar.SetRot(chrono.ChQuaterniond(1, 0, 0, 0))
lidar.SetAngle(chrono.ChRadians(math.pi / 2))
lidar.SetRange(10)
lidar.SetResolution(1)
lidar.AddFilter(sens.ChFilterMedian(3))
sensor_manager.AddSensor(lidar)

# Create randomly placed boxes
for i in range(5):
    box = chrono.ChBodyEasyBox(0.2, 0.2, 0.2, 10, True, True)
    box.SetPos(chrono.ChVector3d(np.random.uniform(-5, 5), np.random.uniform(-5, 5), 0))
    system.Add(box)

# Define motion control function for Turtlebot
def move(mode):
    if mode == 'straight':
        robot.SetMotorSpeed(math.pi / 2, 0)
        robot.SetMotorSpeed(math.pi / 2, 1)
    elif mode == 'left':
        robot.SetMotorSpeed(math.pi / 2, 0)
        robot.SetMotorSpeed(0, 1)
    elif mode == 'right':
        robot.SetMotorSpeed(0, 0)
        robot.SetMotorSpeed(math.pi / 2, 1)

# Create run-time visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Turtlebot Robot - Rigid terrain')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 1.5, 0.2), chrono.ChVector3d(0, 0, 0.2))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0.5), 3, 4, 10, 40, 512)

# Enable shadows (commented out to improve performance)
# vis.EnableShadows()

# Set the simulation time step
time_step = 2e-3

# Simulation loop
time = 0
while vis.Run():
    # Turtlebot movement
    move('straight')

    # Update sensor manager
    sensor_manager.Update()

    # Increment time counter
    time += time_step

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance the simulation by one time step
    system.DoStepDynamics(time_step)
```



**Explanation of Changes:**

1. **Ground Position:** Adjusted the ground position from `chrono.ChVector3d(0, 0, -1)` to `chrono.ChVector3d(0, 0, -0.6)` as requested.

2. **Sensor Manager and Lidar:**
   - Imported `pychrono.sensor` as `sens`.
   - Created a `ChSensorManager` instance and added it to the system.
   - Configured a `ChLidarSensor` with specified parameters (name, robot, position, rotation, angle, range, resolution) and added a median filter for noise reduction.

3. **Random Boxes:**
   - Added a loop to create 5 boxes with random positions within a specified range.

4. **Motion Control Function:**
   - Defined a `move(mode)` function to control the Turtlebot's movement based on the provided `mode` ('straight', 'left', 'right').

5. **Simulation Loop:**
   - Called the `move('straight')` function to make the Turtlebot move straight.
   - Added `sensor_manager.Update()` to update the sensor data within the loop.

**Key Points:**

- The script now includes a lidar sensor that can be used to collect environmental data.
- The randomly placed boxes provide obstacles for the Turtlebot to interact with.
- The `move()` function provides a simple way to control the robot's movement.
- Remember to adjust the lidar sensor parameters and the box positions as needed for your specific simulation scenario.
