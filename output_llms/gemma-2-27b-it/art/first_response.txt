```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as chronovehicle

# ---------------------------------------------------------------------
#
#  Create the simulation system and add items

# ---------------------------------------------------------------------

# Create the physical system
system = chrono.ChSystemNSC()

system.SetSolverMaxIterations(100)
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle("ARTcar Simulation")
vis.Initialize()
vis.AddCameraLight()
vis.AddSkyBox()

# Create a rigid terrain
ground = chrono.ChBodyEasyBox(
    50, 50, 1, 1000, True, True, chrono.ChMaterialSurface.NSC
)
ground.SetPos(chrono.ChVectorD(0, -0.5, 0))
ground.SetBodyFixed(True)
system.Add(ground)

# Add a texture to the terrain
texture = chronoirr.ChTexture()
texture.SetFile("path/to/your/texture.jpg")
ground.AddAsset(texture)

# ---------------------------------------------------------------------
#
#  Create the ARTcar vehicle
#
# ---------------------------------------------------------------------

# Set vehicle parameters
vehicle_location = chrono.ChVectorD(0, 1, 0)
vehicle_orientation = chrono.ChQuaternionD(1, 0, 0, 0)
contact_method = chrono.ChContactMethod.NSC
visualization_type = chronovehicle.ChVehicleVisualType.POVRAY

# Create the ARTcar vehicle
vehicle = chronovehicle.ChVehicle("ARTcar", system)
vehicle.SetChassisVisualizationType(visualization_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitialPosition(vehicle_location)
vehicle.SetInitialOrientation(vehicle_orientation)
vehicle.SetContactMethod(contact_method)

# ---------------------------------------------------------------------
#

#

# Create a driver system
driver = chronovehicle.ChDriver(vehicle)

# Set driver controls
driver.SetSteeringControl(chrono.ChFunction.ChFunction_Const(0))  # Initial steering
driver.SetThrottleControl(chrono.ChFunction.ChFunction_Const(0))  # Initial throttle
driver.SetBrakingControl(chrono.ChFunction.ChFunction_Const(0))  # Initial braking

# Connect driver controls to keyboard input
# (You'll need to implement keyboard input handling here)

# ---------------------------------------------------------------------
#
#  Simulation loop
#
# ---------------------------------------------------------------------

while vis.Run():

    # Update vehicle dynamics

```

**Explanation:**

1. **Initialization:**
   - Creates a `ChSystemNSC` for the physical simulation.
   - Sets solver parameters for better stability.
   - Initializes the Irrlicht visualization system (`chronoirr`).

2. **Terrain:**
   - Creates a `ChBodyEasyBox` for the ground.
   - Sets its position, size, and makes it fixed.
   - Adds a texture to the ground using `ChTexture`.

3. **ARTcar:**
   - Defines vehicle parameters like location, orientation, contact method, and visualization type.
   - Creates the `ChVehicle` object.
   - **Important:** You need to add the specific wheel and suspension setup for your ARTcar model.

4. **Driver System:**
   - Creates a `ChDriver` object to control the vehicle.
   - Sets initial steering, throttle, and braking controls.
   - You'll need to implement keyboard input handling to allow user control.

5. **Simulation Loop:**
   - Runs the Irrlicht visualization loop (`vis.Run()`).
   - Updates driver controls based on keyboard input (which you need to implement).
   - Performs a single simulation step (`system.DoStepDynamics()`).
   - Renders the scene using Irrlicht (`vis.Render()`).
   - Limits the frame rate to 50 FPS using `chrono.ChTime.Wait()`.
