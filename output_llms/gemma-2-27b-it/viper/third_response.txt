```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as viper
from pychrono import irrlicht as chronoirr
import pychrono.vehicle as veh

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))  
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create SCM deformable terrain
terrain = veh.SCMDeformableTerrain(system)
terrain.Initialize("plane", 20, 20, 0.2)  # Plane, length, width, mesh resolution
terrain.SetSoilParameters(1e4, 0.3, 0.01, 100)  # Cohesion, friction, damping, stiffness

# Create Viper rover
rover = viper.Viper(system)
driver = viper.ViperDCMotorControl()
rover.SetDriver(driver)

# Initialize rover position and orientation
init_pos = chrono.ChVector3d(0, 0.2, 0)
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)
rover.Initialize(chrono.ChFramed(init_pos, init_rot))

# Create run-time visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Viper rover - Deformable terrain')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 2.5, 1.5), chrono.ChVector3d(0, 0, 1))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0.5), 3, 4, 10, 40, 512)

# Enable shadows (commented out to improve performance)
# vis.EnableShadows()

# Set the simulation time step
time_step = 1e-3

# Simulation loop
time = 0
while vis.Run():
    time += time_step

    # Set constant steering
    steering = 0.0
    driver.SetSteering(steering)

    rover.Update()

    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    
    system.DoStepDynamics(time_step)
```

**Changes made:**

1. **Import `pychrono.vehicle`:** Added `import pychrono.vehicle as veh` to use the SCM deformable terrain.
2. **Removed ground body:** Deleted the code for creating the ground body.
3. **Created SCM terrain:** Added code to create an `SCMDeformableTerrain` object:
   - `terrain = veh.SCMDeformableTerrain(system)` initializes the terrain within the Chrono system.
   - `terrain.Initialize("plane", 20, 20, 0.2)` sets the terrain type to "plane", length and width to 20, and mesh resolution to 0.2.
   - `terrain.SetSoilParameters(1e4, 0.3, 0.01, 100)` defines soil parameters for cohesion, friction, damping, and stiffness.
4. **Removed steering behavior:** Deleted the code block that defined the time-dependent steering behavior.
5. **Set constant steering:** Added `steering = 0.0` to set a constant steering angle of 0.
