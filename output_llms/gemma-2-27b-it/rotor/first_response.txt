```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.fea as fea

# ---------------------------------------------------------------------
#
#  Create the physical system
#
# ---------------------------------------------------------------------

mysystem = chrono.ChSystemNSC()
mysystem.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
mysystem.Add(ground)

# ---------------------------------------------------------------------
#
#  Create the IGA beam
#
# ---------------------------------------------------------------------

# Define the beam geometry and material properties
beam_length = 1.0
beam_radius = 0.05
beam_material = fea.ChMaterialFEA(rho=7850, E=200e9, nu=0.3)

# Define the knot vector and control points for the IGA beam
knots = [0, 0, 0.25, 0.5, 0.75, 1, 1, 1]
control_points = [
    chrono.ChVectorD(0, 0, 0),
    chrono.ChVectorD(beam_length / 4, 0, 0),
    chrono.ChVectorD(beam_length / 2, 0, 0),
    chrono.ChVectorD(3 * beam_length / 4, 0, 0),
    chrono.ChVectorD(beam_length, 0, 0),
]

# Create the IGA beam object
beam = fea.ChBeamIGA(beam_material, knots, control_points, beam_radius)
beam.SetNumElements(10)

# Add the beam to the system
mysystem.Add(beam)

# ---------------------------------------------------------------------
#
#  Create the flywheel
#
# ---------------------------------------------------------------------

# Define the flywheel geometry and mass properties
flywheel_radius = 0.2
flywheel_mass = 10

# Create the flywheel body
flywheel = chrono.ChBodyEasyCylinder(flywheel_radius, flywheel_radius, flywheel_mass)

# Attach the flywheel to the center of the beam
flywheel.SetPos(chrono.ChVectorD(beam_length / 2, 0, 0))
chrono.ChLinkMateFix(flywheel, beam, chrono.ChCoordsysD(chrono.ChVectorD(beam_length / 2, 0, 0)))
mysystem.Add(flywheel)

# ---------------------------------------------------------------------
#
#  Create the rotational motor
#
# ---------------------------------------------------------------------

# Create the motor body
motor = chrono.ChBody()
motor.SetBodyFixed(True)
motor.SetPos(chrono.ChVectorD(0, 0, 0))
mysystem.Add(motor)

# Create the motor constraint
motor_constraint = chrono.ChLinkMotorRotationSpeed(motor, beam, chrono.ChVectorD(1, 0, 0))
motor_constraint.SetSpeedFunction(chrono.ChFunction_Const(10))
mysystem.Add(motor_constraint)

# ---------------------------------------------------------------------
#
#  Visualization setup
#
# ---------------------------------------------------------------------

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(mysystem)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle("Jeffcott Rotor Simulation")
vis.Initialize()

# Set up FEM visualization
fea.ChVisualizationFEA(vis, beam)

# Run the simulation
while vis.Run():
    mysystem.DoStepDynamics(0.01)
    vis.Render()

```

This script defines a Jeffcott rotor system with an IGA beam, a flywheel, and a rotational motor. It utilizes PyChrono's features for IGA beam modeling, rigid body dynamics, and Irrlicht visualization. Remember to install the necessary PyChrono and Irrlicht libraries before running this script.
