```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as chronovehicle

# ---------------------------------------------------------------------
#
#  Create the simulation system and add the vehicle
#
# ---------------------------------------------------------------------

system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Set simulation parameters
system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)
system.SetSolverMaxIterations(100)
system.SetTimestepperType(chrono.ChTimestepper.Type_EULER_IMPLICIT_LINEARIZED)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle('City Bus Simulation')
vis.Initialize()
vis.AddCamera(chrono.ChVectorD(0, 2, -5))
vis.AddTypicalLights()

# Create the terrain
terrain = chrono.ChRigidTerrain(system)
heightmap_file = "path/to/your/heightmap.png"
texture_file = "path/to/your/texture.png"
terrain.SetTexture(texture_file)
terrain.Initialize(heightmap_file)

# Create the vehicle
vehicle = chronovehicle.ChVehicle(system)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChVectorD(0, 0, 0))
vehicle.SetInitRot(chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 1, 0)))

# Add the vehicle's chassis and wheels
vehicle.AddChassisRedundant(
    chrono.ChBodyEasyBox(2.5, 1, 1, 1000),
    chrono.ChVectorD(0, 0.5, 0),
    chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 1, 0))
)

# Load vehicle mesh (replace with your vehicle model)
vehicle_mesh = chrono.ChBodyEasyMesh(chrono.GetChronoDataPath() + 'vehicle/bus/bus.obj')
vehicle_mesh.SetPos(chrono.ChVectorD(0, 0.5, 0))
vehicle.AddBody(vehicle_mesh)

# Add wheels with tire models
for i in range(4):
    wheel = chronovehicle.ChWheel(vehicle.GetSystem())
    wheel.SetRadius(0.3)
    wheel.SetFriction(2.0)
    wheel.SetTireModel(chronovehicle.ChTireModelNSC())
    vehicle.AddWheel(wheel)

# ---------------------------------------------------------------------
#
#  Create the driver system
#
# ---------------------------------------------------------------------

class Driver:
    def __init__(self, vehicle):
        self.vehicle = vehicle
        self.steering = 0
        self.throttle = 0
        self.brake = 0

    def Update(self, dt):
        # Implement your driver logic here
        # Example:
        if vis.GetDevice().IsKeyDown(chrono.irr.KEY_LEFT):
            self.steering = -1
        elif vis.GetDevice().IsKeyDown(chrono.irr.KEY_RIGHT):
            self.steering = 1
        else:
            self.steering = 0

        if vis.GetDevice().IsKeyDown(chrono.irr.KEY_UP):
            self.throttle = 1
        elif vis.GetDevice().IsKeyDown(chrono.irr.KEY_DOWN):
            self.brake = 1
        else:
            self.throttle = 0
            self.brake = 0

        self.vehicle.SetSteering(self.steering)
        self.vehicle.SetThrottle(self.throttle)
        self.vehicle.SetBraking(self.brake)

driver = Driver(vehicle)

# ---------------------------------------------------------------------
#
#  Simulation loop
#
# ---------------------------------------------------------------------

while vis.Run():
    dt = vis.GetTimeStep()
    driver.Update(dt)
    system.DoStepDynamics(dt)
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

```

**Explanation:**

1. **Initialization:**
   - Imports necessary Chrono libraries.
   - Creates a Chrono simulation system (`system`) and sets gravity.
   - Configures the solver and timestepper for numerical stability.
   - Initializes the Irrlicht visualization system (`vis`) and sets up the camera.
   - Creates a rigid terrain (`terrain`) using a heightmap and texture.

2. **Vehicle Creation:**
   - Creates a `ChVehicle` object (`vehicle`) and sets its initial position and orientation.
   - Adds a chassis to the vehicle (using a box shape in this example).
   - Loads a vehicle mesh (replace `bus.obj` with your model).
   - Adds four wheels to the vehicle, each with a tire model.

3. **Driver System:**
   - Defines a `Driver` class to handle vehicle controls.
   - The `Update` method takes keyboard input to control steering, throttle, and braking.

4. **Simulation Loop:**
   - Enters the main simulation loop, which runs until the Irrlicht window is closed.
   - Updates the driver logic using `driver.Update(dt)`.
   - Performs a simulation step (`system.DoStepDynamics(dt)`).
   - Renders the scene using Irrlicht (`vis.Render()`).

**Key Points:**

- Replace placeholder file paths with your actual heightmap, texture, and vehicle model files.
- Customize the driver logic in the `Driver` class to implement your desired control behavior.
- Adjust simulation parameters (solver type, timestep, etc.) as needed for stability and performance.
