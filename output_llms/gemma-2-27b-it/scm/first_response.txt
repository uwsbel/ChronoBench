```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as chronovehicle

# 1. Initialize PyChrono environment and core components
chrono.SetChronoDataPath("data/")
my_system = chrono.ChSystemNSC()
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))
my_system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)

# 2. Add physical systems and objects

# 2.1. Vehicle
vehicle = chronovehicle.ChVehicle("HMMWV")
vehicle.SetSystem(my_system)

# Load vehicle chassis and components (replace with actual HMMWV model files)
vehicle.SetChassis(chronovehicle.ChChassis(
    "Chassis",
    chrono.ChBodyEasyBox(2.5, 1.2, 0.8, 1000),
    chrono.ChVectorD(0, 1, 0),
    chrono.ChQuaternionD(1, 0, 0, 0)
))

# Add wheels with rigid tire model (replace with actual HMMWV wheel geometry)
wheel_radius = 0.4
wheel_width = 0.2
for i in range(4):
    wheel = chronovehicle.ChWheel(
        "Wheel_" + str(i),
        chrono.ChBodyEasyCylinder(wheel_radius, wheel_width, 100),
        chrono.ChVectorD(i % 2 * 1.5 - 1, -0.6, i // 2 * 1.5 - 0.6)
    )
    wheel.SetTireModel(chronovehicle.ChTireModelRigid())
    vehicle.AddWheel(wheel)

# 2.2. SCM Terrain
terrain = chrono.ChTerrainSCM()
terrain.SetSystem(my_system)
terrain.SetSize(50, 50)
terrain.Initialize()

# Set custom soil parameters (adjust values as needed)
terrain.SetFriction(0.5)
terrain.SetCohesion(100)
terrain.SetPsi(30)

# Enable moving patch feature
terrain.SetMovingPatch(True)
terrain.SetPatchSize(5, 5)
terrain.SetPatchOffset(vehicle.GetChassis().GetPos())

# 2.3. Visualization
vis = chronoirr.ChIrrApp(my_system, "HMMWV Simulation", chronoirr.dimension2du(1280, 720))
vis.AddTypicalLights()
vis.AddSkyBox()

# Visualize vehicle components
vehicle.GetChassis().SetCollide(True)
vehicle.GetChassis().SetBodyFixed(False)
vis.AddVisualModel(vehicle.GetChassis())
for wheel in vehicle.GetWheels():
    vis.AddVisualModel(wheel)

# Visualize terrain with sinkage visualization
vis.AddVisualModel(terrain, chrono.ChVisualModel.VisualizationType_WIREFRAME)
terrain.SetVisualizeSinkage(True)

# 3. Set default parameters

# 3.1. Vehicle position and orientation
vehicle.GetChassis().SetPos(chrono.ChVectorD(0, 1, 0))
vehicle.GetChassis().SetRot(chrono.ChQuaternionD(1, 0, 0, 0))

# 3.2. Interactive driver system
driver = chronovehicle.ChDriver(vehicle)
driver.SetSteering(0)
driver.SetThrottle(0)
driver.SetBraking(0)

# 4. Simulation loop
while vis.GetDevice().run():
    vis.BeginScene()
    vis.Render()

    # Update driver inputs (replace with actual input handling)
    driver.SetSteering(vis.GetDevice().GetKeyboard()->GetCursorPos().X / 100)
    driver.SetThrottle(vis.GetDevice().GetKeyboard()->GetCursorPos().Y / 100)

    # Update vehicle dynamics
    my_system.DoStepDynamics(0.02)

    # Update terrain moving patch
    terrain.SetPatchOffset(vehicle.GetChassis().GetPos())

    vis.EndScene()
    vis.GetDevice().sleep(0.02)
```

5. **Simulation Loop:**
