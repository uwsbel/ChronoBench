```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# ---------------------------------------------------------------------
#
#  Create the simulation system and add items
#
# ---------------------------------------------------------------------

# Create the system
system = chrono.ChSystemNSC()

# Set gravitational acceleration
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the Irrlicht visualizer
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle("Epicyclic Gears")
vis.Initialize()
vis.AddCamera(chrono.ChCamera(chrono.ChVectorD(2, 2, 2)))

# Create a fixed truss
truss = chrono.ChBody()
truss.SetBodyFixed(True)
truss.SetPos(chrono.ChVectorD(0, 0, 0))

# Add a visualization shape to the truss
truss_shape = chrono.ChBoxShape()
truss_shape.SetSize(chrono.ChVectorD(1, 0.1, 0.1))
truss.AddShape(truss_shape)
vis.Add(truss, True)

# Create a rotating bar
bar = chrono.ChBody()
bar.SetPos(chrono.ChVectorD(0, 0.5, 0))
bar.SetMass(1)

# Add a visualization shape to the bar
bar_shape = chrono.ChCylinderShape()
bar_shape.SetRadius(0.05)
bar_shape.SetHeight(1)
bar.AddShape(bar_shape)
vis.Add(bar, True)

# Create a revolute joint to connect the bar to the truss
joint_bar_truss = chrono.ChLinkRevolute()
joint_bar_truss.Initialize(truss, bar, chrono.ChCoordsysD(chrono.ChVectorD(0, 0.5, 0), chrono.Q_from_AngAxis(chrono.CH_C_PI_2, chrono.VECT_X)))
system.AddLink(joint_bar_truss)

# Create two gears
gear1 = chrono.ChBodyEasyCylinder(0.2, 0.05, 1000, True, True)
gear1.SetPos(chrono.ChVectorD(0.5, 0.5, 0))
gear2 = chrono.ChBodyEasyCylinder(0.1, 0.05, 1000, True, True)
gear2.SetPos(chrono.ChVectorD(0.5, 0.5, 0.1))

# Add visualization shapes to the gears
vis.Add(gear1, True)
vis.Add(gear2, True)

# Create revolute joints to connect the gears to the bar
joint_gear1_bar = chrono.ChLinkRevolute()
joint_gear1_bar.Initialize(bar, gear1, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngAxis(chrono.CH_C_PI_2, chrono.VECT_X)))
system.AddLink(joint_gear1_bar)

joint_gear2_bar = chrono.ChLinkRevolute()
joint_gear2_bar.Initialize(bar, gear2, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0.1), chrono.Q_from_AngAxis(chrono.CH_C_PI_2, chrono.VECT_X)))
system.AddLink(joint_gear2_bar)

# Create a gear motor to enforce constant rotation of gear1
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(gear1, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))
motor.SetTargetSpeed(10)
system.AddLink(motor)

# ---------------------------------------------------------------------
#
#  Simulation loop
#
# ---------------------------------------------------------------------

while vis.Run():
    system.DoStepDynamics(0.01)
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
```

**Explanation:**

1. **Initialization:**
   - Import necessary PyChrono and Irrlicht modules.
   - Create a `ChSystemNSC` object to manage the simulation.
   - Set gravitational acceleration.
   - Initialize the Irrlicht visualizer (`chronoirr.ChVisualSystemIrrlicht`) and set up the window.

2. **Adding Objects:**
   - Create a fixed truss (`ChBody`) and add a box shape for visualization.
   - Create a rotating bar (`ChBody`) and add a cylinder shape.
   - Create two gears (`ChBodyEasyCylinder`) and add cylinder shapes.

3. **Joints and Constraints:**
   - Use `ChLinkRevolute` to create revolute joints connecting:
     - The bar to the truss.
     - Gear 1 to the bar.
     - Gear 2 to the bar.
   - Use `ChLinkMotorRotationSpeed` to create a motor that enforces a constant rotation speed for gear 1.

4. **Visualization:**
   - Add all bodies to the Irrlicht visualizer using `vis.Add()`.
   - Set appropriate materials and lighting using Irrlicht's functionalities (not shown in this example).

5. **Simulation Loop:**
   - Run the Irrlicht visualization loop (`vis.Run()`).
   - Inside the loop:
     - Perform a simulation step using `system.DoStepDynamics(0.01)`.
     - Render the scene using `vis.BeginScene()`, `vis.Render()`, and `vis.EndScene()`.

**Remember:**

- This script provides a basic framework. You can customize it further by:
  - Adding more complex geometries and materials.
  - Implementing different types of joints and constraints.
  - Modifying the simulation parameters (e.g., time step, solver settings).
  - Adding user input or interaction.
