```python
import pychrono.core as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import math

# ---------------------------------------------------------------------
#
#  Create the simulation system and add items
#
# ---------------------------------------------------------------------

system = chrono.ChSystemNSC()

# Set the default material properties
material = chrono.ChMaterialSurfaceNSC()
material.SetFriction(0.8)
material.SetRestitution(0.2)

# Create the terrain
ground = chrono.ChBodyEasyBox(
    200, 10, 200, 1000, True, True, material
)
ground.SetPos(chrono.ChVectorD(0, -1, 0))
ground.SetBodyFixed(True)
system.Add(ground)

# Create the M113 vehicle
vehicle = veh.ChVehicle(system)
vehicle.SetChassis(
    veh.ChChassis(
        "M113 Chassis",
        chrono.ChVectorD(0, 1, 0),
        chrono.ChVectorD(1.5, 0.5, 3),
        1500,
        chrono.ChVectorD(0.4, 0.2, 0.3),
        chrono.ChMatrix33D(1),
        material,
    )
)
vehicle.SetInitialSpeed(chrono.ChVectorD(0, 0, 0))
vehicle.SetSteeringAngle(0)

# Add wheels to the vehicle
wheel_radius = 0.4
wheel_width = 0.2
wheel_mass = 50
for i in range(6):
    wheel = veh.ChWheel(
        "Wheel " + str(i),
        chrono.ChVectorD(
            (-1.5 + i * 0.8, -0.3, 1.5) if i < 3 else (-1.5 + (i - 3) * 0.8, -0.3, -1.5)
        ),
        wheel_radius,
        wheel_width,
        wheel_mass,
        material,
    )
    vehicle.AddWheel(wheel)

# Create the driver system
driver = veh.ChDriver(vehicle)

# ---------------------------------------------------------------------
#
#  Set up Irrlicht visualization
#
# ---------------------------------------------------------------------

vis = irr.ChIrrApp(system, "M113 Simulation")
vis.SetWindowSize(1280, 720)
vis.SetCameraPosition(chrono.ChVectorD(5, 5, 5))
vis.SetCameraLookAt(chrono.ChVectorD(0, 1, 0))
vis.AddLight(irr.ChLight(chrono.ChVectorD(5, 5, 5)))
vis.Initialize()

# ---------------------------------------------------------------------
#
#  Simulation loop
#
# ---------------------------------------------------------------------

while vis.GetDevice().run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Driver control
    driver.SetSteeringAngle(math.sin(vis.GetTime() * 2))
    driver.SetThrottle(0.5)

    # Advance the simulation
    system.DoStepDynamics(vis.GetTimeStep())

    # Synchronize the visualization with the simulation
    vis.Synchronize(vis.GetTimeStep(), vis.GetTime())

```

**Explanation:**

1. **Initialization:** The script starts by importing necessary Chrono libraries and initializing the simulation system (`chrono.ChSystemNSC`). It defines default material properties for friction and restitution.
2. **Terrain:** A simple rectangular terrain is created using `chrono.ChBodyEasyBox` with specified dimensions and material. It is fixed in place.
3. **Vehicle:** An M113 vehicle is instantiated using `veh.ChVehicle`. The chassis is defined with its dimensions, mass, and center of gravity. Six wheels are added to the vehicle with appropriate positions, radii, widths, and masses.
4. **Driver:** A driver system (`veh.ChDriver`) is created to control the vehicle's steering and throttle.
5. **Visualization:** Irrlicht visualization is set up using `irr.ChIrrApp`. The window size, camera position, and lighting are configured.
6. **Simulation Loop:** The main loop runs as long as the Irrlicht window is open. Inside the loop:
   - Visualization is updated using `vis.BeginScene()`, `vis.Render()`, and `vis.EndScene()`.
   - The driver applies steering and throttle inputs based on the simulation time.
   - The simulation is advanced one timestep using `system.DoStepDynamics()`.
   - Visualization is synchronized with the simulation using `vis.Synchronize()`.



This script provides a basic framework for simulating an M113 vehicle on a rigid terrain with visualization. You can further customize it by adding more complex terrain, vehicle models, sensors, and control systems.
