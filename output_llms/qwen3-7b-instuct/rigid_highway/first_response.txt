```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.postprocess as postprocess

# Initialize PyChrono environment
chrono.Ch纪元.Initialize()
chrono.Ch纪元.SetRealTimeScale(1.0)

# Create a 3D scene
scene = chrono.ChScene()
scene.SetGravity(chrono.ChVectorD(0, -9.81, 0))

# Create Irrlicht visualization
vis = chronoirr.ChIrrApp(scene, "HMMWV Simulation", chronoirr.dimension2du(1280, 720))
vis.SetTimestep(1/50)

# Define the terrain mesh
terrain_mesh = chrono.ChTriangleMeshConnected()
terrain_mesh.LoadMeshFromObjFile("Highway_col.obj")
terrain_visual = chrono.ChTriangleMeshConnected()
terrain_visual.LoadMeshFromObjFile("Highway_vis.obj")

# Create a collision geometry
terrain_collision = chrono.ChTriangleMeshCollision(terrain_mesh)
terrain_collision.SetMaxContacts(10)
terrain_collision.SetMesh(terrain_mesh)
terrain_collision.SetMaterialSurface(chrono.ChMaterialSurfaceNSC())
terrain_collision.SetMeshColor(chrono.ChColor(0.5, 0.5, 0.5))

# Add terrain to the scene
scene.Add(terrain_collision)
scene.Add(terrain_visual)

# Load HMMWV model
hmmwv_model = chrono.ChChaseModel("hmmwv.obj")
hmmwv_model.SetMaterialSurface(chrono.ChMaterialSurfaceNSC())
hmmwv_model.SetMeshColor(chrono.ChColor(1, 0.5, 0))
hmmwv_model.SetCollisionModel(chrono.ChCollisionModel())
hmmwv_model.SetContactMethod(chrono.ChContactMethod.TMEASY)
hmmwv_model.SetTireModel(chrono.ChTireModel.TMEASY)
hmmwv_model.SetContactMaterial(chrono.ChMaterialSurfaceNSC())
hmmwv_model.SetPosition(chrono.ChVectorD(0, 0, 0))

# Add HMMWV to the scene
scene.Add(hmmwv_model)

# Define driver control
driver = chrono.ChDriverSteeringThrottleBrake()
driver.SetVehicle(hmmwv_model)
driver.SetSteering(chrono.ChControllerSteeringInput(0.0))
driver.SetThrottle(chrono.ChControllerThrottleInput(0.0))
driver.SetBrake(chrono.ChControllerBrakeInput(0.0))

# Run the simulation loop
vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 10, 10), chrono.ChVectorD(0, 0, 0))
vis.AddLight(chrono.ChVectorD(10, 10, 10), chrono.ChVectorD(1, 1, 1))
vis.AddLight(chrono.ChVectorD(-10, -10, 10), chrono.ChVectorD(1, 1, 1))
vis.AddLight(chrono.ChVectorD(0, -10, -10), chrono.ChVectorD(1, 1, 1))
vis.AddLight(chrono.ChVectorD(0, 10, -10), chrono.ChVectorD(1, 1, 1))
vis.AddLight(chrono.ChVectorD(-10, 10, 0), chrono.ChVectorD(1, 1, 1))
vis.AddLight(chrono.ChVectorD(10, -10, 0), chrono.ChVectorD(1, 1, 1))
vis.AddLight(chrono.ChVectorD(-10, -10, -10), chrono.ChVectorD(1, 1, 1))
vis.AddLight(chrono.ChVectorD(10, 10, -10), chrono.ChVectorD(1, 1, 1))

vis.SetInteractiveDriver(driver)
vis.AddDriverFeedback(driver)

vis.Execute()
vis.Run()

# Clean up
chrono.Ch纪元.Finalize()
```

This script initializes a PyChrono environment, sets up the scene, terrain, and vehicle, configures the driver control system, and runs the simulation with Irrlicht visualization. The vehicle's dynamics, including steering, throttle, and braking, are controlled interactively through the visualization window. The simulation is set to run at 50 frames per second, providing a real-time experience.
