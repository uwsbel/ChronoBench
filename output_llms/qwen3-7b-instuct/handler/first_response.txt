```python
import pychrono as chrono
import pychrono.core as chcore
import pychrono.sensor as chsensor
import pychrono.postprocess as chpost
import pychrono.engine as chengine
import pychrono.ros as chros

# 1. Initialize PyChrono environment and core components
chrono.Ch纪元.initialize()
chrono.Ch纪元.set_real_time_mode(True)

# 2. Add the required physical systems and objects
# Create a new system with gravity
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create a material for the box
material = chrono.ChMaterialSurfaceNSC()
material.SetFriction(0.3)
material.SetDampingF(0.5)
material.SetCompliance(0)
material.SetComplianceT(0)

# Create a fixed floor
floor = chrono.ChBodyEasyBox(10, 10, 0.1, 100, True, True, material)
floor.SetPos(chrono.ChVectorD(0, -10, 0))
system.Add(floor)

# Create a movable box
box = chrono.ChBodyEasyBox(1, 1, 1, 1000, True, True, material)
box.SetPos(chrono.ChVectorD(0, 0, 0))
system.Add(box)

# 3. Set necessary default parameters
# Set default positions and forces
box.SetPos(chrono.ChVectorD(0, 0, 0))
box.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))

# 4. Create a custom ROS handler
# Define the custom ROS handler class
class CustomROSHandler(chros.ChRosHandler):
    def __init__(self):
        super().__init__()
        self.pub_int = None

    def setup_ros(self, topic_name):
        # Initialize ROS node and publisher
        self.node = chros.ChRosNode()
        self.pub_int = self.node.create_publisher(chros.ChRosIntMsg, topic_name)

    def publish_int(self, data):
        # Publish integer message
        msg = chros.ChRosIntMsg()
        msg.data = data
        self.pub_int.publish(msg)

# Create an instance of the custom ROS handler
ros_handler = CustomROSHandler()

# 5. Configure the ROS manager with clock, body, and transform handlers
ros_manager = chros.ChRosManager(system)
ros_manager.AddClockHandler()
ros_manager.AddBodyHandler(box)
ros_manager.AddTransformHandler(floor)

# Register the custom handler
ros_manager.AddHandler(ros_handler)

# 6. Run the simulation loop
# Set the simulation time step
dt = 0.001

# Main simulation loop
while system.GetChTime() < 10:
    system.DoStepDynamics(dt)
    ros_manager.Advance(dt)
    ros_handler.publish_int(chrono.Ch纪元.GetChTime())

# Clean up
chrono.Ch纪元.finalize()
```

This script sets up a PyChrono simulation with a fixed floor and a movable box, initializes a custom ROS handler to publish integer messages, and integrates the ROS manager to publish data at each timestep. The simulation runs for 10 seconds, updating the ROS data at each step.
