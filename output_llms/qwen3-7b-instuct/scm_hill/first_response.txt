```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.postprocess as postprocess

# 1. Initialize PyChrono environment
chrono.SetChronoDataPath('path_to_data_folder')  # Set path to Chrono data folder
chrono.SetChronoConfigPath('path_to_config_folder')  # Set path to Chrono configuration folder

# 2. Add physical systems and objects
vehicle = chrono.ChBody()
vehicle.SetName("HMMWV")
vehicle.SetMass(1000)  # Mass of the vehicle
vehicle.SetPos(chrono.ChVectorD(0, 0, 0))  # Initial position
vehicle.SetInertiaXX(chrono.ChVectorD(1000, 1000, 1000))  # Inertia tensor
vehicle.SetBodyFixed(False)  # Vehicle is not fixed
vehicle.SetCollide(True)  # Enable collision detection
vehicle.SetMaterialSurface(chrono.ChMaterialSurfaceNSC())  # Surface material
vehicle.SetPos_dt(chrono.ChVectorD(0, 0, 0))  # Initial velocity

terrain = chrono.ChTerrain()
terrain.SetName("SCM terrain")
terrain.SetMaterialSurface(chrono.ChMaterialSurfaceNSC())  # Surface material
terrain.SetSoilParams(chrono.ChSoilParams(0.5, 0.1, 0.5))  # Soil parameters
terrain.LoadHeightMap('path_to_height_map')  # Load height map for terrain

driver = chrono.ChDriverSystem()
driver.SetVehicle(vehicle)
driver.SetSteeringAngle(0)  # Initial steering angle
driver.SetThrottle(0)  # Initial throttle

# 3. Set default parameters
vehicle.GetBodyNode().SetCollide(True)
terrain.GetTerrainNode().SetCollide(True)
driver.SetVehicle(vehicle)

# Visualization setup
gui = chronoirr.ChIrrApp(chrono.GetSystem(), "PyChrono Simulation", chronoirr.SCIFFontMode::e_SCIFFontMode_Courier)
gui.AddLogo('path_to_logo', 150)
gui.AddAdditionalModule(postprocess.ChPostProcessing())
gui.AddSkyBox()
gui.AddCamera(chrono.ChVectorD(10, 10, 10), chrono.ChVectorD(0, 0, 0))
gui.AddLight(chrono.ChVectorD(10, 10, 10), chrono.ChVectorD(1, 1, 1), 10, 10, 10)
gui.AddDirectionalLight(chrono.ChVectorD(0, -1, 0), chrono.ChVectorD(1, 1, 1))

# Simulation loop
dt = 0.001  # Time step
t = 0  # Simulation time
while gui.GetRun():
    gui.BeginScene(True, True, chrono.ChColor(0.8, 0.8, 0.8))  # Enable lighting
    gui.DrawAll()
    gui.EndScene()

    # Advance the simulation
    vehicle.CalcCollisions(chrono.ChVectorD(0, 0, 0))
    terrain.CalcCollisions(chrono.ChVectorD(0, 0, 0))
    driver.CalcThrottleSteering()
    vehicle.DoStepDynamics(dt)
    terrain.DoStepPhysics(dt)
    gui.SetTimestep(dt)
    t += dt

    gui.GetDevice().display()
```

This script initializes a PyChrono simulation with an HMMWV vehicle, an SCM deformable terrain, and a driver system. It sets up real-time visualization using Irrlicht, including camera and lighting settings. The simulation loop advances the vehicle, terrain, driver, and visualization at each timestep, maintaining real-time execution.
