```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import math

# Initialize PyChrono environment
chrono.Ch纪元()
chrono.Ch纪元().SetChkFreqEvery(10000)
chrono.Ch纪元().SetChkCollsEvery(1000)

# Create the physical system
system = chrono.ChSystemNSC()

# Add the fixed truss
truss = chrono.ChBodyEasyBox(0.1, 0.1, 1.0, 1000, False, False)
truss.SetPos(chrono.ChVectorD(0, 0, 0))
truss.SetBodyFixed(True)
system.AddBody(truss)

# Add the rotating bar
bar = chrono.ChBodyEasyCylinder(0.1, 1.0, 1000, False, False)
bar.SetPos(chrono.ChVectorD(0, 0, 0.5))
bar.SetRot(chrono.Q_from_AngX(math.pi))
bar.SetBodyFixed(False)
system.AddBody(bar)

# Add the gears
gear1 = chrono.ChBodyEasyCylinder(0.05, 0.1, 1000, False, False)
gear1.SetPos(chrono.ChVectorD(0, 0, 0.5))
gear1.SetRot(chrono.Q_from_AngX(math.pi / 2))
gear1.SetBodyFixed(False)
system.AddBody(gear1)

gear2 = chrono.ChBodyEasyCylinder(0.05, 0.1, 1000, False, False)
gear2.SetPos(chrono.ChVectorD(0, 0, 0.5))
gear2.SetRot(chrono.Q_from_AngX(-math.pi / 2))
gear2.SetBodyFixed(False)
system.AddBody(gear2)

# Connect the gears with a gear mesh
gear_pair = chrono.ChGearPair()
gear_pair.Initialize(gear1, gear2, 10, 1, 1)
system.Add(gear_pair)

# Add a gear motor to enforce a constant rotation speed
gear_motor = chrono.ChMotorGearSet()
gear_motor.Initialize(system.GetGroundBody(), gear2, 1, 1000)
system.Add(gear_motor)

# Set default parameters for the system
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))
system.Set_GyroEnable(True)

# Set up visualization
vis_system = chronoirr.ChVisualSystemIrrlicht()
vis_system.AttachSystem(system)
vis_system.SetWindowSize(800, 600)
vis_system.SetWindowTitle('Epicyclic Gears Simulation')
vis_system.Initialize()

# Set materials and lighting
material = chrono.ChMaterialSurfaceNSC()
material.SetFriction(0.5)
material.SetDampingF(0.5)
material.SetCompliance(0.00001)
material.SetComplianceT(0.00001)
vis_system.SetMaterialSurface(material)

light = chrono.ChLight()
light.SetType(chrono.ChLight.DIRECTIONAL)
light.SetDirection(chrono.ChVectorD(0, -1, -1))
light.SetAmbientColor(chrono.ChColor(1, 1, 1))
light.SetDiffuseColor(chrono.ChColor(1, 1, 1))
light.SetSpecularColor(chrono.ChColor(1, 1, 1))
vis_system.AddLight(light)

# Start the simulation
vis_system.AddSceneCallback(chrono.callback.SceneDrawCallback())
vis_system.AddSceneCallback(chrono.callback.SceneUpdateCallback())
vis_system.AddStepSimulationCallback(chrono.callback.StepSimulationCallback())
vis_system.AddPostStepSimulationCallback(chrono.callback.PostStepSimulationCallback())

# Run the simulation
while vis_system.GetDevice().run():
    vis_system.BeginScene()
    vis_system.DrawAll()
    vis_system.EndScene()
    system.DoStepDynamics(0.001)
```
This script initializes a PyChrono environment, sets up a system with a fixed truss, a rotating bar, and two gears connected with a gear mesh, and enforces a constant rotation speed using a gear motor. It then visualizes the system using the Irrlicht visualization system with appropriate materials and lighting. The simulation runs until the Irrlicht device stops.
