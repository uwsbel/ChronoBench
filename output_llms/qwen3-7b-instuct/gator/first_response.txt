```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import os

# Initialize PyChrono environment
chrono.SetChronoDataPath(os.path.join(os.path.dirname(__file__), "data"))

# Create a new Chrono simulation engine
chrono_engine = chrono.ChSystemNSC()

# Add the RigidTerrain
terrain = chrono.ChTerrain()
terrain.SetType(chrono.ChTerrainType.RIGID)
terrain.SetSize(50, 50, 0.1)
terrain.SetMaterial(chrono.ChMaterialSurfaceNSC())
terrain.SetTexture("terrain_texture.jpg")
terrain.SetFriction(0.5)
terrain.SetRestitution(0.1)
chrono_engine.Add(terrain)

# Add the Gator vehicle
vehicle = chrono.ChBodyEasyBox(2, 1, 1, 1000, False, False)
vehicle.SetPos(chrono.ChVectorD(0, 0, 0))
vehicle.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI_2))
vehicle.SetBodyFixed(False)
vehicle.SetCollide(True)
chrono_engine.Add(vehicle)

# Set the driver system (steering, throttle, braking)
driver = chrono.ChDriverSteeringThrottleBrake(vehicle, 0.5, 0.5, 0.5)
driver.SetDriverType(chrono.ChDriverSteeringThrottleBrake.DriverType.INTERACTIVE)
chrono_engine.Add(driver)

# Add the tire model (TMEASY)
tire_model = chrono.ChTMEASYTireModel()
tire_model.SetTireModelType(chrono.ChTMEASYTireModel.ModelType.STANDARD)
tire_model.SetTireDiameter(0.3)
tire_model.SetTireWidth(0.1)
tire_model.SetTirePressure(200)
vehicle.AddTireModel(tire_model)

# Set mesh visualization for all components
for body in chrono_engine.GetSystem():
    body.SetVisualShapeType(chrono.ChVisualShapeType.SHAPE_MESH)
    body.SetMeshFile("vehicle_mesh.obj")

# Create Irrlicht visualization
irr_engine = chronoirr.ChIrrApp(chrono_engine, "Gator Vehicle Simulation", chronoirr.dimension2du(1280, 720))
irr_engine.AddTypicalCameraController()
irr_engine.AddTypicalLights()

# Set the simulation loop parameters
chrono_engine.SetTimestepperType(chrono.ChTimestepper::EULER)
chrono_engine.SetTimestep(0.005)
chrono_engine.SetRealtimePrediction(True)
chrono_engine.SetRealtimeVisualisation(True)
chrono_engine.SetRealtimeVisualisationFPS(50)

# Start the simulation loop
irr_engine.AssetBindAll()
irr_engine.AssetUpdateAll()
irr_engine.InitGL()
irr_engine.AddLogo("pychrono.png", 0.1)
irr_engine.AddSkyBox()
irr_engine.Start()
```

This script initializes a PyChrono simulation environment, adds a Gator vehicle on a flat rigid terrain, sets up an interactive driver system, and configures visualization with Irrlicht. The vehicle's dynamics, tire model, and rendering are properly configured for a realistic simulation experience.
