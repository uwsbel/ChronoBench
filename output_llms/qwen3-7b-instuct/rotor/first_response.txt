```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment
chrono.Ch纪.initSystem()

# 2. Add the required physical systems and objects
# Create a new system
system = chrono.ChSystemNSC()

# Create a beam
beam = chrono.ChLinkLockHinge()
beam.SetName("beam")
beam.Initialize(system, chrono.ChFrameRef(chrono.ChVectorD(0, 0, 0), chrono.QUNIT))
beam.SetFramePos(chrono.ChFrameD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT))

# Create a flywheel
flywheel = chrono.ChLinkLockHinge()
flywheel.SetName("flywheel")
flywheel.Initialize(system, chrono.ChFrameRef(chrono.ChVectorD(0, 0, 0), chrono.QUNIT))
flywheel.SetFramePos(chrono.ChFrameD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT))

# Create a motor to drive the beam
motor = chrono.ChLinkMotorRevolute()
motor.SetName("motor")
motor.Initialize(system, chrono.ChFrameRef(chrono.ChVectorD(0, 0, 0), chrono.QUNIT), chrono.ChFrameD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT))

# 3. Set necessary default parameters
# Beam properties
beam.SetMass(100)
beam.SetInertia(chrono.ChVectorD(1, 1, 1))

# Flywheel properties
flywheel.SetMass(50)
flywheel.SetInertia(chrono.ChVectorD(1, 1, 1))

# Motor properties
motor.SetMaxTorque(500)

# 4. Create visualization components
# FEM visualization
vis = chrono.ChVisualItem()
vis.SetFEMVisualization(chrono.ChVisualItem.FEM_MESH_FEM_LINES_3D_FEM_CUBES)
vis.SetMaterialSurface(chrono.ChMaterialSurfaceNSC())
vis.SetColor(chrono.ChColor(0.5, 0.5, 0.5))
vis.SetFEMModel(beam.GetFEMModel())
system.AddVisualItem(vis)

# Irrlicht window
window = chronoirr.ChIrrApp(system, "Jeffcott Rotor Simulation", chronoirr.dimension2du(800, 600))
window.AddTypicalCameraController()
window.AddLightSource(chrono.ChVectorD(10, 10, 10), chrono.ChColor(1, 1, 1))
window.AddTypicalPostProcessing()
window.AddTypicalGUI()

# 5. Run the simulation
chrono.Ch纪.runSystem(system, window)
```

This script initializes a PyChrono environment, creates a Jeffcott rotor model with a beam, flywheel, and motor, sets default parameters, and visualizes the system using FEM and an Irrlicht window. The simulation runs until the Irrlicht window is closed.
