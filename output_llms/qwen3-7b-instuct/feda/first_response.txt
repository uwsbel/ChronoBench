```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.core as chcore
import pychrono.postprocess as chpost

# Initialize PyChrono environment
chrono.Ch纪元()
system = chrono.ChSystemNSC()

# Load terrain
terrain = chrono.ChRigidTerrain()
terrain.SetGWorld(chrono.ChVectorD(0, 0, 0))
terrain.SetFriction(0.5)
terrain.SetRestitution(0.1)
terrain.SetElasticity(0.0)
terrain.SetGravity(chrono.ChVectorD(0, -9.81, 0))
terrain.SetTexture(chrono.ChTexture("terrain.png"))

# Add terrain to the system
system.Add(terrain)

# Load FEDA vehicle model
vehicle = chrono.ChVehicleFEDA()
vehicle.Initialize(system)
vehicle.SetPos(chrono.ChVectorD(0, 0, 0))
vehicle.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI / 4))
vehicle.SetContactMethod(chrono.ChContactMethod.SPHERICAL)
vehicle.SetTireModel(chrono.ChTireModel.FULL)

# Set initial velocity
vehicle.SetVel(chrono.ChVectorD(0, 5, 0))

# Set camera to follow the vehicle
camera = chronoirr.ChIrrApp.NewCameraLookAt(chrono.ChVectorD(10, 10, 10), vehicle.GetPos(), chrono.ChVectorD(0, 0, 0))
camera.SetVRMode(False)
camera.SetVRWindowSize(800, 600)
camera.SetVRFramerate(50)

# Add vehicle to the system
system.Add(vehicle)

# Load mesh visualization for vehicle parts
vehicle.SetMeshVisualizationType(chcore.ChMeshVisualizationType.MESH_VISUALIZATION_TYPE_3D)

# Set Irrlicht visualization
app = chronoirr.ChIrrApp(system, "FEDA Vehicle Simulation", chronoirr.SIUnits::mToMeter)
app.AddTypicalSky()
app.AddTypicalLogo("logo水印.png")
app.AddTypicalCamera(chrono.ChVectorD(10, 10, 10), chrono.ChVectorD(0, 0, 0))
app.AddTypicalLights()
app.AddTypicalLights(True)
app.AddTypicalUI()

# Driver system for controlling vehicle dynamics
driver = chpost.ChDriverSteeringThrottleBrake()
driver.SetVehicle(vehicle)
driver.SetSteeringAngle(chrono.CH_C_PI / 4)
driver.SetThrottle(1)
driver.SetBrake(0)

# Simulation loop
while app.GetDevice().run():
    # Update driver inputs
    driver.UpdateInputs()
    # Update vehicle dynamics
    system.DoStepDynamics(1 / 50)
    # Render the scene
    app.BeginScene(True, True, chrono.ChColorB(0.5, 0.5, 0.5))
    app.DrawAll()
    app.EndScene()

# Finalize PyChrono environment
chrono.Ch纪元().Finalize()
```

This script initializes a PyChrono simulation of a FEDA vehicle on a rigid terrain with Irrlicht visualization. It includes setting up the vehicle's parameters, camera, terrain, and driver system, as well as the necessary visualization and simulation loop to run at 50 frames per second.
