```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.postprocess as postprocess
import pychrono.geometry as geom

# Initialize PyChrono environment
chrono.Ch纪元Initialize()

# Create a new Chrono simulation system
sim = chrono.ChSystemNSC()

# Set the timestep
sim.SetTimestep(0.001)

# Add the ground
ground = chrono.ChBodyEasyBox(20, 20, 1, 0, 0, 0, 0.1)
ground.SetPos(chrono.ChVectorD(0, -1, 0))
sim.Add(ground)

# Define the HMMWV parameters
hmmwv_mass = 3000
hmmwv_length = 5
hmmwv_width = 2
hmmwv_height = 1.5

# Create the HMMWV model
hmmwv_model = geom.ChTriangleMeshShape()
hmmwv_model.SetMesh(geom.ChTriangleMeshGenerator().GenerateFromWavefrontFile("hmmwv.obj"))
hmmwv_model.SetInertiaXX(chrono.ChVectorD(100, 100, 100))
hmmwv_model.SetColor(chrono.ChColor(0.8, 0.8, 0.8))

# Create the HMMWV body
hmmwv_body = chrono.ChBody()
hmmwv_body.SetBodyFixed(False)
hmmwv_body.SetPos(chrono.ChVectorD(0, 0, hmmwv_height / 2))
hmmwv_body.SetMass(hmmwv_mass)
hmmwv_body.SetInertiaXX(chrono.ChVectorD(hmmwv_mass * hmmwv_length * hmmwv_width * hmmwv_height))
hmmwv_body.AddAsset(hmmwv_model)
sim.Add(hmmwv_body)

# Add interactive driver controls
driver = chrono.ChDriverSteeringThrottleBrake()
driver.SetVehicle(hmmwv_body)
driver.SetSteeringMaxAngle(chrono.CH_C_PI / 6)
driver.SetThrottleMax(1.0)
driver.SetBrakeMax(-1.0)
sim.Add(driver)

# Create the terrain patches
terrain_patches = [
    chrono.ChTerrainPatch(),
    chrono.ChTerrainPatch(),
    chrono.ChTerrainPatch()
]

# Set terrain parameters for each patch
for i, patch in enumerate(terrain_patches):
    patch.SetMaterial(chrono.ChMaterial())
    patch.SetMesh(geom.ChTriangleMeshGenerator().GenerateFromWavefrontFile(f"terrain_patch_{i}.obj"))
    patch.SetHeightmap(chrono.ChHeightmap())
    patch.SetPosition(chrono.ChVectorD(i * 5, 0, 0))
    sim.Add(patch)

# Set Irrlicht visualization
vis = chronoirr.ChIrrApp(sim, "Chrono: HMMWV on Complex Terrain", chronoirr.dimension2du(1280, 720))
vis.AddTypicalCameraController()
vis.AddTypicalLights()
vis.AddTypicalSky()
vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"), 0.1)
vis.AddTypicalPass()
vis.AddRenderObject(hmmwv_body, chrono.ChVisualParams().SetMaterialColor(chrono.ChColor(0.8, 0.8, 0.8)).SetMeshColor(chrono.ChColor(1, 1, 1)))

# Run the simulation
vis.SetTimestep(sim.GetTimestep())
vis.AssetBindAll()
vis.AssetUpdateAll()
vis.Run()

# Finalize the PyChrono environment
chrono.Ch纪元Finalize()
```

This script initializes a PyChrono simulation of an HMMWV on a complex rigid terrain with diverse surface types, including flat patches with different textures and a mesh-based patch for a bump, as well as a heightmap-based patch for varying elevations. It also incorporates an interactive driver system to control steering, throttle, and braking, and renders the scene in real time using Irrlicht visualization.
