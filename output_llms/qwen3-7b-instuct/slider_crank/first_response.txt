```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import os

# Initialize PyChrono environment
chrono.Ch纪元.init_spring_system()

# 1. Initialize core components
system = chrono.ChSystemNSC()

# 2. Add physical systems and objects
# Floor (truss)
floor = chrono.ChBodyEasyBox(10, 1, 1, 100, False, True)
floor.SetPos(chrono.ChVectorD(0, -1, 0))
system.AddBody(floor)

# Crankshaft
crankshaft = chrono.ChBodyEasyCylinder(0.1, 0.5, 1, 100, False, True)
crankshaft.SetPos(chrono.ChVectorD(0, 0, 0))
crankshaft.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI_2))
crankshaft.SetBodyFixed(True)
system.AddBody(crankshaft)

# Connecting rod
connecting_rod = chrono.ChBodyEasyCylinder(0.05, 0.5, 1, 100, False, True)
connecting_rod.SetPos(chrono.ChVectorD(0.5, 0, 0))
connecting_rod.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI_2))
connecting_rod.SetBodyFixed(False)
system.AddBody(connecting_rod)

# Piston
piston = chrono.ChBodyEasyCylinder(0.05, 0.1, 1, 100, False, True)
piston.SetPos(chrono.ChVectorD(1, 0, 0))
piston.SetRot(chrono.Q_from_AngX(chrono.CH_C_PI_2))
piston.SetBodyFixed(False)
system.AddBody(piston)

# Motor to drive the crankshaft
motor = chrono.ChMotorShaft()
motor.SetBody(crankshaft)
motor.SetTargetSpeed(chrono.ChVectorD(1.0, 0, 0))
system.AddMotor(motor)

# 3. Set default parameters
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# 4. Initialize visualization
vis_system = chronoirr.ChIrrApp(system, "PyChrono Crank-Slider Mechanism", chronoirr.Sclr(1, 1, 1, 1), 1024, 768)
vis_system.AddTypicalLogo("logo.ppm")
vis_system.AddTypicalSky()
vis_system.AddTypicalLights()
vis_system.AddTypicalCamera(chrono.ChVectorD(1, 1, 1))

# Run the simulation
vis_system.AssetBindAll()
vis_system.AssetUpdateAll()
vis_system.SetTimestep(0.001)
vis_system.AssetUpdateAll()
vis_system.AssetRenderAll()
vis_system.Execute()
vis_system.Run()
```

This script initializes a PyChrono environment, adds the required physical components for a crank-slider mechanism, sets default parameters, and visualizes the simulation using the Irrlicht visualization system. The motor drives the crankshaft, simulating the movement of the system over time.
