```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Initialize the PyChrono environment
chrono.SetChTimeStep(0.001)
chrono.Initialize()

# Create the ground
ground = chrono.ChBody()
ground.SetIdentifier(0)
ground.SetBodyFixed(True)
ground.SetPos(chrono.ChVectorD(0, -10, 0))
ground.SetMass(0)
ground.SetCollide(False)
ground.SetName("ground")
ground.Initialize()

# Create the mass
mass = chrono.ChBodyEasyBox(1, 1, 1, 1000, True, True)
mass.SetPos(chrono.ChVectorD(0, 0, 0))
mass.SetName("mass")
mass.Initialize()

# Create the spring and damper
spring = chrono.ChLinkTSDA()
spring.SetName("spring")
spring.Initialize(mass, ground, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 1, 0))
spring.SetK(1000)  # Spring constant
spring.SetC(100)   # Damping coefficient
spring.SetRestLength(1)  # Rest length of the spring

# Create the visualization scene
scene = chronoirr.ChIrrApp(chrono.GetSystem(), "PyChrono Mass-Spring-Damper", chronoirr.dimension2du(1280, 720))
scene.AddTypicalLights()
scene.AddCamera(chrono.ChVectorD(-10, 10, 10), chrono.ChVectorD(0, 0, 0))

# Add shapes to the mass and spring for visualization
mass_shape = chrono.ChTriangleMeshShape()
mass_shape.SetMesh(chrono.ChTriangleMesh())
mass_shape.SetColor(chrono.ChColor(1, 0, 0))
mass_shape.SetName("mass_shape")
mass_shape.AddMesh(chrono.ChTriangleMesh())
mass_shape.GetMesh().LoadWavefrontMesh("box.obj")
mass.SetVisualShape(mass_shape)

spring_shape = chrono.ChTriangleMeshShape()
spring_shape.SetMesh(chrono.ChTriangleMesh())
spring_shape.SetColor(chrono.ChColor(0, 1, 0))
spring_shape.SetName("spring_shape")
spring_shape.AddMesh(chrono.ChTriangleMesh())
spring_shape.GetMesh().LoadWavefrontMesh("spring.obj")
spring_shape.GetMesh().SetTexture("spring_texture.png")

# Add the spring shape to the scene
scene.AddSceneObject(spring_shape)

# Run the simulation
scene.AddLogItem(chrono.LogPos, mass.GetPos(), "mass position")
scene.AddLogItem(chrono.LogVel, mass.GetPos_dt(), "mass velocity")
scene.AddLogItem(chrono.LogTorque, chrono.ChVectorD(0, 0, 0), "torque")
scene.AddLogItem(chrono.LogForce, chrono.ChVectorD(0, 0, 0), "force")
scene.AddLogItem(chrono.LogEnergy, chrono.ChVectorD(0), "energy")
scene.AddLogItem(chrono.LogSpringLength, chrono.ChVectorD(0), "spring length")
scene.AddLogItem(chrono.LogSpringForce, chrono.ChVectorD(0), "spring force")
scene.AddLogItem(chrono.LogSpringEnergy, chrono.ChVectorD(0), "spring energy")
scene.AddLogItem(chrono.LogDamping, chrono.ChVectorD(0), "damping force")
scene.AddLogItem(chrono.LogKinetic, chrono.ChVectorD(0), "kinetic energy")
scene.AddLogItem(chrono.LogPotential, chrono.ChVectorD(0), "potential energy")
scene.AddLogItem(chrono.LogTotalEnergy, chrono.ChVectorD(0), "total energy")

scene.SetTimestep(0.001)
scene.SetTryRealtime(True)
scene.AssetUpdate()
scene.AssetPostUpdate()
scene.AssetPreUpdate()
scene.AssetPrePhysicsUpdate()
scene.AssetPostPhysicsUpdate()
scene.AssetPostUpdate()
scene.AssetPreRenderUpdate()
scene.AssetPostRenderUpdate()

chrono.RunStep(scene.GetSystem(), scene.GetSystem().GetChTimeStep(), 1000)

# Finalize the PyChrono environment
chrono.Finalize()
```

This script initializes a PyChrono environment, creates a ground, a mass, and a spring-damper system. It also sets up Irrlicht visualization with appropriate lighting and camera settings, and provides visual shapes for the mass and spring. The script runs the simulation for a specified number of steps and finalizes the PyChrono environment.
