```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import math

# Initialize PyChrono environment
chrono.Ch纪元()
system = chrono.ChSystemNSC()

# Ground body
ground = chrono.ChBodyEasyBox(1000, 1, 1, 1000)
ground.SetPos(chrono.ChVectorD(0, -10, 0))
ground.SetBodyFixed(True)
system.Add(ground)

# Pendulum mass and inertia
mass = 10
Ixx = 100
Iyy = 100
Izz = 100
pendulum = chrono.ChBodyEasyCylinder(mass, 0.5, 10, True, False)
pendulum.SetPos(chrono.ChVectorD(0, 0, 0))
pendulum.SetBodyFixed(False)
system.Add(pendulum)

# Inertia tensor
I = chrono.ChVectorD(Ixx, Iyy, Izz)

# Calculate the inertia matrix
I_mat = chrono.ChMatrixSpd()
I_mat.SetDiag(I)
I_mat(0, 2) = I(2)
I_mat(2, 0) = I_mat(0, 2)
I_mat(1, 2) = I(2)
I_mat(2, 1) = I_mat(1, 2)

# Apply inertia to the pendulum
pendulum.SetInertiaXX(I_mat)

# Revolute joint connecting ground and pendulum
joint = chrono.ChLinkTSDA()
joint.Initialize(ground, pendulum, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT))
joint.SetName("pendulum_joint")
joint.SetPos(chrono.ChVectorD(0, 0, 0))
joint.SetMaxTorque(chrono.ChVectorD(1000, 0, 0))
system.Add(joint)

# Visualization system
vis_system = chronoirr.ChVisualSystemIrrlicht()
vis_system.AttachSystem(system)
vis_system.SetWindowSize(800, 600)
vis_system.SetTitle("Simple Pendulum Simulation")
vis_system.SetLookAt(chrono.ChVectorD(0, 10, 10))
vis_system.SetPerspectiveFieldOfView(math.pi / 3)
vis_system.SetTarget(chrono.ChVectorD(0, 0, 0))
vis_system.SetZoom(1.5)
vis_system.Initialize()

# Log position and velocity
log_file = open("pendulum_log.txt", "w")
log_file.write("Time, Position, Velocity\n")

# Simulation loop
step = 0.001
time = 0
while vis_system.GetDevice().run():
    time += step
    system.DoStepDynamics(step)
    
    # Log pendulum position and velocity
    pos = pendulum.GetPos()
    vel = pendulum.GetPos_dt()
    log_file.write(f"{time}, {pos.x}, {vel.x}\n")
    
    # Update visualization
    vis_system.BeginScene()
    vis_system.Render()
    vis_system.EndScene()

log_file.close()
```

This script sets up a simple pendulum simulation using PyChrono and Irrlicht for visualization. It initializes the system, adds a ground body, a pendulum with mass and inertia, and connects them with a revolute joint. The script also logs the pendulum's position and velocity over time and renders the simulation in a window.
