```python
import pychrono as chrono
import pychrono.fea as fea
import pychrono.irrlicht as chronoirr
import math

# Create the Chrono physical system
system = chrono.ChSystemSMC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, -9.81, 0))  # Set gravity in the negative Y direction

# Create the mesh for the beam
mesh = fea.ChMesh()

# Create the beam section properties
section = fea.ChBeamSectionEulerAdvanced()
section.SetDensity(7800)  # Set density of the beam material
section.SetYoungModulus(2.1e11)  # Set Young's modulus
section.SetGshear(0.4 * 2.1e11)  # Set shear modulus
section.SetSectionAsRectangular(0.02, 0.02)  # Set rectangular cross-section with width and height of 0.02 m
section.SetIyy(0.0001)  # Set moment of inertia about the Y-axis
section.SetIzz(0.0001)  # Set moment of inertia about the Z-axis
section.SetAsCircular(0.01, 0.01)  # Set circular cross-section with diameter of 0.01 m

# Create nodes for the beam elements
nodeA = fea.ChNodeFEAxyzrot(chrono.ChVector3d(0, 0, 0), chrono.ChQuaterniond(1, 0, 0, 0))
nodeB = fea.ChNodeFEAxyzrot(chrono.ChVector3d(0, 0, 1, 0), chrono.ChQuaterniond(1, 0, 0, 0))
nodeC = fea.ChNodeFEAxyzrot(chrono.ChVector3d(0, 0, 2, 0), chrono.ChQuaterniond(1, 0, 0, 0))
nodeD = fea.ChNodeFEAxyzrot(chrono.ChVector3d(0, 0, 3, 0), chrono.ChQuaterniond(1, 0, 0, 0))

# Create and add beam elements to the mesh
elementAB = fea.ChElementBeamEulerAdvanced()
elementAB.SetNodes(nodeA, nodeB)
elementAB.SetSection(section)
elementAB.SetCoordinates(chrono.ChVector3d(0, 0, 0), chrono.ChVector3d(0, 0, 1))
elementAB.SetId(1)
mesh.AddElement(elementAB)

elementBC = fea.ChElementBeamEulerAdvanced()
elementBC.SetNodes(nodeB, nodeC)
elementBC.SetSection(section)
elementBC.SetCoordinates(chrono.ChVector3d(0, 0, 0), chrono.ChVector3d(0, 0, 1))
elementBC.SetId(2)
mesh.AddElement(elementBC)

elementCD = fea.ChElementBeamEulerAdvanced()
elementCD.SetNodes(nodeC, nodeD)
elementCD.SetSection(section)
elementCD.SetCoordinates(chrono.ChVector3d(0, 0, 0), chrono.ChVector3d(0, 0, 1))
elementCD.SetId(3)
mesh.AddElement(elementCD)

# Add nodes to the mesh
mesh.AddNode(nodeA)
mesh.AddNode(nodeB)
mesh.AddNode(nodeC)
mesh.AddNode(nodeD)

# Create and add constraints to the mesh
constraintA = fea.ChLinkBeamNodeNode()
constraintA.Initialize(nodeA, nodeB)
constraintA.SetConstrainedDirection(fea.ChLinkBeamNodeNode.ConstrainedDirection_X)
mesh.AddConstraint(constraintA)

constraintB = fea.ChLinkBeamNodeNode()
constraintB.Initialize(nodeB, nodeC)
constraintB.SetConstrainedDirection(fea.ChLinkBeamNodeNode.ConstrainedDirection_X)
mesh.AddConstraint(constraintB)

constraintC = fea.ChLinkBeamNodeNode()
constraintC.Initialize(nodeC, nodeD)
constraintC.SetConstrainedDirection(fea.ChLinkBeamNodeNode.ConstrainedDirection_X)
mesh.AddConstraint(constraintC)

# Create and add the mesh to the system
system.Add(mesh)

# Create and initialize the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(chrono.ChVector2i(1024, 768))
vis.SetWindowTitle('Beam buckling')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 0.5, 3), chrono.ChVector3d(0, 0.5, 0))
vis.AddTypicalLights()

# Set the solver and timestepper for the system
system.SetSolverType(chrono.ChSolver.Type_SOR)
system.GetSolver().SetMaxIterations(25)
system.SetTimestepperType(chrono.ChTimestepper.Type_EULER_IMPLICIT_LINEARIZED)

# Initialize the simulation parameters
time = 0
time_end = 10
time_step = 0.01

# Create a custom motor function for the simulation
def mymotorfunction(myt):
    return 0.5 * (1 - math.cos(myt * 2 * math.pi / 2))

# Create and initialize the motor for the simulation
motor = fea.ChLinkMotorRotationSpeed()
motor.Initialize(nodeD, nodeC, fea.ChFramed(nodeD.GetPos()))
motor.SetMotorFunction(chrono.ChFunctionConst(mymotorfunction))
mesh.AddLink(motor)

# Run the simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    system.DoStepDynamics(time_step)
    time += time_step
```
