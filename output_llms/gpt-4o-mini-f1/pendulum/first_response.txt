```python
import pychrono as chrono
import pychrono.irrlicht as irr
import math

# Create the Chrono simulation system
system = chrono.ChSystemNSC()

# Create the ground body
ground = chrono.ChBody()
system.Add(ground)

# Create the pendulum body
pendulum = chrono.ChBody()
system.Add(pendulum)

# Set the mass and inertia of the pendulum body
pendulum.SetMass(1.0)
pendulum.SetInertiaXX(chrono.ChVector3d(0.1, 0.1, 0.1))

# Set the initial position of the pendulum body
pendulum.SetPos(chrono.ChVector3d(1.0, 0.0, 0.0))

# Create a revolute joint to connect the ground and the pendulum
joint = chrono.ChLinkLockRevolute()
joint.Initialize(ground, pendulum, chrono.ChCoordsysd(chrono.ChVector3d(0.0, 0.0, 0.0), chrono.ChQuaterniond(1, 0, 0, 0)))
system.Add(joint)

# Create the Irrlicht visualization system
vis = chrono.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(chrono.ChVector2i(1024, 768))
vis.SetWindowTitle('Simple Pendulum Simulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 2, -5), chrono.ChVector3d(0, 1, 0))
vis.AddTypicalLights()

# Set the simulation time step
time_step = 0.01

# Log the pendulum's position and velocity every 100 time steps
log_interval = 100

# Run the simulation
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    system.DoStepDynamics(time_step)

    # Log the pendulum's position and velocity
    if int(system.GetChTime() / time_step) % log_interval == 0:
        pos = pendulum.GetPos()
        vel = pendulum.GetPosDt()
        print(f'Time: {system.GetChTime():.2f}, Position: {pos}, Velocity: {vel}')
```
