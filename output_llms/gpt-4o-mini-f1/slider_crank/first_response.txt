```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import math

# Create the Chrono::Engine physical system
sys = chrono.ChSystemNSC()
sys.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# Create the floor (truss)
floor = chrono.ChBodyEasyBox(20, 1, 20, 1000, True, True)
floor.SetPos(chrono.ChVector3d(0, -1, 0))
floor.SetFixed(True)
floor.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
sys.Add(floor)

# Create the crankshaft
crankshaft = chrono.ChBodyEasyCylinder(0.2, 5, 1000, True, True)
crankshaft.SetPos(chrono.ChVector3d(0, 0, 0))
crankshaft.SetRot(chrono.ChQuaterniond(1, 0, 0, 0))
crankshaft.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
sys.Add(crankshaft)

# Create the connecting rod
connecting_rod = chrono.ChBodyEasyBox(0.2, 0.5, 1.5, 1000, True, True)
connecting_rod.SetPos(chrono.ChVector3d(0, 0, 0))
connecting_rod.SetRot(chrono.ChQuaterniond(1, 0, 0, 0))
connecting_rod.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
sys.Add(connecting_rod)

# Create the piston
piston = chrono.ChBodyEasyCylinder(0.5, 1, 1000, True, True)
piston.SetPos(chrono.ChVector3d(0, 0, 0))
piston.SetRot(chrono.ChQuaterniond(1, 0, 0, 0))
piston.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
sys.Add(piston)

# Create the crankshaft joint
crankshaft_joint = chrono.ChLinkLockRevolute()
crankshaft_joint.Initialize(crankshaft, floor, chrono.ChFramed(crankshaft.GetPos(), chrono.ChQuaterniond(1, 0, 0, 0)))
sys.Add(crankshaft_joint)

# Create the connecting rod joint
connecting_rod_joint = chrono.ChLinkLockRevolute()
connecting_rod_joint.Initialize(connecting_rod, crankshaft, chrono.ChFramed(connecting_rod.GetPos(), chrono.ChQuaterniond(1, 0, 0, 0)))
sys.Add(connecting_rod_joint)

# Create the piston joint
piston_joint = chrono.ChLinkLockRevolute()
piston_joint.Initialize(piston, connecting_rod, chrono.ChFramed(piston.GetPos(), chrono.ChQuaterniond(1, 0, 0, 0)))
sys.Add(piston_joint)

# Create the crankshaft motor
crankshaft_motor = chrono.ChLinkMotorRotationSpeed()
crankshaft_motor.Initialize(crankshaft, floor, chrono.ChFramed(crankshaft.GetPos(), chrono.ChQuaterniond(1, 0, 0, 0)))
crankshaft_motor.SetMotorFunction(chrono.ChFunctionConst(chrono.CH_PI))
sys.Add(crankshaft_motor)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(chrono.ChVector2i(1280, 720))
vis.SetWindowTitle('Crank-slider mechanism')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 2, -6), chrono.ChVector3d(0, 1, 0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, 4.5, -2.5), chrono.ChVector3d(0, 0.5, 0), 3, 4, 10, 40, 512)

# Simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(1e-2)
```
