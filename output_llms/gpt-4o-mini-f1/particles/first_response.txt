```python
import pychrono as chrono
import pychrono.irrlicht as irr
import math
import random

# Create the Chrono simulation system
sys = chrono.ChSystemNSC()
sys.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, 0))  # No gravity in the system

# Create a particle emitter to generate particles
class ParticleEmitter:
    def __init__(self, sys, num_particles):
        self.sys = sys
        self.num_particles = num_particles
        self.particles = []

    def emit(self):
        for _ in range(self.num_particles):
            # Create a particle body
            particle = chrono.ChBodyEasySphere(0.5, 1000, True, True)
            particle.SetPos(chrono.ChVector3d(random.uniform(-10, 10), random.uniform(-10, 10), random.uniform(-10, 10)))
            particle.SetPosDt(chrono.ChVector3d(random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1)))
            particle.SetRot(chrono.ChQuaterniond(1, 0, 0, 0))
            particle.SetRotDt(chrono.ChVector3d(0, 0, 0))
            particle.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
            particle.GetCollisionModel().SetDefaultSuggestedEnvelope(0.0025)
            particle.GetCollisionModel().SetDefaultSuggestedMargin(0.0025)
            particle.SetFixed(False)
            particle.GetMaterialSurfaceNSC().SetFriction(0.4)
            particle.GetMaterialSurfaceNSC().SetRestitution(0.1)
            particle.GetCollisionModel().BuildModel()
            self.sys.Add(particle)
            self.particles.append(particle)

# Create the particle emitter
emitter = ParticleEmitter(sys, 100)

# Create the Irrlicht visualization
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(chrono.ChVector2ui(1280, 720))
vis.SetWindowTitle('Particle Emitter with Custom Gravity')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 0, 10))
vis.AddTypicalLights()

# Simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    emitter.emit()  # Emit particles
    for particle in emitter.particles:
        # Apply custom gravitational force to each particle
        force = chrono.ChVector3d(0, 0, 0)
        for other in emitter.particles:
            if particle is not other:
                r = other.GetPos() - particle.GetPos()
                distance = r.Length()
                if distance > 0:
                    force += r.GetNormalized() * (particle.GetMass() * other.GetMass()) / (distance * distance)
        particle.SetForce(force)
    sys.DoStepDynamics(0.01)  # Advance the simulation by one step
```
