```python
import pychrono as chrono  # Import the main PyChrono module.
import pychrono.fea as fea  # Import the finite element analysis (FEA) module.
import pychrono.pardisomkl as mkl  # Import the Pardiso MKL linear solver module.
import pychrono.irrlicht as chronoirr  # Import the Irrlicht visualization module.

print("Example: PyChrono using beam finite elements")  # Print an introductory statement.

# Create the physical system that will be simulated.
sys = chrono.ChSystemSMC()

# Create a mesh, which is a container for elements and their referenced nodes.
mesh = fea.ChMesh()

# --------- Start Euler-Bernoulli beam setup using builder helper ---------------
# Create a section object for Euler-Bernoulli beam properties.
euler_section = fea.ChBeamSectionEuler()

# Set beam section properties - using rectangular cross section.
beam_wy = 0.012
beam_wz = 0.025
euler_section.SetAsRectangularSection(beam_wy, beam_wz)

# Material properties.
euler_section.SetYoungModulus(0.01e9)  # Young's modulus
euler_section.SetGwithPoissonRatio(0.01e9, 0.3)  # Shear modulus via G and Poisson ratio (correct method for Euler section)

# Create a builder for Euler beam elements
builder = fea.ChBuilderBeamEuler()

# Set the mesh where the beam will be built
builder.BuildBeam(mesh,                  # the mesh to put elements and nodes in
                  euler_section,         # the beam section
                  chrono.ChVectorD(0, 0, -0.1),  # beam start point (A)
                  chrono.ChVectorD(0.2, 0, -0.1), # beam end point (B)
                  chrono.VECT_Y,         # Y up direction for beam section reference
                  5)                     # number of beam elements

# Fix the last node of the created beam
builder.GetLastBeamNodes().back().SetFixed(True)

# Apply a force (0, -1, 0) to the first node of the created beam
builder.GetLastBeamNodes().front().SetForce(chrono.ChVectorD(0, -1, 0))
# --------- End Euler-Bernoulli beam setup using builder helper -----------------

# --- Original beam section and nodes/elements (kept for contrast; they remain in code but can be optionally removed) ---
# Commenting out this section is optional, but instructions don't say to remove it,
# so we keep it here without modification except for node fixing approach below.

# Create a section object for advanced Euler beam properties. (for original beams)
msection = fea.ChBeamSectionEulerAdvanced()

# Set the width and height of the rectangular section of the beam.
beam_wy_adv = 0.012
beam_wz_adv = 0.025
msection.SetAsRectangularSection(beam_wy_adv, beam_wz_adv)  # Set the cross-sectional shape and size of the beam.

# Set the material properties of the beam.
msection.SetYoungModulus(0.01e9)  # Young's modulus.
msection.SetShearModulus(0.01e9 * 0.3)  # Shear modulus.
msection.SetRayleighDamping(0.000)  # Damping coefficient.
msection.SetCentroid(0, 0.02)  # Position of centroid.
msection.SetShearCenter(0, 0.1)  # Position of shear center.
# Correct the rotation: original code uses 45 * CH_RAD_TO_DEG but CH_RAD_TO_DEG is ~57.2958; for degrees to radians it should be CH_DEG_TO_RAD
# Let's assume the intent was 45 degrees converted to radians for SetSectionRotation:
msection.SetSectionRotation(45 * chrono.CH_DEG_TO_RAD)  # Rotation angle in radians.

# Define the length of the beam elements.
beam_L = 0.1

# Create nodes for the positions that will be used for beams.
hnode1 = fea.ChNodeFEAxyzrot(chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))  # Node at origin.
hnode2 = fea.ChNodeFEAxyzrot(chrono.ChFrameD(chrono.ChVectorD(beam_L, 0, 0)))  # Node at beam length.
hnode3 = fea.ChNodeFEAxyzrot(chrono.ChFrameD(chrono.ChVectorD(beam_L * 2, 0, 0)))  # Node at twice the beam length.

# Add nodes to the mesh.
mesh.AddNode(hnode1)
mesh.AddNode(hnode2)
mesh.AddNode(hnode3)

# Create the first beam element and set its properties.
belement1 = fea.ChElementBeamEuler()
belement1.SetNodes(hnode1, hnode2)  # Connect the first beam between node 1 and node 2.
belement1.SetSection(msection)  # Assign the section
mesh.AddElement(belement1)  # Add the first beam element.

# Create the second beam element and set its properties.
belement2 = fea.ChElementBeamEuler()
belement2.SetNodes(hnode2, hnode3)  # Connect beam between node 2 and node 3.
belement2.SetSection(msection)  # Assign the section
mesh.AddElement(belement2)  # Add the second beam element.

# Apply a force to node 2: 4 N in x-direction and 2 N in y-direction.
hnode2.SetForce(chrono.ChVectorD(4, 2, 0))

# Apply a torque to node 3: -0.04 Nm in y-direction.
hnode3.SetTorque(chrono.ChVectorD(0, -0.04, 0))

# Replace direct fixing of hnode1 with a constraint:
# hnode1.SetFixed(True)  # <-- This line is commented out per instruction.

# Create a fixed truss, which is a rigid body that won't move.
mtruss = chrono.ChBody()
mtruss.SetFixed(True)
sys.Add(mtruss)

# Create and initialize a constraint that connects node 3 to the fixed truss.
constr_bc = chrono.ChLinkMateGeneric()
constr_bc.Initialize(hnode3, mtruss, False, hnode3.Frame(), hnode3.Frame())
sys.Add(constr_bc)
constr_bc.SetConstrainedCoords(True, True, True, True, True, True)

# Create and initialize constraint to fix node 1 observed in original mesh (instead of SetFixed)
constr_d = chrono.ChLinkMateGeneric()
constr_d.Initialize(hnode1, mtruss, False, hnode1.Frame(), hnode1.Frame())
sys.Add(constr_d)
# Constrain only y,z translation; no rotation constraints
constr_d.SetConstrainedCoords(False, True, True, False, False, False)

# Disable automatic gravity for FEA elements in this demo.
mesh.SetAutomaticGravity(False)

# Add the mesh to the physical system.
sys.Add(mesh)

# Add visualization for the beams in the mesh.
visualizebeamA = chrono.ChVisualShapeFEA(mesh)
visualizebeamA.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_ELEM_BEAM_MZ)  # Visualize bending moments.
visualizebeamA.SetColorscaleMinMax(-0.4, 0.4)
visualizebeamA.SetSmoothFaces(True)
visualizebeamA.SetWireframe(False)
mesh.AddVisualShapeFEA(visualizebeamA)

# Add visualization for the nodes in the mesh.
visualizebeamC = chrono.ChVisualShapeFEA(mesh)
visualizebeamC.SetFEMglyphType(chrono.ChVisualShapeFEA.GlyphType_NODE_CSYS)  # Coordinate systems at nodes.
visualizebeamC.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_NONE)  # No data type visualization.
visualizebeamC.SetSymbolsThickness(0.006)
visualizebeamC.SetSymbolsScale(0.01)
visualizebeamC.SetZbufferHide(False)
mesh.AddVisualShapeFEA(visualizebeamC)

# Create an Irrlicht visualization window.
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('FEA beams')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0.1, 0.1, 0.2))
vis.AddTypicalLights()

# Set the MKL Pardiso linear solver for system.
msolver = mkl.ChSolverPardisoMKL()
sys.SetSolver(msolver)

# Simulation loop.
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(0.001)
```
