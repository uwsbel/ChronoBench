```python
import pychrono as chrono
import pychrono.fea as fea
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Ensure data path is set

# Create a physical system
system = chrono.ChSystemSMC()

# 2. Add the required physical systems and objects as specified

# Create a mesh to hold the beam elements and nodes
mesh = fea.ChMesh()

# Beam properties
beam_length = 1.0  # length of each beam element
num_elements = 5   # number of beam elements in a chain

# Material properties for the beam
beam_material = chrono.ChBeamSectionEulerAdvanced()
beam_material.SetYoungModulus(2.1e11)  # Steel approx Young's modulus in Pa
beam_material.SetGwithPoissonRatio(8.1e10, 0.3)  # Shear modulus and Poisson ratio
beam_material.SetDensity(7800)  # Steel density in kg/m^3
beam_material.SetAsRectangularSection(0.02, 0.04)  # cross-section 2cm x 4cm

# Create nodes for the beam chain
nodes = []
for i in range(num_elements + 1):
    node = fea.ChNodeFEAxyzrot(chrono.ChVectorD(i * beam_length, 0, 0))
    node.SetMass(0)  # mass is defined by beam section, so set node mass to zero
    mesh.AddNode(node)
    nodes.append(node)

# Create beam elements connecting the nodes
for i in range(num_elements):
    beam = fea.ChElementBeamEuler()
    beam.SetNodes(nodes[i], nodes[i + 1])
    beam.SetSection(beam_material)
    beam.SetAlphaDamp(0.05)  # some damping
    mesh.AddElement(beam)

# Fix the first node in space (clamped boundary condition)
nodes[0].SetFixed(True)

# Add the mesh to the system
system.Add(mesh)

# 3. Set necessary default parameters such as positions, forces, and interactions

# Add gravity
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Visualization of the mesh and beams
visualize = fea.ChVisualizationFEAmesh(mesh)
visualize.SetFEMdataType(fea.ChVisualizationFEAmesh.E_PLOT_NODE_POS)
visualize.SetColorscaleMinMax(-0.1, 0.1)
visualize.SetShrinkElements(True, 0.85)
visualize.SetSmoothFaces(True)
mesh.AddVisualizationLayer(visualize)

# Visualize beam elements as cylinders
beam_vis = fea.ChVisualizationFEAmesh(mesh)
beam_vis.SetFEMglyphType(fea.ChVisualizationFEAmesh.E_GLYPH_BEAM_MATERIAL)
beam_vis.SetSymbolsThickness(0.002)
beam_vis.SetDefaultSymbolsColor(chrono.ChColor(0.2, 0.2, 0.8))
mesh.AddVisualizationLayer(beam_vis)

# 4. Setup Irrlicht visualization
application = chronoirr.ChIrrApp(system, "Beam Finite Element Simulation", chronoirr.dimension2du(1024, 768))

application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(chronoirr.vector3df(1.5, 1.0, 1.5))

application.AssetBindAll()
application.AssetUpdateAll()

# Simulation loop parameters
application.SetTimestep(1e-3)

# Run the simulation loop
application.SetPaused(False)
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```
