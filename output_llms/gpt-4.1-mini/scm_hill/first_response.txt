```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.vehicle.scm as veh_scm
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Ensure data path is set

# Simulation parameters
step_size = 1e-3  # simulation step size
t_end = 10        # simulation end time

# 2. Add the required physical systems and objects as specified

# Create the HMMWV vehicle system
vehicle = veh.HMMWV_Full()
vehicle.SetContactMethod(chrono.ChContactMethod_SMC)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 1.0), chrono.QUNIT))  # initial position (x,y,z)
vehicle.SetTireType(veh.TireModelType_TMEASY)
vehicle.SetInitFwdVel(0.0)  # initial forward velocity
vehicle.Initialize()

# Create the SCM deformable terrain
terrain = veh_scm.SCMDeformableTerrain(vehicle.GetSystem())
terrain.SetSoilParameters(
    1.2e6,   # Bekker Kphi
    0,       # Bekker Kc
    1.1e6,   # Bekker n exponent
    0,       # Mohr cohesive limit (Pa)
    30,      # Mohr friction limit (degrees)
    2e7,     # Janosi shear coefficient (Pa/m)
    0.01,    # Elastic stiffness (N/m^3)
    3e4      # Damping (N s/m^3)
)
terrain.Initialize(0.6, 4.0, 8.0)  # length, width, max height

# Load a height map for terrain initialization
# For demonstration, we create a flat height map with zero height
# In practice, you would load a height map file or generate one
import numpy as np
hm_rows = 100
hm_cols = 100
height_map = np.zeros((hm_rows, hm_cols))
terrain.SetHeightMap(height_map.tolist(), 0.6, 4.0)

# 3. Initialize a driver system for vehicle control
driver = veh.ChIrrGuiDriver(vehicle)
driver.Initialize()

# 4. Create a real-time visualization using Irrlicht with camera settings and lighting
application = veh.ChWheeledVehicleIrrApp(vehicle, "HMMWV SCM Terrain Simulation")
application.SetTimestep(step_size)
application.SetChaseCamera(chrono.ChVectorD(0.0, 0.0, 1.75), 6.0, 0.5)
application.AddTypicalLights()
application.AssetBindAll()
application.AssetUpdateAll()

# 5. Implement the simulation loop that synchronizes and advances all systems

time = 0.0
while application.GetDevice().run() and time < t_end:
    # Begin scene
    application.BeginScene()
    application.DrawAll()

    # Synchronize modules
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver.GetInputs(), terrain)
    application.Synchronize(time)

    # Advance simulation by step size
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    application.Advance(step_size)

    # End scene
    application.EndScene()

    # Increment time
    time += step_size
```
