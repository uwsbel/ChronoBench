```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Ensure data path is set

# Create the physical system (ChSystem)
system = chrono.ChSystemNSC()

# 2. Add the required physical systems and objects as specified

# Create the UAZBUS vehicle system
vehicle = veh.UAZBUS()
vehicle.SetContactMethod(chrono.ChContactMethod_NSC)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0.5), chrono.QUNIT))  # initial position (x,y,z) and no rotation
vehicle.SetTireType(veh.TireModelType_TMEASY)
vehicle.SetStepSize(1e-3)
vehicle.Initialize()

# Create the rigid terrain
terrain = veh.RigidTerrain(vehicle.GetSystem())
patch = terrain.AddPatch(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT),
                         chrono.ChVectorD(50, 50, 1))
patch.SetColor(chrono.ChColor(0.3, 0.3, 0.3))
patch.SetFriction(0.9)
patch.SetRestitution(0.01)
terrain.Initialize()

# Initialize the driver system for vehicle control
driver = veh.ChIrrGuiDriver(vehicle)
driver.Initialize()

# 3. Set necessary default parameters such as positions, forces, and interactions
# (Already set initial position above; driver and terrain friction/restitution set)

# Create the Irrlicht visualization system
vis = chronoirr.ChIrrApp(vehicle.GetSystem(), "UAZBUS Simulation", chronoirr.dimension2du(1280, 720))
vis.AddTypicalLights()
vis.AddTypicalSky()
vis.AddTypicalCamera(chronoirr.vector3df(0, -8, 3))
vis.AssetBindAll()
vis.AssetUpdateAll()

# Simulation parameters
timestep = 1e-3
vis.SetTimestep(timestep)

# 4. Implement the simulation loop that synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.

while vis.Run():
    vis.BeginScene()
    vis.DrawAll()

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Synchronize modules
    time = vehicle.GetSystem().GetChTime()
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    vis.Synchronize(time)

    # Advance simulation by timestep
    driver.Advance(timestep)
    terrain.Advance(timestep)
    vehicle.Advance(timestep)
    vis.Advance(timestep)

    vis.EndScene()
```
