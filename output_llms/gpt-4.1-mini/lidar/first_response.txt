```python
import pychrono as chrono
import pychrono.sensor as sens
import os
import math

# 1. Initialize the PyChrono environment and core components.
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Ensure Chrono data path is set for meshes, shaders, etc.

# Create the physical system
system = chrono.ChSystemSMC()
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))  # No gravity since lidar scanning a fixed mesh

# 2. Add the required physical systems and objects as specified.

# Create a fixed body with the triangular mesh (from Wavefront .obj file)
mesh_body = chrono.ChBody()
mesh_body.SetBodyFixed(True)
mesh_body.SetCollide(False)

# Load .obj mesh:
mesh_obj_path = os.path.join(chrono.GetChronoDataPath(), "sensor", "objects", "terrain_simple.obj")
# Replace "terrain_simple.obj" with your specific obj filename if different.

# Create a visualization asset from the mesh obj file
mesh_asset = chrono.ChTriangleMeshShape()
tm = chrono.ChTriangleMeshConnected()
if not tm.LoadWavefrontMesh(mesh_obj_path, True, True):
    raise Exception(f"Failed to load mesh from {mesh_obj_path}")
mesh_asset.SetMesh(tm)
mesh_asset.SetName("mesh_visualization")
mesh_asset.SetStatic(True)
mesh_asset.SetColor(chrono.ChColor(0.7, 0.7, 0.7))

mesh_body.AddAsset(mesh_asset)
system.Add(mesh_body)

# 2. Add the sensor manager and a lidar sensor to the body

# Create sensor manager with the system and update rate
sensor_manager = sens.SensorManager(system)
sensor_manager.SetVerbose(False)
update_rate = 60  # Hz
sensor_manager.SetSensorUpdateRate(update_rate)
sensor_manager.SetWindowHandle(800, 600)

# Create and position the Lidar sensor on mesh_body:
lidar = sens.ChLidarSensor(mesh_body)

# Configure lidar parameters
lidar.SetName("lidar_sensor")
lidar.SetOffsetPose(chrono.ChFrameD(chrono.ChVectorD(1, 0, 0), chrono.Q_from_AngY(math.radians(90))))
lidar.SetScanWindow(math.radians(180))  # 180 degrees scan angle horizontally
lidar.SetNumberOfVerticalChannels(3)
lidar.SetVerticalFOV(math.radians(10))  # 10 degrees vertical field of view total
lidar.SetHorizontalResolution(200)  # Number of points per scan line
lidar.SetMaxDistance(10)  # Max lidar distance in meters

# Enable saving of lidar scans to a CSV file
output_folder = "lidar_output"
if not os.path.exists(output_folder):
    os.makedirs(output_folder)
lidar.SetFilePathCSV(output_folder + "/lidar_scan.csv")
lidar.SetFilePathPCD(output_folder + "/lidar_scan.pcd")

# Add noise filter: e.g. Gaussian noise with sigma=0.1m
gauss_noise = sens.ChLidarNoiseGaussian(0.1)
lidar.AddFilter(gauss_noise)

# Add visualization filter to see lidar points
viz_filter = sens.ChFilterVisualizePointCloud()
lidar.AddFilter(viz_filter)

# Add a filter to save scan data (CSV and PCD)
csvfilter = sens.ChFilterSaveCSV()
lidar.AddFilter(csvfilter)

pcdfilter = sens.ChFilterSavePCD()
lidar.AddFilter(pcdfilter)

sensor_manager.AddSensor(lidar)

# 3. Set necessary default parameters such as positions, forces, and interactions.

# Since mesh body is fixed with no gravity, no external forces needed

# Simulation parameters
simulation_time = 10  # seconds
step_size = 1.0 / 500  # simulation step size

# For dynamic updating, orbit the lidar around the mesh in a circle radius=1m at 0.5 Hz
orbit_radius = 1.0
orbit_freq = 0.5  # Hz

# Run simulation: advance time, update lidar position in orbit, print lidar buffer data each step
print_interval = 1.0 / update_rate  # print once per lidar scan update

num_steps = int(simulation_time / step_size)
next_print_time = 0

for step in range(num_steps):
    time = step * step_size

    # Advance the physical system
    system.DoStepDynamics(step_size)

    # Update lidar position orbiting around mesh center (0,0,0)
    angle = 2 * math.pi * orbit_freq * time
    x = orbit_radius * math.cos(angle)
    y = orbit_radius * math.sin(angle)
    z = 0.2  # slightly above the mesh plane

    # Update lidar offset pose relative to body (which is at origin fixed)
    # We want lidar to orbit so we change the relative pose of lidar sensor to mesh_body each step
    lidar.SetOffsetPose(chrono.ChFrameD(chrono.ChVectorD(x, y, z),
                                       chrono.Q_from_AngAxis(angle + math.pi/2, chrono.ChVectorD(0, 0, 1))))

    # Update sensors and filters (needed so lidar simulates and updates buffers)
    sensor_manager.Update()

    # Print lidar buffer data once per lidar update rate (print first 10 points from the last scan)
    if time >= next_print_time:
        next_print_time += print_interval
        # Access lidar scan buffer (range and points)
        points = lidar.GetPointCloudBuffer()
        if points is not None:
            print(f"Time {time:.3f} s: Lidar points count = {len(points)}")
            max_print = min(10, len(points))
            for i in range(max_print):
                p = points[i]
                print(f"  Point[{i}] = ({p.x:.3f}, {p.y:.3f}, {p.z:.3f}) depth={p.z}")
        else:
            print(f"Time {time:.3f} s: No lidar points available yet.")

print("Simulation complete.")
```
