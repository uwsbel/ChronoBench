```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr


def main():
    # 1. Initialize the PyChrono environment and core components
    chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Ensure data path is set
    time_step = 0.02  # 50 FPS -> 1/50 = 0.02 sec

    # Contact method for simulation
    contact_method = chrono.ChMaterialSurfaceSMC()

    # 2. Add the required physical systems and objects

    # --------------------- VEHICLE SETUP ---------------------
    # Vehicle initial location and orientation
    initLoc = chrono.ChVectorD(0, 0, 0.5)  # 0.5 m above ground
    initRot = chrono.ChQuaternionD(1, 0, 0, 0)  # No rotation (identity quaternion)

    # Create the ARTcar vehicle
    vehicle = veh.ARTiculationVehicle(contact_method)
    vehicle.SetInitPosition(chrono.ChCoordsysD(initLoc, initRot))
    vehicle.SetChassisFixed(False)
    vehicle.SetInitFwdVel(0)

    # Choose between visualization types (MESH, PRIMITIVES, NONE)
    vis_type = veh.VisualizationType_MESH
    vehicle.SetVisualizationType(vis_type)

    vehicle.Initialize()

    # --------------------- TERRAIN SETUP ---------------------
    # Create rigid terrain of specific dimensions
    terrainLength = 150.0
    terrainWidth = 150.0
    terrainHeight = 0.4

    terrain = veh.RigidTerrain(vehicle.GetSystem())
    patch_material = chrono.ChMaterialSurfaceSMC()
    patch = terrain.AddPatch(patch_material, chrono.ChVector2D(terrainLength, terrainWidth))
    patch.SetTexture(veh.GetDataFile("terrain/textures/grass.jpg"), 200, 200)

    terrain.Initialize()

    # --------------------- DRIVER SETUP ---------------------
    # Use InteractiveDriver to control steering, throttle, braking from keyboard
    driver = veh.InteractiveDriver(vehicle)
    driver.SetInputTimeDelay(0.1)  # small delay to smooth input
    driver.SetSteeringDelta(0.02)
    driver.SetThrottleDelta(0.01)
    driver.SetBrakingDelta(0.1)
    driver.Initialize()

    # 3. Set necessary default parameters (already set during initialization above)

    # --------------------- IRRLICHT VISUALIZATION ---------------------
    app = chronoirr.ChIrrApp(vehicle.GetSystem(), "ARTcar Simulation", chronoirr.dimension2du(1280, 720))
    app.AddTypicalLights()
    app.AddTypicalSky()
    app.AddTypicalCamera(chronoirr.vector3df(10, -12, 6))
    app.SetTimestep(time_step)
    app.AssetBindAll()
    app.AssetUpdateAll()

    # 4. Simulation loop running at 50 FPS
    while app.GetDevice().run():

        app.BeginScene()
        app.DrawAll()

        time = vehicle.GetSystem().GetChTime()

        # Get driver inputs
        driverInputs = driver.Advance(time_step)

        # Apply inputs to vehicle
        vehicle.GetSteeringController().SetInput(driverInputs.m_steering)
        vehicle.GetActuatorThrottle().SetInput(driverInputs.m_throttle)
        vehicle.GetActuatorBrake().SetInput(driverInputs.m_braking)

        # Advance simulation
        driver.Synchronize(time)
        terrain.Synchronize(time)
        vehicle.Synchronize(time, driverInputs, terrain)
        vehicle.Advance(time_step)
        terrain.Advance(time_step)
        driver.Advance(time_step)

        app.EndScene()

        # Optional: Print simulation time for debugging
        # print(f"Time: {time:.2f}s")

if __name__ == "__main__":
    main()
```
