```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as robot
from pychrono import irrlicht as chronoirr
from pychrono import sensor as sens  # 1. Added sensor module import

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create ground body with contact material and add it to the system
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVector3d(0, 0, -0.5))  # Position the ground slightly below the origin
ground.SetFixed(True)  # Fix the ground in place
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# create a long box for rover to cross
box = chrono.ChBodyEasyBox(0.25, 5, 0.25, 1000, True, True, ground_mat)
box.SetPos(chrono.ChVector3d(0, 0, 0.0))
box.SetFixed(True)
box.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
system.Add(box)

# Create Curiosity rover and add it to the system
rover = robot.Curiosity(system)

# Create driver for rover
driver = robot.CuriosityDCMotorControl()
rover.SetDriver(driver)

# Initialize rover position and orientation
init_pos = chrono.ChVector3d(-5, 0.0, 0)
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)
rover.Initialize(chrono.ChFrame(init_pos, init_rot))  # Corrected ChFramed -> ChFrame

# 2. Create sensor manager
manager = sens.ChSensorManager(system)

# 3. Add lidar sensor to the rover
# Define lidar parameters
lidar_update_rate = 10  # Hz
horizontal_samples = 720
vertical_samples = 1
horizontal_fov = math.radians(360)  # 360 degrees in radians
vertical_fov = math.radians(1)      # 1 degree in radians
max_distance = 10.0  # meters

# Create lidar sensor
lidar = sens.ChLidarSensor(
    rover.GetChassisBody(),  # attach to rover chassis
    lidar_update_rate,
    horizontal_samples,
    vertical_samples,
    horizontal_fov,
    vertical_fov,
    max_distance,
)

# Position lidar sensor relative to chassis
# For example, place it at the front top of the chassis
lidar_pose = chrono.ChFrameD(chrono.ChVectorD(0.5, 0, 0.5), chrono.ChQuaternionD(1, 0, 0, 0))
lidar.SetFrame(lidar_pose)

# Set up filters for lidar data
# Example: add a point cloud filter to save data to a file (optional)
# Here, just add a visualization filter to see lidar points in Irrlicht
pc_filter = sens.ChFilterVisualizePointCloud()
lidar.PushFilter(pc_filter)

# Add lidar sensor to sensor manager
manager.AddSensor(lidar)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Curiosity rover - Rigid terrain with Lidar')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 3), chrono.ChVector3d(0, 0, 0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0), 3, 4, 10, 40, 512)

# Enable shadows (commented out to improve performance)
# vis.EnableShadows()

# Set the simulation time step
time_step = 1e-3

# Simulation loop
time = 0
while vis.Run():
    time += time_step

    # ask rover to move forward
    driver.SetSteering(0.0)

    # Update rover dynamics
    rover.Update()

    # 4. Update sensor manager
    manager.Update()

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance simulation by one time step
    system.DoStepDynamics(time_step)
```
---

### Explanation of corrections and modifications:

  - Added the missing import for `pychrono.sensor` as `sens`.
  - Used `chrono.ChFrameD` and `chrono.ChVectorD` for the lidar pose (double precision) consistent with PyChrono conventions.
  - Confirmed that `rover.GetChassisBody()` is the correct method to get the chassis body for sensor attachment.
  - Added a visualization filter (`ChFilterVisualizePointCloud`) to the lidar sensor so that lidar points are visible in the Irrlicht window.
  - Added `manager.Update()` inside the simulation loop to update sensors every step.

- **Modifications per instructions:**
  - Created a sensor manager attached to the system.
  - Created and configured a lidar sensor with specified parameters.
  - Attached the lidar sensor to the rover chassis with a pose offset.
  - Added the lidar sensor to the sensor manager.
  - Updated the sensor manager in the simulation loop.
