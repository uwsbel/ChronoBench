```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Ensure data path is set for textures, meshes, etc.

# Create the physical system
system = chrono.ChSystemNSC()

# Set gravity
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# 2. Add the required physical systems and objects

# --- Floor (truss) ---
floor = chrono.ChBody()
floor.SetBodyFixed(True)
floor.SetPos(chrono.ChVectorD(0, 0, 0))
floor_shape = chrono.ChBoxShape()
floor_shape.GetBoxGeometry().Size = chrono.ChVectorD(2.0, 0.1, 0.5)
floor.AddAsset(floor_shape)
floor_texture = chrono.ChTexture()
floor_texture.SetTextureFilename(chrono.GetChronoDataPath() + "textures/concrete.jpg")
floor.AddAsset(floor_texture)
system.Add(floor)

# --- Crankshaft ---
crank = chrono.ChBody()
crank.SetPos(chrono.ChVectorD(0, 0.15, 0))
crank.SetMass(2.0)
crank.SetInertiaXX(chrono.ChVectorD(0.01, 0.01, 0.01))
crank_shape = chrono.ChBoxShape()
crank_shape.GetBoxGeometry().Size = chrono.ChVectorD(0.2, 0.05, 0.05)
crank.AddAsset(crank_shape)
crank_texture = chrono.ChTexture()
crank_texture.SetTextureFilename(chrono.GetChronoDataPath() + "textures/metal.jpg")
crank.AddAsset(crank_texture)
system.Add(crank)

# --- Connecting rod ---
rod = chrono.ChBody()
rod.SetPos(chrono.ChVectorD(0.4, 0.15, 0))
rod.SetMass(1.0)
rod.SetInertiaXX(chrono.ChVectorD(0.005, 0.005, 0.005))
rod_shape = chrono.ChBoxShape()
rod_shape.GetBoxGeometry().Size = chrono.ChVectorD(0.4, 0.03, 0.03)
rod.AddAsset(rod_shape)
rod_texture = chrono.ChTexture()
rod_texture.SetTextureFilename(chrono.GetChronoDataPath() + "textures/metal.jpg")
rod.AddAsset(rod_texture)
system.Add(rod)

# --- Piston ---
piston = chrono.ChBody()
piston.SetPos(chrono.ChVectorD(0.8, 0.15, 0))
piston.SetMass(1.5)
piston.SetInertiaXX(chrono.ChVectorD(0.01, 0.01, 0.01))
piston_shape = chrono.ChBoxShape()
piston_shape.GetBoxGeometry().Size = chrono.ChVectorD(0.1, 0.1, 0.1)
piston.AddAsset(piston_shape)
piston_texture = chrono.ChTexture()
piston_texture.SetTextureFilename(chrono.GetChronoDataPath() + "textures/metal.jpg")
piston.AddAsset(piston_texture)
system.Add(piston)

# 3. Set necessary default parameters such as positions, forces, and interactions

# --- Joints ---

# Fix crankshaft to floor with revolute joint (rotation about Z axis)
joint_crank_floor = chrono.ChLinkLockRevolute()
joint_crank_floor.Initialize(crank, floor, chrono.ChCoordsysD(crank.GetPos(), chrono.Q_from_AngAxis(chrono.CH_C_PI_2, chrono.VECT_X)))
system.Add(joint_crank_floor)

# Revolute joint between crank and connecting rod at crank end
joint_crank_rod = chrono.ChLinkLockRevolute()
joint_crank_rod_pos = crank.GetPos() + chrono.ChVectorD(0.2, 0, 0)
joint_crank_rod.Initialize(rod, crank, chrono.ChCoordsysD(joint_crank_rod_pos, chrono.Q_from_AngAxis(chrono.CH_C_PI_2, chrono.VECT_X)))
system.Add(joint_crank_rod)

# Revolute joint between connecting rod and piston
joint_rod_piston = chrono.ChLinkLockRevolute()
joint_rod_piston_pos = rod.GetPos() + chrono.ChVectorD(0.4, 0, 0)
joint_rod_piston.Initialize(piston, rod, chrono.ChCoordsysD(joint_rod_piston_pos, chrono.Q_from_AngAxis(chrono.CH_C_PI_2, chrono.VECT_X)))
system.Add(joint_rod_piston)

# Prismatic joint between piston and floor (piston moves vertically along Y axis)
joint_piston_floor = chrono.ChLinkLockPrismatic()
joint_piston_floor.Initialize(piston, floor, chrono.ChCoordsysD(piston.GetPos(), chrono.Q_from_AngAxis(chrono.CH_C_PI_2, chrono.VECT_X)))
system.Add(joint_piston_floor)

# --- Motor to drive crankshaft at constant angular speed ---
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(crank, floor, chrono.ChFrameD(joint_crank_floor.GetMarker1().GetAbsCoord()))
system.Add(motor)

# Define constant angular speed function (e.g., 2 rad/s)
class ConstantSpeed(chrono.ChFunction):
    def __init__(self, omega):
        super().__init__()
        self.omega = omega

    def Get_y(self, t):
        return self.omega * t

motor.SetSpeedFunction(ConstantSpeed(2.0))  # 2 rad/s constant speed

# --- Visualization setup ---

# Create the Irrlicht application
application = chronoirr.ChIrrApp(system, "Crank-Slider Mechanism", chronoirr.dimension2du(1024, 768))

# Add typical elements
application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(chronoirr.vector3df(1.5, 0.8, 1.5))

# Customize camera: position and target
cam = application.GetCameraManager().GetActiveCamera()
cam.setPosition(chronoirr.vector3df(1.5, 0.8, 1.5))
cam.setTarget(chronoirr.vector3df(0.4, 0.15, 0))

# Add floor texture visualization
floor_asset = chrono.CastToChBoxShape(floor.GetAssets()[0])
if floor_asset:
    # Already added texture above, so no extra needed here
    pass

# Add a logo or text in the scene
logo = chronoirr.IrrlichtDevice_GetGUIEnvironment(application.GetDevice()).addStaticText(
    "PyChrono Crank-Slider Demo", 
    chronoirr.recti(10, 10, 300, 30), 
    True)
logo.setBackgroundColor(chronoirr.SColor(100, 60, 60, 60))
logo.setTextAlignment(chronoirr.EGUI_ALIGNMENT.EGUIA_UPPERLEFT)

# Finalize assets
application.AssetBindAll()
application.AssetUpdateAll()

# Simulation loop
application.SetTimestep(0.01)

while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```
