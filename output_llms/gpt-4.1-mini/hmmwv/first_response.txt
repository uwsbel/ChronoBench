```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Ensure data path is set

# Simulation parameters
time_step = 1.0 / 1000  # simulation step size
render_step_size = 1.0 / 50  # render at 50 FPS

# 2. Add the required physical systems and objects as specified

# Create the vehicle system
contact_method = chrono.ChContactMethod_NSC  # Non-smooth contact (default)
# For TMEASY tire model, NSC contact is required
vehicle = veh.HMMWV_Full(contact_method)

# Initialize vehicle at specified location and orientation
init_loc = chrono.ChVectorD(0, 0, 0.5)  # 0.5 m above ground to avoid initial penetration
init_rot = chrono.ChQuaternionD(1, 0, 0, 0)  # no rotation (identity quaternion)
vehicle.SetInitPosition(chrono.ChCoordsysD(init_loc, init_rot))

# Set tire model to TMEASY
vehicle.SetTireType(veh.TireModelType_TMEASY)

# Use primitive visualization for vehicle components
vehicle.SetChassisVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetWheelVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetTireVisualizationType(veh.VisualizationType_PRIMITIVES)

# Initialize the vehicle system
vehicle.Initialize()

# Create the terrain (RigidTerrain)
terrain = veh.RigidTerrain(vehicle.GetSystem())

# Define terrain patch dimensions and location
patch_length = 100.0
patch_width = 100.0
patch_height = 0.1  # thickness of the patch

patch = terrain.AddPatch(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.ChQuaternionD(1, 0, 0, 0)),
                         patch_length, patch_width, patch_height)
patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)
patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
patch.SetFriction(0.9f)
patch.SetMaterialSurface(veh.MaterialSurfaceType_TMEASY)

terrain.Initialize()

# 3. Set necessary default parameters such as positions, forces, and interactions.

# Create the interactive driver system
driver = veh.ChIrrGuiDriver(vehicle)

# Set steering, throttle, and braking time response (for smooth control)
driver.SetSteeringDelta(0.04)
driver.SetThrottleDelta(0.1)
driver.SetBrakingDelta(0.1)

# Initialize the driver system
driver.Initialize()

# Create the Irrlicht application for visualization
app = chronoirr.ChIrrApp(vehicle.GetSystem(), "HMMWV Full Vehicle Simulation", chronoirr.dimension2du(1280, 720))

# Add typical elements to Irrlicht scene
app.AddTypicalSky()
app.AddTypicalLights()
app.AddTypicalCamera(chronoirr.vector3df(0, -8, 3))

# Bind all assets and update all assets
app.AssetBindAll()
app.AssetUpdateAll()

# Set timestep for the Irrlicht application
app.SetTimestep(time_step)

# Simulation loop variables
time = 0
render_frame = 0
render_steps = int(1.0 / (time_step * 50))  # number of simulation steps per render frame

# Simulation loop
while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()

    # Driver inputs
    steering_input = driver.GetSteering()
    throttle_input = driver.GetThrottle()
    braking_input = driver.GetBraking()

    # Apply driver inputs to vehicle
    vehicle.GetSteering().SetInput(steering_input)
    vehicle.GetPowertrain().SetThrottle(throttle_input)
    vehicle.GetBraking().SetBrake(braking_input)

    # Advance simulation by time_step
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, steering_input, throttle_input, braking_input)
    app.Synchronize("", steering_input, throttle_input, braking_input)

    vehicle.Advance(time_step)
    terrain.Advance(time_step)
    driver.Advance(time_step)
    app.Advance(time_step)

    app.EndScene()

    time += time_step
    render_frame += 1

    # Sleep to maintain real-time at 50 FPS rendering
    if render_frame >= render_steps:
        chrono.ChRealtimeStepTimer().Sleep(render_step_size)
        render_frame = 0
```
