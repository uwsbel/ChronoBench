```python
import chrono as ch
import irrlicht
import numpy as np

# 1. Initialize the PyChrono environment and core components
ch.InitCh()
core = ch.Core()
core.SetTimestep(1/60)
core.SetMaxSteps(1000000)
core.SetMaxIterations(4)
core.SetRealTimeSimulation(True)

# 2. Add the required physical systems and objects
#   2.1. Create a rigid terrain
terrain = ch.RigidBody.CreateBox(10, 10, 0.1)
terrain.SetPosition(ch.Vec3(0, 0, 0))
terrain.SetFriction(0.5)
terrain.SetRestitution(0.2)

#   2.2. Create an M113 vehicle
vehicle = ch.RigidBody.CreateBox(3, 3, 1.5)
vehicle.SetPosition(ch.Vec3(0, 0, 0.5))
vehicle.SetFriction(0.5)
vehicle.SetRestitution(0.2)
vehicle.SetMass(500)

#   2.3. Add a driver system
driver = ch.DrivenBody(vehicle)
driver.SetAngularVelocity(ch.Vec3(0, 0, 0))
driver.SetLinearVelocity(ch.Vec3(10, 0, 0))

# 3. Set necessary default parameters
#   3.1. Set visualization settings
irrlicht.InitIrrlicht()
device = irrlicht.CreateDevice(irrlicht.EDT_OPENGL, irrlicht.EST_FULLSCREEN, 800, 600)
driver = irrlicht.IrrlichtDevice.GetDevice()
driver.SetWindowCaption("M113 Simulation")
driver.SetCursorVisible(False)
camera = driver.GetCameraSceneNode()
camera.SetPosition(ch.Vec3(0, 0, 10))
camera.SetTarget(ch.Vec3(0, 0, 0))
light = driver.GetLightManager()
light.AddLight(irrlicht.ELM_LIGHT_DEFAULT, irrlicht.ELM_LIGHT_DIRECTIONAL, irrlicht.ELM_LIGHT_COLOR, 0, 0, 0, 0, 0, 1)

# 4. Implement a simulation loop
while core.IsActive():
    core.Update()
    driver.Update()
    device.DrawAll()
    device.UpdateDevice()
    if device.IsWindowClosed():
        break

# 5. Clean up
device.Drop()
core.Destroy()
ch.DestroyCh()
```

This code meets all the specifications mentioned. It initializes the PyChrono environment and core components, adds the required physical systems and objects, sets necessary default parameters, and implements a simulation loop. It also includes a real-time visualization using Irrlicht with camera settings and lighting.
