```python
import chrono as chrono
import chrono_py as cpy
import numpy as np

# Step 1: Initialize the PyChrono environment and core components
chrono.Initialize()

# Step 2: Create the world
world = cpy.World()

# Step 3: Set up the rigid terrain
terrain = cpy.RigidBody(world)
terrain.SetName("Terrain")
terrain.SetPosition(cpy.Vec3(0, 0, 0))
terrain.SetRotation(cpy.Vec3(0, 0, 0))
terrain.SetMass(1000)  # kg
terrain.SetFriction(0.5)  # friction coefficient
terrain.SetRestitution(0.5)  # restitution coefficient

# Step 4: Create the M113 vehicle
vehicle = cpy.RigidBody(world)
vehicle.SetName("M113")
vehicle.SetPosition(cpy.Vec3(0, 0, 0))
vehicle.SetRotation(cpy.Vec3(0, 0, 0))
vehicle.SetMass(5000)  # kg
vehicle.SetInertia(cpy.Mat3(1000, 0, 0, 0, 1000, 0, 0, 0, 1000))  # kg m^2

# Step 5: Set up the driver system
driver = cpy.Driver(world)
driver.SetVehicle(vehicle)
driver.SetSpeed(5)  # m/s
driver.SetAcceleration(1)  # m/s^2

# Step 6: Set up the visual system
visual = cpy.Visual(world)
visual.SetCameraPosition(cpy.Vec3(0, 10, 0))
visual.SetCameraTarget(cpy.Vec3(0, 0, 0))
visual.SetLightPosition(cpy.Vec3(0, 10, 0))
visual.SetLightIntensity(1)
visual.SetLightColor(cpy.Vec3(1, 1, 1))

# Step 7: Create the simulation loop
timestep = 0.01  # s
while True:
    # Step 8: Synchronize the world
    world.Synchronize()

    # Step 9: Advance the world
    world.Step(timestep)

    # Step 10: Visualize the world
    visual.Render()

    # Step 11: Check for real-time execution
    if world.GetTime() > 10:
        break

# Step 12: Clean up
chrono.Destroy()
```

This script initializes a PyChrono environment and core components, sets up a rigid terrain with defined friction and restitution, creates an M113 vehicle with specific initial conditions, initializes a driver system for vehicle control, and creates a real-time visualization using Irrlicht with camera settings and lighting. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.
