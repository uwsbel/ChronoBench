```python
import chrono as chrono
from chrono import ChSystem, ChBody, ChMesh, ChVisualization, ChVisualizationType, ChCamera
import irrlicht
from irrlicht import video, scene, core

# Initialize PyChrono environment and core components
system = ChSystem()
system.Set_Grav(0, 0, -9.81)

# Add rigid terrain
terrain = ChBody()
terrain.Set_Mass(0)
terrain.Set_AngularMass(0)
terrain.Set_Geometry(ChMesh())
terrain.Set_Vis(ChVisualization(ChVisualizationType.VIS_MESH))
terrain.Set_Texture(video.ESF_TEXTURE_2D("terrain.png"))
terrain.Set_Name("Terrain")
system.Add(terrain)
terrain.Set_Position(0, 0, 0)

# Add CityBus vehicle
vehicle = ChBody()
vehicle.Set_Mass(1000)
vehicle.Set_AngularMass(1000)
vehicle.Set_Geometry(ChMesh())
vehicle.Set_Vis(ChVisualization(ChVisualizationType.VIS_MESH))
vehicle.Set_Name("CityBus")
system.Add(vehicle)
vehicle.Set_Position(0, 0, -5)
vehicle.Set_Orientation(0, 0, 0)
vehicle.Set_TireModel("tire_model.txt")

# Add driver system
driver = ChBody()
driver.Set_Mass(50)
driver.Set_AngularMass(50)
driver.Set_Geometry(ChMesh())
driver.Set_Vis(ChVisualization(ChVisualizationType.VIS_MESH))
driver.Set_Name("Driver")
system.Add(driver)
driver.Set_Position(0, 0, 0)
driver.Set_Orientation(0, 0, 0)

# Add steering, throttle, and braking controls
steering = ChForce()
steering.Set_Type(ChForce.FORCE_STEERING)
steering.Set_Name("Steering")
vehicle.Add_Force(steering)
driver.Add_Force(steering)

throttle = ChForce()
throttle.Set_Type(ChForce.FORCE_THROTTLE)
throttle.Set_Name("Throttle")
vehicle.Add_Force(throttle)
driver.Add_Force(throttle)

braking = ChForce()
braking.Set_Type(ChForce.FORCE_BRAKING)
braking.Set_Name("Braking")
vehicle.Add_Force(braking)
driver.Add_Force(braking)

# Set camera to follow vehicle
camera = ChCamera()
camera.Set_Name("Camera")
camera.Set_Position(0, 0, 5)
camera.Set_Target(vehicle.Get_Position())
camera.Set_Orientation(0, 0, 0)
system.Add(camera)

# Set simulation loop
system.Set_TimeStep(0.02)
system.Set_MaxSteps(50)
system.Set_Rendering(True)

# Run simulation loop
while True:
    system.DoStep()
    system.Render()
    irrlicht.getDevice().run()
```
This script initializes the PyChrono environment and core components, adds the rigid terrain and CityBus vehicle, sets necessary default parameters, and implements an interactive driver system. The simulation loop runs at 50 frames per second, updating the vehicle dynamics and rendering the scene in real time. The camera follows the vehicle from a specified position, allowing for real-time visualization of the simulation.
