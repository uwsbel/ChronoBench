Here's the code that meets the given specifications:

```python
import chrono as chrono
import irrlight as irrlight

# Initialize the PyChrono environment and core components
chrono.Initialize()

# Create a world
world = chrono.ChWorld()

# Create a terrain
terrain = chrono.ChTerrain()
terrain.SetWidth(100)
terrain.SetHeight(10)
terrain.SetDepth(100)
terrain.SetName("Terrain")
terrain.SetTexture("terrain.png")
terrain.SetSkybox("skybox.png")
terrain.SetLogo("logo.png")
terrain.SetChaseCamera(irrlight.Camera())
terrain.SetDirectionalLight(irrlight.DirectionalLight())
world.Add(terrain)

# Create a vehicle
vehicle = chrono.ChVehicle()
vehicle.SetName("MAN 10t")
vehicle.SetVisualization(chrono.ChVehicleVisualization())
vehicle.SetCollision(chrono.ChVehicleCollision())
vehicle.SetTireModel(chrono.ChTireModelTMEasy())
vehicle.SetDriverControl(chrono.ChDriverControl())
world.Add(vehicle)

# Set default parameters
vehicle.SetPosition(chrono.ChVector(0, 0, 0))
vehicle.SetOrientation(chrono.ChQuaternion(0, 0, 0, 1))
vehicle.SetLinearVelocity(chrono.ChVector(0, 0, 0))
vehicle.SetAngularVelocity(chrono.ChVector(0, 0, 0))
vehicle.SetMass(10000)
vehicle.SetInertia(chrono.ChMatrix33(0, 0, 0, 0, 0, 0, 0, 0, 0))
vehicle.SetMaxSpeed(50)
vehicle.SetMaxTorque(1000)
vehicle.SetMaxBrakeForce(1000)

# Set up the visualization
irrlight.SetWindow("PyChrono Simulation")
irrlight.SetWindowSize(800, 600)
irrlight.SetCamera(irrlight.Camera())
irrlight.SetDirectionalLight(irrlight.DirectionalLight())
irrlight.SetSkybox("skybox.png")
irrlight.SetTerrainTexture("terrain.png")
irrlight.SetLogo("logo.png")

# Run the simulation
while True:
    world.DoStep()
    irrlight.Render()
    if irrlight.IsKeyDown(irrlight.KEY_ESCAPE):
        break

# Clean up
chrono.Finalize()
```

This code initializes the PyChrono environment, creates a world, terrain, and vehicle, sets default parameters, and sets up the visualization. It then runs the simulation in an infinite loop until the escape key is pressed. Finally, it cleans up the environment and exits.
