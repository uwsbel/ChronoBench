```python
import chrono as ch
import numpy as np

# Initialize the PyChrono environment
ch.InitChronics()

# Create a new simulation
sim = ch.Simulation()

# Create a new body from a Wavefront .obj file
body = ch.BodyFromObj("path_to_your_file.obj")

# Set the body as fixed
body.SetBodyFixed(True)

# Create a new sensor manager
sensor_manager = ch.SensorManager()

# Add a lidar sensor to the body
lidar = ch.SensorLidar()
lidar.SetSensorManager(sensor_manager)
lidar.SetBody(body)
lidar.SetPosition(ch.Vec3(0, 0, 0))  # Initial position
lidar.SetDirection(ch.Vec3(0, 0, 1))  # Initial direction

# Set the lidar's noise filter
lidar.SetNoiseFilter(ch.Vec3(0.1, 0.1, 0.1))  # Set the noise filter values

# Set the lidar's visualization options
lidar.SetVisualization(True)
lidar.SetVisualizationOptions(ch.Vec3(0.1, 0.1, 0.1))  # Set the visualization options

# Set the lidar's saving options
lidar.SetSave(True)
lidar.SetSaveOptions(ch.Vec3(0.1, 0.1, 0.1))  # Set the saving options

# Create a new motion for the lidar
motion = ch.MotionLinear()
motion.SetPosition(ch.Vec3(0, 0, 0))  # Initial position
motion.SetVelocity(ch.Vec3(0, 0, 1))  # Initial velocity
motion.SetAcceleration(ch.Vec3(0, 0, 0))  # Initial acceleration

# Add the motion to the lidar
lidar.AddMotion(motion)

# Create a new integrator
integrator = ch.IntegratorRungeKutta()

# Add the integrator to the simulation
sim.AddIntegrator(integrator)

# Add the lidar to the simulation
sim.AddSensor(lidar)

# Set the simulation time
sim.SetTime(0)

# Set the simulation step
sim.SetStep(0.01)

# Set the simulation iterations
sim.SetIterations(1000)

# Simulate the system
while sim.Step() == 0:
    # Print the lidar buffer data
    print(lidar.GetBufferData())

# Finalize the PyChrono environment
ch.FinalizeChronics()
```

This script creates a PyChrono simulation with a triangular mesh from a Wavefront .obj file as a fixed body. It adds a lidar sensor to the body, applies noise filters, visualizations, and saving options to the lidar data, and simulates the system while dynamically updating the lidar's position in an orbit around the mesh. The lidar buffer data is printed at each simulation step.
