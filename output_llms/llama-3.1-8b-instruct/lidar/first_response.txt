Here is the Python code that meets the given instructions:

```python
import chrono as chrono
from chrono import ChColor, ChLoadable
from chrono import ChLoadableMesh, ChLoadableMeshWavefront
import numpy as np

# Initialize the PyChrono environment
chrono.Initialize()

# Create a reference to the scene
m_world = chrono.ChWorld()

# Load the triangular mesh from the Wavefront .obj file
m_mesh = ChLoadableMeshWavefront()
m_mesh.Load('path_to_your_mesh.obj')
m_mesh.SetName('Mesh')

# Create a fixed body from the mesh and add it to the scene
m_body = m_mesh.CreateBody()
m_body.SetMass(0)  # Set mass to zero to make it a fixed body
m_world.Add(m_body)

# Create a lidar sensor
m_lidar = chrono.ChSensor()
m_lidar.SetName('Lidar')
m_lidar.SetPosition([1, 0, 0])  # Initial position of the lidar

# Create a sensor manager
m_sensor_manager = m_world.AddSensorManager()
m_sensor_manager.AddSensor(m_lidar)

# Apply noise filters to the lidar data
m_lidar.SetNoiseFilter(0.1)  # 10% noise filter

# Set visualization options for the lidar data
m_lidar.SetVisualization(chrono.ChColor(1, 0, 0))  # Red color
m_lidar.SetBufferSize(1000)  # Set buffer size to 1000 points

# Set saving options for the lidar data
m_lidar.SetSaveData(True)  # Save lidar data to file
m_lidar.SetSaveFileName('lidar_data.txt')  # Set file name

# Define the lidar's orbit around the mesh
def update_lidar_position(step):
    angle = step * 0.1  # Increase angle by 0.1 rad/s
    m_lidar.SetPosition([1 * np.cos(angle), 0, 1 * np.sin(angle)])  # Update lidar position

# Simulate the system
m_world.SetGravity(0, 0, 0)  # Disable gravity
m_world.SetTimestep(0.01)  # Set time step
m_world.SetMaxStep(1.0)  # Set maximum step

# Print lidar buffer data at each simulation step
while m_world.GetStep() < 100:
    update_lidar_position(m_world.GetStep())
    m_world.DoStep()
    print(m_lidar.GetBuffer())

# Clean up
chrono_Finalize()
```

Note that you need to replace `'path_to_your_mesh.obj'` with the actual path to your Wavefront .obj file. Also, this script assumes that the lidar sensor is initially positioned at `[1, 0, 0]` and orbits around the mesh with a radius of `1`. You can adjust these parameters as needed.
