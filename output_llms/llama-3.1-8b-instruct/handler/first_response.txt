```python
import chrono as ch
import time
import rospy
from std_msgs.msg import Int32

# Initialize PyChrono environment
ch.InitializeChSystem()

# Set up the Chrono system with gravitational acceleration and physical material properties
g = 9.81  # gravitational acceleration (m/s^2)
m_floor = 0  # mass of the floor (kg)
m_box = 10  # mass of the box (kg)
E_floor = 1e7  # Young's modulus of the floor (Pa)
nu_floor = 0.3  # Poisson's ratio of the floor
E_box = 1e7  # Young's modulus of the box (Pa)
nu_box = 0.3  # Poisson's ratio of the box
rho_floor = 7850  # density of the floor (kg/m^3)
rho_box = 7850  # density of the box (kg/m^3)

# Create a fixed floor
floor = ch.ChBodyEasyBox(1, 1, 0.01, m_floor, E_floor, nu_floor, rho_floor)
floor.SetPos(0, 0, 0)
floor.SetRot(ch.Vec3(0, 0, 0))
floor.SetBodyFixed(True)
ch.World.Add(floor)

# Create a movable box
box = ch.ChBodyEasyBox(1, 1, 0.01, m_box, E_box, nu_box, rho_box)
box.SetPos(0, 0, 0.01)
box.SetRot(ch.Vec3(0, 0, 0))
box.SetBodyFixed(False)
ch.World.Add(box)

# Create a custom ROS handler to publish integer messages to a specified topic
class CustomRosHandler:
    def __init__(self):
        self.publisher = rospy.Publisher('custom_topic', Int32, queue_size=10)

    def publish_message(self, message):
        self.publisher.publish(message)

# Configure the ROS manager with clock, body, and transform handlers
rospy.init_node('chrono_ros_node', anonymous=True)
ros_handler = CustomRosHandler()

# Register the custom handler
ch.World.RegisterRosHandler(ros_handler)

# Run a simulation loop, advancing the system and updating ROS communication at each timestep
while not rospy.is_shutdown():
    # Update the simulation
    ch.World.DoStep()

    # Update ROS communication
    ros_handler.publish_message(1)  # publish an integer message to the custom topic

    # Maintain real-time execution
    rate = rospy.Rate(100)  # 100 Hz
    rate.sleep()

# Clean up
ch.DestroyChSystem()
```

This code sets up a PyChrono simulation with a fixed floor and a movable box, and integrates it with ROS by publishing integer messages to a custom topic at each timestep. The simulation loop maintains real-time execution and updates ROS communication at each step.
