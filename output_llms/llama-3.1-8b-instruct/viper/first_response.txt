```python
import chrono as ch
import numpy as np

# Initialize the PyChrono environment
ch.InitChrono()

# Create a rigid terrain
ground_body = ch.BodyNew()
ground_body.SetBodyType(ch.BT_RIGID)
ground_body.SetPos(ch.Vec3(0, 0, 0))
ground_body.SetRot(ch.QfromEuler(0, 0, 0))

# Create a Viper rover
viper_body = ch.BodyNew()
viper_body.SetBodyType(ch.BT_RIGID)
viper_body.SetPos(ch.Vec3(0, 0, 1))
viper_body.SetRot(ch.QfromEuler(0, 0, 0))

# Create a driver for the Viper rover
driver = ch.BodyNew()
driver.SetBodyType(ch.BT_RIGID)
driver.SetPos(ch.Vec3(0, 0, 2))
driver.SetRot(ch.QfromEuler(0, 0, 0))

# Create a link between the Viper rover and the driver
link = ch.LinkNew()
link.SetBodyA(viper_body)
link.SetBodyB(driver)
link.SetPos(ch.Vec3(0, 0, 1))
link.SetRot(ch.QfromEuler(0, 0, 0))

# Set the ground as a collision object
ground_collision = ch.CollisionObjectNew()
ground_collision.SetBody(ground_body)
ground_collision.SetShape(ch.CS_BOX, ch.Vec3(100, 100, 1))

# Set the Viper rover as a collision object
viper_collision = ch.CollisionObjectNew()
viper_collision.SetBody(viper_body)
viper_collision.SetShape(ch.CS_BOX, ch.Vec3(1, 1, 1))

# Set the driver as a collision object
driver_collision = ch.CollisionObjectNew()
driver_collision.SetBody(driver)
driver_collision.SetShape(ch.CS_BOX, ch.Vec3(1, 1, 1))

# Set the gravity
ch.SetGravity(ch.Vec3(0, -9.81, 0))

# Create a real-time visualization using Irrlicht
irrlicht = ch.CreateIrrlichtDevice()
irrlicht.SetWindowTitle("Viper Rover Simulation")
irrlicht.SetWindowDimensions(800, 600)

# Set camera settings
irrlicht.AddCamera(ch.Vec3(0, 0, 5), ch.Vec3(0, 0, 0))

# Set lighting
irrlicht.AddLight(ch.Vec3(0, 0, 0), ch.Vec3(1, 1, 1), 1)

# Simulation loop
start_time = 0
end_time = 10
dt = 0.01
t = 0

while t < end_time:
    # Update the Viper rover's steering
    steering_angle = np.sin(t)
    viper_body.SetRot(ch.QfromEuler(0, steering_angle, 0))

    # Update and render the simulation
    ch.Update()
    irrlicht.Render()

    # Increment time
    t += dt

# Clean up
ch.DestroyIrrlichtDevice()
ch.FinalizeChrono()
```

This script initializes the PyChrono environment, creates a rigid terrain and a Viper rover, sets up the Chrono system with gravity and ground body, initializes the Viper rover and its driver, creates a real-time visualization using Irrlicht with camera settings and lighting, and implements a simulation loop where the rover's steering gradually changes over a specified time period, updating and rendering the rover at each timestep.
