```python
import chrono as ch
import chrono.irrlicht as irr

# Initialize the PyChrono environment and core components
ch.begin()

# Create a world object
world = ch.ChWorld()

# Create a ground object
ground = ch.ChBodyEasyPlane(world, 0, 0, 0, 1, 0, 0)
ground.SetBodyFixed(True)

# Create a mass object
mass = ch.ChBodyEasyBox(world, 0, 0, 0.1, 0.1, 0.1, 0.1)
mass.SetBodyFixed(False)

# Create a spring-damper object
spring = ch.ChLinkTSDA(mass, ground)

# Set default linear spring-damper parameters
spring.SetSpringDamp(0.5, 0.1, 0.0, 0.0)
spring.SetSpringStiff(1000, 0, 0, 0)

# Set the initial position of the mass
mass.SetPos(0, 0, 0.5)

# Set the initial velocity of the mass
mass.SetV(0, 0, 0)

# Create an Irrlicht visualizer
vis = irr.Irrlicht()
vis.AddLight(0, 0, 0, 0, 0, 1, 1)
vis.AddLight(0, 0, 0, 0, 1, 0, 1)
vis.AddCamera(0, 0, 0, 0, 0, 0, 0)

# Add visual shapes for the mass and spring
mass.AddVisual(irr.IrrlichtMesh("mass.obj"))
spring.AddVisual(irr.IrrlichtMesh("spring.obj"))

# Set up the simulation
ch.SetDT(0.01)
ch.SetMaxStep(100)

# Start the simulation
ch.Run()

# Clean up
ch.end()
```

This script initializes the PyChrono environment, creates a world and its components, sets up a spring-damper connection, and configures the Irrlicht visualizer. The simulation is then run, and the visual shapes for the mass and spring are added to the scene.
