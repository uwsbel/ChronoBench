class MySpringForce(chrono.ForceFunctor):

        self.spring_coef = spring_coef
        self.damping_coef = damping_coef

    def ComputeForce(self, body, pos_on_body, pos_on_other, normal_on_other):
        # Calculate spring force
        spring_force = self.spring_coef * (pos_on_other - pos_on_body) * normal_on_other
        # Calculate damping force
        damping_force = -self.damping_coef * chrono.ChVector3d(body.GetLinearVelocity()).Cross(normal_on_other)
        return spring_force + damping_force
