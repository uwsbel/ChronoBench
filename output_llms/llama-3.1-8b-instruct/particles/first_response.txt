```python
import chrono as chrono
import irrlicht
import random

# Step 1: Initialize the PyChrono environment and core components
chrono.Initialize()
solver = chrono.ChronoSolver()
irrlicht.InitAll(irrlicht.IrrlichtDeviceCreateDevice(irrlicht.E_DRIVER_TYPE_VIDEO_HD))

# Step 2: Define the particle emitter
class ParticleEmitter:
    def __init__(self):
        self.particles = []

    def emit(self):
        for _ in range(10):
            particle = chrono.ChronoBodyDynamics()
            particle.SetMass(1)
            particle.SetPosition(chrono.ChronoVector(0, 0, 0))
            particle.SetOrientation(chrono.ChronoQuaternion(0, 0, 0, 1))
            particle.SetVelocity(chrono.ChronoVector(random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1)))
            particle.SetAngularVelocity(chrono.ChronoVector(random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1)))

            # Generate a random shape for the particle
            shape = chrono.ChronoSphere(0.1)
            particle.AddAsset(shape)

            self.particles.append(particle)

# Step 3: Create a particle emitter and emit particles
emitter = ParticleEmitter()
emitter.emit()

# Step 4: Define a custom gravitational force
class CustomGravitationalForce:
    def __init__(self):
        self.gravity = chrono.ChronoVector(0, 0, -9.81)

    def apply(self, body):
        body.AddForce(self.gravity)

# Step 5: Set up the simulation and apply custom gravitational forces
world = chrono.ChronoWorld()
for particle in emitter.particles:
    world.Add(particle)
force = CustomGravitationalForce()
for particle in emitter.particles:
    force.apply(particle)

# Step 6: Set up the visualization
device = irrlicht.IrrlichtDeviceCreateDevice(irrlicht.E_DRIVER_TYPE_VIDEO_HD)
driver = device.getDriver()
scene = device.getSceneManager()
camera = scene.addCameraSceneNodeFPS(0, 0, 10, 20)
camera.setTarget(chrono.ChronoVector(0, 0, 0))
camera.setOrthographic(True)

# Step 7: Run the simulation
while True:
    solver.Solve()
    device.beginScene(irrlicht.SCENE_MANAGER)
    for particle in emitter.particles:
        node = scene.addMeshSceneNode(particle.GetAsset(0))
        node.setPosition(particle.GetPos())
        node.setRotation(particle.GetRot())
    device.endScene()
    driver.updateDevice()
    device.draw()
```

This script initializes the PyChrono environment, creates a particle emitter, and generates particles with random shapes, positions, velocities, and orientations. It then defines a custom gravitational force and applies it to each particle. Finally, it sets up the visualization using Irrlicht and runs the simulation.
