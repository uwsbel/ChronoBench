Here's a Python script that meets the requirements:

```python
import chrono as chrono
import irrlicht

# 1. Initialize the PyChrono environment and core components
chrono.Initialize()

# 2. Create a new physical system
sys = chrono.System()

# 3. Define the beam finite elements
beam1 = chrono.ChBeams()
beam1.SetName('Beam1')
beam1.SetMaterial(chrono.ChMaterialSurface(chrono.EM_KINEMATIC))
beam1.SetLength(1)
beam1.SetRadius(0.1)
beam1.SetMass(0.1)
beam1.SetDensity(1000)
beam1.SetColor(irrlicht.color(255, 0, 0))
sys.Add(beam1)

# 4. Define the nodes for the beam
node1 = chrono.ChBody()
node1.SetName('Node1')
node1.SetPosition(chrono.ChVectorD(0, 0, 0))
node1.SetMass(0)
node1.SetDensity(0)
node1.SetColor(irrlicht.color(0, 255, 0))
sys.Add(node1)

node2 = chrono.ChBody()
node2.SetName('Node2')
node2.SetPosition(chrono.ChVectorD(1, 0, 0))
node2.SetMass(0)
node2.SetDensity(0)
node2.SetColor(irrlicht.color(0, 0, 255))
sys.Add(node2)

# 5. Connect the nodes to the beam
beam1.AddNode(node1)
beam1.AddNode(node2)

# 6. Set the initial conditions
node1.SetPosition(chrono.ChVectorD(0, 0, 0))
node2.SetPosition(chrono.ChVectorD(1, 0, 0))
node1.SetVelocity(chrono.ChVectorD(0, 0, 0))
node2.SetVelocity(chrono.ChVectorD(0, 0, 0))

# 7. Set the simulation parameters
sys.SetTimestep(0.01)
sys.SetMaxStep(0.1)
sys.SetSolverType(chrono.SolverType.HHT)

# 8. Create the visualizer
visualizer = irrlicht.IrrlichtApplication(sys.GetDeviceName(), irrlicht.dimension2D(800, 600), False)
visualizer.AddRootNode(irrlicht.IrrlichtSceneNode.CreateSceneNode(visualizer.GetDevice(), irrlicht.SCT_CAMERA))

# 9. Run the simulation loop
while visualizer.GetDevice().run() and not visualizer.IsClosed():
    sys.DoStep()
    visualizer.BeginScene()
    visualizer.GetScene().DrawAll()
    visualizer.EndScene()

# 10. Clean up
chrono.Deinitialize()
```

This script creates a physical system with a mesh composed of a beam and two nodes. It sets the necessary default parameters such as positions, forces, and interactions. The simulation is run using the Irrlicht engine for real-time rendering.
