```python
import chrono as c
import numpy as np

# Initialize the PyChrono environment
c.init()

# Create a rigid body for the ground
ground_body = c.ChBody()
ground_body.SetBodyFixed(True)
ground_body.SetCollision(True)
ground_body.SetPos(c.Vec3(0, 0, 0))
ground_body.SetRot(c.Mat3(c.PI/2, 0, 0))

# Create a rigid body for the rover
rover_body = c.ChBody()
rover_body.SetPos(c.Vec3(0, 0, 0))
rover_body.SetRot(c.Mat3(0, 0, 0))
rover_body.SetMass(100)
rover_body.SetDensity(1000)
rover_body.SetFriction(0.5)
rover_body.SetRestitution(0.2)

# Create a motor for the rover's wheels
motor = c.ChMotor()
motor.SetPos(0)
motor.SetRot(0)
motor.SetSpeed(100)

# Create a motor control driver for the rover
motor_driver = c.ChMotorDriver()
motor_driver.SetPos(0)
motor_driver.SetRot(0)
motor_driver.SetSpeed(100)

# Add the physical systems and objects to the simulation
c.Space().Add(ground_body)
c.Space().Add(rover_body)
c.Space().Add(motor)
c.Space().Add(motor_driver)

# Set the default parameters
c.Space().SetGravity(c.Vec3(0, -9.81, 0))
c.Space().SetTimeStep(0.01)

# Set the visualization settings
c.IrrlichtVisualizer().SetCameraPos(c.Vec3(0, 10, 10))
c.IrrlichtVisualizer().SetCameraTarget(c.Vec3(0, 0, 0))
c.IrrlichtVisualizer().SetAmbientLight(c.Color(0.5, 0.5, 0.5))
c.IrrlichtVisualizer().SetDiffuseLight(c.Color(1, 1, 1))
c.IrrlichtVisualizer().SetSpecularLight(c.Color(1, 1, 1))
c.IrrlichtVisualizer().SetShadows(True)
c.IrrlichtVisualizer().SetTexture(c.Texture("path_to_texture.png"))
c.IrrlichtVisualizer().SetLogo(c.Logo("path_to_logo.png"))

# Run the simulation
c.SimulationLoop()
```

Note that this code assumes that you have the PyChrono and Irrlicht libraries installed, and that you have the necessary files (e.g. textures, logos) in the correct locations. You will need to replace the `path_to_texture.png` and `path_to_logo.png` placeholders with the actual paths to your files.
