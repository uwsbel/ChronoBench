# Create a new rigid body for the chassis
chassis = sim.AddRigidBody(chrono.ChBodySimple())

terrain_patch = sim.AddRigidBody(chrono.ChBodyPatch())

driver_system.AddOutput(chrono.ChOutputAcceleration())

# Set timestep and rendering settings
sim.SetTimestep(1 / 60)
sim.SetRenderingSettings(chrono.ChRenderingSettings())
sim.SetRenderingSettings().SetRealtime(True)

# Run the simulation
while True:
    sim.DoStep()
    chronoirr.RenderScene(sim)

# Clean up
chrono.Finalize()
