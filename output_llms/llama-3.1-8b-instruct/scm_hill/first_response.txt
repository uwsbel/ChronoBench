```python
# Import necessary modules
import chrono as chrono
import irrlicht
import math

# Initialize PyChrono environment and core components
chrono.Initialize()
core = chrono.Core()
visual = chrono.Visual()
visual.SetDebugMode(visual.DB_NORMAL)

# Create HMMWV vehicle
hmmwv = chrono.CreateHMMWV()
hmmwv.SetPosition(chrono.Vec3(0, 0, 0))
hmmwv.SetMass(2000)
hmmwv.SetFriction(0.5)
hmmwv.SetRollingFriction(0.2)
hmmwv.SetSuspensionStiffness(10000)
hmmwv.SetDamping(50)

# Create SCM deformable terrain
terrain = chrono.CreateSCMTerrain()
terrain.SetPosition(chrono.Vec3(0, 0, 0))
terrain.SetSize(100, 100)
terrain.SetSoilParameters(100, 0.5, 0.2)
terrain.SetHeightMap("path/to/heightmap.txt")

# Create driver system
driver = chrono.CreateDriverSystem()
driver.SetPosition(chrono.Vec3(0, 0, 0))
driver.SetSpeed(0)
driver.SetSteering(0)

# Create Irrlicht visualization
irrlicht.Initialize()
camera = irrlicht.CreateCamera()
camera.SetPosition(chrono.Vec3(0, 0, 10))
camera.SetTarget(chrono.Vec3(0, 0, 0))
camera.SetFOV(math.pi / 3)
irrlicht.SetLighting(chrono.LIGHTING_ENABLED)

# Set up simulation loop
timestep = 0.01
while True:
    # Synchronize systems
    core.Synchronize()
    driver.Synchronize()
    terrain.Synchronize()
    visual.Synchronize()

    # Advance systems
    core.Step(timestep)
    driver.Step(timestep)
    terrain.Step(timestep)
    visual.Step(timestep)

    # Visualize
    visual.Render()
    visual.Update()

    # Check for termination condition
    if core.GetTime() > 10:
        break

# Clean up
chrono.Finalize()
irrlicht.Finalize()
```
