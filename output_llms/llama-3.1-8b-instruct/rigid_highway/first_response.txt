```python
# Import required modules
import chrono as chrono
from chrono import ChVector, ChFrame, ChBody, ChMesh, ChVisualization, ChContactMethod, ChTireModel, ChDriverSystem

# Initialize PyChrono environment and core components
chrono.Initialize()

# Define terrain meshes
terrain_collision_mesh = ChMesh('Highway_col.obj')
terrain_visual_mesh = ChMesh('Highway_vis.obj')

# Create terrain body with collision and visual meshes
terrain_body = ChBody()
terrain_body.AddMesh(terrain_collision_mesh, 1.0)
terrain_body.AddMesh(terrain_visual_mesh, 1.0)
terrain_body.SetName('Terrain')

# Create vehicle body with custom mesh
vehicle_body = ChBody()
vehicle_body.AddMesh('HMMWV.obj', 1.0)
vehicle_body.SetName('HMMWV')

# Set vehicle position, orientation, and contact method
vehicle_body.SetPos(ChVector(0, 0, 0))
vehicle_body.SetOrient(ChFrame())
vehicle_body.SetContactMethod(ChContactMethod.EM_LCP)
vehicle_body.SetName('HMMWV')

# Create vehicle chassis
chassis_body = ChBody()
chassis_body.AddMesh('HMMWV_chassis.obj', 1.0)
chassis_body.SetName('Chassis')
chassis_body.SetPos(ChVector(0, 0, 0))
chassis_body.SetOrient(ChFrame())
chassis_body.SetContactMethod(ChContactMethod.EM_LCP)

# Create vehicle wheels
wheel1_body = ChBody()
wheel1_body.AddMesh('HMMWV_wheel.obj', 1.0)
wheel1_body.SetName('Wheel1')
wheel1_body.SetPos(ChVector(-1.5, -1.5, -0.5))
wheel1_body.SetOrient(ChFrame())

wheel2_body = ChBody()
wheel2_body.AddMesh('HMMWV_wheel.obj', 1.0)
wheel2_body.SetName('Wheel2')
wheel2_body.SetPos(ChVector(1.5, -1.5, -0.5))
wheel2_body.SetOrient(ChFrame())

wheel3_body = ChBody()
wheel3_body.AddMesh('HMMWV_wheel.obj', 1.0)
wheel3_body.SetName('Wheel3')
wheel3_body.SetPos(ChVector(-1.5, 1.5, -0.5))
wheel3_body.SetOrient(ChFrame())

wheel4_body = ChBody()
wheel4_body.AddMesh('HMMWV_wheel.obj', 1.0)
wheel4_body.SetName('Wheel4')
wheel4_body.SetPos(ChVector(1.5, 1.5, -0.5))
wheel4_body.SetOrient(ChFrame())

# Set wheel positions and orientations
wheel1_body.SetPos(ChVector(-1.5, -1.5, -0.5))
wheel1_body.SetOrient(ChFrame())
wheel2_body.SetPos(ChVector(1.5, -1.5, -0.5))
wheel2_body.SetOrient(ChFrame())
wheel3_body.SetPos(ChVector(-1.5, 1.5, -0.5))
wheel3_body.SetOrient(ChFrame())
wheel4_body.SetPos(ChVector(1.5, 1.5, -0.5))
wheel4_body.SetOrient(ChFrame())

# Create TMEASY tire model for each wheel
tire_model1 = ChTireModel()
tire_model1.SetName('Tire1')
wheel1_body.AddTireModel(tire_model1)

tire_model2 = ChTireModel()
tire_model2.SetName('Tire2')
wheel2_body.AddTireModel(tire_model2)

tire_model3 = ChTireModel()
tire_model3.SetName('Tire3')
wheel3_body.AddTireModel(tire_model3)

tire_model4 = ChTireModel()
tire_model4.SetName('Tire4')
wheel4_body.AddTireModel(tire_model4)

# Create driver system
driver_system = ChDriverSystem()
driver_system.SetName('Driver')
driver_system.SetSteering(0.5)
driver_system.SetThrottle(0.5)
driver_system.SetBraking(0.5)

# Add driver system to vehicle body
vehicle_body.AddDriverSystem(driver_system)

# Create simulation loop
chrono.SetRealTimeSimulation(True)
chrono.SetTargetRealTimeFPS(50)

# Run simulation loop
while True:
    chrono.DoStep()
    chrono.Render()
```

This script initializes the PyChrono environment and core components, creates a custom terrain with collision and visual meshes, and a vehicle with specific parameters such as location, orientation, contact method, and TMEASY tire model. It also implements an interactive driver system for steering, throttle, and braking control. The simulation loop runs in real time at 50 frames per second, updating vehicle dynamics and rendering the scene.
