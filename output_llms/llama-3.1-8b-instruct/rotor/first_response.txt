```python
import chrono as ch
import numpy as np

# Initialize the PyChrono environment
ch.InitChSystem()

# Create a world
world = ch.ChWorld()

# Create a beam
beam = ch.ChBeam()
beam.SetName("JeffcottRotor")
beam.SetDensity(1.0)
beam.SetYoungModulus(200e9)
beam.SetPoissonRatio(0.3)
beam.SetLength(1.0)
beam.SetRadius(0.05)
beam.SetNumElements(10)
world.Add(beam)

# Create a flywheel
flywheel = ch.ChBody()
flywheel.SetName("Flywheel")
flywheel.SetDensity(1.0)
flywheel.SetMass(1.0)
flywheel.SetInertia(0.5, 0.5, 0.5)
flywheel.SetPos(ch.Vec(0, 0, 0.5))
flywheel.SetAng(0, 0, 0)
world.Add(flywheel)

# Create a motor
motor = ch.ChMotor()
motor.SetName("Motor")
motor.SetType(ch.ChMotorType.ROTATIONAL)
motor.SetAxis(ch.Vec(1, 0, 0))
motor.SetSpeed(100)
motor.SetMaxForce(100)
motor.SetPos(ch.Vec(0, 0, 0))
motor.SetAng(0, 0, 0)
world.Add(motor)

# Attach the flywheel to the beam
flywheelAttach = ch.ChAttachMode()
flywheelAttach.SetBodyA(beam)
flywheelAttach.SetBodyB(flywheel)
flywheelAttach.SetPos(ch.Vec(0, 0, 0))
flywheelAttach.SetAng(0, 0, 0)
flywheelAttach.SetType(ch.ChAttachMode.FIXED)
world.Add(flywheelAttach)

# Set the beam's FEM visualization
beam.SetFemVisualization(True)
beam.SetFemColor(ch.ChColor(1, 0, 0))

# Create an Irrlicht window
irrlicht = ch.ChIrrAppSimple(world, "Jeffcott Rotor Simulation", ch.dimension2du(800, 600))
irrlicht.AddFrameScene(world)

# Run the simulation
while not irrlicht.KeyCloseDown():
    world.DoStep()
    irrlicht.BeginScene()
    irrlicht.DrawAll()
    irrlicht.EndScene()
    irrlicht.ShowImage()

# Clean up
ch.FinalCleanUp()
```

This script initializes the PyChrono environment, creates a world, a beam, a flywheel, and a motor, and sets up the necessary physical systems and objects. It also creates an Irrlicht window to visualize the simulation and runs the simulation until the user closes the window. The beam's FEM visualization is enabled and set to red.
