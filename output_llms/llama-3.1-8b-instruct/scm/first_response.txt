Here is the PyChrono simulation script based on the provided instructions:

```python
import chrono as ch
import irrlicht as irr
import time

# Initialize the PyChrono environment
ch.initialize()

# Create the simulation core
core = ch.Core()

# Set up the Irrlicht renderer
renderer = irr.IrrlichtDevice.create_device(irr.video.E_DRIVER_TYPE_OPENGL, irr.dimension2di(800, 600), irr.video.E_FULLSCREEN_MODE.FALSE, irr.video.E_DEFAULT_RTT_MODE, False)
renderer.setTTargetFrameRate(50)

# Create the terrain
terrain = ch.SCM_terrain()
terrain.set_width(100)
terrain.set_length(100)
terrain.set_depth(1)
terrain.set_soil_parameters(0.1, 0.01, 0.001, 0.1, 0.01, 0.001, 0.1, 0.01, 0.001)
terrain.set_sinkage_visualization(True)
terrain.set_sinkage_color_map(irr.make_color(1, 0, 0))  # red color for sinkage

# Create the vehicle
vehicle = ch.MovingBody()
vehicle.set_mass(2000)
vehicle.set_inertia_tensor(ch.Mat3x3(1000, 0, 0, 0, 1000, 0, 0, 0, 1000))
vehicle.set_position(ch.Vec3(0, 0, 0))
vehicle.set_orientation(ch.Vec3(0, 0, 0))

# Create the chassis
chassis = ch.RigidBody()
chassis.set_mass(500)
chassis.set_inertia_tensor(ch.Mat3x3(200, 0, 0, 0, 200, 0, 0, 0, 200))
chassis.set_position(ch.Vec3(0, 0, 0))
chassis.set_orientation(ch.Vec3(0, 0, 0))
vehicle.add_child(chassis)

# Create the tire
tire = ch.RigidBody()
tire.set_mass(100)
tire.set_inertia_tensor(ch.Mat3x3(20, 0, 0, 0, 20, 0, 0, 0, 20))
tire.set_position(ch.Vec3(0, 0, 0))
tire.set_orientation(ch.Vec3(0, 0, 0))
vehicle.add_child(tire)

# Create the driver
driver = ch.InteractiveDriver()
driver.set_steering_gain(1)
driver.set_throttle_gain(1)
driver.set_braking_gain(1)

# Add the vehicle to the simulation
core.add_body(vehicle)

# Set up the moving patch feature
moving_patch = ch.MovingPatch()
moving_patch.set_body(vehicle)
moving_patch.set_radius(1)
moving_patch.set_speed(1)
terrain.add_moving_patch(moving_patch)

# Set up the visualization
for body in core.get_bodies():
    body.set_mesh_visualization(True)

# Main simulation loop
while True:
    # Update the simulation
    core.update()

    # Update the driver
    driver.update()

    # Render the scene
    renderer.begin_scene()
    renderer.draw_scene()
    renderer.end_scene()

    # Cap the frame rate
    time.sleep(1 / 50)
```

This script initializes the PyChrono environment, creates the simulation core, sets up the Irrlicht renderer, and creates the terrain and vehicle. It also configures the SCM terrain with custom soil parameters and enables the moving patch feature. The script then sets up the interactive driver system and ensures the simulation runs in real time, updating vehicle dynamics and rendering the scene at 50 frames per second.
