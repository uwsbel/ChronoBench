```python
# Import necessary modules
import chrono as chrono
import chrono.irrlicht as irrl
import numpy as np

# Initialize the PyChrono environment and core components
chrono.Initialize()
irrl.Initialize()

# Create the simulation time
t = chrono.ChronoTime()

# Create the vehicle and its components
vehicle = chrono.Chassis()
vehicle.SetName("HMMWV")
vehicle.SetPosition(chrono.ChVectorD(0, 0, 0))
vehicle.SetRotation(chrono.ChQuaternionD(1, 0, 0, 0))

# Create the terrain
terrain = chrono.ChTerrain()
terrain.SetName("Terrain")
terrain.SetPosition(chrono.ChVectorD(0, 0, 0))
terrain.SetSize(100, 100, 0.1)
terrain.SetFriction(0.5)
terrain.SetRestitution(0.2)

# Create the driver
driver = chrono.ChDriver()
driver.SetName("Driver")
driver.SetPosition(chrono.ChVectorD(0, 0, 0))
driver.SetRotation(chrono.ChQuaternionD(1, 0, 0, 0))

# Create the visual system
visual_system = irrl.IrrlichtVisualizer()
visual_system.SetName("Visual System")
visual_system.SetPosition(chrono.ChVectorD(0, 0, 0))
visual_system.SetRotation(chrono.ChQuaternionD(1, 0, 0, 0))

# Create the IMU and GPS sensors
imu = chrono.ChIMUSensor()
imu.SetName("IMU")
imu.SetPosition(chrono.ChVectorD(0, 0, 0))
imu.SetRotation(chrono.ChQuaternionD(1, 0, 0, 0))
imu.SetSamplingPeriod(0.01)

gps = chrono.ChGPSSensor()
gps.SetName("GPS")
gps.SetPosition(chrono.ChVectorD(0, 0, 0))
gps.SetRotation(chrono.ChQuaternionD(1, 0, 0, 0))
gps.SetSamplingPeriod(0.01)

# Attach the sensors to the vehicle's chassis
vehicle.AddSensor(imu)
vehicle.AddSensor(gps)

# Set the vehicle's mass and moment of inertia
vehicle.SetMass(2000)
vehicle.SetMomentOfInertia(chrono.ChVectorD(500, 500, 500))

# Set the terrain's friction and restitution
terrain.SetFriction(0.5)
terrain.SetRestitution(0.2)

# Set the driver's inputs
driver.SetThrottle(0.5)
driver.SetBrake(0.0)
driver.SetSteering(0.0)

# Set the visual system's camera
visual_system.SetCameraPosition(chrono.ChVectorD(0, 0, 0))
visual_system.SetCameraRotation(chrono.ChQuaternionD(1, 0, 0, 0))

# Set the simulation time step
t.SetTimestep(0.01)

# Set the simulation loop
while t.GetStep() < 1000:
    # Synchronize and advance the vehicle, terrain, driver, and visual system modules
    vehicle.Synchronize()
    terrain.Synchronize()
    driver.Synchronize()
    visual_system.Synchronize()
    
    # Update the sensor data
    imu.Update()
    gps.Update()
    
    # Get the vehicle's mass
    mass = vehicle.GetMass()
    
    # Render the scene
    visual_system.Render()
    
    # Output the vehicle's mass
    print("Vehicle Mass:", mass)
    
    # Advance the simulation time
    t.Advance()

# Clean up
chrono.Finalize()
irrl.Finalize()
```

It then sets up the simulation loop, synchronizes and advances the vehicle, terrain, driver, and visual system modules, updates the sensor data, renders the scene, and outputs the vehicle's mass. Finally, it cleans up by finalizing the PyChrono environment and core components.
