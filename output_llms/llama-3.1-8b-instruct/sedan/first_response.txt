```python
# Import necessary modules
import chrono as chrono
import chrono_irrlicht as chrono_irrlicht
import numpy as np

# Initialize PyChrono environment
chrono.Initialize()

# Create a new simulation
sim = chrono.ChSystem()

# Create a new rigid terrain
terrain = chrono.ChBodyEasyPlane(
    name='terrain',
    dimensions=[100, 100, 0.1],
    material=chrono.ChMaterialSurface(
        name='terrain_material',
        friction=0.5,
        restitution=0.1
    )
)
sim.Add(terrain)

# Create a new vehicle
vehicle = chrono.ChBodyEasyBox(
    name='vehicle',
    dimensions=[4.5, 1.7, 1.4],
    material=chrono.ChMaterialSurface(
        name='vehicle_material',
        friction=0.5,
        restitution=0.1
    )
)
sim.Add(vehicle)

# Create a new tire model
tire = chrono.ChTireModelTMEasy(
    name='tire',
    radius=0.35,
    stiffness=100000,
    damping=100,
    rolling_resistance=1000
)
sim.Add(tire)

# Set the tire model on the vehicle
vehicle.AddAsset(tire)

# Create a new driver system
driver = chrono.ChDriver(
    name='driver',
    steering_gain=0.1,
    throttle_gain=0.1,
    braking_gain=0.1
)
sim.Add(driver)

# Set the driver system on the vehicle
vehicle.AddAsset(driver)

# Set the vehicle's initial position and velocity
vehicle.SetPos(chrono.ChVector(0, 0, 0))
vehicle.SetVel(chrono.ChVector(10, 0, 0))

# Set the terrain's initial position
terrain.SetPos(chrono.ChVector(0, 0, 0))

# Create a new Irrlicht visualizer
visualizer = chrono_irrlicht.ChIrrApp(
    sim,
    'BMW E90 Sedan Simulation',
    np.array([0, 0, 0]),
    np.array([0, 0, 1])
)

# Set up the visualizer
visualizer.SetChaseCamera(True)
visualizer.SetLighting(True)
visualizer.SetSkybox(True)
visualizer.SetTexture('terrain_texture.jpg', 'terrain')
visualizer.SetLogo('BMW_logo.png', 'BMW')
visualizer.SetLogo('E90_logo.png', 'E90')

# Run the simulation
while visualizer.GetDevice().Run():
    sim.DoStep()
    visualizer.BeginScene()
    visualizer.DrawAll()
    visualizer.EndScene()

# Clean up
chrono.Finalize()
```
This code creates a PyChrono simulation of a BMW E90 Sedan driving on a rigid terrain, with a TMEASY tire model and an interactive driver system. The simulation is visualized using the Irrlicht system with a chase camera, directional lighting, a skybox, and customizable textures and logos on the terrain surface.
