```python
# Import necessary modules
import chrono as chrono
import chrono.irrlicht as chronoirr
import chrono.terrain as terr
import chrono.vehicle as veh

# Initialize the PyChrono environment
chrono.Initialize()

# Create the simulation core
sim = chrono.Simulation()

# Create the Irrlicht visualization
irrlicht = chronoirr.Irrlicht(sim)

# Create the rigid terrain
terrain = terr.RigidTerrain(sim, 100, 100, 1, "terrain.png")

# Create the vehicle
gator = veh.Vehicle(sim, "gator")
gator.SetContactMethod(chrono.ContactMethod.PENETRATION_DETECTION)
gator.SetTireModel(chrono.TireModel.TMEASY)

# Set vehicle parameters
gator.SetLocation(chrono.ChVectorD(0, 0, 0))
gator.SetOrientation(chrono.QFromEulerD(0, 0, 0))
gator.SetMass(5000)
gator.SetInertia(chrono.ChMatrix33D([[1000, 0, 0], [0, 1000, 0], [0, 0, 1000]]))
gator.SetType(chrono.VehicleType.VEHICLE)

# Create vehicle components
wheel1 = veh.Wheel(sim, "wheel1")
wheel1.SetLocation(chrono.ChVectorD(-2.5, 0, 0))
wheel1.SetOrientation(chrono.QFromEulerD(0, 0, 0))
wheel1.SetRadius(0.5)
wheel1.SetMass(100)
wheel1.SetInertia(chrono.ChMatrix33D([[100, 0, 0], [0, 100, 0], [0, 0, 100]]))
wheel1.SetType(chrono.WheelType.FOUR_WHEEL_DRIVE)

wheel2 = veh.Wheel(sim, "wheel2")
wheel2.SetLocation(chrono.ChVector(2.5, 0, 0))
wheel2.SetOrientation(chrono.QFromEulerD(0, 0, 0))
wheel2.SetRadius(0.5)
wheel2.SetMass(100)
wheel2.SetInertia(chrono.ChMatrix33D([[100, 0, 0], [0, 100, 0], [0, 0, 100]]))
wheel2.SetType(chrono.WheelType.FOUR_WHEEL_DRIVE)

# Set up interactions between vehicle components
gator.AddWheel(wheel1)
gator.AddWheel(wheel2)

# Set up driver system
driver = veh.Driver(sim, "driver")
driver.SetSteeringAngle(0)
driver.SetThrottle(0)
driver.SetBraking(0)

# Set up simulation loop
sim.SetRealTimeLoop(True)
sim.SetMaxSteps(50)

# Run the simulation
sim.Run()
```
