```python
import chrono as ch
import numpy as np

# Initialize the PyChrono environment
ch.InitChrono()

# Create a ground body
ground_body = ch.ChBody()
ground_body.SetPos(ch.Vec3(0, 0, 0))
ground_body.SetRot(ch.Mat3(ch.PI/2, 0, 0))
ground_body.SetMass(0)
ground_body.SetInertia(ch.Moments(0, 0, 0))
ground_body.SetBodyType(ch.CH_BTYPE_GROUND)
ground_body.SetBodyMaterial(0.0, 0.0, 0.0)
ground_body.SetElasticity(0.0)
ground_body.SetFriction(0.0)
ground_body.SetDensity(0.0)

# Create a pendulum body
pendulum_body = ch.ChBody()
pendulum_body.SetPos(ch.Vec3(0, 0, 1))
pendulum_body.SetRot(ch.Mat3(0, 0, 0))
pendulum_body.SetMass(1.0)
pendulum_body.SetInertia(ch.Moments(1, 0, 0))
pendulum_body.SetBodyType(ch.CH_BTYPE_DYNAMIC)
pendulum_body.SetBodyMaterial(0.0, 0.0, 0.0)
pendulum_body.SetElasticity(0.0)
pendulum_body.SetFriction(0.0)
pendulum_body.SetDensity(0.0)

# Create a revolute joint
joint = ch.ChRevoluteJoint()
joint.SetBody1(ground_body)
joint.SetBody2(pendulum_body)
joint.SetAx(0, 0, 1)
joint.SetLoch(0, 0, 0)
joint.SetLo(0, 0, 0)
joint.SetRot(0, 0, 0)

# Add the ground and pendulum bodies to the simulation
ch.ChAddBody(ground_body)
ch.ChAddBody(pendulum_body)

# Set the time step and simulation duration
tstep = 0.001
tend = 10

# Create an Irrlicht visualization system
vis = ch.ChIrrApp(ch.ChSystem(), "Simple Pendulum", ch.dimension2D(800, 600))

# Main simulation loop
while vis.GetDriver().run() and vis.GetSim().GetChTime() < tend:
    # Update the simulation
    vis.GetSim().DoStepDynamics(tstep)

    # Log the pendulum's position and velocity
    pos = pendulum_body.GetPos()
    vel = pendulum_body.GetVel()
    print(f"Time: {vis.GetSim().GetChTime():.4f}, Position: {pos.x:.4f} {pos.y:.4f} {pos.z:.4f}, Velocity: {vel.x:.4f} {vel.y:.4f} {vel.z:.4f}")

# Clean up
ch.ChCloseChSystem()
```

This code creates a simple pendulum simulation with a ground body and a pendulum body connected by a revolute joint. The simulation is run for 10 seconds, and the pendulum's position and velocity are logged at each time step. The simulation is visualized using the Irrlicht visualization system.
