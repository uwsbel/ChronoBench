```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import math

# Initialize the Chrono system
chrono.SetChronoDataPath("C:/Program Files/Chrono/data/")
system = chrono.ChSystemNSC()

# Set gravity
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the ground body
ground = chrono.ChBodyEasyBox(100, 1, 100, 1000, True, True)
ground.SetBodyFixed(True)
system.Add(ground)

# Initialize the Viper rover
viper = veh.Viper()
viper.SetContactMethod(chrono.ChContactMethod_NSC)
viper.SetChassisFixed(False)
viper.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0.5, 0), chrono.ChQuaternionD(1, 0, 0, 0)))
viper.Initialize()

# Set up the Viper driver
driver = veh.ViperDriver(viper)

# Add the Viper's subsystems to the Chrono system
viper.GetSystem().Add(viper.GetChassis())
viper.GetSystem().AddAssembly(viper.GetVehicle())

# Create the Irrlicht visualization
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle("Viper Rover Simulation")
vis.Initialize()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 5, -10), chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()

# Simulation parameters
step_size = 1e-3
steering_time = 5.0  # Time to complete steering change
max_steering = 0.5  # Maximum steering angle in radians

# Simulation loop
time = 0
while vis.Run():
    # Calculate steering angle based on time
    if time < steering_time:
        steering = (time / steering_time) * max_steering
    else:
        steering = max_steering

    # Update driver inputs
    driver_inputs = veh.ViperDriverInputs()
    driver_inputs.m_steering = steering
    driver_inputs.m_throttle = 0.5  # Constant throttle

    # Update the Viper and its subsystems
    driver.Synchronize(time)
    viper.Synchronize(time, driver_inputs, system.GetContactContainer())
    system.DoStepDynamics(step_size)

    # Update visualization
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    time += step_size

# Clean up
vis.GetDevice().closeDevice()
```

This script does the following:

1. Initializes the PyChrono environment and core components, including the system, gravity, and ground.

2. Sets up the Viper rover and its driver.

3. Creates an Irrlicht visualization with camera settings and lighting.

4. Implements a simulation loop that:
   - Gradually changes the rover's steering over a specified time period.
   - Updates the rover's position and dynamics at each timestep.
   - Renders the scene using Irrlicht.

5. Cleans up resources after the simulation ends.
