Here's a PyChrono script that implements the simulation as described in your instructions:

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import os

# Initialize the ChronoEngine and vehicle components
chrono.SetChronoDataPath(os.path.join(os.path.dirname(__file__), "data/"))
veh.SetDataPath(chrono.GetChronoDataPath() + "vehicle/")

# Create the simulation system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the Irrlicht visualization
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle("Gator Vehicle Simulation")
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataPath() + "logo_pychrono_alpha.png")
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 5, -15))
vis.AddTypicalLights()

# Create the terrain
terrain = veh.RigidTerrain(system)
patch_mat = chrono.ChMaterialSurfaceNSC()
patch = terrain.AddPatch(patch_mat, 
                         chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT),
                         100.0, 100.0)
patch.SetTexture(veh.GetDataFile("terrain/textures/grass.jpg"), 200, 200)
terrain.Initialize()

# Create the Gator vehicle
gator = veh.Gator()
gator.SetContactMethod(chrono.ChContactMethod_NSC)
gator.SetChassisFixed(False)
gator.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 1.0, 0), chrono.QUNIT))
gator.SetTireType(veh.TireModelType_TMEASY)
gator.SetTireStepSize(1e-3)
gator.Initialize()

gator.SetChassisVisualizationType(veh.VisualizationType_MESH)
gator.SetSuspensionVisualizationType(veh.VisualizationType_MESH)
gator.SetSteeringVisualizationType(veh.VisualizationType_MESH)
gator.SetWheelVisualizationType(veh.VisualizationType_MESH)
gator.SetTireVisualizationType(veh.VisualizationType_MESH)

# Create the driver system
driver = veh.ChIrrGuiDriver(vis)
driver.Initialize()

# Simulation loop
step_size = 1e-3
realtime_timer = chrono.ChRealtimeStepTimer()

while vis.Run():
    time = system.GetChTime()

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Update vehicle
    gator.Synchronize(time, driver_inputs, terrain)
    terrain.Synchronize(time)
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance simulation
    gator.Advance(step_size)
    terrain.Advance(step_size)
    system.DoStepDynamics(step_size)

    # Delay to achieve real-time simulation
    realtime_timer.Spin(step_size)

    # Exit if 'ESC' key is pressed
    if vis.GetKeyManager().IsKeyPressed(irr.KEY_ESCAPE):
        break

# Clean up
vis.GetDevice().closeDevice()
```

This script does the following:

1. Initializes the PyChrono environment and imports necessary modules.
2. Sets up the ChronoEngine system with gravity.
3. Creates an Irrlicht visualization system with a camera and lights.
4. Generates a flat rigid terrain with a grass texture.
5. Initializes a Gator vehicle with TMEASY tire model and mesh visualization for all components.
6. Sets up an interactive driver system for user input.
7. Implements a simulation loop that:
   - Updates vehicle dynamics based on driver inputs
   - Renders the scene
   - Advances the simulation in real-time at 50 FPS (using a step size of 1e-3 seconds)
   - Allows exiting the simulation with the 'ESC' key
