```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import math

# Initialize the ChronoEngine and other core components
chrono.SetChronoDataPath("/path/to/chrono/data/")
veh.SetDataPath("/path/to/vehicle/data/")

# Create the simulation system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the Irrlicht visualization system
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle("HMMWV Simulation")
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 1.5, -8))
vis.AddTypicalLights()

# Create the terrain
terrain = veh.RigidTerrain(system)
terrain_file_vis = "/path/to/Highway_vis.obj"
terrain_file_col = "/path/to/Highway_col.obj"

terrain_mesh = terrain.AddMesh(chrono.GetChronoDataFile(terrain_file_col),
                               chrono.GetChronoDataFile(terrain_file_vis),
                               chrono.VNULL, chrono.QUNIT, 1.0)
terrain.Initialize()

# Create the vehicle
vehicle_file = veh.GetDataFile("hmmwv/vehicle/HMMWV_Vehicle.json")
tire_file = veh.GetDataFile("hmmwv/tire/HMMWV_TMeasyTire.json")
powertrain_file = veh.GetDataFile("hmmwv/powertrain/HMMWV_ShaftsPowertrain.json")

vehicle = veh.WheeledVehicle(system, vehicle_file)
vehicle.Initialize(chrono.ChCoordsysD(chrono.ChVectorD(0, 1.0, 0), chrono.ChQuaternionD(1, 0, 0, 0)))

vehicle.SetChassisVisualizationType(veh.VisualizationType_MESH)
vehicle.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetWheelVisualizationType(veh.VisualizationType_MESH)

# Create and initialize the powertrain
powertrain = veh.ReadPowertrainJSON(powertrain_file)
vehicle.InitializePowertrain(powertrain)

# Create and initialize the tires
for axle in vehicle.GetAxles():
    tireL = veh.ReadTireJSON(tire_file)
    vehicle.InitializeTire(tireL, axle.m_wheels[0], veh.VisualizationType_MESH)
    tireR = veh.ReadTireJSON(tire_file)
    vehicle.InitializeTire(tireR, axle.m_wheels[1], veh.VisualizationType_MESH)

# Create the driver system
driver = veh.ChIrrGuiDriver(vis)
driver.SetSteeringDelta(0.02)
driver.SetThrottleDelta(0.02)
driver.SetBrakingDelta(0.02)
driver.Initialize()

# Create the vehicle Irrlicht interface
vehicle_vis = veh.ChWheeledVehicleIrrApp(vehicle, vis)
vehicle_vis.SetHUDLocation(500, 20)
vehicle_vis.SetChassisTrackPoints(chrono.ChVectorD(0.0, 0.0, -1.75),
                                  chrono.ChVectorD(0.0, 0.0, 1.75),
                                  1.0)
vehicle_vis.Initialize()

# Simulation loop
realtime_timer = chrono.ChRealtimeStepTimer()
step_size = 1.0 / 50  # Time step size (50 Hz)

while vis.Run():
    time = vehicle.GetSystem().GetChTime()

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Update modules (process inputs from other modules)
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    vis.Synchronize("", driver_inputs)

    # Advance simulation for one timestep for all modules
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    # Render scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Ensure real-time simulation
    realtime_timer.Spin(step_size)

# Clean up
vehicle_vis.Shutdown()
```
