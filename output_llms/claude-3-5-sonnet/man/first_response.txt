```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import pychrono.vehicle.irrlicht as virr
import os

# Initialize the ChronoVehicle system
chrono.SetChronoDataPath(chrono.GetChronoDataPath())

# Create the simulation system
system = chrono.ChSystemNSC()

# Create the vehicle
vehicle = veh.MAN_10t(system)
vehicle.Initialize(chrono.ChCoordsysD(chrono.ChVectorD(0, 1, 0), chrono.ChQuaternionD(1, 0, 0, 0)))
vehicle.SetChassisVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
vehicle.SetWheelVisualizationType(veh.VisualizationType_NONE)

# Create and initialize the powertrain
powertrain = veh.SimplePowertrain("Powertrain")
vehicle.InitializePowertrain(powertrain)

# Create and initialize the tires
tire_data = veh.TMeasyTire("tire_data")
for axle in vehicle.GetAxles():
    for wheel in axle.GetWheels():
        tire = veh.TMeasyTire(tire_data)
        vehicle.InitializeTire(tire, wheel, veh.VisualizationType_MESH)

# Create the terrain
terrain = veh.RigidTerrain(system)
patch_mat = chrono.ChMaterialSurfaceNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)
terrain_file = chrono.GetChronoDataFile("vehicle/terrain/RigidPlane.obj")
patch = terrain.AddPatch(patch_mat, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 1, 0), terrain_file)
patch.SetTexture(chrono.GetChronoDataFile("vehicle/terrain/textures/tile4.jpg"), 200, 200)
terrain.Initialize()

# Create the driver system
driver = veh.ChIrrGuiDriver(vehicle.GetVehicle())
driver.Initialize()

# Create the vehicle Irrlicht interface
vis = virr.ChWheeledVehicleIrrApp(vehicle.GetVehicle(), 'MAN 10t Truck Simulation')

vis.SetHUDLocation(500, 20)
vis.SetSkyBox()
vis.AddTypicalLogo()
vis.AddTypicalLights(irr.vector3df(30, -30, 100), irr.vector3df(30, 50, 100), 250, 130)
vis.AddTypicalCamera(irr.vector3df(6, 3, -6), irr.vector3df(0, 0, 0))
vis.SetChaseCamera(chrono.ChVectorD(0, 0, -10), 5, 0.05)
vis.Initialize()

# Simulation loop
time_step = 1e-3
while vis.GetDevice().run():
    time = system.GetChTime()

    # Render scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Update modules (process inputs from other modules)
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    vis.Synchronize(driver.GetInputModeAsString(), driver_inputs)

    # Advance simulation for one timestep for all modules
    driver.Advance(time_step)
    terrain.Advance(time_step)
    vehicle.Advance(time_step)
    vis.Advance(time_step)

    # Increment system time
    system.DoStepDynamics(time_step)

# Clean up
vis.GetDevice().drop()
```

This script does the following:

1. Initializes the PyChrono environment and core components.
2. Creates a MAN 10t truck vehicle model with configurable visualization settings.
3. Sets up a TMEASY tire model for each wheel.
4. Creates a rigid terrain with customizable textures.
5. Implements a driver system with real-time controls for steering, throttle, and braking.
6. Sets up the Irrlicht visualization system with a chase camera, directional lighting, a skybox, and customizable terrain textures and logos.
7. Runs the simulation loop, updating all components and rendering the scene in real-time.
