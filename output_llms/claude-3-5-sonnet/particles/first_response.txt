```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import random
import math

# Initialize the ChSystem
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))  # Disable default gravity

# Create the Irrlicht visualization
application = chronoirr.ChIrrApp(system, "Particle Gravitational Attraction", chronoirr.dimension2du(1024, 768))
application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(chronoirr.vector3df(0, 5, -10))

# Particle emitter parameters
emission_rate = 5  # particles per second
max_particles = 100
particle_mass = 1.0
particle_density = 1000
emission_area = 10
gravitational_constant = 6.67430e-11 * 1e10  # Scaled up for visibility

particles = []

# Custom gravitational force function
def apply_gravitational_forces():
    for i, particle1 in enumerate(particles):
        for particle2 in particles[i+1:]:
            pos1 = particle1.GetPos()
            pos2 = particle2.GetPos()
            direction = pos2 - pos1
            distance = direction.Length()
            
            if distance > 0:
                force_magnitude = gravitational_constant * particle1.GetMass() * particle2.GetMass() / (distance ** 2)
                force = direction.GetNormalized() * force_magnitude
                
                particle1.Accumulate_force(force, pos1, False)
                particle2.Accumulate_force(-force, pos2, False)

# Particle emitter function
def emit_particle():
    if len(particles) < max_particles:
        # Random shape
        shape_type = random.choice(["sphere", "box", "cylinder"])
        
        if shape_type == "sphere":
            radius = random.uniform(0.1, 0.3)
            shape = chrono.ChSphereShape(radius)
        elif shape_type == "box":
            size = chrono.ChVectorD(random.uniform(0.1, 0.3), random.uniform(0.1, 0.3), random.uniform(0.1, 0.3))
            shape = chrono.ChBoxShape(size)
        else:  # cylinder
            radius = random.uniform(0.1, 0.2)
            height = random.uniform(0.2, 0.4)
            shape = chrono.ChCylinderShape(radius, height)
        
        # Create the rigid body
        body = chrono.ChBodyEasySphere(radius, particle_density)
        
        # Set random position
        x = random.uniform(-emission_area/2, emission_area/2)
        y = random.uniform(5, 10)
        z = random.uniform(-emission_area/2, emission_area/2)
        body.SetPos(chrono.ChVectorD(x, y, z))
        
        # Set random velocity
        vx = random.uniform(-1, 1)
        vy = random.uniform(-1, 1)
        vz = random.uniform(-1, 1)
        body.SetPos_dt(chrono.ChVectorD(vx, vy, vz))
        
        # Set random orientation
        angle = random.uniform(0, 2*math.pi)
        axis = chrono.ChVectorD(random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1))
        rotation = chrono.Q_from_AngAxis(angle, axis)
        body.SetRot(rotation)
        
        # Add to the system and particle list
        system.Add(body)
        particles.append(body)

# Set up the simulation loop
application.SetTimestep(0.01)
application.SetTryRealtime(True)

while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    
    # Emit new particles
    if system.GetChTime() % (1.0 / emission_rate) < application.GetTimestep():
        emit_particle()
    
    # Apply gravitational forces
    apply_gravitational_forces()
    
    # Step the simulation
    application.DoStep()
    application.EndScene()

```
