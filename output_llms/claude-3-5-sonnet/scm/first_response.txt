Here's a PyChrono script that implements the simulation as described in your instructions:

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import pychrono.terrain as terrain

# Initialize the ChronoEngine and vehicle module
chrono.SetChronoDataPath('/path/to/chrono/data/')
veh.SetDataPath('/path/to/vehicle/data/')

# Create the simulation system
system = chrono.ChSystemSMC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the Irrlicht visualization
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('HMMWV on SCM Terrain')
vis.Initialize()

# Create the HMMWV vehicle
hmmwv = veh.HMMWV_Full(system)
hmmwv.SetChassisFixed(False)
hmmwv.SetChassisVis(veh.VisualizationType_MESH)
hmmwv.SetSuspensionVis(veh.VisualizationType_PRIMITIVES)
hmmwv.SetSteeringVis(veh.VisualizationType_PRIMITIVES)
hmmwv.SetWheelVis(veh.VisualizationType_MESH)

# Set initial vehicle location and orientation
hmmwv.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 1.0, 0), chrono.QUNIT))
hmmwv.Initialize()

# Set up rigid tire model
tire_type = veh.RigidTire_Type()
for axle in hmmwv.GetAxles():
    for wheel in axle.GetWheels():
        tire = veh.RigidTire(tire_type)
        hmmwv.InitializeTire(tire, wheel, veh.VisualizationType_MESH)

# Create SCM terrain
terrain_params = terrain.SCMParameters()
terrain_params.Bekker_Kphi = 0.2e6
terrain_params.Bekker_Kc = 0
terrain_params.Bekker_n = 1.1
terrain_params.Mohr_cohesion = 0
terrain_params.Mohr_friction = 30
terrain_params.Janosi_shear = 0.01
terrain_params.elastic_K = 4e7
terrain_params.damping_R = 3e4

scm_terrain = terrain.SCMDeformableTerrain(system)
scm_terrain.SetPlane(chrono.ChCoordsysD(chrono.VNULL, chrono.Q_from_AngX(-0.1)))
scm_terrain.Initialize(300, 300, 1.0)

scm_terrain.SetSoilParameters(terrain_params)

# Enable moving patch feature
patch_width = 5.0
patch_length = 12.0
scm_terrain.AddMovingPatch(hmmwv.GetChassisBody(), chrono.ChVectorD(0, 0, 0),
                           chrono.ChVectorD(patch_width, 0, patch_length))

# Set up visualization of SCM terrain
scm_terrain.SetPlotType(terrain.SCMDeformableTerrain.PLOT_SINKAGE, 0, 0.1)

# Create the interactive driver system
driver = veh.ChIrrGuiDriver(vis)
driver.SetSteeringDelta(0.02)
driver.SetThrottleDelta(0.02)
driver.SetBrakingDelta(0.06)
driver.Initialize()

# Simulation loop
time_step = 1.0 / 50.0  # 50 FPS
while vis.Run():
    t = system.GetChTime()

    # Synchronize driver inputs
    driver.Synchronize(t)
    hmmwv.Synchronize(t, driver.GetInputs())

    # Advance simulation
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    system.DoStepDynamics(time_step)

    # Update visualization
    hmmwv.Advance(time_step)
    driver.Advance(time_step)

# Clean up
vis.GetDevice().closeDevice()
```

This script creates a PyChrono simulation of a full HMMWV on SCM deformable terrain with Irrlicht visualization, as per your instructions. Here's a breakdown of the key components:

1. The HMMWV vehicle is initialized with mesh visualization for all components.
2. A rigid tire model is applied to all wheels.
3. The SCM terrain is configured with custom soil parameters and a moving patch feature.
4. Terrain visualization includes sinkage with false color plotting.
5. An interactive driver system is set up to control the vehicle.
6. The simulation runs in real-time, updating at 50 frames per second.
