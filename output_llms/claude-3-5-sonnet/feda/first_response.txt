import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import pychrono.vehicle.irrlicht as virr
import os

# Initialize the ChronoVehicle system
chrono.SetChronoDataPath(os.path.join(os.path.dirname(__file__), "data/"))
veh.SetDataPath(chrono.GetChronoDataPath() + "vehicle/")

# Create the simulation system
system = chrono.ChSystemNSC()

# Create the vehicle
vehicle = veh.WheeledVehicle(system, veh.GetDataFile("hmmwv/vehicle/HMMWV_Vehicle.json"))
vehicle.Initialize(chrono.ChCoordsysD(chrono.ChVectorD(0, 1, 0), chrono.ChQuaternionD(1, 0, 0, 0)))

# Set collision family
vehicle.SetCollisionFamily(1)

# Set the tire model (Pacejka)
vehicle.SetTireType(veh.TireModelType_PACEJKA)

# Create and initialize the powertrain
powertrain = veh.ReadPowertrainJSON(veh.GetDataFile("hmmwv/powertrain/HMMWV_ShaftsPowertrain.json"))
vehicle.InitializePowertrain(powertrain)

# Create the terrain
terrain = veh.RigidTerrain(system, veh.GetDataFile("terrain/RigidPlane.json"))

# Set terrain properties
patch_mat = chrono.ChMaterialSurfaceNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)
terrain.SetContactMaterial(patch_mat)

# Set terrain visualization properties
terrain.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
terrain.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)

# Create the Irrlicht visualization
vis = virr.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle("FEDA Vehicle Simulation")
vis.SetWindowSize(1280, 720)
vis.SetChaseCamera(chrono.ChVectorD(0, 0, 1.75), 6.0, 0.5)
vis.Initialize()
vis.AddLightDirectional()
vis.AttachVehicle(vehicle)

# Create the interactive driver system
driver = virr.ChIrrGuiDriver(vis)

# Set the time response for steering, throttle and braking inputs
steering_time = 1.0
throttle_time = 1.0
braking_time = 0.3
driver.SetSteeringDelta(0.1 * steering_time)
driver.SetThrottleDelta(0.1 * throttle_time)
driver.SetBrakingDelta(0.1 * braking_time)

# Set the vehicle's driver
vehicle.SetDriver(driver)

# Simulation loop
step_size = 1e-3
render_step_size = 1.0 / 50  # 50 FPS
render_steps = int(render_step_size / step_size)

realtime_timer = chrono.ChRealtimeStepTimer()

while vis.Run():
    time = vehicle.GetSystem().GetChTime()

    # Render scene and output POV-Ray data
    if vis.GetVisualizationType() == virr.VisualizationType_MESH:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Update modules (process inputs from other modules)
    driver.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    terrain.Synchronize(time)
    vis.Synchronize(driver.GetInputModeAsString(), driver_inputs)

    # Advance simulation for one timestep for all modules
    driver.Advance(step_size)
    vehicle.Advance(step_size)
    terrain.Advance(step_size)
    vis.Advance(step_size)

    # Spin in place for real time to catch up
    realtime_timer.Spin(step_size)

    if time > 20:
        break

# Clean up
vis.Shutdown()
