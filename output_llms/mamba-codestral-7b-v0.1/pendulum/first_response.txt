Here is a Python script that uses the PyChrono library to simulate a simple pendulum with Irrlicht for visualization:

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Initialize the PyChrono environment and core components
chrono.SetChronoDataPath('path_to_chrono_data')
chrono.SetChronoOutputLevel(chrono.ChLogLevel_WARNING)

system = chrono.ChSystemNSC()

# Add the required physical systems and objects
ground = chrono.ChBody()
ground.SetBodyFixed(True)
system.AddBody(ground)

pendulum = chrono.ChBody()
pendulum.SetMass(1.0)
pendulum.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
pendulum.SetPos(chrono.ChVectorD(0, 1, 0))
system.AddBody(pendulum)

joint = chrono.ChRevoluteJoint()
joint.SetName('pendulum_joint')
joint.Initialize(ground, pendulum, chrono.ChVectorD(0, 1, 0))
system.AddLink(joint)

# Set necessary default parameters such as positions, forces, and interactions
pendulum.SetPos_dt(chrono.ChVectorD(0, -1, 0))

# Visualize the simulation using Irrlicht
application = chronoirr.ChIrrApp(system, 'Simple Pendulum Simulation', chronoirr.dimension2du(800, 600))
application.AddTypicalLogo()
application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(chronoirr.camera_follow_mode.CAMERA_FOLLOW_NONE)

# Periodically log the pendulum's position and velocity
class PendulumLogger(chrono.ChFunction_Timer):
    def __init__(self, pendulum):
        chrono.ChFunction_Timer.__init__(self)
        self.pendulum = pendulum

    def GetTimeInfo(self):
        return 'Pendulum Position: ' + str(self.pendulum.GetPos()) + ', Velocity: ' + str(self.pendulum.GetPos_dt())

logger = PendulumLogger(pendulum)
logger.SetTimestep(1.0)
system.AddChFunction(logger)

# Run the simulation
application.SetTimestep(0.01)
application.SetTryRealtime(True)
application.Run()
```

This script initializes the PyChrono environment, creates a ground body, a pendulum body with mass and inertia, connects them with a revolute joint, sets the pendulum's initial position and velocity, and logs the pendulum's position and velocity periodically. The simulation is then run using the Irrlicht visualization library.
