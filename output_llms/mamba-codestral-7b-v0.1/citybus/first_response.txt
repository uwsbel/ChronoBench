To generate a simulation script based on the given instructions, we will use the PyChrono library for simulation and Irrlicht for visualization. Here is the code:

```python
import pychrono
import pychrono.irrlicht

# Initialize PyChrono
system = pychrono.ChSystemSMC()

# Add required physical systems and objects
terrain = pychrono.RigidTerrain(system)
terrain.SetTexture("terrain_texture.png")

bus = pychrono.ChVehicle(system)
bus.SetChassisCollisionType(pychrono.ChCollisionModel.ChCollisionModelSphere)
bus.SetChassisFixed(False)
bus.SetInitPosition(pychrono.ChVectorD(0, 0, 0))
bus.SetInitOrientation(pychrono.ChQuaternionD(0, 0, 0))

tire_model = pychrono.RigidTireModel(bus, terrain)
tire_model.SetTireType(pychrono.TireModelType.RIGID_TIRE)

# Set default parameters
bus.SetChassisCollisionHalfSize(pychrono.ChVectorD(2, 1, 1))
bus.SetChassisMass(2000)
bus.SetChassisInertiaXX(pychrono.ChVectorD(500, 500, 500))
bus.SetChassisInertiaXY(pychrono.ChVectorD(0, 0, 0))
bus.SetChassisInertiaYZ(pychrono.ChVectorD(0, 0, 0))
bus.SetChassisCollisionMargin(0.01)
bus.SetChassisVisualizationType(pychrono.VisualizationType.MESH)
bus.SetChassisMesh(pychrono.GetChronoDataFile("shapes/bus.obj"))
bus.SetChassisMeshScale(pychrono.ChVectorD(1, 1, 1))
bus.SetChassisColor(pychrono.ChColor(1, 0, 0))

camera = pychrono.ChCameraDummy()
camera.SetPosition(pychrono.ChVectorD(0, 10, -10))
camera.SetTarget(pychrono.ChVectorD(0, 0, 0))
camera.SetVertical(pychrono.ChVectorD(0, 0, 1))

# Set Irrlicht visualization
visualization = pychrono.ChIrrlichtVisualizationData()
visualization.screen_size = (800, 600)
visualization.window_title = "CityBus Simulation"
visualization.camera = camera
visualization.add_vehicle(bus)

# Run simulation loop
system.SetTimestep(0.02)
system.SetSolverType(pychrono.ChSolver.Type.BARZILAIBORWEIN)
system.SetMaxIterationNumber(50)

simulator = pychrono.ChSimulation(system)
simulator.SetVisualizationType(pychrono.ChVisualizationType.IRRLICHT)
simulator.SetIrrlichtVisualization(visualization)
simulator.Start()

while simulator.GetTime() < 10:
    simulator.DoStep()

simulator.Stop()
```

This code initializes the PyChrono environment, creates a rigid terrain with a custom texture, adds a CityBus vehicle with specified parameters, and sets up the camera for visualization. The simulation loop runs at 50 frames per second, updating the vehicle dynamics and rendering the scene in real time. The simulation stops after 10 seconds.
