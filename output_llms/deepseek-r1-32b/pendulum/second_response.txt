import pychrono.irrlicht as chronoirr

pend_cyl = chrono.ChVisualShapeCylinder(0.1, 1.5)  # Cylinder with radius 0.1 and height 1.5
pend_cyl.SetColor(chrono.ChColor(0.6, 0, 0))  # Set the color of the cylinder (RGB)

pend_1.SetPos(chrono.ChVector3d(0.75, 0, 1))  # Position adjusted for the new pendulum length
