```python
importchrono
import math

# Initialize the Chrono system
def initialize_system():
    # Create the Chrono system with standard settings
    system = chrono.ChSystemNSC()
    system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity
    system.Set_time_step(0.001)  # Set time step
    return system

# Create vehicle, terrain, and driver
def create_simulation(system):
    # Create M113 vehicle
    vehicle = chrono.vehicle.M113()
    vehicle.Initialize(system, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 1), chrono.ChQuaternionD(1, 0, 0, 0)))
    
    # Create terrain
    terrain = chrono.ChBody()
    terrain.SetBodyFixed(True)
    terrain.SetCollide(True)
    terrain.SetMaterialSurface(chrono.ChMaterialSurface())
    terrain.GetMaterialSurface().SetFriction(0.8)
    terrain.GetMaterialSurface().SetRestitution(0.1)
    
    # Create ground plane
    plane = chrono.ChPlane()
    plane.GetPlaneEquation().SetNormal(chrono.ChVectorD(0, 1, 0))
    plane.GetPlaneEquation().SetOffset(0)
    terrain.AddCollisionShape(plane)
    
    terrain.SetPos(chrono.ChVectorD(0, 0, 0))
    system.Add(terrain)
    
    # Create driver
    driver = chrono.vehicle.ChDriver()
    driver.Initialize(vehicle.GetDriverInterface())
    
    return vehicle, terrain, driver

# Set up visualization
def setup_visualization(system):
    # Create Irrlicht application
    app = chrono.ChIrrApp(system, "M113 Vehicle Simulation", chrono.CHDimension(800, 600))
    app.SetCamera(chrono.ChVectorD(5, 5, 5), chrono.ChVectorD(0, 0, 0))
    app.AddTypicalLights()
    return app

# Main simulation loop
def simulate(vehicle, terrain, driver, app):
    time = 0
    while app.Run():
        # Update vehicle
        vehicle.Update(time)
        
        # Update driver
        driver.Update(time)
        
        # Update terrain (if needed)
        #terrain.Update(time)
        
        # Advance simulation
        app.BeginFrame()
        system.DoStepDynamics(system.Get_time_step())
        app.EndFrame()
        
        time += system.Get_time_step()

# Main function
def main():
    try:
        system = initialize_system()
        vehicle, terrain, driver = create_simulation(system)
        app = setup_visualization(system)
        simulate(vehicle, terrain, driver, app)
    except Exception as e:
        print(f"An error occurred: {e}")
    finally:
        app.Close()

if __name__ == "__main__":
    main()
```
