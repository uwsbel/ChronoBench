```python
import pychrono as chrono
import pychrono.vehicle as vehicle
import pychrono.irrlicht as irr

# Initialize the Chrono system
system = chrono.ChSystem()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Gravity

# Create the Irrlicht visualization
vis = irr.ChIrrApp(system, "HMMWV Simulation", irr.dimension2du(800, 600))
vis.SetCameraPosition(chrono.ChVectorD(5, 5, 5))
vis.SetCameraLookatPoint(chrono.ChVectorD(0, 0, 0))
vis.AddLight(chrono.ChVectorD(5, 5, 5), chrono.ChColor(1, 1, 1))
vis.AddLight(chrono.ChVectorD(-5, 5, 5), chrono.ChColor(1, 1, 1))

# Create the HMMWV vehicle
hmmwv = vehicle.ChHMMWV()
hmmwv.SetChassisMass(2500)  # Vehicle mass
hmmwv.SetChassisInertia(chrono.ChVectorD(500, 1000, 500))
hmmwv.SetChassisDimensions(chrono.ChVectorD(4, 2, 1.5))
hmmwv.SetSuspensionType(vehicle.ChSuspensionType.FINDEDGE)
hmmwv.SetTireType(vehicle.ChTireType.RIGID)
hmmwv.SetPowertrainType(vehicle.ChPowertrainType.SIMPLE)
hmmwv.SetSteeringType(vehicle.ChSteeringType.RACKPINION)

# Initialize vehicle
hmmwv.Initialize(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0.5), chrono.ChQuaternionD(1, 0, 0, 0)))

# Create driver inputs
driver = vehicle.ChDriverInputs()
driver.m_throttle = 0.5  # Throttle input (0-1)
driver.m_steering = 0   # Steering input (-1 to 1)

# Create terrain
terrain = vehicle.ChTerrain()
terrain.SetSize(100, 100)  # Terrain dimensions
terrain.SetTexture(vehicle.ChTexture("textures/grass.jpg"))  # Add texture
terrain.Initialize()

# Add sensors
imu = vehicle.ChIMU()
imu.Attach(hmmwv.GetChassis())
imu.SetName("IMU")
system.Add(imu)

gps = vehicle.ChGPS()
gps.Attach(hmmwv.GetChassis())
gps.SetName("GPS")
system.Add(gps)

# Add visualization assets
vis.AddChassis(hmmwv.GetChassis())
vis.AddSuspension(hmmwv.GetSuspension())
vis.AddWheels(hmmwv.GetWheels())

# Add camera and lights
camera = irr.ChIrr_Camera(vis.GetDevice())
camera.SetPosition(chrono.ChVectorD(5, 5, 5))
camera.SetLookAtPoint(chrono.ChVectorD(0, 0, 0))
vis.AddAsset(camera)

# Simulation parameters
simulation_steps = 1000
step_size = 0.01

# Main simulation loop
for step in range(simulation_steps):
    # Update driver inputs
    hmmwv.GetDriver().SetInputs(driver)
    
    # Advance vehicle
    hmmwv.Advance(step_size)
    
    # Advance terrain
    terrain.Advance(step_size)
    
    # Update sensors
    imu.Update()
    gps.Update()
    
    # Update visualization
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    
    # Process sensor data
    acceleration = imu.GetAcceleration()
    position = gps.GetPosition()
    
    # Output vehicle mass
    print(f"Vehicle Mass: {hmmwv.GetMass()} kg")
    print(f"Acceleration: {acceleration}")
    print(f"Position: {position}")
    
    # Check for window closure
    if vis.GetDevice().IsQuit():
        break

# Stop visualization
vis.Stop()
```
