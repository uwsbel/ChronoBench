Here's a PyChrono simulation script that creates an epicyclic gear system with a fixed truss, rotating bar, and two gears with a motor:

```python
importchrono
import math
import time

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.Set_Gravity(chrono.ChVectorD(0, -9.81, 0))

# Create the Irrlicht visualization
vis = chrono.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle("Epicyclic Gear System")
vis.SetLightPosition(chrono.ChVectorD(0, 1, 3))
vis.SetLightDir(chrono.ChVectorD(0, -1, -1))
vis.SetCameraPosition(chrono.ChVectorD(0, 1.5, 2))
vis.SetCameraRotation(chrono.ChVectorD(0, 0, 0))

# Create fixed truss
truss_node1 = chrono.ChBody()
truss_node1.SetBodyFixed(True)
truss_node1.SetPos(chrono.ChVectorD(0, 0, 0))
system.Add(truss_node1)

truss_node2 = chrono.ChBody()
truss_node2.SetBodyFixed(True)
truss_node2.SetPos(chrono.ChVectorD(0, 0, 0.5))
system.Add(truss_node2)

truss_joint = chrono.ChLinkRev()
truss_joint.Initialize(truss_node1, truss_node2, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.ChMatrix33D()))
system.Add(truss_joint)

# Create rotating bar
rotating_bar = chrono.ChBody()
rotating_bar.SetMass(1)
rotating_bar.SetPos(chrono.ChVectorD(0, 0, 0.25))
rotating_bar.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
system.Add(rotating_bar)

bar_joint = chrono.ChLinkRev()
bar_joint.Initialize(truss_node1, rotating_bar, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.ChMatrix33D()))
system.Add(bar_joint)

# Create gears
sun_gear = chrono.ChBody()
sun_gear.SetMass(1)
sun_gear.SetPos(chrono.ChVectorD(0, 0, 0.25))
sun_gear.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
sun_gear.SetAxisSize(0.1)
sun_gear.SetRadius(0.2)
system.Add(sun_gear)

planet_gear = chrono.ChBody()
planet_gear.SetMass(1)
planet_gear.SetPos(chrono.ChVectorD(0.3, 0, 0.25))
planet_gear.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
planet_gear.SetAxisSize(0.1)
planet_gear.SetRadius(0.2)
system.Add(planet_gear)

# Create gear joints
sun_joint = chrono.ChLinkRev()
sun_joint.Initialize(truss_node1, sun_gear, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.ChMatrix33D()))
system.Add(sun_joint)

planet_joint = chrono.ChLinkRev()
planet_joint.Initialize(rotating_bar, planet_gear, chrono.ChCoordsysD(chrono.ChVectorD(0.3, 0, 0), chrono.ChMatrix33D()))
system.Add(planet_joint)

# Add motor to sun gear
motor = chrono.ChLinkMotorRotation()
motor.Initialize(sun_gear, truss_node1, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.ChMatrix33D()))
motor.SetSpeed(chrono.ChFunction_Const(10))  # 10 rad/s
system.Add(motor)

# Add contact properties
sun_gear.GetCollisionModel().AddMaterial(chrono.ChMaterialSurface())
sun_gear.GetCollisionModel().SetMaterial(0, chrono.ChMaterialSurface())
sun_gear.GetCollisionModel().SetFriction(10)

planet_gear.GetCollisionModel().AddMaterial(chrono.ChMaterialSurface())
planet_gear.GetCollisionModel().SetMaterial(0, chrono.ChMaterialSurface())
planet_gear.GetCollisionModel().SetFriction(10)

# Add visualization assets
sun_gear.GetVisualShape().SetMaterial(chrono.ChVisualMaterial())
sun_gear.GetVisualShape().SetMaterial().SetDiffuseColor(chrono.ChColor(0.8, 0.6, 0.2))  # Bronze color
sun_gear.GetVisualShape().SetMaterial().SetSpecularColor(chrono.ChColor(0.3, 0.3, 0.3))
sun_gear.GetVisualShape().SetMaterial().SetShininess(10)

planet_gear.GetVisualShape().SetMaterial(chrono.ChVisualMaterial())
planet_gear.GetVisualShape().SetMaterial().SetDiffuseColor(chrono.ChColor(0.8, 0.6, 0.2))  # Bronze color
planet_gear.GetVisualShape().SetMaterial().SetSpecularColor(chrono.ChColor(0.3, 0.3, 0.3))
planet_gear.GetVisualShape().SetMaterial().SetShininess(10)

truss_node1.GetVisualShape().SetMaterial(chrono.ChVisualMaterial())
truss_node1.GetVisualShape().SetMaterial().SetDiffuseColor(chrono.ChColor(0.7, 0.7, 0.7))  # Silver color
truss_node1.GetVisualShape().SetMaterial().SetSpecularColor(chrono.ChColor(0.3, 0.3, 0.3))
truss_node1.GetVisualShape().SetMaterial().SetShininess(10)

# Add visualization shapes
sun_gear.GetVisualShape().SetShape(chrono.ChBox(0.2, 0.2, 0.1))
planet_gear.GetVisualShape().SetShape(chrono.ChBox(0.2, 0.2, 0.1))
truss_node1.GetVisualShape().SetShape(chrono.ChBox(0.1, 0.1, 0.1))

# Run simulation
vis.BeginScene()
vis.Render()
vis.EndScene()

while vis.GetDevice().Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    system.DoStepDynamics(0.001)
    time.sleep(0.001)

vis.GetDevice().Close()
```

This script creates:
1. A fixed truss structure
2. A rotating bar connected to the truss
3. Two gears (sun gear and planet gear)
4. A motor to drive the sun gear at a constant speed
5. Proper materials and lighting for visualization

The simulation demonstrates:
- Epicyclic gear motion
- Rotational dynamics
- Gear-to-gear contact and interaction
- Proper visualization of the mechanical system

The gears are made of bronze material, and the truss is silver-colored. The system is illuminated with a point light source, and the camera is positioned to view the entire mechanism.
