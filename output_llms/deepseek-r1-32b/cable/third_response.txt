import pychrono.core as chrono
import pychrono.fea as fea
import pychrono.irrlicht as chronoirr

# ----------------------------------------------------------------------------
# Model1: Multiple beams composed of ANCF cable elements, with one end hinged to
# ground, moving under gravity alone.
# ----------------------------------------------------------------------------

class Model1:
    def __init__(self, system, mesh, n_chains=6):
        self.system = system
        self.mesh = mesh
        self.n_chains = n_chains
        self.end_bodies = []

        # Create sections for the beams
        msection_cable = fea.ChBeamSectionCable()
        msection_cable.SetDiameter(0.015)
        msection_cable.SetYoungModulus(0.01e9)
        msection_cable.SetRayleighDamping(0.0001)

        # Create a builder for ANCF beams
        builder = fea.ChBuilderCableANCF()

        # Angle between each chain
        angle_step = 2 * chrono.CH_C_PI / n_chains

        for i in range(n_chains):
            # Calculate position for each chain
            theta = i * angle_step
            start_pos = chrono.ChVector3d(0.2 * chrono.ChCos(theta), 0.2 * chrono.ChSin(theta), -0.1)
            end_pos = start_pos + chrono.ChVector3d(0.5 * chrono.ChCos(theta), 0.5 * chrono.ChSin(theta), 0)

            # Create a new truss body for each chain
            mtruss = chrono.ChBody()
            mtruss.SetFixed(True)
            system.Add(mtruss)

            # Build the beam for this chain
            num_elements = 10 + i  # Increase number of elements per chain
            builder.BuildBeam(
                mesh,
                msection_cable,
                num_elements,
                start_pos,
                end_pos
            )

            # Apply forces to the front node
            builder.GetLastBeamNodes().front().SetForce(chrono.ChVector3d(0, -0.7, 0))

            # Create hinge constraint between beam end and truss
            constraint_hinge = fea.ChLinkNodeFrame()
            constraint_hinge.Initialize(builder.GetLastBeamNodes().back(), mtruss)
            system.Add(constraint_hinge)

            # Create and attach a box body to the beam end
            box = chrono.ChBody()
            box.SetBodyType(chrono.ChBodyType.BODY_TYPE_RIGID)
            box.SetMass(1.0)
            box.SetPos(end_pos)
            box_shape = chrono.ChBoxGeometry(0.05, 0.05, 0.05)
            box.AddGeometry(box_shape)
            chrono.ChCollisionModel.SetMaterialProperties(box.GetCollisionGeometry(), 10000, 0.3)
            system.Add(box)

            # Add constraint between beam end and box
            constraint = fea.ChLinkNodeFrame()
            constraint.Initialize(builder.GetLastBeamNodes().back(), box)
            system.Add(constraint)

            # Add visualization for the box
            box_vis = chronoirr.ChVisualShape()
            box_vis.SetGeometry(box_shape)
            box_vis.SetColor(chrono.ChColor(0.5, 0.5, 0.5))
            box.AddVisualShape(box_vis)

            self.end_bodies.append(box)

            # Reset builder for next chain
            builder.Reset()

        # Add visualization for all beams
        visualizebeamA = chrono.ChVisualShapeFEA(mesh)
        visualizebeamA.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_ELEM_BEAM_MZ)
        visualizebeamA.SetColorscaleMinMax(-0.4, 0.4)
        visualizebeamA.SetSmoothFaces(True)
        visualizebeamA.SetWireframe(False)
        mesh.AddVisualShapeFEA(visualizebeamA)

        visualizebeamB = chrono.ChVisualShapeFEA(mesh)
        visualizebeamB.SetFEMglyphType(chrono.ChVisualShapeFEA.GlyphType_NODE_DOT_POS)
        visualizebeamB.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_NONE)
        visualizebeamB.SetSymbolsThickness(0.006)
        visualizebeamB.SetSymbolsScale(0.01)
        visualizebeamB.SetZbufferHide(False)
        mesh.AddVisualShapeFEA(visualizebeamB)

    def PrintBodyPositions(self):
        for i, body in enumerate(self.end_bodies):
            pos = body.GetPos()
            print(f"Chain {i+1} end position: ({pos.x}, {pos.y}, {pos.z})")

# Initialize the physical system and mesh container
sys = chrono.ChSystemSMC()
mesh = fea.ChMesh()

# Create the model with multiple chains
model = Model1(sys, mesh, n_chains=6)
sys.Add(mesh)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('FEA cables with multiple chains')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 0.6, -1))
vis.AddTypicalLights()

# Set solver and timestepper
solver = chrono.ChSolverMINRES()
if solver.GetType() == chrono.ChSolver.Type_MINRES:
    print("Using MINRES solver")
    sys.SetSolver(solver)
    solver.SetMaxIterations(200)
    solver.SetTolerance(1e-10)
    solver.EnableDiagonalPreconditioner(True)
    solver.EnableWarmStart(True)
    solver.SetVerbose(False)

ts = chrono.ChTimestepperEulerImplicitLinearized(sys)
sys.SetTimestepper(ts)

# Simulation loop
step_counter = 0
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(0.01)
    
    # Print positions every 100 steps
    if step_counter % 100 == 0:
        model.PrintBodyPositions()
