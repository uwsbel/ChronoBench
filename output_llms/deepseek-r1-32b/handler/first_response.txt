Here's a PyChrono simulation script integrated with ROS based on your requirements:

```python
importchrono
import rospy
from std_msgs.msg import Int32

# Initialize PyChrono system
def initialize_chrono():
    # Create Chrono system with gravity
    system = chrono.ChSystemNSC()
    system.Set_Gravity(chrono.ChVectorD(0, -9.81, 0))
    
    # Create fixed floor
    floor = chrono.ChGroundPlane()
    floor.SetNormal(chrono.ChVectorD(0, 1, 0))
    floor.SetHeight(0)
    system.Add(floor)
    
    # Create movable box
    box_dims = chrono.ChVectorD(0.5, 0.5, 0.5)
    box = chrono.ChBody()
    box.SetBodyType(chrono.CHBODY_TYPE_RIGID)
    box.SetPos(chrono.ChVectorD(0, 1, 0))
    box.SetMass(1)
    box.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
    
    box_shape = chrono.ChBoxShape()
    box_shape.SetDimensions(box_dims)
    box.AddVisualShape(box_shape)
    
    system.Add(box)
    
    return system, box

# ROS handler class
class ROSHandler(chrono.Ch_rosHandler):
    def __init__(self):
        super(ROSHandler, self).__init__()
        self.publisher = rospy.Publisher('chrono_ros_topic', Int32, queue_size=10)
        self.count = 0
        
    def OnROS.Publish(self):
        msg = Int32()
        msg.data = self.count
        self.publisher.publish(msg)
        self.count += 1

# Main simulation function
def main():
    # Initialize ROS node
    rospy.init_node('chrono_ros_simulation', anonymous=True)
    
    # Create Chrono system and objects
    system, box = initialize_chrono()
    
    # Create ROS manager
    ros_manager = chrono.ChRosManager()
    ros_manager.SetSystem(system)
    
    # Add ROS handlers
    clock_handler = chrono.ChRosClockHandler()
    body_handler = chrono.ChRosBodyHandler()
    transform_handler = chrono.ChRosTransformHandler()
    
    custom_handler = ROSHandler()
    
    ros_manager.RegisterHandler(clock_handler)
    ros_manager.RegisterHandler(body_handler)
    ros_manager.RegisterHandler(transform_handler)
    ros_manager.RegisterHandler(custom_handler)
    
    # Run simulation
    step_count = 1000
    time_step = 0.001
    
    for step in range(step_count):
        system.DoStepFrame(time_step)
        ros_manager.Update()
        
        # Sleep to maintain real-time
        rospy.sleep(time_step)
        
        # Publish custom message
        custom_handler.OnROS.Publish()

if __name__ == "__main__":
    main()
```

This script:
1. Initializes a PyChrono system with gravitational acceleration
2. Creates a fixed floor and a movable box
3. Sets up ROS communication with a custom handler
4. Runs a simulation loop with ROS integration

To run this simulation:

2. Set up your ROS environment

The simulation will publish integer messages to the 'chrono_ros_topic' topic while maintaining real-time execution through ROS timing.
