```python
importchrono
import chronoirr
import chvis

# Initialize simulation settings
def main():
    # Create Chrono::Engine system
    sys = chrono.ChSystemDFS()
    sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))
    
    # Set up visualization with Irrlicht
    vis = chronoirr.ChIrrApp(sys, "Gator Vehicle Simulation", chronoirr.dimension2du(1024, 768))
    vis.SetCameraPosition(chrono.ChVectorD(5, 5, 5))
    vis.SetCameraRotation(chrono.ChVectorD(45, 45, 0))
    vis.AddLight(chrono.ChVectorD(5, 5, 5), chrono.ChColor(1, 1, 1))
    
    # Create terrain
    terrain = chrono.ChRigidTerrain(sys)
    terrain.SetDimensions(100, 100)  # X and Z dimensions
    terrain.SetHeight(0)  # Flat terrain
    terrain.SetTexture("terrain_texture.jpg")  # Add your custom texture
    sys.Add(terrain)
    
    # Create Gator vehicle
    vehicle = chrono.ChVehicleGator(sys)
    vehicle.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 1)))  # Position above terrain
    vehicle.SetInitRotation(chrono.ChQuaternionD(1, 0, 0, 0))  # Default orientation
    vehicle.SetContactMethod(chrono.ChContactMethodTypeEnum.NSM)  # Use default contact method
    vehicle.SetTireModelType(chrono.ChTireModelType.TMEASY)  # Use TMEASY tire model
    
    # Enable mesh visualization for all vehicle components
    vehicle.SetRenderType(chrono.ChVisualizationType.MESH)
    
    # Add vehicle to system
    sys.Add(vehicle)
    
    # Create driver system
    driver = chrono.ChDriver()
    driver.SetSteering(0)  # Initial steering
    driver.SetThrottle(0)  # Initial throttle
    driver.SetBraking(0)  # Initial braking
    
    # Add driver to vehicle
    vehicle.AddDriver(driver)
    
    # Set up simulation loop
    simulation_time = 60  # Total simulation time in seconds
    frame_count = 0
    while vis.Run() and frame_count < simulation_time * 50:  # 50 FPS
        # Update vehicle
        vehicle.Update()
        
        # Handle user input
        if vis.KeyHit(chronoirr.KEY_ESCAPE):
            break
        
        # Update simulation
        sys.DoStepDynamics(1/50)  # Timestep for 50 FPS
        
        # Render scene
        vis.Render()
        frame_count += 1
    
    # Close simulation
    vis.Close()

if __name__ == "__main__":
    main()
```
