```python
importchrono
import random
import math
from chrono import ChVector, ChQuaternion, ChSystem, ChConstraintSolver, ChIrrApp, ChIrrGui, ChIrrRenderer
from chrono.core import ChSystem, ChConstraintSolver, ChIrrApp, ChIrrGui, ChIrrRenderer
from chrono.physics import ChBody, ChBodyEasySphere, ChBodyEasyBox, ChBodyEasyCylinder
from chrono.visual import ChVisualization, ChVisualShape, ChVisualSphere, ChVisualBox, ChVisualCylinder

# Initialize the Chrono system
system = ChSystem()
system.SetSolver(ChConstraintSolver())
system.SetTimestep(0.005)

# Create a particle emitter function
def create_particle(system, x, y, z):
    # Randomly choose between sphere, box, or cylinder
    shape_type = random.choice(['sphere', 'box', 'cylinder'])
    
    # Random mass between 1 and 10
    mass = random.uniform(1, 10)
    
    # Random dimensions
    size = random.uniform(0.1, 0.5)
    
    # Create the particle
    if shape_type == 'sphere':
        particle = ChBodyEasySphere(mass, size, True)
    elif shape_type == 'box':
        particle = ChBodyEasyBox(mass, size, size, size, True)
    elif shape_type == 'cylinder':
        particle = ChBodyEasyCylinder(mass, size, size, True)
    
    # Random position
    particle.SetPos(ChVector(x + random.uniform(-1, 1), y + random.uniform(-1, 1), z + random.uniform(-1, 1)))
    
    # Random velocity
    velocity = ChVector(random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1))
    particle.SetPos_dt(velocity)
    
    # Random orientation
    particle.SetRot(ChQuaternion(random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1)).Normalize())
    
    # Add particle to system
    system.Add(particle)
    return particle

# Create particles
particles = []
for _ in range(50):
    particles.append(create_particle(system, 0, 0, 0))

# Set up Irrlicht visualization
app = ChIrrApp(system, "Particle Gravitation Simulation", ChVector(800, 600, 0))
app.SetCameraPosition(ChVector(10, -10, 10))
app.SetCameraRotation(ChVector(45, 45, 0))
app.AddTypicalLights()

# Add a grid for better visualization
grid = ChIrrRenderer()
grid.AddGrid(20, 20, 1)
app.AddRenderer(grid)

# Simulation loop
while app.Run():
    # Calculate gravitational forces between particles
    for i in range(len(particles)):
        for j in range(i+1, len(particles)):
            p1 = particles[i]
            p2 = particles[j]
            
            # Calculate distance between particles
            dx = p2.GetPos().x - p1.GetPos().x
            dy = p2.GetPos().y - p1.GetPos().y
            dz = p2.GetPos().z - p1.GetPos().z
            distance = math.sqrt(dx**2 + dy**2 + dz**2)
            
            if distance > 0:
                # Calculate gravitational force
                G = 1000  # Gravitational constant
                force = G * p1.GetMass() * p2.GetMass() / (distance**2)
                
                # Normalize direction
                dir = ChVector(dx, dy, dz) / distance
                
                # Apply forces
                p1.AddForce(force * dir)
                p2.AddForce(-force * dir)
    
    # Update the system
    system.DoStepDynamics()
```
