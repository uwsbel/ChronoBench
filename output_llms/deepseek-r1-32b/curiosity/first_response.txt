Here's a PyChrono simulation script for the Curiosity rover on a rigid terrain:

```python
importchrono
import math
import time

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the ground
ground = chrono.ChBody()
ground.SetMass(0)
ground.SetPos(chrono.ChVectorD(0, 0, 0))
ground.GetCollisionModel().AddPlane(chrono.ChVectorD(0, 1, 0), chrono.ChVectorD(0, 0, 0))
ground.GetCollisionModel().SetMaterialSurface(chrono.ChMaterialSurface())
ground.GetVisualShape().SetTexture(chrono.GetChronoDataPath() + 'textures/ground.jpg')
system.Add(ground)

# Create the rover
rover = chrono.ChBody()
rover.SetMass(200)
rover.SetPos(chrono.ChVectorD(0, 0.5, 0))
rover.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
rover.GetCollisionModel().AddBox(chrono.ChVectorD(2, 1, 0.5))
rover.GetCollisionModel().SetMaterialSurface(chrono.ChMaterialSurface())
rover.GetVisualShape().SetTexture(chrono.GetChronoDataPath() + 'textures/mars_rover.png')
system.Add(rover)

# Create wheels
wheel_radius = 0.5
wheel_width = 0.3
wheel_mass = 10

# Front left wheel
fl_wheel = chrono.ChBody()
fl_wheel.SetMass(wheel_mass)
fl_wheel.SetPos(chrono.ChVectorD(-0.8, -0.4, 0))
fl_wheel.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
fl_wheel.GetCollisionModel().AddCylinder(wheel_radius, wheel_width)
fl_wheel.GetCollisionModel().SetMaterialSurface(chrono.ChMaterialSurface())
fl_wheel.GetVisualShape().SetTexture(chrono.GetChronoDataPath() + 'textures/wheel.png')
system.Add(fl_wheel)

# Front right wheel
fr_wheel = chrono.ChBody()
fr_wheel.SetMass(wheel_mass)
fr_wheel.SetPos(chrono.ChVectorD(0.8, -0.4, 0))
fr_wheel.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
fr_wheel.GetCollisionModel().AddCylinder(wheel_radius, wheel_width)
fr_wheel.GetCollisionModel().SetMaterialSurface(chrono.ChMaterialSurface())
fr_wheel.GetVisualShape().SetTexture(chrono.GetChronoDataPath() + 'textures/wheel.png')
system.Add(fr_wheel)

# Rear left wheel
rl_wheel = chrono.ChBody()
rl_wheel.SetMass(wheel_mass)
rl_wheel.SetPos(chrono.ChVectorD(-0.8, 0.4, 0))
rl_wheel.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
rl_wheel.GetCollisionModel().AddCylinder(wheel_radius, wheel_width)
rl_wheel.GetCollisionModel().SetMaterialSurface(chrono.ChMaterialSurface())
rl_wheel.GetVisualShape().SetTexture(chrono.GetChronoDataPath() + 'textures/wheel.png')
system.Add(rl_wheel)

# Rear right wheel
rr_wheel = chrono.ChBody()
rr_wheel.SetMass(wheel_mass)
rr_wheel.SetPos(chrono.ChVectorD(0.8, 0.4, 0))
rr_wheel.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
rr_wheel.GetCollisionModel().AddCylinder(wheel_radius, wheel_width)
rr_wheel.GetCollisionModel().SetMaterialSurface(chrono.ChMaterialSurface())
rr_wheel.GetVisualShape().SetTexture(chrono.GetChronoDataPath() + 'textures/wheel.png')
system.Add(rr_wheel)

# Create motor controls
motor_fl = chrono.ChMotorRotation()
motor_fl.SetName("fl_motor")
motor_fl.SetMotorType(chrono.ChMotorRotation.MOTOR_TYPE_ROTVEL_DEM)
motor_fl.SetRotSpeed(0)
motor_fl.SetTorqueLimit(100)
motor_fl.Initialize(rover, fl_wheel, chrono.ChFrameD(chrono.ChVectorD(-0.8, -0.4, 0), chrono.ChQuaternionD(1, 0, 0, 0)))
system.Add(motor_fl)

motor_fr = chrono.ChMotorRotation()
motor_fr.SetName("fr_motor")
motor_fr.SetMotorType(chrono.ChMotorRotation.MOTOR_TYPE_ROTVEL_DEM)
motor_fr.SetRotSpeed(0)
motor_fr.SetTorqueLimit(100)
motor_fr.Initialize(rover, fr_wheel, chrono.ChFrameD(chrono.ChVectorD(0.8, -0.4, 0), chrono.ChQuaternionD(1, 0, 0, 0)))
system.Add(motor_fr)

motor_rl = chrono.ChMotorRotation()
motor_rl.SetName("rl_motor")
motor_rl.SetMotorType(chrono.ChMotorRotation.MOTOR_TYPE_ROTVEL_DEM)
motor_rl.SetRotSpeed(0)
motor_rl.SetTorqueLimit(100)
motor_rl.Initialize(rover, rl_wheel, chrono.ChFrameD(chrono.ChVectorD(-0.8, 0.4, 0), chrono.ChQuaternionD(1, 0, 0, 0)))
system.Add(motor_rl)

motor_rr = chrono.ChMotorRotation()
motor_rr.SetName("rr_motor")
motor_rr.SetMotorType(chrono.ChMotorRotation.MOTOR_TYPE_ROTVEL_DEM)
motor_rr.SetRotSpeed(0)
motor_rr.SetTorqueLimit(100)
motor_rr.Initialize(rover, rr_wheel, chrono.ChFrameD(chrono.ChVectorD(0.8, 0.4, 0), chrono.ChQuaternionD(1, 0, 0, 0)))
system.Add(motor_rr)

# Visualization setup
vis = chrono.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowTitle("Curiosity Rover Simulation")
vis.SetWindowSize(1024, 768)
vis.SetNumberOfRenderFrames(100)
vis.SetLightDirection(chrono.ChVectorD(0.5, -0.5, 1))
vis.SetLightIntensity(1)
vis.AddLight(chrono.ChVectorD(0, 10, 0), chrono.ChColor(1, 1, 1), 100)
vis.AddSky()
vis.SetCameraPosition(chrono.ChVectorD(5, 5, 5))
vis.SetCameraRotation(chrono.ChQuaternionD(1, 0, 0, 0))

# Simulation parameters
simulation_time = 10
step_size = 0.001
render_step = 0.01

# Simulation loop
time = 0
while time < simulation_time:
    # Update system
    system.DoStepDynamics(step_size)
    
    # Update visualization
    if (time % render_step) < step_size:
        vis.Render()
    
    # Update motor controls
    motor_fl.SetRotSpeed(10)
    motor_fr.SetRotSpeed(10)
    motor_rl.SetRotSpeed(10)
    motor_rr.SetRotSpeed(10)
    
    # Print status
    print(f"Time: {time:.2f}s")
    
    time += step_size

# Close visualization
vis.Close()
```
