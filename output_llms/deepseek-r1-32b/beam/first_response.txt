I'll help you create a PyChrono simulation demonstrating beam finite elements. Here's a complete script that creates a simple beam structure with visualization using the Irrlicht engine:

```python
importchrono
importchrono.visualization

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Gravity

# Create visualization application
vis = chrono.visualization.ChIrrApp(system, "Beam Finite Elements Demo", 1024, 768)
vis.SetCamera(chrono.ChVectorD(0, 5, 5), chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()

# Create ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
ground.SetMass(0)
ground.AddVisualShape(chrono.ChBoxShape(10, 1, 10), chrono.ChColor(0.2, 0.2, 0.2))
system.Add(ground)

# Create beam nodes
nodes = []
for i in range(4):
    node = chrono.ChBody()
    node.SetMass(0.1)
    node.AddVisualShape(chrono.ChSphereShape(0.1), chrono.ChColor(0.5, 0.5, 0.5))
    if i == 0:
        node.SetPos(chrono.ChVectorD(-1, 0, 0))
    elif i == 1:
        node.SetPos(chrono.ChVectorD(1, 0, 0))
    elif i == 2:
        node.SetPos(chrono.ChVectorD(1, 0, 2))
    else:
        node.SetPos(chrono.ChVectorD(-1, 0, 2))
    system.Add(node)
    nodes.append(node)

# Create beam elements
for i in range(4):
    beam = chrono.ChBeam()
    beam.SetNodes(nodes[i], nodes[(i+1)%4])
    beam.SetYoungModulus(2e5)
    beam.SetPoissonRatio(0.3)
    beam.SetArea(0.01)
    beam.SetMoment(0.0001)
    rotation = chrono.ChMatrix33D()
    rotation.Set_A_XY(chrono.CH_C_PI / 2)
    beam.SetRotation(rotation)
    system.Add(beam)

# Add coordinate system for visualization
coordsys = chrono.ChCoordsys()
coordsys.SetPos(chrono.ChVectorD(0, 0, 0))
coordsys.SetRot(chrono.ChMatrix33D())
vis.AddAsset(coordsys)

# Create meshes for visualization
for i in range(4):
    mesh = chrono.ChMesh()
    mesh.AddVertex(nodes[i].GetPos())
    mesh.AddVertex(nodes[(i+1)%4].GetPos())
    mesh.AddFace(0, 1, 2)
    mesh.SetColor(chrono.ChColor(0.8, 0.8, 0.8))
    vis.AddAsset(mesh)

# Run simulation
for _ in range(1000):
    system.DoStepDynamics(0.01)
    vis.Render()
    if vis.GetDevice().IsKeyDown(chrono.CH_KEY_ESCAPE):
        break

vis.Close()
```
