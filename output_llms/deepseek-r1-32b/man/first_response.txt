Here's a PyChrono simulation script that implements the requested MAN 10t truck dynamics simulation with the specified features:

```python
import pychrono.core as chrono
import pychrono.postprocess as postprocess
import pychrono.irrlicht as chronoirr
from pychrono.vehicle import *
from pychrono.vehicle.tire import *

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the vehicle
vehicle = Man10tTruck()
vehicle.SetChassisVisualizationType(ChassisVisualizationType.TEXTURE)
vehicle.SetChassisCollisionType(ChassisCollisionType.BOX)
vehicle.SetTireType(TireType.TMEASY)
vehicle.Initialize()

# Create the terrain
terrain = ChRigidTerrain(system)
terrain.SetSize(1000, 1000)
terrain.SetHeight(0)
terrain.SetTexture(chrono.GetChronoDataFile("textures/grass.jpg"))
terrain.SetTextureScale(10, 10)
terrain.SetContactMaterial(vehicle.GetChassisMaterial())
terrain.Initialize()

# Create the Irrlicht application
application = chronoirr.ChIrrApp(system, "MAN 10t Truck Simulation", chronoirr.dimension2du(1280, 720))
application.AddTypicalSky()
application.AddTypicalLights()
application.AddLightWithShadow(chrono.ChVectorD(0, 100, 100), chrono.ChVectorD(0, 0, 0), 200, 1000, 100000, 100000)
application.SetCameraPosition(chrono.ChVectorD(0, 5, 10))
application.SetCameraRotation(chrono.ChQuaternionD(1, 0, 0, 0))

# Add a chase camera
chase_camera = chronoirr.ChIrrChaseCamera()
chase_camera.SetDistance(10)
chase_camera.SetRotation(chrono.ChQuaternionD(1, 0, 0, 0))
chase_camera.SetPositionOffset(chrono.ChVectorD(0, 2, 0))
application.AddChaseCamera(chase_camera)

# Add a directional light
dir_light = chronoirr.ChIrrLight()
dir_light.SetDirection(chrono.ChVectorD(0, -1, 0))
dir_light.SetIntensity(1)
application.AddLight(dir_light)

# Add a skybox
skybox = chronoirr.ChIrrSkyBox()
skybox.SetClouds(chrono.GetChronoDataFile("skybox/clouds.jpg"))
skybox.SetDistance(1000)
application.AddSkyBox(skybox)

# Add the vehicle and terrain to the application
application.AddDrawable(vehicle.GetChassis())
application.AddDrawable(terrain.GetGround())

# Create driver inputs
driver = ChDriver()
driver.SetThrottle(0.5)
driver.SetSteering(0)
driver.SetBraking(0)

# Simulation loop
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    driver.Update()
    vehicle.Update(driver)
    application.EndScene()

# Cleanup
application.GetDevice().close()
```
