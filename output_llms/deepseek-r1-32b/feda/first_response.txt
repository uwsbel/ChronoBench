```python
importchrono
import chronoirr
importchrono.vehicle
importchrono.geometry
importchrono.sensor
importchrono.visualization

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the Irrlicht application
app = chronoirr.ChIrrApp(system, "FEDA Vehicle Simulation", chronoirr.dimension2du(1024, 768))
app.SetCameraPosition(chrono.ChVectorf(0, 5, 10))
app.SetCameraRotation(chrono.ChQuaternion(0.7071, 0, 0.7071, 0))

# Create the vehicle system
vehicle_system = chrono.vehicle.ChVehicleSystem()
vehicle_system.SetSystem(system)

# Vehicle parameters
vehicle_pos = chrono.ChVectorD(0, 0, 1)
vehicle_rot = chrono.ChQuaternionD(1, 0, 0, 0)
contact_method = chrono.ChContactMethodType.NSC
tire_model = chrono.vehicle.ChTireModelType.TMEASY

# Create the FEDA vehicle
vehicle = chrono.vehicle.fedagazelle.ChFEDAGazelle()
vehicle.SetContactMethod(contact_method)
vehicle.SetTireType(tire_model)
vehicle.SetInitPosition(vehicle_pos, vehicle_rot)
vehicle_system.Add(vehicle)

# Set visualization for all vehicle parts
for body in vehicle.GetVehicle().GetAllCollisionModels():
    body.SetVisualizationType(chrono.ChVisualizationType.MESH)

# Create rigid terrain
terrain = chrono.ChRigidTerrain(system)
terrain.SetSize(100, 100)
terrain.SetHeight(0)
terrain.SetTexture(chrono.ChTexture())
terrain.GetMaterial().SetYoungModulus(1e8)
terrain.GetMaterial().SetPoissonRatio(0.3)
terrain.SetIdentifier("terrain")
app.GetSceneManager().AddMaterial(terrain.GetMaterial())

# Set up camera to follow the vehicle
camera = app.GetCamera()
camera.SetPosition(chrono.ChVectorf(0, 5, 10))
camera.SetRotation(chrono.ChQuaternion(0.7071, 0, 0.7071, 0))

# Driver system
driver = chrono.vehicle.ChDriver()
driver.SetSteering(0)
driver.SetThrottle(0)
driver.SetBraking(0)

# Interactive controls
def on_key_press(event):
    if event.Key == ord('W'):
        driver.SetThrottle(0.5)
    elif event.Key == ord('S'):
        driver.SetBraking(0.5)
    elif event.Key == ord('A'):
        driver.SetSteering(-0.5)
    elif event.Key == ord('D'):
        driver.SetSteering(0.5)

def on_key_release(event):
    if event.Key in [ord('W'), ord('S'), ord('A'), ord('D')]:
        driver.SetThrottle(0)
        driver.SetBraking(0)
        driver.SetSteering(0)

app.SetEventCallback(on_key_press, chronoirr.KEY_PRESS)
app.SetEventCallback(on_key_release, chronoirr.KEY_RELEASE)

# Simulation loop
fps = 50
step = 1.0 / fps

while app.Run():
    # Update vehicle dynamics
    vehicle_system.Update(step)
    
    # Update driver inputs
    driver.Update(step)
    
    # Update camera position
    camera.SetPosition(vehicle.GetVehicle().GetChassis().GetPos() + chrono.ChVectorf(0, 5, 10))
    
    # Render scene
    app.Render()
    
    # Handle events
    app.HandleEvents()
```
