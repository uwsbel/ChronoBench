Here's a PyChrono simulation script that implements the specified Turtlebot simulation:

```python
importchrono
import math

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.SetGravity(chrono.ChVectorD(0, -9.81, 0))

# Create ground body
ground = chrono.ChBody()
ground.SetMass(0)
ground.SetPos(chrono.ChVectorD(0, 0, 0))
ground.GetCollisionModel().AddBox(chrono.ChCollisionModel(), 10, 10, 10)
ground.GetVisualShape().AddBox(chrono.ChVisualMaterial(), 10, 10, 10)
system.Add(ground)

# Create Turtlebot body
turtlebot = chrono.ChBody()
turtlebot.SetMass(1.0)
turtlebot.SetPos(chrono.ChVectorD(0, 1, 0))
turtlebot.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
system.Add(turtlebot)

# Create Turtlebot wheels
wheel_left = chrono.ChBody()
wheel_left.SetMass(0.1)
wheel_left.SetPos(chrono.ChVectorD(-0.15, 0.05, 0))
wheel_left.GetCollisionModel().AddCylinder(chrono.ChCollisionModel(), 0.05, 0.15)
wheel_left.GetVisualShape().AddCylinder(chrono.ChVisualMaterial(), 0.05, 0.15)
turtlebot.Add(wheel_left)

wheel_right = chrono.ChBody()
wheel_right.SetMass(0.1)
wheel_right.SetPos(chrono.ChVectorD(0.15, 0.05, 0))
wheel_right.GetCollisionModel().AddCylinder(chrono.ChCollisionModel(), 0.05, 0.15)
wheel_right.GetVisualShape().AddCylinder(chrono.ChVisualMaterial(), 0.05, 0.15)
turtlebot.Add(wheel_right)

# Create motor for wheels
motor_left = chrono.ChMotorRotation()
motor_left.SetName("left_motor")
motor_left.SetLimitMin(-10)
motor_left.SetLimitMax(10)
motor_left.Initialize(wheel_left, chrono.ChFrameD(chrono.ChVectorD(-0.15, 0.05, 0), chrono.ChQuaternionD(1, 0, 0, 0)), True)
system.Add(motor_left)

motor_right = chrono.ChMotorRotation()
motor_right.SetName("right_motor")
motor_right.SetLimitMin(-10)
motor_right.SetLimitMax(10)
motor_right.Initialize(wheel_right, chrono.ChFrameD(chrono.ChVectorD(0.15, 0.05, 0), chrono.ChQuaternionD(1, 0, 0, 0)), True)
system.Add(motor_right)

# Create Irrlicht application
app = chrono.ChIrrApp(system, "Turtlebot Simulation", 640, 480)
app.AddTypicalLights()
app.AddTypicalCamera(chrono.ChVectorD(0, 5, 5), chrono.ChVectorD(0, 0, 0))
app.SetCameraRotation(chrono.ChQuaternionD(1, 0, 0, 0))

# Simulation parameters
timestep = 0.001
time = 0
duration = 10

# Simulation loop
while time < duration:
    # Update motor velocities based on time
    if int(time) % 2 == 0:
        # Turn left
        motor_left.SetDesiredSpeed(-1)
        motor_right.SetDesiredSpeed(1)
    else:
        # Turn right
        motor_left.SetDesiredSpeed(1)
        motor_right.SetDesiredSpeed(-1)
    
    # Update system
    system.DoStepDynamics(timestep)
    time += timestep
    
    # Render
    app.BeginScene()
    app.DrawAll()
    app.EndScene()
    
    # Check for exit
    if app.GetDevice().GetKey(chrono.CH_KEY_ESCAPE):
        break

# Close application
app.Close()
```
