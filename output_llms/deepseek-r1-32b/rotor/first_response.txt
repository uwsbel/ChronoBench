To create a PyChrono simulation of a Jeffcott rotor using an IGA beam, follow these steps:

1. **Initialize the Environment**: Import necessary modules and create the Chrono system with gravity.

2. **Set Up Visualization**: Create an Irrlicht application and FEM visualization plugin.

3. **Create the Beam**: Define beam properties and create an IGA beam element, adding it to a body in the simulation.

4. **Attach the Flywheel**: Create a body for the flywheel, set its properties, and position it at the beam's center.

5. **Add Rotational Motor**: Create a motor to drive one end of the beam, linking it between the beam and ground.

6. **Apply Constraints**: Fix one end of the beam and allow the other end to roll.

7. **Compile and Run Simulation**: Set up the FEM visualization and run the simulation with the Irrlicht window.

Here's the code implementing these steps:

```python
importchrono
importchrono.fea
importchrono.visual

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Visualization setup
application = chrono.ChIrrApp(system, "Jeffcott Rotor Simulation", chronoirr.dimension2D(800, 600))
application.AddTypicalLights()
application.AddTypicalCamera(chrono.ChVectorD(0, 5, 5), chrono.ChVectorD(0, 0, 0))

# FEM visualization
fem_vis = chrono.ChFEMVisualization()
application.GetSystem().SetFEMVisualization(fem_vis)

# Beam properties
beam_length = 2.0
beam_area = 0.01
beam_inertia = 0.001
density = 1000.0
young_modulus = 2e9
damping = 0.1

# Create IGA beam
beam = chrono.ChIgaBeamElement()
beam.SetNodes(4)
beam.SetLength(beam_length)
beam.SetArea(beam_area)
beam.SetInertia(beam_inertia)
beam.SetDensity(density)
beam.SetYoungModulus(young_modulus)
beam.SetDamping(damping)

# Beam body
beam_body = chrono.ChBody()
beam_body.SetPos(chrono.ChVectorD(0, 0, 0))
beam_body.AddAsset(chrono.ChColorAsset(0.8, 0.8, 0.8))
system.Add(beam_body)
beam_body.AddElement(beam)

# Flywheel
flywheel_mass = 10.0
flywheel_inertia = chrono.ChMatrix33D(1, 0, 0, 0, 1, 0, 0, 0, 1)

flywheel_body = chrono.ChBody()
flywheel_body.SetPos(chrono.ChVectorD(beam_length / 2, 0, 0))
flywheel_body.SetMass(flywheel_mass)
flywheel_body.SetInertia(flywheel_inertia)
flywheel_body.AddAsset(chrono.ChColorAsset(0.5, 0.5, 0.5))
system.Add(flywheel_body)

# Rotational motor
motor = chrono.ChRotationalMotor()
motor.SetTorque(10.0)
motor.SetAngularVelocity(2 * chrono.CH_PI)
motor.SetLink(beam_body, system.GetGround())
system.Add(motor)

# Constraints
constraint_fixed = chrono.ChConstraintLock(beam_body, system.GetGround())
constraint_fixed.SetLockLinX(True)
constraint_fixed.SetLockLinY(True)
constraint_fixed.SetLockLinZ(True)
constraint_fixed.SetLockRotX(True)
constraint_fixed.SetLockRotY(True)
constraint_fixed.SetLockRotZ(True)
system.Add(constraint_fixed)

constraint_roller = chrono.ChConstraintRolling()
constraint_roller.SetBodyA(beam_body)
constraint_roller.SetBodyB(system.GetGround())
constraint_roller.SetPoint(chrono.ChVectorD(beam_length, 0, 0))
constraint_roller.SetNormal(chrono.ChVectorD(0, 1, 0))
system.Add(constraint_roller)

# Compile FEM visualization
fem_vis.Compile(beam)

# Run simulation
application.SetTimestep(0.001)
application.Run()
```

This code sets up a simulation where a Jeffcott rotor with an IGA beam and flywheel is driven by a rotational motor, visualized using FEM and Irrlicht.
