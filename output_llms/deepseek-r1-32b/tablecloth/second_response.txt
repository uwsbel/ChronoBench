nsections_x, nsections_z = 40, 40

mynodes = []
for iz in range(nsections_z + 1):

        mesh.AddNode(mnode)
        mynodes.append(mnode)

# Monitoring and loading setup
nodePlotA = None
nodePlotB = None
nodesLoad = []
ref_X = lambda x: x

                           mynodes[iz * (nsections_x + 1) + ix + 1], boundary_1, boundary_2, boundary_3)
