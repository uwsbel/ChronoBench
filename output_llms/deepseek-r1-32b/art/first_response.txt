```python
import pychrono as chrono
import pychrono irrlicht as chronoirr
import pychrono.vehicle as vehicle
import pychrono.sensor as sensor

# Create the Irrlicht application
app = chronoirr.ChIrrApp(chrono.GetChronoSystem(), "ARTcar Simulation", chronoirr.dimension2du(800, 600))
app.SetCamera(chronoirr.vector3df(0, 5, 10), chronoirr.vector3df(0, 0, 0))
app.AddLight(chronoirr.vector3df(0, 5, 10), chronoirr.vector3df(0, 0, 0), 100)

# Create the vehicle
vehicle_system = vehicle.ChVehicleSystem()
vehicle_system.SetContactMethod(chrono.ChContactMethod罚Contacts)

# Initialize vehicle parameters
vehicle_params = vehicle.ChVehicleParams()
vehicle_params.vehicle_type = vehicle.ChVehicleType_ARTC
vehicle_params.init_position = chronoirr.vector3df(0, 0.5, 1)
vehicle_params.init_rotation = chronoirr.ChQuaternionD(chrono.ChRotY(chrono.CH_C_PI / 2))

# Create the vehicle
my_vehicle = vehicle.ChARTCar(vehicle_params)
vehicle_system.AddVehicle(my_vehicle)

# Create the terrain
terrain = vehicle.ChRigidTerrain(vehicle_system.GetSystem())
terrain.SetDimensions(100, 100, 0)  # X, Y, Z dimensions
terrain.SetTexture("terrain_texture.jpg")  # Add custom texture
vehicle_system.AddTerrain(terrain)

# Set up visualization
app.AddTypicalLights()
app.AddAllTypicalCameraTargets()
app.AddAllTypicalLights()

# Set up driver controls
driver = vehicle.ChDriver(my_vehicle.GetVehicle())
driver.SetThrottle(0)
driver.SetSteering(0)

# Set up keyboard controls
def on_key_press(event):
    if event.Key == 'W':
        driver.SetThrottle(0.5)
    elif event.Key == 'S':
        driver.SetThrottle(-0.5)
    elif event.Key == 'A':
        driver.SetSteering(-0.5)
    elif event.Key == 'D':
        driver.SetSteering(0.5)
    elif event.Key == ' ':
        driver.SetBraking(1)

def on_key_release(event):
    if event.Key in ['W', 'S']:
        driver.SetThrottle(0)
    elif event.Key in ['A', 'D']:
        driver.SetSteering(0)
    elif event.Key == ' ':
        driver.SetBraking(0)

app.SetEventCallback(on_key_press, on_key_release)

# Simulation loop
frame_rate = 50
time_step = 1.0 / frame_rate

while app.Run():
    # Update driver inputs
    driver.Update()

    # Update vehicle
    my_vehicle.Update(time_step)

    # Render scene
    app.BeginScene()
    app.DrawAll()
    app.EndScene()

    # Check for exit condition
    if app.GetKeyboard().KeyDown(chronoirr.KEY_ESCAPE):
        break

# Close the application
app.Close()
```
