world.add(ground)

beam_width = 0.1  # Width of the beam in meters
beam_height = 0.5  # Height of the beam in meters
beam_material = chrono.Material(chrono.Elasticity(2e11), chrono.Density(7850))

# Create beam elements
beam_elements = []
for i in range(10):  # Number of elements in the beam

    beam_elements.append(beam_element)

# Create beam object
beam = chrono.Beam(beam_elements)
world.add(beam)

# Define hinge at the ground
hinge_point = chrono.Vector3(0, 0, 0)
hinge_force = chrono.Force(chrono.Vector3(0, 0, 1), 0)
hinge_constraint = chrono.HingeConstraint(beam, hinge_point, hinge_force)
world.add(hinge_constraint)

# Define gravity
gravity = chrono.Vector3(0, -9.81, 0)
world.add(gravity)
