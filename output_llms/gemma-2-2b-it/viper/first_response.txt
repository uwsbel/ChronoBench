```python
import pychrono as chrono
import pychrono.irrlicht as irrlicht
import pychrono.math as math

# Initialize PyChrono
chrono.init()

# Create Chrono system
chrono.create_system("chrono_system")

# Define ground body
ground_body = chrono.create_rigid_body("ground_body", 
                                       chrono.Vector3(0, 0, 0), 
                                       chrono.Vector3(0, 0, 0), 
                                       chrono.Vector3(0, 0, 0), 
                                       chrono.Vector3(0, 0, 0))
ground_body.set_mass(1000)
ground_body.set_inertia(chrono.Vector3(0, 0, 0))

# Create the Viper rover
viper_rover = chrono.create_rigid_body("viper_rover", 
                                       chrono.Vector3(0, 0, 0), 
                                       chrono.Vector3(0, 0, 0), 
                                       chrono.Vector3(0, 0, 0), 
                                       chrono.Vector3(0, 0, 0))
viper_rover.set_mass(100)
viper_rover.set_inertia(chrono.Vector3(0, 0, 0))

# Initialize driver
driver = chrono.create_rigid_body("driver", 
                                  chrono.Vector3(0, 0, 0), 
                                  chrono.Vector3(0, 0, 0), 
                                  chrono.Vector3(0, 0, 0), 
                                  chrono.Vector3(0, 0, 0))
driver.set_mass(1)
driver.set_inertia(chrono.Vector3(0, 0, 0))

# Set initial positions and velocities
ground_body.set_position(chrono.Vector3(0, 0, 0))
viper_rover.set_position(chrono.Vector3(1, 0, 0))
driver.set_position(chrono.Vector3(0, 0, 0))

# Set initial forces and interactions
# (You can add more forces and interactions here)

# Define simulation parameters
dt = 0.01  # Time step
duration = 10  # Simulation duration

# Create Irrlicht visualization
irrlicht_scene = irrlicht.IrrlichtScene()
irrlicht_camera = irrlicht.IrrlichtCamera(irrlicht_scene)
irrlicht_scene.add_camera(irrlicht_camera)

# Set Irrlicht rendering parameters
irrlicht_scene.set_ambient_light(chrono.Vector3(0.2, 0.2, 0.2))
irrlicht_scene.set_light_position(chrono.Vector3(1, 1, 1))

# Set up Irrlicht visualization loop
irrlicht_scene.run()

# Simulation loop
for t in range(int(duration / dt)):
    # Update simulation
    chrono.update_simulation(dt)
    
    # Render the simulation
    irrlicht_scene.render()

    # Update Irrlicht visualization
    irrlicht_scene.update()

# Clean up
chrono.shutdown()
```
