# Initialize PyChrono
chrono.init()

world.add_body(ground)

# Create Turtlebot body
turtlebot = chrono.Body("turtlebot", chrono.Vec3(0, 0, 0))
turtlebot.set_mass(100)  # Set mass of the turtlebot
turtlebot.set_collision_shape(chrono.SphereShape(chrono.Vec3(0.1, 0.1, 0.1)))  # Set collision shape
world.add_body(turtlebot)

# Initialize Turtlebot's position and orientation
turtlebot.set_position(chrono.Vec3(0, 0, 0))
