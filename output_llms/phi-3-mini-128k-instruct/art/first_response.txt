```python

import pychronoid as pc

import irrlicht as ir


# Initialize PyChrono environment and core components

pc.init()

core = pc.core()


# Define vehicle parameters

vehicle_params = {

    'location': (0, 0, 0),

    'orientation': (0, 0, 0),

    'contact_method': 'point',

    'visualization_type': 'mesh'

}


# Create the vehicle object

vehicle = pc.Vehicle(vehicle_params)


# Initialize the RigidTerrain

terrain = pc.RigidTerrain(dimensions=(100, 100, 1))

terrain.addTexture("terrain.jpg")


# Add terrain to the simulation

core.addPhysicsSystem(terrain)


# Create the driver system

driver = pc.DriverSystem(vehicle)

core.addPhysicsSystem(driver)


# Set up simulation loop parameters

frame_rate = 50


# Create the Irrlicht visualization

scene = ir.Scene()

camera = ir.IrrlichtDevice(scene, 'test', 1280, 720, 32, 16, ir.core.dimension.D3D|ir.core.dimension.Z4)

camera.setPosition(0, 10, 50)

camera.setOrientation(0, 0, -0.78)


# Add vehicle to the scene for visualization

vehicle_mesh = ir.Mesh()

vehicle_mesh.load("artcar.obj")

scene.addMesh(vehicle_mesh)


# Start the simulation loop

while not core.isStopped():

    # Update vehicle dynamics

    vehicle.update()


    # Update visualization

    camera.setPosition(vehicle.getLocation()[0], vehicle.getLocation()[1], vehicle.getLocation()[2])

    camera.setOrientation(vehicle.getOrientation()[0], vehicle.getOrientation()[1], vehicle.getOrientation()[2])


    # Render the scene

    scene.drawScene(camera.getVideoDriver())


    # Sleep to maintain the frame rate

    ir.waitChars(frame_rate)


# Stop the simulation

core.stop()

```


This script sets up a PyChrono simulation with an ARTcar vehicle on a rigid terrain, incorporating an interactive driver system and visualizing the simulation using Irrlicht. The simulation loop updates the vehicle's dynamics and visualization in real-time at a frame rate of 50 frames per second.
