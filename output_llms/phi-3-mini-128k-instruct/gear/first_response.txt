```python

import PyChrono

import PyChronoUtils as ch


# Initialize PyChrono environment

pyChrono = PyChrono.ChronoData()


# Add physical systems

pyChrono.AddSystem(PyChrono.System_Gravity)

pyChrono.AddSystem(PyChrono.System_Contact)

pyChrono.AddSystem(PyChrono.System_CoulombFriction)

pyChrono.AddSystem(PyChrono.System_Inertia)

pyChrono.AddSystem(PyChrono.System_Constraint)

pyChrono.AddSystem(PyChrono.System_Collision)


# Add objects

truss = ch.Object_Box(pyChrono, center=(0, 0, 0), size=(0.1, 0.1, 1))

bar = ch.Object_Box(pyChrono, center=(0, 0, 0), size=(1, 0.01, 0.01))

gear1 = ch.Object_Box(pyChrono, center=(0, 0, 0.5), size=(0.05, 0.05, 0.05))

gear2 = ch.Object_Box(pyChrono, center=(0, 0, 1.5), size=(0.05, 0.05, 0.05))

gear_motor = ch.Object_Box(pyChrono, center=(0, 0, 2), size=(0.05, 0.05, 0.05))


# Set default parameters

pyChrono.SetGravity((0, 0, -9.81))


# Set truss material

truss.SetMaterial(PyChrono.Material_Iso, PyChrono.Material_Iso_Elastic, 1e3, 0.3)


# Set bar material

bar.SetMaterial(PyChrono.Material_Iso, PyChrono.Material_Iso_Elastic, 1e3, 0.3)


# Set gear material

gear1.SetMaterial(PyChrono.Material_Iso, PyChrono.Material_Iso_Elastic, 1e3, 0.3)

gear2.SetMaterial(PyChrono.Material_Iso, PyChrono.Material_Iso_Elastic, 1e3, 0.3)


# Set gear motor material

gear_motor.SetMaterial(PyChrono.Material_Iso, PyChrono.Material_Iso_Elastic, 1e3, 0.3)


# Add constraints for gears and motor

pyChrono.AddConstraint(PyChrono.Constraint_Joint_Pulley, gear1, gear2, 0.5)

pyChrono.AddConstraint(PyChrono.Constraint_Joint_Pulley, gear2, gear_motor, 0.5)


# Set motor speed

gear_motor.SetSpeed(20)


# Visualization with Irrlicht

visualization = ch.Visualization_Irrlicht(pyChrono)

visualization.SetWindowTitle("Epicyclic Gears Simulation")

visualization.SetWindowSize(1024, 768)

visualization.SetWindowVisible(True)

visualization.AddLogicalScene(truss, bar, gear1, gear2, gear_motor)


# Set materials and lighting

visualization.SetSceneGraphMaterials(truss, bar, gear1, gear2, gear_motor)

visualization.SetLightPosition(10, 10, 10)

visualization.SetAmbientLight(0.5, 0.5, 0.5)

visualization.SetDiffuseLight(0.5, 0.5, 0.5)

visualization.SetSpecularLight(0.5, 0.5, 0.5)


# Run the simulation

pyChrono.Step(1e-3, 100)


# Keep the window open

visualization.RenderScene()

PyChrono.Application_Get()->Run()

```
