```python

import pychoro as pc

from pychoro.systems import Pendulum

from pychoro.visualization import IrrlichtVisualization


# Initialize PyChrono environment

pc.init()


# Initialize Irrlicht visualization

vis = IrrlichtVisualization(pc.get_default_context())


# Create ground body

ground = pc.create_box(pc.create_scene(), 100, 1, 0.01)


# Create pendulum with mass and inertia

mass = 1.0

length = 1.0

inertia = (mass * length ** 2) / 12

pendulum = Pendulum(pc.create_sphere(pc.create_scene(), mass, inertia), length)


# Create revolute joint between ground and pendulum

ground_body = pc.create_body(ground)

pendulum_body = pc.create_body(pendulum)

joint = pc.create_revolute_joint(ground_body, pendulum_body, 0, 0, 0, 0, 0, 0)


# Set default parameters for the simulation

pc.set_default_gravity(pc.create_vector(0, -9.81, 0))


# Main simulation loop

while not pc.is_quit_requested():

    # Update the simulation

    pc.update_systems()


    # Render the simulation using Irrlicht

    vis.render(pc.get_default_context())


    # Log the pendulum's position and velocity

    print(f"Position: {pendulum_body.get_position()}, Velocity: {pendulum_body.get_velocity()}")


# Clean up

pc.cleanup()

```
