```python

import pychrono as chrono

import pychrono.vehicle as vehicle

import pychrono.core as core

import pychrono.io as chronoio

import pychrono.sensor as sensor

import pychrono.visualization as visualization

import pychrono.analytical_geometry as ag

import pychrono.irrlicht as irrlicht


# Initialize the PyChrono core

core.Initialize()


# Create the simulation environment

my_chrono_instance = core.Chroma()


# Create the terrain

terrain = vehicle.Terrain()

terrain.CreatePolygonalSurface(chrono.Core.Vector3(-100, -100, 0), chrono.Core.Vector3(100, 100, 0), 20)

my_chrono_instance.Add(terrain)


# Create the vehicle (HMMWV)

vehicle_model = vehicle.Vehicle()

vehicle_model.CreateVehicle(chrono.Vehicle.VehicleType_HMMWV)

vehicle_model.Add(terrain)


# Add IMU and GPS sensors to the vehicle

imu = sensor.IMU()

imu.CreateSensor(vehicle_model, core.Vector3(0, 1, 0))

gps = sensor.GPS()

gps.CreateSensor(vehicle_model, core.Vector3(0, 0, 0))

vehicle_model.Add(imu)

vehicle_model.Add(gps)


# Create the visualization system

vis = visualization.ChromaVisualization(my_chrono_instance)

vis.SetWindowTitle('HMMWV Simulation')

vis.SetWindowSize(1920, 1080)


# Setup default parameters

vehicle_model.SetDefaultParameters(mass=10000, gear_ratios=[0.25, 0.17, 0.11, 0.08])

imu.SetDefaultParameters(accelerometer_noise=chrono.Scalar(0.01, 0.01, 0.01), gyroscope_noise=chrono.Scalar(0.001, 0.001, 0.001))

gps.SetDefaultParameters(noise=chrono.Scalar(0.1, 0.1, 0.1))


# Setup the simulation loop

for _ in range(1000):

    # Update sensors

    imu.Update()

    gps.Update()


    # Update vehicle dynamics

    vehicle_model.ComputeDynamics()


    # Update visualization

    vis.BeginScene()

    vis.DrawVehicle(vehicle_model)

    vis.DrawTerrain(terrain)

    vis.DrawSensors(imu, gps)

    vis.EndScene()


    # Synchronize and advance the simulation

    vis.DoStep(0.001, True)


    # Output vehicle mass

    print('Vehicle mass:', vehicle_model.GetMass())


# Cleanup

vis.CloseWindow()

core.Shutdown()

```

Please note that this script assumes you have the PyChrono library installed and properly configured with the necessary dependencies. The vehicle model, sensor types, and parameters should be adapted to your specific use case and available models within the PyChrono library.
