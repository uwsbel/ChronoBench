```python

import PyChrono

import PyIrrlicht as Ir

import numpy as np


# Initialize PyChrono environment

ch = PyChrono.Chrono()

ch.SetGravity(0, -9.81, 0)


# Initialize Irrlicht for visualization

scene = Ir.SceneManager()

scene.SetWindowCaption("HMMWV Simulation")

scene.SetWindowSize(1920, 1080)

scene.AddFile("Highway_col.obj", Ir.MESSAGE_PRIORITY_HIGHEST)

scene.AddFile("Highway_vis.obj", Ir.MESSAGE_PRIORITY_HIGHEST)


# Create vehicle

vehicle = ch.AddNewVehicle("HMMWV", Ir.VehicleType.HMMWV)

vehicle.SetLocation(0, 0, 0)

vehicle.SetOrientation(0, 0, np.pi/4)

vehicle.SetContactMethod(Irr.VEHICLE_CONTACT_METHOD.AUTO_TOUCH)

vehicle.SetTMEASYTireModel()

vehicle.SetMesh("HMMWV_mesh.obj")


# Add terrain

terrain = scene.AddMesh("Highway_col.obj", Ir.MESSAGE_PRIORITY_HIGHEST)

terrain.SetMaterialFlag(Irr.MESH_MATERIAL_FLAG.WIRE_FRAME, True)

terrain.SetMaterialFlag(Irr.MESH_MATERIAL_FLAG.WIRE_FRAME_COLOR, Ir.Vector3(1, 0, 0))


# Setup interactive driver

driver = ch.AddNewDriver(Irr.VEHICLE_DRIVER_TYPE.USER_CONTROL)

driver.SetVehicle(vehicle)

driver.SetSteeringWheelSpeed(1.0)


# Simulation loop

while ch.GetSimulationStatus() == Irr.SIM_STATUS.RUNNING:

    ch.StepSimulation(0.0016)

    scene.BeginScene()

    scene.DrawAll()

    scene.EndScene()

    Ir.Application.Run()

```


Please note that the above script is a simplified example and assumes that the necessary files and meshes are correctly set up and that the PyChrono and PyIrrlicht libraries are installed and configured.
