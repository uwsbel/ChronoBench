```python

import pychronoid as pc

import irrlicht as ir

import irrlicht.scene as isc

import irrlicht.core as cpl

import irrlicht.keyboard as kb

import irrlicht.video as video


# Initialize PyChrono

pc.init()


# Initialize Irrlicht for visualization

ir.core.IrrContext.setDeviceName("drm")

scene = isc.QuadMeshBuilder()

scene.addPrimitiveWithMeshPath(isc.PRIMITIVE_TRIANGLE_LIST, "textures/terrain.png")

scene.addPrimitiveWithMeshPath(isc.PRIMITIVE_TRIANGLE_LIST, "textures/gator.png")

render_manager = cpl.RenderWindowManager()

render_manager.addSceneGraphRoot(scene.getRootBone())


# Define the RigidTerrain

terrain_dimensions = pc.vector3(100, 100, 0)

terrain = pc.RigidTerrain(terrain_dimensions)

terrain.setTexture("textures/terrain.png")


# Define the Gator vehicle

gator = pc.RigidBody(pc.vector3(0, 0, 0),

                      pc.Quaternion(0, 0, 0, 1),

                      pc.RigidBodyType.ROTATABLE_RIGIDBODY)

gator.setMass(1000)

gator.setFriction(0.5)

gator.setRestitution(0.0)

gator.setContactMethod(pc.ContactMethod.NONE)

gator.setTMEASYTireModel()


# Set up vehicle components

gator.addShape(pc.CapsuleShape(pc.vector3(0.5, 0.5, 0.5), 1.0))

gator.addShape(pc.CapsuleShape(pc.vector3(0.5, 0.5, 0.5), 0.5))

gator.addShape(pc.CapsuleShape(pc.vector3(0.5, 0.5, 0.5), 0.2))


# Set initial position and orientation

gator.setPosition(pc.vector3(0, 0, 0))

gator.setOrientation(pc.Quaternion(0, 0, 0, 1))


# Set up the vehicle's interaction with the terrain

gator.setContactMethod(pc.ContactMethod.FRICTION)

terrain.setRigidBody(gator)

terrain.setRigidBodyCollisionFlags(pc.RigidBodyCollisionFlags.COLLIDE_WITH_BOTH_BODIES)


# Set up the driver system for steering, throttle, and braking

driver_system = pc.DriverSystem(gator,

                                   steering_controller=pc.SteeringController(0.1, 1.0),

                                   throttle_controller=pc.ThrottleController(0.1, 1.0),

                                   brake_controller=pc.BrakeController(0.1, 1.0))


# Set up the simulation loop

frame_count = 0

while True:

    # Update the simulation

    pc.stepSimulation(gator, terrain, driver_system)


    # Render the scene

    render_manager.drawAll()

    video.VideoDriver.update()

    video.VideoDriver.drawAll()


    # Check for quit event

    if kb.waitKey(True, False) == irr.EDT_ESCAPE:

        break


    # Increment frame count and maintain 50 FPS

    frame_count += 1

    if frame_count % 20 == 0:

        pc.waitSimulationFrame()


# Clean up

render_manager.close()

pc.shutdown()

```
