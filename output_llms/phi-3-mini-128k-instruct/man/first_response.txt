```python

import pychrono as chrono

import pychrono.visualization as visual

import pychrono.core as core

import pychrono.vehicle as vehicle

import pychrono.collision as collision

import pychrono.irrlicht as irrlicht


# Initialize the PyChrono environment

core.Initialize()


# Create a core component for the simulation

my_chrono_instance = core.ChromaCore()


# Create the vehicle model

vehicle_model = vehicle.VehicleModel()

vehicle_model.SetName('MAN 10t Truck')

vehicle_model.SetColor(chrono.MaterialProps_Metal)

vehicle_model.SetVisualization(True)

vehicle_model.SetCollision(True)

vehicle_model.SetMass(10000)  # Mass of the truck in kg

vehicle_model.SetInertia(chrono.ChVectorD(10000, 20000, 15000))  # Inertia tensor

vehicle_model.SetPosition(chrono.ChVectorD(0, 0, 0))

vehicle_model.SetVelocity(chrono.ChVectorD(0, 0, 0))

vehicle_model.SetSteeringAngle(0)

vehicle_model.SetTirePressure(21)

vehicle_model.SetTireRadius(0.35)

vehicle_model.SetTireModel(vehicle.VehicleTireModel_TMEASY)


# Create the terrain

terrain = vehicle.Terrain()

terrain.SetName('Rigid Terrain')

terrain.SetColor(chrono.MaterialProps_RoughConcrete)

terrain.SetGeometry(chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(100, 100, 0), 0.1)


# Add the vehicle and terrain to the simulation

my_chrono_instance.Add(vehicle_model)

my_chrono_instance.Add(terrain)


# Set up the visualization system

my_visualizer = visual.VisionSystem()

my_visualizer.SetName('Truck Simulation')

my_visualizer.SetCameraType(visual.VisionSystem_ChaseCamera)

my_visualizer.SetCameraPosition(chrono.ChVectorD(10, 10, 10))

my_visualizer.SetCameraTarget(vehicle_model)

my_visualizer.SetDirectionalLight(chrono.ChVectorD(0, -1, -1))

my_visualizer.SetSkyboxTexture(chrono.GetDataFilePath() + '/skybox/skybox.png')

my_visualizer.SetTerrainTexture(chrono.GetDataFilePath() + '/terrain/terrain.jpg')


# Set up the Irrlicht visualization system

my_irrlicht_visualizer = irrlicht.IrrlichtVisualizer(my_visualizer)

my_irrlicht_visualizer.SetWidth(1280)

my_irrlicht_visualizer.SetHeight(720)


# Add the visualization system to the simulation

my_chrono_instance.AddVisualizationSystem(my_irrlicht_visualizer)


# Set up the real-time driver controls

steering_controller = vehicle.VehicleController()

throttle_controller = vehicle.VehicleController()

brake_controller = vehicle.VehicleController()


# Add the controllers to the vehicle model

vehicle_model.AddController(steering_controller)

vehicle_model.AddController(throttle_controller)

vehicle_model.AddController(brake_controller)


# Run the simulation

my_chrono_instance.DoStep(0.001, True)


# Clean up

core.Shutdown()

```


This script initializes the PyChrono environment, creates a vehicle model for a MAN 10t truck, adds a rigid terrain, sets up the visualization system using Irrlicht, and includes real-time driver controls for steering, throttle, and braking. The script also configures the visualization with a chase camera, directional lighting, a skybox, and customizable terrain textures and logos.
