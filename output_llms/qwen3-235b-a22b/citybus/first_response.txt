```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.terrain as terrain
import pychrono.irrlicht as chronoirr

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the CityBus vehicle
bus = veh.CityBus(system)
bus.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 1.0, 0), chrono.QUNIT))
bus.SetTireType(veh.TireModelType_TMEASY)  # Use TMeasy tire model
bus.Initialize()

# Set visualization assets for the vehicle
# Chassis with mesh
chassis = bus.GetChassis().GetBody()
chassis_mesh = chrono.ChTriangleMeshConnected()
chassis_mesh.LoadWavefrontMesh(chrono.GetChronoDataFile("vehicle/CityBus/CityBus.obj"))
chassis.AddAsset(chassis_mesh)

# Wheels with primitive cylinders
for i in range(bus.GetNumberofWheels()):
    wheel = bus.GetWheelBody(i)
    wheel_cyl = chrono.ChCylinderShape(0.3, 0.2)  # Radius, width
    wheel.AddAsset(wheel_cyl)

# Create Rigid Terrain with custom texture
terrain_obj = terrain.RigidTerrain(system)
patch = terrain_obj.AddPatch(chrono.ChMaterialSurfaceNSC(), chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(200, 1, 200))
patch.SetTexture(chrono.GetChronoDataFile("terrain/textures/tile.png"))
terrain_obj.Initialize()

# Set up Irrlicht visualization
application = chronoirr.ChIrrApp(system, 'CityBus Simulation', chronoirr.dimension2du(1024, 768))
application.AddTypicalLogo()
application.AddTypicalCamera(chronoirr.vector3df(0, 3, -6), chronoirr.vector3df(0, 1, 0))
application.AddLightWithShadow(chronoirr.vector3df(10, 20, 10), chronoirr.vector3df(0, 0, 0), 100, 1, 100, 45, chronoirr.video.SColorf(0.8, 0.8, 1.0))

# Set up interactive driver system
driver = veh.ChIrrGuiDriver(application.GetDevice())
driver.SetSteeringDelta(0.05)
driver.SetThrottleDelta(0.05)
driver.SetBrakingDelta(0.05)

# Simulation loop parameters
step_size = 1.0 / 50.0  # 50 FPS
rt_timer = chrono.ChRealtimeStepTimer()

# Camera follow settings
camera = application.GetSceneManager().getActiveCamera()
camera_offset = chronoirr.vector3df(0, 5, -10)  # Offset from vehicle

# Main simulation loop
while application.GetDevice().run():
    time = system.GetChTime()

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Synchronize vehicle and terrain
    bus.Synchronize(time, driver_inputs, terrain_obj)
    terrain_obj.Synchronize(time)

    # Advance simulation
    system.DoStepDynamics(step_size)

    # Update camera to follow vehicle
    chassis_pos = bus.GetChassis().GetPos()
    camera.setTarget(chronoirr.vector3df(chassis_pos.x, chassis_pos.y, chassis_pos.z))
    camera.setPosition(chronoirr.vector3df(
        chassis_pos.x + camera_offset.X,
        chassis_pos.y + camera_offset.Y,
        chassis_pos.z + camera_offset.Z
    ))

    # Render scene
    application.BeginScene()
    application.DrawAll()
    application.EndScene()

    # Maintain real-time execution
    rt_timer.Spin(step_size)
```
