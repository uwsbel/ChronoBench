```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import pychrono.driver as driver

# Initialize the PyChrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVector(0, 0, -9.81))

# Create the ARTcar vehicle
artcar = veh.ARTcar(system)
artcar.SetContactMethod(chrono.ChContactMethod_NSC)
artcar.SetChassisVisualizationType(veh.VisualizationType_MESH)
artcar.SetWheelVisualizationType(veh.VisualizationType_MESH)
# Set initial position and orientation (x, y, z) and rotate around Y-axis
artcar.Initialize(chrono.ChCoordsysd(chrono.ChVector(0, 0, 1), chrono.Q_from_AngAxis(0, chrono.ChVector(0, 1, 0))))

# Create the Rigid Terrain
terrain = veh.RigidTerrain(system)
patch_mat = chrono.ChMaterialSurfaceNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)
# Add a patch with dimensions 200m x 100m x 1m
patch = terrain.AddPatch(patch_mat, chrono.ChVector(0, 0, 0), chrono.ChVector(200, 100, 1), True, True)
# Apply a custom texture (ensure the texture file exists in the data path)
patch.SetTexture(chrono.GetChronoDataFile("terrain/textures/grass.png"), 200, 100)
terrain.Initialize()

# Create Irrlicht application for visualization
application = irr.ChIrrApp(system, 'ARTcar Simulation', irr.dimension2d(1024, 768))
application.AddTypicalLights()
application.AddCamera(chrono.ChVector(0, -5, 2), chrono.ChVector(0, 0, 1))
application.AssetBindAll()
application.AssetUpdateAll()

# Create the interactive driver system
driver = veh.ChIrrGuiDriver(application.GetDevice(), artcar, 10.0, 0.3, 0.3)
driver.SetSteeringDelta(0.05)
driver.SetThrottleDelta(0.05)
driver.SetBrakingDelta(0.05)

# Simulation loop at 50 frames per second
step_size = 1.0 / 50.0
rt_timer = chrono.ChRealtimeTimer()

while application.GetDevice().run():
    rt_timer.Spin(step_size)
    
    # Get driver inputs
    driver_inputs = driver.GetInputs()
    
    # Update vehicle dynamics
    artcar.Synchronize(system.GetChTime(), driver_inputs)
    system.DoStepDynamics(step_size)
    
    # Update visualization
    application.BeginScene()
    application.DrawAll()
    application.EndScene()
```
