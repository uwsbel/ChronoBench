```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import pychrono.driver as driver

# Initialize the PyChrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVector3d(0, -9.81, 0))

# Set simulation step size for 50 FPS (0.02 seconds)
step_size = 0.02

# Create the HMMWV vehicle
vehicle = veh.HMMWV_Full(system)
vehicle.SetContactMethod(chrono.ChContactMethod_NSC)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(chrono.ChVector3d(0, 1.5, 0), chrono.QUNIT))
vehicle.SetTireType(veh.TireModelType_TMeasy)
vehicle.SetTireStepSize(step_size)
vehicle.Initialize()

# Create custom terrain using mesh files
# Load collision and visual meshes
mesh_col = chrono.ChTriangleMeshConnected()
mesh_col.LoadWavefrontMesh("Highway_col.obj", True, True)

mesh_vis = chrono.ChTriangleMeshConnected()
mesh_vis.LoadWavefrontMesh("Highway_vis.obj", True, True)

# Create terrain body
terrain_body = chrono.ChBody()
terrain_body.SetBodyFixed(True)
terrain_body.SetCollide(True)
system.Add(terrain_body)

# Add collision mesh to terrain
terrain_body.GetCollisionModel().ClearModel()
terrain_body.GetCollisionModel().AddTriangleMesh(
    mesh_col, 
    chrono.ChVector3d(0, 0, 0), 
    chrono.ChMatrix33d(chrono.QUNIT), 
    True, 
    True
)
terrain_body.GetCollisionModel().BuildModel()

# Add visual mesh to terrain
vis_shape = chrono.ChVisualShapeTriangleMesh()
vis_shape.SetMesh(mesh_vis)
vis_shape.SetName("Highway Visualization")
terrain_body.AddVisualShape(vis_shape)

# Initialize Irrlicht application
application = irr.ChIrrApp(system, "HMMWV Simulation", irr.dimension2du(1024, 768))
application.AddTypicalLogo()
application.AddTypicalSky()
application.AddTypicalLights(irr.vector3df(30, 100, 30), irr.vector3df(30, 0, 30), 128)
application.AddCamera(chrono.ChVector3d(0, 5, -10), chrono.ChVector3d(0, 0, 0))

# Initialize interactive driver
my_driver = driver.InteractiveDriverIRR(application)
my_driver.SetSteeringDelta(0.05)
my_driver.SetThrottleDelta(0.05)
my_driver.SetBrakingDelta(0.05)
my_driver.Initialize()

# Configure step management for real-time simulation
application.SetStepManage(True)
application.SetTimestep(step_size)
application.SetTryRealtime(True)

# Simulation loop
while application.GetDevice().run():
    time = system.GetChTime()

    # Update driver inputs
    my_driver.Advance(step_size)
    driver_inputs = my_driver.GetInputs()

    # Synchronize vehicle with driver inputs
    vehicle.Synchronize(time, driver_inputs)
    vehicle.Advance(step_size)

    # Advance the simulation
    system.DoStepDynamics(step_size)

    # Render scene
    application.BeginScene()
    application.DrawAll()
    application.EndScene()
```
