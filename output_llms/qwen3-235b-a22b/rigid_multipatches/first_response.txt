```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irrlicht

# Initialize the PyChrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Set data paths for Chrono and vehicle models
chrono.SetChronoDataPath('path/to/chrono/data')
veh.SetDataPath('path/to/vehicle/data')

# Create Irrlicht application for visualization
application = irrlicht.ChIrrApp(system, 'HMMWV Simulation', irrlicht.dimension2du(1024, 768))
application.AddTypicalLights()
application.AddCamera(chrono.ChVectorD(0, 5, -10), chrono.ChVectorD(0, 0, 0))

# Create HMMWV vehicle
vehicle = veh.HMMWV_Full(system)
vehicle.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 1.5, 0), chrono.QUNIT))
vehicle.SetEngineType(veh.EngineModelType_SIMPLE)
vehicle.SetDrivetrainType(veh.DrivetrainModelType_SHAFTS)
vehicle.SetTireType(veh.TireModelType_TMEASY)
vehicle.Initialize()

# Apply mesh visualization to vehicle components
vehicle.GetChassis().SetMeshFilename(veh.GetDataFile("hmmwv/chassis.obj"))
for i in range(vehicle.GetNumberWheels()):
    wheel_mesh = veh.GetDataFile(f"hmmwv/wheel_{i}.obj")
    vehicle.GetWheel(i).SetMeshFilename(wheel_mesh)

# Create complex terrain with multiple patches
terrain = veh.ChTerrain()

# Flat patch with texture
patch1 = terrain.AddPatch(chrono.ChMaterialSurfaceNSC(), chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(100, 1, 100))
patch1.SetTexture(veh.GetDataFile("terrain/textures/concrete.jpg"))

# Mesh-based bump patch
bump_mesh = chrono.ChTriangleMeshConnected()
bump_mesh.LoadWavefrontMesh(veh.GetDataFile("terrain/meshes/bump.obj"))
patch2 = terrain.AddPatch(bump_mesh, chrono.ChMaterialSurfaceNSC(), chrono.ChVectorD(20, 0, 0), chrono.ChQuaternionD(1, 0, 0, 0))
patch2.SetTexture(veh.GetDataFile("terrain/textures/dirt.jpg"))

# Heightmap-based patch
heightmap = chrono.ChHeightmap()
heightmap.LoadFromFile(veh.GetDataFile("terrain/heightmaps/rough.png"))
patch3 = terrain.AddPatch(heightmap, chrono.ChMaterialSurfaceNSC(), chrono.ChVectorD(50, 0, 0), chrono.ChVectorD(30, 10, 30))
patch3.SetTexture(veh.GetDataFile("terrain/textures/grass.jpg"))

terrain.Initialize()

# Set up interactive driver
driver = veh.ChIrrGuiDriver(application.GetDevice())
driver.Initialize()

# Set up real-time step timer
step_timer = chrono.ChRealtimeStepTimer()
step_size = 1e-3

# Main simulation loop
while application.GetDevice().run():
    time = system.GetChTime()
    
    # Update driver inputs
    driver.Synchronize(time)
    vehicle.SetSteering(driver.GetSteering())
    vehicle.SetThrottle(driver.GetThrottle())
    vehicle.SetBraking(driver.GetBraking())
    
    # Advance simulation
    system.DoStepDynamics(step_size)
    
    # Render scene
    application.BeginScene()
    application.DrawAll()
    application.EndScene()
    
    step_timer.Spin(step_size)
```
