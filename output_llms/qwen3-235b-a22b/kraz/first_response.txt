```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create and initialize the Kraz vehicle
kraz = veh.ChKraz(system)
kraz.SetInitPosition(chrono.ChCoordsysD(
    chrono.ChVectorD(0, 1.5, 0),  # Position (adjust y to be above terrain)
    chrono.ChQuaternionD(1, 0, 0, 0)  # Orientation (unit quaternion)
))
kraz.SetEngineType(veh.EngineModelType_SIMPLE)
kraz.SetTireModel(veh.TireModelType_TMEASY)
kraz.Initialize()

# Create the terrain
terrain = chrono.ChBody()
terrain.SetBodyFixed(True)
terrain.SetCollide(True)
terrain.GetCollisionModel().ClearModel()
terrain.GetCollisionModel().AddBox(100, 1, 100)  # x, y, z half-dimensions
terrain.GetCollisionModel().BuildModel()

# Set terrain material properties
mat = chrono.ChMaterialSurfaceNSC()
mat.SetFriction(0.9)
mat.SetRestitution(0.01)
terrain.SetMaterialSurface(mat)

# Add visualization assets to terrain
box = chrono.ChBoxShape()
box.Size = chrono.ChVectorD(100, 1, 100)
terrain.AddAsset(box)

texture = chrono.ChTexture()
texture.SetTextureFilename(chrono.GetChronoDataFile("textures/concrete.jpg"))
terrain.AddAsset(texture)

system.Add(terrain)

# Create the driver system
driver = veh.ChIrrGuiDriver(kraz)
driver.SetSteeringDelta(0.05)
driver.SetThrottleDelta(0.05)
driver.SetBrakingDelta(0.05)

# Set up Irrlicht visualization
application = chronoirr.ChIrrApp(system, 'Kraz Simulation', chronoirr.dimension2du(1024, 768))
application.AddTypicalLogo()
application.AddTypicalSky()
application.AddTypicalLights()
application.AddCamera(chrono.ChVectorD(0, 5, -10), chrono.ChVectorD(0, 0, 0))

application.AssetBindAll()
application.AssetUpdateAll()

# Simulation loop
step_size = 1e-3  # 0.001 seconds
time = 0.0

while application.GetDevice().run():
    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Synchronize components
    driver.Synchronize(time)
    kraz.Synchronize(time, driver_inputs, terrain)

    # Advance dynamics
    system.DoStepDynamics(step_size)
    time += step_size

    # Update visualization
    application.BeginScene()
    application.DrawAll()
    application.EndScene()
```
