```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr

# Initialize the Chrono simulation system
sys = chrono.ChSystemNSC()

# Set gravitational acceleration for the Moon (1.62 m/s²)
sys.SetGravitationalAcceleration(chrono.ChVector3d(0, -1.62, 0))

# Create the ground body and add it to the simulation
ground = chrono.ChBody()
sys.Add(ground)
ground.SetFixed(True)  # Ground is fixed
ground.EnableCollision(False)  # Disable collision detection

# Add visualization sphere to represent the spherical joint
joint_sphere = chrono.ChVisualShapeSphere(2)  # Radius 2 as requested
ground.AddVisualShape(joint_sphere, chrono.ChFramed(chrono.ChVector3d(0, 0, 1)))

# Create the pendulum body and add it to the simulation
pend_1 = chrono.ChBody()
sys.AddBody(pend_1)
pend_1.SetFixed(False)  # Pendulum can move
pend_1.EnableCollision(False)  # Disable collision detection
pend_1.SetMass(2)  # Set mass to 2 kg
pend_1.SetInertiaXX(chrono.ChVector3d(0.4, 1.5, 1.5))  # Updated inertia tensor

# Add visualization cylinder to the pendulum
pend_cylinder = chrono.ChVisualShapeCylinder(0.1, 1.5)  # Radius 0.1, height 1.5
pend_cylinder.SetColor(chrono.ChColor(0.6, 0, 0))  # Red color
pend_1.AddVisualShape(pend_cylinder, chrono.ChFramed(chrono.VNULL, chrono.QuatFromAngleY(chrono.CH_PI_2)))

# Set initial position of the pendulum (center of mass)
# Adjusted to match new pendulum length (half of 1.5 = 0.75)
pend_1.SetPos(chrono.ChVector3d(0.75, 0, 1))

# Set initial angular velocity (example: 5 rad/s around Z-axis)
pend_1.SetWvel_par(chrono.ChVector3d(0, 0, 5))

# Create a spherical joint to connect the pendulum to the ground
sph_joint = chrono.ChLinkLockSpherical()
sph_joint.Initialize(
    ground, 
    pend_1, 
    chrono.ChFramed(chrono.ChVector3d(0, 0, 1), chrono.ChQuaterniond(1, 0, 0, 0))
)
sys.AddLink(sph_joint)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Spherical Pendulum Simulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 6))
vis.AddTypicalLights()

# Simulation loop
log_info = True
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(1e-3)  # 0.001 second time step

    # Log information after 1 second of simulation
    if log_info and sys.GetChTime() > 1:
        pos = pend_1.GetPos()
        print(f"t = {sys.GetChTime():.2f}s")
        print(f"Position: {pos.x:.4f}, {pos.y:.4f}, {pos.z:.4f}")
        vel = pend_1.GetPosDt()
        print(f"Velocity: {vel.x:.4f}, {vel.y:.4f}, {vel.z:.4f}")
        log_info = False
```
