```python
import pychrono as chrono
import pychrono.fea as fea
import pychrono.irrlicht as irr

# Initialize the system
sys = chrono.ChSystemNSC()
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Set solver
solver = chrono.ChSolverMINRES()
sys.SetSolver(solver)

# Beam parameters
L = 1.0  # Length of the beam
radius = 0.05  # Radius of the beam's cross-section
E = 2.1e11  # Young's modulus (Pa)
nu = 0.3  # Poisson ratio
density = 7800  # Density (kg/m^3)
G = E / (2 * (1 + nu))  # Shear modulus

# Create mesh
mesh = fea.ChMesh()

# Create material and section for the IGA beam
material = fea.ChMaterialBeamIGA()
material.SetYoungModulus(E)
material.SetShearModulus(G)
material.SetDensity(density)
material.SetPoissonRatio(nu)

section = fea.ChSectionBeamIGA()
section.SetMaterial(material)
section.SetAsCircularSection(radius)

# Build the IGA beam using the builder
builder = fea.ChBuilderBeamIGA()
builder.BuildBeam(mesh, section, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(L, 0, 0), chrono.ChVectorD(0, 1, 0), 3)

# Add mesh to the system
sys.Add(mesh)

# Retrieve nodes and attach flywheel at the center
nodes = [node for node in mesh.GetNodes()]
flywheel_node = nodes[len(nodes) // 2]

# Create flywheel body
flywheel = chrono.ChBody()
flywheel.SetMass(10)  # Example mass (kg)
flywheel.SetInertiaXX(chrono.ChVectorD(0.1, 0.1, 0.2))  # Example inertia
flywheel.SetPos(flywheel_node.GetPos())
sys.Add(flywheel)

# Link the flywheel to the beam node
link = fea.ChLinkNodeBody()
link.Initialize(flywheel_node, flywheel)
sys.Add(link)

# Create ground and add rotational motor to one end
ground = chrono.ChBody()
ground.SetBodyFixed(True)
sys.Add(ground)

motor_node = nodes[0]
motor = chrono.ChLinkMotorRotationAngle()
motor.Initialize(motor_node, ground, chrono.ChFrameD(motor_node.GetPos()))
sys.Add(motor)

# Set motor angular velocity (e.g., 100 rad/s)
omega = 100.0
function = chrono.ChFunction_Ramp(omega, 0)  # theta(t) = omega * t
motor.SetAngleFunction(function)

# Visualization setup
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Jeffcott Rotor IGA Simulation')
vis.Initialize()
vis.AddCamera(chrono.ChVectorD(0, 0, -3))
vis.AddLight(chrono.ChVectorD(2, 2, 2), chrono.ChColor(1, 1, 1))

# Add FEM visualization for the beam
mesh_vis = fea.ChVisualizationFEAmesh(mesh)
mesh_vis.SetFEMdataType(fea.ChVisualizationFEAmesh.E_PLOT_SURFACE)
mesh_vis.SetWireframe(True)
mesh.AddAsset(mesh_vis)

# Run simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(0.001)
```

### Explanation:
1. **System Initialization**: The simulation environment is initialized with gravity and a MINRES solver for efficient handling of FEM systems.
2. **Beam Creation**: An IGA beam is constructed using `ChBuilderBeamIGA`, specifying material properties and a circular cross-section. The beam spans from (0,0,0) to (L,0,0) with 3 elements.
3. **Flywheel Attachment**: A flywheel is modeled as a rigid body and attached to the central node of the beam using `ChLinkNodeBody`, ensuring it follows the node's motion.
4. **Rotational Motor**: A motor is applied to one end of the beam, driving it with a constant angular velocity using a ramp function to simulate rotation.
5. **Visualization**: The Irrlicht engine visualizes the system, including the IGA beam's deformation and the flywheel's dynamics.

This script provides a foundational simulation of a Jeffcott rotor using IGA-based FEM, suitable for further refinement and analysis.
