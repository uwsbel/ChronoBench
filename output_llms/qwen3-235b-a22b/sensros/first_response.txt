

```python
import pychrono as chrono
import pychrono.sensor as sens
import pychrono.ros as ros
import math

# Initialize the Chrono system
system = chrono.ChSystemNSC()

# Create the ground body
ground = chrono.ChBody()
ground.SetMass(1000)
ground.SetInertiaXX(chrono.ChVector(100, 100, 100))
ground.SetPos(chrono.ChVectorD(0, 0, 0))
ground.SetLinearVel(chrono.ChVectorD(1, 0, 0))  # Moving along x-axis
ground.SetBodyFixed(False)
system.Add(ground)

# Add a box as the ground's visualization mesh
box = chrono.ChBoxShape()
box.GetBoxGeometry().Size = chrono.ChVector(5, 1, 5)
ground.AddAsset(box)

# Create sensor manager
manager = sens.ChSensorManager(system)

# Add sensors to the manager
# Camera
camera = sens.ChCameraSensor(
    ground,
    60.0,
    chrono.ChFrameD(chrono.ChVectorD(0, 1, 0), chrono.QUNIT),
    1280,
    720,
    chrono.CH_C_PI / 3
)
manager.AddSensor(camera)

# Lidar
lidar = sens.ChLidarSensor(
    ground,
    10.0,
    chrono.ChFrameD(chrono.ChVectorD(0, 1.5, 0), chrono.QUNIT),
    1000,
    32,
    chrono.CH_C_PI,
    chrono.CH_C_PI / 12,
    100
)
manager.AddSensor(lidar)

# GPS
gps = sens.ChGPSSensor(
    ground,
    10.0,
    chrono.ChFrameD(chrono.ChVectorD(0, 2, 0), chrono.QUNIT),
    chrono.ChVectorD(0, 0, 0),
    0.1
)
manager.AddSensor(gps)

# IMU (combines accelerometer, gyroscope, and magnetometer)
imu = sens.ChImuSensor(
    ground,
    100.0,
    chrono.ChFrameD(chrono.ChVectorD(0, 0.5, 0), chrono.QUNIT)
)
manager.AddSensor(imu)

# Initialize ROS manager and register sensors with topics
ros_manager = ros.ChROSManager()
ros_manager.Initialize()
ros_manager.RegisterSensor(camera, "/camera/image_raw")
ros_manager.RegisterSensor(lidar, "/laser/scan")
ros_manager.RegisterSensor(gps, "/gps/fix")
ros_manager.RegisterSensor(imu, "/imu/data")

# Simulation loop
step_size = 1e-3  # 1 ms step
time_step = chrono.ChRealtimeStepTimer()

while system.GetChTime() < 20.0:  # Run for 20 seconds
    # Update sensors and ROS communication
    manager.Update()
    ros_manager.Update()
    
    # Advance the simulation
    system.DoStepDynamics(step_size)
    
    # Maintain real-time execution
    time_step.Sleep()

# Clean up ROS
ros_manager.Shutdown()
```
