```python
import pychrono as chrono
import pychrono.fea as fea
import pychrono.irrlicht as irr

# Initialize the system with PardisoMKL solver
sys = chrono.ChSystemNSC()
solver = chrono.ChSolverPardisoMKL()
sys.SetSolver(solver)
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the finite element mesh
mesh = fea.ChMesh()

# Mesh parameters
nx = 20
ny = 20
Lx = 1.0
Ly = 1.0
dx = Lx / (nx - 1)
dy = Ly / (ny - 1)
thickness = 0.001
density = 1000
E = 2.2e9  # Young's modulus (Pa)
nu = 0.3   # Poisson ratio

# Create nodes (ChNodeFEAxyzD for ANCF shell elements)
nodes = []
for i in range(nx):
    row = []
    for j in range(ny):
        x = i * dx - Lx / 2
        y = j * dy - Ly / 2
        z = 0.0
        director = chrono.ChVectorD(0, 0, thickness / 2)  # Initial director vector
        node = fea.ChNodeFEAxyzD(chrono.ChVectorD(x, y, z), director)
        mesh.AddNode(node)
        row.append(node)
    nodes.append(row)

# Create shell elements (ChElementShell44ANCF)
for i in range(nx - 1):
    for j in range(ny - 1):
        node0 = nodes[i][j]
        node1 = nodes[i+1][j]
        node2 = nodes[i+1][j+1]
        node3 = nodes[i][j+1]

        element = fea.ChElementShell44ANCF()
        element.SetNodes(node0, node1, node2, node3)

        # Assign isotropic material
        material = fea.ChMaterialShellANCF()
        material.SetDensity(density)
        material.SetYoungModulus(E)
        material.SetPoissonRatio(nu)
        element.SetMaterial(material)

        mesh.AddElement(element)

# Add mesh to the system
sys.Add(mesh)

# Fix corner nodes
corner_nodes = [nodes[0][0], nodes[-1][0], nodes[0][-1], nodes[-1][-1]]
for node in corner_nodes:
    node.SetFixed(True)

# Set up Irrlicht visualization
vis = irr.ChVisualSystemIrrlicht()
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Tablecloth Folding Simulation')
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 0.5, 1.5), chrono.ChVectorD(0, 0, 0))
vis.AttachSystem(sys)

# Add mesh visualization
mesh_vis = fea.ChVisualizationFEAmesh(mesh)
mesh_vis.SetFEMdataType(fea.ChVisualizationFEAmesh.E_FEM_DATATYPE_ELEM_STRESS)
mesh_vis.SetColorscaleMinMax(-0.1, 0.1)
mesh_vis.AddFaces(True)
mesh_vis.SetDrawWireframe(True)
mesh.AddAsset(mesh_vis)

# Simulation loop
step_size = 0.001
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(step_size)
```
