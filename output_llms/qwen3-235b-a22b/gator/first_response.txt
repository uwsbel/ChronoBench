```python
import pychrono as chrono
import pychrono.vehicle as vehicle
import pychrono.irrlicht as irrlicht
import pychrono.driver as driver

# Initialize the PyChrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

# Create the Gator vehicle
gator = vehicle.Gator(system)
gator.SetContactMethod(vehicle.ChContactMethod_SMC)  # Contact method (SMC)
initial_pos = chrono.ChVectorD(0, 0, 0.5)  # Initial position (x, y, z)
initial_rot = chrono.ChQuaternionD(1, 0, 0, 0)  # Initial orientation (no rotation)
gator.Initialize(chrono.ChCoordsysD(initial_pos, initial_rot))
gator.SetTireType(vehicle.TMEASY)  # TMEASY tire model
gator.SetChassisVisualizationType(vehicle.VisualizationType_MESH)
gator.SetWheelVisualizationType(vehicle.VisualizationType_MESH)

# Create the Rigid Terrain
terrain = vehicle.RigidTerrain(system)
patch = terrain.AddPatch(
    chrono.ChCoordsysD(), 
    chrono.ChVectorD(200, 100, 1),  # Dimensions (length, width, height)
    vehicle.MaterialInfo()  # Default material
)
patch.SetTexture(chrono.GetChronoDataFile("terrain/textures/grass.jpg"))  # Custom texture
terrain.Initialize()

# Create the interactive driver system
driver = vehicle.ChIrrGuiDriver(gator)
driver.SetSteeringDelta(0.05)  # Steering sensitivity
driver.SetThrottleDelta(0.05)  # Throttle sensitivity
driver.SetBrakingDelta(0.05)  # Braking sensitivity

# Set up Irrlicht visualization
application = irrlicht.ChIrrApp(
    system, 
    'PyChrono Gator Simulation', 
    irrlicht.dimension2du(1024, 768)  # Window size
)
application.AddTypicalLights()
application.AddCamera(chrono.ChVectorD(0, -6, 3), chrono.ChVectorD(0, 0, 0))  # Camera position and target

# Bind visualization assets
application.AssetBindAll()
application.AssetUpdateAll()

# Simulation loop parameters
time_step = 0.02  # 50 FPS
realtime_timer = chrono.ChRealtimeStepTimer()

# Simulation loop
while application.GetDevice().run():
    current_time = system.GetChTime()
    driver_inputs = driver.GetInputs()

    # Synchronize modules
    driver.Synchronize(current_time)
    terrain.Synchronize(current_time)
    gator.Synchronize(current_time, driver_inputs, terrain)

    # Advance simulation
    system.DoStepDynamics(time_step)

    # Update visualization
    application.BeginScene()
    application.DrawAll()
    application.EndScene()

    # Maintain real-time execution
    realtime_timer.Spin(time_step)
```
