Here is a well-documented PyChrono simulation script that implements a **crank-slider mechanism** with a **floor (truss)**, **crankshaft**, **connecting rod**, and **piston**, driven by a **motor**. The simulation uses **Irrlicht** for visualization with customizable camera settings, lighting, and textures.

---

### ✅ Features Implemented

- **Crank-slider mechanism** with appropriate joints:
  - **Revolute joints** for crankshaft-floor and connecting rod-crankshaft/piston.
  - **Prismatic joint** for piston-floor.
- **Motor** to drive the crankshaft at a constant angular speed.
- **Irrlicht visualization** with:
  - Custom camera position and target.
  - Lighting and sky.
  - Textures for visual elements.
  - Logo and skybox.

---

### 📜 Full Simulation Script

```cpp
#include "chrono/physics/ChSystemNSC.h"
#include "chrono/physics/ChBody.h"
#include "chrono/assets/ChBoxShape.h"
#include "chrono/assets/ChTexture.h"
#include "chrono/core/ChRealtimeStep.h"
#include "chrono_irrlicht/ChIrrApp.h"
#include "chrono/physics/ChLinkLockRevolute.h"
#include "chrono/physics/ChLinkLockPrismatic.h"
#include "chrono/physics/ChLinkMotorRotationSpeed.h"
#include "chrono/functions/ChFunction_Const.h"

using namespace chrono;
using namespace chrono::irrlicht;
using namespace irr;

int main() {
    // Create the physical system
    ChSystemNSC my_system;
    my_system.Set_G_acc(ChVector<>(0, -9.81, 0)); // Gravity in Y direction

    // Create the floor (truss-like structure using a box)
    auto floor = chrono_types::make_shared<ChBodyEasyBox>(2.0, 0.1, 1.0, 1000, true, false);
    floor->SetPos(ChVector<>(0, 0, 0));
    floor->SetBodyFixed(true);
    my_system.Add(floor);

    // Create crankshaft (a box rotating around Z-axis)
    auto crank = chrono_types::make_shared<ChBodyEasyBox>(0.2, 0.05, 0.05, 1000, true, false);
    crank->SetPos(ChVector<>(0, 0.1, 0));
    my_system.Add(crank);

    // Create connecting rod
    auto connecting_rod = chrono_types::make_shared<ChBodyEasyBox>(0.5, 0.05, 0.05, 1000, true, false);
    connecting_rod->SetPos(ChVector<>(0.35, 0.1, 0));
    my_system.Add(connecting_rod);

    // Create piston
    auto piston = chrono_types::make_shared<ChBodyEasyBox>(0.1, 0.1, 0.1, 1000, true, false);
    piston->SetPos(ChVector<>(0.6, 0.1, 0));
    my_system.Add(piston);

    // Joints
    // Revolute joint between floor and crankshaft
    auto revolute_floor_crank = chrono_types::make_shared<ChLinkLockRevolute>();
    revolute_floor_crank->Initialize(floor, crank, ChCoordsys<>(ChVector<>(0, 0.1, 0), QUNIT));
    my_system.AddLink(revolute_floor_crank);

    // Revolute joint between crankshaft and connecting rod
    auto revolute_crank_connecting_rod = chrono_types::make_shared<ChLinkLockRevolute>();
    revolute_crank_connecting_rod->Initialize(crank, connecting_rod, ChCoordsys<>(ChVector<>(0.1, 0.1, 0), QUNIT));
    my_system.AddLink(revolute_crank_connecting_rod);

    // Revolute joint between connecting rod and piston
    auto revolute_connecting_rod_piston = chrono_types::make_shared<ChLinkLockRevolute>();
    revolute_connecting_rod_piston->Initialize(connecting_rod, piston, ChCoordsys<>(ChVector<>(0.6, 0.1, 0), QUNIT));
    my_system.AddLink(revolute_connecting_rod_piston);

    // Prismatic joint between piston and floor (along X-axis)
    auto prismatic_piston_floor = chrono_types::make_shared<ChLinkLockPrismatic>();
    prismatic_piston_floor->Initialize(floor, piston, ChCoordsys<>(piston->GetPos(), QUNIT));
    my_system.AddLink(prismatic_piston_floor);

    // Motor to drive the crankshaft at constant angular speed
    auto motor = chrono_types::make_shared<ChLinkMotorRotationSpeed>();
    motor->Initialize(floor, crank, ChFrame<>(ChVector<>(0, 0.1, 0), QUNIT));
    auto mfun = chrono_types::make_shared<ChFunction_Const>(CH_C_PI); // Angular speed: π rad/s
    motor->SetSpeedFunction(mfun);
    my_system.AddLink(motor);

    // Set up Irrlicht visualization
    ChIrrApp application(&my_system, L"Crank-Slider Mechanism", core::dimension2d<u32>(800, 600));
    application.AddTypicalLogo();
    application.AddTypicalSky();
    application.AddTypicalLights();
    application.AddCamera(core::vector3df(0, 0.5, -1.5), core::vector3df(0, 0.5, 0));

    // Apply textures
    auto red = chrono_types::make_shared<ChTexture>();
    red->SetTextureFilename(GetChronoDataFile("textures/red.png"));
    crank->AddAsset(red);

    auto gray = chrono_types::make_shared<ChTexture>();
    gray->SetTextureFilename(GetChronoDataFile("textures/gray.png"));
    connecting_rod->AddAsset(gray);
    floor->AddAsset(gray);

    auto blue = chrono_types::make_shared<ChTexture>();
    blue->SetTextureFilename(GetChronoDataFile("textures/blue.png"));
    piston->AddAsset(blue);

    application.AssetBindAll();
    application.AssetUpdateAll();

    // Simulation loop
    while (application.GetDevice()->run()) {
        application.BeginScene();
        application.DrawAll();
        application.DoStep();
        application.EndScene();
    }

    return 0;
}
```

---

### 📌 Notes

- **Coordinate System**: The simulation is set up in the X-Y plane with the Y-axis as vertical (default in PyChrono).
- **Joints**:
  - Revolute joints allow rotation around the Z-axis.
  - Prismatic joint allows translation along the X-axis.
- **Motor**: A `ChLinkMotorRotationSpeed` is used to impose a constant angular velocity on the crankshaft.
- **Visualization**:
  - Custom camera position and target.
  - Textures applied to differentiate components.
  - Sky and lighting for realism.
