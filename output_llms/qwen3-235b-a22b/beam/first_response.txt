```python
import pychrono as chrono
import pychrono.fea as fea
import pychrono.irrlicht as irrlicht

# Create the physical system
system = chrono.ChSystemNSC()

# Create the FEA mesh
mesh = fea.ChMesh()

# Material properties (Steel)
E = 2e11  # Young's modulus in Pa
nu = 0.3   # Poisson ratio
rho = 7800 # Density in kg/m^3
area = 0.01 # Cross-sectional area in m^2
Iy = 0.0001 # Area moment of inertia in m^4 (Y-axis)
Iz = 0.0001 # Area moment of inertia in Z-axis
J = 0.0001  # Torsion constant in m^4

# Create beam material
material = fea.ChMaterialBeam()
material.SetYoungModulus(E)
material.SetPoissonRatio(nu)
material.SetDensity(rho)
material.SetArea(area)
material.SetIyy(Iy)
material.SetIzz(Iz)
material.SetJ(J)

# Beam geometry
L = 1.0  # Length in meters
num_elements = 10
dx = L / num_elements

# Create nodes
nodes = []
for i in range(num_elements + 1):
    pos = chrono.ChVectorD(i * dx, 0, 0)
    node = fea.ChNodeFEAxyz(pos)
    nodes.append(node)
    mesh.AddNode(node)

# Fix the first node
nodes[0].SetFixed(True)

# Create beam elements
for i in range(num_elements):
    element = fea.ChElementBeamEuler()
    element.SetNodes(nodes[i], nodes[i+1])
    element.SetMaterial(material)
    mesh.AddElement(element)

# Add the mesh to the system
system.Add(mesh)

# Add visualization assets for the mesh
vis = fea.ChVisualizationFEAmesh(mesh)
vis.SetFEMdataType(fea.ChVisualizationFEAmesh.E_PLOT_BEAM)
vis.SetBeamRadius(0.01)  # Visual radius of the beam elements
mesh.AddAsset(vis)
mesh.AddAsset(chrono.ChColorAsset(chrono.ChColor(0.4, 0.4, 1.0)))  # Blue color

# Set up Irrlicht visualization
application = irrlicht.ChIrrApp(system, 'PyChrono Beam Simulation', irrlicht.dimension2du(1024, 768))
application.AddTypicalLogo()
application.AddTypicalSky()
application.AddTypicalLights(irrlicht.vector3df(30, 100, 30))
application.AddTypicalCamera(chrono.ChVectorD(0, 1, -3), chrono.ChVectorD(0, 0, 0))

# Bind and update assets
application.AssetBindAll()
application.AssetUpdateAll()

# Simulation loop
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    system.DoStepDynamics(0.01)
    application.EndScene()
```
