```python
import pychrono as chrono
import pychrono.fea as fea
import pychrono.pardisomkl as mkl
import pychrono.irrlicht as chronoirr

print("Example: PyChrono using beam finite elements with Euler-Bernoulli formulation")

# Create the physical system
sys = chrono.ChSystemSMC()

# Create a mesh container
mesh = fea.ChMesh()

# Create a beam section with Euler-Bernoulli properties
msection = fea.ChBeamSectionEulerAdvanced()
beam_wy = 0.012
beam_wz = 0.025
msection.SetAsRectangularSection(beam_wy, beam_wz)
msection.SetYoungModulus(0.01e9)
msection.SetShearModulus(0.01e9 * 0.3)
msection.SetRayleighDamping(0.0)
msection.SetCentroid(0, 0.02)
msection.SetShearCenter(0, 0.1)
# Corrected rotation to degrees (was erroneously converted from radians)
msection.SetSectionRotation(45.0)

# Instruction 1 & 3: Use ChBuilderBeamEuler for beam creation
print("Creating beam using ChBuilderBeamEuler helper class")
builder = fea.ChBuilderBeamEuler()
builder.BuildBeam(
    mesh,
    msection,
    5,  # Number of elements
    chrono.ChVector3d(0, 0, -0.1),  # Start point A
    chrono.ChVector3d(0.2, 0, -0.1),  # End point B
    chrono.ChVector3d(0, 1, 0)  # Y-up direction
)

# Instruction 3: Fix last node and apply force to first node
nodes = builder.GetLastBeamNodes()
first_node = nodes[0]
last_node = nodes[-1]

# Apply force to the first node
first_node.SetForce(chrono.ChVector3d(0, -1, 0))

# Instruction 2: Replace direct node fixing with constraints
mtruss = chrono.ChBody()
mtruss.SetFixed(True)
sys.Add(mtruss)

# Constraint to fix the last node (replaces SetFixed(True))
constr_last = chrono.ChLinkMateGeneric()
constr_last.Initialize(last_node, mtruss, False, last_node.Frame(), chrono.ChFrameD())
sys.Add(constr_last)
constr_last.SetConstrainedCoords(True, True, True, True, True, True)

# Disable automatic gravity for FEA elements
mesh.SetAutomaticGravity(False)

# Add the mesh to the system
sys.Add(mesh)

# Visualization settings (unchanged per instructions)
visualizebeamA = chrono.ChVisualShapeFEA(mesh)
visualizebeamA.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_ELEM_BEAM_MZ)
visualizebeamA.SetColorscaleMinMax(-0.4, 0.4)
visualizebeamA.SetSmoothFaces(True)
visualizebeamA.SetWireframe(False)
mesh.AddVisualShapeFEA(visualizebeamA)

visualizebeamC = chrono.ChVisualShapeFEA(mesh)
visualizebeamC.SetFEMglyphType(chrono.ChVisualShapeFEA.GlyphType_NODE_CSYS)
visualizebeamC.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_NONE)
visualizebeamC.SetSymbolsThickness(0.006)
visualizebeamC.SetSymbolsScale(0.01)
visualizebeamC.SetZbufferHide(False)
mesh.AddVisualShapeFEA(visualizebeamC)

# Irrlicht visualization setup
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('FEA beams with Euler-Bernoulli formulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0.1, 0.1, 0.2))
vis.AddTypicalLights()

# Use MKL Pardiso solver for precision
msolver = mkl.ChSolverPardisoMKL()
sys.SetSolver(msolver)

# Simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(0.001)
```
