Here is a well-documented PyChrono simulation script that models gravitational attraction between particles with random shapes, positions, velocities, and orientations. The simulation uses the PyChrono physics engine and visualizes the results using the Irrlicht engine. A particle emitter dynamically generates new particles over time, and gravitational forces are applied dynamically during each simulation step.

---

### 🧠 **Key Features Implemented**
- **Random Particle Generation**: Particles are created with random shapes (sphere, box, cylinder), positions, velocities, and orientations.
- **Gravitational Attraction**: A custom gravitational force is applied between all pairs of particles using Newton's law of universal gravitation.
- **Dynamic Emitter**: Particles are emitted periodically during the simulation.
- **Visualization**: The simulation is visualized using the Irrlicht engine with custom colors for each particle.

---

### 📜 **Python Script**

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import random
import math

# Initialize the physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))  # Disable default gravity

# Set up Irrlicht visualization
application = chronoirr.ChIrrApp(system, 'Gravitational Particle Attraction', chronoirr.dimension2du(1024, 768))
application.AddTypicalLogo()
application.AddTypicalCamera(chronoirr.vector3df(0, 0, 10))
application.AddTypicalLights()
application.AssetBindAll()
application.AssetUpdateAll()

# List to store all particles
particles = []

# Gravitational constant (tune for visual effect)
G = 0.01

def create_particle():
    """Create a new particle with random shape, position, velocity, and orientation."""
    body = chrono.ChBody()

    # Random position in a cube
    pos = chrono.ChVectorD(
        random.uniform(-5, 5),
        random.uniform(-5, 5),
        random.uniform(-5, 5)
    )
    body.SetPos(pos)

    # Random velocity
    vel = chrono.ChVectorD(
        random.uniform(-1, 1),
        random.uniform(-1, 1),
        random.uniform(-1, 1)
    )
    body.SetPos_dt(vel)

    # Random orientation using axis-angle
    axis = chrono.ChVectorD(
        random.uniform(-1, 1),
        random.uniform(-1, 1),
        random.uniform(-1, 1)
    )
    axis.Normalize()
    angle = random.uniform(0, 2 * math.pi)
    q = chrono.ChQuaternionD()
    q.Q_from_AngAxis(angle, axis)
    body.SetRot(q)

    # Random mass
    mass = random.uniform(0.5, 2.0)
    body.SetMass(mass)

    # Simple isotropic inertia
    body.SetInertiaXX(chrono.ChVectorD(1, 1, 1))

    # Random shape
    shape_type = random.choice(['sphere', 'box', 'cylinder'])
    if shape_type == 'sphere':
        radius = random.uniform(0.2, 0.5)
        sphere = chrono.ChSphereShape()
        sphere.GetSphereGeometry().rad = radius
        body.AddAsset(sphere)
    elif shape_type == 'box':
        box_size = chrono.ChVectorD(
            random.uniform(0.2, 0.5),
            random.uniform(0.2, 0.5),
            random.uniform(0.2, 0.5)
        )
        box = chrono.ChBoxShape()
        box.GetBoxGeometry().Size = box_size
        body.AddAsset(box)
    elif shape_type == 'cylinder':
        radius = random.uniform(0.1, 0.3)
        height = random.uniform(0.2, 0.6)
        cylinder = chrono.ChCylinderShape()
        cylinder.GetCylinderGeometry().rad = radius
        cylinder.GetCylinderGeometry().p1 = chrono.ChVectorD(0, -height / 2, 0)
        cylinder.GetCylinderGeometry().p2 = chrono.ChVectorD(0, height / 2, 0)
        body.AddAsset(cylinder)

    # Random color
    color = chrono.ChColorAsset()
    color.SetColor(chrono.ChColor(random.random(), random.random(), random.random()))
    body.AddAsset(color)

    # Add to system and list
    system.Add(body)
    particles.append(body)

# Create initial particles
for _ in range(10):
    create_particle()

# Emitter settings
emitter_interval = 1.0  # Emit a new particle every 1 second
last_emit_time = 0.0

# Set simulation time step
application.SetTimestep(0.01)
application.SetTryRealtime(True)

# Simulation loop
while application.GetDevice().run():
    current_time = system.GetChTime()

    # Emit new particles
    if current_time > last_emit_time + emitter_interval:
        create_particle()
        last_emit_time = current_time

    # Apply gravitational forces
    for bodyA in particles:
        force = chrono.ChVectorD(0, 0, 0)
        for bodyB in particles:
            if bodyA == bodyB:
                continue
            posA = bodyA.GetPos()
            posB = bodyB.GetPos()
            r_vec = posB - posA
            distance_sq = r_vec.x**2 + r_vec.y**2 + r_vec.z**2
            if distance_sq < 0.001:  # Avoid singularity
                continue
            distance = math.sqrt(distance_sq)
            force_mag = G * bodyA.GetMass() * bodyB.GetMass() / distance_sq
            force_dir = r_vec / distance
            force += force_dir * force_mag
        bodyA.SetForce(force)

    # Step the simulation
    application.BeginScene()
    application.DrawAll()
    system.DoStepDynamics(application.GetTimestep())
    application.EndScene()
```

---

### 📌 **Notes and Recommendations**
- **Gravitational Constant (`G`)**: Adjust this value to control the strength of the gravitational pull. A higher value will cause faster attraction.
- **Emitter Interval**: You can change `emitter_interval` to control how often new particles are added.
- **Inertia**: The inertia matrix is simplified for demonstration. For more accurate simulations, you can compute inertia based on the actual shape and mass.
- **Collision Handling**: This script assumes no collisions between particles. If needed, you can enable collision detection and response using `ChCollisionModel`.

This script provides a complete and visually engaging simulation of gravitational attraction between particles with randomized properties.
