```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import pychrono.driver as driver

# Initialize the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

# Create the M113 vehicle
m113 = veh.M113()
m113.SetContactMethod(chrono.ChContactMethod_NSC)
m113.SetChassisFixed(False)
m113.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 1.2), chrono.QUNIT))
m113.Initialize(system)

# Create the rigid terrain
terrain = veh.RigidTerrain(system)
patch_mat = chrono.ChMaterialSurfaceNSC()
patch_mat.SetFriction(0.8)
patch_mat.SetRestitution(0.2)
patch = terrain.AddPatch(patch_mat, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(200, 200, 1))
patch.SetTexture(chrono.GetChronoDataFile("terrain/textures/grass.jpg"))
terrain.Initialize()

# Create the Irrlicht application
application = irr.ChIrrApp(system, 'M113 Simulation', irr.dimension2du(1024, 768))
application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(chrono.Vector(0, -5, 2), m113.GetChassisBody().GetPos())

# Create the driver system
driver = veh.ChIrrGuiDriver(application.GetDevice(), m113.GetVehicle(), "M113 Driver")
driver.SetSteeringDelta(0.05)
driver.SetThrottleDelta(0.2)
driver.SetBrakingDelta(0.2)
driver.Initialize()

# Set up the simulation loop
step_size = 0.01
time = 0
timer = chrono.ChRealtimeStepTimer()

while application.GetDevice().run():
    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Synchronize the vehicle, terrain, and driver
    m113.Synchronize(time, driver_inputs, terrain)
    terrain.Synchronize(time)
    driver.Synchronize(time)

    # Update visualization
    application.BeginScene()
    application.DrawAll()
    application.EndScene()

    # Advance the simulation
    system.DoStepDynamics(step_size)
    time += step_size

    # Maintain real-time
    timer.Spin(step_size)
```
