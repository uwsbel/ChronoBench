```python
import chrono
import chrono.irrlicht as chronoirr
import chrono.vehicle as veh
import math

# ---
#  Initialize the PyChrono environment and core components.
# ---

# Create a Chrono physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# ---
# Add the required physical systems and objects as specified.
# ---

# Create the terrain
terrain = chrono.ChRigidBody()
terrain.SetBodyFixed(True)
terrain.SetMass(0)
terrain.AddAsset(chrono.ChTriangleMeshConnected())
terrain.GetAsset().LoadFromFile("terrain.obj")  # Replace with your terrain file
terrain.SetCollide(True)
system.Add(terrain)

# Create the vehicle
bus = veh.ChBus()
bus.SetBodyFixed(False)
bus.SetMass(12000)
bus.SetPos(chrono.ChVectorD(0, 1, 0))
bus.SetTransform(chrono.ChQuaternionD(0, 0, 0))
system.Add(bus)

# ---
# Set necessary default parameters such as positions, forces, and interactions.
# ---

# Set tire model
bus.GetVehicleModel().SetTireModel(veh.ChTireModel_TMeasy)

# Set initial speed
bus.SetForwardSpeed(0)

# ---
# Create an interactive driver system that allows for control of the vehicle’s steering, throttle, and braking.
# ---

# Create a driver system
driver = veh.ChDriver()
driver.SetSteering(0)
driver.SetThrottle(0)
driver.SetBraking(0)
bus.SetDriver(driver)

# ---
# Irrlicht visualization setup
# ---

# Create an Irrlicht application
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('CityBus Simulation')
vis.Initialize()
vis.AddCamera(chrono.ChVectorD(0, 10, -10))
vis.AddTypicalLights()

# ---
# Simulation loop
# ---

# Set simulation parameters
time_step = 0.005
simulation_fps = 50
max_simulation_time = 60

# Main simulation loop
simulation_time = 0
while vis.Run() and simulation_time < max_simulation_time:
    # Update driver controls (example - you would replace this with input handling)
    if simulation_time < 10:
        driver.SetThrottle(1)
    else:
        driver.SetThrottle(0)
        driver.SetBraking(0.5)

    # Advance the simulation
    system.DoStepDynamics(time_step)
    simulation_time += time_step

    # Update the camera to follow the bus
    camera_pos = bus.GetPos() + chrono.ChVectorD(0, 5, -10)
    vis.GetRenderDevice().viewerManager.GetCamera().SetPosition(camera_pos)
    vis.GetRenderDevice().viewerManager.GetCamera().LookAt(bus.GetPos(), chrono.ChVectorD(0, 1, 0))

    # Update the visualization
    vis.DrawMultithreaded(True)
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

# Close the visualization
vis.GetRenderDevice().closeDevice()
```
