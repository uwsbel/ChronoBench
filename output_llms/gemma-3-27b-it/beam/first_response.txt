```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# ---
#  Create the simulation system and add items
# ---

# Create a Chrono physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# Create a simple beam mesh
# Define node coordinates
nodes = [
    chrono.ChVectorD(0, 0, 0),
    chrono.ChVectorD(1, 0, 0),
    chrono.ChVectorD(2, 0, 0),
    chrono.ChVectorD(0, 1, 0),
    chrono.ChVectorD(1, 1, 0),
    chrono.ChVectorD(2, 1, 0)
]

# Define beam elements (pairs of node indices)
beams = [
    (0, 1),
    (1, 2),
    (0, 3),
    (3, 4),
    (4, 5),
    (1, 4)
]

# Create a beam mesh
beam_mesh = chrono.ChBeamMesh()
beam_mesh.Set_Nodes(nodes)
beam_mesh.Set_Beams(beams)

# Set beam properties
beam_mesh.Set_Young_Modulus(2.0e11)  # Young's modulus (Pa)
beam_mesh.Set_Poisson_Ratio(0.3)  # Poisson's ratio
beam_mesh.Set_Cross_Section_Area(0.01)  # Cross-section area (m^2)
beam_mesh.Set_Moment_Of_Inertia(0.0001)  # Moment of inertia (m^4)
beam_mesh.Set_Density(7850)  # Density (kg/m^3)

# Create a ChBody from the beam mesh
body = chrono.ChBodyEasy()
body.AddAsset(beam_mesh)
body.SetBodyFixed(False)
body.SetMass(1.0)
body.SetPos(chrono.ChVectorD(0, 0, 0))
system.Add(body)

# Add a force at the end of the beam
force_point = chrono.ChVectorD(2, 1, 0)
force_magnitude = 100
force_direction = chrono.ChVectorD(1, 0, 0)
force = chrono.ChForce()
force.SetPos(force_point)
force.SetForce(force_direction * force_magnitude)
force.SetApplyOnBody(True)
body.AddForce(force)

# ---
#  Create an Irrlicht application to visualize the system
# ---

vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('PyChrono Beam Example')
vis.Initialize()
vis.AddCamera(chrono.ChVectorD(0, 2, -3))
vis.AddTypicalLights()

# ---
#  Run the simulation
# ---

system.SetTimestepperType(chrono.ChTimestepper.Type_EulerExplicit)
system.SetStep(0.001)
time_end = 5.0

while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    system.DoStepDynamics()
    if system.GetChTime() > time_end:
        break

vis.GetIrrlichtApplication().Close()
```
