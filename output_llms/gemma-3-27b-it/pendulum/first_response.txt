```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import math

# ---
#  Create the simulation system and add items
# ---

# Create a Chrono physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# Create the ground body
body_ground = chrono.ChBodyEasy()
body_ground.SetBodyFixed(True)
body_ground.SetPos(chrono.ChVectorD(0, -1, 0))
body_ground.SetShapeBoxes(0.5, 0.1, 0.5)
body_ground.AddAsset(chrono.ChColor(0.2, 0.2, 0.2))
system.Add(body_ground)

# Create the pendulum body
body_pendulum = chrono.ChBodyEasy()
body_pendulum.SetMass(1.0)
body_pendulum.SetPos(chrono.ChVectorD(0, 0, 0))
body_pendulum.SetShapeCylinder(0.1, 0.5)
body_pendulum.AddAsset(chrono.ChColor(0.8, 0.2, 0.2))

# Calculate inertia tensor (assuming cylinder along Z axis)
inertia = chrono.ChMatrix33()
inertia.SetDiagonal([1/12 * body_pendulum.GetMass() * (3*0.1*0.1 + 0.5*0.5),
                     1/12 * body_pendulum.GetMass() * (3*0.1*0.1 + 0.5*0.5),
                     1/2 * body_pendulum.GetMass() * 0.1*0.1])
body_pendulum.SetInertia(inertia)

system.Add(body_pendulum)

# Create a revolute joint to connect the pendulum to the ground
joint = chrono.ChLinkRevolute()
joint.Initialize(body_ground, body_pendulum, chrono.ChCoordsys(chrono.ChVectorD(0, 0, 0)))
system.Add(joint)

# ---
#  Create an Irrlicht application to visualize the system
# ---

vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Simple Pendulum')
vis.Initialize()
vis.AddTypicalCamera(chrono.ChVectorD(0, 1, -2), chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()

# ---
#  Simulation loop
# ---

time_step = 0.01
time = 0.0

while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    system.DoStepDynamics(time_step)
    time += time_step

    # Log pendulum position and velocity every 10 steps
    if int(time / time_step) % 10 == 0:
        pos = body_pendulum.GetPos()
        vel = body_pendulum.GetPos_WVEL()
        print(f"Time: {time:.2f}, Pendulum Position: {pos.x:.2f}, {pos.y:.2f}, {pos.z:.2f}, Velocity: {vel.x:.2f}, {vel.y:.2f}, {vel.z:.2f}")
```
