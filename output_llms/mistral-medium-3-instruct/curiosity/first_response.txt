# PyChrono Simulation: Curiosity Rover on Rigid Terrain

Below is a complete PyChrono simulation script that creates a Curiosity rover navigating on rigid terrain with all the requested features:

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.driver as driver
import numpy as np
from math import pi

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoOutputMode(chrono.ChOutputMode.VERBOSE)
chrono.SetNumThreads(4)

# Create the physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# Set collision system parameters
collision_system = chrono.ChCollisionSystemBullet()
system.SetCollisionSystem(collision_system)

# 2. Add the required physical systems and objects

# Create the ground
ground = chrono.ChBody()
ground.SetBodyFixed(True)
ground.SetCollide(True)

# Create ground geometry
ground_geometry = chrono.ChBoxShape()
ground_geometry.GetBoxGeometry().SetLengths(chrono.ChVectorD(20, 0.1, 20))
ground.AddAsset(ground_geometry)

# Create ground visualization
ground_visual = chrono.ChVisualization()
ground_visual.SetModel(ground_geometry)
ground_visual.SetTexture(chrono.GetChronoDataFile("textures/terrain.jpg"))
ground.AddAsset(ground_visual)

# Add ground to system
system.Add(ground)

# Create the rover chassis
chassis = chrono.ChBody()
chassis.SetMass(200)
chassis.SetInertiaXX(chrono.ChVectorD(20, 20, 20))
chassis.SetPos(chrono.ChVectorD(0, 0.5, 0))
chassis.SetCollide(True)

# Create chassis geometry
chassis_geometry = chrono.ChBoxShape()
chassis_geometry.GetBoxGeometry().SetLengths(chrono.ChVectorD(1.5, 0.4, 1.0))
chassis.AddAsset(chassis_geometry)

# Create chassis visualization
chassis_visual = chrono.ChVisualization()
chassis_visual.SetModel(chassis_geometry)
chassis_visual.SetTexture(chrono.GetChronoDataFile("textures/metal.jpg"))
chassis.AddAsset(chassis_visual)

# Add chassis to system
system.Add(chassis)

# Create wheels (6 wheels for Curiosity-like rover)
wheel_radius = 0.25
wheel_width = 0.15
wheel_positions = [
    chrono.ChVectorD(0.8, 0.3, 0.5),   # Front right
    chrono.ChVectorD(0.8, 0.3, -0.5),  # Front left
    chrono.ChVectorD(-0.8, 0.3, 0.5),  # Rear right
    chrono.ChVectorD(-0.8, 0.3, -0.5), # Rear left
    chrono.ChVectorD(0, 0.3, 0.5),     # Middle right
    chrono.ChVectorD(0, 0.3, -0.5)     # Middle left
]

wheels = []
for i, pos in enumerate(wheel_positions):
    wheel = chrono.ChBody()
    wheel.SetMass(10)
    wheel.SetInertiaXX(chrono.ChVectorD(0.1, 0.1, 0.1))
    wheel.SetPos(pos)
    wheel.SetCollide(True)

    # Create wheel geometry
    wheel_geometry = chrono.ChCylinderShape()
    wheel_geometry.GetCylinderGeometry().rad = wheel_radius
    wheel_geometry.GetCylinderGeometry().p1 = chrono.ChVectorD(0, 0, -wheel_width/2)
    wheel_geometry.GetCylinderGeometry().p2 = chrono.ChVectorD(0, 0, wheel_width/2)
    wheel.AddAsset(wheel_geometry)

    # Create wheel visualization
    wheel_visual = chrono.ChVisualization()
    wheel_visual.SetModel(wheel_geometry)
    wheel_visual.SetTexture(chrono.GetChronoDataFile("textures/rubber.jpg"))
    wheel.AddAsset(wheel_visual)

    # Add wheel to system
    system.Add(wheel)
    wheels.append(wheel)

# Create suspension system (simplified)
for i, wheel in enumerate(wheels):
    # Create a revolute joint between chassis and wheel
    joint = chrono.ChLinkLockRevolute()
    joint.Initialize(chassis, wheel,
                    chrono.ChCoordinatorSlerp(1, 0, 0, 0),
                    chrono.ChCoordinatorSlerp(0, 1, 0, 0))
    joint.SetLimitAngle(0, pi/4)  # Limit steering angle
    system.Add(joint)

    # Add a motor to each wheel
    motor = chrono.ChLinkMotorRotationSpeed()
    motor.Initialize(wheel, chassis, chrono.ChFrameD(wheel.GetPos()))
    motor.SetSpeedFunction(chrono.ChFunction_Const(0))  # Initially stopped
    system.Add(motor)

# 3. Set necessary default parameters

# Create the visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle("Curiosity Rover Simulation")
vis.SetUseSkyBox(True)
vis.SetSkyBoxTextures(
    chrono.GetChronoDataFile("skybox/right.jpg"),
    chrono.GetChronoDataFile("skybox/left.jpg"),
    chrono.GetChronoDataFile("skybox/top.jpg"),
    chrono.GetChronoDataFile("skybox/bottom.jpg"),
    chrono.GetChronoDataFile("skybox/front.jpg"),
    chrono.GetChronoDataFile("skybox/back.jpg")
)
vis.SetCameraVertical(chrono.ChVectorD(0, 1.5, 0))
vis.SetCameraPointing(chrono.ChVectorD(0, 0.5, 0))
vis.SetCameraMoveScale(0.01)
vis.SetLightIntensity(1.5)
vis.SetLightDirection(chrono.ChVectorD(1, -1, 1))
vis.SetShadows(True)
vis.SetLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.SetLogoPosition(0.9, 0.9)
vis.SetLogoSize(0.1)
vis.AttachSystem(system)

# Create a driver for motor control
class RoverDriver(driver.ChDriver):
    def __init__(self):
        super().__init__("Rover Driver")
        self.time = 0
        self.steering_angle = 0
        self.speed = 0

    def OnInitialize(self):
        print("Rover driver initialized")

    def OnStep(self, time):
        self.time = time

        # Simple control logic - move forward and turn slightly
        if time < 5:
            self.speed = 1.0
            self.steering_angle = 0.1
        elif time < 10:
            self.speed = 1.0
            self.steering_angle = -0.1
        else:
            self.speed = 0
            self.steering_angle = 0

        # Apply to all wheels
        for i, wheel in enumerate(wheels):
            # Find the motor for this wheel
            for link in system.Get_linklist():
                if isinstance(link, chrono.ChLinkMotorRotationSpeed):
                    if link.GetBody1() == wheel or link.GetBody2() == wheel:
                        # Set speed (positive for forward, negative for backward)
                        link.SetSpeedFunction(chrono.ChFunction_Const(self.speed))

                        # For steering wheels (front and rear), apply steering angle
                        if i in [0, 1, 2, 3]:  # Front and rear wheels
                            # Get the revolute joint for this wheel
                            for joint in system.Get_linklist():
                                if isinstance(joint, chrono.ChLinkLockRevolute):
                                    if joint.GetBody1() == wheel or joint.GetBody2() == wheel:
                                        joint.SetLimitAngle(0, self.steering_angle)

# Create and add the driver to the system
rover_driver = RoverDriver()
system.AddDriver(rover_driver)

# Simulation loop
time_step = 0.01
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(5, 3, 0), chrono.ChVectorD(0, 0.5, 0))
vis.AddLight(chrono.ChVectorD(5, 5, 5), chrono.ChVectorD(0, 0, 0), 10)

while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance simulation by one step
    system.DoStepDynamics(time_step)
```

## Key Features of This Simulation:

1. **Physical System Setup**:
   - NSC (Non-Smooth Contact) solver for accurate contact dynamics
   - Bullet collision system for robust collision detection

2. **Rover Model**:
   - Main chassis with realistic mass and inertia properties
   - Six wheels with proper geometry and collision properties
   - Revolute joints for wheel attachment with steering limits
   - Motor control for each wheel

3. **Terrain**:
   - Fixed ground plane with collision properties
   - Textured visualization for realism

4. **Visualization**:
   - Irrlicht visualization system with customizable camera
   - Skybox for immersive environment
   - Shadows and lighting effects
   - Textures for realistic appearance
   - PyChrono logo overlay

5. **Control System**:
   - Custom driver class for motor control
   - Simple steering logic that changes over time
   - Speed and steering angle control for all wheels

6. **Simulation Parameters**:
   - Realistic gravity setting
   - Proper time stepping for stable simulation

To run this simulation, you'll need to have PyChrono installed with all its dependencies. The script assumes you have the standard PyChrono data files (textures, skybox images) available in your environment.
