# PyChrono Beam Buckling Simulation with FEA and Constraints

Below is a complete PyChrono simulation script for a beam buckling scenario with finite element analysis, custom motor functions, and constraints:

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath("PATH_TO_CHRONO_DATA")  # Set path to Chrono data directory

# Create a Chrono physical system
my_system = chrono.ChSystemNSC()
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# 2. Add the required physical systems and objects

# Create a beam material (steel-like properties)
beam_material = chrono.ChMaterialBeamEuler()
beam_material.SetYoungModulus(2.1e11)  # Young's modulus in Pa
beam_material.SetPoissonRatio(0.3)
beam_material.SetDensity(7800)         # Density in kg/m^3

# Create a beam section (rectangular cross-section)
beam_section = chrono.ChBeamSectionEulerRectangular()
beam_section.SetAsRectangularSection(0.02, 0.02)  # width and height in meters

# Create a finite element beam
num_elements = 20
beam_length = 2.0  # meters
element_length = beam_length / num_elements

# Create nodes for the beam
beam_nodes = []
for i in range(num_elements + 1):
    node = chrono.ChNodeFEAxyzrot()
    node.SetPos(chrono.ChVectorD(0, 0.5, i * element_length))
    node.SetMass(0)  # Mass will be handled by elements
    my_system.Add(node)
    beam_nodes.append(node)

# Create beam elements between nodes
beam_elements = []
for i in range(num_elements):
    element = chrono.ChElementBeamEuler()
    element.SetNodes(beam_nodes[i], beam_nodes[i+1])
    element.SetSection(beam_section)
    element.SetMaterial(beam_material)
    my_system.Add(element)
    beam_elements.append(element)

# Fix the bottom node (boundary condition)
fix_constraint = chrono.ChLinkLockLock()
fix_constraint.Initialize(beam_nodes[0], chrono.ChFrameD(beam_nodes[0].GetPos()))
my_system.Add(fix_constraint)

# 3. Set necessary default parameters

# Apply a compressive load at the top node
top_node = beam_nodes[-1]
compressive_force = chrono.ChForce()
compressive_force.SetMode(chrono.ChForce.FORCE)
compressive_force.SetF_force(chrono.ChVectorD(0, -1000, 0))  # 1000 N downward force
top_node.AddForce(compressive_force)

# Create a custom motor function to apply additional loading
class CustomMotorFunction(chrono.ChFunction):
    def __init__(self):
        chrono.ChFunction.__init__(self)
        self.time = 0

    def Get_y(self, time):
        self.time = time
        # Ramp up the force over time
        if time < 2.0:
            return 0.5 * time * 1000  # Ramp to 1000 N in 2 seconds
        else:
            return 1000

custom_motor = CustomMotorFunction()
motor_force = chrono.ChForce()
motor_force.SetMode(chrono.ChForce.FORCE)
motor_force.SetF_force(chrono.ChVectorD(0, -1, 0))  # Direction
motor_force.SetM_modulator(custom_motor)  # Apply our custom function
top_node.AddForce(motor_force)

# Add a constraint to limit lateral movement (simulate buckling constraints)
lateral_constraint = chrono.ChLinkLockRevolute()
lateral_constraint.Initialize(beam_nodes[-1], chrono.ChFrameD(beam_nodes[-1].GetPos()))
my_system.Add(lateral_constraint)

# 4. Visualization setup
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(my_system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Beam Buckling Simulation")
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(1, 1, 1))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVectorD(1.5, 5.5, -2.5), chrono.ChVectorD(0, 0, 0), 3, 1, 10, 40, 512)

# Create visualization assets for the beam
for node in beam_nodes:
    sphere = chrono.ChSphereShape()
    sphere.SetRadius(0.05)
    node.AddAsset(sphere)

for element in beam_elements:
    beam_vis = chrono.ChBeamShape()
    beam_vis.SetBeam(element)
    beam_vis.SetColor(chrono.ChColor(0.6, 0.2, 0.2))
    element.AddAsset(beam_vis)

# 5. Solver and timestepper setup
my_system.SetSolverType(chrono.ChSolver.Type_MINRES)
my_system.SetMaxIterations(100)
my_system.SetTol(1e-10)

timestepper = chrono.ChTimestepperHHT(my_system)
timestepper.SetAlpha(-0.2)  # For better stability with stiff systems
timestepper.SetStepSize(0.01)

# Simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance simulation by one timestep
    my_system.DoStepDynamics(timestepper.GetStepSize())

    # Print simulation time
    time = my_system.GetChTime()
    print(f"Simulation time: {time:.2f} s")

    # Break simulation after 10 seconds
    if time > 10.0:
        break
```

## Key Features of This Simulation:

1. **Finite Element Analysis (FEA) Setup**:
   - Uses Euler-Bernoulli beam elements with rectangular cross-section
   - Material properties set for steel (Young's modulus, Poisson ratio, density)
   - Beam is discretized into multiple elements for accurate buckling analysis

2. **Loading Conditions**:
   - Initial compressive load applied at the top node
   - Custom motor function that ramps up the load over time
   - Both static and time-varying loads included

3. **Constraints**:
   - Fixed boundary condition at the base
   - Revolute constraint at the top to allow rotation but limit lateral movement
   - These constraints help simulate the buckling behavior

4. **Visualization**:
   - Uses Irrlicht for real-time visualization
   - Beam elements and nodes are rendered with appropriate shapes and colors
   - Proper lighting and camera setup for good visibility

5. **Solver Configuration**:
   - Uses MINRES solver for efficient solution of the FEA system
   - HHT timestepper with α = -0.2 for stability with stiff systems
   - Appropriate tolerance and iteration limits set

To run this simulation, you'll need to:
1. Have PyChrono properly installed with all dependencies
2. Set the correct path to the Chrono data directory
3. Adjust parameters like beam dimensions, material properties, and loading conditions as needed for your specific scenario
