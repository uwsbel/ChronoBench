# PyChrono Simulation with ROS Integration

Below is a complete PyChrono simulation script that integrates with ROS, following your specifications:

```python
#!/usr/bin/env python3

import pychrono as chrono
import pychrono.irrlich as chrono_irrlich
import rospy
from std_msgs.msg import Int32

# Custom ROS handler to publish integer messages
class CustomROSHandler(chrono.ChRosHandler):
    def __init__(self, topic_name, queue_size=10):
        super().__init__()
        self.pub = rospy.Publisher(topic_name, Int32, queue_size=queue_size)
        self.counter = 0

    def update(self, time):
        # Publish an incrementing integer message
        msg = Int32()
        msg.data = self.counter
        self.pub.publish(msg)
        self.counter += 1

def main():
    # Initialize ROS node
    rospy.init_node('chrono_ros_simulation', anonymous=True)

    # 1. Initialize the PyChrono environment and core components
    chrono.SetChronoDataPath('PATH_TO_CHRONO_DATA')  # Update this path

    # Create the Chrono physical system
    system = chrono.ChSystemNSC()
    system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravitational acceleration

    # Set collision system
    collision_system = chrono.ChCollisionSystemBullet()
    system.SetCollisionSystem(collision_system)

    # 2. Add physical systems and objects

    # Add a fixed floor
    floor_body = chrono.ChBodyEasyBox(10, 0.5, 10, 1000, True, True)
    floor_body.SetPos(chrono.ChVectorD(0, -1, 0))
    floor_body.SetBodyFixed(True)
    floor_body.GetVisualModel().SetColor(chrono.ChColor(0.5, 0.5, 0.5))
    system.Add(floor_body)

    # Add a movable box
    box_body = chrono.ChBodyEasyBox(1, 1, 1, 1000, True, True)
    box_body.SetPos(chrono.ChVectorD(0, 2, 0))
    box_body.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
    box_body.GetVisualModel().SetColor(chrono.ChColor(0.8, 0.2, 0.2))
    system.Add(box_body)

    # Set material properties
    floor_material = chrono.ChMaterialSurfaceNSC()
    floor_material.SetFriction(0.4)
    floor_body.GetMaterialSurfaceNSC().SetMaterial(floor_material)

    box_material = chrono.ChMaterialSurfaceNSC()
    box_material.SetFriction(0.4)
    box_body.GetMaterialSurfaceNSC().SetMaterial(box_material)

    # 3. Set up ROS manager and handlers
    ros_manager = chrono.ChRosManager()

    # Add standard handlers
    clock_handler = chrono.ChRosClockHandler()
    ros_manager.AddHandler(clock_handler)

    body_handler = chrono.ChRosBodyHandler()
    ros_manager.AddHandler(body_handler)

    transform_handler = chrono.ChRosTransformHandler()
    ros_manager.AddHandler(transform_handler)

    # Add our custom handler
    custom_handler = CustomROSHandler('/chrono_counter', queue_size=10)
    ros_manager.AddHandler(custom_handler)

    # Register bodies with ROS handlers
    body_handler.AddBody(box_body)

    # Create visualization
    app = chrono_irrlich.ChIrrApp(system, 'PyChrono ROS Simulation', chrono.dimensionalize(1280), chrono.dimensionalize(720))
    app.AddTypicalLogo()
    app.AddTypicalSky()
    app.AddTypicalLights()
    app.AddTypicalCamera(chrono.ChVectorD(0, 3, -5))
    app.AssetBindAll()
    app.AssetUpdateAll()

    # Simulation parameters
    time_step = 0.01
    max_time = 10.0
    real_time_factor = 1.0

    # Simulation loop
    while app.GetDevice().run() and not rospy.is_shutdown():
        current_time = system.GetChTime()

        # Update ROS
        ros_manager.Update(system.GetChTime())

        # Advance simulation
        app.BeginScene()
        app.DrawAll()
        app.DoStepDynamics(time_step)
        app.EndScene()

        # Maintain real-time execution
        if real_time_factor > 0:
            time_to_wait = time_step - (system.GetChTime() - current_time)
            if time_to_wait > 0:
                rospy.sleep(rospy.Duration(time_to_wait))

        if system.GetChTime() > max_time:
            break

if __name__ == '__main__':
    try:
        main()
    except rospy.ROSInterruptException:
        pass
```

## Key Features of This Implementation:

1. **PyChrono Environment Setup**:
   - Initializes the Chrono system with NSC (Non-Smooth Contact) solver
   - Sets up gravitational acceleration
   - Configures collision system using Bullet

2. **Physical Objects**:
   - Fixed floor (gray) with dimensions 10x0.5x10
   - Movable box (red) with dimensions 1x1x1 positioned above the floor
   - Material properties with friction set to 0.4

3. **ROS Integration**:
   - Custom ROS handler (`CustomROSHandler`) that publishes incrementing integers
   - Standard ROS handlers for clock, body, and transform data
   - Real-time synchronization between simulation and ROS

4. **Visualization**:
   - Uses Irrlicht for 3D rendering
   - Includes typical camera, lighting, and sky setup

5. **Simulation Loop**:
   - Advances simulation with proper time stepping
   - Maintains real-time execution
   - Updates ROS communication at each timestep

To use this script:
1. Make sure you have both PyChrono and ROS properly installed
2. Update the `PATH_TO_CHRONO_DATA` to point to your Chrono data directory
3. The script will publish integer messages to the `/chrono_counter` topic
4. You can visualize the simulation in the Irrlicht window

The simulation will run for 10 seconds (simulation time) by default, with the box falling onto the floor under gravity.
