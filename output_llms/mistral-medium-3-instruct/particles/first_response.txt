# PyChrono Simulation: Gravitational Particle Attraction

Below is a complete PyChrono script that models gravitational attraction between particles with random properties, visualized using Irrlicht:

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import numpy as np
import random

# =============================================================================
# Initialize PyChrono and set up the simulation
# =============================================================================

# Create a Chrono physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))  # Disable default gravity

# =============================================================================
# Simulation parameters
# =============================================================================

# Time step for the simulation
time_step = 0.01

# Number of particles to emit
num_particles = 50

# Gravitational constant (scaled for visualization)
G = 1.0

# Particle properties
min_radius = 0.1
max_radius = 0.3
min_mass = 0.1
max_mass = 1.0
min_velocity = 0.5
max_velocity = 2.0

# Simulation duration
sim_duration = 10.0

# =============================================================================
# Create visualization
# =============================================================================

# Create an Irrlicht application
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Gravitational Particle Attraction')
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(5, 5, 5))
vis.AddLight(chrono.ChVectorD(5, 5, 5), chrono.ChVectorD(1, 1, 1), 10)

# =============================================================================
# Custom force callback for gravitational attraction
# =============================================================================

class GravitationalForceCallback(chrono.ChForceCallback):
    def __init__(self, G):
        super().__init__()
        self.G = G

    def ForceCallback(self, time, body):
        # Calculate gravitational forces between all bodies
        force = chrono.ChVectorD(0, 0, 0)
        for other_body in system.Get_bodylist():
            if other_body != body:
                # Calculate distance vector
                delta = other_body.GetPos() - body.GetPos()
                distance = delta.Length()

                # Avoid division by zero and very small distances
                if distance > 0.01:
                    # Calculate gravitational force (F = G * m1 * m2 / r^2)
                    force_direction = delta.GetNormalized()
                    force_magnitude = self.G * body.GetMass() * other_body.GetMass() / (distance * distance)
                    force += force_direction * force_magnitude

        # Apply the calculated force
        body.ForceAccumulate(force)

# Create and add the force callback
gravity_callback = GravitationalForceCallback(G)
system.AddForceCallback(gravity_callback)

# =============================================================================
# Particle emitter function
# =============================================================================

def emit_particles(num_particles):
    particles = []

    for i in range(num_particles):
        # Create a new body
        body = chrono.ChBody()

        # Random radius
        radius = min_radius + (max_radius - min_radius) * random.random()

        # Random mass (proportional to volume)
        mass = min_mass + (max_mass - min_mass) * random.random()
        body.SetMass(mass)

        # Random position within a sphere
        theta = random.uniform(0, 2 * np.pi)
        phi = random.uniform(0, np.pi)
        r = random.uniform(2, 5)  # Distance from origin
        x = r * np.sin(phi) * np.cos(theta)
        y = r * np.sin(phi) * np.sin(theta)
        z = r * np.cos(phi)
        body.SetPos(chrono.ChVectorD(x, y, z))

        # Random velocity
        vx = random.uniform(-min_velocity, max_velocity)
        vy = random.uniform(-min_velocity, max_velocity)
        vz = random.uniform(-min_velocity, max_velocity)
        body.SetPos_dt(chrono.ChVectorD(vx, vy, vz))

        # Random rotation
        qw = random.random()
        qx = random.random()
        qy = random.random()
        qz = random.random()
        quat = chrono.ChQuaternionD(qw, qx, qy, qz)
        quat.Normalize()
        body.SetRot(quat)

        # Random angular velocity
        wx = random.uniform(-0.5, 0.5)
        wy = random.uniform(-0.5, 0.5)
        wz = random.uniform(-0.5, 0.5)
        body.SetWvel_loc(chrono.ChVectorD(wx, wy, wz))

        # Set collision properties
        body.SetCollide(True)
        body.GetCollisionModel().ClearModel()
        body.GetCollisionModel().AddSphere(radius)
        body.GetCollisionModel().BuildModel()

        # Set visualization properties
        vis_shape = chrono.ChSphereShape()
        vis_shape.SetRadius(radius)
        vis_shape.SetColor(chrono.ChColor(random.random(), random.random(), random.random()))
        body.AddVisualShape(vis_shape)

        # Add to system
        system.Add(body)
        particles.append(body)

    return particles

# =============================================================================
# Main simulation loop
# =============================================================================

# Emit initial particles
particles = emit_particles(num_particles)

# Simulation loop
time = 0
while vis.Run() and time < sim_duration:
    # Render scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance simulation
    system.DoStepDynamics(time_step)
    time += time_step

    # Print simulation time
    print(f"Simulation time: {time:.2f} s", end='\r')

# =============================================================================
# Clean up
# =============================================================================

# The Irrlicht visualization system is automatically cleaned up when the program ends
```

## Key Features of This Simulation:

1. **Initialization**:
   - Sets up a Chrono physical system with no default gravity
   - Configures Irrlicht visualization with camera and lighting

2. **Particle System**:
   - Emits particles with random:
     - Shapes (spheres with varying radii)
     - Positions (within a spherical region)
     - Velocities
     - Orientations and angular velocities
     - Colors

3. **Gravitational Forces**:
   - Custom force callback calculates gravitational attraction between all particles
   - Uses Newton's law of universal gravitation (F = G * m1 * m2 / r²)
   - Forces are applied continuously during simulation

4. **Visualization**:
   - Real-time rendering with Irrlicht
   - Each particle has a unique color
   - Camera positioned to view the entire system

5. **Simulation Control**:
   - Configurable time step and duration
   - Adjustable gravitational constant
   - Particle properties can be easily modified

To run this simulation, you'll need to have PyChrono installed with the Irrlicht visualization module. The particles will start in random positions and velocities, then gradually attract each other through gravitational forces, forming dynamic clusters.
