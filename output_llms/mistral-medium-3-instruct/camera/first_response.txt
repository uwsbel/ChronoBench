Here's a PyChrono simulation script that meets your requirements. This script loads a triangular mesh from a Wavefront .obj file, visualizes it as a fixed body, adds a camera sensor with noise filters, and simulates an orbiting camera around the mesh while printing camera buffer data.

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.sensor as sensors
import numpy as np
import os

# Set the path to the Chrono data directory
chrono.SetChronoDataPath("PATH_TO_CHRONO_DATA")  # Update this path

# Create a Chrono physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle("Orbiting Camera Simulation")
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 0, 6))
vis.AddTypicalLights()
vis.SetCameraVertical(chrono.ChVectorD(0, 0, 0))

# Load the triangular mesh from an .obj file
mesh = chrono.ChTriangleMeshConnected()
mesh.LoadWavefrontMesh("path_to_your_mesh.obj")  # Update this path

# Create a body with the mesh
body = chrono.ChBody()
body.SetBodyFixed(True)
body.GetVisualModel().AddTriangleMesh(mesh)
system.Add(body)

# Create a sensor manager
sensor_manager = sensors.ChSensorManager(system)

# Create a camera sensor
camera = sensors.ChCameraSensor(
    body,                     # parent body
    10,                       # update rate in Hz
    chrono.ChFrameD(chrono.ChVectorD(2, 0, 0), chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 1, 0))),  # initial position and orientation
    640,                      # horizontal resolution
    480,                      # vertical resolution
    1.4,                      # focal length
    1.4                       # lens aperture
)
camera.SetName("camera_sensor")
camera.SetLag(0.1)            # lag in seconds
camera.SetNoise(sensors.ChNoiseWhite(0.05))  # add white noise
sensor_manager.AddSensor(camera)

# Create a filter for the camera
filter = sensors.ChFilterRGBD()
camera.PushFilter(filter)

# Create a visualizer for the camera images
visualizer = sensors.ChVisualizerSensor("camera_visualizer", 800, 600)
visualizer.AddSensor(camera)
visualizer.SetActiveCamera(camera)
sensor_manager.AddVisualizer(visualizer)

# Simulation parameters
time_step = 0.01
simulation_time = 10.0
current_time = 0.0

# Simulation loop
while current_time < simulation_time:
    # Update the camera position in an orbit around the mesh
    angle = 2 * np.pi * current_time / simulation_time
    radius = 3.0
    camera_pos = chrono.ChVectorD(radius * np.cos(angle), 0, radius * np.sin(angle))
    camera_frame = chrono.ChFrameD(camera_pos, chrono.Q_from_AngAxis(angle, chrono.ChVectorD(0, 1, 0)))
    camera.SetOffsetPose(camera_frame)

    # Advance the simulation
    system.DoStepDynamics(time_step)
    current_time += time_step

    # Update the sensor manager
    sensor_manager.Update()

    # Print camera buffer data
    if camera.HasData():
        data = camera.GetMostRecentBuffer()
        print(f"Time: {current_time:.2f}s - Camera buffer size: {len(data)} bytes")

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Update the visualizer
    visualizer.Update()

# Clean up
vis.Run()
```

### Important Notes:
1. **Paths**: You need to update `PATH_TO_CHRONO_DATA` and the path to your `.obj` file.
2. **Dependencies**: Ensure you have PyChrono and its dependencies (like Irrlicht) properly installed.
3. **Camera Orbit**: The camera orbits the mesh in a circular path with a radius of 3 units.
4. **Sensor Data**: The script prints the camera buffer size at each step. You can modify this to print more detailed data if needed.

This script provides a complete simulation with visualization, sensor management, and dynamic camera movement. Adjust parameters like the orbit radius, simulation time, or noise levels as needed.
