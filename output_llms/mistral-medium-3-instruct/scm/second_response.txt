# New class to manage SCM terrain parameters
class SCMTerrainParameters:
    def __init__(self, preset="mid"):
        self.preset = preset
        self.parameters = {
            "soft": {
                "Bekker_Kphi": 1e6,
                "Bekker_Kc": 0,

                "Bekker_Kc": 1e4,

                "Mohr_friction": 35,
                "Janosi_shear": 0.005,
                "Elastic_stiffness": 5e8,
                "Damping": 5e4
            }
        }

    def get_parameters(self):
        return self.parameters[self.preset]

    def set_preset(self, preset):
        if preset in self.parameters:
            self.preset = preset
        else:
            print(f"Warning: Unknown preset '{preset}'. Using default 'mid'.")
            self.preset = "mid"
