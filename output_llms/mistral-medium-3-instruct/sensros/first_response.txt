# PyChrono Simulation with Multiple Sensors and ROS Integration

Below is a comprehensive PyChrono simulation script that creates a ground body with various sensors, integrates with ROS, and runs a real-time simulation loop:

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.sensor as sensors
import pychrono.ros as chronoros
import numpy as np
import time

# ==============================================================================
# Initialize Chrono system and visualization
# ==============================================================================

# Create a Chrono physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle("PyChrono Sensor Simulation")
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(5, 3, 2))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVectorD(5, 10, 5), chrono.ChVectorD(0, 0, 0), 10, 1, 10, 40, 512)

# ==============================================================================
# Create ground body and mesh for visualization
# ==============================================================================

# Create a ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
ground.SetCollide(True)
system.Add(ground)

# Create a mesh for the ground (a simple box)
ground_mesh = chrono.ChTriangleMeshConnected()
ground_mesh.LoadWavefrontMesh(chrono.GetChronoDataFile("models/box.obj"), False, True)
ground_mesh.Transform(chrono.ChVectorD(0, -0.5, 0), chrono.ChMatrix33D(1))  # Scale if needed
ground.GetVisualModel().AddTriangleMesh(ground_mesh, chrono.ChMatrix33D(1), chrono.ChVectorD(0, 0, 0))

# ==============================================================================
# Create sensor manager and configure sensors
# ==============================================================================

# Create sensor manager
sensor_manager = sensors.ChSensorManager(system)

# Update frequency for all sensors (Hz)
update_rate = 30
update_step = 1.0 / update_rate

# 1. Camera sensor
camera = sensors.ChCameraSensor(
    ground,                          # body to attach to
    update_step,                     # update rate
    chrono.ChFrameD(chrono.ChVectorD(0, 1, 0), chrono.Q_from_AngAxis(chrono.CH_C_PI/4, chrono.ChVectorD(1, 0, 0))),  # pose
    640, 480,                        # resolution
    chrono.CH_C_PI/4,                # fov
    1.0                              # max range
)
sensor_manager.AddSensor(camera)

# 2. LiDAR sensor
lidar = sensors.ChLidarSensor(
    ground,                          # body to attach to
    update_step,                     # update rate
    chrono.ChFrameD(chrono.ChVectorD(0, 1, 0), chrono.Q_from_AngAxis(chrono.CH_C_PI/4, chrono.ChVectorD(1, 0, 0))),  # pose
    30,                              # horizontal samples
    1,                               # vertical samples
    chrono.CH_C_PI/4,                # horizontal fov
    0,                               # vertical fov
    10.0,                            # max range
    0.1,                             # min range
    0.1                              # range precision
)
sensor_manager.AddSensor(lidar)

# 3. GPS sensor
gps = sensors.ChGPSSensor(
    ground,                          # body to attach to
    update_step                      # update rate
)
sensor_manager.AddSensor(gps)

# 4. IMU sensors (accelerometer, gyroscope, magnetometer)
imu = sensors.ChIMUSensor(
    ground,                          # body to attach to
    update_step                      # update rate
)
sensor_manager.AddSensor(imu)

# ==============================================================================
# Initialize ROS interface
# ==============================================================================

# Create ROS manager
ros_manager = chronoros.ChRosManager()
ros_manager.Initialize("pychrono_sensor_simulation")

# Register sensors with ROS topics
ros_manager.RegisterSensor(camera, "/camera/image_raw", "sensor_msgs/Image")
ros_manager.RegisterSensor(lidar, "/lidar/points", "sensor_msgs/PointCloud2")
ros_manager.RegisterSensor(gps, "/gps/fix", "sensor_msgs/NavSatFix")
ros_manager.RegisterSensor(imu, "/imu/data", "sensor_msgs/Imu")

# ==============================================================================
# Simulation loop
# ==============================================================================

# Simulation parameters
time_step = 0.01
max_time = 10.0
current_time = 0

# Real-time simulation parameters
real_time_factor = 1.0
last_time = time.time()

# Main simulation loop
while vis.Run() and current_time < max_time:
    # Start timing
    start_time = time.time()

    # Update sensors
    sensor_manager.Update()

    # Advance simulation
    system.DoStepDynamics(time_step)
    current_time += time_step

    # Update visualization
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Publish ROS data
    ros_manager.PublishData()

    # Calculate time for this step
    step_time = time.time() - start_time
    sleep_time = (time_step / real_time_factor) - step_time

    # Sleep to maintain real-time
    if sleep_time > 0:
        time.sleep(sleep_time)

    # Update last time for real-time calculation
    last_time = time.time()

# Clean up
ros_manager.Shutdown()
```

## Key Features of This Implementation:

1. **System Initialization**:
   - Creates a Chrono NSC system with gravity
   - Sets up Irrlicht visualization with camera and lighting

2. **Ground Body Setup**:
   - Creates a fixed ground body
   - Adds a visual mesh for the ground

3. **Sensor Configuration**:
   - Camera sensor with 640x480 resolution
   - LiDAR sensor with 30 horizontal samples
   - GPS sensor for position data
   - IMU sensor providing accelerometer, gyroscope, and magnetometer data

4. **ROS Integration**:
   - Initializes ROS manager
   - Registers each sensor with appropriate ROS topics and message types
   - Handles data publishing during simulation

5. **Real-time Simulation Loop**:
   - Updates all sensors at their specified rates
   - Advances the physics simulation
   - Maintains real-time execution with sleep timing
   - Publishes sensor data to ROS topics

## Customization Options:

1. To change sensor positions or orientations, modify the `ChFrameD` parameters in each sensor constructor.
2. Adjust the `update_rate` to change how frequently sensors publish data.
3. Modify ROS topic names or message types by changing the parameters in `RegisterSensor` calls.
4. Change the simulation parameters (time_step, max_time) to control simulation duration and accuracy.

This script provides a complete foundation for a sensor simulation in PyChrono with ROS integration, ready for further customization based on specific requirements.
