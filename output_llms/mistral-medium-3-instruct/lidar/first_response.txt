# PyChrono Simulation: Fixed Triangular Mesh with Orbiting Lidar Sensor

Below is a complete PyChrono simulation script that creates a fixed triangular mesh from an OBJ file with an orbiting lidar sensor:

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.sensor as sensors
import numpy as np
import os

# 1. Initialize PyChrono environment and core components
chrono.SetChronoOutputPath("output/")

# Create the physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))  # No gravity needed for this simulation

# 2. Add physical systems and objects
# Create a fixed body for the mesh
mesh_body = chrono.ChBody()
mesh_body.SetBodyFixed(True)  # Make it a fixed body
system.Add(mesh_body)

# Load the triangular mesh from OBJ file
mesh_path = "path/to/your/mesh.obj"  # Replace with your OBJ file path
if not os.path.exists(mesh_path):
    raise FileNotFoundError(f"Mesh file not found at {mesh_path}")

mesh = chrono.ChTriangleMeshConnected()
mesh.LoadWavefrontMesh(mesh_path, False, True)  # False=no vertex colors, True=compute normals

# Create visualization assets for the mesh
mesh_asset = chrono.ChTriangleMeshShape()
mesh_asset.SetMesh(mesh)
mesh_asset.SetName("mesh_visualization")
mesh_asset.SetStatic(True)
mesh_body.AddAsset(mesh_asset)

# Create a collision model for the mesh
mesh_body.GetCollisionModel().ClearModel()
mesh_body.GetCollisionModel().AddTriangleMesh(mesh)
mesh_body.GetCollisionModel().BuildModel()
mesh_body.SetCollide(True)

# 3. Set up the lidar sensor
# Create a sensor manager
sensor_manager = sensors.ChSensorManager(system)

# Create the lidar sensor
lidar = sensors.ChLidarSensor(
    name="orbiting_lidar",
    update_rate=100,  # Hz
    sampling_rate=1000,  # Hz
    max_distance=10.0,  # meters
    min_distance=0.1,  # meters
    horizontal_resolution=0.5,  # degrees
    vertical_resolution=0.5,  # degrees
    horizontal_angle=360,  # degrees
    vertical_angle=30,  # degrees
    point_cloud=True
)

# Add noise filters to the lidar
noise_model = sensors.ChGaussianNoiseModel()
noise_model.SetNoise(0.01)  # 1% noise
lidar.AddNoiseModel(noise_model)

# Configure visualization for the lidar
lidar.SetVisualize(True)
lidar.SetVisualizePointCloud(True)
lidar.SetVisualizeFov(True)

# Configure data saving options
lidar.SetSaveData(True)
lidar.SetSaveDirectory("output/lidar_data/")

# Attach the lidar to a moving body (we'll create one for the orbit)
lidar_body = chrono.ChBody()
system.Add(lidar_body)

# Position the lidar body initially
lidar_body.SetPos(chrono.ChVectorD(5, 0, 2))
lidar_body.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))

# Attach the lidar to this body
lidar.Attach(lidar_body)
sensor_manager.AddSensor(lidar)

# 4. Set up visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Triangular Mesh with Orbiting Lidar")
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(10, 0, 5))
vis.AddTypicalLights()
vis.AddLightWithShadow(
    chrono.ChVectorD(10, 10, 10),
    chrono.ChVectorD(0, 0, 0),
    10,
    1, 20,
    40
)

# 5. Simulation parameters
time_step = 0.01
simulation_time = 10.0  # seconds
current_time = 0.0

# Orbit parameters
orbit_radius = 5.0
orbit_height = 2.0
orbit_speed = 0.5  # radians per second

# Simulation loop
while current_time < simulation_time:
    # Update the lidar position in an orbit around the mesh
    angle = orbit_speed * current_time
    x = orbit_radius * np.cos(angle)
    z = orbit_radius * np.sin(angle)

    # Update lidar body position and orientation
    lidar_body.SetPos(chrono.ChVectorD(x, 0, z + orbit_height))
    lidar_body.SetRot(chrono.Q_from_AngAxis(-angle, chrono.VECT_Y))

    # Advance the simulation
    system.DoStepDynamics(time_step)
    current_time += time_step

    # Update the sensor manager
    sensor_manager.Update()

    # Print lidar buffer data
    if lidar.HasData():
        print(f"Time: {current_time:.2f}s")
        print(f"Lidar position: {lidar_body.GetPos()}")
        print(f"Points detected: {lidar.GetPointCloud().size()}")

        # Print first 5 points if available
        point_cloud = lidar.GetPointCloud()
        for i in range(min(5, point_cloud.size())):
            point = point_cloud[i]
            print(f"  Point {i+1}: {point.x}, {point.y}, {point.z}")

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

# Save the final state of the lidar data
sensor_manager.WriteData()
```

## Key Features of This Implementation:

1. **Fixed Triangular Mesh**:
   - Loads a mesh from an OBJ file and makes it a fixed body in the simulation
   - Includes both visualization and collision models

2. **Lidar Sensor Setup**:
   - Creates a lidar sensor with configurable parameters (range, resolution, etc.)
   - Adds Gaussian noise to simulate real-world sensor imperfections
   - Configures visualization of the sensor's field of view and point cloud

3. **Orbiting Motion**:
   - The lidar is attached to a moving body that follows a circular orbit
   - The orbit is defined by radius, height, and speed parameters

4. **Data Collection**:
   - Prints lidar buffer data at each simulation step
   - Saves all lidar data to output files
   - Includes visualization of the point cloud in real-time

5. **Visualization**:
   - Uses Irrlicht for 3D rendering
   - Includes typical scene elements (lights, camera, skybox)

To use this script, you'll need to:
1. Replace "path/to/your/mesh.obj" with the actual path to your OBJ file
2. Ensure you have PyChrono and its dependencies properly installed
3. Adjust simulation parameters (time step, duration, orbit parameters) as needed

The script will create an "output" directory where simulation results and lidar data will be saved.
