        self.mesh = mesh

        # Create chains
        for i in range(n_chains):

    def _create_chain(self, chain_idx):
        # Create a section for the cable beam
        msection_cable = fea.ChBeamSectionCable()
        msection_cable.SetDiameter(0.015)  # 15 mm diameter
        msection_cable.SetYoungModulus(0.01e9)  # 0.01 GPa
        msection_cable.SetRayleighDamping(0.0001)

        # Create a builder for the ANCF cable
        builder = fea.ChBuilderCableANCF()

        # Calculate positions for this chain (spread them out)
        x_offset = chain_idx * 0.1  # Horizontal spacing between chains
        num_elements = 10 + chain_idx * 2  # Increase elements per chain

        # Build the beam
        builder.BuildBeam(
            self.mesh,
            msection_cable,
            num_elements,
            chrono.ChVector3d(x_offset, 0, -0.1),  # Start point
            chrono.ChVector3d(x_offset + 0.5, 0, -0.1)  # End point
        )

        # Create a fixed truss body for this chain
        mtruss = chrono.ChBody()
        mtruss.SetFixed(True)
        self.system.Add(mtruss)

        # Apply boundary conditions
        # Fix the first node of the chain
        constraint_fix = fea.ChLinkNodeFrame()
        constraint_fix.Initialize(builder.GetFirstBeamNodes().front(), mtruss)
        self.system.Add(constraint_fix)

        # Apply force to the last node
        builder.GetLastBeamNodes().front().SetForce(chrono.ChVector3d(0, -0.7, 0))

        # Create a box body at the end of the chain
        end_body = chrono.ChBody()
        end_body.SetPos(builder.GetLastBeamNodes().front().GetPos())
        end_body.SetMass(0.1)
        end_body.SetInertiaXX(chrono.ChVector3d(0.001, 0.001, 0.001))
        end_body.SetCollide(True)

        # Add visualization for the box
        box_vis = chrono.ChBoxShape()
        box_vis.GetBoxGeometry().SetLengths(chrono.ChVector3d(0.05, 0.05, 0.05))
        end_body.AddVisualShape(box_vis)

        self.system.Add(end_body)
        self.end_bodies.append(end_body)

        # Connect the end of the chain to the box
        constraint_end = fea.ChLinkNodeFrame()
        constraint_end.Initialize(builder.GetLastBeamNodes().front(), end_body)
        self.system.Add(constraint_end)

        # Store chain information
        self.chains.append({
            'builder': builder,
            'truss': mtruss,
            'end_body': end_body
        })

        visualize_beam.SetSmoothFaces(True)
