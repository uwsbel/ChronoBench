# Simple Pendulum Simulation with PyChrono

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

def main():
    # Initialize the Chrono system
    print("Creating the Chrono system...")
    my_system = chrono.ChSystemNSC()
    
    # Set gravitational acceleration
    my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))
    
    # Create the ground body
    ground = chrono.ChBodyEasyBox(10, 0.2, 10, 1000, True, True)
    ground.SetPos(chrono.ChVectorD(0, -5, 0))
    ground.SetBodyFixed(True)  # Fix the ground so it doesn't move
    ground.GetVisualShape(0).SetColor(chrono.ChColor(0.5, 0.5, 0.8))
    my_system.Add(ground)
    
    # Create the pendulum body
    pendulum_radius = 0.2
    pendulum_length = 2.0
    pendulum_mass = 1.0
    
    # Create pendulum as a cylinder
    pendulum = chrono.ChBodyEasyCylinder(pendulum_radius, pendulum_length, 1000, True, True)
    pendulum.SetPos(chrono.ChVectorD(0, 0, 0))
    
    # Rotate cylinder to align with pendulum direction (horizontal)
    pendulum.SetRot(chrono.Q_from_AngZ(chrono.CH_C_PI_2))
    
    # Set mass properties
    pendulum.SetMass(pendulum_mass)
    # Set inertia based on cylinder formula
    inertia_xx = pendulum_mass * (3 * pendulum_radius**2 + pendulum_length**2) / 12
    inertia_yy = pendulum_mass * (3 * pendulum_radius**2 + pendulum_length**2) / 12
    inertia_zz = pendulum_mass * pendulum_radius**2 / 2
    pendulum.SetInertiaXX(chrono.ChVectorD(inertia_xx, inertia_yy, inertia_zz))
    
    # Set initial position (hanging vertically)
    pendulum.SetPos(chrono.ChVectorD(0, -2, 0))
    
    # Add to the system
    pendulum.GetVisualShape(0).SetColor(chrono.ChColor(0.8, 0.2, 0.2))
    my_system.Add(pendulum)
    
    # Create a revolute joint between ground and pendulum
    # The joint is at the top end of the pendulum
    joint = chrono.ChLinkLockRevolute()
    joint.Initialize(ground, pendulum, chrono.ChCoordsysD(
        chrono.ChVectorD(0, 0, 0),     # position
        chrono.Q_from_AngX(chrono.CH_C_PI_2)))  # rotation to align joint axis with Z
    my_system.Add(joint)
    
    # Create the visualization window
    vis = chronoirr.ChVisualSystemIrrlicht()
    vis.AttachSystem(my_system)
    vis.SetWindowSize(1024, 768)
    vis.SetWindowTitle("Simple Pendulum Demo")
    vis.Initialize()
    vis.AddCamera(chrono.ChVectorD(0, 0, 6))
    vis.AddTypicalLights()
    
    # Simulation settings
    step_size = 0.001
    my_system.SetSolverType(chrono.ChSolver.Type_PSOR)
    my_system.SetSolverMaxIterations(100)
    my_system.SetTimestepperType(chrono.ChTimestepper.Type_EULER_IMPLICIT_LINEARIZED)
    
    # Data logging
    log_interval = 0.05  # Log data every 0.05 seconds
    last_log_time = -log_interval  # To ensure we log at the start of the simulation
    
    # Simulation loop
    print("Starting simulation...")
    
    while vis.Run():
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        
        # Log data at specified intervals
        time = my_system.GetChTime()
        if time - last_log_time >= log_interval:
            pos = pendulum.GetPos()
            vel = pendulum.GetPos_dt()
            print(f"Time: {time:.3f} s, Position: ({pos.x:.3f}, {pos.y:.3f}, {pos.z:.3f}), "
                  f"Velocity: ({vel.x:.3f}, {vel.y:.3f}, {vel.z:.3f})")
            last_log_time = time
        
        # Step the simulation
        my_system.DoStepDynamics(step_size)
    
    print("Simulation ended.")

if __name__ == "__main__":
    main()
```

This script creates a simple pendulum simulation with PyChrono and Irrlicht visualization. It includes:

1. A fixed ground body represented as a box
2. A pendulum body represented as a cylinder with appropriate mass and inertia
3. A revolute joint connecting the ground and pendulum
4. Visualization with Irrlicht
5. Periodic logging of the pendulum's position and velocity

The pendulum starts in a vertical position and swings under the influence of gravity. The simulation runs until the visualization window is closed.
