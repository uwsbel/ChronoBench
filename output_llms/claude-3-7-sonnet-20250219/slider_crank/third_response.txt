import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import matplotlib.pyplot as plt
import numpy as np

# Set ChronoDataPath if needed
# chrono.SetChronoDataPath('path/to/data')

sys = chrono.ChSystemNSC()

# Some data shared in the following
crank_center = chrono.ChVector3d(-1, 0.5, 0)
crank_rad = 0.4
crank_thick = 0.1
rod_length = 1.5

# Create four rigid bodies: the truss, the crank, the rod, the piston.

# Create the floor truss
mfloor = chrono.ChBodyEasyBox(3, 1, 3, 1000)
mfloor.SetPos(chrono.ChVector3d(0, -0.5, 0))
mfloor.SetBodyFixed(True)  # Corrected: SetFixed() to SetBodyFixed()
sys.Add(mfloor)

# Create the flywheel crank
mcrank = chrono.ChBodyEasyCylinder(chrono.ChAxis.AXIS_Y, crank_rad, crank_thick, 1000)  # Corrected: ChAxis_Y to ChAxis.AXIS_Y
mcrank.SetPos(crank_center + chrono.ChVector3d(0, 0, -0.1))
# Since ChBodyEasyCylinder creates a vertical (y up) cylinder, here rotate it:
mcrank.SetRot(chrono.Q_from_AngY(chrono.CH_C_PI_2))  # Corrected: Q_ROTATE_Y_TO_Z to Q_from_AngY(CH_C_PI_2)
sys.Add(mcrank)

# Create a stylized rod
mrod = chrono.ChBodyEasyBox(rod_length, 0.1, 0.1, 1000)
mrod.SetPos(crank_center + chrono.ChVector3d(crank_rad + rod_length / 2, 0, 0))
sys.Add(mrod)

# Create a stylized piston
mpiston = chrono.ChBodyEasyCylinder(chrono.ChAxis.AXIS_Y, 0.2, 0.3, 1000)  # Corrected: ChAxis_Y to ChAxis.AXIS_Y
mpiston.SetPos(crank_center + chrono.ChVector3d(crank_rad + rod_length, 0, 0))
mpiston.SetRot(chrono.Q_from_AngZ(chrono.CH_C_PI_2))  # Corrected: Q_ROTATE_Y_TO_X to Q_from_AngZ(CH_C_PI_2)
sys.Add(mpiston)

# Create crank-truss joint: a motor that spins the crank flywheel
my_motor = chrono.ChLinkMotorRotationSpeed()
my_motor.Initialize(mcrank,  # the first connected body
                    mfloor,  # the second connected body
                    chrono.ChFrame(crank_center))  # Corrected: ChFramed to ChFrame
my_angularspeed = chrono.ChFunction_Const(chrono.CH_C_PI)  # Corrected: ChFunctionConst to ChFunction_Const
my_motor.SetSpeedFunction(my_angularspeed)  # Corrected: SetMotorFunction to SetSpeedFunction
sys.Add(my_motor)

# Create crank-rod joint (Changed from revolute to spherical)
mjointA = chrono.ChLinkLockSpherical()  # Changed from ChLinkLockRevolute to ChLinkLockSpherical
mjointA.Initialize(mrod,
                   mcrank,
                   chrono.ChFrame(crank_center + chrono.ChVector3d(crank_rad, 0, 0)))  # Corrected: ChFramed to ChFrame
sys.Add(mjointA)

# Create rod-piston joint (Changed from revolute to spherical)
mjointB = chrono.ChLinkLockSpherical()  # Changed from ChLinkLockRevolute to ChLinkLockSpherical
mjointB.Initialize(mpiston,
                   mrod,
                   chrono.ChFrame(crank_center + chrono.ChVector3d(crank_rad + rod_length, 0, 0)))  # Corrected: ChFramed to ChFrame
sys.Add(mjointB)

# Create piston-truss joint (Changed from prismatic to planar)
mjointC = chrono.ChLinkLockPlanePlane()  # Changed from ChLinkLockPrismatic to ChLinkLockPlanePlane
mjointC.Initialize(mpiston,
                   mfloor,
                   chrono.ChFrame(crank_center + chrono.ChVector3d(crank_rad + rod_length, 0, 0),
                                  chrono.Q_from_AngX(chrono.CH_C_PI_2))  # Corrected: ChFramed to ChFrame and rotation
                  )
sys.Add(mjointC)

vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Crank demo')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(1, 1, 3), chrono.ChVector3d(0, 1, 0))
vis.AddTypicalLights()

# Initialize these lists to store values to plot.
array_time = []
array_angle = []
array_pos = []
array_speed = []

# Run the interactive simulation loop
while vis.Run():
    # for plotting, append instantaneous values:
    array_time.append(sys.GetChTime())
    array_angle.append(my_motor.GetMotorRot())  # Corrected: GetMotorAngle to GetMotorRot
    array_pos.append(mpiston.GetPos().x)
    array_speed.append(mpiston.GetPos_dt().x)  # Corrected: GetPosDt to GetPos_dt
    # here happens the visualization and step time integration
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(1e-3)

    # stop simulation after 20 seconds
    if sys.GetChTime() > 20:
        vis.GetDevice().closeDevice()

# Use matplotlib to make two plots when simulation ended:
fig, (ax1, ax2) = plt.subplots(2, sharex=True)

ax1.plot(array_angle, array_pos)
ax1.set(ylabel='position [m]')
ax1.grid()

ax2.plot(array_angle, array_speed, 'r--')
ax2.set(ylabel='speed [m]', xlabel='angle [rad]')
ax2.grid()

# trick to plot \pi on x axis of plots instead of 1 2 3 4 etc.
plt.xticks(np.linspace(0, 2 * np.pi, 5), ['0', r'$\pi/2$', r'$\pi$', r'$3\pi/2$', r'$2\pi$'])
plt.show()  # Added to ensure plots are displayed
