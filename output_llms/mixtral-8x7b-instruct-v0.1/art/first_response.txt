```python
import chrono
import chrono.irrlicht as irrl
import chrono.collision as col
import chrono.vehicle as veh
import irrl_simple_app

# Initialize PyChrono and Irrlicht
myChrono = chrono.ChronoEngine()
myChrono.Set_G_Accel(chrono.ChVector(0, -9.81, 0))
myIrrApp = irrl_simple_app.IrrlichtApplication(myChrono.GetChSystem())

# Create a terrain
terrain_size = 50.0
terrain_mesh = myChrono.Add(col.RigidTerrain(myChrono, terrain_size, terrain_size))
terrain_texture = myIrrApp.AddTexture('path/to/your/texture.jpg')
terrain_mesh.SetTexture(terrain_texture)

# Create the vehicle
vehicle_pos = chrono.ChVectorD(0, 1, 0)
vehicle_ori = chrono.Q_from_AngY(chrono.CH_C_PI/4)
vehicle_type = veh.VehicleType_ARTCAR
contact_method = veh.VehicleContactMethod_CSM
visualization_type = veh.VisualizationType_USER_DEFINED
myVehicle = myChrono.Add(veh.ChVehicle(myChrono, vehicle_type, contact_method, visualization_type))
myVehicle.SetChassisPos(vehicle_pos)
myVehicle.SetChassisRot(vehicle_ori)

# Set up the driver system
driver = veh.ChVehicleDriver()
myVehicle.SetDriver(driver)
driver.SetSteeringCommand(0)
driver.SetThrottleCommand(0)
driver.SetBrakeCommand(0)

# Set up the simulation loop
myIrrApp.SetTimestep(1./50)
while myIrrApp.GetDevice().run():
    myIrrApp.BeginScene()
    myChrono.DoStepDynamics(myIrrApp.GetTimestep())
    myVehicle.Update()
    myIrrApp.DrawAll()
    myIrrApp.EndScene()
```

Replace `'path/to/your/texture.jpg'` with the actual path to your custom texture file. This script initializes the PyChrono environment, creates a rigid terrain with a custom texture, and sets up an ARTcar vehicle with a driver system to control the steering, throttle, and braking. The simulation loop runs at 50 frames per second, updating the vehicle dynamics and visualization in real time.
