```python
import pychrono as chrono
import pychrono.irrlicht as irrlitch
import pychrono.ros as chronoros
import rospy
from std_msgs.msg import Int32

# Initialize PyChrono environment and core components
chrono.SetChronoDataPath('path/to/chrono/data')

# Initialize the Chrono system with gravitational acceleration and physical material properties
chrono_system = chrono.ChSystem()
chrono_system.Set_G_acc(chrono.ChVector(0, -9.81, 0))

# Define custom ROS handler
class RosHandler(chronoros.ChRosHandler):
    def __init__(self):
        super().__init__()
        self.pub = rospy.Publisher('custom_topic', Int32, queue_size=10)

    def OnUpdate(self, chrono_system, time):
        msg = Int32()
        msg.data = int(time * 1000)  # Convert to milliseconds
        self.pub.publish(msg)

# Add a fixed floor to the system
floor_shape = chrono.ChBoxShape()
floor_shape.SetBox(chrono.ChVector(100, 0.1, 100))
floor_body = chrono.ChBodyEasyBox(100, 0.1, 100, 1000, chrono.ChVector(0, -0.05, 0), floor_shape)
chrono_system.Add(floor_body)

# Add a movable box to the system
box_shape = chrono.ChBoxShape()
box_shape.SetBox(chrono.ChVector(1, 1, 1))
box_body = chrono.ChBodyEasyBox(1, 1, 1, 100, chrono.ChVector(0, 5, 0), box_shape)
box_body.SetPos(chrono.ChVector(0, 1, 0))
chrono_system.Add(box_body)

# Initialize ROS manager with clock, body, and transform handlers
ros_manager = chronoros.ChRosManager()
ros_manager.Initialize(None)
ros_manager.AddHandler(chronoros.ChRosClockHandler())
ros_manager.AddHandler(chronoros.ChRosBodyHandler())
ros_manager.AddHandler(chronoros.ChRosTransformHandler())
ros_manager.AddHandler(RosHandler())

# Run simulation loop
sim_step = 0.001  # Simulation timestep
ros_step = 0.01  # ROS communication timestep
while not rospy.is_shutdown():
    if sim_step >= ros_step:
        # Update ROS communication
        ros_manager.Synchronize()
        sim_step -= ros_step

    # Advance the Chrono system
    chrono_system.DoStepDynamics(sim_step)

    # Update simulation time
    sim_time = chrono_system.GetChTime()
    ros_manager.GetRosNode().set_param('sim_time', sim_time)

    # Sleep to maintain real-time execution
    rospy.sleep(ros_step - sim_step)
```

Replace `'path/to/chrono/data'` with the actual path to the Chrono data directory. This script initializes the PyChrono environment, adds a fixed floor and a movable box to the system, and creates a custom ROS handler to publish integer messages to a specified topic. The ROS manager is configured with clock, body, and transform handlers, and the custom handler is registered. The simulation loop advances the system and updates ROS communication at each timestep while maintaining real-time execution.
