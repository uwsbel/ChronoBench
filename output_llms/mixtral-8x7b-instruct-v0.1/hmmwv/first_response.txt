```python
my_system.SetSolverType(chrono.ChSolver.Type_PSS_DANTZIG)

my_system.SetSolverRelTol(1e-6)

# Set up the driver system

irrl.ChIrrApp::AddTypicalSky()

irrl.ChIrrApp::SetTimestep(1.0 / 50)
irrl.ChIrrApp::Initialize()
irrl.ChIrrApp::SetShowInfos(True)
irrl.ChIrrApp::SetShowPerformanceCounters(True)
irrl.ChIrrApp::SetShowFPS(True)
```
