```python
import chrono
from chrono import ChVector, ChMatrix33, ChQuaternion
from chrono.irrlicht import IrrApp
from chrono.collision import ChCollisionModel
from chrono.irrlicht import ChIrrDevice, ChIrrDriver, ChIrrSceneManager, ChIrrCamera
from chrono.irrlicht.ChIrrBox import ChIrrBox
from chrono.irrlicht.ChIrrSphere import ChIrrSphere
from chrono.irrlicht.ChIrrCylinder import ChIrrCylinder
from chrono.irrlicht.ChIrrText import ChIrrText
from chrono.vehicle import ChWheel, ChTire, ChTSDAxle, ChVehicle
from chrono.chrono import ChLinkMotorRotation


def createFloor(mscene):
    # Create a truss object for the floor
    truss = mscene.addChild(ChIrrBox(100, 1, 100, 0, 0, 0, 0.5, 0.5, 0.5, True))
    truss.setName('Floor')
    truss.setBodyFixed(True)


def createCrankshaft(mscene, mbody):
    # Create a cylinder for the crankshaft
    crankshaft = mscene.addChild(ChIrrCylinder(0.1, 0.5, 0, 0, 0, 0, 0.5, 32, 32, True))
    crankshaft.setName('Crankshaft')
    crankshaft.addBox(ChVectorD(0, -0.25, 0), ChVectorD(0.1, 0.5, 0.1))
    crankshaft.setPos(ChVectorD(0, 0.25, 0))
    crankshaft.setBody(mbody)


def createConnectingRod(mscene, mbody):
    # Create a cylinder for the connecting rod
    connecting_rod = mscene.addChild(ChIrrCylinder(0.1, 1, 0, 0, 0, 0, 0.5, 32, 32, True))
    connecting_rod.setName('ConnectingRod')
    connecting_rod.addBox(ChVectorD(0, -0.05, 0), ChVectorD(0.1, 0.1, 0.1))
    connecting_rod.setPos(ChVectorD(0.5, 0.2, 0))
    connecting_rod.setBody(mbody)


def createPiston(mscene, mbody):
    # Create a sphere for the piston
    piston = mscene.addChild(ChIrrSphere(0.15, 0, 0, 0, 0.5, 32, 32, True))
    piston.setName('Piston')
    piston.addBox(ChVectorD(0, -0.1, 0), ChVectorD(0.15, 0.15, 0.15))
    piston.setPos(ChVectorD(1, 0.15, 0))
    piston.setBody(mbody)


def main():
    # Initialize PyChrono and Irrlicht
    mydata = chrono.ChData()
    mydata.SetSimType(chrono.ChSystemSMC)

    msystem = mydata.GetSystem()

    mdevice = ChIrrDevice(mydata)
    mdriver = ChIrrDriver(mdevice)
    mscene = ChIrrSceneManager(mdevice)
    mcamera = ChIrrCamera(mdevice.GetSceneManager(), chrono.ChVectorD(3, 3, 3))

    msystem.Set_G_acc(ChVectorD(0, -9.81, 0))

    # Create the floor
    createFloor(mscene)

    # Create the crankshaft, connecting rod, and piston
    mbody = msystem.AddBody(ChBodyEasyCreate(1, True, True))
    createCrankshaft(mscene, mbody)
    mbody = msystem.AddBody(ChBodyEasyCreate(1, True, True))
    createConnectingRod(mscene, mbody)
    mbody = msystem.AddBody(ChBodyEasyCreate(1, True, True))
    createPiston(mscene, mbody)

    # Add a motor to spin the crankshaft
    motor = ChLinkMotorRotation()
    motor.Initialize(mbody, mbody, chrono.ChFrameD(ChVectorD(0, 0, 0), ChQuaternionD(1, 0, 0, 0)))
    motor.SetAngSpeed(10)
    msystem.AddLink(motor)

    # Set up the Irrlicht visualization
    mdevice.SetWindowCaption(u'PyChrono Crank-Slider Example')
    mdevice.Initialize()
    mdevice.getSceneManager().addCamera(mcamera)
    mdevice.getSceneManager().setActiveCamera(mcamera)
    mdevice.getSceneManager().getActiveCamera().setPosition(chrono.ChVectorD(3, 3, 3))
    mdevice.getSceneManager().getActiveCamera().setTarget(chrono.ChVectorD(0, 0, 0))

    # Add lights and text
    light = mscene.addLightSceneNode(0, chrono.ChVectorD(3, 5, -5))
    light.setType(irr.ELightType.ELT_POINT)
    light.setColor(irr.SColor(255, 255, 255, 255))
    light.setCastShadows(True)

    text = ChIrrText(mdevice.getGUIEnvironment(), u'Crank-Slider Example', chrono.ChVector2D(10, 10), 16,
                      irr.video.SColor(255, 255, 255, 255), irr.core.rect<irr::s32>(0, 0, 100, 30))
    text.setTextAlignment(irr.video.ETEXT_ALIGN_LEFT, irr.video.ETEXT_ALIGN_TOP)
    mdevice.getGUIEnvironment().addControl(text)

    # Run the simulation
    step_size = 1e-4
    for i in range(10000):
        msystem.DoStepDynamics(step_size)
        mdevice.getSceneManager().drawAll()
        mdevice.getSceneManager().getActiveCamera().setPosition(mdevice.getSceneManager().getActiveCamera().getPosition() +
                                                                 chrono.ChVectorD(0.01, 0, 0))
        mdevice.sleep(1)

    # Clean up
    mdevice.getSceneManager().clear()
    mdevice.drop()


if __name__ == '__main__':
    main()
```
This script creates a simple crank-slider mechanism with a floor, crankshaft, connecting rod, and piston. The crankshaft is spun by a motor at a constant angular speed. The Irrlicht visualization system is used to display the simulation with customizable camera settings, lighting, and additional visual elements like textures and logos.
