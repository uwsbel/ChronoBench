```python
import time
import pychrono as pc
import pychrono.irrlicht as irr
import pychrono.vehicle as veh

# Initialize the PyChrono environment and core components
pc.set_default_collision_validation(False)
pc.set_default_contact_printing(False)

# Create a new Chrono::Engine physical system (with default collision manager)
system = pc.ChSystemSMC()

# -------------------------------------------------------------------
# Add the required physical systems and objects as specified.

# Create the HMMWV vehicle
hmmwv = veh.HMMWV()
system.add_vehicle(hmmwv)

# Set up an SCM deformable terrain with defined soil parameters
terrain = veh.SCMDeformableTerrain()
terrain.set_young_modulus(1e6)
terrain.set_poisson_ratio(0.3)
terrain.set_density(1000)
terrain.set_max_principal_strain(0.1)
terrain.set_max_shear_strain(0.05)
terrain.set_viscous_coefficient(1e4)
system.add_terrain(terrain)

# Initialize the terrain with a height map
height_map_file = "path/to/your/heightmap.png"
terrain.initialize(height_map_file)

# Initialize a driver system for vehicle control
driver = veh.ChVehicleDriver(hmmwv)
system.add_vehicle_driver(driver)

# -------------------------------------------------------------------
# Set necessary default parameters such as positions, forces, and interactions.

# Set initial vehicle pose
hmmwv.set_chassis_pose(pc.ChCoordsysD(pc.ChVectorD(0, 0, 0)))

# Set initial driver inputs
driver.set_steering(0.5)
driver.set_brake(0.2)
driver.set_throttle(0.5)

# -------------------------------------------------------------------
# Real-time visualization using Irrlicht

# Create an Irrlicht visualization system
device = irr.create_device(irr.video_mode(800, 600))
driver = irr.ChIrrlichtDriver(device)
system.set_driver(driver)

# Set up the camera
camera_pos = pc.ChVectorD(10, 10, 10)
camera_target = pc.ChVectorD(0, 0, 0)
driver.set_camera_position(camera_pos)
driver.set_camera_target(camera_target)
driver.set_camera_up_vector(pc.ChVectorD(0, 1, 0))

# Set up lighting
driver.set_ambient_light(pc.ChColor(0.2, 0.2, 0.2))
driver.add_light(pc.ChVectorD(10, 10, -10), pc.ChColor(0.8, 0.8, 0.8))

# -------------------------------------------------------------------
# Simulation loop

timestep = 1e-3  # in seconds
end_time = 10.0  # in seconds

start_time = time.time()
current_time = 0.0

while current_time < end_time:
    # Advance the simulation one timestep
    system.do_step(timestep)

    # Synchronize the Irrlicht visualization with the simulation
    driver.sleep(timestep * 1000.0)
    driver.run()

    # Update the current time
    current_time = system.get_time()

elapsed_time = time.time() - start_time
print(f"Simulation completed in {elapsed_time:.3f} seconds")

# Clean up and shut down the Irrlicht device
device.drop()
```

Replace `path/to/your/heightmap.png` with the actual path to your heightmap image file. This script creates an HMMWV vehicle on an SCM deformable terrain with a specified height map, controlled by a driver system and visualized using Irrlicht. The simulation loop runs for a specified duration, maintaining real-time execution.
