```python
import chrono
from chrono import ChVector, ChCoordsys, ChFrame, ChQuaternion, ChMatrix33
from chrono.irrlicht import *
from chrono.vehicle import *

# Initialize the PyChrono environment and core components
chrono.SetChDir('path/to/your/chrono/data/directory')
my_application = chrono.ChApplication()
my_application.SetSolverType(chrono.ChSolver.Type_PSS_DANTZIG)
my_application.Set_G_acc(ChVector(0, -9.81, 0))

# Add the required physical systems and objects as specified
my_system = chrono.ChSystemNSC()

# Ground body
ground_shape = chrono.ChBoxShape(1e7, 1e7, 1)
ground_body = chrono.ChBodyEasyCreate dynamics, ground_shape, 1000, chrono.ChVector(0, 0, 0)
my_system.AddBody(ground_body)

# Initialize the Viper rover and its driver
viper_filename = 'path/to/your/viper/model.obj'
viper_driver_filename = 'path/to/your/viper/driver.py'
viper_rover = chrono.ChVehicleNSC(viper_filename)
viper_driver = chrono.ChVehicleDriver(viper_driver_filename)
my_system.Add(viper_rover)
viper_rover.SetDriver(viper_driver)

# Create a real-time visualization using Irrlicht
my_vis = chrono.ChVisualSystemIrrlicht()
my_application.SetVisualSystem(my_vis)
my_vis.SetChaseCamera(chrono.ChVector(10, 10, 10), chrono.ChVector(0, 0, 0))
my_vis.SetBackfaceCulling(False)
my_vis.AddLogo()
my_vis.AddSkyBox()
my_vis.AddCamera(chrono.ChVector(0, 15, -20), chrono.ChVector(0, 0, 0))

# Implement a simulation loop
duration = 10  # seconds
time_step = 1e-3  # seconds
timesteps = int(duration / time_step)
steering_time = 5  # seconds
steering_angle = 0.5  # radians
steering_start_time = timesteps // 2

for i in range(timesteps):
    if i >= steering_start_time and i < steering_start_time + (timesteps // (2 * steering_time)):
        steering_angle = 0.5 * (i - steering_start_time) / (timesteps // steering_time)
    else:
        steering_angle = 0.5 - 0.5 * abs((i - steering_start_time) / (timesteps // steering_time))

    viper_rover.GetVehicle().GetChassisBody().SetWvel(chrono.ChVector(0, 0, 0))
    viper_rover.GetVehicle().GetChassisBody().SetWrot(chrono.ChQuaternion(1, 0, 0, 0))
    viper_rover.GetVehicle().SetSteer(steering_angle)

    my_system.DoStepDynamics(time_step)
    my_application.DoStepVisualization(time_step, my_vis)

# Finalize the simulation
my_vis.WaitKey(0)
del my_application
```

Replace the placeholders with the correct file paths for the Viper rover model and driver. This script initializes the PyChrono environment, sets up the required physical systems and objects, and implements a simulation loop where the rover's steering gradually changes over a specified time period. The real-time visualization is created using Irrlicht with camera settings and lighting.
