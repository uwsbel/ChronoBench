```python
import chrono
from chrono import ChSystem, ChBody, ChLinkMate, ChVisualization, ChMaterialSurface, ChColor
from chrono.vehicle import ChVehicle, ChChassis, ChTire, ChSteer, ChSuspension, ChDriveshaft, ChAxle, ChWheel
from chrono.sensor import ChVisualSensor, ChPointLight, ChCamera

# Initialize PyChrono environment and core components
my_system = ChSystem()

# Add the required physical systems and objects

# Create a rigid terrain
terrain_shape = chrono.ChBox()
terrain_shape.size = chrono.ChVectorD(100, 5, 100)
terrain = ChBody(mass=0, visualization=ChVisualization())
terrain.AddShape(terrain_shape)
terrain.SetPos(chrono.ChVectorD(0, -2.5, 0))
my_system.AddBody(terrain)

# Create a Gator vehicle
chassis_shape = chrono.ChBox()
chassis_shape.size = chrono.ChVectorD(1.5, 0.5, 3)
chassis = ChChassis(visualization=ChVisualization())
chassis.AddShape(chassis_shape)
chassis.SetPos(chrono.ChVectorD(0, 2.5, 0))
chassis.SetBodyFixed(True)

tire_shape = chrono.ChCylinder()
tire_shape.p1 = chrono.ChVectorD(0, 0.25, 0)
tire_shape.p2 = chrono.ChVectorD(0, -0.25, 0)
tire_shape.radius = 0.3
tire1 = ChTire()
tire1.AddShape(tire_shape)
tire1.SetPos(chrono.ChVectorD(-0.75, 2.5, 0))
tire1.SetBodyFixed(True)

tire2 = ChTire()
tire2.AddShape(tire_shape)
tire2.SetPos(chrono.ChVectorD(0.75, 2.5, 0))
tire2.SetBodyFixed(True)

steer = ChSteer()
steer.SetChassisBody(chassis)
steer.AddTire(tire1)
steer.AddTire(tire2)

suspension = ChSuspension()
suspension.SetChassisBody(chassis)
suspension.SetTireBody(tire1)
suspension.SetTireBody(tire2)

driveshaft = ChDriveshaft()
axle = ChAxle()
wheel = ChWheel()

vehicle = ChVehicle()
vehicle.SetChassisBody(chassis)
vehicle.AddTire(tire1)
vehicle.AddTire(tire2)
vehicle.AddSuspension(suspension)
vehicle.AddSteer(steer)
vehicle.AddDriveshaft(driveshaft)
vehicle.AddAxle(axle)
vehicle.AddWheel(wheel)

my_system.AddBody(chassis)
my_system.AddBody(tire1)
my_system.AddBody(tire2)

# Set necessary default parameters
chassis.SetVisualizationType(ChVisualization.VISUALIZATION_TYPE_MESH)
chassis.SetMesh("path/to/gator_chassis.obj")
chassis.SetColor(ChColor(0.5, 0.5, 0.5))

tire1.SetVisualizationType(ChVisualization.VISUALIZATION_TYPE_MESH)
tire1.SetMesh("path/to/gator_tire.obj")
tire1.SetColor(ChColor(0.5, 0.5, 0.5))

tire2.SetVisualizationType(ChVisualization.VISUALIZATION_TYPE_MESH)
tire2.SetMesh("path/to/gator_tire.obj")
tire2.SetColor(ChColor(0.5, 0.5, 0.5))

# Initialize sensor manager
sensor_manager = ChSensorManager()

# Add point lights
point_light1 = ChPointLight()
point_light1.SetPos(chrono.ChVectorD(-2, 5, -2))
point_light1.SetColor(ChColor(1, 1, 1))
sensor_manager.AddPointLight(point_light1)

point_light2 = ChPointLight()
point_light2.SetPos(chrono.ChVectorD(2, 5, 2))
point_light2.SetColor(ChColor(1, 1, 1))
sensor_manager.AddPointLight(point_light2)

# Add camera
camera = ChCamera()
camera.SetPos(chrono.ChVectorD(0, 5, -10))
camera.SetDirection(chrono.ChVectorD(0, -1, 1))
camera.SetLookAt(chrono.ChVectorD(0, 0, 0))
sensor_manager.AddCamera(camera)

# Run simulation loop
for i in range(1000):
    my_system.DoStepDynamics(1.0 / 60.0)

    # Update driver
    vehicle.GetDriver().Synchronize(my_system.GetChTime())

    # Update terrain
    terrain.Synchronize(my_system.GetChTime())

    # Update vehicle
    vehicle.Synchronize(my_system.GetChTime())

    # Update sensor manager
    sensor_manager.Synchronize(my_system.GetChTime())

    # Advance simulation modules
    my_system.DoStepDynamics(1.0 / 60.0)
```

Replace `"path/to/gator_chassis.obj"` and `"path/to/gator_tire.obj"` with the actual paths to the 3D models of the Gator chassis and tires. This script initializes the PyChrono environment, creates a Gator vehicle with rigid terrain and a driver system, and runs a simulation loop updating the specified components at each timestep. Additionally, it sets up a sensor manager with point lights and a camera attached to the vehicle's chassis for rendering the camera images.
