To create the requested PyChrono simulation, follow the steps below to generate a Python script.

1. Import required modules:

```python
import time
from chrono import ChronoEngine, ChronoPyRunner
from chrono.collision import ChCollisionModel, ChCollisionModelWorld
from chrono.irrlicht import ChIrrApp
from chrono.vehicle import ChVehicleModelData, ChTmeasyTire
from chrono.utils.math import ChFrame
from chrono.utils.functions import ChVector
```

2. Initialize the PyChrono environment and core components:

```python
chrono_engine = ChronoEngine()
chrono_runner = ChronoPyRunner(chrono_engine)
```

3. Add the required physical systems and objects as specified:

```python
# Create a collision model for the HMMWV
hmmwv_collision_model = ChCollisionModel()

# Create a visualization model for the HMMWV
hmmwv_visualization_model = ChCollisionModel()

# Create a HMMWV vehicle model data object
hmmwv_data = ChVehicleModelData()

# Create a tire model using the TMEASY model
hmmwv_tire_model = ChTmeasyTire()

# Create a collision model for the terrain
terrain_collision_model = ChCollisionModel()

# Create a visualization model for the terrain
terrain_visualization_model = ChCollisionModel()

# Create a collision model world
collision_model_world = ChCollisionModelWorld()
collision_model_world.AddCollisionModel(hmmwv_collision_model)
collision_model_world.AddCollisionModel(terrain_collision_model)

# Create a vehicle system
vehicle_system = chrono_engine.CreateVehicleSystem(hmmwv_data)
```

4. Set necessary default parameters such as positions, forces, and interactions:

```python
# Set HMMWV location, orientation, contact method, and TMEASY tire model
hmmwv_location = ChVector(0, 0, 0)
hmmwv_orientation = ChQuaternion(1, 0, 0, 0)
hmmwv_chassis_frame = ChFrame(hmmwv_location, hmmwv_orientation)
vehicle_system.SetChassisWorldTransform(hmmwv_chassis_frame)
vehicle_system.SetSuspensionType(ChVehicleSystem.SUSPENSION_TYPE_INDEPENDENT)
vehicle_system.SetTireType(ChVehicleSystem.TIRE_TYPE_TMEASY)

# Set terrain collision and visual meshes
terrain_collision_model.AddAssetFile("Highway_col.obj")
terrain_visualization_model.AddAssetFile("Highway_vis.obj")

# Set HMMWV collision and visual meshes
hmmwv_collision_model.AddAssetFile("path/to/hmmwv_col.obj")
hmmwv_visualization_model.AddAssetFile("path/to/hmmwv_vis.obj")

# Set HMMWV tire model parameters
hmmwv_tire_model.SetWidth(0.3)
hmmwv_tire_model.SetRadius(0.4)
hmmwv_tire_model.SetMass(40)

# Add tire models to the vehicle system
vehicle_system.AddTireModel(0, hmmwv_tire_model)
vehicle_system.AddTireModel(1, hmmwv_tire_model)
vehicle_system.AddTireModel(2, hmmwv_tire_model)
vehicle_system.AddTireModel(3, hmmwv_tire_model)

# Add the collision models to the collision model world
collision_model_world.BuildModel()
```

5. Implement a custom terrain using collision and visual meshes:

```python
# Initialize Irrlicht visualization
my_app = ChIrrApp(chrono_engine.GetSystem(), "HMMWV Simulation")

# Add the terrain visualization model to Irrlicht
my_app.AddVisualSystem(terrain_visualization_model)

# Add the HMMWV visualization model to Irrlicht
my_app.AddVisualSystem(hmmwv_visualization_model)

# Configure the driver system for steering, throttle, and braking control
driver = my_app.GetDevice().get_driver()
my_app.AddUserInput(driver, vehicle_system.GetSteering())
my_app.AddUserInput(driver, vehicle_system.GetThrottle())
my_app.AddUserInput(driver, vehicle_system.GetBrake())

# Set up the simulation loop
my_app.SetTimestep(1 / 50.0)
my_app.SetSimulationStep(chrono_engine.GetSystem().GetTimestep())

# Initialize the Irrlicht event receiver
my_app.Initialize()

# Run the simulation loop
while my_app.GetDevice().run():
    my_app.BeginScene()
    my_app.DrawAll()
    my_app.EndScene()
    my_app.DoStep()

    # Update vehicle dynamics
    vehicle_system.DoStepDynamics(chrono_engine.GetSystem().GetTimestep())

    # Wait for the next frame
    time.sleep(my_app.GetTimestep())

# Clean up
my_app.CleanUp()
```

Replace `path/to/hmmwv_col.obj` and `path/to/hmmwv_vis.obj` with the actual file paths for the HMMWV collision and visual meshes. This script initializes the PyChrono environment, creates the specified systems and objects, and runs the simulation loop in real time at 50 frames per second.
