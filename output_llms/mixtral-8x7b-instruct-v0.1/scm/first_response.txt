```python
import chrono
import irrlicht

# Initialize the PyChrono environment and core components
chrono.ChSystem.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))
system = chrono.ChSystemNSC()

# Add the required physical systems and objects as specified

# Create HMMWV
vehicle = chrono.ChBodyEasyBox(2.5, 1.5, 0.75, 5000, chrono.ChVectorD(0, 0, 0))
system.AddBody(vehicle)

# Add chassis visualization
visualization = irrlicht.ChIrrVisualSystem(irrlicht.irr::video::EDT_DOUBLE)
visualization.AddLogo(chrono.GetChronoDataPath() + "logo_white_small.png")
visualization.SetWindowSize(1280, 720)
visualization.SetWindowCaption(u"PyChrono HMMWV Simulation")
visualization.SetDefault SkyBox()

# Add wheel visualization
wheel_geometry = chrono.ChCylinderShape(0.35, 0.35, chrono.ChVectorD(0, 0, 0))
wheel_geometry.SetTexture(chrono.GetChronoDataPath() + "textures/rubber.png")
wheel_geometry.SetTextureScale(4, 4)
wheel_body = chrono.ChBodyEasyCylinder(0.35, 0.35, 1.2, 2500, chrono.ChVectorD(0, 1.2, 0))
wheel_body.AddVisualShape(wheel_geometry)
wheel_body.SetPos(chrono.ChVectorD(0.9, 0.75, 0))
vehicle.AddChild(wheel_body)

# Add more wheels by cloning and modifying the position
for x_offset in [-0.9, 0.9]:
    wheel_body_clone = wheel_body.GetChBody().GetScopedCopy(system.Get_ChSystem())
    wheel_body_clone.SetPos(chrono.ChVectorD(x_offset, 0.75, 0))
    vehicle.AddChild(wheel_body_clone)

# Add SCM terrain
terrain = chrono.SCM_Terrain(system, 50, 50, 0.1)
terrain.SetSoilType(chrono.SCM_Terrain.SOIL_TYPE_SAND)
terrain.SetSoilParams(0.5, 0.1, 0.02, 0.005, 0.001, 0.0001)
terrain.SetVisualization(True)
terrain.SetMovingPatch(vehicle)

# Configure the interactive driver system
driver = chrono.ChInteractiveDriver()
driver.AddKeys({'W': (1, 0, 0), 'S': (-1, 0, 0), 'A': (0, 0, 1), 'D': (0, 0, -1),
                'Q': (0, -1, 0), 'E': (0, 1, 0), 'Space': (0, 0, 0), 'Shift': (0, 0, -1)})
driver.SetDeltaDisplacement(0.01)
system.Add(driver)

# Run the simulation in real time
system.Set_Delta_t(0.001)
system.Set_TimestepperType(chrono.ChTimestepper.Type.HHT)

vis = chrono.ChVisualSystemIrrlicht(system, visualization)
vis.SetChronoDataPath(chrono.GetChronoDataPath())
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataPath() + "logo_white_small.png")
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 5, 5), chrono.ChVectorD(0, 1, 0))
vis.AddTypicalLights()

while vis.Run():
    system.DoStepDynamics(system.Get_Timestepper().Get_Predictor(system).GetStepSize())
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
```
