```python
import chrono
import irrlicht

# Initialize the PyChrono environment
chrono.ChronoPy_SetLog(chrono.ChronoLog_NONE)
my_chrono = chrono.ChronoEngine()

# Initialize Irrlicht visualization
my_irrlicht = irrlicht.ChIrrApp(my_chrono.GetChSystem(), "HMMWV Simulation", chrono.ChIrrApp::window_size, false)
my_irrlicht.SetTimestep(1.0 / 50.0)
my_irrlicht.AddTypicalSky()
my_irrlicht.AddTypicalLogo()
my_irrlicht.AddTypicalCamera(chrono.ChVector(0, 5, -10))

# 1. Initialize HMMWV
vehicle = create_hmmwv(my_chrono, my_irrlicht)

# Set vehicle location and orientation
vehicle.SetPos(chrono.ChVector(0, 0, 0))
vehicle.SetRot(chrono.ChQuaternion(1, 0, 0, 0))

# 2. Initialize SCM terrain
terrain = create_scm_terrain(my_chrono, my_irrlicht, vehicle)

# Set custom soil parameters
terrain.SetSoilParameters(1.5, 0.5, 0.2, 0.05, 0.1)

# Enable moving patch feature
terrain.SetMovingPatch(vehicle.GetChassisBody())

# Enable sinkage visualization
terrain.EnableSinkageVisualization()

# 3. Initialize interactive driver system
driver = create_interactive_driver(my_chrono, vehicle)

# Run the simulation in real time
while my_irrlicht.GetDevice().run():
    my_chrono.DoStepDynamics(my_irrlicht.GetTimestep())
    my_irrlicht.BeginScene(true, true)
    my_irrlicht.DrawAll()
    my_irrlicht.EndScene()

# Destroy the PyChrono system
my_chrono.Destroy()
```

This script provides the required structure for the simulation. Note that the functions `create_hmmwv`, `create_scm_terrain`, and `create_interactive_driver` are not defined in this example. These functions should be implemented according to the specific HMMWV model, SCM terrain, and interactive driver requirements.
