# Initialize the PyChrono environment and core components

beam_radius = 0.01  # radius

beam.SetFixed(True)

# Add the flywheel at the center of the beam
flywheel_radius = 0.05
flywheel_mass = 1.0
flywheel_inertia = flywheel_mass * (flywheel_radius ** 2) / 2
flywheel_body = pc.ChBodyEasyBox(flywheel_mass, flywheel_radius, flywheel_radius, flywheel_radius, True, True)
flywheel_body.SetPos(beam.GetBody().GetPos())
flywheel_body.SetPos(pc.Vector3d(0, 0, 0))
flywheel_body.SetInertiaXX(pc.VectorD(flywheel_inertia, flywheel_inertia, flywheel_inertia))
beam.GetBody().GetChLinkParent()->AddVisualShape(flywheel_body, fea.ChColor(1, 0, 0))

# Create a revolute joint at the end of the beam
