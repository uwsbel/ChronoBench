```python
import chrono
from chrono import chrono_irrlicht as irrlib
from chrono import chrono_postprocess as postprocess
from chrono.collision import collision as chrono_collision
from chrono.vehicle import vehicle as chrono_vehicle

# Initialize the PyChrono environment and core components
chrono.Chrono::SetDefaultOptions(False, True, True, True, True)
my_ Chrono = chrono.Chrono()

# Add gravity
my_ Chrono.GetSystem().Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Add ground body
ground = my_ Chrono.AddBody(chrono.ChBodyEasyBox(1e6, 1e6, 1e-3, 1000, True, True))
ground.SetPos(chrono.ChVectorD(0, 0, 0))
ground.SetBodyFixed(True)

# Initialize the Turtlebot with a specific position and orientation
turtlebot_file = 'path/to/turtlebot.urdf'
turtlebot_options = chrono_vehicle.ChVehicleOptions()
turtlebot_options.SetCollide(True)
turtlebot_options.SetChassisVisualizationType(chrono_vehicle.ChVehicle.VISUALIZATION_TYPE_BODIES)
turtlebot = chrono_vehicle.ChVehicleNSKin(my_ Chrono, turtlebot_file, turtlebot_options)
turtlebot.SetPos(chrono.ChVectorD(0, 0, 0.1))
turtlebot.SetChassisRot(chrono.ChQuaternionD(1, 0, 0, 0))

# Create Irrlicht real-time visualization
device = irrlib.createDeviceIrrlicht(irrlib.CH_IrrlichtDevice_window_size_800_600, my_ Chrono.GetSystem(), 32)
driver = device.getVideoDriver()
scene = device.getSceneManager()
smgr = scene.getSceneManager()

# Set up the camera
camera_focus = turtlebot.GetChassisBody().GetPos()
camera_focus.z += 5
device.getSceneManager().addCameraSceneNode(irrlib.createCamera(device))
camera = device.getSceneManager().getActiveCamera()
camera.setPosition(chrono.ChVectorD(10, 10, 10))
camera.setTarget(camera_focus)
camera.setNearValue(0.1)
camera.setFarValue(1000.0)

# Add lighting
light_ambient = irrlib.CLightAmbient(scene)
light_ambient.setAmbientColor(chrono.ChColor(0.2f, 0.2f, 0.2f))
scene.addLightNode(light_ambient)

light_directional = irrlib.CLightDirectional(scene)
light_directional.setDirection(chrono.ChVectorD(0, -1, 0))
light_directional.setDiffuseColor(chrono.ChColor(0.6f, 0.6f, 0.6f))
light_directional.setSpecularColor(chrono.ChColor(0.2f, 0.2f, 0.2f))
scene.addLightNode(light_directional)

# Set up the simulation loop
timestep = 1.0 / 60.0
run_time = 10.0
turtlebot.SetMotorSpeeds(0, 0)

for t in my_ Chrono.GetSystem().GetChTime() :
    if t > 2.0 and t <= 4.0:
        turtlebot.SetMotorSpeeds(-1, 1)
    elif t > 6.0 and t <= 8.0:
        turtlebot.SetMotorSpeeds(1, -1)
    else:
        turtlebot.SetMotorSpeeds(0, 0)

    my_ Chrono.DoStepDynamics(timestep)

    # Update Irrlicht scene
    device.getSceneManager().clear()
    turtlebot.UpdateMesh()
    turtlebot.Draw(scene, driver)
    device.getVideoDriver().endScene()

# Clean up
device.drop()
```

This script creates a PyChrono simulation with a Turtlebot robot on a rigid terrain. It initializes the Chrono system with gravity and a ground body, then initializes the Turtlebot with a specific position and orientation. A real-time visualization is created using Irrlicht, with camera settings and lighting. The simulation loop updates and renders the robot at each timestep, controlling left and right turns at specified times.
