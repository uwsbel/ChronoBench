To create the requested simulation script, first, install PyChrono and Irrlicht. You can find the installation instructions in the official PyChrono documentation (https://projectchrono.org/documentation.html). After installing both libraries, you can start writing your script.

Here's an example script that follows the provided instructions:

```python
import time
import chrono
from chrono import ChSystemNSC, ChBody, ChLinkMate, ChLinkLockRevolute, ChShaft, ChVisualization, ChMaterialSurfaceNSC, ChLoadBody
from chrono.irrlicht import ChIrrApp
from chrono.irrlicht.utils import ChIrrRigidShapeDisplay, ChIrrBodyVisualization, ChIrrMarker
from chrono.vehicle import ChVehicleModelData, ChTireData, ChTMeasyTire
from chrono.collision import ChCollisionModel
from chrono.fea import ChElementTetra

# Initialize the PyChrono environment and core components
my_system = ChSystemNSC()

# Add the required physical systems and objects as specified
# CityBus vehicle
citybus_data = ChVehicleModelData()
citybus_data.SetChassisData(filename="citybus_chassis.obj", mass=12000.0, inertia=ChVectorD(9000, 12000, 15000))
citybus_data.AddTireData(filename="tire_model.obj", radius=0.5, width=0.2, mass=50.0, suspension_stiffness=1e6,
                         suspension_damping=1e4, max_suspension_travel=0.15, tire_type=ChTMeasyTire.TIRE_TYPE_CAR)
vehicle = citybus_data.CreateVehicle(my_system)
vehicle.SetPos(ChVectorD(0, 0, 0))
vehicle.SetChassisWorldOrientation(ChQuaternionD(0.7071, 0, 0, 0.7071))

# RigidTerrain
terrain_mesh = ChIrrMesh("terrain_mesh.obj")
terrain = ChBody(my_system, 0)
terrain.SetCollisionMesh(terrain_mesh)
terrain.SetPos(ChVectorD(0, -1, 0))
terrain.SetBodyFixed(True)
my_system.AddBody(terrain)

# Set necessary default parameters such as positions, forces, and interactions
terrain_shape = ChIrrRigidShapeDisplay.CreateBox(terrain, 50, 1, 50, 50, ChColor(0.5, 0.5, 0.5))
terrain_shape.SetTexture("terrain_texture.jpg")

# Camera
my_app = ChIrrApp(my_system, "CityBus Simulation", 800, 600, False, False, 50)
my_app.AddTypicalLogo()
my_app.AddCamera(ChVectorD(-10, 15, 5), ChVectorD(0, 0, 0))
my_app.AddCamera(ChVectorD(10, 15, 5), ChVectorD(0, 0, 0))
my_app.AddTypicalSky()
my_app.AddTypicalLights()

# Interactive driver system
def my_custom_update(vehicle, system):
    user_input = my_app.GetDevice().getEventReceiver().getInputEvent()
    if user_input.type == irr.EET_KEY_INPUT_EVENT:
        if user_input.Key == irr.KEY_KEY_UP:
            vehicle.GetChassis().AddForce(ChVectorD(0, 0, 100))
        elif user_input.Key == irr.KEY_KEY_DOWN:
            vehicle.GetChassis().AddForce(ChVectorD(0, 0, -100))
        elif user_input.Key == irr.KEY_KEY_LEFT:
            vehicle.GetChassis().AddTorque(ChVectorD(0, 100, 0))
        elif user_input.Key == irr.KEY_KEY_RIGHT:
            vehicle.GetChassis().AddTorque(ChVectorD(0, -100, 0))

# Implement the simulation loop
while my_app.GetDevice().run():
    my_app.BeginScene()
    my_app.DrawAll()
    my_app.DoStep()

    my_custom_update(vehicle, my_system)

    my_app.EndScene()

# Save the results
my_system.SaveCSV("citybus_simulation.csv", my_system.GetStep(), True)
```

Replace the object files (citybus_chassis.obj, tire_model.obj, terrain_mesh.obj) and texture file (terrain_texture.jpg) with your custom meshes and texture.

This script demonstrates a CityBus vehicle on a rigid terrain using PyChrono and Irrlicht for visualization. The vehicle is initialized with specified parameters, and the camera is set to follow the vehicle from a specified position. The interactive driver system allows controlling the vehicle's steering, throttle, and braking. The simulation loop runs at 50 frames per second, updating the vehicle dynamics and rendering the scene in real time.
