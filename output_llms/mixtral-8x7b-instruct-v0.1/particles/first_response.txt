To create the requested PyChrono simulation, follow the instructions below. Make sure to install and import the necessary packages before proceeding.

```python
# Import required modules
import chrono
import chrono.irrlicht as irrlicht
import chrono.collision as collision
import chrono.utils as utils
import math
import random

# Initialize the PyChrono environment and core components
myChrono = chrono.ChronoEngine()
myData = myChrono.GetDataFileManager()
myCollision = myChrono.GetCollisionSystem()
mySystem = myChrono.GetSystem()

# Set the Irrlicht visualization
myIrrApp = irrlicht.IrrlichtChrono(myChrono, 'PyChrono Particle Attraction Simulation')
myIrrApp.SetTimestep(1./240.)
myIrrApp.AddLogo(myData.GetUserDataPath() + 'logo_irrlicht_white.png')
myIrrApp.AddLogo(myData.GetUserDataPath() + 'logo_chrono_engine_black.png')
myIrrApp.SetShowInfos(True)
myIrrApp.SetShowLogo(True)
myIrrApp.SetShowAxes(True)
myIrrApp.SetShowBoundingBoxes(False)
myIrrApp.SetShowContacts(True)
myIrrApp.SetShowPairs(False)
myIrrApp.SetShowCollisionMargin(False)
myIrrApp.SetShowMasses(True)
myIrrApp.SetShowMassCenters(True)
myIrrApp.SetShowJointAxes(True)
myIrrApp.SetShowJointLimits(True)

# Create a particle emitter
num_particles = 100
particles = []
for i in range(num_particles):
    particle = myChrono.Add(chrono.ChParticles(myCollision))
    particle.SetPos(chrono.ChVectorD(random.uniform(-10, 10), random.uniform(-10, 10), 0))
    particle.SetMass(random.uniform(0.1, 1))
    particle.SetVel(chrono.ChVectorD(random.uniform(-1, 1), random.uniform(-1, 1), 0))
    particle.SetWvel(chrono.ChVectorD(random.uniform(-1, 1), random.uniform(-1, 1), 0))
    particle.SetChassisWorldTransform(chrono.ChFrameD(particle.GetPos(), chrono.ChQuaternionD(random.uniform(0, math.pi), random.uniform(0, math.pi), random.uniform(0, math.pi))))
    particles.append(particle)

# Add the required physical systems and objects
mySystem.Add(particles)
myCollision.AddPhysicsItem(particles)

# Set necessary default parameters such as positions, forces, and interactions
for i in range(num_particles):
    for j in range(i+1, num_particles):
        dist = (particles[i].GetPos() - particles[j].GetPos()).Length()
        myCollision.AddPair(particles[i], particles[j], chrono.ChContactMethod.SMC, 0.01, 0.1, 0.1)
        mySystem.AddForce(chrono.ChForce().SetGravity(chrono.ChVectorD(0, 0, -9.81)))

# Set up the Irrlicht visualization
myIrrApp.AssetScene()
myIrrApp.Add(particles)
myIrrApp.Run()

# Clean up the PyChrono environment and core components
myChrono.Clear()
```

This script initializes the PyChrono environment, creates a particle emitter, and adds the necessary physical systems and objects. It also sets up the Irrlicht visualization to display the simulation with custom forces for particle attraction.
