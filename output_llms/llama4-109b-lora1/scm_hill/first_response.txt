# Initialize PyChrono environment and core components
chrono.SetChairsThreading(True)

vehicle.SetTireStepSize(0.001)

vehicle.Initialize()

# Add vehicle to the simulation
vehicle.GetSystem().SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))
