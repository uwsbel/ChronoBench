```python
# =============================================================================
# PyChrono Simulation: Triangular Mesh with Orbiting Camera Sensor
# =============================================================================

import pychrono as chrono
import pychrono.sensor as sens
import numpy as np
import math
import os

# =============================================================================
# Simulation Parameters
# =============================================================================
SIMULATION_STEP_SIZE = 1e-3        # Simulation time step (seconds)
SIMULATION_END_TIME  = 20.0        # Total simulation time (seconds)

# Camera parameters
CAMERA_WIDTH        = 1280
CAMERA_HEIGHT       = 720
CAMERA_FOV          = math.pi / 3  # 60 degrees field of view
CAMERA_UPDATE_RATE  = 30           # Hz
CAMERA_ORBIT_RADIUS = 5.0          # Metres – orbit radius around the mesh
CAMERA_ORBIT_SPEED  = 0.5          # rad / s

# Mesh file – replace with your own .obj if available
MESH_OBJ_FILE = "shape.obj"        # Expected relative to the working directory

# =============================================================================
# Helper: create a minimal fallback .obj if none exists
# =============================================================================
def ensure_mesh_file(path: str) -> None:
    """Write a simple tetrahedron .obj so the demo always runs."""
    if os.path.isfile(path):
        return
    print(f"[INFO] '{path}' not found – generating a placeholder tetrahedron.")
    content = """# Simple tetrahedron
v  0.0  1.0  0.0
v -1.0 -1.0  1.0
v  1.0 -1.0  1.0
v  0.0 -1.0 -1.0
f 1 2 3
f 1 3 4
f 1 4 2
f 2 4 3
"""
    with open(path, "w") as f:
        f.write(content)

# =============================================================================
# 1. Initialise the Chrono system
# =============================================================================
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

print("[INFO] Chrono system initialised.")

# =============================================================================
# 2. Load the triangular mesh and create a fixed body
# =============================================================================
ensure_mesh_file(MESH_OBJ_FILE)

# -- Load mesh geometry -------------------------------------------------------
trimesh = chrono.ChTriangleMeshConnected()
trimesh.LoadWavefrontMesh(MESH_OBJ_FILE, True, True)
trimesh.Transform(
    chrono.ChVectorD(0, 0, 0),
    chrono.ChMatrix33D(chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 1, 0)))
)

# -- Create the body ----------------------------------------------------------
mesh_body = chrono.ChBody()
mesh_body.SetBodyFixed(True)
mesh_body.SetPos(chrono.ChVectorD(0, 0, 0))
mesh_body.SetName("MeshBody")

# -- Visual shape -------------------------------------------------------------
vis_mesh = chrono.ChTriangleMeshShape()
vis_mesh.SetMesh(trimesh)
vis_mesh.SetName("MeshVisual")
vis_mesh.SetMutable(False)
mesh_body.AddVisualShape(vis_mesh)

# -- Collision shape (optional, mesh is fixed) --------------------------------
mesh_material = chrono.ChMaterialSurfaceNSC()
collision_mesh = chrono.ChTriangleMeshShape()
collision_mesh.SetMesh(trimesh)

system.Add(mesh_body)
print(f"[INFO] Mesh body '{mesh_body.GetName()}' added to system (fixed={mesh_body.GetBodyFixed()}).")

# =============================================================================
# 3. Create the Sensor Manager
# =============================================================================
manager = sens.ChSensorManager(system)

# Set the scene lighting so the camera can see something
manager.scene.AddPointLight(
    chrono.ChVectorF(3, 3, 3),   # position
    chrono.ChColor(1, 1, 1),     # white
    50.0                          # intensity
)
manager.scene.AddPointLight(
    chrono.ChVectorF(-3, 3, -3),
    chrono.ChColor(0.8, 0.8, 1.0),
    30.0
)

print("[INFO] Sensor manager created.")

# =============================================================================
# 4. Define the initial camera pose (offset = orbit start)
# =============================================================================
# Initial position: on the +X side of the mesh at orbit radius
initial_cam_pos = chrono.ChVectorD(CAMERA_ORBIT_RADIUS, 1.0, 0.0)
look_at         = chrono.ChVectorD(0, 0, 0)
up              = chrono.ChVectorD(0, 1, 0)

# Build rotation so the camera looks at the origin
cam_z = (look_at - initial_cam_pos).GetNormalized()       # forward
cam_x = up.Cross(cam_z).GetNormalized()                   # right
cam_y = cam_z.Cross(cam_x).GetNormalized()                # up (recomputed)

rot_matrix = chrono.ChMatrix33D()
rot_matrix.Set_A_axis(cam_x, cam_y, cam_z)
cam_rotation = chrono.ChQuaternionD()
cam_rotation.Set_A_Rmatrix(rot_matrix)

camera_offset_pose = chrono.ChFrameD(initial_cam_pos, cam_rotation)

# =============================================================================
# 5. Create the Camera sensor
# =============================================================================
camera = sens.ChCameraSensor(
    mesh_body,            # body the sensor is attached to
    CAMERA_UPDATE_RATE,   # update rate (Hz)
    camera_offset_pose,   # offset pose relative to body
    CAMERA_WIDTH,
    CAMERA_HEIGHT,
    CAMERA_FOV
)
camera.SetName("OrbitCamera")
camera.SetLag(0.0)           # no lag
camera.SetCollectionWindow(1.0 / CAMERA_UPDATE_RATE)

# ---- Filters -----------------------------------------------------------------

# 1. Noise filter: add Gaussian noise to RGBA8 data
noise_model = sens.ChNoiseNormal(
    chrono.ChVectorF(0, 0, 0),          # mean  (unused for image noise)
    chrono.ChVectorF(0.01, 0.01, 0.01)  # std   (per channel, 0-1 range)
)
camera.PushFilter(sens.ChFilterCameraNoiseConstNormal(0.0, 0.02))

# 2. Visualise the raw RGBA image in an interactive window
camera.PushFilter(sens.ChFilterVisualize(CAMERA_WIDTH, CAMERA_HEIGHT, "Raw Camera Feed"))

# 3. Access the RGBA8 buffer from the host
camera.PushFilter(sens.ChFilterRGBA8Access())

# 4. Convert to greyscale
camera.PushFilter(sens.ChFilterGrayscale())

# 5. Visualise the greyscale image
camera.PushFilter(sens.ChFilterVisualize(CAMERA_WIDTH, CAMERA_HEIGHT, "Greyscale Feed"))

# 6. Access the greyscale (R8) buffer from the host
camera.PushFilter(sens.ChFilterR8Access())

# Add the sensor to the manager
manager.AddSensor(camera)

print("[INFO] Camera sensor added to manager with noise + visualisation filters.")

# =============================================================================
# 6. Simulation loop
# =============================================================================
sim_time   = 0.0
step_count = 0
orbit_angle = 0.0

print("[INFO] Starting simulation loop …")
print(f"       step size = {SIMULATION_STEP_SIZE} s  |  end time = {SIMULATION_END_TIME} s")
print("-" * 60)

while sim_time < SIMULATION_END_TIME:

    # ------------------------------------------------------------------
    # 6a. Dynamically update the camera offset (orbit around Y-axis)
    # ------------------------------------------------------------------
    orbit_angle = CAMERA_ORBIT_SPEED * sim_time   # radians

    cam_x_pos = CAMERA_ORBIT_RADIUS * math.cos(orbit_angle)
    cam_z_pos = CAMERA_ORBIT_RADIUS * math.sin(orbit_angle)
    cam_y_pos = 1.5 * math.sin(orbit_angle * 0.3)  # gentle vertical oscillation

    new_cam_pos = chrono.ChVectorD(cam_x_pos, cam_y_pos, cam_z_pos)
    look_at     = chrono.ChVectorD(0, 0, 0)

    # Recompute camera orientation so it keeps looking at the origin
    forward = (look_at - new_cam_pos).GetNormalized()
    world_up = chrono.ChVectorD(0, 1, 0)

    right   = world_up.Cross(forward)
    if right.Length() < 1e-6:          # degenerate case (looking straight up/down)
        world_up = chrono.ChVectorD(0, 0, 1)
        right    = world_up.Cross(forward)
    right = right.GetNormalized()
    cam_up  = forward.Cross(right).GetNormalized()

    new_rot_mat = chrono.ChMatrix33D()
    new_rot_mat.Set_A_axis(right, cam_up, forward)
    new_quat = chrono.ChQuaternionD()
    new_quat.Set_A_Rmatrix(new_rot_mat)

    new_offset = chrono.ChFrameD(new_cam_pos, new_quat)
    camera.SetOffsetPose(new_offset)

    # ------------------------------------------------------------------
    # 6b. Update the sensor manager (renders + applies filters)
    # ------------------------------------------------------------------
    manager.Update()

    # ------------------------------------------------------------------
    # 6c. Step the physics
    # ------------------------------------------------------------------
    system.DoStepDynamics(SIMULATION_STEP_SIZE)

    # ------------------------------------------------------------------
    # 6d. Retrieve and print camera buffer data
    # ------------------------------------------------------------------
    # Print at most every ~1 s of simulated time to keep output readable
    if step_count % int(1.0 / SIMULATION_STEP_SIZE) == 0:

        rgba_buffer = camera.GetMostRecentRGBA8Buffer()
        grey_buffer = camera.GetMostRecentR8Buffer()

        rgba_info = "unavailable"
        grey_info = "unavailable"

        if rgba_buffer and rgba_buffer.HasData():
            rgba_data  = rgba_buffer.GetRGBA8Data()          # numpy array (H, W, 4)
            rgba_info  = (f"shape={rgba_data.shape}  "
                          f"dtype={rgba_data.dtype}  "
                          f"mean(R,G,B,A)=("
                          f"{rgba_data[:,:,0].mean():.3f}, "
                          f"{rgba_data[:,:,1].mean():.3f}, "
                          f"{rgba_data[:,:,2].mean():.3f}, "
                          f"{rgba_data[:,:,3].mean():.3f})")

        if grey_buffer and grey_buffer.HasData():
            grey_data = grey_buffer.GetChar8Data()            # numpy array (H, W, 1)
            grey_info = (f"shape={grey_data.shape}  "
                         f"dtype={grey_data.dtype}  "
                         f"mean={grey_data.mean():.3f}  "
                         f"max={grey_data.max()}  "
                         f"min={grey_data.min()}")

        print(f"[t={sim_time:6.3f}s | step={step_count:6d}]")
        print(f"  Camera pos : ({cam_x_pos:.3f}, {cam_y_pos:.3f}, {cam_z_pos:.3f})  "
              f"orbit angle = {math.degrees(orbit_angle):.1f}°")
        print(f"  RGBA buffer: {rgba_info}")
        print(f"  Grey buffer: {grey_info}")
        print()

    sim_time   += SIMULATION_STEP_SIZE
    step_count += 1

# =============================================================================
# 7. Cleanup / summary
# =============================================================================
print("-" * 60)
print(f"[INFO] Simulation finished.  Total steps: {step_count}")
print(f"[INFO] Simulated time: {sim_time:.4f} s")
```

### What the script does

| Section | Description |
|---|---|
| **Mesh loading** | Loads a Wavefront `.obj` file via `ChTriangleMeshConnected`; if the file does not exist a minimal tetrahedron is auto-generated. |
| **Fixed body** | The mesh is attached to a `ChBody` that is pinned in space (`SetBodyFixed(True)`). |
| **Sensor manager** | A `ChSensorManager` controls rendering and houses two point lights for illumination. |
| **Camera sensor** | A `ChCameraSensor` (1280 × 720 @ 30 Hz, 60° FOV) is attached to the body. |
| **Filter chain** | ① Gaussian noise → ② RGBA visualisation window → ③ RGBA8 host access → ④ greyscale conversion → ⑤ greyscale visualisation window → ⑥ R8 host access. |
| **Orbit update** | Every simulation step, `SetOffsetPose` repositions the camera along a circular (+ gentle vertical sine) orbit so it always looks at the origin. |
| **Buffer printing** | Every simulated second, RGBA and greyscale buffer shapes, dtypes, and pixel statistics are printed to the console. |
