```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr
import math

# =============================================================================
# Simulation parameters
# =============================================================================

# Simulation step sizes
step_size = 1e-3
tire_step_size = step_size

# Time interval between two render frames
render_step_size = 1.0 / 50  # FPS = 50

# Output directories
out_dir = chrono.GetChronoOutputPath() + "MAN_10t"

# Initial vehicle location and orientation
initLoc = chrono.ChVector3d(0, 0, 0.5)
initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# Visualization type for vehicle parts (PRIMITIVES, MESH, NONE)
chassis_vis_type = veh.VisualizationType_MESH
suspension_vis_type = veh.VisualizationType_PRIMITIVES
steering_vis_type = veh.VisualizationType_PRIMITIVES
wheel_vis_type = veh.VisualizationType_MESH
tire_vis_type = veh.VisualizationType_MESH

# Collision type for chassis (NONE, PRIMITIVES, HULLS, MESH)
chassis_collision_type = veh.CollisionType_NONE

# Type of tire model (RIGID, TMEASY)
tire_model = veh.TireModelType_TMEASY

# Rigid terrain parameters
terrainHeight = 0
terrainLength = 200.0
terrainWidth = 100.0

# Point on chassis tracked by the chase camera
trackPoint = chrono.ChVector3d(0.0, 0.0, 1.75)

# Contact method
contact_method = chrono.ChContactMethod_NSC
contact_vis = False

# =============================================================================
# Create the MAN 10t vehicle
# =============================================================================

print("Creating MAN 10t truck simulation...")

# Create vehicle system
vehicle = veh.MAN_10t()
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
vehicle.SetTireType(tire_model)
vehicle.SetTireStepSize(tire_step_size)

# Initialize the vehicle
vehicle.Initialize()

# Set visualization types for vehicle components
vehicle.SetChassisVisualizationType(chassis_vis_type)
vehicle.SetSuspensionVisualizationType(suspension_vis_type)
vehicle.SetSteeringVisualizationType(steering_vis_type)
vehicle.SetWheelVisualizationType(wheel_vis_type)
vehicle.SetTireVisualizationType(tire_vis_type)

# Get the vehicle system
vehicle_system = vehicle.GetSystem()

# =============================================================================
# Create the terrain
# =============================================================================

terrain = veh.RigidTerrain(vehicle_system)

# Create a patch with contact material
patch_mat = chrono.ChContactMaterialNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)

patch = terrain.AddPatch(
    patch_mat,
    chrono.ChCoordsysd(chrono.ChVector3d(0, 0, 0), chrono.QUNIT),
    terrainLength,
    terrainWidth
)

# Set terrain visualization
patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)
patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))

terrain.Initialize()

# =============================================================================
# Create the Irrlicht visualization system
# =============================================================================

vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle("MAN 10t Truck Demo")
vis.SetWindowSize(1280, 720)
vis.SetChaseCamera(trackPoint, 10.0, 0.5)
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetVehicle())

# =============================================================================
# Create the interactive driver system
# =============================================================================

driver = veh.ChInteractiveDriverIRR(vis)

# Set the time response for steering and throttle keyboard inputs
steering_time = 1.0   # time to go from 0 to +1 (or from 0 to -1)
throttle_time = 1.0   # time to go from 0 to +1
braking_time = 0.3    # time to go from 0 to +1

driver.SetSteeringDelta(render_step_size / steering_time)
driver.SetThrottleDelta(render_step_size / throttle_time)
driver.SetBrakingDelta(render_step_size / braking_time)

driver.Initialize()

# =============================================================================
# Output vehicle mass
# =============================================================================

print(f"Vehicle mass: {vehicle.GetVehicle().GetMass():.0f} kg")
print(f"Terrain: {terrainLength}m x {terrainWidth}m rigid terrain")
print(f"Tire model: TMEASY")
print("Controls: Arrow keys for steering/throttle, Space for braking")

# =============================================================================
# Simulation loop
# =============================================================================

# Number of simulation steps between miscellaneous events
render_steps = math.ceil(render_step_size / step_size)

# Initialize simulation frame counter and simulation time
step_number = 0
render_frame = 0
simulation_time = 0

# Real-time timer
realtime_timer = chrono.ChRealtimeStepTimer()

print("\nStarting simulation...")
print("Use arrow keys to drive the truck. Press ESC to exit.\n")

while vis.Run():
    simulation_time = vehicle.GetSystem().GetChTime()

    # Render scene and output data
    if step_number % render_steps == 0:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        render_frame += 1

    # Get driver inputs
    driver_inputs = driver.GetInputs()

    # Update modules (process inputs from other modules)
    driver.Synchronize(simulation_time)
    terrain.Synchronize(simulation_time)
    vehicle.Synchronize(simulation_time, driver_inputs, terrain)
    vis.Synchronize(simulation_time, driver_inputs)

    # Advance simulation for one timestep for all modules
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    # Increment frame number
    step_number += 1

    # Spin in place for real time to catch up
    realtime_timer.Spin(step_size)

print("Simulation complete!")
```

### Key Features of This Simulation:

1. **MAN 10t Truck**: Full vehicle model with configurable visualization types (mesh, primitives, or none) for each component (chassis, suspension, steering, wheels, tires).

2. **TMEASY Tire Model**: Physically accurate tire model that handles lateral and longitudinal slip realistically.

3. **Rigid Terrain**: A flat terrain with:
   - Configurable friction (0.9) and restitution (0.01)
   - Texture mapping applied to the surface
   - Configurable dimensions (200m × 100m)

4. **Irrlicht Visualization**:
   - Chase camera that follows the vehicle
   - Directional lighting for realistic shadows
   - Skybox for environmental context
   - Logo display
   - 1280×720 window resolution

5. **Interactive Driver Controls**:
   - Arrow keys for steering and throttle
   - Spacebar for braking
   - Configurable response times for smooth control

6. **Real-time Simulation**: Uses `ChRealtimeStepTimer` to synchronize simulation with wall-clock time.
