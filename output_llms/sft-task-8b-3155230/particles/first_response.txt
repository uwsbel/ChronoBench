

```python
import pychrono as chrono
from pychrono import irrlicht
import numpy as np

system = chrono.ChSystemNSC()
system.SetTimeStep(1e-3)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, 0))
visual_system = irrlicht.ChVisualSystemIrrlicht()
visual_system.SetWindowSize(1024, 768)
visual_system.SetWindowTitle('Particle Gravity Simulation')
visual_system.Initialize()
visual_system.AddSkyBox()
visual_system.AddCamera(chrono.ChVector3d(0, 3, 5), chrono.ChVector3d(0, 0, 0))
visual_system.AddLightDirectional()

G = 1e-5

def add_particle(system):
    radius = np.random.uniform(0.1, 0.5)
    color = chrono.ChColor(0.5 + 0.5*np.random.rand(), 0.5 + 0.5*np.random.rand(), 0.5 + 0.5*np.random.rand())
    pos = chrono.ChVector3d(np.random.uniform(-5, 5), np.random.uniform(-5, 5), np.random.uniform(-5, 5))
    vel = chrono.ChVector3d(np.random.uniform(-1, 1), np.random.uniform(-1, 1), np.random.uniform(-1, 1))
    orientation = chrono.QuatFromAngleZ(np.random.uniform(0, 2*np.pi))
    body = chrono.ChBodyEasySphere(radius, 1000, True, False)
    body.SetMass(1.0)
    body.SetInertiaXX(chrono.ChVector3d(0.01, 0.01, 0.01))
    body.SetPos(pos)
    body.SetPosDt(vel)
    body.SetRot(orientation)
    body.GetVisualShape(0).SetColor(color)
    system.Add(body)
    return body

particles = [add_particle(system) for _ in range(50)]

step = 0
while visual_system.Run():
    step += 1
    if step % 100 == 0:
        particles.append(add_particle(system))
    for i in range(len(particles)):
        for j in range(i+1, len(particles)):
            body1 = particles[i]
            body2 = particles[j]
            r1 = body1.GetPos()
            r2 = body2.GetPos()
            r = r2 - r1
            dist = r.Length()
            if dist < 1e-6:
                continue
            force_mag = G * body1.GetMass() * body2.GetMass() / (dist**2)
            force = r.Normalize() * force_mag
            body1.ApplyForceAtPos(force, r1, True)
            body2.ApplyForceAtPos(-force, r2, True)
    system.DoStepDynamics(1e-3)
    visual_system.BeginScene()
    visual_system.DrawAll()
    visual_system.EndScene()
```
