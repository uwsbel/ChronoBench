

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import math

# Initialize PyChrono system
system = chrono.ChSystemNSC()
system.SetGravitationalAcceleration(chrono.ChVector3d(0, -9.81, 0))

# Create ground body (rigid terrain)
ground = chrono.ChBodyEasyBox(100, 100, 0.1, 1000, True, False)
ground.SetFixed(True)
ground.SetPos(chrono.ChVector3d(0, -0.05, 0))
system.AddBody(ground)

# Create Viper rover
class ViperRover:
    def __init__(self, system):
        self.body = chrono.ChBodyEasyBox(1.5, 0.8, 0.3, 1000, True, False)
        self.body.SetFixed(False)
        self.body.SetPos(chrono.ChVector3d(0, 0.15, 0))
        self.body.SetMass(100)
        self.body.SetInertiaXX(chrono.ChVector3d(10, 20, 10))
        system.AddBody(self.body)

        # Wheels
        self.wheels = []
        wheel_radius = 0.2
        wheel_mass = 10
        for i in range(4):
            wheel = chrono.ChBodyEasySphere(wheel_radius, 1000, True, False)
            wheel.SetFixed(True)
            wheel.SetPos(chrono.ChVector3d(0, 0, 0))
            system.AddBody(wheel)
            self.wheels.append(wheel)

        # Joints
        self.joints = []
        for i in range(4):
            joint = chrono.ChLinkRevolute()
            joint.Initialize(self.body, self.wheels[i], chrono.ChFramed(chrono.ChVector3d(0, 0, 0), chrono.QUNIT))
            system.AddLink(joint)
            self.joints.append(joint)

        # Driver
        self.steering = 0.0
        self.throttle = 0.5

    def update(self, time):
        # Update steering angle over time
        self.steering = 0.3 * math.sin(2 * math.pi * time / 10.0)
        # Apply steering to front wheels
        for i in range(2):
            self.joints[i].GetRevolute().SetAngle(self.steering)
        # Apply throttle to rear wheels
        for i in range(2, 4):
            self.wheels[i].ApplyForce(chrono.ChVector3d(self.throttle * 1000, 0, 0), True)

# Create rover
rover = ViperRover(system)

# Create Irrlicht visualization
application = chronoirr.ChVisualSystemIrrlicht()
application.AssetBindAll()
application.AssetUpdateAll()

# Set camera
camera = chronoirr.ChCameraAuxA()
application.GetSceneManager().AddCamera(camera)
camera.SetPosition(chrono.ChVector3d(2, 2, 2))
camera.SetTarget(chrono.ChVector3d(0, 0.15, 0))

# Add lights
light = chronoirr.ChLightSun()
light.SetDirection(chrono.ChVector3d(0, -1, -1))
application.GetSceneManager().AddLight(light)

# Simulation loop
time = 0.0
time_step = 1e-3
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.EndScene()
    system.DoStepDynamics(time_step)
    time += time_step
```
