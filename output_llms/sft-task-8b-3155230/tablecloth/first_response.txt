

```python
import pychrono as chrono
from pychrono import irrlicht
import numpy as np

system = chrono.ChSystemNSC()
system.SetSolverType(chrono.SolverType_PardisoMKL)
system.SetGravitationalAcceleration(chrono.Vect(0, -9.81, 0))

material = chrono.ChMaterialShellKirchhoff()
material.SetE(1e6)
material.SetNu(0.3)
material.SetThickness(0.005)

nx = 20
ny = 20
dx = 1.0 / (nx - 1)
dy = 1.0 / (ny - 1)
nodes = []
for i in range(nx):
    for j in range(ny):
        x = i * dx
        y = j * dy
        node = chrono.ChNodeXYZ()
        node.SetPos(chrono.ChVector3d(x, y, 0))
        system.Add(node)
        nodes.append(node)

elements = []
for i in range(nx - 1):
    for j in range(ny - 1):
        n1 = nodes[i * ny + j]
        n2 = nodes[(i + 1) * ny + j]
        n3 = nodes[(i + 1) * ny + j + 1]
        n4 = nodes[i * ny + j + 1]
        shell = chrono.ChShellElement()
        shell.SetNodes([n1, n2, n3, n4])
        shell.SetMaterial(material)
        system.Add(shell)
        elements.append(shell)

mesh_data = chrono.ChTriangleMeshConnected()
visual_shape = chrono.ChVisualShapeMesh()
visual_shape.SetMesh(mesh_data)
visual_shape.SetColor(chrono.ChColor(0.5, 0.5, 0.5))
system.AddVisualShape(visual_shape)

application = irrlicht.ChIrrApp(system, "Tablecloth Folding Simulation", irrlicht.dimension2d_i(800, 600))
application.AddSkyBox()
application.AddCamera(chrono.ChVector3d(1.5, 1.5, 1.5))
application.AddTypicalLights()
application.SetTimestep(1e-3)

while application.GetDevice().run():
    application.BeginScene()
    mesh_data.Clear()
    for shell in elements:
        n1 = shell.GetNode(0)
        n2 = shell.GetNode(1)
        n3 = shell.GetNode(2)
        n4 = shell.GetNode(3)
        mesh_data.AddTriangle(n1.GetPos(), n2.GetPos(), n3.GetPos())
        mesh_data.AddTriangle(n1.GetPos(), n3.GetPos(), n4.GetPos())
    visual_shape.SetMesh(mesh_data)
    application.DrawAll()
    application.EndScene()
```
