

```python
import chrono
import chrono_irrlicht as chronoirr
import math

# 1. Initialize system and set gravity
system = chrono.ChSystemNSC()
system.SetGravity(chrono.ChVector3d(0, 0, -9.81))

# 2. Create ground (truss) with texture
ground = chrono.ChBody()
ground.SetFixed(True)
floor_shape = chrono.ChVisualShapeBox(20, 1, 20)
floor_shape.SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
ground.AddVisualShape(floor_shape, chrono.ChFramed(chrono.ChVector3d(0, -0.5, 0), chrono.QUNIT))
system.Add(ground)

# 3. Create crankshaft (rotating body)
crank_radius = 0.2
crank = chrono.ChBodyEasyCylinder(0.05, 0.1, 1000)  # Thin cylinder for visualization
crank.SetPos(chrono.ChVector3d(0, 0, 0))
crank.SetRot(chrono.Q_ROTATE_Y_TO_X)  # Orient cylinder along X-axis
crank.GetVisualShape(0).SetColor(chrono.ChColor(0.8, 0.2, 0.2))
system.Add(crank)

# 4. Add rotation motor between crank and ground
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(crank, ground, chrono.ChFramed(chrono.ChVector3d(0, 0, 0), chrono.QUNIT))
motor.SetSpeedFunction(chrono.ChFunction_Const(math.pi * 2))  # 1 full rotation per second
system.Add(motor)

# 5. Create connecting rod
rod_length = 0.5
rod = chrono.ChBodyEasyBox(rod_length, 0.1, 0.1, 1000)
rod.SetPos(chrono.ChVector3d(crank_radius + rod_length/2, 0, 0))
rod.GetVisualShape(0).SetColor(chrono.ChColor(0.2, 0.6, 0.8))
system.Add(rod)

# 6. Connect crank and rod with revolute joint
revolute1 = chrono.ChLinkLockRevolute()
revolute1.Initialize(crank, rod, chrono.ChFramed(chrono.ChVector3d(crank_radius, 0, 0), chrono.QUNIT))
system.Add(revolute1)

# 7. Create piston
piston = chrono.ChBodyEasyBox(0.3, 0.3, 0.3, 1000)
piston.SetPos(chrono.ChVector3d(crank_radius + rod_length, 0, 0))
piston.GetVisualShape(0).SetColor(chrono.ChColor(0.8, 0.8, 0.2))
system.Add(piston)

# 8. Connect rod and piston with revolute joint
revolute2 = chrono.ChLinkLockRevolute()
revolute2.Initialize(rod, piston, chrono.ChFramed(chrono.ChVector3d(crank_radius + rod_length, 0, 0), chrono.QUNIT))
system.Add(revolute2)

# 9. Constrain piston with prismatic joint
prismatic = chrono.ChLinkLockPrismatic()
prismatic.Initialize(piston, ground, chrono.ChFramed(chrono.ChVector3d(0, 0, 0), chrono.QUNIT))
system.Add(prismatic)

# 10. Set up Irrlicht visualization
app = chronoirr.ChIrrApp(system, "Crank-Slider Mechanism", chronoirr.Dimension2du(1280, 720))
app.AddTypicalLogo(chrono.GetChronoDataFile("logo_chronoengine.png"))
app.AddTypicalSky()
app.AddTypicalLights(chronoirr.vector3df(2, 4, 2), chronoirr.vector3df(-2, -4, -2))
app.AddTypicalCamera(chronoirr.vector3df(2, 2, 2), chronoirr.vector3df(0, 0, 0))

# Configure visualization settings
app.AssetBindAll()
app.AssetUpdateAll()
app.SetTimestep(0.01)
app.SetTryRealtime(True)

# Main simulation loop
while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()
    app.DoStep()
    app.EndScene()
```
