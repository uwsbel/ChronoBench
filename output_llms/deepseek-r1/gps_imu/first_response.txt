Here's a complete PyChrono simulation script implementing the requested features with detailed comments:

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import pychrono.sensor as sens

def main():
    # ==========================
    # 1. Initialize Environment
    # ==========================
    chrono.SetChronoDataPath(chrono.GetChronoDataPath())  # Use default data path
    system = chrono.ChSystemNSC()
    system.SetGravitationalAcceleration(chrono.ChVectorD(0, 0, -9.81))
    system.SetSolverMaxIterations(100)

    # =====================
    # 2. Create Terrain
    # =====================
    terrain = veh.RigidTerrain(system)
    patch_mat = chrono.ChMaterialSurfaceNSC()
    patch = terrain.AddPatch(patch_mat, chrono.ChCoordsysD(), 100, 100)
    patch.SetTexture(veh.GetDataFile("terrain/textures/concrete.jpg"), 100, 100)
    terrain.Initialize()

    # =====================
    # 3. Create Vehicle
    # =====================
    # Initialize HMMWV vehicle
    vehicle = veh.HMMWV_Full(system)
    vehicle.Initialize(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0.5), chrono.QUNIT))
    vehicle.SetChassisVisualizationType(veh.VisualizationType_PRIMITIVES)
    vehicle.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
    vehicle.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
    vehicle.SetWheelVisualizationType(veh.VisualizationType_MESH)
    
    # Get vehicle systems
    chassis = vehicle.GetChassisBody()
    powertrain = veh.SimplePowertrain()

    # =====================
    # 4. Add Driver Inputs
    # =====================
    driver = veh.ChDriver(vehicle.GetVehicle())
    # Set constant driving controls (throttle=50%, steering=0%, braking=0%)
    driver.SetThrottle(0.5)
    driver.SetSteering(0.0)
    driver.SetBraking(0.0)

    # =====================
    # 5. Add Sensors
    # =====================
    # Create sensor manager
    manager = sens.ChSensorManager(system)
    manager.scene.AddPointLight(chrono.ChVectorF(100, 100, 100), chrono.ChColor(1, 1, 1), 5000)

    # IMU sensor on chassis
    imu_offset = chrono.ChVectorD(0, 0, 0.5)
    imu_rot = chrono.ChQuaternionD(1, 0, 0, 0)
    imu = sens.ChAccelerometerSensor(
        chassis,              # Parent body
        100,                  # Update rate [Hz]
        chrono.ChFrameD(imu_offset, imu_rot),  # Offset pose
    )
    imu.SetName("Vehicle IMU")
    imu.PushFilter(sens.ChFilterAccelAccess())
    manager.AddSensor(imu)

    # GPS sensor on chassis
    gps_offset = chrono.ChVectorD(-1.5, 0, 0.5)
    gps = sens.ChGPSSensor(
        chassis,
        10,                   # Update rate [Hz]
        chrono.ChFrameD(gps_offset, imu_rot),
    )
    gps.SetName("Vehicle GPS")
    gps.PushFilter(sens.ChFilterGPSAccess())
    manager.AddSensor(gps)

    # ===========================
    # 6. Setup Visualization
    # ===========================
    vis = irr.ChVisualSystemIrrlicht()
    vis.AttachSystem(system)
    vis.SetWindowSize(1280, 720)
    vis.SetWindowTitle("HMMWV Simulation with Sensors")
    vis.Initialize()
    vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
    vis.AddSkyBox()
    vis.AddCamera(chrono.ChVectorD(8, 8, 1.5))
    vis.AddTypicalLights()
    vehicle.GetVehicle().AddVisualizationAssets(irr)

    # ===========================
    # 7. Simulation Loop
    # ===========================
    step_size = 0.02
    realtime_step = True

    while vis.Run():
        time = system.GetChTime()
        
        # Update vehicle systems
        driver.Synchronize(time)
        vehicle.Synchronize(time, driver.GetInputs(), terrain)
        terrain.Synchronize(time)
        manager.Update()
        
        # Advance simulation
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        system.DoStepDynamics(step_size)
        
        # =============================================
        # 8. Output Information & Sensor Data Handling
        # =============================================
        # Print vehicle mass
        if abs(time % 0.1) < step_size:  # Print every 0.1s
            print(f"Time: {time:.2f}s | Vehicle Mass: {vehicle.GetVehicle().GetMass():.1f} kg")
        
        # Process IMU data
        imu_data = imu.GetMostRecentBuffer()
        if imu_data.HasData():
            accel_data = imu_data.GetAccelData()
            print(f"IMU Acceleration: X={accel_data[0]:.2f} Y={accel_data[1]:.2f} Z={accel_data[2]:.2f} m/s²")
        
        # Process GPS data
        gps_data = gps.GetMostRecentBuffer()
        if gps_data.HasData():
            gps_pos = gps_data.GetGPSData()
            print(f"GPS Position: Lat={gps_pos.Latitude:.6f} Lon={gps_pos.Longitude:.6f} Alt={gps_pos.Altitude:.2f} m")

if __name__ == "__main__":
    main()
```

Key components of this implementation:

1. **Environment Initialization**:
   - Sets up physical system with gravity
   - Configures collision and solver parameters

2. **Vehicle System**:
   - Full HMMWV model with realistic mass properties
   - Proper visualization settings for different components

3. **Terrain**:
   - Rigid flat terrain with concrete texture
   - Customizable size and material properties

4. **Driver Inputs**:
   - Constant throttle input (50%)
   - Can be extended with interactive controls

5. **Sensors**:
   - IMU sensor @ 100Hz on chassis center
   - GPS sensor @ 10Hz on front of chassis
   - Data filters for direct access to measurements

6. **Visualization**:
   - Irrlicht with proper camera setup
   - Vehicle visualization assets
   - Lighting and skybox enhancements

7. **Simulation Loop**:
   - Real-time synchronization
   - Physics stepping with fixed time increment
   - Sensor data polling and processing
   - Periodic output of vehicle mass

8. **Data Output**:
   - Console output of vehicle mass
   - Real-time sensor data printing
   - Visual rendering at each frame
