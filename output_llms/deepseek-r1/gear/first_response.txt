
To create a simulation of an epicyclic gear system in PyChrono, we'll set up a fixed sun gear, a rotating carrier arm, and a planetary gear meshed with the sun gear. A motor drives the carrier arm at a constant speed. Here's the complete solution:

```python
import pychrono as chrono
import pychrono.irrlicht as irr

def main():
    # 1. Initialize system
    system = chrono.ChSystemNSC()
    system.SetSolverType(chrono.ChSolver.Type_BARZILAIBORWEIN)

    # Create materials for visualization
    red_material = chrono.ChVisualMaterial()
    red_material.SetDiffuseColor(chrono.ChColor(0.8, 0.2, 0.2))
    blue_material = chrono.ChVisualMaterial()
    blue_material.SetDiffuseColor(chrono.ChColor(0.2, 0.2, 0.8))
    green_material = chrono.ChVisualMaterial()
    green_material.SetDiffuseColor(chrono.ChColor(0.2, 0.8, 0.2))

    # 2. Create ground body
    ground = chrono.ChBody()
    ground.SetBodyFixed(True)
    system.Add(ground)

    # 3. Create sun gear (fixed to ground)
    sun_radius = 0.5
    sun = chrono.ChBody()
    sun.SetPos(chrono.ChVectorD(0, 0, 0))
    sun_shape = chrono.ChCylinderShape()
    sun_shape.GetCylinderGeometry().rad = sun_radius
    sun_shape.GetCylinderGeometry().p1 = chrono.ChVectorD(0, 0, -0.2)
    sun_shape.GetCylinderGeometry().p2 = chrono.ChVectorD(0, 0, 0.2)
    sun.AddVisualShape(sun_shape)
    sun.GetVisualShape(0).SetMaterial(0, red_material)
    system.Add(sun)

    # Fix sun gear to ground
    sun_to_ground = chrono.ChLinkLockRevolute()
    sun_to_ground.Initialize(sun, ground, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0)))
    system.AddLink(sun_to_ground)

    # 4. Create carrier arm
    arm_length = sun_radius * 1.8
    arm = chrono.ChBody()
    arm.SetPos(chrono.ChVectorD(0, 0, 0))
    arm_shape = chrono.ChBoxShape()
    arm_shape.GetBoxGeometry().Size = chrono.ChVectorD(arm_length/2, 0.05, 0.05)
    arm.AddVisualShape(arm_shape, chrono.ChFrameD(chrono.ChVectorD(arm_length/2, 0, 0)))
    arm.GetVisualShape(0).SetMaterial(0, green_material)
    system.Add(arm)

    # Connect arm to ground with motor
    motor = chrono.ChLinkMotorRotationAngle()
    motor.Initialize(arm, ground, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)))
    motor_func = chrono.ChFunction_Ramp(0, chrono.CH_C_PI / 4)  # 45 deg/s rotation
    motor.SetAngleFunction(motor_func)
    system.AddLink(motor)

    # 5. Create planetary gear
    planet_radius = sun_radius * 0.4
    planet = chrono.ChBody()
    planet.SetPos(chrono.ChVectorD(arm_length, 0, 0))
    planet_shape = chrono.ChCylinderShape()
    planet_shape.GetCylinderGeometry().rad = planet_radius
    planet_shape.GetCylinderGeometry().p1 = chrono.ChVectorD(0, 0, -0.2)
    planet_shape.GetCylinderGeometry().p2 = chrono.ChVectorD(0, 0, 0.2)
    planet.AddVisualShape(planet_shape)
    planet.GetVisualShape(0).SetMaterial(0, blue_material)
    system.Add(planet)

    # Connect planetary gear to arm
    planet_to_arm = chrono.ChLinkLockRevolute()
    planet_to_arm.Initialize(planet, arm, 
                            chrono.ChCoordsysD(chrono.ChVectorD(arm_length, 0, 0)))
    system.AddLink(planet_to_arm)

    # 6. Create gear constraint between sun and planet
    gear_train = chrono.ChLinkGear()
    gear_train.Initialize(sun, planet, ground, 
                         chrono.ChFrameD(chrono.ChVectorD(0, 0, 0)),
                         chrono.ChFrameD(chrono.ChVectorD(arm_length, 0, 0)))
    ratio = planet_radius / sun_radius  # Negative ratio for external gears
    gear_train.SetTransmissionRatio(-ratio)
    system.AddLink(gear_train)

    # 7. Set up Irrlicht visualization
    vis = irr.ChVisualSystemIrrlicht()
    vis.AttachSystem(system)
    vis.SetWindowSize(1024, 768)
    vis.SetWindowTitle('Epicyclic Gear System')
    vis.Initialize()
    vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
    vis.AddSkyBox()
    vis.AddCamera(chrono.ChVectorD(2, 1.5, 1))
    vis.AddTypicalLights()

    # 8. Run simulation
    time_step = 0.01
    while vis.Run():
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        system.DoStepDynamics(time_step)

if __name__ == '__main__':
    main()
```

Key components of this simulation:
1. **System Setup**: Uses NSC (Non-Smooth Contacts) solver for mechanical simulation
2. **Gear System**:
   - Fixed red sun gear at origin
   - Green rotating carrier arm driven by a motor
   - Blue planetary gear attached to the arm's end
3. **Mechanical Constraints**:
   - Revolute joints for rotational connections
   - Gear constraint to enforce meshing between sun and planetary gears
   - Motor with constant angular speed (45 deg/s) driving the arm
4. **Visualization**:
   - Color-coded materials for clear differentiation
   - Camera positioned for optimal viewing
   - Skybox and lighting for enhanced visuals

The simulation demonstrates fundamental epicyclic gear behavior where:
- The motor rotates the carrier arm
- The planetary gear orbits around the sun gear
- The gear constraint maintains proper meshing between gears
- Rotation speeds follow the gear ratio (sun_radius/planet_radius)

To run this, ensure you have PyChrono installed. The simulation will show the gear system in action with the planetary gear both rotating and revolving around the sun gear.
