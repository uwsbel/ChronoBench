```python
import pychrono as chrono
import pychrono.irrlicht as irrlicht
import pychrono.fea as fea
from pychrono import ChVectorD, ChQuaternionD

def main():
    # 1. Initialize system
    system = chrono.ChSystemNSC()
    system.Set_G_acc(ChVectorD(0, 0, 0))  # Optional: Set gravity if needed

    # 2. Create a mesh for finite elements
    mesh = fea.ChMesh()
    system.Add(mesh)

    # 3. Create beam material properties
    density = 7850  # kg/m³ (steel)
    E = 2.1e11      # Young's modulus (Pa)
    nu = 0.3        # Poisson ratio
    beam_material = fea.ChMaterialBeamEuler(density, E, nu)

    # Beam cross-section properties
    area = 0.001        # Cross-sectional area (m²)
    Iyy = 8.33e-8       # Area moment of inertia
    Izz = 8.33e-8
    section = fea.ChBeamSectionEuler(area, Iyy, Izz, beam_material)

    # 4. Create nodes and elements
    num_elements = 10
    length = 1.0  # Total beam length
    dx = length / num_elements

    # Create first node (fixed)
    node1 = fea.ChNodeFEAxyzrot(chrono.ChFrameD(ChVectorD(0, 0, 0)))
    node1.SetFixed(True)
    mesh.AddNode(node1)

    prev_node = node1
    for i in range(1, num_elements + 1):
        # Create subsequent nodes
        new_node = fea.ChNodeFEAxyzrot(chrono.ChFrameD(ChVectorD(i * dx, 0, 0)))
        mesh.AddNode(new_node)

        # Create beam element between nodes
        element = fea.ChElementBeamEuler()
        element.SetNodes(prev_node, new_node)
        element.SetSection(section)
        mesh.AddElement(element)

        prev_node = new_node

    # 5. Apply force at the free end
    last_node = prev_node
    last_node.SetForce(ChVectorD(0, 0, -0.5))  # Apply vertical force

    # 6. Add visualization
    vis_shape = fea.ChVisualShapeFEA(mesh)
    vis_shape.SetFEMdataType(fea.ChVisualShapeFEA.DataType_ELEM_BEAM_MZ)
    vis_shape.SetColorscaleMinMax(-50, 50)
    vis_shape.SetSmoothFaces(True)
    vis_shape.SetWireframe(True)
    mesh.AddVisualShapeFEA(vis_shape)

    # 7. Set up Irrlicht visualization
    vis = irrlicht.ChIrrApp(system, "Beam FEM Demo", irrlicht.dimension2du(1280, 720))
    vis.AddTypicalLights()
    vis.AddTypicalCamera(irrlicht.vector3df(1.5, 0.5, 1.5))
    vis.AssetBindAll()
    vis.AssetUpdateAll()

    # 8. Simulation loop
    timestep = 0.001
    while vis.GetDevice().run():
        vis.BeginScene()
        vis.DrawAll()
        vis.GetVideoDriver().setMaterial(vis.GetVideoDriver().getMaterial2D())
        vis.GetSystem().DoStepDynamics(timestep)
        vis.EndScene()

if __name__ == '__main__':
    main()
```

This script:
1. Creates a cantilever beam using Euler-Bernoulli beam elements
2. Fixes the first node and applies vertical force at the free end
3. Implements color-coded visualization of bending moments
4. Uses Irrlicht for real-time 3D visualization
5. Sets up proper beam material and section properties
6. Runs an interactive simulation loop

The beam will bend under the applied force, demonstrating FEM beam element behavior. Adjust parameters like force magnitude, beam length, or material properties to observe different responses.
