

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr

def main():
    # 1. Initialize system and contact method
    system = chrono.ChSystemSMC()
    system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

    # 2. Create vehicle and set parameters
    vehicle = veh.WheeledVehicle(system, veh.GetDataFile('artcar/ARTcar_Vehicle.json'))
    init_pos = chrono.ChVectorD(0, 0, 0.5)
    init_rot = chrono.ChQuaternionD(1, 0, 0, 0)
    vehicle.Initialize(chrono.ChCoordsysD(init_pos, init_rot))
    vehicle.SetChassisVisualizationType(veh.VisualizationType_PRIMITIVES)
    vehicle.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
    vehicle.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
    vehicle.SetWheelVisualizationType(veh.VisualizationType_MESH)

    # 3. Create rigid terrain
    terrain = veh.RigidTerrain(system)
    patch_mat = chrono.ChMaterialSurfaceSMC()
    patch_mat.SetFriction(0.9)
    patch_mat.SetRestitution(0.01)
    patch = terrain.AddPatch(patch_mat, 
                            chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT),
                            200, 200)  # 200x200 meter terrain
    patch.SetTexture(veh.GetDataFile('terrain/textures/concrete.jpg'), 200, 200)
    terrain.Initialize()

    # 4. Initialize Irrlicht visualization
    app = chronoirr.ChIrrApp(system, 'ARTcar Simulation', chronoirr.dimension2du(1280, 720))
    app.AddTypicalLights()
    app.AddTypicalCamera(chronoirr.vector3df(2, 2, 1))
    app.AssetBindAll()
    app.AssetUpdateAll()

    # 5. Create interactive driver system
    driver = veh.InteractiveDriverIRR(app)
    driver.SetSteeringDelta(0.02)
    driver.SetThrottleDelta(0.02)
    driver.SetBrakingDelta(0.02)

    # 6. Simulation loop
    time_step = 0.01
    frame_step = 0.02  # 50 FPS
    time = 0
    while app.GetDevice().run():
        time += time_step
        
        # Update driver inputs
        driver_inputs = driver.GetInputs()
        vehicle.Synchronize(time, driver_inputs)
        terrain.Synchronize(time)
        
        # Advance simulation
        app.BeginScene()
        app.DrawAll()
        system.DoStepDynamics(time_step)
        app.EndScene()
        
        # Maintain 50 FPS
        chrono.ChRealtimeStepTimer.Sleep(frame_step)

if __name__ == "__main__":
    main()
```

**Important Notes:**
1. The ARTcar vehicle specification assumes the existence of `artcar/ARTcar_Vehicle.json` in Chrono's data directory. Adjust paths according to your Chrono installation.
2. Replace `'terrain/textures/concrete.jpg'` with your actual texture path
3. Adjust terrain dimensions and material properties as needed
4. The interactive driver uses arrow keys for steering and WASD for throttle/braking
5. Visualization types can be changed between `PRIMITIVES` and `MESH` based on requirements
6. Camera position and lighting can be adjusted in the visualization setup
