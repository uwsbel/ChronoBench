```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as robot
from pychrono import irrlicht as chronoirr

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create ground body with contact material and add it to the system
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVector3d(0, 0, -0.5))  # Position the ground slightly below the origin
ground.SetFixed(True)  # Fix the ground in place
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# Create obstacle (long box for rover to cross)
obstacle_mat = chrono.ChContactMaterialNSC()
obstacle = chrono.ChBodyEasyBox(8.0, 0.5, 0.2, 1000, True, True, obstacle_mat)
obstacle.SetPos(chrono.ChVector3d(0, 0, 0.1))  # Positioned at center, slightly above ground
obstacle.SetFixed(True)  # Fix the obstacle in place
obstacle.GetVisualShape(0).SetColor(chrono.ChColor(0.9, 0.2, 0.2))  # Red color for visibility
system.Add(obstacle)

# Create Curiosity rover and add it to the system
rover = robot.Curiosity(system)

# Create driver for rover
driver = robot.CuriosityDCMotorControl()
rover.SetDriver(driver)

# Initialize rover position and orientation (modified to new starting point)
init_pos = chrono.ChVector3d(-5, 0.0, 0)  # Changed from (0, 0.2, 0) to (-5, 0.0, 0)
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)
rover.Initialize(chrono.ChFrameD(init_pos, init_rot))  # Fixed typo: ChFramed -> ChFrameD

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Curiosity rover - Rigid terrain')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 3), chrono.ChVector3d(0, 0, 0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0), 3, 4, 10, 40, 512)

# Enable shadows (commented out to improve performance)
# vis.EnableShadows()

# Set the simulation time step
time_step = 1e-3

# Simulation loop
time = 0
while vis.Run():
    time += time_step

    # Set rover to move forward with zero steering (modified per instructions)
    driver.SetSteering(0.0)        # Constant zero steering
    driver.SetSpeed(3.0)           # Constant forward speed (3 rad/s wheel rotation)

    # Update rover dynamics
    rover.Update()

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance simulation by one time step
    system.DoStepDynamics(time_step)
```

Key changes made:
1. **Added Obstacle**: Created an 8m long, 0.5m wide, 0.2m high box obstacle at position (0, 0, 0.1) with red color for visibility
2. **Fixed Rover Initialization**: Corrected `ChFramed` to `ChFrameD` and changed starting position to (-5, 0.0, 0)
3. **Modified Rover Control**: 
   - Removed time-based steering logic
   - Set constant zero steering
   - Added constant forward speed (3 rad/s wheel rotation)
4. **Visual Improvements**: 
   - Added red color to obstacle for better visibility
   - Maintained shadow setup (commented but available if needed)

The rover will now start 5 meters behind the origin, drive straight forward (X-direction) at constant speed, and encounter the obstacle centered at the origin.
