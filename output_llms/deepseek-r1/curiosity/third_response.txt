```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as robot
import pychrono.sensor as sens  # Added sensor module import
from pychrono import irrlicht as chronoirr

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create ground body with contact material and add it to the system
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVector3d(0, 0, -0.5))  # Position the ground slightly below the origin
ground.SetFixed(True)  # Fix the ground in place
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# create a long box for rover to cross
box = chrono.ChBodyEasyBox(0.25, 5, 0.25, 1000, True, True, ground_mat)
box.SetPos(chrono.ChVector3d(0, 0, 0.0))
box.SetFixed(True)
box.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/blue.png"))
system.Add(box)

# Create Curiosity rover and add it to the system
rover = robot.Curiosity(system)

# Create driver for rover
driver = robot.CuriosityDCMotorControl()
rover.SetDriver(driver)

# Initialize rover position and orientation - FIXED initialization frame
init_pos = chrono.ChVector3d(-5, 0.0, 0)
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)
rover.Initialize(chrono.ChFrameD(init_pos, init_rot))  # Fixed ChFrameD initialization

# Create sensor manager - ADDED as per instructions
manager = sens.ChSensorManager(system)
manager.scene.AddPointLight(chrono.ChVector3d(2, 2.5, 5), chrono.ChColor(1, 1, 1), 500.0)

# Create and configure lidar sensor - ADDED as per instructions
lidar_offset_pose = chrono.ChFrameD(chrono.ChVector3d(0, 0, 0.5), chrono.Q_from_AngAxis(0, chrono.ChVector3d(0, 0, 1)))
lidar = sens.ChLidarSensor(
    rover.GetChassisBody(),  # Parent body
    10,                      # Update rate (Hz)
    lidar_offset_pose,       # Offset pose
    1000,                    # Horizontal samples
    64,                      # Vertical samples
    chrono.CH_PI,            # Horizontal field of view (180°)
    chrono.CH_PI / 9,        # Vertical field of view (20°)
    100.0,                   # Max distance
)
lidar.SetName("Rover Lidar")
lidar.SetLag(0.0)
lidar.SetCollectionWindow(0.0)

# Add noise model - ADDED as per instructions
noise_model = sens.ChNoiseNormalDist(0.0, 0.01)
lidar.SetNoiseModel(noise_model)

# Add filters - ADDED as per instructions
lidar.PushFilter(sens.ChFilterPCfromDepth())
lidar.PushFilter(sens.ChFilterVisualize(1000, 64, "Lidar Point Cloud"))

# Add sensor to manager - ADDED as per instructions
manager.AddSensor(lidar)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Curiosity Rover with Lidar Sensor')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 3), chrono.ChVector3d(0, 0, 0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0), 3, 4, 10, 40, 512)

# Set the simulation time step
time_step = 1e-3

# Simulation loop
time = 0
while vis.Run():
    time += time_step

    # ask rover to move forward
    driver.SetSteering(0.0)

    # Update rover dynamics
    rover.Update()

    # Update sensor manager - ADDED as per instructions
    manager.Update()

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance simulation by one time step
    system.DoStepDynamics(time_step)
```

Key changes made:
1. **Sensor Module Import**: Added `import pychrono.sensor as sens`
2. **Fixed Rover Initialization**: Changed `ChFramed` to correct `ChFrameD`
3. **Sensor Manager**: Created `sens.ChSensorManager`
4. **Lidar Configuration**:
   - Added to rover's chassis body at height 0.5m
   - Configured with 10Hz update rate
   - 1000 horizontal samples and 64 vertical samples
   - 180° horizontal FOV and 20° vertical FOV
   - 100m max distance
5. **Noise Model**: Added Gaussian noise with σ=0.01
6. **Filters**: Added point cloud generation and visualization
7. **Sensor Update**: Added `manager.Update()` in simulation loop
8. **Visualization**: Updated window title to reflect lidar addition

The lidar will now generate and visualize a point cloud from the rover's perspective as it moves across the terrain.
