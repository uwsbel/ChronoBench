

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

def main():
    # 1. Initialize Chrono system and core components
    chrono.SetChronoDataPath("path/to/chrono/data/")  # Update with actual path
    system = chrono.ChSystemNSC()
    system.SetSolverMaxIterations(150)
    system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

    # 2. Create BMW E90 Sedan vehicle
    vehicle = veh.Sedan()
    vehicle.Initialize(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0.5), chrono.QUNIT))
    vehicle.SetChassisVisualizationType(veh.VisualizationType_PRIMITIVES)
    vehicle.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
    vehicle.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
    vehicle.SetWheelVisualizationType(veh.VisualizationType_MESH)

    # 3. Create and configure TMeasy tires
    tire_l = veh.TMeasyTire("LeftTire")
    tire_r = veh.TMeasyTire("RightTire")
    for tire in [tire_l, tire_r]:
        tire.SetTireParams(veh.TMparams())
        tire.Initialize(vehicle.GetVehicle().GetWheelBody(veh.LEFT), veh.LEFT)
        tire.Initialize(vehicle.GetVehicle().GetWheelBody(veh.RIGHT), veh.RIGHT)
        system.Add(tire)

    # 4. Create rigid terrain with textures
    terrain = veh.RigidTerrain(system)
    patch_mat = chrono.ChMaterialSurfaceNSC()
    patch = terrain.AddPatch(patch_mat, chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0, 0, 1), 200, 200)
    patch.SetTexture(veh.GetDataFile("terrain/textures/concrete.jpg"), 200, 200)
    terrain.Initialize()

    # 5. Set up Irrlicht visualization
    vis = irr.ChVisualSystemIrrlicht()
    vis.AttachSystem(system)
    vis.SetWindowSize(1280, 720)
    vis.SetWindowTitle("BMW E90 Dynamics Simulation")
    vis.Initialize()
    vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
    vis.AddSkyBox()
    vis.AddCamera(chrono.ChVectorD(8, 8, 1.5), chrono.ChVectorD(0, 0, 1.5))
    vis.AddTypicalLights()
    vis.AddLightWithShadow(chrono.ChVectorD(-5, 0, 8), chrono.ChVectorD(0, 0, 2.5), 50, 2, 30, 50, 512)

    # 6. Create interactive driver system
    driver = veh.ChInteractiveDriverIRR(vis)
    driver.SetSteeringDelta(0.02)
    driver.SetThrottleDelta(0.02)
    driver.SetBrakingDelta(0.06)
    driver.Initialize()

    # 7. Simulation loop
    while vis.Run():
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        
        # Get driver inputs
        driver_inputs = driver.GetInputs()
        vehicle.GetVehicle().Synchronize(chrono.ChTime, driver_inputs, terrain)
        terrain.Synchronize(chrono.ChTime)
        vis.Synchronize(chrono.ChTime, driver_inputs)
        
        # Advance simulation
        system.DoStepDynamics(1e-3)

if __name__ == "__main__":
    main()
```

Key features:
- Uses Chrono's built-in Sedan model with TMeasy tires
- Implements interactive driver controls (arrow keys + WSAD)
- Features Irrlicht visualization with chase camera and shadows
- Includes textured terrain and skybox
- Proper physics setup with NSC contact system
- Real-time simulation with vehicle-terrain interaction

Note: Requires Chrono::Vehicle module and proper data path configuration. Adjust paths to match your Chrono installation.
