import pychrono as ch
import pychrono.ros as chros
import pychrono.irrlicht as ch_irr
import rclpy.publisher
import rclpy.qos
from std_msgs.msg import Int64

# Define a custom ROS handler for publishing integer messages.
class MyCustomHandler(chros.ChROSHandler):
    """This custom handler will publish integer messages to a specified ROS topic."""

    def __init__(self, topic):
        super().__init__(10)  # Set publishing rate to 10 Hz
        self.topic = topic
        self.publisher: rclpy.publisher.Publisher = None
        self.ticker = 0

    def Initialize(self, interface: chros.ChROSPythonInterface) -> bool:
        print(f"Creating publisher for topic {self.topic} ...")
        qos_profile = rclpy.qos.QoSProfile(depth=1)
        self.publisher = interface.GetNode().create_publisher(Int64, self.topic, qos_profile)
        return True

    def Tick(self, time: float):
        print(f"Publishing {self.ticker} ...")
        msg = Int64()
        msg.data = self.ticker
        self.publisher.publish(msg)
        self.ticker += 1

def main():
    sys = ch.ChSystemNSC()
    sys.SetGravitationalAcceleration(ch.ChVector3d(0, 0, -9.81))

    # Create Irrlicht visualization system
    vis = ch_irr.ChVisualSystemIrrlicht()
    vis.AttachSystem(sys)
    vis.SetWindowSize(1280, 720)
    vis.SetWindowTitle('PyChrono Simulation')
    vis.Initialize()
    vis.AddCamera(ch.ChVector3d(0, -3, 2), ch.ChVector3d(0, 0, 0))
    vis.AddTypicalLights()

    phys_mat = ch.ChContactMaterialNSC()
    phys_mat.SetFriction(0.5)

    # Create floor with texture
    floor = ch.ChBodyEasyBox(10, 10, 1, 1000, True, True, phys_mat)
    floor.SetPos(ch.ChVector3d(0, 0, -1))
    floor.SetFixed(True)
    floor.SetName("base_link")
    floor.GetVisualShape(0).SetTexture(ch.GetChronoDataFile("textures/concrete.jpg"))
    sys.Add(floor)

    # Create box with texture
    box = ch.ChBodyEasyBox(1, 1, 1, 1000, True, True, phys_mat)
    box.SetPos(ch.ChVector3d(0, 0, 5))
    box.SetRot(ch.QuatFromAngleAxis(.2, ch.ChVector3d(1, 0, 0)))
    box.SetName("box")
    box.GetVisualShape(0).SetTexture(ch.GetChronoDataFile("textures/blue.png"))
    sys.Add(box)

    # Configure ROS manager with 10 Hz publishing rate
    ros_manager = chros.ChROSPythonManager()
    ros_manager.RegisterHandler(chros.ChROSClockHandler(10))
    ros_manager.RegisterHandler(chros.ChROSBodyHandler(10, box, "~/box"))
    
    tf_handler = chros.ChROSTFHandler(10)
    tf_handler.AddTransform(floor, floor.GetName(), box, box.GetName())
    ros_manager.RegisterHandler(tf_handler)
    
    custom_handler = MyCustomHandler("~/my_topic")
    ros_manager.RegisterPythonHandler(custom_handler)
    ros_manager.Initialize()

    # Simulation parameters
    time_step = 1e-3
    time_end = 30
    time = 0
    step_number = 0
    render_step_size = 1/60  # 60 FPS
    render_steps = max(1, int(render_step_size / time_step))

    realtime_timer = ch.ChRealtimeStepTimer()
    while time < time_end:
        sys.DoStepDynamics(time_step)
        time = sys.GetChTime()

        if not ros_manager.Update(time, time_step):
            break

        # Conditional rendering
        if step_number % render_steps == 0:
            vis.BeginScene()
            vis.Render()
            vis.EndScene()

        realtime_timer.Spin(time_step)
        step_number += 1

if __name__ == "__main__":
    main()
