

```python
import chrono
import chrono.fea as fea
import chrono.irrlicht as irrlicht
from chrono import ChVectorD, ChLinkMotorLinearPosition, ChFunction_Ramp

# 1. Initialize system and set parameters
system = chrono.ChSystemNSC()
system.Set_G_acc(ChVectorD(0, 0, 0))  # Disable gravity

# 2. Create FEA mesh
mesh = fea.ChMesh()
system.Add(mesh)

# 3. Define beam properties
beam_length = 2.0
n_elements = 10
beam_section = fea.ChBeamSectionEuler()
beam_section.SetAsRectangularSection(0.05, 0.05)  # 5x5 cm cross-section
beam_section.SetYoungModulus(2e11)  # Steel
beam_section.SetGwithPoissonRatio(0.3)
beam_section.SetDensity(7850)

# 4. Create nodes and elements
nodes = []
for i in range(n_elements + 1):
    node = fea.ChNodeFEAxyzrot(chrono.ChFrameD(ChVectorD(i * beam_length/n_elements, 0, 0)))
    if i == 0:
        node.SetFixed(True)  # Fixed end
    nodes.append(node)
    mesh.AddNode(node)

for i in range(n_elements):
    element = fea.ChElementBeamEuler()
    element.SetNodes(nodes[i], nodes[i+1])
    element.SetSection(beam_section)
    mesh.AddElement(element)

# 5. Create motorized constraint for displacement
motor_body = chrono.ChBodyEasyBox(0.1, 0.1, 0.1, 1000)
motor_body.SetPos(nodes[-1].GetPos())
system.Add(motor_body)

motor_link = ChLinkMotorLinearPosition()
motor_link.Initialize(motor_body, chrono.ChBody(), 
                     chrono.ChCoordsysD(nodes[-1].GetPos()))
motor_link.SetMotorFunction(ChFunction_Ramp(0, -0.0005))  # Linear displacement
system.Add(motor_link)

# Connect motor body to end node
node_body_link = fea.ChLinkNodeBody()
node_body_link.Initialize(nodes[-1], motor_body)
system.Add(node_body_link)

# 6. Configure visualization
vis = irrlicht.ChIrrApp(system, "Beam Buckling Analysis", 800, 600)
vis.AddTypicalCamera(irrlicht.vector3df(-1, 0.5, 1))
vis.AddTypicalLights()
vis.AddFEAmesh(mesh)  # Add FEA mesh visualization
vis.AssetBindAll()
vis.AssetUpdateAll()

# 7. Configure solver and timestepper
solver = chrono.ChSolverMINRES()
solver.SetMaxIterations(200)
solver.SetTolerance(1e-10)
system.SetSolver(solver)

system.SetTimestepperType(chrono.ChTimestepper.Type_HHT)
hht_stepper = system.GetTimestepper().AsTimestepperHHT()
hht_stepper.SetAlpha(-0.2)
hht_stepper.SetGamma(0.5)
hht_stepper.SetStepControl(True)

# 8. Run simulation
vis.Run()

while vis.GetDevice().run():
    vis.BeginScene()
    vis.DrawAll()
    vis.DoStepDynamics(0.001)
    vis.EndScene()
```

This script:
- Creates a fixed-free Euler-Bernoulli beam using FEA
- Applies compressive displacement via motor constraint
- Uses HHT timestepper for numerical stability
- Implements MINRES solver for linear systems
- Includes Irrlicht visualization
- Demonstrates proper FEA constraints and loads

The beam will compress until buckling occurs due to the applied displacement. Parameters can be adjusted by modifying:
- Beam dimensions in `SetAsRectangularSection()`
- Material properties in beam section
- Displacement rate in `ChFunction_Ramp
