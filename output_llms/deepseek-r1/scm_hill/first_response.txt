

```python
import os
import math
import time
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# ----------------------------
# Simulation Initialization
# ----------------------------
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# ----------------------------
# HMMWV Vehicle Setup
# ----------------------------
hmmwv = veh.HMMWV_Full(system)
hmmwv.SetContactMethod(chrono.ChContactMethod_NSC)
hmmwv.SetChassisFixed(False)
hmmwv.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 1, 0), chrono.QUNIT))
hmmwv.SetTireStepSize(0.002)
hmmwv.Initialize()

# Set visualization parameters
hmmwv.SetChassisVisualizationType(veh.VisualizationType_PRIMITIVES)
hmmwv.SetSuspensionVisualizationType(veh.VisualizationType_PRIMITIVES)
hmmwv.SetSteeringVisualizationType(veh.VisualizationType_PRIMITIVES)
hmmwv.SetWheelVisualizationType(veh.VisualizationType_NONE)
hmmwv.SetTireVisualizationType(veh.VisualizationType_MESH)

# ----------------------------
# Deformable Terrain Setup
# ----------------------------
terrain = veh.SCMDeformableTerrain(system)
terrain.SetPlane(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT))

# Soil parameters
terrain.SetSoilParameters(
    Bekker_Kphi=1.209e6,
    Bekker_Kc=0.999e4,
    Bekker_n=1.01,
    Mohr_cohesion=3.96e4,
    Mohr_friction=31.1,
    Janosi_shear=0.0178
)

# Terrain visualization
terrain.Initialize(800, 400, 0.05)
terrain.SetPlotType(veh.SCMDeformableTerrain.PlotType_PRESSURE_YELD, 0, 30000)

# ----------------------------
# Driver System Setup
# ----------------------------
driver = veh.ChIrrGuiDriver(irr.ChVisualSystemIrrlicht())
driver.SetSteeringDelta(0.02)
driver.SetThrottleDelta(0.02)
driver.SetBrakingDelta(0.02)
hmmwv.GetVehicle().SetDriver(driver)

# ----------------------------
# Visualization System Setup
# ----------------------------
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('HMMWV on Deformable Terrain')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(-5, 2, -5), chrono.ChVectorD(0, 0, 0))
vis.AddLightWithShadow(chrono.ChVectorD(2,4,2), chrono.ChVectorD(0,0,0), 10, 2, 10, 40, 512)
vis.EnableShadows()

# ----------------------------
# Simulation Loop
# ----------------------------
time_step = 0.002
realtime_step = 0.02

while vis.Run():
    time = system.GetChTime()
    
    # Update driver inputs
    driver.Synchronize(time)
    
    # Update vehicle and terrain
    hmmwv.Synchronize(time, driver.GetInputs(), terrain)
    terrain.Synchronize(time)
    
    # Advance simulation
    system.DoStepDynamics(time_step)
    
    # Update visualization
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    
    # Maintain real-time pacing
    pause_time = system.GetChTime() - time
    if pause_time < realtime_step:
        time.sleep(realtime_step - pause_time)
```
