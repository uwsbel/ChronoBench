

```python
import os
import math
import chrono
import chrono.irrlicht as chronoirr
import chrono.sensor as sens

# Set data path
chrono.SetChronoDataPath(os.path.join(os.environ.get('CHRONO_DATA_DIR', ''), 'data/'))

# System setup
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))  # No gravity

# Create fixed triangular mesh body
mesh = chrono.ChTriangleMeshConnected()
mesh.LoadWavefrontMesh(chrono.GetChronoDataFile("models/cube.obj"))  # Replace with your mesh

mesh_body = chrono.ChBody()
mesh_body.SetBodyFixed(True)
mesh_body.SetPos(chrono.ChVectorD(0, 0, 0))
mesh_shape = chrono.ChTriangleMeshShape()
mesh_shape.SetMesh(mesh)
mesh_body.AddAsset(mesh_shape)
system.Add(mesh_body)

# Create orbiting lidar body
lidar_body = chrono.ChBody()
lidar_body.SetBodyFixed(False)
lidar_body.SetPos(chrono.ChVectorD(5, 1, 0))
lidar_body.SetBodyType(chrono.ChBody.BODY_KINEMATIC)
system.Add(lidar_body)

# Sensor manager
manager = sens.ChSensorManager(system)
manager.scene.AddPointLight(chrono.ChVectorD(2, 2.5, 0), chrono.ChColor(0.8, 0.8, 0.8), 500)

# Lidar configuration
lidar_offset = chrono.ChFrameD()
lidar = sens.ChLidarSensor(
    lidar_body,             # Parent body
    30,                     # Update rate (Hz)
    lidar_offset,           # Offset pose
    1280,                   # Horizontal samples
    720,                    # Vertical samples
    chrono.CH_C_PI / 1.5,  # Horizontal FOV (120°)
    chrono.CH_C_PI / 4.5,  # Vertical FOV (40°)
    100.0                   # Max detection distance
)

# Noise model
noise_model = sens.ChNoiseNormalDist(0.0, 0.02)
lidar.AddNoiseModel(noise_model)

# Visualization filter
lidar.PushFilter(sens.ChFilterVisualize(1280, 720, "Lidar Output"))

# Save point cloud
lidar.PushFilter(sens.ChFilterPCfromDepth())
lidar.PushFilter(sens.ChFilterSavePtCloud("lidar_data/"))

manager.AddSensor(lidar)

# Visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.Initialize()
vis.AddCamera(chrono.ChVectorD(0, 3, 3), chrono.ChVectorD(0, 0, 0))
vis.AddTypicalLights()

# Simulation parameters
time_step = 0.01
current_time = 0.0
radius = 5.0
angular_speed = 0.5  # rad/s

def Q_from_angle_axis(angle, axis):
    quat = chrono.ChQuaternionD()
    quat.Q_from_AngAxis(angle, axis)
    return quat

while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    
    # Update lidar position and orientation
    angle = angular_speed * current_time
    x = radius * math.cos(angle)
    z = radius * math.sin(angle)
    lidar_body.SetPos(chrono.ChVectorD(x, 1.0, z))
    
    # Calculate rotation to face origin
    direction = (-lidar_body.GetPos()).GetNormalized()
    up = chrono.ChVectorD(0, 1, 0)
    rot = chrono.ChMatrix33D()
    rot.Set_A_axis(direction, up, direction.Cross(up))
    lidar_body.SetRot(rot.Get_A_quaternion())
    
    # Update sensor manager
    manager.Update()
    
    # Print lidar data
    frame = lidar.GetMostRecentFrame()
    if frame.HasData():
        buffer = frame.GetDepthBuffer()
        print(f"Time: {current_time:.2f}s, Points: {buffer.GetWidth()}x{buffer.GetHeight()}")
        print("Sample distance:", buffer.GetPixel(640, 360))
    
    system.DoStepDynamics(time_step)
    current_time += time_step
```

This script:
1. Creates a fixed mesh body from an OBJ file
2. Sets up a kinematic body with a lidar sensor orbiting around the mesh
3. Configures lidar with noise, visualization, and data saving
4. Dynamically updates the lidar's position and orientation
5. Prints lidar buffer data at each time step

The lidar orbits in a circular path while maintaining orientation towards the origin. The sensor data is visualized in real-time and saved to PCD files. Make sure to replace "models/cube.obj" with your actual OBJ file path.
