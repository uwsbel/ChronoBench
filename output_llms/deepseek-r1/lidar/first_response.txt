
```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.sensor as sens

def main():
    # 1. Initialize PyChrono system
    chrono.SetChronoDataPath(os.path.join(os.environ.get('CHRONO_DATA_DIR', ''), 'data/'))
    system = chrono.ChSystemNSC()
    system.SetGravity(chrono.ChVectorD(0, 0, 0))  # No gravity for static scene

    # 2. Create fixed body with triangular mesh
    mesh_body = chrono.ChBody()
    mesh_body.SetBodyFixed(True)
    mesh_body.SetPos(chrono.ChVectorD(0, 0, 0))
    
    # Load OBJ mesh
    mesh = chrono.ChTriangleMeshConnected()
    mesh.LoadWavefrontMesh(chrono.GetChronoDataFile("models/cube.obj"))  # Default cube mesh
    
    # Add visualization asset
    mesh_shape = chrono.ChTriangleMeshShape()
    mesh_shape.SetMesh(mesh)
    mesh_body.AddAsset(mesh_shape)
    system.Add(mesh_body)

    # 3. Create sensor manager
    manager = sens.ChSensorManager(system)
    manager.scene.AddPointLight(chrono.ChVectorF(100, 100, 100), chrono.ChColor(1, 1, 1), 5000)

    # Lidar specifications
    lidar_offset_pose = chrono.ChFrameD(chrono.ChVectorD(0, 0, 0))  # Relative to body
    horizontal_samples = 90
    vertical_samples = 45
    horizontal_fov = 360  # degrees
    vertical_fov = 40     # degrees
    lag = 0
    exposure_time = 0
    lens_model = sens.DualDepthModel()
    max_distance = 100

    # Create lidar and add to manager
    lidar = sens.ChLidarSensor(
        mesh_body,            # Parent body
        update_rate=10,       # Hz
        lidar_offset_pose,
        horizontal_samples,
        vertical_samples,
        horizontal_fov,
        vertical_fov,
        max_distance,
        lens_model,
        lag,
        exposure_time
    )
    lidar.SetName("Orbiting Lidar")
    lidar.PushFilter(sens.ChFilterDIAccess())
    lidar.PushFilter(sens.ChFilterPCfromDepth())
    lidar.PushFilter(sens.ChFilterVisualize(horizontal_samples, vertical_samples, "Lidar Data"))
    lidar.PushFilter(sens.ChFilterSavePtCloud("lidar_data/lidar_%d.ply"))  # Saves to folder
    manager.AddSensor(lidar)

    # 4. Simulation parameters
    orbit_radius = 5.0
    orbit_height = 2.0
    orbit_speed = 0.1  # rad/s
    time_step = 0.01
    total_time = 5.0

    # 5. Simulation loop
    time = 0.0
    while time < total_time:
        # Update lidar position in circular orbit
        angle = orbit_speed * time
        x = orbit_radius * math.cos(angle)
        y = orbit_radius * math.sin(angle)
        lidar_offset_pose = chrono.ChFrameD(chrono.ChVectorD(x, y, orbit_height))
        lidar.SetOffsetPose(lidar_offset_pose)

        # Update sensors and system
        manager.Update()
        system.DoStepDynamics(time_step)
        time += time_step

        # Print lidar data
        buffer = lidar.GetMostRecentBuffer()
        if buffer.HasData():
            di_buffer = sens.DepthFrameAccess(buffer)
            print(f"Time: {time:.2f}s | "
                  f"Points: {di_buffer.width * di_buffer.height} | "
                  f"Min Dist: {np.min(di_buffer.depth):.3f}m | "
                  f"Max Dist: {np.max(di_buffer.depth):.3f}m")

if __name__ == "__main__":
    main()
```

### Key Features:
1. **Fixed Mesh Body**: Loads a cube.obj mesh as a fixed object
2. **Orbiting Lidar**:
   - Position updated dynamically in a circular orbit
   - Uses ChLidarSensor with realistic parameters
3. **Sensor Processing**:
   - Depth-to-point cloud conversion
   - Real-time visualization
   - Point cloud saving to PLY files
4. **Data Output**:
   - Prints point count and distance metrics each step
   - Saves accumulated data to `lidar_data/` folder

### Requirements:
- Place your OBJ file in `CHRONO_DATA_DIR/models/` (uses built-in cube.obj by default)
- PyChrono with sensor module built
- Creates `lidar_data/` directory for output files

Adjust `orbit_radius`, `orbit_height`, and `orbit_speed` to modify the lidar's path. Change the OBJ path to use custom meshes.
