

```python
import os
import math
import chrono.chrono as chrono
import chrono.chrono_irrlicht as chronoirr
import chrono.chrono_sensor as sens

def main():
    # Initialize Chrono system
    system = chrono.ChSystemNSC()
    system.Set_G_acc(chrono.ChVectorD(0, 0, 0))  # No gravity

    # Create fixed triangular mesh body from OBJ file
    mesh_body = chrono.ChBodyEasyMesh(
        os.path.join(chrono.GetChronoDataPath(), "models", "cube.obj"),  # Replace with your OBJ file
        1000,  # Density
        True,  # Collision
        True   # Visualization
    )
    mesh_body.SetFixed(True)
    system.Add(mesh_body)

    # Create sensor manager
    manager = sens.ChSensorManager(system)
    manager.scene.AddPointLight(chrono.ChVectorD(2, 2.5, 0), chrono.ChColor(1, 1, 1), 500)

    # Lidar configuration parameters
    orbit_radius = 5.0
    orbit_speed = 0.5  # Radians per second
    lidar_params = sens.LidarModelParams(
        horz_res=360,    # Horizontal resolution
        vert_res=180,    # Vertical resolution
        hfov=2*math.pi,  # Horizontal FOV (360 degrees)
        vfov=math.pi,    # Vertical FOV (180 degrees)
        max_distance=100,
        lag=0.0,
        angle_lag=0.0
    )
    noise_model = sens.ChNoiseNormal(chrono.ChVectorD(0, 0, 0), chrono.ChVectorD(0.05, 0.05, 0.05))

    # Create orbiting parent body for lidar
    parent_body = chrono.ChBody()
    parent_body.SetBodyFixed(False)
    parent_body.SetPos(chrono.ChVectorD(orbit_radius, 0, 0))
    system.Add(parent_body)

    # Create and configure lidar sensor
    lidar = sens.ChLidarSensor(
        parent_body,             # Parent body
        30,                      # Update rate (Hz)
        chrono.ChFrameD(),       # Offset pose
        lidar_params             # Sensor parameters
    )
    lidar.SetName("Orbiting Lidar")
    lidar.SetNoiseModel(noise_model)
    lidar.PushFilter(sens.ChFilterVisualize(640, 480, "Lidar Data"))
    lidar.PushFilter(sens.ChFilterPCfromDepth())
    lidar.PushFilter(sens.ChFilterXYZIAccess())
    lidar.PushFilter(sens.ChFilterSavePtCloud("lidar_data/"))
    manager.AddSensor(lidar)

    # Set up Irrlicht visualization
    vis = chronoirr.ChVisualSystemIrrlicht()
    vis.AttachSystem(system)
    vis.SetWindowSize(1024, 768)
    vis.SetWindowTitle('Lidar Orbit Simulation')
    vis.Initialize()
    vis.AddCamera(chrono.ChVectorD(0, 3, -5), chrono.ChVectorD(0, 0, 0))
    vis.AddTypicalLights()

    # Simulation loop
    time_step = 0.01
    time = 0.0
    while vis.Run():
        vis.BeginScene()
        vis.Render()
        vis.EndScene()

        # Update orbital position and orientation
        theta = orbit_speed * time
        new_pos = chrono.ChVectorD(
            orbit_radius * math.cos(theta),
            0,
            orbit_radius * math.sin(theta)
        )
        parent_body.SetPos(new_pos)
        parent_body.SetRot(chrono.Q_from_AngAxis(theta + math.pi/2, chrono.ChVectorD(0, 1, 0)))

        # Update sensor manager
        manager.Update()

        # Access and print lidar data
        buffer = lidar.GetMostRecentBuffer()
        if buffer and buffer.HasData():
            print(f"Time: {time:.2f} | Points captured: {buffer.GetXYZ().shape[0]}")
            if buffer.GetXYZ().shape[0] > 0:
                print(f"Sample point: {buffer.GetXYZ()[0]}")

        # Advance simulation
        system.DoStepDynamics(time_step)
        time += time_step

if __name__ == "__main__":
    main()
```

This script:
1. Creates a fixed mesh body from an OBJ file
2. Sets up a lidar sensor with noise modeling and data visualization
3. Orbits the lidar around the mesh while maintaining focus on it
4. Outputs captured point cloud data each timestep
5. Saves accumulated point cloud data to disk
6. Includes real-time 3D visualization

Replace the sample OBJ path with your actual mesh file. Adjust orbital parameters (radius/speed) and lidar specifications (resolution/FOV) as needed.
