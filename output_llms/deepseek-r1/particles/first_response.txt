import math
import random
import pychrono as chrono
import pychrono.irrlicht as chronoirr

def compute_gravitational_forces(particles, G):
    for i in range(len(particles)):
        body_i = particles[i]
        for j in range(i + 1, len(particles)):
            body_j = particles[j]
            pos_i = body_i.GetPos()
            pos_j = body_j.GetPos()
            delta = pos_j - pos_i
            r = delta.Length()
            if r < 1e-6:
                continue
            force_magnitude = G * body_i.GetMass() * body_j.GetMass() / (r**2)
            force_dir = delta / r
            force_i = force_dir * force_magnitude
            force_j = -force_dir * force_magnitude
            body_i.AddForce(force_i)
            body_j.AddForce(force_j)

class ParticleEmitter:
    def __init__(self, system):
        self.system = system
        self.time_since_last_emission = 0
        self.particles = []
    
    def emit_particle(self, dt):
        self.time_since_last_emission += dt
        emission_interval = 0.1
        if self.time_since_last_emission >= emission_interval:
            shape_type = random.choice(['sphere', 'box', 'cylinder'])
            density = 1000
            
            if shape_type == 'sphere':
                radius = random.uniform(0.2, 0.5)
                body = chrono.ChBodyEasySphere(radius, density, True, True)
            elif shape_type == 'box':
                sx, sy, sz = [random.uniform(0.2, 0.5) for _ in range(3)]
                body = chrono.ChBodyEasyBox(sx, sy, sz, density, True, True)
            elif shape_type == 'cylinder':
                radius = random.uniform(0.2, 0.5)
                height = random.uniform(0.4, 1.0)
                body = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, radius, height, density, True, True)
            
            pos = chrono.ChVectorD(*[random.uniform(-2, 2) for _ in range(3)])
            body.SetPos(pos)
            
            vel = chrono.ChVectorD(*[random.uniform(-0.5, 0.5) for _ in range(3)])
            body.SetPos_dt(vel)
            
            euler = chrono.ChVectorD(*[random.uniform(0, 2 * math.pi) for _ in range(3)])
            body.SetRot(chrono.Q_from_Euler123(euler))
            
            self.system.Add(body)
            self.particles.append(body)
            self.time_since_last_emission = 0

sys = chrono.ChSystemNSC()
sys.Set_G_acc(chrono.ChVectorD(0, 0, 0))

emitter = ParticleEmitter(sys)

app = chronoirr.ChIrrApp(sys, "Particle Gravity Simulation", 800, 600)
app.AddTypicalCamera(chronoirr.vector3df(0, 14, -20))
app.AddTypicalLights()
app.AssetBindAll()
app.AssetUpdateAll()

G = 1e-3
time_step = 0.01

while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()
    emitter.emit_particle(time_step)
    compute_gravitational_forces(emitter.particles, G)
    app.DoStep(time_step)
    app.EndScene()
