
Here's a complete PyChrono simulation that models gravitational attraction between particles with random shapes, positions, and velocities. The solution includes a custom gravity force calculation and particle emitter system:

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import random
import math

# Constants
G = 0.1        # Gravitational constant (tuned for visible attraction)
MAX_PARTICLES = 50
EMIT_INTERVAL = 0.1  # Seconds between particle emissions
EMIT_RADIUS = 2.0    # Emission area radius

# Custom gravity force calculator
class CustomGravity(chrono.Force):
    def __init__(self, particle_list):
        super().__init__()
        self.particles = particle_list
        self.G = G

    def Update(self, time, update_assets=True):
        n = len(self.particles)
        for i in range(n):
            body_i = self.particles[i]
            for j in range(i + 1, n):
                body_j = self.particles[j]
                
                # Calculate distance between particles
                r_vec = body_j.GetPos() - body_i.GetPos()
                r = r_vec.Length()
                
                # Skip if particles are too close
                if r < 1e-3:
                    continue
                
                # Calculate gravitational force (F = G * m1 * m2 / r^2)
                force_mag = self.G * body_i.GetMass() * body_j.GetMass() / (r * r)
                force_dir = r_vec / r
                force = force_dir * force_mag
                
                # Apply equal and opposite forces
                body_i.AddForce(force)
                body_j.AddForce(-force)

# Create physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))  # Disable default gravity

# Particle container
particles = []
gravity_force = CustomGravity(particles)
system.Add(gravity_force)

# Particle emitter function
def create_particle():
    # Random shape selection
    shape_type = random.choice(['sphere', 'box', 'cylinder'])
    density = 1000  # kg/m³
    
    # Create random-sized body
    if shape_type == 'sphere':
        radius = random.uniform(0.1, 0.3)
        body = chrono.ChBodyEasySphere(radius, density, True, False)
    elif shape_type == 'box':
        size = chrono.ChVectorD(*[random.uniform(0.1, 0.3) for _ in range(3)])
        body = chrono.ChBodyEasyBox(size.x, size.y, size.z, density, True, False)
    else:  # cylinder
        radius = random.uniform(0.1, 0.2)
        height = random.uniform(0.2, 0.4)
        body = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, radius, height, density, True, False)
    
    # Random position in emission disk
    angle = random.uniform(0, 2 * math.pi)
    distance = random.uniform(0, EMIT_RADIUS)
    pos = chrono.ChVectorD(
        distance * math.cos(angle),
        random.uniform(3, 5),
        distance * math.sin(angle)
    )
    
    # Random velocity and rotation
    vel = chrono.ChVectorD(
        random.uniform(-0.5, 0.5),
        random.uniform(-1, 0),
        random.uniform(-0.5, 0.5)
    )
    ang_vel = chrono.ChVectorD(
        random.uniform(-2, 2),
        random.uniform(-2, 2),
        random.uniform(-2, 2)
    )
    
    # Set body properties
    body.SetPos(pos)
    body.SetPos_dt(vel)
    body.SetWvel_par(ang_vel)
    body.GetVisualShape(0).SetColor(chrono.ChColor(random.random(), random.random(), random.random()))
    
    # Add to system
    system.Add(body)
    particles.append(body)
    return body

# Create Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Particle Gravity Simulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 5, -10))
vis.AddTypicalLights()

# Create text GUI for particle count
particle_text = vis.GetGUIEnvironment().addStaticText(
    "Particles: 0", 
    chronoirr.recti(10, 10, 400, 40), 
    False
)

# Simulation parameters
time = 0.0
next_emit_time = 0.0
sim_time = 0

# Main simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    
    # Update particle counter
    particle_text.setText(f"Particles: {len(particles)} Time: {sim_time:.1f}s")
    
    # Emit new particles
    if time >= next_emit_time and len(particles) < MAX_PARTICLES:
        create_particle()
        next_emit_time = time + EMIT_INTERVAL
    
    # Advance simulation
    vis.EndScene()
    system.DoStepDynamics(1e-3)
    time += 1e-3
    sim_time += 1
```

Key features of this implementation:
1. **Custom Gravity System**: Implements n-body gravitational attraction using Newton's law
2. **Random Particle Emitter**:
   - Generates spheres, boxes, and cylinders with random sizes
   - Applies random positions within an emission disk
   - Sets random linear/angular velocities
   - Assigns random colors for visualization
3. **Visualization**:
   - Real-time particle counter
   - Camera positioned for optimal viewing
   - Color-coded particles
4. **Performance Controls**:
   - Particle emission rate limit
   - Maximum particle cap
   - Collision detection disabled for efficiency

To adjust the simulation:
- Modify `G` constant for stronger/weaker gravity
- Change `MAX_PARTICLES` for more/fewer particles
- Adjust `EMIT_INTERVAL` for faster/slower emission
- Tweak `EMIT_RADIUS` to change emission area size

The particles will initially spread out from the emission zone, then gradually attract each other due to gravitational forces, forming clusters over time.
