```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import math

# Initialize the Chrono simulation system
sys = chrono.ChSystemNSC()

# Set gravitational acceleration for moon simulation
sys.SetGravitationalAcceleration(chrono.ChVector3d(0, -1.62, 0))

# Create the ground body
ground = chrono.ChBody()
sys.Add(ground)
ground.SetFixed(True)
ground.EnableCollision(False)

# Add ground visualization cylinder
cyl_ground = chrono.ChVisualShapeCylinder(0.2, 0.4)
ground.AddVisualShape(cyl_ground, chrono.ChFrameD(chrono.ChVector3d(0, 0, 1)))

# Create pendulum body with updated properties
pend_1 = chrono.ChBody()
sys.AddBody(pend_1)
pend_1.SetFixed(False)
pend_1.EnableCollision(False)
pend_1.SetMass(2)  # Updated mass
pend_1.SetInertiaXX(chrono.ChVector3d(0.4, 1.5, 1.5))  # Updated inertia

# Add pendulum visualization cylinder with new dimensions
cyl_pend = chrono.ChVisualShapeCylinder(0.1, 1.5)  # Radius 0.1, height 1.5
cyl_pend.SetColor(chrono.ChColor(0.6, 0, 0))
pend_1.AddVisualShape(cyl_pend, chrono.ChFrameD(chrono.ChVector3d(-0.75, 0, 0), 
                chrono.Q_from_AngY(chrono.CH_PI_2)))

# Set initial position (adjusted for new length)
pend_1.SetPos(chrono.ChVector3d(1.5, 0, 1))

# Set initial angular velocity (2 rad/s around Z-axis)
pend_1.SetAngVelParent(chrono.ChVector3d(0, 0, 2))

# Create spherical joint instead of revolute
sph_joint = chrono.ChLinkLockSpherical()
sph_joint.Initialize(ground, pend_1, True, chrono.ChVector3d(0, 0, 1))
sys.AddLink(sph_joint)

# Add joint visualization sphere
sphere_joint = chrono.ChVisualShapeSphere(0.2)  # Radius 0.2 (not 2 to avoid occlusion)
sphere_joint.SetColor(chrono.ChColor(0, 1, 0))
ground.AddVisualShape(sphere_joint, chrono.ChFrameD(chrono.ChVector3d(0, 0, 1)))

# Visualization setup
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Modified Pendulum Simulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 6))
vis.AddTypicalLights()

# Simulation loop
log_info = True
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(1e-3)

    if log_info and sys.GetChTime() > 1:
        pos_1 = pend_1.GetPos()
        print("t = ", sys.GetChTime())
        print("Position:", pos_1.x, pos_1.y, pos_1.z)
        lin_vel_1 = pend_1.GetPosDt()
        print("Velocity:", lin_vel_1.x, lin_vel_1.y, lin_vel_1.z)
        log_info = False
```
