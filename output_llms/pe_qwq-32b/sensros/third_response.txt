    sys.Add(ground_body)

    sens_manager = sens.ChSensorManager(sys)

    cam.SetName("camera")
    sens_manager.AddSensor(cam)

    # Create and configure a lidar sensor.
    lidar = sens.ChLidarSensor(ground_body, 5., offset_pose, 90, 300, 2*ch.CH_PI, ch.CH_PI/12, -ch.CH_PI/6, 100., 0)
