import pychrono.core as chrono
import pychrono.fea as fea
import pychrono.irrlicht as chronoirr

        msection_cable2 = fea.ChBeamSectionCable()
        msection_cable2.SetDiameter(0.015)
        msection_cable2.SetYoungModulus(0.01e9)
        msection_cable2.SetRayleighDamping(0.0001)  # Modified Rayleigh damping value

        builder = fea.ChBuilderCableANCF()
        builder.BuildBeam(
            mesh,
            msection_cable2,
            10,
            chrono.ChVectorD(0, 0, -0.1),
            chrono.ChVectorD(0.5, 0, -0.1)
        )

        # Apply force to front node (index 0)
        front_node = builder.GetLastBeamNodes()[0]
        front_node.SetForce(chrono.ChVectorD(0, -0.7, 0))  # Modified force value

        mtruss = chrono.ChBody()
        mtruss.SetFixed(True)

        # Use list indices instead of .front()/.back()
        last_node = builder.GetLastBeamNodes()[-1]
        constraint_hinge = fea.ChLinkNodeFrame()
        constraint_hinge.Initialize(last_node, mtruss)
        system.Add(constraint_hinge)

sys = chrono.ChSystemSMC()

model = Model1(sys, mesh)
sys.Add(mesh)

visualizebeamA = chrono.ChVisualShapeFEA(mesh)

mesh.AddVisualShapeFEA(visualizebeamB)
