```python
import pychrono
import pychrono.core as pc
import pychrono.visuals as v
import pychrono.simulation as sc
import numpy as np
import matplotlib.pyplot as plt  # For visualization (optional, but recommended)

# --- Simulation Parameters ---
terrain_width = 10.0
terrain_height = 5.0
terrain_depth = 2.0
terrain_x_start = 0.0
terrain_x_end = 10.0
terrain_y_start = 0.0
terrain_y_end = 5.0
terrain_z_start = 0.0
terrain_z_end = 2.0
vehicle_x_start = 0.0
vehicle_x_end = 10.0
vehicle_y_start = 0.0
vehicle_y_end = 5.0
vehicle_z_start = 0.0
vehicle_z_end = 2.0
vehicle_speed = 0.5  # m/s
vehicle_acceleration = 0.1 # m/s^2
vehicle_brake = 0.0  # m/s^2
vehicle_steering_rate = 0.1  # degrees/second
vehicle_throttle_rate = 0.0  # percentage/second
vehicle_braking_rate = 0.0  # m/s^2


# --- Initial Setup ---
try:
    # Initialize PyChrono
    chrono.init()

    # Create the terrain
    terrain = pc.RigidTerrain(terrain_width, terrain_height, terrain_depth, terrain_x_start, terrain_x_end, terrain_y_start, terrain_y_end, terrain_z_start, terrain_z_end)

    # Create the vehicle
    vehicle = pc.Vehicle(vehicle_x_start, vehicle_y_start, vehicle_z_start, vehicle_z_end, vehicle_speed, vehicle_acceleration, vehicle_brake, vehicle_steering_rate, vehicle_throttle_rate, vehicle_braking_rate)

    # --- Visualization ---
    v.create_visuals(terrain, vehicle,  # Pass the terrain and vehicle to visualize
                   title="ARTcar Simulation",
                   width=500,
                   height=500)  # Adjust width/height as needed


    # --- Simulation Loop ---
    print("Simulation started...")
    simulation_loop = sc.Simulation(terrain, vehicle)
    simulation_loop.start()

    # --- Simulation Loop (Update - this is where the core simulation logic goes) ---
    while simulation_loop.is_running():
        # Update vehicle dynamics
        vehicle.update_dynamics()

        # Update visualization
        v.update_visuals(terrain, vehicle)

        # Print some information (for debugging/monitoring)
        print(f"Time: {simulation_loop.time}")
        print(f"Speed: {vehicle.speed}")
        print(f"Acceleration: {vehicle.acceleration}")
        print(f"Braking: {vehicle.brake}")
        print(f"Steering: {vehicle.steering_rate}")
        print(f"Throttle: {vehicle.throttle_rate}")
        print(f"Current Speed: {vehicle.speed}")

        # Optional:  Add a delay to control the simulation speed.  Adjust as needed.
        #time.sleep(0.01)  # Adjust for desired simulation speed

    # --- Simulation Termination ---
    simulation_loop.stop()
    print("Simulation finished.")

except Exception as e:
    print(f"An error occurred: {e}")
    print("Please ensure PyChrono is correctly installed and the terrain is valid.")
```
