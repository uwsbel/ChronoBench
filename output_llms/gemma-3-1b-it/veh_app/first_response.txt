import pychrono.core as chrono
import pychrono.visuals as visuals

import pychrono.drivers as drivers
import numpy as np

# --- 1. Initialization & Core Components ---

# Define Simulation Parameters
simulation_name = "GatorSimulation"
simulation_time = 10  # Simulation duration in seconds
initial_position = (0, 0, 0)  # Initial vehicle position (x, y, z)
initial_velocity = 0.5  # Initial velocity (m/s)
terrain_heightmap = np.zeros((100, 100))  # Example terrain heightmap - replace with your actual data
sensor_manager = chrono.SensorManager(
    initial_position=initial_position,
    initial_velocity=initial_velocity,
    terrain_heightmap=terrain_heightmap,
    camera_position=initial_position,
    camera_orientation=np.array([0, 0, 0]),  # Default orientation
    sensor_type="camera"
)

# --- 2. Physical Systems & Objects ---

# Gator Vehicle
gator = drivers.Gator(
    initial_position=initial_position,
    initial_velocity=initial_velocity,
    mass=100,  # kg - Adjust as needed
    acceleration=0.1, # m/s^2 - Adjust as needed
    friction=0.05, # m/s^2 - Adjust as needed
    gravity=0.0, # m/s^2 - Adjust as needed
    sensor_type="vehicle"
)

# Terrain
terrain = drivers.Terrain(
    initial_position=initial_position,
    heightmap=terrain_heightmap,
    friction=0.05,
    gravity=0.0,
    sensor_type="terrain"
)

# --- 3. Sensor Manager Setup ---

# Sensor Manager
sensor_manager.add_sensor(
    sensor_type="point_light",
    sensor_name="point_light",
    position=sensor_manager.camera_position,
    orientation=sensor_manager.camera_orientation,
    sensor_type="camera"
)
sensor_manager.add_sensor(
    sensor_type="camera",
    sensor_name="camera",
    position=sensor_manager.camera_position,
    orientation=sensor_manager.camera_orientation,
    sensor_type="camera"
)


# --- 4. Simulation Loop ---

# Simulation Loop
for t in range(simulation_time):
    # Update Driver
    gator.update_driver(t)

    # Update Terrain
    terrain.update_terrain(t)

    # Update Vehicle
    gator.update_vehicle(t)

    # Update Sensor Manager
    sensor_manager.update_sensor(t)

    # Print Status (for debugging) -  Remove in production
    print(f"Time: {t}, Position: {gator.position}, Velocity: {gator.velocity}")

    # Advance Simulation
    chrono.advance(t)
