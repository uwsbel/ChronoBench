**1. Initialization & Core Components**

```python
import pychrono
import pychrono.core as pc
import pychrono.visuals as v
import pychrono.physics as pfx
import pychrono.simulation as ps
import numpy as np
import matplotlib.pyplot as plt  # For visualization (optional, but highly recommended)

# --- Configuration ---
# Vehicle Parameters
vehicle_id = "E90_Sedan_01"  # Unique ID for the vehicle
vehicle_model = "BMW E90"
vehicle_speed = 25.0  # m/s (Adjust for speed)
vehicle_acceleration = 1.0  # m/s^2 (Adjust for acceleration)
vehicle_brake_force = 0.5  # N (Adjust for braking)

# Terrain Parameters
terrain_height = 1.0  # meters
terrain_width = 10.0 # meters
terrain_slope = 0.0  # Slope angle (degrees) -  This is a crucial parameter.
terrain_texture = "grass_terrain.png" # Path to a texture file (optional)
terrain_color = "green" # Color of the terrain

# Driver System Parameters
driver_control_type = "interactive"  # "interactive" or "passive"
driver_control_speed = 20.0  # m/s (Adjust for speed)
driver_brake_pressure = 0.0  # N (Adjust for braking)


# --- Initial Setup ---
# Create the simulation environment
simulation = pc.Simulation()
simulation.create_environment(vehicle_id, terrain_height, terrain_width, terrain_slope, terrain_texture, terrain_color)

# --- Physics Setup ---
# Define the vehicle's dynamics
pfx.VehicleDynamics(vehicle_id, vehicle_speed, vehicle_acceleration, vehicle_brake_force)

# --- Visualization ---
# Initialize Irrlicht
plt = plt.figure()
plt.title("BMW E90 Sedan Simulation")
plt.axis('equal') # Ensure the plot is a square

# Create a chase camera
camera = v.Camera(x=0, y=0, z=0,  # Camera position
                   width=5,  # Camera width
                   height=5,  # Camera height
                   depth=5,  # Camera depth
                   zoom=1.0)

# Create a directional light
light = v.Light(x=0, y=0, z=0, color=pc.Color("white"))

# Create a skybox
skybox = v.Skybox(x=0, y=0, z=0, color=pc.Color("white"))

# Create a texture
texture = v.Texture(x=0, y=0, z=0, color=pc.Color("grass_terrain.png"))

# Create a custom logo (replace with your logo path)
logo = v.Logo(x=0, y=0, z=0, color=pc.Color("red"))

# Set up the visualization
v.SetVisuals(camera, light, skybox, texture, logo,  
              vehicle_id,  # Use the vehicle ID for the camera
              vehicle_model,
              vehicle_speed,
              vehicle_acceleration,
              vehicle_brake_force)

# --- Simulation Setup ---
# Set simulation parameters
simulation.set_simulation_parameters(
    vehicle_id,
    vehicle_speed,
    vehicle_acceleration,
    vehicle_brake_force,
    terrain_height,
    terrain_width,
    terrain_slope,
    terrain_color,
    driver_control_type,
    driver_control_speed,
    driver_brake_pressure
)

# --- Simulation Loop ---
simulation.run()
```
