```python
import pychrono
import pychrono.core as pc
import pychrono.visual as v
import numpy as np
import matplotlib.pyplot as plt
from pychrono.visual import Visual

def simulate_mass_spring_damper(dt=0.01):
    """
    Simulates a single mass-spring-damper system using PyChrono.

    Args:
        dt (float): Time step for the simulation.
    """

    # 1. Initialize PyChrono environment
    chrono = pc.Chchrono()
    chrono.set_environment(pc.Environment(
        time_step=dt,
        gravity=0.0,  # Assuming a standard gravitational acceleration
        mass=1.0,
        spring_constant=0.5,
        damping_coefficient=0.1
    ))

    # 2. Define physical systems and objects
    mass = pc.Mass(0.1, 1.0)  # Mass of 1.0 kg
    spring = pc.Spring(0.2, 0.5)  # Spring constant 0.2 N/m, spring length 0.5 m
    ground = pc.Ground(0.0, 1.0)  # Ground surface

    # 3. Set default parameters
    chrono.set_default_parameters(
        mass=mass,
        spring_constant=spring.constant,
        damping_coefficient=damping_coefficient
    )

    # 4. Create a visual object
    visual = v.Visual(
        title="Mass-Spring-Damper Simulation",
        camera_settings={"width": 640, "height": 480},
        # Add lighting for better visualization
        lighting={"color": "white"}
    )

    # 5. Create a link between the mass and the ground
    chrono.create_link(mass, ground)

    # 6. Run the simulation
    chrono.run(visual)

    # 7.  Optional: Save the simulation
    # chrono.save_simulation(visual)

if __name__ == '__main__':
    simulate_mass_spring_damper()
```
