**2.  Vehicle Setup & Parameters**

```python
# --- Vehicle Parameters ---
vehicle_type = "Gator"
vehicle_speed = 0.5
vehicle_acceleration = 0.1
vehicle_deceleration = 0.1
vehicle_max_speed = 2.0

# --- Terrain Setup ---
terrain = physics.Terrain(width=terrain_width, height=terrain_height, depth=terrain_depth)
terrain.set_texture(terrain_texture)
terrain.set_friction_coefficient(0.8) # Adjust for realism.

# --- Rigid Terrain ---
rigid_terrain = visual.RigidTerrain(width=terrain_width, height=terrain_height, depth=terrain_depth,
                                 texture=terrain_texture,
                                 friction_coefficient=0.8) # Adjust friction
rigid_terrain.set_vehicle_position(vehicle_position)
rigid_terrain.set_vehicle_orientation(vehicle_orientation)
rigid_terrain.set_vehicle_speed(vehicle_speed)
rigid_terrain.set_vehicle_acceleration(vehicle_acceleration)
rigid_terrain.set_vehicle_deceleration(vehicle_deceleration)
rigid_terrain.set_vehicle_max_speed(vehicle_max_speed)


# --- Contact Implementation ---
contact_method = contact_method
# --- End Contact Implementation ---

# ---  Simulation Parameters ---
simulation_time = 50.0
simulation_steps = 50

# ---  Initial State ---
vehicle_state = {
    "position": vehicle_position,
    "orientation": vehicle_orientation,
    "velocity": np.array(vehicle_speed),
    "acceleration": np.array(vehicle_acceleration),
    "contact_status": False # Initially, no contact
}

# ---  Start Simulation ---
print("Simulation started.  Press Ctrl+C to exit.")
print(f"Simulation time: {simulation_time:.2f} seconds")
print(f"Simulation steps: {simulation_steps}")

# ---  Main Simulation Loop ---
try:
    while True:
        # ---  Update Dynamics ---
        vehicle_state["velocity"] += vehicle_acceleration * simulation_time
        vehicle_state["acceleration"] += vehicle_deceleration * simulation_time
        vehicle_state["position"] += vehicle_state["velocity"] * simulation_time

        # ---  Check for Contact ---
        if contact_method == "contact_method_rigid":
            contact_status = True
        elif contact_method == "contact_method_soft":
            contact_status = True
        elif contact_method == "contact_method_semi_rigid":
            contact_status = False # Simulate semi-rigid contact
        else:
            print("Invalid contact method.")
            contact_status = False

        # ---  Rendering ---
        visual.render(rigid_terrain, vehicle_state,  0.5) # Adjust rendering parameters

        # ---  Update Visualization ---
        visual.update(rigid_terrain)

        # ---  Check for End Condition ---
        if contact_status:
            print("Simulation ended due to contact.")
            break

        # ---  Print Status ---
        print(f"Time: {utils.time_since_start:.2f}s")
        print(f"Position: {vehicle_state['position']:.2f}")
        print(f"Orientation: {vehicle_state['orientation']:.2f}")
        print(f"Velocity: {vehicle_state['velocity']:.2f}")
        print(f"Acceleration: {vehicle_state['acceleration']:.2f}")

        # ---  Wait for the next step ---
        utils.sleep(simulation_steps)

except KeyboardInterrupt:
    print("Simulation interrupted.")
```
