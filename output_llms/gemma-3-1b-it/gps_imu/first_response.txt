import pychrono
import pychrono.core as chrono
import pychrono.visual as visual
import numpy as np
import matplotlib.pyplot as plt  # For visualization (optional, but recommended)

# --- Configuration ---
simulation_name = "HMMWV_Simulation"
simulation_time = 60  # Simulation duration in seconds
initial_position = [0, 0, 0]  # Initial vehicle position (x, y, z)
initial_velocity = [0, 0, 0] # Initial velocity (x, y, z)
initial_acceleration = [0, 0, 0] # Initial acceleration (x, y, z)
initial_sensor_data = {
    'imu': {'x': 0, 'y': 0, 'z': 0},
    'gps': {'x': 0, 'y': 0, 'z': 0}
}
sensor_data_update_interval = 1  # Update IMU and GPS every X seconds
sensor_update_interval = 0.5 # Update sensor data every 0.5 seconds


# --- HMMWV Vehicle Model ---
vehicle_model = chrono.Vehicle(
    name=simulation_name,
    mass=1000,  # kg
    position=initial_position,
    velocity=initial_velocity,
    acceleration=initial_acceleration,
    sensor_data=initial_sensor_data
)

# --- Terrain ---
terrain_model = chrono.Terrain(
    name=simulation_name,
    heightmap=np.zeros((100, 100), dtype=np.float32))  # Simple heightmap for demonstration
terrain_model.set_heightmap(terrain_model.heightmap)

# --- Driver Model ---
driver_model = chrono.Driver(
    name=simulation_name,
    position=initial_position,
    velocity=initial_velocity,
    acceleration=initial_acceleration
)

# --- Sensor Management ---
sensor_manager = chrono.SensorManager(
    name=simulation_name,
    sensors=[
        chrono.IMU(name='imu', x=0, y=0, z=0),
        chrono.GPS(name='gps', x=0, y=0, z=0)
    ]
)

# --- Visualization ---
visual.plot_vehicle(vehicle_model, terrain_model, driver_model, sensor_manager,
                title=simulation_name,
                x_range=(0, 100),  # Range for x-axis
                y_range=(0, 100),
                z_range=(0, 100),
                color='blue')


# --- Simulation Loop ---
def run_simulation():
    while True:
        # 1. Update Sensor Data
        imu_data = sensor_manager.get_sensor_data('imu')
        gps_data = sensor_manager.get_sensor_data('gps')

        # Simulate sensor updates (replace with actual sensor readings)
        imu_data['x'] += 0.1
        imu_data['y'] += 0.2
        imu_data['z'] += 0.05
        gps_data['x'] += 0.05
        gps_data['y'] += 0.05
        gps_data['z'] += 0.05

        # 2. Update Vehicle State
        vehicle_model.update_state(imu_data, gps_data)

        # 3. Update Terrain (Simple - just a shift)
        terrain_model.set_heightmap(terrain_model.heightmap)

        # 4. Update Driver State
        driver_model.update_state(vehicle_model)

        # 5. Render the Scene
        visual.plot_vehicle(vehicle_model, terrain_model, driver_model, sensor_manager,
                         title=simulation_name,
                         x_range=(0, 100),
                         y_range=(0, 100),
                         z_range=(0, 100),
                         color='blue')

        # 6. Advance Simulation Time
        simulation_time -= 1
        if simulation_time <= 0:
            break

        # 7.  Display the current state

if __name__ == "__main__":
