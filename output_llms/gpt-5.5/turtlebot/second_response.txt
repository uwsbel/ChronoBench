```python
import math
import pychrono as chrono
import pychrono.robot as turtlebot
import pychrono.irrlicht as chronoirr

# -----------------------------------------------------------------------------
# Create Chrono system
# -----------------------------------------------------------------------------
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))

chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# -----------------------------------------------------------------------------
# Create ground body as terrain plane that robot will drive on
# -----------------------------------------------------------------------------
ground_mat = chrono.ChContactMaterialNSC()

ground = chrono.ChBodyEasyBox(
    20, 20, 1,      # size
    1000,           # density
    True,           # visualization
    True,           # collision
    ground_mat
)

# Adjusted ground position
ground.SetPos(chrono.ChVector3d(0, 0, -0.6))
ground.SetFixed(True)

ground.GetVisualShape(0).SetTexture(
    chrono.GetChronoDataFile("textures/concrete.jpg")
)

system.Add(ground)

# -----------------------------------------------------------------------------
# Create TurtleBot robot
# -----------------------------------------------------------------------------
init_pos = chrono.ChVector3d(0, 0.2, 0)
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)

robot = turtlebot.TurtleBot(system, init_pos, init_rot)
robot.Initialize()

# Wheel IDs
LEFT_DRIVE_WHEEL = 0
RIGHT_DRIVE_WHEEL = 1

# -----------------------------------------------------------------------------
# Robot movement controller
# -----------------------------------------------------------------------------
def move(mode):
    """
    Control the TurtleBot motion.

    Modes:
        straight : both wheels move forward
        left     : right wheel moves, left wheel stops
        right    : left wheel moves, right wheel stops
    """
    forward_speed = -math.pi

    if mode == "straight":
        robot.SetMotorSpeed(forward_speed, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(forward_speed, RIGHT_DRIVE_WHEEL)

    elif mode == "left":
        robot.SetMotorSpeed(0, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(forward_speed, RIGHT_DRIVE_WHEEL)

    elif mode == "right":
        robot.SetMotorSpeed(forward_speed, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0, RIGHT_DRIVE_WHEEL)

    else:
        raise ValueError(
            "Invalid movement mode. Expected 'straight', 'left', or 'right'."
        )

# -----------------------------------------------------------------------------
# Create run-time visualization
# -----------------------------------------------------------------------------
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle("TurtleBot Robot - Rigid terrain")
vis.Initialize()

vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
vis.AddSkyBox()
vis.AddCamera(
    chrono.ChVector3d(0, 1.5, 0.2),
    chrono.ChVector3d(0, 0, 0.2)
)
vis.AddTypicalLights()
vis.AddLightWithShadow(
    chrono.ChVector3d(1.5, -2.5, 5.5),
    chrono.ChVector3d(0, 0, 0.5),
    3,
    4,
    10,
    40,
    512
)

# -----------------------------------------------------------------------------
# Simulation settings
# -----------------------------------------------------------------------------
time_step = 2e-3
time = 0.0
current_mode = None

# -----------------------------------------------------------------------------
# Simulation loop
# -----------------------------------------------------------------------------
while vis.Run():

    # First 5 seconds: move straight
    if time < 5.0:
        if current_mode != "straight":
            print("Robot moving straight")
            current_mode = "straight"
        move("straight")

    # Next 5 seconds: turn left
    elif time < 10.0:
        if current_mode != "left":
            print("Robot turning left")
            current_mode = "left"
        move("left")

    # Thereafter: turn right
    else:
        if current_mode != "right":
            print("Robot turning right")
            current_mode = "right"
        move("right")

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance the simulation
    system.DoStepDynamics(time_step)

    # Increment simulation time
    time += time_step
```
