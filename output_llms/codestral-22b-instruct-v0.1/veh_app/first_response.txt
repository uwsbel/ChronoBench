```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Initialize the PyChrono environment
chrono.SetChronoDataPath('/path/to/chrono/data/')

# Create a system
system = chrono.ChSystemNSC()

# Create a vehicle
vehicle = chrono.vehicle.Gator()
vehicle.Initialize(system)

# Set visualization type for vehicle parts
vehicle.SetChassisVisualizationType(chrono.vehicle.VisualizationType_MESH)
vehicle.SetSuspensionVisualizationType(chrono.vehicle.VisualizationType_PRIMITIVES)
vehicle.SetSteeringVisualizationType(chrono.vehicle.VisualizationType_PRIMITIVES)
vehicle.SetWheelVisualizationType(chrono.vehicle.VisualizationType_MESH)

# Create a rigid terrain
terrain = chrono.vehicle.RigidTerrain(vehicle.GetSystem())
terrain.Initialize(system)

# Create an interactive driver system
driver = chrono.vehicle.ChDriver(vehicle)
driver.Initialize()

# Create a sensor manager
sensor_manager = chrono.sensor.ChSensorManager(system)

# Add point lights to the sensor manager
point_light = chrono.sensor.ChSensorPointLight(system)
point_light.SetColor(chrono.ChColor(1, 1, 1))
point_light.SetIntensity(1000)
sensor_manager.AddSensor(point_light)

# Add a camera to the sensor manager and attach it to the vehicle's chassis
camera = chrono.sensor.ChSensorCamera(system)
camera.SetPosition(chrono.ChVectorD(0, 1, 0))
camera.SetLookAt(chrono.ChVectorD(0, 0, 0))
vehicle.GetChassisBody().AddSensor(camera)

# Create an Irrlicht application to visualize the system
app = chronoirr.ChIrrApp(system, 'Gator Simulation', chronoirr.dimension2du(1024, 768))
app.AddTypicalLogo()
app.AddTypicalSky()
app.AddTypicalCamera(chronoirr.vector3df(0, 3, -6))
app.AddTypicalLights()
app.AddLightWithShadow(chronoirr.vector3df(3, 6, -2.5), chronoirr.vector3df(0, 0, 0), 3, 2.2, 7.2, 40, 512, chronoirr.SColorf(0.8, 0.8, 0.9))
app.SetTimestep(0.01)

# Run the simulation loop
while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()
    app.DoStep()
    app.EndScene()
    system.DoStepDynamics(0.01)
    driver.Synchronize(system.GetChTime())
    terrain.Synchronize(system.GetChTime())
    vehicle.Synchronize(system.GetChTime(), driver.GetInputs(), terrain)
    sensor_manager.Update()
```

This script initializes a PyChrono simulation featuring a Gator vehicle, a rigid terrain, an interactive driver system, and a sensor manager with a point light and a camera attached to the vehicle's chassis. The script also sets up visualization for various vehicle parts and runs a simulation loop, updating the driver, terrain, vehicle, and sensor manager at each timestep.
