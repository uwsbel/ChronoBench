```python
# Import necessary PyChrono modules
from pychrono import core as chrono
from pychrono import irrlicht as chronoirr
from pychrono.fea import ChElementBeamIGA
from pychrono.fea import ChMesh
from pychrono.fea import ChMeshFileLoader
from pychrono.fea import ChVisualizationFEAmesh

# Create a Chrono system
system = chrono.ChSystemNSC()

# Create a mesh for the beam
mesh = ChMesh()
loader = ChMeshFileLoader()
loader.LoadMeshFromFile(mesh, 'beam_mesh.obj')

# Create an IGA beam
beam = ChElementBeamIGA()
beam.SetMesh(mesh)

# Set beam properties
beam.SetYoungModulus(73e9)
beam.SetGwithPoissonRatio(0.3)
beam.SetDensity(2700)
beam.SetBeamRaleyghDamping(0.005)

# Add the beam to the system
system.Add(beam)

# Create a flywheel
flywheel = chrono.ChBody()
flywheel.SetMass(10)
flywheel.SetInertiaXX(chrono.ChVectorD(1, 1, 1))

# Set flywheel position
flywheel.SetPos(chrono.ChVectorD(0, 0, 0))

# Add the flywheel to the system
system.Add(flywheel)

# Create a rotational motor
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(beam, flywheel, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngX(0)))

# Set motor speed
motor.SetSpeedFunction(chrono.ChFunction_Const(5))

# Add the motor to the system
system.Add(motor)

# Create a visualization object for the FEM mesh
visualization = ChVisualizationFEAmesh()
visualization.SetFEMdataType(beam.GetVisualizationType())
visualization.SetMesh(mesh)
visualization.SetWireframe(True)
visualization.SetSmoothFaces(True)
visualization.SetShowElementLabels(True)

# Create an Irrlicht application
application = chronoirr.ChIrrApp(system, 'Jeffcott Rotor Simulation', chronoirr.dimension2du(1024, 768))
application.AddTypicalLogo()
application.AddTypicalCamera(chronoirr.vector3df(0, 3, -6))
application.AddTypicalLights()
application.AddSkyBox()
application.AddLightWithShadow(chronoirr.vector3df(3, 6, -6), chronoirr.vector3df(0, 0, 0), 12, 2, 10, 40, 512, chronoirr.SColorf(0.8, 0.8, 0.8))

# Add the visualization to the application
application.AddAsset(visualization)

# Run the simulation
application.SetTimestep(0.01)
application.SetTryRealtime(True)
application.Run()
```
