```python
import pychrono as chrono
import pychrono.irrlicht as irr

# Initialize the PyChrono environment
chrono.SetChronoDataPath("PATH_TO_CHRONO_DATA_DIRECTORY")

# Create a physical system
my_system = chrono.ChSystemNSC()

# Create a floor (truss)
floor = chrono.ChBodyEasyBox(10, 1, 1, 1000, False, True)
floor.SetPos(chrono.ChVectorD(0, -1, 0))
floor.SetBodyFixed(True)
my_system.Add(floor)

# Create a crankshaft
crankshaft = chrono.ChBodyEasyCylinder(0.5, 2, 1000, False, True)
crankshaft.SetPos(chrono.ChVectorD(0, 0, 0))
my_system.Add(crankshaft)

# Create a connecting rod
connecting_rod = chrono.ChBodyEasyCylinder(0.2, 5, 1000, False, True)
connecting_rod.SetPos(chrono.ChVectorD(2, 0, 0))
my_system.Add(connecting_rod)

# Create a piston
piston = chrono.ChBodyEasyCylinder(0.5, 1, 1000, False, True)
piston.SetPos(chrono.ChVectorD(5, 0, 0))
my_system.Add(piston)

# Connect the crankshaft and connecting rod using a revolute joint
revolute_joint_crank_rod = chrono.ChLinkLockRevolute()
revolute_joint_crank_rod.Initialize(crankshaft, connecting_rod, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 0, 1))))
my_system.Add(revolute_joint_crank_rod)

# Connect the connecting rod and piston using a revolute joint
revolute_joint_rod_piston = chrono.ChLinkLockRevolute()
revolute_joint_rod_piston.Initialize(connecting_rod, piston, chrono.ChCoordsysD(chrono.ChVectorD(5, 0, 0), chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 0, 1))))
my_system.Add(revolute_joint_rod_piston)

# Drive the crankshaft using a motor
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(crankshaft, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 0, 1))))
motor.SetSpeed(5)  # Set the angular speed of the crankshaft
my_system.Add(motor)

# Set the default parameters
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set the gravity acceleration

# Visualize the simulation using the Irrlicht visualization system
my_application = irr.ChIrrApp(my_system, 'Crank-Slider Mechanism', irr.dimension2du(800, 600))
my_application.AddTypicalLights()
my_application.AddTypicalCamera(irr.vector3df(0, 3, -10))
my_application.AddSkyBox()
my_application.AddLogo()
my_application.AddLightWithShadow(irr.vector3df(3, 6, -10), irr.vector3df(0, 0, 0), 3, 2, 10, 40, 512)

# Run the simulation
my_application.SetTimestep(0.01)
my_application.SetTryRealtime(True)
my_application.AssetBindAll()
my_application.AssetUpdateAll()
my_application.Run()
```

This script initializes a PyChrono environment, creates a physical system with a floor, a crankshaft, a connecting rod, and a piston. It connects the crankshaft and connecting rod using a revolute joint, and the connecting rod and piston using another revolute joint. A motor is then used to drive the crankshaft at a constant angular speed. Finally, the simulation is visualized using the Irrlicht visualization system with customizable camera settings, lighting, and additional visual elements.
