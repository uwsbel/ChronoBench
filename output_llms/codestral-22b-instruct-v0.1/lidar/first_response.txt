```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Initialize the PyChrono environment and core components
chrono.SetChronoDataPath("PATH_TO_PYCHRONO_DATA")
my_system = chrono.ChSystemNSC()

# Add the triangular mesh from a Wavefront .obj file as a fixed body in the scene
mesh_shape = chrono.ChBodyEasyMesh("PATH_TO_OBJ_FILE", 1000, True, True)
mesh_body = chrono.ChBody()
mesh_body.AddAsset(mesh_shape)
mesh_body.SetBodyFixed(True)
my_system.Add(mesh_body)

# Add a lidar sensor to the body using a sensor manager
lidar_sensor = chrono.ChLidarSensor()
lidar_sensor.SetName("LidarSensor")
lidar_sensor.SetScanDirection(chrono.ChVectorD(1, 0, 0))
lidar_sensor.SetScanLine(400)
lidar_sensor.SetScanSampling(1)
lidar_sensor.SetScanFrequency(10)
lidar_sensor.SetScanResolution(1)
lidar_sensor.SetScanSegment(0.1)
lidar_sensor.SetNoise(0.01)

sensor_manager = chrono.ChSensorManager(my_system)
sensor_manager.AddSensor(lidar_sensor)

# Apply noise filters, visualizations, and saving options to the lidar data
lidar_sensor.SetFilter(chrono.ChLidarFilter())
lidar_sensor.SetVisualizationType(chrono.LidarVisualizationType_POINTS)
lidar_sensor.SetDataRecording(True)

# Set the lidar's initial position in an orbit around the mesh
lidar_body = chrono.ChBody()
lidar_body.AddAsset(lidar_sensor)
lidar_body.SetPos(chrono.ChVectorD(2, 0, 0))
my_system.Add(lidar_body)

# Simulate the system while dynamically updating the lidar's position in an orbit
application = chronoirr.ChIrrApp(my_system, 'Lidar Sensor Simulation', chronoirr.dimension2du(1024, 768))
application.AddTypicalLogo()
application.AddTypicalSky()
application.AddTypicalCamera(chronoirr.vector3df(0, 3, -6))
application.AddTypicalLights()
application.SetTimestep(0.01)

while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()

    # Dynamically update the lidar's position in an orbit
    t = my_system.GetChTime()
    lidar_body.SetPos(chrono.ChVectorD(2 * np.cos(t), 0, 2 * np.sin(t)))

    # Print lidar buffer data at each simulation step
    lidar_data = lidar_sensor.GetData()
    print(lidar_data.buffer)

    application.DoStep()
    application.EndScene()
```

Note: Replace "PATH_TO_PYCHRONO_DATA" and "PATH_TO_OBJ_FILE" with the appropriate paths.

This script initializes the PyChrono environment, adds a triangular mesh as a fixed body in the scene, adds a lidar sensor to the body using a sensor manager, applies noise filters, visualizations, and saving options to the lidar data, sets the lidar's initial position in an orbit around the mesh, simulates the system while dynamically updating the lidar's position in an orbit, and prints lidar buffer data at each simulation step.
