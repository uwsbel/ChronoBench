```python
import pychrono as chrono
import pychrono.irrlicht as irr
import pychrono.vehicle as veh
import numpy as np

# Initialize the PyChrono environment
chrono.SetChronoDataPath("PYCHRONO_DATA_DIR")

# Create a system
my_system = chrono.ChSystemNSC()

# Create the HMMWV vehicle
hmmwv = veh.HMMWV()
hmmwv.Initialize(my_system)
hmmwv.SetChassisFixed(False)
hmmwv.SetChassisCollisionType(chrono.ChCollisionModel.Type_BOX)
hmmwv.SetChassisVisualizationType(chrono.VisualizationType_MESH)
hmmwv.SetTireType(veh.TireModelType_TMEASY)
hmmwv.SetTireVisualizationType(veh.VisualizationType_MESH)
hmmwv.SetTireStepSize(0.05)
hmmwv.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 1), chrono.ChQuaternionD(1, 0, 0, 0)))

# Create the SCM deformable terrain
terrain = veh.SCMTerrain(hmmwv.GetSystem())
terrain.SetSoilParameters(1e3,  # Bekker Kphi
                          5e4,  # Bekker Kc
                          0.31,  # Bekker n exponent
                          0.0,  # Mohr cohesive limit (Pa)
                          30,   # Mohr friction angle (degrees)
                          0.0,  # Janosi shear coefficient (m)
                          1.0,  # Janosi critical pressure (Pa)
                          0.0)  # Janosi exponent
terrain.SetPlane(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngX(chrono.CH_C_PI_2)))
terrain.Initialize(100, 100, 1, 1, 0, 0)
height_map = np.zeros((100, 100))
terrain.SetHeightMap(height_map)

# Create the driver system
driver = veh.ChDriver(hmmwv.GetVehicle(), 0)
driver.Initialize()

# Create the visualization
app = irr.ChIrrApp(hmmwv.GetSystem(), 'HMMWV on SCM Terrain', irr.dimension2du(1024, 768))
app.AddTypicalLights()
app.AddSkyBox()
app.AddCamera(chrono.ChVectorD(0, -10, 2))
app.AddLogo()
app.AddLightWithShadow(chrono.ChVectorD(20, 20, 20), chrono.ChVectorD(0, 0, 0), 25, 16, 70, 100)
app.AssetBindAll()
app.AssetUpdateAll()

# Simulation loop
my_system.SetChTime(0)
step_size = 1e-3
while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()
    app.DoStep()
    driver.Synchronize(my_system.GetChTime())
    my_system.DoStepDynamics(step_size)
    app.EndScene()
```

This script initializes the PyChrono environment and creates an HMMWV vehicle with specific initial conditions. It sets up an SCM deformable terrain with defined soil parameters and initializes it with a height map. A driver system for vehicle control is created, and real-time visualization using Irrlicht is set up with camera settings and lighting. Finally, a simulation loop is implemented that synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.
