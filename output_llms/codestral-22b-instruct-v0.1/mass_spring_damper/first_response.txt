```python
import pychrono as chrono
import pychrono.irrlicht as irr

# Initialize the PyChrono environment
chrono.SetChronoDataPath("PATH_TO_CHRONO_DATA_DIRECTORY")

# Create a system and add it to the multibody system
system = chrono.ChSystemNSC()

# Create a ground body
ground = chrono.ChBodyEasyBox(20, 1, 20, 1000, True, True)
system.Add(ground)

# Create a mass body
mass = chrono.ChBodyEasySphere(1, 1, 1000, True, True)
mass.SetPos(chrono.ChVectorD(0, 5, 0))
system.Add(mass)

# Create a spring-damper connection between the mass and ground
spring_damper = chrono.ChLinkTSDA()
spring_damper.Initialize(mass, ground, chrono.ChCoordsysD(chrono.ChVectorD(0, 5, 0)))
system.Add(spring_damper)

# Set default linear spring-damper parameters
spring_damper.SetSpringCoefficient(1000)
spring_damper.SetDampingCoefficient(50)

# Create an Irrlicht application to visualize the system
app = irr.ChIrrApp(system, "Mass-Spring-Damper System", irr.dimension2du(1024, 768))
app.AddTypicalLogo()
app.AddTypicalSky()
app.AddTypicalCamera(irr.vector3df(0, 10, -20))
app.AddTypicalLights()

# Create visual shapes for the mass and spring
ground_shape = irr.scene.IBoxSceneNode(20, 1, 20)
ground_shape.SetMaterialTexture(0, irr.video.ITexture("textures/concrete.jpg"))
app.GetSceneManager().GetRootSceneNode().AddChild(ground_shape)

mass_shape = irr.scene.ISphereSceneNode(1)
mass_shape.SetMaterialTexture(0, irr.video.ITexture("textures/redwhite.png"))
mass.AddAsset(mass_shape)

spring_shape = irr.scene.ICylinderSceneNode(0.1, 5)
spring_shape.SetMaterialTexture(0, irr.video.ITexture("textures/bluewhite.png"))
spring_shape.SetRotation(irr.vector3df(90, 0, 0))
spring_shape.SetPosition(irr.vector3df(0, 2.5, 0))
app.GetSceneManager().GetRootSceneNode().AddChild(spring_shape)

# Run the simulation
while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()
    app.DoStep()
    app.EndScene()
```

This script initializes a PyChrono environment and creates a single mass-spring-damper system with visualization using Irrlicht. The mass is connected to the ground using a `ChLinkTSDA` connection with default linear spring-damper parameters. Visual shapes for the mass and spring are also added to the scene for better visualization.
