```python
import pychrono as chrono
import pychrono.irrlicht as irr

# Initialize Chrono and Irrlicht
chrono.SetChronoDataPath("path_to_chrono_data_folder")
my_system = chrono.ChSystemNSC()
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the UAZBUS vehicle
uazbus = chrono.vehicle.WheeledVehicle(my_system)
uazbus.Initialize("UAZBUS", chrono.ChCoordsysD(chrono.ChVectorD(0, 0.5, 0), chrono.Q_from_AngX(0)))

# Create the rigid terrain
terrain = chrono.ChBodyEasyBox(10, 0.1, 10, 1000, True, True)
my_system.Add(terrain)
terrain.SetBodyFixed(True)
terrain.SetFriction(0.8)
terrain.SetRestitution(0.01)

# Create the driver system
driver = chrono.vehicle.ChDriver(uazbus)
driver.SetSteeringAngle(0)
driver.SetEngineSpeed(0)
driver.SetBraking(0)

# Create the visualization system
application = irr.ChIrrApp(my_system, 'UAZBUS Simulation', irr.dimension2du(1024, 768))
application.AddTypicalLogo()
application.AddTypicalCamera(irr.vector3df(0, 3, -10))
application.AddTypicalLights()
application.SetTimestep(0.01)

# Simulation loop
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()

    # Advance the vehicle, terrain, driver, and visual system
    my_system.DoStepDynamics(application.GetTimestep())
    driver.Synchronize(application.GetTimestep())
    application.Synchronize("", my_system)
```

This script initializes the PyChrono environment, creates a UAZBUS vehicle, a rigid terrain, a driver system, and a visualization system. It then enters a simulation loop that advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.
