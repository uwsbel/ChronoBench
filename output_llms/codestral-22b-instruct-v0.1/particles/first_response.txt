To create a PyChrono simulation that models the gravitational attraction between particles, we will follow the steps outlined in the instructions.

Here is the Python script to generate the simulation:

```python
import pychrono as chrono
import pychrono.irrlicht as irr
import random

# Initialize PyChrono environment and core components
chrono.SetChronoDataPath("path_to_chrono_data_folder")
my_system = chrono.ChSystemNSC()

# Set gravitational acceleration
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create an Irrlicht application
app = irr.ChIrrApp(my_system, 'Particle Attraction Simulation', irr.dimension2du(1024, 768))
app.AddTypicalLogo()
app.AddTypicalSky()
app.AddTypicalCamera(irr.vector3df(0, 2, -5))
app.AddTypicalLights()

# Create a particle emitter
particle_emitter = chrono.ChParticleEmitter(my_system)

# Set particle emitter parameters
particle_emitter.SetParticleVelocity(chrono.ChVectorD(0, 0, 0))
particle_emitter.SetParticleSpin(chrono.ChVectorD(0, 0, 0))
particle_emitter.SetParticleMass(0.001)
particle_emitter.SetParticleMaterial(chrono.ChMaterialSurfaceNSC())
particle_emitter.SetParticleDistributionType(chrono.ChParticleEmitter.DISTR_UNIFORM)
particle_emitter.SetParticleDistributionRadius(1)
particle_emitter.SetParticleDistributionCenter(chrono.ChVectorD(0, 2, 0))

# Create a custom gravitational force function
def custom_gravity(particle, time):
    # Calculate the gravitational force between particles
    force = chrono.ChVectorD(0, 0, 0)
    for other_particle in particle_emitter.GetParticles():
        if particle != other_particle:
            distance = particle.GetPos() - other_particle.GetPos()
            force += distance.Normalize() * (particle.GetMass() * other_particle.GetMass() / distance.Length2())
    return force

# Set the custom gravitational force function
particle_emitter.SetCustomForceFunction(custom_gravity)

# Create particles
for _ in range(100):
    # Randomize shape, position, velocity, and orientation
    shape = random.choice([chrono.ChSphereShape, chrono.ChBoxShape, chrono.ChCylinderShape])
    radius = random.uniform(0.01, 0.02)
    position = chrono.ChVectorD(random.uniform(-1, 1), random.uniform(2, 3), random.uniform(-1, 1))
    velocity = chrono.ChVectorD(random.uniform(-0.1, 0.1), random.uniform(-0.1, 0.1), random.uniform(-0.1, 0.1))
    orientation = chrono.ChQuaternionD(random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-1, 1))
    orientation.Normalize()

    # Create the particle
    particle = chrono.ChBody()
    particle.SetPos(position)
    particle.SetPos_dt(velocity)
    particle.SetRot(orientation)
    particle.SetMass(particle_emitter.GetParticleMass())
    particle.SetMaterialSurface(particle_emitter.GetParticleMaterial())
    particle.SetCollide(True)
    particle.AddAsset(shape(radius))
    particle.SetBodyFixed(False)
    particle_emitter.AddParticle(particle)

# Run the simulation
app.SetTimestep(0.01)
app.SetTryRealtime(True)
app.AssetBindAll()
app.AssetUpdateAll()
while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()
    app.DoStep()
    app.EndScene()
```

In this script, we initialize the PyChrono environment and create a system to simulate the gravitational attraction between particles. We then create an Irrlicht application to visualize the simulation.

Next, we create a particle emitter that generates particles with random shapes, positions, velocities, and orientations. We set the particle emitter parameters accordingly.

To simulate particle attraction dynamically, we define a custom gravitational force function that calculates the gravitational force between each pair of particles. This function is then set as the custom force function for the particle emitter.

Finally, we create 100 particles with randomized properties and add them to the particle emitter. We run the simulation using the Irrlicht application.
