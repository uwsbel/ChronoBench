```python
# Import necessary PyChrono modules
import pychrono as chrono
import pychrono.irrlicht as irr
import pychrono.vehicle as veh

# Initialize the PyChrono system
chrono.SetChronoDataPath('PATH_TO_YOUR_CHRONO_DATA')
my_system = chrono.ChSystemNSC()

# Create the vehicle and add it to the system
vehicle = veh.ChVehicle(my_system)
vehicle.Initialize(veh.ChVehicleModelType_FEDA)
vehicle.SetChassisFixed(False)
vehicle.SetChassisCollisionType(chrono.ChCollisionModelType_BOX)
vehicle.SetChassisVisualizationType(veh.ChVehicleVisualizationType_MESH)
vehicle.SetChassisVisualMesh(chrono.GetChronoDataFile('vehicle/hmmwv/hmmwv_chassis.obj'))
vehicle.SetChassisCollisionMesh(chrono.GetChronoDataFile('vehicle/hmmwv/hmmwv_chassis_collision.obj'))
vehicle.SetChassisVisualScale(1)
vehicle.SetChassisPos(chrono.ChVectorD(0, 0, 1))
vehicle.SetChassisRot(chrono.ChQuaternionD(1, 0, 0, 0))

# Create the terrain and add it to the system
terrain = veh.ChRigidTerrain(my_system)
patch = terrain.AddPatch(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngX(0)))
patch.SetTexture(chrono.GetChronoDataFile('terrain/textures/tile4.jpg'), 200, 200)
terrain.Initialize()

# Set up the driver system
driver = veh.ChDriver(vehicle)
driver.Initialize()
driver.SetSteeringControllerType(veh.ChDriver.SteeringControllerType_PID)
driver.SetSpeedControllerType(veh.ChDriver.SpeedControllerType_PID)

# Set up the Irrlicht application
app = irr.ChIrrApp(my_system, 'FEDA Vehicle Simulation', irr.dimension2du(1024, 768))
app.AddTypicalLogo()
app.AddTypicalSky()
app.AddTypicalCamera(chrono.ChVectorD(0, -10, 2), chrono.ChVectorD(0, 0, 0))
app.AddTypicalLights()
app.SetTimestep(0.02)

# Run the simulation loop
while app.GetDevice().run():
    app.BeginScene()
    app.DrawAll()
    app.DoStep()
    app.EndScene()

# Close the application
app.GetDevice().closeDevice()
```

Please replace `'PATH_TO_YOUR_CHRONO_DATA'` with the actual path to your Chrono data directory. This script initializes the PyChrono environment, creates a FEDA vehicle and a rigid terrain, sets up an Irrlicht application for visualization, and runs a simulation loop.
