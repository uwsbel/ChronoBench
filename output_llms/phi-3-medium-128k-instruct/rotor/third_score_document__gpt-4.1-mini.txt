```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Set the data path for Chrono assets
chrono.SetChronoDataPath('/usr/local/share/chrono/data/')

# Create the physical system using NSC contact method
sys = chrono.ChSystemNSC()

# Create a material for contact properties
material = chrono.ChContactMaterialNSC()
material.SetFriction(0.4)
material.SetDampingF(0.2)

# Create a fixed ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
ground.SetPos(chrono.ChVector3d(0, 0, 0))
ground.EnableCollision(True)
ground_shape = chrono.ChVisualShapeBox(chrono.ChVector3d(10, 0.5, 10))
ground.AddVisualShape(ground_shape)
sys.Add(ground)

# Create a dynamic sphere body
sphere = chrono.ChBodyEasySphere(1.0, 1000, True, True, material)
sphere.SetPos(chrono.ChVector3d(0, 5, 0))
sys.Add(sphere)

# Create a revolute joint between the ground and the sphere
joint = chrono.ChLinkLockRevolute()
joint.Initialize(ground, sphere, chrono.ChCoordsysD(chrono.ChVector3d(0, 5, 0)))
sys.Add(joint)

# Setup the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('PyChrono Revolute Joint Example')
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddTypicalLights()
vis.AddCamera(chrono.ChVector3d(0, 5, 15))

# Simulation loop parameters
time_step = 0.01

# Run the simulation loop
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(time_step)
```
