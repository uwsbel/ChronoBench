Below is a self‐contained, corrected PyChrono script that  
1) fixes a couple of minor Python issues (removes a stray semicolon, calls the correct API for setting soil parameters, etc.)  
2) encapsulates the SCM terrain parameters in a small helper class, and  
3) allows you to pick among “soft,” “mid,” or “hard” presets when you build the terrain.

```python
import math

import pychrono.core as chrono
import pychrono.irrlicht as irr  # Never directly used in this example, but available if you want to drop down
import pychrono.vehicle as veh

# -------------------------------------------------------------------
# Helper class to encapsulate SCM soil parameters
# -------------------------------------------------------------------
class SCMTerrainParams:
    """
    A simple container for Bekker / Janosi–Hanamoto / elastic / damping
    parameters for the SCMTerrain model.  You can choose 'soft', 'mid',
    or 'hard' presets (or extend this with your own).
    """
    def __init__(self, preset="mid"):
        if preset == "soft":
            self.Kphi = 1e6       # Bekker Kphi
            self.Kc   = 0         # Bekker Kc
            self.n    = 1.0       # Bekker exponent
            self.coh  = 0.0       # cohesion (Pa)
            self.phi  = 20.0      # friction angle (deg)
            self.J     = 0.005    # Janosi shear coefficient (m)
            self.E     = 1e8      # elastic stiffness (Pa/m)
            self.damp  = 1e3      # damping (Pa·s/m)
        elif preset == "hard":
            self.Kphi = 4e6
            self.Kc   = 0
            self.n    = 1.2
            self.coh  = 0.0
            self.phi  = 35.0
            self.J     = 0.02
            self.E     = 5e8
            self.damp  = 5e4
        else:
            # default "mid" terrain
            self.Kphi = 2e6
            self.Kc   = 0
            self.n    = 1.1
            self.coh  = 0.0
            self.phi  = 30.0
            self.J     = 0.01
            self.E     = 2e8
            self.damp  = 3e4

    def apply(self, terrain):
        """Set these parameters on an SCMTerrain instance."""
        terrain.SetSoilParameters(
            self.Kphi,   # Bekker Kphi
            self.Kc,     # Bekker Kc
            self.n,      # Bekker n exponent
            self.coh,    # Mohr cohesive limit (Pa)
            self.phi,    # Mohr friction limit (degrees)
            self.J,      # Janosi shear coefficient (m)
            self.E,      # Elastic stiffness (Pa/m)
            self.damp    # Damping (Pa·s/m)
        )

# -------------------------------------------------------------------
# Main simulation setup
# -------------------------------------------------------------------

# Make sure Chrono knows where to find its data files
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

# Initial vehicle location and attitude
initLoc = chrono.ChVector3d(-8, 0, 0.6)
initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# Visualization/collision/tire settings
vis_type               = veh.VisualizationType_MESH
chassis_collision_type = veh.CollisionType_NONE
tire_model             = veh.TireModelType_RIGID

# Contact method
contact_method = chrono.ChContactMethod_SMC

# Time stepping
step_size        = 1e-3
tire_step_size   = step_size
render_fps       = 50
render_step_size = 1.0 / render_fps
render_steps     = math.ceil(render_step_size / step_size)

# Create the vehicle
vehicle = veh.HMMWV_Full()     # could also choose HMMWV_Reduced, etc.
vehicle.SetContactMethod(contact_method)
vehicle.SetChassisCollisionType(chassis_collision_type)
vehicle.SetChassisFixed(False)
vehicle.SetInitPosition(chrono.ChCoordsysd(initLoc, initRot))
vehicle.SetTireType(tire_model)
vehicle.SetTireStepSize(tire_step_size)

vehicle.Initialize()

vehicle.SetChassisVisualizationType(vis_type)
vehicle.SetSuspensionVisualizationType(vis_type)
vehicle.SetSteeringVisualizationType(vis_type)
vehicle.SetWheelVisualizationType(vis_type)
vehicle.SetTireVisualizationType(vis_type)

# Use Bullet for collisions (optional)
vehicle.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# Create and initialize the SCM terrain
terrain = veh.SCMTerrain(vehicle.GetSystem())

# Choose one of: 'soft', 'mid', 'hard'
soil_choice = "mid"
soil_params = SCMTerrainParams(soil_choice)
soil_params.apply(terrain)

# Enable a moving patch around the chassis
terrain.AddMovingPatch(
    vehicle.GetChassisBody(),
    chrono.ChVector3d(0, 0, 0),
    chrono.ChVector3d(5, 3, 1)
)

# For debugging, show sinkage as a color plot
terrain.SetPlotType(veh.SCMTerrain.PLOT_SINKAGE, 0, 0.1)

# Initialize the terrain mesh (length, width, grid spacing)
terrain.Initialize(20, 20, 0.02)

# -----------------------------------------------------------------------------
# Irrlicht visualization
# -----------------------------------------------------------------------------
vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle('HMMWV SCMTerrain Demo')
vis.SetWindowSize(1280, 1024)
# chase camera: track the point on the chassis at height 1.71 m
trackPoint = chrono.ChVector3d(0, 0, 1.71)
vis.SetChaseCamera(trackPoint, 6.0, 0.5)
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(vehicle.GetVehicle())

# -----------------------------------------------------------------------------
# Interactive driver (keyboard)
# -----------------------------------------------------------------------------
driver = veh.ChInteractiveDriverIRR(vis)
# steering / throttle / brake response times
driver.SetSteeringDelta(render_step_size / 1.0)
driver.SetThrottleDelta(render_step_size / 1.0)
driver.SetBrakingDelta(render_step_size / 0.3)
driver.Initialize()

# Print out the vehicle mass
print("VEHICLE MASS:", vehicle.GetVehicle().GetMass())

# -----------------------------------------------------------------------------
# Main simulation loop
# -----------------------------------------------------------------------------
realtime_timer = chrono.ChRealtimeStepTimer()
step_number    = 0
render_frame   = 0

while vis.Run():
    time = vehicle.GetSystem().GetChTime()

    # render
    if step_number % render_steps == 0:
        vis.BeginScene()
        vis.Render()
        vis.EndScene()
        render_frame += 1

    # fetch inputs
    driver_inputs = driver.GetInputs()

    # synchronize modules
    driver.Synchronize(time)
    terrain.Synchronize(time)
    vehicle.Synchronize(time, driver_inputs, terrain)
    vis.Synchronize(time, driver_inputs)

    # advance all
    driver.Advance(step_size)
    terrain.Advance(step_size)
    vehicle.Advance(step_size)
    vis.Advance(step_size)

    step_number += 1
    realtime_timer.Spin(step_size)
```

Key points:  
• We removed a stray semicolon.  
• We factored out the eight soil parameters into `SCMTerrainParams`, which you instantiate with “soft,” “mid,” or “hard.”  
• We call `soil_params.apply(terrain)` in place of the old direct `terrain.SetSoilParameters(...)`.  
• All other logic (driver, vehicle, visualization, loop) remains unchanged.
