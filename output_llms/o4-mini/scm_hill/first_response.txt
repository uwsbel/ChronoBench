Below is a self‐contained Python script that uses PyChrono to build exactly what you requested.  It:

 1. Creates a Chrono SMC system  
 2. Instantiates an HMMWV vehicle, places it at (0, 0.8, 0) with zero initial speed  
 3. Builds an SCM deformable terrain using a height‐map and user‐defined soil parameters  
 4. Hooks up an Irrlicht real‐time display (sky, lights, camera)  
 5. Creates an Irrlicht‐based GUI driver to steer/throttle/brake with the keyboard  
 6. Runs a loop that, each time step, synchronizes and advances vehicle, terrain, driver and Irrlicht—attempting real time  

Save this as, e.g., `demo_hmmwv_scm.py` and run with the standard PyChrono Python installation.

```python
import os
import pychrono.core as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr
import pychrono.vehicle.hmmwv as hmmwv

# ------------------------------------------------------------------
# 1) Chrono system and data path
# ------------------------------------------------------------------
chrono.SetChronoDataPath(os.environ.get('CHRONO_DATA_PATH', '')) 
system = chrono.ChSystemSMC()
step_size = 1e-3

# ------------------------------------------------------------------
# 2) Create and initialize the HMMWV vehicle
# ------------------------------------------------------------------
#   full‐model vehicle, SMC contacts, box chassis collision
vehicle = hmmwv.HMMWV_Full()
vehicle.SetContactMethod(chrono.ChContactMethod_SMC)
vehicle.SetChassisCollisionType(veh.ChassisCollisionType_BOX)
# initial position/orientation: at origin, 0.8 m above ground, no rotation
init_pos = chrono.ChCoordsysD(chrono.ChVectorD(0, 0.8, 0), chrono.QUNIT)
vehicle.SetInitPosition(init_pos)
vehicle.SetInitFwdVel(0.0)    # start at rest
vehicle.Initialize(system)

# choose rigid tires (default)
for t in vehicle.GetTireList():
    t.SetRollingMode(veh.TireRollingMode::ROLLING_MODE_RIGID)

# ------------------------------------------------------------------
# 3) Create SCM deformable terrain
# ------------------------------------------------------------------
terrain = veh.SCMDeformableTerrain(system)
# Specify the underlying rigid plane:
terrain.SetPlane(0, 0, 1, 0)  
# Soil parameters: Kphi, Kc, n, cohesion, friction, restitution, saturation
terrain.SetSoilParameters(
    Kphi=1.0e6,   # frictional modulus
    Kc=1.0e5,     # cohesive modulus
    n=1.1,        # exponent
    coh=0.0,      # cohesion
    mu=0.9,       # friction coefficient
    cr=0.01,      # restitution
    beta=0.0      # saturation
)
# Visualization mesh steps (dx, dy)
terrain.SetMeshResolutions(0.05, 0.05)
# Texture scaling
terrain.SetTextureScale(8, 8)
# Initialize from a height‐map image (grayscale), give your own path here
heightmap_file = chrono.GetChronoDataPath() + '/vehicle/hmmwv/terrain/height_maps/height_map.png'
terrain.Initialize(heightmap_file)
# Optional: plot pressure on the terrain
terrain.SetPlotType(veh.SCMDeformableTerrain.PLOT_PRESSURE_Y)

# ------------------------------------------------------------------
# 4) Create the Irrlicht application
# ------------------------------------------------------------------
app = chronoirr.ChIrrApp(
    system,                                    # the physical system
    'HMMWV on Deformable Terrain (SCM)',       # window title
    chronoirr.dimension2du(1280, 720)          # window size
)
app.AddTypicalSky()
app.AddTypicalLights()
app.AddCamera(chronoirr.ChVectorD(0, 2, -6), chronoirr.ChVectorD(0, 0, 0))
app.SetTryRealtime(True)
app.SetTimestep(step_size)

# Bind asset visualization
app.AssetBindAll()
app.AssetUpdateAll()

# ------------------------------------------------------------------
# 5) Create an Irrlicht‐based GUI driver
# ------------------------------------------------------------------
driver = veh.ChIrrGuiDriverIrrlicht(vehicle, app)
driver.SetSteeringDelta(0.02)
driver.SetThrottleDelta(0.02)
driver.SetBrakingDelta(0.02)
driver.Initialize()

# ------------------------------------------------------------------
# 6) Simulation loop
# ------------------------------------------------------------------
while app.Run():
    t = system.GetChTime()

    # 1) Collect driver inputs
    driver_inputs = driver.GetInputs()

    # 2) Synchronize vehicle, terrain, driver
    vehicle.Synchronize(t, driver_inputs, terrain)
    terrain.Synchronize(t)
    driver.Synchronize(t)
    
    # 3) Render scene
    app.BeginScene(True, True, chronoirr.SColor(255, 140, 161, 192))
    app.DrawAll()
    app.EndScene()

    # 4) Advance simulation for one time step
    vehicle.Advance(step_size)
    terrain.Advance(step_size)
    driver.Advance(step_size)
    system.DoStepDynamics(step_size)
```

A few notes:

• Make sure `CHRONO_DATA_PATH` is set (or pass the path directly to `SetChronoDataPath`).  
• The height‐map must exist at the specified location or you can substitute your own PNG/RAW file.  
• With arrow keys you can steer (left/right), throttle (up arrow), and brake (down arrow).  
• You can tune the soil parameters, camera position, step size, etc., to fit your needs.
