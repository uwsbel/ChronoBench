Below is a self‐contained PyChrono script that

 1. fixes a couple of minor API mismatches in your original (e.g. using `ChFrameD` instead of the old `ChFramed`, initializing contact‐surface properties on the bodies rather than trying to pass a raw `ChContactMaterialNSC` into the `ChBodyEasyBox` constructor, etc.),  
 2. imports and instantiates the Chrono Sensor module,  
 3. creates a lidar attached to the Curiosity chassis,  
 4. pushes two simple filters (a point‐cloud visualizer and a PCD‐writer),  
 5. adds the lidar to the sensor manager, and  
 6. calls `manager.Update()` inside the main loop.  

You should be able to run this as‐is assuming you have a working PyChrono + sensor + robot installation and the Chrono data directory visible via `CHRONO_DATA_DIR`.

```python
import os
import math
import numpy as np

import pychrono as chrono
import pychrono.robot as robot
import pychrono.sensor as sens
from pychrono import irrlicht as chronoirr

# -----------------------------------------------------------------------------
# 1) Create Chrono system and set global parameters
# -----------------------------------------------------------------------------
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.Set_G_acc(chrono.ChVector3d(0, 0, -9.81))

# suggested collision envelope/margin (optional tuning)
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# -----------------------------------------------------------------------------
# 2) Create ground (ChBodyEasyBox) and assign contact‐surface properties
# -----------------------------------------------------------------------------
ground = chrono.ChBodyEasyBox(20, 20, 1,    # size
                              1000,         # density
                              True, True)   # collide, visualize
ground.SetPos(chrono.ChVector3d(0, 0, -0.5))
ground.SetBodyFixed(True)
# set friction on the NSC surface material
ground_mat = ground.GetMaterialSurfaceNSC()
ground_mat.SetFriction(0.9)
# optional: restitution, compliance, etc.
system.Add(ground)

# -----------------------------------------------------------------------------
# 3) Create a long fixed box that the rover will cross
# -----------------------------------------------------------------------------
box = chrono.ChBodyEasyBox(0.25, 5, 0.25,
                          1000,
                          True, True)
box.SetPos(chrono.ChVector3d(0, 0, 0.0))
box.SetBodyFixed(True)
box_mat = box.GetMaterialSurfaceNSC()
box_mat.SetFriction(0.9)
system.Add(box)

# -----------------------------------------------------------------------------
# 4) Create the Curiosity rover, attach a DC‐motor driver, and initialize
# -----------------------------------------------------------------------------
rover = robot.Curiosity(system)
driver = robot.CuriosityDCMotorControl()
rover.SetDriver(driver)

init_pos = chrono.ChVector3d(-5, 0, 0)
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)
# Use ChFrameD (position + orientation) to initialize
rover.Initialize(chrono.ChFrameD(init_pos, init_rot))

# -----------------------------------------------------------------------------
# 5) Create the Irrlicht visualizer
# -----------------------------------------------------------------------------
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Curiosity rover - Rigid terrain with LiDAR')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 3),
              chrono.ChVector3d(0, 0, 0))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5),
                       chrono.ChVector3d(0, 0, 0),
                       3, 4, 10, 40, 512)
# vis.EnableShadows()   # enable if you want shadows

# -----------------------------------------------------------------------------
# 6) Create the sensor manager
# -----------------------------------------------------------------------------
manager = sens.ChSensorManager(system)
manager.SetVerbose(False)

# -----------------------------------------------------------------------------
# 7) Add a LiDAR to the rover’s chassis
# -----------------------------------------------------------------------------
# LiDAR specs
update_rate     = 10              # [Hz]
horiz_samples   = 360             # number of rays in the horizontal sweep
vert_samples    = 1               # single layer LiDAR
horiz_fov       = chrono.CH_C_2PI # full 360°
vert_fov        = chrono.CH_C_PI/8   # a small vertical aperture
min_distance    = 0.5             # [m]
max_distance    = 20.0            # [m]

# create the LiDAR sensor, attach to the rover chassis.  No pose offset
lidar = sens.ChLidarSensor(rover.GetChassisBody(),
                           update_rate,
                           horiz_samples,
                           vert_samples,
                           horiz_fov,
                           vert_fov,
                           min_distance,
                           max_distance)

# if you need to move the LiDAR on the chassis, you can uncomment:
# lidar.SetOffsetPose(chrono.ChFrameD(
#     chrono.ChVector3d(0.5, 0, 0.3),  # e.g. 0.5 m forward, 0.3 m up
#     chrono.ChQuaterniond(1, 0, 0, 0)
# ))

# Simple visualize‐pointcloud filter (draw points in the Irrlicht window)
vis_filter = sens.ChFilterVisualizePointCloud(max_points=3000)
lidar.PushFilter(vis_filter)

# PCD‐writer filter (write a PCD file each LiDAR scan)
pcd_folder = os.path.join(os.getcwd(), "lidar_scans")
os.makedirs(pcd_folder, exist_ok=True)
pcd_filter = sens.ChFilterPCDWriter(
    output_folder=pcd_folder,
    filename_format="scan_{:05d}.pcd",
    write_every=1)   # write every scan
lidar.PushFilter(pcd_filter)

# register the sensor
manager.AddSensor(lidar)

# -----------------------------------------------------------------------------
# 8) Main simulation loop
# -----------------------------------------------------------------------------
time_step = 1e-3
while vis.Run():
    # steering = 0: drive straight ahead
    driver.SetSteering(0.0)

    # let the rover internals update
    rover.Update()

    # update the sensor manager: this will fire the LiDAR at its rate
    manager.Update()

    # render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # advance the physical system
    system.DoStepDynamics(time_step)
```

Explanation of the key changes:

  • We removed the incorrect passing of a raw `ChContactMaterialNSC` into the `ChBodyEasyBox` constructor.  Instead we call `GetMaterialSurfaceNSC()` on each body and set friction (and optionally restitution, compliance, etc.).  
  • We replaced the old‐style `ChFramed` with `ChFrameD` when calling `rover.Initialize()`.  
  • We imported `pychrono.sensor as sens`, constructed a `ChSensorManager`, and in the loop call `manager.Update()`.  
  • We created a `ChLidarSensor`, attached it to the chassis, set its parameters, and pushed two filters: a real‐time visualizer and a PCD‐writer.  
  • Finally we added the LiDAR to the manager so it fires at the specified `update_rate`.  

This script should compile and run, populating `./lidar_scans` with PCD files and showing the reconstructed point cloud in the Irrlicht window.
