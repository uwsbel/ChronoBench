Below is a self‐contained Python script that uses PyChrono with the IGA module to build a Jeffcott‐style rotor: a single cantilever beam (modeled as an IGA Cosserat beam), a flywheel rigid body at mid‐span, and a simple rotational motor at the clamped end.  An Irrlicht window is opened so you can watch the vibration of the beam and the flywheel as the motor spins.

```python
import pychrono as chrono
import pychrono.irrlicht as irr
import pychrono.iga as iga
import pychrono.fea as fea

# -----------------------------------------------------------------------------
# 1)  Initialize Chrono system
# -----------------------------------------------------------------------------
system = chrono.ChSystemSMC()
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

# -----------------------------------------------------------------------------
# 2)  Build a straight IGA beam (cantilever) along the X‐axis
# -----------------------------------------------------------------------------
L = 1.0           # beam length
nr_elems = 20     # number of beam elements

# 2a) NURBS curve for the beam centerline (cubic, 4 control points)
degree = 3
ctrl_pts = [
    chrono.ChVectorD(0.0, 0.0, 0.0),
    chrono.ChVectorD(L/3, 0.0, 0.0),
    chrono.ChVectorD(2*L/3, 0.0, 0.0),
    chrono.ChVectorD(L, 0.0, 0.0),
]
# uniform clamped knot vector for 4 pts, degree=3: 0,0,0,0,1,1,1,1
knots = [0,0,0,0, 1,1,1,1]
weights = [1,1,1,1]

curve = iga.ChNurbsCurve(degree, knots, weights, ctrl_pts)

# 2b) Define beam section (circular cross‐section)
section = iga.ChBeamSectionCosserat()
section.SetDensity(7800)              # steel density
section.SetYoungModulus(2e11)         # steel Young's modulus
section.SetGwithPoissonRatio(0.3)     # calculates shear modulus
section.SetAsCircularSection(0.02)    # radius = 2 cm

# 2c) Build the IGA beam mesh
builder = iga.ChBuilderBeamIGA(system)
builder.BuildBeam(curve, section, nr_elems,
                  iga.ChBuilderBeamIGA.BEAM_ELEMENT_TYPE_Iga)

mesh = builder.GetMesh()
system.Add(mesh)

# Optional: set mesh gravity or other properties
mesh.SetAutomaticGravity(False)

# -----------------------------------------------------------------------------
# 3)  Clamp one end of the beam but leave rotation about X free, then add a motor
# -----------------------------------------------------------------------------
# Identify the first node (clamped end) in the mesh
node0 = mesh.GetNode(0)
pos0 = node0.GetPos()  # should be (0,0,0)

# Create a fixed ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
system.Add(ground)

# Attach a marker on the beam node for FEA orientation
marker_beam = chrono.ChMarkerFEAxyzrot(node0)
system.Add(marker_beam)

# Create a matching marker on the ground at the same location
marker_ground = chrono.ChMarker()
marker_ground.SetFrame(chrono.ChFrameD(pos0, chrono.QUNIT))
ground.AddMarker(marker_ground)

# Constrain all translations + rotations about Y,Z. Leave rot‐X free.
link_clamp = chrono.ChLinkMateGeneric()
link_clamp.Initialize(marker_ground, marker_beam)
# order: (X, Y, Z, Rx, Ry, Rz)
link_clamp.SetConstrainedCoords(True, True, True,
                                False, True, True)
system.Add(link_clamp)

# Create a simple motor to spin the beam about its axis (X)
motor = chrono.ChLinkMotorRotationAngleSpeed()
motor.Initialize(marker_ground, marker_beam,
                 chrono.ChFrameD(pos0, chrono.Q_from_AngAxis(chrono.CH_C_PI / 2, chrono.ChVectorD(0,1,0))))
# Define a constant‐speed function: ramp to 1000 rpm in 0.1 s, then hold
rpm_target = 1000
omega_target = rpm_target * 2 * chrono.CH_C_PI / 60.0
f_angle = chrono.ChFunction_Ramp(0, omega_target)  
motor.SetAngleFunction(f_angle)
motor.SetSpeedFunction(chrono.ChFunction_Const(omega_target))
system.Add(motor)

# -----------------------------------------------------------------------------
# 4)  Add a flywheel rigid body at mid‐span and rigidly attach to the beam
# -----------------------------------------------------------------------------
# Find the node closest to mid‐span
num_nodes = mesh.GetNnodes()
i_mid = num_nodes // 2
node_mid = mesh.GetNode(i_mid)
pos_mid = node_mid.GetPos()

# Create the flywheel as a simple cylinder body
flywheel = chrono.ChBody()
flywheel.SetMass(2.0)  # kg
# Approximate inertia of a solid disk radius=0.1 m, thickness=0.02 m
Rw = 0.1
Tw = 0.02
Iw = 0.5 * flywheel.GetMass() * Rw*Rw
flywheel.SetInertiaXX(chrono.ChVectorD(Iw, 0.5*Iw, 0.5*Iw))
flywheel.SetPos(pos_mid)
flywheel.SetRot(chrono.QUNIT)
system.Add(flywheel)

# Markers for flywheel and beam node
marker_fw = chrono.ChMarker()
marker_fw.SetFrame(chrono.ChFrameD(pos_mid, chrono.QUNIT))
flywheel.AddMarker(marker_fw)

marker_beam_mid = chrono.ChMarkerFEAxyz(node_mid)
system.Add(marker_beam_mid)

# Rigidly attach (all 6 DOF) mid‐span node to flywheel
link_fw = chrono.ChLinkMateGeneric()
link_fw.Initialize(marker_fw, marker_beam_mid)
link_fw.SetConstrainedCoords(True, True, True, True, True, True)
system.Add(link_fw)

# -----------------------------------------------------------------------------
# 5)  Visual assets: wireframe beam + disc + Irrlicht window
# -----------------------------------------------------------------------------
# (a) add a FEA visualization asset to the mesh
vis_mesh = fea.ChVisualizationFEAmesh(mesh)
vis_mesh.SetWireframe(True)
vis_mesh.SetDrawUndeformed(False)
mesh.AddAsset(vis_mesh)

# (b) set up the Irrlicht application
application = irr.ChIrrApp(
    system,                            # the physical system to visualize
    "Jeffcott Rotor with IGA Beam",   # window title
    irr.dimension2du(1024, 768)
)
application.AddTypicalLights()
application.AddLogo()
application.AddSkyBox()
application.AddCamera(irr.vector3df(0.5, -1.5, 0.5))
application.SetShowInfo(True)
application.AssetBindAll()
application.AssetUpdateAll()

# -----------------------------------------------------------------------------
# 6)  Solve & animate
# -----------------------------------------------------------------------------
application.SetTimestep(1e-3)
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```

Explanation of the main blocks:

1. **Chrono system setup**  
   We use a ChSystemSMC, apply gravity, and prepare the solver.

2. **IGA beam creation**  
   - We define a cubic NURBS centerline spanning 0→L.  
   - We create a Cosserat beam section (circular cross‐section).  
   - The `ChBuilderBeamIGA` object populates the FEA mesh with beam elements and nodes.

3. **Clamped‐end motor**  
   - The first beam node is “clamped” in all translations and in rotations about Y/Z, but allowed to spin about X.  
   - A `ChLinkMotorRotationAngleSpeed` imposes a prescribed spin speed about X.

4. **Central flywheel**  
   - A small rigid body (disc) is placed at the mid‐span node.  
   - A 6‐DOF generic mate locks the node to the disc, so that the disc spins with the beam and also contributes inertia.

5. **Visualization**  
   - A `ChVisualizationFEAmesh` asset draws the beam in wireframe.  
   - The Irrlicht app renders the beam, motor and flywheel in real time.

When run, you should see the beam cantilever spinning up to the target RPM and begin to exhibit lateral vibration modes typical of a Jeffcott rotor.
