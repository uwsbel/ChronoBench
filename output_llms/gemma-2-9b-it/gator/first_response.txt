import numpy as np

vehicle.SetMaterialSurface(chrono.ChMaterialSurface(0.5, 0.5, 0.5))  # Adjust friction coefficients

    wheel.SetRollingFriction(0.05)  # Adjust rolling friction
    world.Add(wheel)

vehicle.AddVisual(vehicle_visual)

# Simulation loop
while app.run():

vis.close()
