```python
import pychrono as chrono
import chrono.irrlicht as irrlicht

# Initialize Chrono
chrono.SetChronoDataPath("path/to/chrono/data")  # Replace with your Chrono data path
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravitational acceleration

# Create Irrlicht visualization
vis = irrlicht.ChIrrApp(system, 'Particle Attraction')
vis.SetWindowSize(800, 600)
vis.AddCamera(chrono.ChVectorD(0, 10, 10), chrono.ChVectorD(0, 0, 0))
vis.SetChaseTarget(chrono.ChVectorD(0, 0, 0))

# Particle emitter
class ParticleEmitter(chrono.ChBody):
    def __init__(self, system, num_particles, particle_radius, mass_range):
        super().__init__(system)
        self.num_particles = num_particles
        self.particle_radius = particle_radius
        self.mass_range = mass_range

        for i in range(self.num_particles):
            # Generate random particle properties
            shape = chrono.ChBoxShape(chrono.ChVectorD(self.particle_radius, self.particle_radius, self.particle_radius))
            particle = chrono.ChBodyEasyBox(self.particle_radius, self.particle_radius, self.particle_radius, 
                                            mass=chrono.ChRandom.rand_double(self.mass_range[0], self.mass_range[1]),
                                            pos=chrono.ChVectorD(chrono.ChRandom.rand_double(-5, 5), chrono.ChRandom.rand_double(-5, 5), chrono.ChRandom.rand_double(-5, 5)))
            particle.SetBodyFixed(True)
            particle.SetCollide(True)
            particle.SetMaterialSurface(chrono.ChMaterialSurface())
            self.Add(particle)

# Create particle emitter
emitter = ParticleEmitter(system, num_particles=100, particle_radius=0.1, mass_range=(0.1, 1))

# Custom gravitational force
def apply_gravitational_force(particle1, particle2):
    dist = particle1.GetPos() - particle2.GetPos()
    force = chrono.ChVectorD(0, 0, 0)
    if dist.Length() > 0.1:
        force = -chrono.ChVectorD(dist.x / dist.Length(), dist.y / dist.Length(), dist.z / dist.Length()) * 0.1
    particle1.AddForce(force)
    particle2.AddForce(-force)

# Simulation loop
while vis.Run():
    system.DoStepDynamics(0.01)

    # Apply custom gravitational forces between particles
    for particle1 in emitter.GetChildren():
        for particle2 in emitter.GetChildren():
            if particle1 != particle2:
                apply_gravitational_force(particle1, particle2)

    vis.Render()

vis.Close()
```

**Explanation:**

1. **Initialization:**
   - Imports necessary libraries (`pychrono` and `chrono.irrlicht`).
   - Initializes the Chrono environment with a `ChSystemNSC`.
   - Sets the gravitational acceleration to -9.81 m/s² along the Y-axis.
   - Creates an Irrlicht visualization window and sets up a camera.

2. **Particle Emitter:**
   - Defines a custom `ParticleEmitter` class that inherits from `ChBody`.
   - Constructor initializes the number of particles, particle radius, and mass range.
   - Creates particles with random shapes (boxes), positions, velocities, and orientations.
   - Adds each particle to the system.

3. **Custom Gravitational Force:**
   - Defines a function `apply_gravitational_force` that calculates the gravitational force between two particles based on their distance.
   - The force is proportional to the inverse square of the distance and acts along the line connecting the particles.

4. **Simulation Loop:**
   - Enters a loop that runs until the user closes the visualization window.
   - Updates the simulation dynamics using `system.DoStepDynamics(0.01)`.
   - Iterates through all particles in the emitter and applies the custom gravitational force to each pair of particles.
   - Renders the visualization using `vis.Render()`.

5. **Closing:**
   - Closes the Irrlicht visualization window using `vis.Close()`.



**Remember:**

- Replace `"path/to/chrono/data"` with the actual path to your Chrono data directory.
- Adjust the parameters (e.g., `num_particles`, `particle_radius`, `mass_range`) to control the simulation behavior.
- You can further customize the particle shapes, colors, and sizes in the `ParticleEmitter` class.
