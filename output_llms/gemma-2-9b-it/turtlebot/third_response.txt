```python
import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as turtlebot
from pychrono import irrlicht as chronoirr
from pychrono.sensor import *

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))  # Set gravity in negative Z direction
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create ground body as terrain plane that robot will drive on
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVector3d(0, 0, -0.6))  # Position the ground slightly below the origin
ground.SetFixed(True)  # Fix the ground in place
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# Create Turtlebot Robot
init_pos = chrono.ChVector3d(0, 0.2, 0)  # Initial position of the robot
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)  # Initial orientation of the robot
robot = turtlebot.TurtleBot(system, init_pos, init_rot)  # Create Turtlebot instance
robot.Initialize()  # Initialize the robot

# Create sensor manager
sensor_manager = chrono.ChSensorManager()
system.Add(sensor_manager)

# Configure and add lidar sensor
lidar = chrono.ChLidarSensor()
lidar.SetMaxRange(10)  # Set maximum range of the lidar
lidar.SetAngleResolution(0.1)  # Set angle resolution
lidar.SetScanFrequency(10)  # Set scan frequency
lidar.SetNumLaserBeams(360)  # Set number of laser beams
lidar.SetFilterType(chrono.ChLidarSensor.FilterType_GAUSSIAN)  # Set filter type
sensor_manager.AddSensor(lidar)
lidar.SetBody(robot.GetChassis())  # Attach lidar to robot's chassis

# Add randomly placed boxes
for i in range(5):
    box = chrono.ChBodyEasyBox(0.5, 0.5, 0.5, 1000, True, True)
    box.SetPos(chrono.ChVector3d(np.random.uniform(-5, 5), np.random.uniform(-5, 5), 0.1))
    system.Add(box)

# Create run-time visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chrono.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Turtlebot Robot - Rigid terrain')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 1.5, 0.2), chrono.ChVector3d(0, 0, 0.2))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVector3d(1.5, -2.5, 5.5), chrono.ChVector3d(0, 0, 0.5), 3, 4, 10, 40, 512)

# Enable shadows (commented out to improve performance)
# vis.EnableShadows()

# Set the simulation time step
time_step = 2e-3

# Motion control function
def move(mode):
    if mode == 'straight':
        robot.SetMotorSpeed(0.5, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0.5, RIGHT_DRIVE_WHEEL)
    elif mode == 'left':
        robot.SetMotorSpeed(0.5, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(-0.5, RIGHT_DRIVE_WHEEL)
    elif mode == 'right':
        robot.SetMotorSpeed(-0.5, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0.5, RIGHT_DRIVE_WHEEL)

# Simulation loop
time = 0
while vis.Run():
    # Update sensor manager
    sensor_manager.Update()

    # Move the Turtlebot
    move('straight') 

    # Increment time counter
    time += time_step

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance the simulation by one time step
    system.DoStepDynamics(time_step)



```

**Changes Made:**

1. **Ground Position:** Changed `chrono.ChVector3d(0, 0, -1)` to `chrono.ChVector3d(0, 0, -0.6)` in the ground body's position.

2. **Sensor Manager and Lidar:**
   - Imported `pychrono.sensor as sens`.
   - Created a `chrono.ChSensorManager` object and added it to the system.
   - Created a `chrono.ChLidarSensor` object, configured its parameters (range, resolution, frequency, beams, filter type), and added it to the sensor manager.
   - Attached the lidar sensor to the robot's chassis using `lidar.SetBody(robot.GetChassis())`.

3. **Random Boxes:**
   - Added a loop to create 5 boxes with random positions using `np.random.uniform`.
   - Each box is added to the system using `system.Add(box)`.

4. **Motion Control Function:**
   - Created a `move(mode)` function that takes a string argument (`mode`) representing the desired movement:
     - `'straight'`: Moves the robot forward.
     - `'left'`: Turns the robot left.
     - `'right'`: Turns the robot right.
   - The function sets the motor speeds for the left and right wheels accordingly.

5. **Simulation Loop Update:**
   - Called `sensor_manager.Update()` within the loop to process sensor data.
   - Used `move('straight')` to control the robot's movement.
