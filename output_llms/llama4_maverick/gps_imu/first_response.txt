Here's a Python script that implements the simulation as per the given instructions:

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as chronoirr
import numpy as np

# Initialize PyChrono environment and core components
def main():
    # 1. Set up the HMMWV vehicle
    veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')
    my_hmmwv = veh.HMMWV_Full()
    my_hmmwv.SetContactMethod(chrono.ChContactMethod_SMC)
    my_hmmwv.SetChassisFixed(False) 
    my_hmmwv.SetInitPosition(chrono.ChCoordsysD(chrono.ChVectorD(-5, -5, 0.5), chrono.ChQuaternionD(1, 0, 0, 0)))
    my_hmmwv.SetPowertrainType(veh.PowertrainModelType_SHAFTS)
    my_hmmwv.SetDriveType(veh.DrivelineTypeWV_AWD)
    my_hmmwv.SetSteeringType(veh.SteeringTypeWV_PITMAN_ARM)
    my_hmmwv.SetTireType(veh.TireModelType_TMEASY)
    my_hmmwv.Initialize()

    # 2. Create the terrain
    terrain = veh.RigidTerrain(my_hmmwv.GetSystem())
    patch_mat = chrono.ChMaterialSurfaceSMC()
    patch_mat.SetFriction(0.9)
    patch_mat.SetRestitution(0.01)
    patch = terrain.AddPatch(patch_mat, chrono.CSYSNORM, 100, 100)
    patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)
    patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
    terrain.Initialize()

    # 3. Create the driver
    driver = veh.ChDriver(my_hmmwv.GetVehicle())
    driver.Initialize()

    # 4. Create the Irrlicht visualization
    vis = chronoirr.ChVisualSystemIrrlicht()
    vis.AttachSystem(my_hmmwv.GetSystem())
    vis.SetWindowSize(800, 600)
    vis.SetWindowTitle('HMMWV Simulation')
    vis.Initialize()
    vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
    vis.AddSkyBox()
    vis.AddCamera(chrono.ChVectorD(1.0, 1.4, 1.2), chrono.ChVectorD(0, 0, 0))
    vis.AddTypicalLights()

    # 5. Add IMU and GPS sensors
    imu = veh.ChIMUSensor(my_hmmwv.GetChassisBody(), 100, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT))
    imu.PushUpdate()
    my_hmmwv.GetSystem().AddSensor(imu)

    gps = veh.ChGPSSensor(my_hmmwv.GetChassisBody(), 100, chrono.ChFrameD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT), chrono.ChVectorD(0, 0, 0), 0.01, 0.01)
    gps.PushUpdate()
    my_hmmwv.GetSystem().AddSensor(gps)

    # Simulation loop
    realtime_timer = chrono.ChRealtimeStepTimer()
    while vis.Run():
        time = my_hmmwv.GetSystem().GetChTime()

        # Driver inputs
        driver_inputs = driver.GetInputs()
        my_hmmwv.Synchronize(time, driver_inputs, terrain)
        terrain.Synchronize(time)
        vis.Synchronize("HMMWV", driver_inputs)

        # Advance simulation
        my_hmmwv.Advance(0.01)
        terrain.Advance(0.01)
        vis.Advance(0.01)

        # Update sensors
        imu.PushUpdate()
        gps.PushUpdate()

        # Output vehicle mass
        print(f"Vehicle mass: {my_hmmwv.GetVehicle().GetMass()}")

        # Spin in place for real time
        realtime_timer.Spin(0.01)

    return 0

if __name__ == "__main__":
    main()
```

This script initializes a PyChrono simulation with an HMMWV vehicle, sets up a rigid terrain, and includes a driver model. It also adds IMU and GPS sensors to the vehicle's chassis and visualizes the simulation using Irrlicht. The simulation loop updates the vehicle, terrain, and driver states, advances the simulation, and outputs the vehicle's mass at each time step.

To run this script, ensure you have the necessary dependencies installed, including PyChrono and Irrlicht. The script assumes a basic understanding of PyChrono and its vehicle module. Adjustments may be necessary based on the specific requirements of your project, such as changing the terrain properties or the initial conditions of the vehicle.
