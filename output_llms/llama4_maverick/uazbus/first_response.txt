Here's a Python script that follows the given instructions to create a PyChrono simulation featuring a UAZBUS vehicle:

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr
import math

# Initialize the PyChrono environment
print('Copyright (c) 2023 Project Chrono')

# Create the UAZBUS vehicle
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')
uazbus_vehicle = veh.UAZBUS(veh.SteeringType_POWERTRAIN, False, veh.BrakeType_SIMPLE)

# Set initial conditions for the vehicle
initLoc = chrono.ChVectorD(0, 0, 1.0)
initRot = chrono.ChQuaternionD(1, 0, 0, 0)
uazbus_vehicle.Initialize(chrono.ChCoordsysD(initLoc, initRot))

# Set terrain properties
terrain = veh.RigidTerrain(uazbus_vehicle.GetSystem())
patch = terrain.AddPatch(chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.QUNIT), 
                         chrono.ChVectorD(100, 100, 0), 0.0)
patch.SetContactFrictionCoefficient(0.9)
patch.SetContactRestitutionCoefficient(0.01)
patch.SetContactMaterialProperties(2e7, 0.3)
patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)
terrain.Initialize()

# Create the driver system
driver = veh.ChDriver(uazbus_vehicle.GetVehicle())
driver.Initialize()

# Create the Irrlicht visualization
vis = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis.SetWindowTitle('UAZBUS Demo')
vis.SetWindowSize(1280, 720)
vis.SetChaseCamera(chrono.ChVectorD(0.0, 0.0, 1.75), 6.0, 0.5)
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddLightDirectional()
vis.AddSkyBox()
vis.AttachVehicle(uazbus_vehicle.GetVehicle())

# Simulation loop
while vis.Run():
    time = uazbus_vehicle.GetSystem().GetChTime()
    driver_inputs = driver.GetInputs()
    uazbus_vehicle.Advance(time)
    terrain.Advance(time)
    driver.Advance(time)
    vis.Advance(time)
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    # Synchronize with the Irrlicht event loop to maintain real-time execution
    irr.ChIrrAppUtilities.sleep(1)
```

This script creates a PyChrono simulation featuring a UAZBUS vehicle with specified initial conditions and visualization settings. The simulation includes a rigid terrain with defined friction and restitution, a driver system for vehicle control, and a real-time visualization using Irrlicht with camera settings and lighting. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.
