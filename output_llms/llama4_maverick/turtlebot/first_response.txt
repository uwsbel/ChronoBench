Here's a Python script that follows the given instructions to create a PyChrono simulation with a Turtlebot robot on a rigid terrain.

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# Initialize the PyChrono environment
chrono.SetChronoDataPath("/path/to/chrono/data/")

# Create the system and set gravity
sys = chrono.ChSystemNSC()
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the ground body
ground = chrono.ChBody()
ground.SetBodyFixed(True)
sys.Add(ground)

# Initialize the Turtlebot with a specific position and orientation
turtlebot = chrono.ChBody()
turtlebot.SetMass(10)
turtlebot.SetInertiaXX(chrono.ChVectorD(1, 1, 1))
turtlebot.SetPos(chrono.ChVectorD(0, 0.5, 0))
turtlebot.SetRot(chrono.ChQuaternionD(1, 0, 0, 0))
sys.Add(turtlebot)

# Add wheels to the Turtlebot
wheel_L = chrono.ChBody()
wheel_L.SetMass(1)
wheel_L.SetInertiaXX(chrono.ChVectorD(0.1, 0.1, 0.1))
wheel_L.SetPos(chrono.ChVectorD(-0.5, 0.25, 0))
sys.Add(wheel_L)

wheel_R = chrono.ChBody()
wheel_R.SetMass(1)
wheel_R.SetInertiaXX(chrono.ChVectorD(0.1, 0.1, 0.1))
wheel_R.SetPos(chrono.ChVectorD(0.5, 0.25, 0))
sys.Add(wheel_R)

# Create joints between the Turtlebot body and wheels
motor_L = chrono.ChLinkMotorRotationSpeed()
motor_L.Initialize(turtlebot, wheel_L, chrono.ChFrameD(chrono.ChVectorD(-0.5, 0.25, 0)))
sys.Add(motor_L)

motor_R = chrono.ChLinkMotorRotationSpeed()
motor_R.Initialize(turtlebot, wheel_R, chrono.ChFrameD(chrono.ChVectorD(0.5, 0.25, 0)))
sys.Add(motor_R)

# Create the Irrlicht visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Turtlebot Simulation')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 2, -4))
vis.AddTypicalLights()
vis.AttachSystem(sys)

# Simulation loop
time = 0
while vis.Run():
    vis.BeginScene()
    vis.Render()
    sys.DoStepDynamics(0.01)
    time += 0.01
    
    # Control the Turtlebot's motors
    if time < 2:
        motor_L.SetSpeed(1)
        motor_R.SetSpeed(1)
    elif time < 4:
        motor_L.SetSpeed(-1)
        motor_R.SetSpeed(1)
    else:
        motor_L.SetSpeed(1)
        motor_R.SetSpeed(-1)
        
    vis.EndScene()
    ```
