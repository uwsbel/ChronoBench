This script initializes the PyChrono environment, sets up a Gator vehicle, a rigid terrain, an interactive driver system, and a sensor manager with point lights and a camera attached to the vehicle's chassis. The script also includes a simulation loop to update and render the simulation.

```python
import pychrono as chrono
import pychrono.vehicle as veh
import pychrono.postprocess as postprocess
import pychrono.irrlicht as irr

# Initialize the PyChrono environment
chrono.ChSystemNSC msystem = chrono.ChSystemNSC()
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Set up the rigid terrain
terrain = chrono.ChBodyEasyBox(10, 0.1, 10, 1000, True, True)
terrain.SetPos(chrono.ChVectorD(0, -0.05, 0))
terrain.SetBodyFixed(True)
msystem.Add(terrain)

# Create the Gator vehicle
gator = veh.Gator()
gator.SetChassisVisualizationType(veh.ChVisualizationType::PRISM_UV)
gator.SetSuspensionVisualizationType(veh.ChVisualizationType::PRISM_UV)
gator.SetWheelVisualizationType(veh.ChVisualizationType::PRISM_UV)
gator.SetSteeringVisualizationType(veh.ChVisualizationType::PRISM_UV)

# Initialize the vehicle system
vehicle = veh.ChVehicleSystem()
vehicle.SetChassisBody(gator.GetChassisBody())
vehicle.SetTireForceHandler(gator.GetTireForceHandler())
vehicle.SetDriver(gator.GetDriver())
vehicle.Initialize()

# Add the vehicle to the system
msystem.Add(vehicle)

# Set up the sensor manager
sensor_manager = veh.ChSensorManager()
sensor_manager.Initialize(vehicle)

# Add point lights
point_light1 = chrono.ChPointLight()
point_light1.SetPos(chrono.ChVectorD(5, 5, 5))
point_light1.SetDir(chrono.ChVectorD(-1, -1, -1))
point_light1.SetRadius(10)
point_light1.SetIntensity(1.0)
sensor_manager.AddPointLight(point_light1)

point_light2 = chrono.ChPointLight()
point_light2.SetPos(chrono.ChVectorD(-5, 5, 5))
point_light2.SetDir(chrono.ChVectorD(1, -1, -1))
point_light2.SetRadius(10)
point_light2.SetIntensity(1.0)
sensor_manager.AddPointLight(point_light2)

# Add a camera attached to the vehicle's chassis
camera = chrono.ChCamera()
camera.SetPos(chrono.ChVectorD(0, 2, -5))
camera.SetDir(chrono.ChVectorD(0, 0, 1))
camera.SetUp(chrono.ChVectorD(0, 1, 0))
camera.SetNearClip(0.1)
camera.SetFarClip(100)
camera.SetFov(chrono.CH_C_PI_4)
sensor_manager.AddCamera(camera)

# Set up the visualization system
application = irr.ChIrrApp()
application.AddVisualSystem(irr.ChIrrApp::IrrlichtCreateParams())
application.AddSystem(msystem)
application.AddSystem(sensor_manager)

# Create a post-processor for rendering
postprocess.PostProcess()
postprocess.PostProcess.SetRenderFlag(postprocess.ERenderFlags.RENDER_WIREFRAME)

# Run the simulation loop
step_size = 0.01
end_time = 10.0
time = 0.0

while time < end_time:
    application.BeginScene()
    application.DrawAll()
    application.EndScene()

    # Update the driver
    vehicle.GetDriver().SynchronizeAction(vehicle.GetDriver().GetInputEnv(), vehicle.GetDriver().GetOutputEnv())

    # Update the terrain
    terrain.SetPos(chrono.ChVectorD(0, -0.05, 0))

    # Update the vehicle
    vehicle.Synchronize(msystem, step_size)
    vehicle.Advance(step_size)

    # Update the sensor manager
    sensor_manager.Update(msystem, step_size)

    # Advance the system
    msystem.DoStepDynamics(step_size)

    time += step_size

# Close the application
application.Close()
```

### Explanation:
1. **Initialization**: The PyChrono environment and core components are initialized.
2. **Terrain**: A rigid terrain is created and added to the system.
3. **Vehicle**: A Gator vehicle is created with specified visualization types for its parts.
4. **Driver System**: An interactive driver system is set up for the vehicle.
5. **Sensor Manager**: A sensor manager is created with point lights and a camera attached to the vehicle's chassis.
6. **Simulation Loop**: The simulation loop updates the driver, terrain, vehicle, and sensor manager at each timestep, synchronizing and advancing the simulation modules.
