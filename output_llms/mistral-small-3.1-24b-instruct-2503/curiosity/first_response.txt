```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Initialize the Chrono::Engine
chrono.SetChronoDataPath('path/to/chrono/data')  # Set the path to Chrono data files

# Create the simulation system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the ground
ground = chrono.ChBodyEasyBox(50, 1, 50, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -0.5, 0))
ground.SetBodyFixed(True)
system.Add(ground)

# Create the Curiosity rover
# Load the rover model (assuming a URDF file is available)
rover = veh.ChVehicleSystemNSC()
rover.SetChassisBody(veh.ChChassis(chrono.ChVectorD(0, 0.5, 0), 1000, veh.ChChassisBodyType.PRIMITIVE))
rover.SetChassisVisualizationType(veh.ChChassisBodyType.PRIMITIVE)
rover.SetChassisCollisionType(veh.ChChassisBodyType.PRIMITIVE)
rover.SetChassisMaterialSurface(chrono.ChMaterialSurfaceNSC())
rover.SetChassisMaterialSurfaceFriction(0.9)

# Add wheels to the rover
wheel_radius = 0.25
wheel_width = 0.15
wheel_mass = 20
wheel_inertia = chrono.ChVectorD(wheel_mass * wheel_radius**2 / 2, wheel_mass * wheel_radius**2 / 2, wheel_mass * wheel_width**2 / 12)
wheel_positions = [
    chrono.ChVectorD(0.5, wheel_radius, 0.5),
    chrono.ChVectorD(0.5, wheel_radius, -0.5),
    chrono.ChVectorD(-0.5, wheel_radius, 0.5),
    chrono.ChVectorD(-0.5, wheel_radius, -0.5)
]

for pos in wheel_positions:
    wheel = chrono.ChBodyEasyCylinder(wheel_radius, wheel_width, wheel_mass, True, True)
    wheel.SetPos(pos)
    wheel.SetInertiaXX(wheel_inertia)
    wheel.SetBodyFixed(False)
    system.Add(wheel)
    rover.AddWheelBody(wheel)

# Add motor control driver
rover.Init()
rover.SetSteeringInput(0)  # Initial steering input
rover.SetThrottleInput(1)  # Initial throttle input

# Add the rover to the system
system.Add(rover.GetChassisBody())

# Set up visualization
visualization = chronoirr.ChVisualSystemIrrlicht()
visualization.SetWindowSize(1280, 720)
visualization.SetWindowTitle('Curiosity Rover Simulation')
visualization.Initialize()
visualization.AddLogo()
visualization.AddSkyBox()
visualization.AddTypicalLights()
visualization.AddLightWithShadow(chrono.ChVectorD(1, 1, 1), chrono.ChVectorD(0, 0, 0), 1, 5)

# Attach the system to the visualization
visualization.AttachSystem(system)

# Set camera settings
camera = visualization.GetSceneManager().getActiveCamera()
camera.setPosition(chrono.ChVectorD(5, 5, 5))
camera.setTarget(chrono.ChVectorD(0, 0, 0))

# Run the simulation
step_size = 0.01
end_time = 10.0
current_time = 0.0

while current_time < end_time:
    visualization.Render()
    system.DoStepDynamics(step_size)
    current_time += step_size

# Clean up
visualization.Close()
```

### Explanation:
1. **Initialization**: The Chrono::Engine is initialized, and the path to Chrono data files is set.
2. **Ground**: A fixed ground plane is created with collision properties.
3. **Rover**: A simple rover model is created with a chassis and wheels. The wheels are positioned relative to the chassis.
4. **Motor Control**: Basic steering and throttle inputs are set for the rover.
5. **Visualization**: The Irrlicht visualization system is set up with customizable camera settings, lighting, shadows, and additional visual elements like textures and logos.
6. **Simulation Loop**: The simulation runs for a specified duration, updating the physics and rendering the visualization in each step.

### Notes:
- The path to Chrono data files should be set correctly.
- The rover model is simplified. For a more accurate model, you might need to load a detailed URDF or other model files.
