```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import pychrono.solvers as solvers
import pychrono.terrain as ter

# Initialize the Chrono::Engine
chrono.SetChronoDataPath('path/to/chrono/data')  # Set the path to Chrono data directory
chrono.ChSystemNSC my_system()
my_system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Create the HMMWV vehicle
vehicle = veh.ChVehicleSystemNSC()
vehicle.SetChassisBody(veh.ChChassisBodyEasyBox(1500, chrono.ChVectorD(1.5, 1, 0.5)))
vehicle.SetChassisVisualizationType(veh.ChVehicleVisualizationType::MESH)
vehicle.SetChassisMaterialSurface(chrono.ChMaterialSurfaceNSC())
vehicle.SetChassisCollisionType(veh.ChCollisionType::MESH)

# Add wheels to the HMMWV
wheels = []
for i in range(4):
    wheel = veh.ChWheelBodyEasyCylinder(0.4, 0.25)
    wheel.SetPos(chrono.ChVectorD(0, 0, 0.5 * i))
    wheels.append(wheel)
    vehicle.AddWheelBody(wheel)

# Initialize the driver system
driver = veh.ChDriver()
driver.Initialize(vehicle, veh.ChDriverInputSimple::STEERING_SINE, veh.ChDriverInputSimple::THROTTLE_FULL)
vehicle.SetDriver(driver)

# Create the SCM deformable terrain
terrain = ter.ChTerrainToolbox()
terrain.Initialize(my_system, ter.ChTerrain::Parameters())
terrain.CreateTerrain(ter.ChTerrain::Parameters())
terrain.SetHeightField(ter.ChTerrain::Parameters())
terrain.SetSoilParameters(ter.ChTerrain::Parameters())
terrain.SetHeightMap('path/to/heightmap.png')  # Path to your height map file

# Set up the visualization (Irrlicht)
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(my_system)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle('HMMWV on Deformable Terrain')
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(5, -5, 3), chrono.ChVectorD(0, 0, 1))
vis.AddLightWithShadow(chrono.ChVectorD(10, -10, 10), chrono.ChVectorD(0, 0, 0), 1, 0.2, 30, 50)

# Set up the simulation parameters
my_system.SetTimestepperType(chrono.ChTimestepper::Type::HHT)
my_system.SetSolverType(solvers.ChSolverMINRES())
my_system.SetSolverMaxIterations(100)
my_system.SetSolverTolerance(1e-10)

# Simulation loop
run_visualization = True
while vis.Run():
    time = my_system.GetChronoDataTime()
    my_system.DoStepDynamics(0.01)

    # Update the driver and vehicle
    driver.Synchronize(time)
    vehicle.Synchronize(time, my_system.GetSolver())

    # Update the visualization
    vis.Render()
    vis.Update()

    # Exit condition
    if time > 10:  # Run for 10 seconds
        run_visualization = False

# Clean up
vis.Close()
```

### Explanation:
1. **Initialize the PyChrono environment**:
   - Set the path to the Chrono data directory.
   - Create the `ChSystemNSC` object and set the gravity.

2. **Add the HMMWV vehicle**:
   - Create a chassis body and wheels.
   - Add wheels to the vehicle.
   - Initialize the driver system for vehicle control.

3. **Set up the SCM deformable terrain**:
   - Create a `ChTerrainToolbox` object.
   - Initialize the terrain with soil parameters and a height map.

4. **Set up the visualization**:
   - Create an `ChVisualSystemIrrlicht` object.
   - Attach the system to the visualization.
   - Set up the camera and lighting.

5. **Simulation loop**:
   - Run the simulation for a specified duration.
   - Synchronize and advance the vehicle, terrain, driver, and visual system at each timestep.
