```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import pychrono.geom as geom

# Initialize the PyChrono environment
chrono.SetChronoDataPath('../data/')  # Set the path to the Chrono data directory

# Create a Chrono::Engine physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# Create the ground body
ground = chrono.ChBodyEasyBox(10, 10, 0.5, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -0.25, 0))
system.Add(ground)

# Create the Viper rover
rover = veh.Viper()
rover.SetChassisBody(system, chrono.ChVectorD(0, 0.5, 0))  # Set initial position of the rover

# Initialize the driver for the Viper rover
driver = veh.ViperDriver(rover.GetVehicle())
driver.SetSteeringMode(veh.ViperDriver.STEERING_MODE_SPEED)

# Create an Irrlicht visualization
visualization = chronoirr.ChIrrApp(system, 'Viper Rover Simulation', chronoirr.dimension2du(800, 600))
visualization.AddTypicalSky()
visualization.AddTypicalLogo(chrono.GetChronoDataFile('logo_pychrono_red.png'))
visualization.AddLightWithShadow(chrono.ChVectorD(1.0, -1.0, 1.0), chrono.ChVectorD(0.2, 0.2, 0.2), 100, 100, 100, 40, 512)
camera = visualization.AddCamera(chrono.ChVectorD(5, -5, 5), chrono.ChVectorD(0, 0, 0))

# Set up the simulation loop
steering_change_duration = 10  # Time period over which steering changes
steering_change_start_time = 0  # Start time for steering change
steering_change_end_time = steering_change_start_time + steering_change_duration

def update_steering(time):
    if steering_change_start_time <= time <= steering_change_end_time:
        # Gradually change steering from -1 to 1 over the specified duration
        steering = -1 + 2 * (time - steering_change_start_time) / steering_change_duration
        driver.SetSteering(steering)

def simulation_step(time):
    system.DoStepDynamics(0.01)  # Time step for the simulation
    update_steering(time)
    visualization.Render()

# Run the simulation loop
time = 0
while visualization.Run():
    simulation_step(time)
    time += 0.01
```

   - The Viper rover is initialized and positioned in the system.
   - A driver is created to control the rover's steering.

3. **Visualization**:
   - An Irrlicht application is created for real-time visualization.
   - Camera settings and lighting are configured.
   - The rover is added to the visualization.

4. **Simulation Loop**:
   - The steering of the rover is gradually changed over a specified time period.
   - The simulation updates and renders the scene at each timestep.
