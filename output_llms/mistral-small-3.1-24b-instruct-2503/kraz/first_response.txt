```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh

# Step 1: Initialize the PyChrono environment and core components
chrono.SetChronoDataPath('path/to/chrono/data')  # Set the path to Chrono data

# Create the system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Step 2: Add the required physical systems and objects
# Create a rigid terrain
terrain = chrono.ChBodyEasyBox(20, 0.5, 20, 1000, True, True)
terrain.SetPos(chrono.ChVectorD(0, -0.25, 0))
terrain.GetMaterialSurface().SetFriction(0.9)
terrain.GetMaterialSurface().SetRestitution(0.2)
system.Add(terrain)

# Create the Kraz vehicle
vehicle = veh.KrazVehicle()
vehicle.SetChassisBody(system.AddBody(vehicle.GetChassisBody()))

# Set the initial position and orientation of the vehicle
vehicle.SetInitialPosition(chrono.ChVectorD(0, 1, 0))
vehicle.SetInitialOrientation(chrono.Q_from_AngAxis(chrono.ChMatrix33D(1)))

# Add the vehicle to the system
vehicle.Initialize(system)

# Step 3: Set necessary default parameters such as positions, forces, and interactions
# Initialize the driver system
driver = veh.ChDriver()
driver.Initialize(vehicle)
driver.SetSteeringInput(0)  # Initial steering input
driver.SetThrottleInput(0.5)  # Initial throttle input

# Step 4: Create a real-time visualization using Irrlicht with camera settings and lighting
# Create the Irrlicht application
application = chronoirr.ChIrrApp(system, 'Kraz Vehicle Simulation', chronoirr.dimension2du(800, 600))

# Add a skybox and set the camera position
application.AddSkyBox()
application.GetSceneManager().AddCameraSceneNodeMaySkip(True, chrono.ChVectorD(5, 5, 5), chrono.ChVectorD(0, 0, 0))

# Set lighting
application.AddLightDirectional()
application.AddLightPoint(chrono.ChVectorD(2, 2, 2), chrono.ChColorf(1, 1, 1), 100)

# Add the vehicle to the visualization
application.AssetBindAll()
application.AssetUpdateAll()

# Step 5: Implement a simulation loop
# Create a simulation manager
manager = chronoirr.ChIrrAppManager(system, application)

# Set the simulation step size
step_size = 0.01

# Run the simulation loop
while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()

    # Synchronize and advance the simulation
    system.DoStepDynamics(step_size)

    # Update the driver
    driver.Synchronize(step_size)
    driver.Advance(step_size)

    # Update the vehicle
    vehicle.Synchronize(step_size, driver)
    vehicle.Advance(step_size)

    # Update the visualization
    application.DoStep()

    application.EndScene()

print("Simulation completed.")
```

### Explanation:
1. **Initialization**: The script starts by setting the Chrono data path and creating the main Chrono system.
2. **Terrain**: A rigid terrain is created with specified friction and restitution properties.
3. **Vehicle**: A Kraz vehicle is initialized and added to the system with specified initial conditions.
4. **Driver**: A driver system is initialized to control the vehicle.
5. **Visualization**: Irrlicht is used for real-time visualization, including setting up the camera and lighting.
6. **Simulation Loop**: The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep to maintain real-time execution.

Make sure to replace `'path/to/chrono/data'` with the actual path to your Chrono data directory.
