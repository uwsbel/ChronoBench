```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import pychrono.postprocess as postprocess
import pychrono.sensor as sensors
import pychrono.vehicle as veh

# Initialize the PyChrono environment
chrono.SetChronoDataPath('../data/')  # Path to Chrono data directory

# Create the simulation system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# Create the vehicle system
vehicle = veh.HMMWV()
vehicle.SetChassisBodyVisualizationType(veh.ChVisualizationType::MESH)
vehicle.SetInitPosition(chrono.ChVectorD(0, 1, 0), chrono.ChQuaternionD(1, 0, 0, 0))
vehicle.SetChassisBodyFixed(False)
vehicle.SetTireStepSize(0.01)

# Create the driver system
driver = veh.ChDriver()
driver.Initialize(vehicle.GetVehicle())
driver.SetSteeringController(veh.ChGeneric_2WSteering::STEERING_TYPE_PI)
driver.SetSteeringControllerGains(1.0, 0.1)
driver.SetThrottleController(veh.ChGenericPowerController::CONTROLLER_TYPE_PI)
driver.SetThrottleControllerGains(0.2, 0.1)

# Create the terrain system
terrain = veh.ChTerrain()
terrain.SetContactMaterial(vehicle.GetVehicle().GetSystem().GetContactMethod())
terrain.Initialize(chrono.ChVectorD(0, -0.5, 0), chrono.ChQuaternionD(1, 0, 0, 0))
terrain.SetPlane(chrono.ChVectorD(0, 1, 0), -0.5)
terrain.SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"), 200, 200)

# Add vehicle, driver, and terrain to the system
system.Add(vehicle.GetVehicle())
system.Add(driver.GetDriver())
system.Add(terrain.GetTerrain())

# Add IMU and GPS sensors
imu_sensor = sensors.ChImuSensor(vehicle.GetVehicle().GetChassisBody())
gps_sensor = sensors.ChGpsSensor(vehicle.GetVehicle().GetChassisBody())

# Add sensors to the system
system.Add(imu_sensor)
system.Add(gps_sensor)

# Create the Irrlicht visual interface
vis = chronoirr.ChIrrApp(system, 'HMMWV Simulation', chronoirr.dimension2du(800, 600))
vis.AddTypicalSky()
vis.AddTypicalLogo(chrono.GetChronoDataFile('logo_pychrono_small.png'))
vis.AddLightWithShadow(chrono.ChVectorD(1.0, -1.0, 1.0), chrono.ChVectorD(0.7, 0.7, 0.9), 1000, 10, 10, 30, 50)
vis.EnableRTshadows(True)
vis.AddLight(chrono.ChVectorD(0.5, 0.5, 1), chrono.ChColorf(0.6, 0.6, 0.6), 0.5)
vis.AddCamera(chrono.ChVectorD(5, 5, 5), chrono.ChVectorD(0, 0, 0))

# Add vehicle to the visual interface
vis.AddVisualSystem(vehicle.GetVehicle())
vis.AddVisualSystem(terrain.GetTerrain())

# Run the simulation loop
step_size = 0.01
end_time = 10.0
time = 0.0

while time < end_time:
    time += step_size

    # Synchronize and advance the simulation
    system.DoStepDynamics(step_size)

    # Process and update sensor data
    imu_sensor.Synchronize()
    gps_sensor.Synchronize()

    # Output vehicle mass
    chassis_mass = vehicle.GetVehicle().GetChassisBody().GetMass()
    print(f'Time: {time:.2f} s, Vehicle Mass: {chassis_mass:.2f} kg')

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

# Close the visualization
vis.Close()
```
