```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import pychrono.fea as fea

# Initialize the Chrono::Engine
chrono.ChSystemNSC my_system()
chrono.ChCollisionModel.setDefaultSuggestedEnvelope(0.01)
chrono.ChCollisionModel.setDefaultSuggestedMargin(0.005)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(my_system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('UAZBUS Simulation')
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(5, -5, 3), chrono.ChVectorD(0, 0, 1))
vis.AddTypicalLights()

# Create the UAZBUS vehicle
vehicle = veh.ChVehicleUAZBUS()
vehicle.InitSystem(my_system, veh.ChVehicleUAZBUS.InitType.FULL)
vehicle.SetChassisVisualizationType(veh.ChVehicleVisualizationType.CH_VIS_BOTH)
vehicle.SetInitPosition(chrono.ChVectorD(0, 0, 1))
vehicle.SetInitOrientation(chrono.ChQuaternionD(1, 0, 0, 0))
vehicle.SetInitFwdVelocity(5)

# Create the rigid terrain
terrain = chrono.ChBodyEasyBox(20, 0.5, 20, 1000, True, True)
terrain.SetPos(chrono.ChVectorD(0, -0.25, 0))
terrain.GetMaterialSurface().SetFriction(0.9)
terrain.GetMaterialSurface().SetRestitution(0.2)
my_system.Add(terrain)

# Create the driver system
driver = veh.ChDriverVehicleUAZBUS()
driver.SetVehicle(vehicle)
driver.SetSteeringController(veh.ChFunction_PiecewiseLinear(0, 0.2, 1, 0.2, 1))
driver.SetThrottleController(veh.ChFunction_PiecewiseLinear(0, 1, 1, 1))
driver.SetBrakingController(veh.ChFunction_PiecewiseLinear(0, 0, 1, 0))

# Simulation parameters
step_size = 0.01
end_time = 10

# Simulation loop
my_system.SetTimestepperType(chrono.ChTimestepper.Type.HHH)
my_system.SetSolverType(chrono.ChSolver.Type.BARZILAIBORWEIN)
my_system.SetSolverMaxIterations(100)
my_system.SetSolverTolerance(1e-10)

# Run the simulation
start_time = chrono.ChTimer.GetTime()
current_time = 0

while current_time < end_time:
    time = my_system.GetChronoDataTime()
    my_system.DoStepDynamics(step_size)

    # Update the visualization
    vis.Render()

    # Synchronize the driver
    driver.Synchronize(time)

    current_time = chrono.ChTimer.GetTime() - start_time

# Close the visualization
vis.Close()
```

   - **Rigid Terrain**: A rigid terrain is created with defined friction and restitution properties.
   - **Driver System**: A driver system is set up to control the vehicle.
3. **Visualization**: The Irrlicht visualization system is initialized with camera settings and lighting.
4. **Simulation Loop**: The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.
