```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.sensor as sensors
import pychrono.postprocess as postprocess
import numpy as np

# Initialize the PyChrono environment
chrono.SetChronoDataPath('path_to_chrono_data')  # Set the path to Chrono data
system = chrono.ChSystemNSC()

# Create the visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('PyChrono Lidar Simulation')
vis.Initialize()
vis.AddLogo()
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(1, 1, 1), chrono.ChVectorD(0, 0, 0))

# Load the triangular mesh from a Wavefront .obj file
mesh_body = chrono.ChBodyEasyBox(1, 1, 1, 1000, True, True)
mesh_body.SetPos(chrono.ChVectorD(0, 0, 0))
mesh_body.SetFixed(True)

# Load the mesh from the .obj file
mesh_shape = chrono.ChTriangleMeshConnected()
mesh_shape.LoadWavefrontMesh('path_to_your_mesh.obj')
mesh_body.GetVisualShape(0).GetMesh().ImportMesh(mesh_shape)

# Add the mesh body to the system
system.Add(mesh_body)

# Create a sensor manager
sensor_manager = sensors.ChSensorManager(system)

# Create a lidar sensor
lidar = sensors.ChLidarSensor()
lidar.SetName("lidar_sensor")
lidar.SetPointCloudVizColor(chrono.ChColorf(1, 0, 0))  # Red color for lidar points
lidar.SetPointCloudVizType(sensors.ChLidarSensor.PointCloudVizType.POINTS)  # Visualize as points

# Add noise filters to the lidar sensor
lidar.SetNoiseFilterType(sensors.ChLidarSensor.NoiseFilterType.GAUSSIAN)
lidar.SetNoiseFilterParameters(0.01, 0.01, 0.01)  # Example parameters for Gaussian noise

# Add the lidar sensor to the sensor manager
sensor_manager.AddSensor(lidar)

# Attach the lidar sensor to the mesh body
lidar.SetPos(chrono.ChVectorD(1, 1, 1))  # Initial position of the lidar
mesh_body.AddChild(lidar)

# Set up the lidar data saving options
lidar.SetSaveData(True)
lidar.SetSaveDataFilename("lidar_data.csv")

# Simulation parameters
simulation_time = 10  # seconds
time_step = 0.01  # seconds

# Simulation loop
current_time = 0
while current_time < simulation_time:
    # Update the lidar position in an orbit around the mesh
    orbit_radius = 2
    orbit_angle = current_time * np.pi / 2  # Simple circular orbit
    lidar.SetPos(chrono.ChVectorD(orbit_radius * np.cos(orbit_angle),
                                   orbit_radius * np.sin(orbit_angle),
                                   1))

    # Run the simulation step
    system.DoStepDynamics(time_step)

    # Print lidar buffer data
    lidar_buffer = lidar.GetLastPointCloud()
    print(f"Lidar buffer data at time {current_time:.2f}: {lidar_buffer}")

    # Update the visualization
    vis.Run()

    # Increment the time
    current_time += time_step

# Finalize the visualization
vis.Close()
```

### Explanation:
1. **Initialization**:

   - A `ChSystemNSC` object is created to represent the physical system.
   - An Irrlicht visualization system is set up to visualize the simulation.

2. **Mesh Body**:
   - A triangular mesh is loaded from a Wavefront .obj file and added to the system as a fixed body.
   - The mesh is visualized using `ChTriangleMeshConnected`.

3. **Lidar Sensor**:
   - A `ChLidarSensor` is created and configured with noise filters, visualization options, and data saving settings.
   - The lidar sensor is attached to the mesh body and its position is dynamically updated in an orbit around the mesh.

4. **Simulation Loop**:
   - The simulation runs for a specified duration, updating the lidar's position and printing the lidar buffer data at each step.
   - The visualization is updated in each simulation step.

Make sure to replace `'path_to_chrono_data'` and `'path_to_your_mesh.obj'` with the actual paths to your Chrono data directory and the .obj file, respectively.
