```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.vehicle as veh
import pychrono.solvers as solvers

# Initialize the PyChrono environment
chrono.SetChronoDataPath('../data/')  # Set the path to the Chrono data directory

# Create a Chrono::Engine physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# Create the Irrlicht visualization system
visualization = chronoirr.ChIrrApp()
visualization.AddTypicalSky()
visualization.AddTypicalLogo(chrono.GetChronoDataFile('logo_pychrono_small.png'))
visualization.AddTypicalCamera(chrono.ChVectorD(0, 5, -3), chrono.ChVectorD(0, 0, 0))
visualization.AddTypicalLights()
visualization.SetWindowSize(1280, 1024)
visualization.SetWindowTitle('HMMWV on SCM Terrain')
visualization.Initialize()
visualization.AssetBind(system)

# Create the SCM terrain
terrain = veh.ChTerrainSCM()
terrain.Initialize(100, 100, 0.5, 0.5, chrono.ChVectorD(0, -0.5, 0))
terrain.SetContactMaterial(0.3, 0.01, 0.01, 0.01, 10, 10, 10, 10)
terrain.SetCustomSoilParameters(0.2, 100000, 100000, 100000, 100000, 0.01, 0.01, 0.01)
terrain.EnableFalseColorVisualization()
terrain.EnableMovingPatch(True, 2, 2, chrono.ChVectorD(0, 0, 0))

# Add the terrain to the system
system.Add(terrain)

# Create the HMMWV vehicle
vehicle = veh.ChVehicleHMMWV()
vehicle.Initialize()
vehicle.SetChassisBodyVisualizationType(veh.ChVehicleVisualizationType.MESH)
vehicle.SetChassisBodyMeshFilename(chrono.GetChronoDataFile('vehicle/hmmwv/chassis_chassis_body.obj'))
vehicle.SetInitPosition(chrono.ChVectorD(0, 0, 0.5))
vehicle.SetInitOrientation(chrono.Q_from_AngAxis(chrono.ChMatrix33D(1)))

# Add the vehicle to the system
system.Add(vehicle.GetVehicleSystem())

# Set up the driver system
driver = veh.ChDriverVehicle()
driver.SetVehicle(vehicle)
driver.SetSteeringController(veh.ChDriverSteeringController_PID())
driver.SetThrottleController(veh.ChDriverThrottleController_PID())
driver.SetBrakingController(veh.ChDriverBrakingController_PID())
driver.SetSteeringControllerParameters(0.1, 0, 0)
driver.SetThrottleControllerParameters(0.1, 0, 0)
driver.SetBrakingControllerParameters(0.1, 0, 0)
driver.SetSteeringInput(0)
driver.SetThrottleInput(0)
driver.SetBrakingInput(0)

# Set up the simulation parameters
system.SetSolverType(solvers.ChSolverMINRES())
system.SetMaxPenetrationRecoverySpeed(1.0)
system.SetTimestepperType(chrono.ChTimestepperEulerImplicit)
system.SetMaxIterationsSpeed(100)
system.SetMaxIterationsStab(100)

# Run the simulation in real-time
run_time = 10.0  # Total simulation time in seconds
step_size = 1.0 / 50.0  # Time step for 50 FPS
current_time = 0.0

while current_time < run_time:
    visualization.BeginScene()
    visualization.Render()

    system.DoStepDynamics(step_size)

    visualization.EndScene()
    current_time += step_size

# Close the Irrlicht visualization
visualization.Close()
```

### Explanation:
1. **Initialize the PyChrono environment**: Set the Chrono data path and create the physical system and visualization system.
2. **Create the SCM terrain**: Initialize the terrain with custom soil parameters, enable false color visualization, and moving patch feature.
3. **Create the HMMWV vehicle**: Initialize the vehicle with specific parameters, including position, orientation, and mesh visualization.
4. **Set up the driver system**: Configure steering, throttle, and braking controllers.
5. **Run the simulation**: Execute the simulation in real-time with a frame rate of 50 FPS.
