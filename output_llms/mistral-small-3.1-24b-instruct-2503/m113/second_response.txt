soil_cohesion = 5000   # cohesion in Pa
soil_friction = 30     # friction angle in degrees
soil_adhesion = 100    # adhesion in Pa

vehicle.Initialize()

vehicle.SetChassisVisualizationType(vis_type)
vehicle.SetSprocketVisualizationType(vis_type)
vehicle.SetIdlerVisualizationType(vis_type)

terrain.SetSoilMaterialProperties(soil_Young, soil_density, soil_cohesion, soil_friction, soil_adhesion, soil_penetration, soil_plastic_hardening, soil_plastic_exponent)
