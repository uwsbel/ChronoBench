```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import pychrono.fem as fem

# Initialize the PyChrono environment and core components
chrono.SetChronoDataPath('path/to/chrono/data')  # Set the path to Chrono data files

# Create the physical system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# Create a mesh with beam finite elements
mesh = fem.ChFemGeneralizedElementBeam()
mesh.SetName("beam_mesh")

# Define beam properties
beam_length = 1.0
beam_diameter = 0.05
beam_young_modulus = 210e9  # Young's modulus (steel)
beam_poisson_ratio = 0.3    # Poisson ratio (steel)
beam_density = 7800         # Density (steel)

# Create beam section
beam_section = fem.ChBeamSectionCircle()
beam_section.SetRadius(beam_diameter / 2)
beam_section.SetYoungModulus(beam_young_modulus)
beam_section.SetGantModulus(beam_young_modulus / (2.0 * (1.0 + beam_poisson_ratio)))
beam_section.SetDensity(beam_density)

# Add nodes to the beam mesh
node1 = fem.ChFemNodeBeam()
node1.SetName("node1")
node1.SetPos(chrono.ChVectorD(0, 0, 0))
node1.SetMass(0.1)  # Mass of the node
mesh.AddNode(node1)

node2 = fem.ChFemNodeBeam()
node2.SetName("node2")
node2.SetPos(chrono.ChVectorD(beam_length, 0, 0))
node2.SetMass(0.1)  # Mass of the node
mesh.AddNode(node2)

# Add beam element to the mesh
beam_element = fem.ChBeamElementBeam()
beam_element.SetSection(beam_section)
beam_element.SetNodes(node1, node2)
mesh.AddElement(beam_element)

# Add the mesh to the physical system
system.Add(mesh)

# Set up visualization
visualization = chronoirr.ChIrrApp()
visualization.AddSystem(system)
visualization.AddLogo(chrono.GetChronoDataFile('logo_pychrono_small.png'))
visualization.SetWindowSize(800, 600)
visualization.SetWindowTitle('PyChrono Beam Finite Element Simulation')
visualization.SetTryFullscreen(False)
visualization.Initialize()
visualization.AddTypicalSky()
visualization.AddTypicalLogo()
visualization.AddTypicalFloor()

# Set up the camera
camera = visualization.GetSceneManager().getActiveCamera()
camera.setPosition(chrono.ChVectorD(0.5, 0.5, 1.0))
camera.setTarget(chrono.ChVectorD(0.5, 0.0, 0.0))

# Run the simulation loop
while visualization.Run():
    visualization.BeginScene()
    visualization.Render()
    system.DoStepDynamics(0.01)  # Time step of 0.01 seconds
    visualization.EndScene()

# Clean up
visualization.Close()
```

   - Define beam properties such as length, diameter, Young's modulus, Poisson ratio, and density.
   - Create beam sections and nodes.
   - Add nodes and beam elements to the mesh.
   - Add the mesh to the physical system.

3. **Set necessary default parameters:**
   - Define positions and masses for the nodes.
   - Set up the visualization using the Irrlicht engine.
   - Configure the camera for viewing the simulation.

4. **Run the simulation loop:**
   - Use a while loop to run the simulation and update the visualization in real-time.
