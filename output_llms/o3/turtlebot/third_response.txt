```python
#
# Turtlebot on flat terrain – with Chrono::Sensor lidar
#

import os
import math
import random
import numpy as np
import pychrono as chrono
import pychrono.robot as turtlebot
import pychrono.sensor as sens                           # 2.  sensor module
from pychrono import irrlicht as chronoirr

# ---------------------------------------------------------------------
# 1.  Create Chrono system
# ---------------------------------------------------------------------
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# NOTE: correct Chrono call for setting gravity
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))

chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# ---------------------------------------------------------------------
# 2.  Build the ground
# ---------------------------------------------------------------------
ground_mat = chrono.ChContactMaterialNSC()
ground = chrono.ChBodyEasyBox(20, 20, 1,          # size X-Y-Z
                              1000,               # density
                              True, True, ground_mat)
ground.SetPos(chrono.ChVectorD(0, 0, -0.6))       # ❶ ground raised from -1 -> -0.6
ground.SetFixed(True)
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# ---------------------------------------------------------------------
# 3.  Build the Turtlebot
# ---------------------------------------------------------------------
init_pos = chrono.ChVectorD(0, 0.2, 0)                                  # position
init_rot = chrono.ChQuaternionD(1, 0, 0, 0)                             # orientation
robot    = turtlebot.TurtleBot(system, init_pos, init_rot)              # create
robot.Initialize()

# Handy wheel IDs (Chrono’s TurtleBot uses 0 = left, 1 = right)
LEFT_DRIVE_WHEEL  = 0
RIGHT_DRIVE_WHEEL = 1

# ---------------------------------------------------------------------
# 4.  Add a few random boxes for the lidar to “see”
# ---------------------------------------------------------------------
box_mat = chrono.ChContactMaterialNSC()
for i in range(5):                                                      # ❸ five boxes
    bx = chrono.ChBodyEasyBox(0.25, 0.25, 0.25, 1000, True, True, box_mat)
    x = random.uniform(-3.0, 3.0)
    y = random.uniform(-3.0, 3.0)
    bx.SetPos(chrono.ChVectorD(x, y, 0.125))                            # top at 0.25/2
    bx.SetMass(5.0)
    bx.SetCollide(True)
    bx.GetVisualShape(0).SetColor(chrono.ChColor(0.8, 0.1, 0.1))
    system.Add(bx)

# ---------------------------------------------------------------------
# 5.  Sensor manager and lidar sensor
# ---------------------------------------------------------------------
manager = sens.ChSensorManager(system)                                  # ❷ manager

# Simple point-light so we can see the lidar rays (not required)
manager.scene.AddPointLight(chrono.ChVectorF(100, 100, 100),
                            chrono.ChVectorF(1, 1, 1), 500)

# parent body for sensor (Turtlebot’s chassis)
parent_body = robot.GetBody()                                           # chassis body

# Pose of the lidar w.r.t. body                              (x,y,z, q)
lidar_offset = chrono.ChFrameD(chrono.ChVectorD(0, 0, 0.20),
                               chrono.ChQuaternionD(1, 0, 0, 0))

# Create lidar: 10 Hz, 32 × 720 samples, 90° × 360° FOV, 80 m range
lidar = sens.ChLidarSensor(parent_body,                   # parent
                           10.0,                          # update rate [Hz]
                           lidar_offset,                  # offset pose
                           32,                            # V samples
                           720,                           # H samples
                           math.radians(90),              # V-FOV
                           math.radians(360),             # H-FOV
                           80.0)                          # max range

# Filters: add noise and produce a point-cloud buffer
lidar.PushFilter(sens.ChFilterLidarNoise(0.00))
lidar.PushFilter(sens.ChFilterLidarXYZReturn())           # return XYZ list

manager.AddSensor(lidar)

# ---------------------------------------------------------------------
# 6.  Run-time visualisation
# ---------------------------------------------------------------------
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Turtlebot Robot – Lidar demo')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVectorD(1.5, -2.5, 5.5),
                       chrono.ChVectorD(0, 0, 0.5),
                       3, 4, 10, 40, 512)
vis.AddCamera(chrono.ChVectorD(0, 1.5, 0.7),
              chrono.ChVectorD(0, 0, 0.3))

vis.SetCameraVertical(chrono.CameraVerticalDir_Z)

# ---------------------------------------------------------------------
# 7.  Simple Turtlebot motion helper
# ---------------------------------------------------------------------
def move(mode='straight', speed=math.pi):
    """
    Differential drive helper:
        mode = 'straight'  -> both wheels same speed
        mode = 'left'      -> pivot left   (right wheel forward)
        mode = 'right'     -> pivot right  (left  wheel forward)
    """
    if mode == 'straight':
        robot.SetMotorSpeed(speed, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(speed, RIGHT_DRIVE_WHEEL)
    elif mode == 'left':
        robot.SetMotorSpeed(0.0,  LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(speed, RIGHT_DRIVE_WHEEL)
    elif mode == 'right':
        robot.SetMotorSpeed(speed, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0.0,  RIGHT_DRIVE_WHEEL)
    else:     # stop
        robot.SetMotorSpeed(0.0, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0.0, RIGHT_DRIVE_WHEEL)

# ---------------------------------------------------------------------
# 8.  Simulation loop
# ---------------------------------------------------------------------
time_step = 2e-3
time      = 0.0

while vis.Run():
    # --- motion command ------------------------------------------------
    move('straight')                       # ❹ keep moving straight

    # --- Rendering -----------------------------------------------------
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # --- Step both physics and sensors --------------------------------
    system.DoStepDynamics(time_step)
    manager.Update()                       # ❺ sensor manager update
    time += time_step
```
