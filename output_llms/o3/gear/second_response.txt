```python
###############################################################################
#  GEAR–TRAIN DEMO – corrected and updated to the requested specifications   #
#  Tested with PyChrono 7.x                                                   #
###############################################################################

import math
import pychrono as chrono
import pychrono.irrlicht as chronoirr

# --------------------------------------------------------------------------- #
# Helper aliases (shorter, readable)                                          #
# --------------------------------------------------------------------------- #
Vec3  = chrono.ChVectorD
Quat  = chrono.ChQuaternionD
Frame = chrono.ChFrameD
Csys  = chrono.ChCoordsysD
QUNIT = chrono.QUNIT          # unit quaternion (0-rotation)

# --------------------------------------------------------------------------- #
# Global parameters (MODIFIED where required)                                 #
# --------------------------------------------------------------------------- #
radA = 1.5          # radius of gear A  (was 2.0)
radB = 3.5          # radius of gear B  (was 4.0)
radC = 2 * radB + radA           # radius of internal gear C (truss)

density       = 1000.0           # generic density for helper “Easy” bodies
timestep      = 1.0e-3           # integration step

# --------------------------------------------------------------------------- #
# Create the physical system                                                  #
# --------------------------------------------------------------------------- #
sys = chrono.ChSystemNSC()

# One shared contact material
mat_surf = chrono.ChContactMaterialNSC()

# --------------------------------------------------------------------------- #
# Truss (fixed reference)  – size MODIFIED                                    #
# --------------------------------------------------------------------------- #
truss = chrono.ChBodyEasyBox(
            15, 8, 2,            # w, h, t  (was 20, 10, 2)
            density,
            True,                # visualisation ON
            False,               # collision OFF
            mat_surf)
truss.SetPos(Vec3(0, 0, 3))
truss.SetFixed(True)
sys.Add(truss)

# --------------------------------------------------------------------------- #
# A small coloured visual material                                            #
# --------------------------------------------------------------------------- #
vis_mat = chrono.ChVisualMaterial()
vis_mat.SetKdTexture(chrono.GetChronoDataFile('textures/pinkwhite.png'))

# --------------------------------------------------------------------------- #
# Rotating support bar (planet carrier)                                       #
# --------------------------------------------------------------------------- #
carrier = chrono.ChBodyEasyBox(8, 1.5, 1.0, density, True, False, mat_surf)
carrier.SetPos(Vec3(3, 0, 0))
sys.Add(carrier)

# Revolute joint: truss <--> carrier
rev_truss_carrier = chrono.ChLinkLockRevolute()
rev_truss_carrier.Initialize(truss, carrier, Csys(Vec3(0, 0, 0), QUNIT))
sys.AddLink(rev_truss_carrier)

# --------------------------------------------------------------------------- #
# Gear A (driver)                                                             #
# --------------------------------------------------------------------------- #
gearA = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, radA, 0.5,
                                  density, True, False, mat_surf)
gearA.SetPos(Vec3(0, 0, -1))
gearA.SetRot(chrono.Q_from_AngX(math.pi / 2))
gearA.GetVisualShape(0).SetMaterial(0, vis_mat)
sys.Add(gearA)

# Thin shaft purely for graphics – size MODIFIED
shaft_shape = chrono.ChVisualShapeCylinder(radA * 0.3, 10)   # (was 0.4 , 13)
gearA.AddVisualShape(shaft_shape,
                     Frame(Vec3(0,  3.5, 0),
                           chrono.Q_from_AngX(math.pi/2)))

# Motor imposing angular speed to gear A – speed MODIFIED
motor_A = chrono.ChLinkMotorRotationSpeed()
motor_A.Initialize(gearA, truss, Csys(Vec3(0, 0, 0), QUNIT))
motor_A.SetSpeedFunction(chrono.ChFunctionConst(3.0))         # (was 6 rad/s)
sys.AddLink(motor_A)

# --------------------------------------------------------------------------- #
# Gear B (planet)                                                             #
# --------------------------------------------------------------------------- #
interaxisAB = radA + radB
gearB = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, radB, 0.4,
                                  density, True, False, mat_surf)
gearB.SetPos(Vec3(interaxisAB, 0, -2))        # z-position MODIFIED (was -1)
gearB.SetRot(chrono.Q_from_AngX(math.pi / 2))
gearB.GetVisualShape(0).SetMaterial(0, vis_mat)
sys.Add(gearB)

# Revolute joint: carrier <--> gear B
rev_carrier_B = chrono.ChLinkLockRevolute()
rev_carrier_B.Initialize(gearB, carrier,
                         Csys(Vec3(interaxisAB, 0, 0), QUNIT))
sys.AddLink(rev_carrier_B)

# --------------------------------------------------------------------------- #
# Gear constraint A <-> B  (external mesh)                                    #
# --------------------------------------------------------------------------- #
gear_AB = chrono.ChLinkLockGear()
gear_AB.Initialize(gearA, gearB, Csys())                      # main reference
gear_AB.SetFrameShaft1(Frame(Vec3(0, 0, 0),
                             chrono.Q_from_AngX(-math.pi / 2)))
gear_AB.SetFrameShaft2(Frame(Vec3(0, 0, 0),
                             chrono.Q_from_AngX(-math.pi / 2)))
gear_AB.SetTransmissionRatio(radA / radB)   # speed_B = ratio * speed_A
gear_AB.SetEnforcePhase(True)
sys.AddLink(gear_AB)

# --------------------------------------------------------------------------- #
# Gear constraint B <-> C (internal / epicyclic)                              #
# here body “C” is the fixed truss                                            #
# --------------------------------------------------------------------------- #
gear_BC = chrono.ChLinkLockGear()
gear_BC.Initialize(gearB, truss, Csys())
gear_BC.SetFrameShaft1(Frame(Vec3(0, 0, 0),
                             chrono.Q_from_AngX(-math.pi / 2)))
gear_BC.SetFrameShaft2(Frame(Vec3(0, 0, -4), QUNIT))
gear_BC.SetTransmissionRatio(radB / radC)
gear_BC.SetEpicyclic(True)                   # internal (ring) gear
sys.AddLink(gear_BC)

# --------------------------------------------------------------------------- #
# Visualisation with Irrlicht                                                 #
# --------------------------------------------------------------------------- #
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Gears and Pulleys – corrected demo')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(Vec3(12, 15, -20))
vis.AddTypicalLights()

# --------------------------------------------------------------------------- #
# Integrator                                                                  #
# --------------------------------------------------------------------------- #
sys.SetTimestepperType(chrono.ChTimestepper.Type_EULER_IMPLICIT_PROJECTED)

# --------------------------------------------------------------------------- #
# Simulation loop                                                             #
# --------------------------------------------------------------------------- #
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(timestep)
```
