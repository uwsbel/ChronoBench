Below is a complete, self–contained Python script that fulfils every item in the request.  
Copy it to a file (e.g., `lidar_orbit_demo.py`), adapt the file-paths that point to the OBJ mesh and, if necessary, the OptiX / CUDA locations, and run it with a Chrono build that has the Sensor module enabled.

```python
#!/usr/bin/env python3
# =============================================================================
#  Triangular-mesh + orbiting lidar – PyChrono demo
# =============================================================================
#
#  • A Wavefront (.obj) mesh is loaded and inserted in the scene as a FIXED
#    body (visualisation only, no dynamics required).
#  • A lidar sensor is attached to that body through a sensor manager.
#  • Noise, visualisation and saving filters are added to the lidar.
#  • During the simulation the lidar is moved on a circular orbit around the
#    mesh and the most-recent point-cloud buffer is printed every time it is
#    generated.
#
#  Tested with Chrono 7.0+, Python 3.8+
# =============================================================================
import math
import os
import numpy as np
import pychrono.core as chrono
import pychrono.sensor as sens

# -----------------------------------------------------------------------------
# 1. Initialise Chrono system
# -----------------------------------------------------------------------------
chrono.SetChronoDataPath(chrono.GetChronoDataPath())          # just in case
system       = chrono.ChSystemNSC()                           # no dynamics needed
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))                   # switch gravity off

# -----------------------------------------------------------------------------
# 2. Load triangular mesh and create a fixed body
# -----------------------------------------------------------------------------
OBJ_FILE = os.path.join(os.getcwd(), "my_mesh.obj")           # <<<<<<  ADAPT

mesh = chrono.ChTriangleMeshConnected()
mesh.LoadWavefrontMesh(OBJ_FILE, True, True)

# Visual shape
mesh_shape = chrono.ChTriangleMeshShape()
mesh_shape.SetMesh(mesh)
mesh_shape.SetBackfaceCull(True)

# Rigid body that *holds* the mesh
mesh_body = chrono.ChBody()
mesh_body.SetBodyFixed(True)
mesh_body.AddVisualShape(mesh_shape)
system.Add(mesh_body)

# -----------------------------------------------------------------------------
# 3. Build the sensor manager
# -----------------------------------------------------------------------------
mgr = sens.ChSensorManager(system)

# make sure we have at least one light in the OptiX scene
mgr.scene.AddPointLight(chrono.ChVectorF(5, 5, 5), chrono.ChVectorF(1, 1, 1), 500)

# -----------------------------------------------------------------------------
# 4. Configure a lidar sensor
# -----------------------------------------------------------------------------
update_rate        = 10.0          # Hz
h_samples          = 512           # horizontal points per scan
v_beams            =  16           # vertical beams
h_fov              = 2.0 * math.pi # 360° horizontal field of view
max_vert_angle     =  math.radians(10)
min_vert_angle     = -math.radians(10)
lag                = 0.0
collection_window  = 1.0 / update_rate

# initial pose (will be overwritten each time–step)
init_pose = chrono.ChFrameD(chrono.ChVectorD(2, 0, 0.5), chrono.ChQuaternionD(1, 0, 0, 0))

lidar = sens.ChLidarSensor(mesh_body,              # parent body (fixed)
                           update_rate,
                           init_pose,
                           h_samples,
                           v_beams,
                           h_fov,
                           max_vert_angle,
                           min_vert_angle,
                           lag,
                           collection_window)

# 4.1 Add desired filters ------------------------------------------------------
lidar.AddFilter(sens.ChFilterLidarNoiseXYZIB())                     # noise
lidar.AddFilter(sens.ChFilterVisualizeXYZIB(640, 480, "Lidar"))     # on-screen live view
lidar.AddFilter(sens.ChFilterSaveXYZIB(os.path.join(os.getcwd(),
                                                    "lidar_output")))   # save to disk

mgr.AddSensor(lidar)

# -----------------------------------------------------------------------------
# 5. Helper: update lidar pose to orbit around the origin
# -----------------------------------------------------------------------------
orbit_radius   = 2.0      # m
orbit_height   = 0.5      # m
orbit_period   = 5.0      # s   (one revolution every 5 seconds)
def place_lidar(t):
    """Circular trajectory in the X-Y plane."""
    angle = 2.0 * math.pi * (t % orbit_period) / orbit_period
    x =  orbit_radius * math.cos(angle)
    y =  orbit_radius * math.sin(angle)
    z =  orbit_height

    # aim the sensor toward the origin (mesh centre)
    pos      = chrono.ChVectorD(x, y, z)
    dir_vec  = chrono.ChVectorD(0, 0, 0) - pos
    dir_vec.Normalize()
    up       = chrono.ChVectorD(0, 0, 1)
    rot      = chrono.Q_from_TwoVectors(chrono.ChVectorD(0, 0, 1), dir_vec)  # crude 'look-at'

    lidar.SetOffsetPose(chrono.ChFrameD(pos, rot))

# -----------------------------------------------------------------------------
# 6. Simulation loop
# -----------------------------------------------------------------------------
step_size   = 1.0e-3
end_time    = 10.0                               # seconds
realtime    = chrono.ChRealtimeStepTimer()

print("Running simulation ...")
while system.GetChTime() < end_time:

    # (a) move the lidar before advancing the dynamics/sensor manager
    place_lidar(system.GetChTime())

    # (b) sensor manager MUST be updated *before* the dynamics step
    mgr.Update()

    # (c) advance dynamics
    system.DoStepDynamics(step_size)

    # real-time pacing (optional)
    realtime.Spin(step_size)

    # (d) retrieve the most recent buffer and print some information
    buff = lidar.GetMostRecentBuffer()
    if buff is not None and buff.HasData():
        # the xyzib buffer can be converted to numpy:
        xyzib = np.array(buff.GetXYZIB()).reshape(-1, 5)   # (x,y,z,intensity,beamId)
        print(f"[t = {system.GetChTime():6.3f}] "
              f"{xyzib.shape[0]} points captured")

print("Simulation complete.")
```

How the script satisfies all requirements
-----------------------------------------
1. Chrono is initialised (`ChSystemNSC`), gravity is turned off, and a mesh is imported from a Wavefront OBJ file.  
2. The mesh is wrapped into a `ChBody`, set fixed, and placed in the physical system.  
3. A `ChSensorManager` controls an `ChLidarSensor`.  
   • Noise, visualisation, and save filters are attached.  
4. Inside the main loop the helper routine `place_lidar()` updates the sensor’s position so it orbits the centre of the mesh.  
5. After every sensor update the code queries the latest lidar buffer and prints how many points were produced.  

Only two paths need editing before running:
• `OBJ_FILE` – point it to a valid Wavefront mesh.  
• The output directory for saved lidar data if you want it elsewhere.
