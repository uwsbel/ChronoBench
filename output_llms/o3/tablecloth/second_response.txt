```python
# =============================================================================
#  PYCHRONO – FEA SHELLS (BST) EXAMPLE
#  Corrected & extended in accordance with the requested instructions
# =============================================================================
#
#  Main additions / fixes:
#    • Added node monitoring & loading infrastructure
#    • Robust boundary–node creation with explicit edge-checks
#    • Element/Node monitor assignment
#    • Upper–belt nodes are fixed
#    • Visual shapes adjusted
#    • Pardiso-MKL sparsity pattern locked
#    • Timestep increased to 0.005 s
#
# =============================================================================

import os, errno
import pychrono.core       as chrono
import pychrono.irrlicht   as chronoirr
import pychrono.fea        as fea
import pychrono.pardisomkl as mkl

# -------------------------------------------------------------------------
#  Output folder
# -------------------------------------------------------------------------
out_dir = chrono.GetChronoOutputPath() + "FEA_SHELLS_BST"
try:
    os.mkdir(out_dir)
except OSError as exc:
    if exc.errno != errno.EEXIST:
        print("Error creating output directory", exc)

# -------------------------------------------------------------------------
#  Chrono physical system
# -------------------------------------------------------------------------
sys  = chrono.ChSystemSMC()

# -------------------------------------------------------------------------
#  FEA mesh and material
# -------------------------------------------------------------------------
mesh = fea.ChMesh()
sys.Add(mesh)

# material data -----------------------------------------------------------
density   = 100.0           # kg/m^3
E         = 6.0e4           # Young modulus (Pa)
nu        = 0.0             # Poisson ratio
thickness = 0.01            # shell thickness (m)

melasticity = fea.ChElasticityKirchhoffIsothropic(E, nu)
material    = fea.ChMaterialShellKirchhoff(melasticity)
material.SetDensity(density)

# -------------------------------------------------------------------------
#  Geometry – regular plate
# -------------------------------------------------------------------------
L_x = 1.0        # length in x  (m)
L_z = 1.0        # length in z  (m)

nsections_x = 40
nsections_z = 40

# -------------------------------------------------------------------------
#  Node containers for later reference
# -------------------------------------------------------------------------
mynodes          = []    # all nodes
nodePlotA        = None  # requested plotting node A
nodePlotB        = None  # requested plotting node B
nodesLoad        = []    # nodes on which external loads could be applied
mnodemonitor     = None  # specific node to monitor during simulation
melementmonitor  = None  # specific element to monitor during simulation

# -------------------------------------------------------------------------
#  Build nodes
# -------------------------------------------------------------------------
for iz in range(nsections_z + 1):
    for ix in range(nsections_x + 1):
        p = chrono.ChVector3d(ix * (L_x / nsections_x), 0.0,
                              iz * (L_z / nsections_z))
        mnode = fea.ChNodeFEAxyz(p)
        mesh.AddNode(mnode)
        mynodes.append(mnode)

        # Select some representative nodes for the user-requested variables
        if ix == 0 and iz == 0:
            nodePlotA = mnode
        if ix == nsections_x and iz == nsections_z:
            nodePlotB = mnode

# Keep a list of nodes that will receive an external load
nodesLoad.append(nodePlotB)   # (example: load at the far corner)

# -------------------------------------------------------------------------
#  Build shell BST elements
# -------------------------------------------------------------------------
for iz in range(nsections_z):
    for ix in range(nsections_x):

        # Utility lambda to access node indices safely
        def node_at(ix_, iz_):
            return mynodes[iz_ * (nsections_x + 1) + ix_]

        # ------------- first triangle (melementA) -------------------------
        melementA = fea.ChElementShellBST()

        n0 = node_at(ix,     iz)
        n1 = node_at(ix + 1, iz)
        n2 = node_at(ix,     iz + 1)

        # boundary nodes with edge checks
        boundary_1 = node_at(ix + 1, iz + 1)                        # always valid
        boundary_2 = node_at(ix - 1, iz + 1) if ix > 0       else None
        boundary_3 = node_at(ix + 1, iz - 1) if iz > 0       else None

        melementA.SetNodes(n0, n1, n2, boundary_1, boundary_2, boundary_3)
        melementA.AddLayer(thickness, 0.0, material)
        mesh.AddElement(melementA)

        # Assign element monitor on the second cell of first row
        if iz == 0 and ix == 1:
            melementmonitor = melementA

        # ------------- second triangle (melementB) ------------------------
        melementB = fea.ChElementShellBST()

        n3 = node_at(ix + 1, iz + 1)
        n4 = node_at(ix,     iz + 1)
        n5 = node_at(ix + 1, iz)

        boundary_1B = node_at(ix,     iz)                   # always valid
        boundary_2B = node_at(ix + 2, iz) if ix < nsections_x - 1 else None
        boundary_3B = node_at(ix,     iz + 2) if iz < nsections_z - 1 else None

        melementB.SetNodes(n3, n4, n5, boundary_1B, boundary_2B, boundary_3B)
        melementB.AddLayer(thickness, 0.0, material)
        mesh.AddElement(melementB)

# -------------------------------------------------------------------------
#  Choose a node to monitor (example: centre node)
# -------------------------------------------------------------------------
mid_ix = nsections_x // 2
mid_iz = nsections_z // 2
mnodemonitor = mynodes[mid_iz * (nsections_x + 1) + mid_ix]

# -------------------------------------------------------------------------
#  Fix upper belt of nodes (first 30×30 in the grid)
# -------------------------------------------------------------------------
for j in range(30):
    for k in range(30):
        mynodes[j * (nsections_x + 1) + k].SetFixed(True)

# -------------------------------------------------------------------------
#  LOAD definition (can be expanded if needed)
# -------------------------------------------------------------------------
load_force = chrono.ChVector3d(0.0, -100.0, 0.0)     # example vertical load
for n in nodesLoad:
    load = fea.ChLoadXYZnode(n, load_force)
    mesh.AddLoad(load)

# -------------------------------------------------------------------------
#  “Reference” interpolation functions to be filled during runtime
# -------------------------------------------------------------------------
ref_X = chrono.ChFunction_Recorder()
ref_Y = chrono.ChFunction_Recorder()

# -------------------------------------------------------------------------
#  Visualisation helpers
# -------------------------------------------------------------------------
mvisualizeshellA = chrono.ChVisualShapeFEA(mesh)
mvisualizeshellA.SetShellResolution(2)
mvisualizeshellA.SetSmoothFaces(True)
# mvisualizeshellA.SetWireframe(True)        # optional – keep commented
# mvisualizeshellA.SetBackfaceCull(True)     # optional – keep commented
mesh.AddVisualShapeFEA(mvisualizeshellA)

mvisualizeshellB = chrono.ChVisualShapeFEA(mesh)
mvisualizeshellB.SetFEMglyphType(chrono.ChVisualShapeFEA.GlyphType_NODE_DOT_POS)
mvisualizeshellB.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_NONE)
mvisualizeshellB.SetSymbolsThickness(0.006)
mesh.AddVisualShapeFEA(mvisualizeshellB)

# -------------------------------------------------------------------------
#  Irrlicht visual system
# -------------------------------------------------------------------------
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle("Shells FEA test: triangle BST elements")
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(1.0, 0.3, 1.3),
              chrono.ChVector3d(0.5, -0.3, 0.5))
vis.AddTypicalLights()

# -------------------------------------------------------------------------
#  Solver – Pardiso MKL
# -------------------------------------------------------------------------
mkl_solver = mkl.ChSolverPardisoMKL()
mkl_solver.LockSparsityPattern(True)     # was False – now locked
sys.SetSolver(mkl_solver)

# -------------------------------------------------------------------------
#  Final setup
# -------------------------------------------------------------------------
sys.Setup()
sys.Update()

# -------------------------------------------------------------------------
#  Time stepping
# -------------------------------------------------------------------------
timestep = 0.005     # increased from 0.001

# -------------------------------------------------------------------------
#  Simulation loop
# -------------------------------------------------------------------------
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # (Optional) – record monitor data
    ref_X.AddPoint(sys.GetChTime(), mnodemonitor.GetPos().x)
    ref_Y.AddPoint(sys.GetChTime(), mnodemonitor.GetPos().y)

    sys.DoStepDynamics(timestep)
```

The script now fulfils every requirement:

1. Node monitoring, load vectors, and interpolation functions are declared.  
2. Boundary nodes are created with explicit edge checks to avoid out-of-range access.  
3. A dedicated element (`melementmonitor`) is chosen when `iz == 0 and ix == 1`.  
4. The top 30 × 30 nodes are fixed.  
5. Visual shapes receive the requested configuration (with optional toggles kept in comments).  
6. The Pardiso-MKL solver locks its sparsity pattern.  
7. The simulation time step is adjusted to `0.005 s`.
