```python
# ======================================================================
#  Crank–slider mechanism: version with spherical joints on the rod and
#  a planar (plane–plane) joint that constrains the piston to stay in
#  the x–y plane.  Wrong method / class names that existed in the
#  original file have been fixed as well.
# ======================================================================

import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import matplotlib.pyplot as plt
import numpy as np

# ----------------------------------------------------------------------
# Create the physical system
# ----------------------------------------------------------------------
sys = chrono.ChSystemNSC()

# ----------------------------------------------------------------------
# Some geometric data – shared by several bodies / joints
# ----------------------------------------------------------------------
crank_center = chrono.ChVectorD(-1, 0.5, 0)
crank_rad    = 0.4
crank_thick  = 0.1
rod_length   = 1.5

# ----------------------------------------------------------------------
# Create four bodies: ground, crank, rod, piston
# ----------------------------------------------------------------------

# 1) Ground (fixed) – just a box used as a truss
ground = chrono.ChBodyEasyBox(3, 1, 3, 1000)
ground.SetPos(chrono.ChVectorD(0, -0.5, 0))
ground.SetFixed(True)
sys.Add(ground)

# 2) Crank – a thin cylinder, its axis must be along Z
crank = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, crank_rad, crank_thick, 1000)
crank.SetPos(crank_center + chrono.ChVectorD(0, 0, -0.1))
# ChBodyEasyCylinder is created with its axis parallel to +Y; rotate it to +Z
crank.SetRot(chrono.Q_ROTATE_Y_TO_Z)
sys.Add(crank)

# 3) Connecting rod – slender box
rod = chrono.ChBodyEasyBox(rod_length, 0.1, 0.1, 1000)
rod.SetPos(crank_center + chrono.ChVectorD(crank_rad + rod_length * 0.5, 0, 0))
sys.Add(rod)

# 4) Piston – short cylinder, axis along +X (motion direction)
piston = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, 0.2, 0.3, 1000)
piston.SetPos(crank_center + chrono.ChVectorD(crank_rad + rod_length, 0, 0))
piston.SetRot(chrono.Q_ROTATE_Y_TO_X)
sys.Add(piston)

# ----------------------------------------------------------------------
# Joints and actuators
# ----------------------------------------------------------------------

# Motor between ground and crank (rotation-speed type)
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(crank,                     # driven body
                 ground,                    # support
                 chrono.ChFrameD(crank_center))  # motor location (abs.)
motor_speed = chrono.ChFunction_Const(chrono.CH_PI)   # 180 deg/s
motor.SetSpeedFunction(motor_speed)
sys.Add(motor)

# --- MODIFIED   -------------------------------------------------------
# 1) CRANK <-> ROD : spherical (ball-and-socket) joint
joint_crank_rod = chrono.ChLinkLockSpherical()
joint_crank_rod.Initialize(rod,
                           crank,
                           chrono.ChCoordsysD(crank_center + chrono.ChVectorD(crank_rad, 0, 0)))
sys.Add(joint_crank_rod)

# 2) ROD <-> PISTON : spherical (ball-and-socket) joint
joint_rod_piston = chrono.ChLinkLockSpherical()
joint_rod_piston.Initialize(piston,
                            rod,
                            chrono.ChCoordsysD(crank_center + chrono.ChVectorD(crank_rad + rod_length, 0, 0)))
sys.Add(joint_rod_piston)

# 3) PISTON <-> GROUND : plane-plane (planar) joint – allows translation
#    in x-y and rotation about z (piston constrained to the x-y plane)
joint_plane = chrono.ChLinkLockPlanePlane()
plane_frame = chrono.ChCoordsysD(crank_center + chrono.ChVectorD(crank_rad + rod_length, 0, 0),
                                 chrono.QUNIT)              # plane normal = +Z
joint_plane.Initialize(piston, ground, plane_frame)
sys.Add(joint_plane)

# ----------------------------------------------------------------------
# Irrlicht visualisation
# ----------------------------------------------------------------------
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Crank-slider – spherical & planar joints')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(1, 1, 3), chrono.ChVectorD(0, 1, 0))
vis.AddTypicalLights()

# ----------------------------------------------------------------------
# Data collection for post-processing
# ----------------------------------------------------------------------
time_data   = []
angle_data  = []
pos_data    = []
speed_data  = []

# ----------------------------------------------------------------------
# Simulation loop
# ----------------------------------------------------------------------
step = 1e-3
while vis.Run():
    # store current values
    time_data.append(sys.GetChTime())
    angle_data.append(motor.GetMotorRot())        # actual crank angle
    pos_data.append(piston.GetPos().x)
    speed_data.append(piston.GetPos_dt().x)

    # render
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    sys.DoStepDynamics(step)

    # stop after 20 s of simulated time
    if sys.GetChTime() > 20:
        vis.GetDevice().closeDevice()

# ----------------------------------------------------------------------
# Matplotlib plots
# ----------------------------------------------------------------------
fig, (ax1, ax2) = plt.subplots(2, sharex=True)

ax1.plot(angle_data, pos_data)
ax1.set(ylabel='piston position [m]')
ax1.grid()

ax2.plot(angle_data, speed_data, 'r--')
ax2.set(ylabel='piston speed [m/s]', xlabel='crank angle [rad]')
ax2.grid()

# Put π ticks on the x-axis
plt.xticks(np.linspace(0, 2 * np.pi, 5),
           ['0', r'$\pi/2$', r'$\pi$', r'$3\pi/2$', r'$2\pi$'])

plt.show()
```

Explanation of main corrections and requested changes
1. Fixed wrong Chrono class / method names  
   • `ChFunctionConst` → `ChFunction_Const`  
   • `SetMotorFunction()` → `SetSpeedFunction()`  
   • `GetMotorAngle()`   → `GetMotorRot()`  
   • `ChFramed`          → `ChFrameD` / `ChCoordsysD`

2. Replaced joints as required  
   • `ChLinkLockRevolute` → `ChLinkLockSpherical` (crank-rod and rod-piston)  
   • `ChLinkLockPrismatic`→ `ChLinkLockPlanePlane` (piston-ground)

3. The new planar joint constrains the piston to stay in the global x-y
   plane, leaving two translations (x, y) and one rotation (about z).

The rest of the script is identical to the original, so the simulation,
visualisation, and plotting behaviour remain unchanged except for the
new joint kinematics.
