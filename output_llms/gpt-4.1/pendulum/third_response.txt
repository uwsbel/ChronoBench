Here is the **corrected and modified PyChrono script** for a **double pendulum** system, with all errors fixed and improvements for clarity and correctness.

---

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import math

# Initialize the Chrono simulation system
sys = chrono.ChSystemNSC()

# Set the gravitational acceleration for the system (in m/s^2)
sys.Set_G_acc(chrono.ChVector3d(0, -9.81, 0))  # Correct method name

# Create the ground body and add it to the simulation
ground = chrono.ChBody()
ground.SetBodyFixed(True)  # Correct method name
ground.SetCollide(False)   # Correct method name
sys.Add(ground)

# Add a visualization cylinder to the ground (vertical post)
cyl_ground = chrono.ChVisualShapeCylinder(0.2, 0.4)
cyl_ground.SetColor(chrono.ChColor(0.4, 0.4, 0.4))
ground.AddVisualShape(cyl_ground, chrono.ChFrameD(chrono.ChVector3d(0, 0, +1)))

# --- First Pendulum (pend_1) ---
pend_1 = chrono.ChBody()
pend_1.SetBodyFixed(False)
pend_1.SetCollide(False)
pend_1.SetMass(1)
pend_1.SetInertiaXX(chrono.ChVector3d(0.2, 1, 1))
sys.Add(pend_1)

# Add a visualization cylinder to the first pendulum (length 2, radius 0.2)
cyl_pend1 = chrono.ChVisualShapeCylinder(0.2, 2)
cyl_pend1.SetColor(chrono.ChColor(0.6, 0, 0))
# The cylinder is centered at the body's origin, so shift it so that one end is at the joint
pend_1.AddVisualShape(cyl_pend1, chrono.ChFrameD(chrono.ChVector3d(0, -1, 0), chrono.Q_from_AngAxis(math.pi/2, chrono.ChVector3d(1,0,0))))

# Set the initial position of the first pendulum (pivot at (0,0,1), length 2, horizontal)
pend_1.SetPos(chrono.ChVector3d(1, 0, 1))

# --- Second Pendulum (pend_2) ---
pend_2 = chrono.ChBody()
pend_2.SetBodyFixed(False)
pend_2.SetCollide(False)
pend_2.SetMass(1)
pend_2.SetInertiaXX(chrono.ChVector3d(0.2, 1, 1))
sys.Add(pend_2)

# Add a visualization cylinder to the second pendulum (length 2, radius 0.2)
cyl_pend2 = chrono.ChVisualShapeCylinder(0.2, 2)
cyl_pend2.SetColor(chrono.ChColor(0, 0, 0.6))
pend_2.AddVisualShape(cyl_pend2, chrono.ChFrameD(chrono.ChVector3d(0, -1, 0), chrono.Q_from_AngAxis(math.pi/2, chrono.ChVector3d(1,0,0))))

# Set the initial position of the second pendulum (attached to end of first pendulum)
# First pendulum's pivot: (0,0,1), length 2, horizontal (along +X)
# So its end is at (2,0,1)
pend_2.SetPos(chrono.ChVector3d(3, 0, 1))

# --- Joints ---

# Revolute joint between ground and first pendulum at (0,0,1), axis Z
rev_1 = chrono.ChLinkLockRevolute()
rev_1.Initialize(ground, pend_1, chrono.ChFrameD(chrono.ChVector3d(0, 0, 1)))
sys.AddLink(rev_1)

# Revolute joint between first and second pendulum at (2,0,1), axis Z
rev_2 = chrono.ChLinkLockRevolute()
rev_2.Initialize(pend_1, pend_2, chrono.ChFrameD(chrono.ChVector3d(2, 0, 1)))
sys.AddLink(rev_2)

# --- Visualization ---
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Double Pendulum Demo')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 3, 8))
vis.AddTypicalLights()

# --- Simulation loop ---
log_info = True
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(1e-3)

    # Log information after 1 second of simulation time
    if log_info and sys.GetChTime() > 1:
        pos_1 = pend_1.GetPos()
        pos_2 = pend_2.GetPos()
        print("t = ", sys.GetChTime())
        print("Pendulum 1: ", pos_1.x, pos_1.y, pos_1.z)
        print("Pendulum 2: ", pos_2.x, pos_2.y, pos_2.z)
        lin_vel_1 = pend_1.GetPos_dt()
        lin_vel_2 = pend_2.GetPos_dt()
        print("Pendulum 1 velocity: ", lin_vel_1.x, lin_vel_1.y, lin_vel_1.z)
        print("Pendulum 2 velocity: ", lin_vel_2.x, lin_vel_2.y, lin_vel_2.z)
        log_info = False
```

**Key corrections and modifications:**

- Fixed method names:  
  - `SetGravitationalAcceleration` → `Set_G_acc`  
  - `SetFixed` → `SetBodyFixed`  
  - `EnableCollision` → `SetCollide`  
  - `AddBody` → `Add`  
  - `ChFramed` → `ChFrameD`  
  - `ChQuaterniond` → `chrono.QUNIT` or use `Q_from_AngAxis` for rotations  
  - `GetPosDt` → `GetPos_dt`  
- Added the second pendulum (`pend_2`) and its visualization.
- Added the second revolute joint (`rev_2`) connecting `pend_1` and `pend_2`.
- Set initial positions so the pendulums are horizontal and connected.
- Improved visualization: colors, camera position, and cylinder orientation.
- Improved logging to show both pendulums' positions and velocities.
- Ensured all objects are added to the system before simulation.

**This script will simulate a double pendulum, with both pendulums able to move independently, and will visualize the system using Irrlicht.**
