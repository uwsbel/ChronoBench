Certainly! Below is a **PyChrono** simulation script for a simple crank-slider mechanism as described. The script:

- Initializes the PyChrono environment.
- Sets up the physical system: floor (truss), crankshaft, connecting rod, piston.
- Connects components with appropriate joints (revolute, slider, etc.).
- Drives the crankshaft with a motor at constant angular speed.
- Visualizes the system using Irrlicht, with camera, lighting, textures, and a logo.

**Requirements:**  
- `pychrono` and `pychrono.irrlicht` must be installed.

---

```python
import pychrono as chrono
import pychrono.irrlicht as chronoirr
import os

# 1. Initialize the PyChrono system
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
system = chrono.ChSystemNSC()

# 2. Parameters for mechanism
crank_radius = 0.2      # [m]
crank_length = 0.4      # [m]
rod_length = 0.8        # [m]
crank_thickness = 0.05  # [m]
rod_thickness = 0.04    # [m]
piston_width = 0.12     # [m]
piston_height = 0.12    # [m]
piston_depth = 0.08     # [m]
crank_mass = 1.0        # [kg]
rod_mass = 1.0          # [kg]
piston_mass = 2.0       # [kg]
motor_speed = chrono.CH_C_PI  # [rad/s] (0.5 rev/sec)

# 3. Create the floor (truss)
truss = chrono.ChBody()
truss.SetBodyFixed(True)
truss.SetName("Truss")
system.Add(truss)

# Optional: Add a visual box for the floor
floor_shape = chrono.ChBoxShape()
floor_shape.GetBoxGeometry().Size = chrono.ChVectorD(1, 0.05, 0.5)
truss.AddAsset(floor_shape)
floor_texture = chrono.ChTexture()
floor_texture.SetTextureFilename(chrono.GetChronoDataFile("textures/concrete.jpg"))
truss.AddAsset(floor_texture)

# 4. Create the crankshaft (crank)
crank = chrono.ChBodyEasyBox(crank_length, crank_thickness, crank_thickness,  # size
                             1000, True, True)  # density, visualization, collision
crank.SetPos(chrono.ChVectorD(crank_length/2, 0, 0))
crank.SetMass(crank_mass)
crank.SetName("Crank")
system.Add(crank)

# 5. Create the connecting rod
rod = chrono.ChBodyEasyBox(rod_length, rod_thickness, rod_thickness,
                           1000, True, True)
rod.SetPos(chrono.ChVectorD(crank_length + rod_length/2, 0, 0))
rod.SetMass(rod_mass)
rod.SetName("Rod")
system.Add(rod)

# 6. Create the piston
piston = chrono.ChBodyEasyBox(piston_width, piston_height, piston_depth,
                              1000, True, True)
piston.SetPos(chrono.ChVectorD(crank_length + rod_length, 0, 0))
piston.SetMass(piston_mass)
piston.SetName("Piston")
system.Add(piston)

# 7. Add joints

# Revolute joint: crank to truss (crankshaft axis)
crankshaft_joint = chrono.ChLinkLockRevolute()
crankshaft_joint.Initialize(crank, truss,
    chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngAxis(0, chrono.ChVectorD(0,0,1))))
system.AddLink(crankshaft_joint)

# Revolute joint: crank to rod (crank pin)
crank_pin_pos = chrono.ChVectorD(crank_length, 0, 0)
crank_rod_joint = chrono.ChLinkLockRevolute()
crank_rod_joint.Initialize(crank, rod,
    chrono.ChCoordsysD(crank_pin_pos, chrono.Q_from_AngAxis(0, chrono.ChVectorD(0,0,1))))
system.AddLink(crank_rod_joint)

# Revolute joint: rod to piston (wrist pin)
rod_piston_joint_pos = chrono.ChVectorD(crank_length + rod_length, 0, 0)
rod_piston_joint = chrono.ChLinkLockRevolute()
rod_piston_joint.Initialize(rod, piston,
    chrono.ChCoordsysD(rod_piston_joint_pos, chrono.Q_from_AngAxis(0, chrono.ChVectorD(0,0,1))))
system.AddLink(rod_piston_joint)

# Prismatic (slider) joint: piston to truss (constrain piston to move along X)
slider_axis = chrono.ChVectorD(1, 0, 0)
slider_joint = chrono.ChLinkLockPrismatic()
slider_joint.Initialize(piston, truss,
    chrono.ChCoordsysD(piston.GetPos(), chrono.Q_from_AngAxis(chrono.CH_C_PI/2, chrono.ChVectorD(0,0,1))))
system.AddLink(slider_joint)

# 8. Add a motor to drive the crankshaft at constant speed
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(crank, truss, chrono.ChFrameD(chrono.ChVectorD(0,0,0), chrono.QUNIT))
motor_speed_fun = chrono.ChFunction_Const(motor_speed)
motor.SetSpeedFunction(motor_speed_fun)
system.AddLink(motor)

# 9. Visualization with Irrlicht
application = chronoirr.ChIrrApp(system, "Crank-Slider Mechanism", chronoirr.dimension2du(1024,768))

# Camera settings
application.AddTypicalCamera(chronoirr.vector3df(1.5, 0.5, 1.5), chronoirr.vector3df(0.5, 0, 0))

# Lighting
application.AddTypicalLights()
application.AddLightWithShadow(chronoirr.vector3df(2,2,2), chronoirr.vector3df(0,0,0), 3, 1, 3, 30)

# Add logo
logo_path = os.path.join(chrono.GetChronoDataPath(), "logo_pychrono_alpha.png")
application.AddLogo(logo_path)

# Add skybox
application.AddTypicalSky()

# Bind all assets for visualization
application.AssetBindAll()
application.AssetUpdateAll()

# 10. Simulation loop
application.SetTimestep(0.001)
application.SetTryRealtime(True)

while(application.GetDevice().run()):
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```

---

**How to Customize:**
- **Camera:** Change the `AddTypicalCamera` parameters for different views.
- **Lighting:** Adjust `AddLightWithShadow` or add more lights.
- **Textures:** Replace texture filenames for different looks.
- **Logo:** Use your own logo image if desired.

**How it works:**  
- The crank rotates at a constant speed, moving the connecting rod and piston.
- The piston is constrained to slide along the X-axis.
- The system is visualized in real time with lighting, textures, and a logo.
