Certainly! Below is a **PyChrono** simulation script that fulfills your requirements:

- Loads a triangular mesh from a Wavefront `.obj` file as a fixed body.
- Attaches a lidar sensor to the mesh via a sensor manager.
- Applies noise filters, visualization, and saving options to the lidar.
- During simulation, the lidar orbits the mesh by updating its position each step.
- Prints lidar buffer data at each simulation step.

**Assumptions:**
- You have a mesh file named `"mesh.obj"` in your working directory.
- PyChrono and PyChrono Sensor modules are installed.

```python
import pychrono as chrono
import pychrono.sensor as sens
import numpy as np
import math
import time

# 1. Initialize the PyChrono environment and core components
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
system = chrono.ChSystemSMC()

# 2. Add the required physical systems and objects as specified

# Load a triangular mesh from a Wavefront .obj file
mesh = chrono.ChTriangleMeshConnected()
mesh.LoadWavefrontMesh("mesh.obj", False, True)
mesh.RepairDuplicateVertexes(1e-9)

# Create a fixed body and attach the mesh
body_mesh = chrono.ChBody()
body_mesh.SetBodyFixed(True)
body_mesh.SetPos(chrono.ChVectorD(0, 0, 0))
body_mesh.GetAssets().push_back(chrono.ChTriangleMeshShape(mesh))
system.Add(body_mesh)

# 3. Set necessary default parameters such as positions, forces, and interactions

# Create a sensor manager
manager = sens.ChSensorManager(system)

# Lidar parameters
lidar_update_rate = 10.0  # Hz
lidar_horiz_samples = 512
lidar_vert_samples = 32
lidar_horiz_fov = 2 * math.pi  # 360 deg
lidar_vert_fov = math.radians(30)  # 30 deg
lidar_max_dist = 10.0  # meters

# Initial lidar position (orbit radius)
orbit_radius = 2.0
lidar_height = 1.0
lidar_angle = 0.0

# Function to compute lidar pose in orbit
def get_lidar_pose(angle):
    x = orbit_radius * math.cos(angle)
    y = lidar_height
    z = orbit_radius * math.sin(angle)
    # Look at the origin (mesh center)
    pos = chrono.ChVectorD(x, y, z)
    target = chrono.ChVectorD(0, 0, 0)
    up = chrono.ChVectorD(0, 1, 0)
    # Build a rotation matrix to look at the target
    dir = (target - pos).GetNormalized()
    right = up.Cross(dir).GetNormalized()
    up2 = dir.Cross(right)
    rot = chrono.ChMatrix33D()
    rot.Set_A_axis(right, up2, dir)
    quat = chrono.ChQuaternionD()
    rot.Get_A_quaternion(quat)
    return chrono.ChFrameD(pos, quat)

# Create the lidar sensor and attach to the mesh body
lidar = sens.ChLidarSensor(
    body_mesh,
    lidar_update_rate,
    get_lidar_pose(lidar_angle),
    lidar_horiz_samples,
    lidar_vert_samples,
    lidar_horiz_fov,
    lidar_vert_fov,
    lidar_max_dist
)

# Add noise filter
lidar.PushFilter(sens.ChFilterLidarNoise(0.01))  # 1cm stddev

# Add visualization filter
lidar.PushFilter(sens.ChFilterVisualize(640, 480, "Lidar Visualization"))

# Add data saving filter
lidar.PushFilter(sens.ChFilterSaveLidarPCD("lidar_output/"))

# Add to sensor manager
manager.AddSensor(lidar)

# Simulation parameters
step_size = 1.0 / 100.0  # 100 Hz physics
end_time = 2.0  # seconds
orbit_speed = 2 * math.pi / end_time  # one full orbit in end_time

# Main simulation loop
sim_time = 0.0
while sim_time < end_time:
    # Update lidar's pose to orbit the mesh
    lidar_angle = (sim_time / end_time) * 2 * math.pi
    lidar.SetOffsetPose(get_lidar_pose(lidar_angle))

    # Advance simulation
    system.DoStepDynamics(step_size)
    manager.Update()

    # Access lidar buffer data
    data = lidar.GetMostRecentBuffer()
    if data is not None:
        # Print some info about the point cloud
        print(f"Time: {sim_time:.2f}s | Lidar points: {data.GetPointCloud().shape}")
    else:
        print(f"Time: {sim_time:.2f}s | Lidar data not ready yet.")

    sim_time += step_size
    time.sleep(0.01)  # Slow down for visualization (optional)

print("Simulation complete.")
```

**Notes:**
- Replace `"mesh.obj"` with your actual mesh file path.
- The script creates a directory `"lidar_output/"` for saving point clouds.
- The lidar orbits the mesh at a fixed height, always looking at the mesh center.
- The script prints the shape of the point cloud at each step.
- Visualization window will pop up if you have the necessary dependencies.
