import os
import math
import numpy as np
import pychrono as chrono
import pychrono.robot as turtlebot
from pychrono import irrlicht as chronoirr

# ------------------- Error Corrections and Improvements -------------------
# 1. ChSystemNSC uses Set_G_acc() for gravity, not SetGravitationalAcceleration().
# 2. ChVector3d is not a valid class in PyChrono; use chrono.ChVectorD.
# 3. ChQuaterniond is not a valid class; use chrono.ChQuaternionD.
# 4. ChContactMaterialNSC requires a system as the first argument.
# 5. ChBodyEasyBox expects half-dimensions, not full dimensions.
# 6. SetTexture() expects a string path, not a chrono.GetChronoDataFile() object.
# 7. CameraVerticalDir_Z should be chronoirr.CameraVerticalDir_Z.
# 8. ChVector3d in AddCamera and AddLightWithShadow should be chrono.ChVectorD.
# 9. SetMotorSpeed expects (speed, wheel_id), not (wheel_id, speed).
# 10. The simulation loop should not redefine LEFT_DRIVE_WHEEL and RIGHT_DRIVE_WHEEL every iteration.
# 11. The ground's visual shape index may be -1 if no visual shape is present.
# 12. ChContactMaterialNSC should be created with the system as the first argument.

# ------------------- Create Chrono system -------------------
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.Set_G_acc(chrono.ChVectorD(0, 0, -9.81))  # Correct gravity setter
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# ------------------- Create ground body as terrain plane -------------------
ground_mat = chrono.ChContactMaterialNSC(system)  # Correct: pass system as argument
# ChBodyEasyBox expects half-dimensions, so for 20x20x1, use 10,10,0.5
ground = chrono.ChBodyEasyBox(10, 10, 0.5, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVectorD(0, 0, -0.6))  # Instruction 1: Adjusted ground position
ground.SetFixed(True)
# Set texture if visual shape exists
if ground.GetVisualShape(0):
    ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# ------------------- Create Turtlebot Robot -------------------
init_pos = chrono.ChVectorD(0, 0.2, 0)
init_rot = chrono.ChQuaternionD(1, 0, 0, 0)
robot = turtlebot.TurtleBot(system, init_pos, init_rot)
robot.Initialize()

# ------------------- Visualization -------------------
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetCameraVertical(chronoirr.CameraVerticalDir_Z)
vis.SetWindowSize(1280, 720)
vis.SetWindowTitle('Turtlebot Robot - Rigid terrain')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVectorD(0, 1.5, 0.2), chrono.ChVectorD(0, 0, 0.2))
vis.AddTypicalLights()
vis.AddLightWithShadow(chrono.ChVectorD(1.5, -2.5, 5.5), chrono.ChVectorD(0, 0, 0.5), 3, 4, 10, 40, 512)
# Instruction 2: Removed commented-out EnableShadows line

# ------------------- Motor Control Function -------------------
LEFT_DRIVE_WHEEL = 0
RIGHT_DRIVE_WHEEL = 1

def move(mode):
    """Set the robot's wheel speeds according to the movement mode."""
    if mode == 'straight':
        robot.SetMotorSpeed(2.0, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(2.0, RIGHT_DRIVE_WHEEL)
    elif mode == 'left':
        robot.SetMotorSpeed(0.5, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(2.0, RIGHT_DRIVE_WHEEL)
    elif mode == 'right':
        robot.SetMotorSpeed(2.0, LEFT_DRIVE_WHEEL)
        robot.SetMotorSpeed(0.5, RIGHT_DRIVE_WHEEL)
    else:
        raise ValueError(f"Invalid movement mode: {mode}")

# ------------------- Simulation Loop -------------------
time_step = 2e-3
time = 0
current_mode = None

while vis.Run():
    # Determine movement mode based on simulation time
    if time < 5.0:
        mode = 'straight'
    elif time < 10.0:
        mode = 'left'
    else:
        mode = 'right'

    # Only update and print if mode changes
    if mode != current_mode:
        move(mode)
        print(f"Time {time:.2f}s: Moving {mode}")
        current_mode = mode

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance the simulation by one time step
    system.DoStepDynamics(time_step)
    time += time_step
