import pychrono as chrono
import pychrono.irrlicht as chronoirr
from itertools import combinations


class MyCreatorForAll(chrono.ChRandomShapeCreator_AddBodyCallback):
    def __init__(self, vis, coll):
        super().__init__()
        self.vis = vis
        self.coll = coll

    def OnAddBody(self, body, coords, creator):
        body.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/bluewhite.png"))
        self.vis.BindItem(body)
        self.coll.BindItem(body)
        body.SetUseGyroTorque(False)


class MyParticleEmitter(chrono.ChParticleEmitter):
    def __init__(self, vis, coll):
        super().__init>(20)
        self.vis = vis
        self.coll = coll
        self.creator_for_all = MyCreatorForAll(vis, coll)

    def CreateParticle(self, particle, coords):
        particle.SetMass(0.02)
        particle.SetDamping(0.0)
        particle.SetInertiaXX(chrono.ChVector3d(0.02, 0.02, 0.02))
        particle.SetPos(coords)
        particle.SetRot(chrono.QUNIT)
        particle.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/bluewhite.png"))
        self.vis.BindItem(particle)
        self.coll.BindItem(particle)
        particle.SetUseGyroTorque(False)


class MyForces(chrono.ChBodyAuxRef):
    def __init__(self, bodylist):
        super().__init__()
        self.bodylist = bodylist

    def UpdateForce(self, body, force, torque, time):
        G = 1.0
        for other in self.bodylist:
            if other != body:
                d = (other.GetPos() - body.GetPos()).Length()
                F = G * other.GetMass() / (d * d)
                force += (other.GetPos() - body.GetPos()).GetNormalized() * F


class SimBody(chrono.ChBodyEasyBox):
    def __init__(self, vis, coll):
        super().__init__(0.2, 0.2, 0.2, 1000, True, True, None, None)
        self.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
        vis.BindItem(self)
        coll.BindItem(self)
        self.SetUseGyroTorque(False)


def main():
    chrono.SetChronoDataPath(chrono.GetChronoDataPath())
    irr_vis = chronoirr.ChVisualSystemIrrlicht()
    irr_vis.SetWindowTitle("Particle emitter demo")
    irr_vis.SetWindowSize(1024, 768)
    irr_vis.Initialize()
    irr_vis.AddLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
    irr_vis.AddSkyBox()
    irr_vis.AddCamera(chrono.ChVector3d(0, 0.6, -1))
    irr_vis.AddTypicalLights()

    mphysicalSystem = chrono.ChSystemNSC()
    mphysicalSystem.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
    mcollisiontype = chrono.ChCollisionShape.Type_NONE

    emitter = chrono.ChParticleEmitter()
    emitter.SetParticleCreator(chrono.ChRandomShapeCreator())
    emitter.GetParticleCreator().SetUseParticleReservoir(True)
    emitter.GetParticleCreator().SetParticleReservoirAmount(200)
    emitter.GetParticleCreator().SetParticleMass(0.02)
    emitter.GetParticleCreator().SetParticleInertia(chrono.ChInertiaEllipsoid(0.02, 0.02, 0.02))
    emitter.GetParticleCreator().SetParticleVelocityDistribution(
        chrono.ChUniformDistribution(2, 4), chrono.ChUniformDistribution(-1, 1), chrono.ChUniformDistribution(-1, 1)
    )
    emitter.SetParticlePositionDistribution(chrono.ChRandomDistributionPointInSphere(0.2))

    sphere = chrono.ChBodyEasySphere(0.3, 1000, True, True)
    sphere.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/pebbles.jpg"))
    mphysicalSystem.Add(sphere)

    while (not irr_vis.GetDevice().GetDeviceShutDown()):
        emitter.EmitParticles(mphysicalSystem, 0.02)

        mphysicalSystem.DoStepDynamics(0.01)

        irr_vis.BeginScene()
        irr_vis.Render()
        irr_vis.EndScene()
        mphysicalSystem.GetSystem()->GetApplication().GetLog()->Print("SYS: step time = %g", mphysicalSystem.GetChTime())


main()
