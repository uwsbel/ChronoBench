```python
import pychrono.core as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# 1. Initialize the Chrono system and core components
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# Create the terrain (ground body)
ground = chrono.ChBodyEasyBox(50, 1, 50, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -0.5, 0))
ground.SetBodyFixed(True)
system.Add(ground)

# 2. Initialize the Viper rover and its driver
# Create the vehicle
viper = veh.Viper(system)

# Set initial position
initial_pos = chrono.ChVectorD(0, 0.5, 0)
viper.GetChassisBody().SetPos(initial_pos)

# Initialize the driver
driver = veh.ChIrrGuiDriver()
driver.Initialize(viper.GetChassisBody())

# 3. Set up Irrlicht visualization
application = irr.ChIrrApp(system, "Viper Rover Simulation", irr.dimension2du(800, 600))
application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(irr.vector3df(0, 5, -10))
application.AssetBindAll()
application.AssetUpdateAll()

# Simulation parameters
time_step = 0.01
total_time = 20  # seconds
steering_change_duration = 10  # seconds
max_steering_angle = 0.5  # radians (~28.6 degrees)

# Initialize steering angle
current_steering = 0.0

# Simulation loop
t = 0.0
while application.GetDevice().run():
    # Update driver inputs
    driver.BeginScene()
    driver.Advance(t)
    driver.EndScene()

    # Gradually change steering over the first 'steering_change_duration' seconds
    if t <= steering_change_duration:
        # Linearly interpolate steering from 0 to max_steering_angle
        current_steering = (t / steering_change_duration) * max_steering_angle
    else:
        current_steering = max_steering_angle

    # Apply steering to the vehicle
    viper.GetSteering().SetAngle(current_steering)

    # Perform simulation step
    system.DoStepDynamics(time_step)

    # Render the scene
    application.BeginScene()
    application.DrawAll()
    application.EndScene()

    t += time_step
    if t >= total_time:
        break
```
