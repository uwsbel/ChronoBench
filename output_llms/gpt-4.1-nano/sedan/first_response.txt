```python
import pychrono.core as chrono
import pychrono.vehicle as veh

# 1. Initialize the Chrono system
system = chrono.ChSystemNSC()

# 2. Create the terrain (rigid ground)
ground = chrono.ChBodyEasyBox(50, 0.2, 50, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -0.1, 0))
ground.SetBodyFixed(True)
system.Add(ground)

# Optional: Customize terrain appearance (textures, logos)
# Note: In PyChrono, visual customization can be done via visualization assets
# For simplicity, we'll add a basic texture
vis_terrain = chrono.ChVisualShape()
vis_terrain.SetTextureFilename("textures/terrain_texture.jpg")  # Placeholder path
ground.AddVisualShape(vis_terrain)

# 3. Create the vehicle
# Initialize the vehicle with default parameters
vehicle = veh.ChVehicleAssembly()

# Create the chassis
chassis = veh.ChBodyAuxRef()
chassis.SetBodyFixed(False)
chassis.SetPos(chrono.ChVectorD(0, 0.5, 0))
system.Add(chassis)

# Initialize the vehicle with the chassis
my_vehicle = veh.ChVehicle()
my_vehicle.SetChassis(chassis)

# Add the vehicle to the assembly
vehicle.AddVehicle(my_vehicle)

# 4. Configure the vehicle visualization
# Enable visualization of the vehicle
my_vehicle.SetVisualizationType(veh.VEHICLE_VISUALIZATION_TYPE.VEHICLE_VISUALIZATION_TYPE_MESH)

# 5. Add tires with TMEASY model
# Define tire parameters
tire_radius = 0.33
tire_width = 0.2
tire_mass = 20
tire_inertia = chrono.ChVectorD(0.5, 0.5, 0.5)

# Positions for four tires
tire_positions = [
    chrono.ChVectorD(1.2, 0, 1.0),   # Front right
    chrono.ChVectorD(-1.2, 0, 1.0),  # Front left
    chrono.ChVectorD(1.2, 0, -1.0),  # Rear right
    chrono.ChVectorD(-1.2, 0, -1.0)  # Rear left
]

# Create and add tires
for i, pos in enumerate(tire_positions):
    tire = veh.ChTMeasyTire()
    tire.SetRadius(tire_radius)
    tire.SetWidth(tire_width)
    tire.SetMass(tire_mass)
    tire.SetInertia(chrono.ChVectorD(0.5, 0.5, 0.5))
    tire.Initialize(my_vehicle.GetChassisBody(), pos, chrono.QUNIT)
    tire.SetVisualizationType(veh.VEHICLE_TIRE_VISUALIZATION_TYPE.VEHICLE_TIRE_VISUALIZATION_TYPE_MESH)
    tire.SetUseMeshVisualization(True)
    tire.SetTextureFilename("textures/tire_texture.jpg")  # Placeholder path
    my_vehicle.AddTire(tire)

# 6. Set collision and contact parameters
# Enable collision detection for tires
for tire in my_vehicle.GetTires():
    tire.GetTireBody().SetCollide(True)

# 7. Add driver system for real-time control
driver = veh.ChIrrGuiDriver()
driver.Initialize(system, my_vehicle.GetChassisBody())

# 8. Set up the Irrlicht visualization
# Create the Irrlicht application
application = veh.ChIrrApp(system, "BMW E90 Simulation", 800, 600, False, True)

# Add typical visualization assets
application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(veh.ChVectorD(0, 2, -5))
application.SetChaseCamera(veh.ChVectorD(0, 1.5, -3),  # Camera position
                           veh.ChVectorD(0, 1, 0))     # Look-at point

# Enable logo overlay or custom textures if needed
# (Placeholder: actual implementation depends on available assets)

# 9. Simulation loop
application.AssetBindAll()
application.AssetUpdateAll()

# Simulation parameters
time_step = 0.005
application.SetTimestep(time_step)

# Run the simulation
while application.GetDevice().run():
    # Get user inputs for driver controls
    driver.Synchronize()
    driver.Advance(time_step)

    # Apply driver inputs to the vehicle
    my_vehicle.GetDrive().SetThrottle(driver.GetThrottle());
    my_vehicle.GetDrive().SetSteering(driver.GetSteering());
    my_vehicle.GetBrake().SetBraking(driver.GetBraking());

    # Advance the system
    system.DoStepDynamics(time_step)

    # Update visualization
    application.BeginScene()
    application.DrawAll()
    application.EndScene()
```
