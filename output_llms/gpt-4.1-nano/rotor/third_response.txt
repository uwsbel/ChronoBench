```python
import math as m
import pychrono as chrono
import pychrono.fea as fea
import pychrono.pardisomkl as mkl
import pychrono.irrlicht as chronoirr

# Create the physical system
sys = chrono.ChSystemSMC()

# Create mesh for FEA
mesh = fea.ChMesh()
sys.Add(mesh)

# Enable automatic gravity with at least 2 integration points per element
mesh.SetAutomaticGravity(True, 2)
sys.SetGravitationalAcceleration(chrono.ChVector3d(0, -9.81, 0))

# Beam parameters
beam_L = 6
beam_ro = 0.050
beam_ri = 0.045
CH_PI = 3.141592653589793  # Use math.pi for better precision

# Create inertia object for the beam cross-section
minertia = fea.ChInertiaCosseratSimple()
minertia.SetDensity(7800)
minertia.SetArea(CH_PI * (m.pow(beam_ro, 2) - m.pow(beam_ri, 2)))
minertia.SetIyy((CH_PI / 4.0) * (m.pow(beam_ro, 4) - m.pow(beam_ri, 4)))
minertia.SetIzz((CH_PI / 4.0) * (m.pow(beam_ro, 4) - m.pow(beam_ri, 4)))

# Create elasticity object for the beam
melasticity = fea.ChElasticityCosseratSimple()
melasticity.SetYoungModulus(210e9)
melasticity.SetShearModulusFromPoisson(0.3)
melasticity.SetIyy((CH_PI / 4.0) * (m.pow(beam_ro, 4) - m.pow(beam_ri, 4)))
melasticity.SetIzz((CH_PI / 4.0) * (m.pow(beam_ro, 4) - m.pow(beam_ri, 4)))
melasticity.SetJ((CH_PI / 2.0) * (m.pow(beam_ro, 4) - m.pow(beam_ri, 4)))

# Create the beam section
msection = fea.ChBeamSectionCosserat(minertia, melasticity)
msection.SetCircular(True)
msection.SetDrawCircularRadius(beam_ro)

# Build the beam using IGA
builder = fea.ChBuilderBeamIGA()
builder.BuildBeam(mesh,
                  msection,
                  20,
                  chrono.ChVector3d(0, 0, 0),
                  chrono.ChVector3d(beam_L, 0, 0),
                  chrono.VECT_Y,
                  1)

# Get the middle node for attaching the flywheel
node_mid_index = int(m.floor(builder.GetLastBeamNodes().size() / 2.0))
node_mid = builder.GetLastBeamNodes()[node_mid_index]

# Create the flywheel and attach it to the middle node
mbodyflywheel = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, 0.24, 0.1, 7800)
mbodyflywheel.SetCoordsys(
    chrono.ChCoordsysd(
        node_mid.GetPos() + chrono.ChVector3d(0, 0.05, 0),
        chrono.QuatFromAngleAxis(CH_PI / 2.0, chrono.VECT_Z)
    )
)
sys.Add(mbodyflywheel)

# Fix the flywheel to the node
myjoint = chrono.ChLinkMateFix()
myjoint.Initialize(node_mid, mbodyflywheel)
sys.Add(myjoint)

# Create a fixed truss
truss = chrono.ChBody()
truss.SetFixed(True)
sys.Add(truss)

# Create the end bearing
bearing = chrono.ChLinkMateGeneric(False, True, True, False, True, True)
bearing.Initialize(
    builder.GetLastBeamNodes().back(),
    truss,
    chrono.ChFramed(builder.GetLastBeamNodes().back().GetPos())
)
sys.Add(bearing)

# Define a custom motor function class
class ChFunctionMyFun(chrono.ChFunction):
    def __init__(self, A1=10, A2=20, T1=1, T2=2, T3=3, w=1):
        super().__init__()
        self.A1 = A1
        self.A2 = A2
        self.T1 = T1
        self.T2 = T2
        self.T3 = T3
        self.w = w

    def GetVal(self, x):
        # Piecewise function based on time x
        if x < self.T1:
            return self.A1
        elif x < self.T2:
            return self.A2
        elif x < self.T3:
            return self.A1
        else:
            return self.A1 + self.w * (x - self.T3)

# Create the rotation motor with the custom function
rotmotor1 = chrono.ChLinkMotorRotationSpeed()

# Initialize the motor between the first node and the truss
rotmotor1.Initialize(
    builder.GetLastBeamNodes().front(),
    truss,
    chrono.ChFrameD(
        builder.GetLastBeamNodes().front().GetPos(),
        chrono.QuatFromAngleAxis(CH_PI / 2.0, chrono.VECT_Y)
    )
)

# Set the custom motor function
custom_func = ChFunctionMyFun(A1=10, A2=20, T1=1, T2=2, T3=3, w=5)
rotmotor1.SetMotorFunction(custom_func)

sys.Add(rotmotor1)

# Visualization of the FEM mesh
visual_shape_surface = chrono.ChVisualShapeFEA(mesh)
visual_shape_surface.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_SURFACE)
visual_shape_surface.SetSmoothFaces(True)
mesh.AddVisualShapeFEA(visual_shape_surface)

visual_shape_nodes = chrono.ChVisualShapeFEA(mesh)
visual_shape_nodes.SetFEMglyphType(chrono.ChVisualShapeFEA.GlyphType_NODE_CSYS)
visual_shape_nodes.SetFEMdataType(chrono.ChVisualShapeFEA.DataType_NONE)
visual_shape_nodes.SetSymbolsThickness(0.006)
visual_shape_nodes.SetSymbolsScale(0.01)
visual_shape_nodes.SetZbufferHide(False)
mesh.AddVisualShapeFEA(visual_shape_nodes)

# Create the Irrlicht visualization system
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Test FEA: the Jeffcott rotor with IGA beams')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(0, 1, 4), chrono.ChVector3d(beam_L / 2, 0, 0))
vis.AddTypicalLights()

# Use HHT timestepper for higher accuracy (optional)
# sys.SetTimestepperType(chrono.ChTimestepper.Type_HHT)

# Set solver
msolver = mkl.ChSolverPardisoMKL()
sys.SetSolver(msolver)

# Perform static analysis
sys.DoStaticLinear()

# Run the simulation
while vis.Run():
    vis.BeginScene()
    vis.Render()
    vis.EndScene()
    sys.DoStepDynamics(0.002)
```
