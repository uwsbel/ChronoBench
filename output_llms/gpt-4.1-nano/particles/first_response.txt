Certainly! Below is a comprehensive PyChrono simulation script that fulfills your requirements:

- Initializes the PyChrono environment.
- Sets up the physical system.
- Implements a particle emitter that generates particles with random shapes, positions, velocities, and orientations.
- Applies custom gravitational forces to simulate particle attraction dynamically.
- Visualizes the simulation using Irrlicht.

```python
import pychrono.core as chrono
import pychrono.irrlicht as irr
import random
import math

# Initialize the Chrono system
system = chrono.ChSystemNSC()

# Set gravity to zero since we'll implement custom gravity
system.Set_G_acc(chrono.ChVectorD(0, 0, 0))

# Create a fixed central body to attract particles (e.g., a massive sphere at the origin)
central_body = chrono.ChBodyEasySphere(1.0,  # radius
                                       1000,  # density
                                       True,  # visualization
                                       True)  # collision
central_body.SetPos(chrono.ChVectorD(0, 0, 0))
central_body.SetBodyFixed(True)
system.Add(central_body)

# Parameters for particle emitter
num_particles = 50
spawn_area_radius = 20
min_velocity = 1
max_velocity = 5

# Store particles for force calculations
particles = []

# Function to create a random shape (box, sphere, or cylinder)
def create_random_shape():
    shape_type = random.choice(['box', 'sphere', 'cylinder'])
    if shape_type == 'box':
        size = random.uniform(0.2, 0.5)
        shape = chrono.ChBodyEasyBox(size, size, size,  # dimensions
                                     1.0,  # density
                                     True,  # visualization
                                     True)  # collision
    elif shape_type == 'sphere':
        radius = random.uniform(0.2, 0.5)
        shape = chrono.ChBodyEasySphere(radius, 1.0, True, True)
    else:  # cylinder
        radius = random.uniform(0.2, 0.4)
        height = random.uniform(0.2, 0.6)
        shape = chrono.ChBodyEasyCylinder(radius, height, 1.0, True, True)
    return shape

# Particle emitter: generate particles with random properties
for _ in range(num_particles):
    particle = create_random_shape()
    # Random position within spawn_area_radius
    theta = random.uniform(0, 2 * math.pi)
    phi = random.uniform(0, math.pi)
    r = random.uniform(0, spawn_area_radius)
    x = r * math.sin(phi) * math.cos(theta)
    y = r * math.sin(phi) * math.sin(theta)
    z = r * math.cos(phi)
    particle.SetPos(chrono.ChVectorD(x, y, z))
    
    # Random velocity directed outward or inward
    speed = random.uniform(min_velocity, max_velocity)
    direction_theta = random.uniform(0, 2 * math.pi)
    direction_phi = random.uniform(0, math.pi)
    vx = speed * math.sin(direction_phi) * math.cos(direction_theta)
    vy = speed * math.sin(direction_phi) * math.sin(direction_theta)
    vz = speed * math.cos(direction_phi)
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    particle.SetPos(chrono.ChVectorD(x, y, z))
    # Set initial velocity
    particle.SetVel(chrono.ChVectorD(vx, vy, vz))
    
    # Random orientation
    quat = chrono.ChQuaternionD()
    quat.Q_from_AngX(random.uniform(0, 2*math.pi))
    quat.Q_from_AngY(random.uniform(0, 2*math.pi))
    quat.Q_from_AngZ(random.uniform(0, 2*math.pi))
    particle.SetRot(quat)
    
    system.Add(particle)
    particles.append(particle)

# Define gravitational force parameters
G = 1.0  # Gravitational constant (arbitrary units)

# Create Irrlicht visualization
application = irr.ChIrrApp(system, "Particle Gravitational Attraction", irr.dimension2du(800, 600))
application.AddTypicalSky()
application.AddTypicalCamera(irr.vector3df(0, 10, -30))
application.AssetBindAll()
application.AssetUpdateAll()

# Simulation loop
time_step = 0.01
while application.GetDevice().run():
    # Apply custom gravitational forces between particles and central body
    for p in particles:
        # Vector from particle to central body
        dir_vec = central_body.GetPos() - p.GetPos()
        distance = dir_vec.Length()
        if distance > 0.1:  # avoid singularity
            force_magnitude = G * central_body.GetMass() * p.GetMass() / (distance ** 2)
            force_dir = dir_vec / distance
            force = force_dir * force_magnitude
            p.ApplyForce(force)
        # Optional: add mutual attraction between particles
        # for other_p in particles:
        #     if other_p != p:
        #         dir_vec = other_p.GetPos() - p.GetPos()
        #         distance = dir_vec.Length()
        #         if distance > 0.1:
        #             force_magnitude = G * other_p.GetMass() * p.GetMass() / (distance ** 2)
        #             force_dir = dir_vec / distance
        #             force = force_dir * force_magnitude
        #             p.ApplyForce(force)
    
    system.DoStepDynamics(time_step)
    application.BeginScene()
    application.DrawAll()
    application.EndScene()

# Note: To stop the simulation, close the Irrlicht window.
```

### Explanation:
- **System Initialization:** Sets up the `ChSystemNSC` with zero gravity since we implement custom forces.
- **Central Body:** A fixed sphere at the origin acts as the massive object attracting particles.
- **Particle Emitter:** Creates particles with random shapes (`box`, `sphere`, `cylinder`), positions within a sphere, random velocities, and orientations.
- **Forces:** In each simulation step, applies a gravitational force from the central body to each particle based on Newton's law of universal gravitation.
- **Visualization:** Uses Irrlicht to visualize the scene, with a camera and sky added for better visualization.
- **Simulation Loop:** Advances the simulation, applies forces, and renders each frame until the window is closed.

Feel free to adjust parameters such as `G`, number of particles, or spawn area to suit your specific needs!
