```python
import pychrono.core as chrono
import pychrono.animate as animate
import random
import math

# Initialize the Chrono engine
chrono.SetChronoDataPath('/path/to/data/')  # Set to your data path if needed
system = chrono.ChSystemNSC()

# Set gravitational acceleration (vector pointing downward)
gravity = chrono.ChVectorD(0, -9.81, 0)
system.Set_G_acc(gravity)

# Create a ground body (fixed, optional; can be used as a reference)
ground = chrono.ChBodyEasyBox(10, 1, 10, 1000, True, True)
ground.SetPos(chrono.ChVectorD(0, -0.5, 0))
system.Add(ground)

# Particle emitter parameters
num_particles = 50  # Total particles to emit
emission_area_center = chrono.ChVectorD(0, 2, 0)
emission_area_size = 1.0  # Particles spawn within a cube of size 1 around the center

# Store particles for force calculations
particles = []

# Function to generate a random shape (sphere or box)
def create_random_shape():
    shape_type = random.choice(['sphere', 'box'])
    density = 1000  # kg/m^3
    if shape_type == 'sphere':
        radius = random.uniform(0.05, 0.2)
        shape = chrono.ChBodyEasySphere(radius, density, True, True)
    else:
        size = [random.uniform(0.05, 0.2) for _ in range(3)]
        shape = chrono.ChBodyEasyBox(size[0], size[1], size[2], density, True, True)
    return shape

# Function to generate random initial properties for particles
def generate_particle():
    body = create_random_shape()
    # Random position within emission area
    pos = chrono.ChVectorD(
        emission_area_center.x + random.uniform(-emission_area_size/2, emission_area_size/2),
        emission_area_center.y + random.uniform(-emission_area_size/2, emission_area_size/2),
        emission_area_center.z + random.uniform(-emission_area_size/2, emission_area_size/2)
    )
    body.SetPos(pos)
    # Random initial velocity
    vel = chrono.ChVectorD(
        random.uniform(-1, 1),
        random.uniform(0, 2),
        random.uniform(-1, 1)
    )
    body.SetPos(pos)
    body.SetRot(chrono.Q_from_EulerAngles(
        random.uniform(0, math.pi),
        random.uniform(0, math.pi),
        random.uniform(0, math.pi)
    ))
    body.SetPos(pos)
    body.SetMassProperties(body.GetMass(), body.GetInertia())
    body.SetPos(pos)
    body.SetRot(chrono.Q_from_EulerAngles(
        random.uniform(0, math.pi),
        random.uniform(0, math.pi),
        random.uniform(0, math.pi)
    ))
    # Set initial velocity
    body.SetVelocity(vel)
    system.Add(body)
    return body

# Emit particles
for _ in range(num_particles):
    particle = generate_particle()
    particles.append(particle)

# Define a force functor for gravitational attraction
class CustomGravityForce:
    def __init__(self, particles, G=1.0):
        self.particles = particles
        self.G = G  # Gravitational constant for simulation

    def apply(self):
        for i, p1 in enumerate(self.particles):
            for j, p2 in enumerate(self.particles):
                if i >= j:
                    continue  # Avoid double calculation or self-force
                # Calculate vector from p1 to p2
                dir = p2.GetPos() - p1.GetPos()
                dist_sq = dir.Length2()
                if dist_sq < 1e-4:
                    continue  # Avoid singularity
                dist = math.sqrt(dist_sq)
                # Compute gravitational force magnitude
                mass1 = p1.GetMass()
                mass2 = p2.GetMass()
                force_mag = self.G * mass1 * mass2 / dist_sq
                # Force vector
                force_vec = dir * (force_mag / dist)
                # Apply forces in opposite directions
                p1.ApplyForce(force_vec)
                p2.ApplyForce(-force_vec)

# Create an instance of the custom force
gravity_force = CustomGravityForce(particles, G=0.5)

# Set up the Irrlicht visualization
visualization = animate.ChApiIrrlicht()
visualization.SetSystem(system)
visualization.Initialize()
visualization.AddTypicalSky()
visualization.AddTypicalCamera(chrono.ChVectorD(0, 2, 5))
visualization.AddTypicalLights()

# Simulation parameters
time_step = 0.01
total_time = 10.0
current_time = 0.0

# Main simulation loop
while visualization.GetDevice().run():
    # Apply custom gravitational forces between particles
    gravity_force.apply()

    # Step the simulation
    system.DoStepDynamics(time_step)

    # Optionally, add code here to remove or reset particles if needed

    # Render the scene
    visualization.BeginScene()
    visualization.DrawAll()
    visualization.EndScene()

    # Increment time
    current_time += time_step
    if current_time >= total_time:
        break

# Cleanup
visualization.GetDevice().closeDevice()
```

### Explanation:
- **Initialization**: Sets up the `ChSystemNSC`, gravity, and ground.
- **Particle Emitter**: Randomly creates particles of different shapes (`sphere` or `box`) with random positions, orientations, and velocities.
- **Custom Gravitational Force**: Applies pairwise gravitational forces between particles dynamically each time step, simulating attraction.
- **Visualization**: Uses Irrlicht to render the simulation with typical sky, camera, and lighting setup.
- **Simulation Loop**: Advances the simulation by fixed time steps, applying forces, stepping physics, and rendering each frame.

### Notes:
- Adjust the `G` parameter in `CustomGravityForce` to control the strength of attraction.
- Ensure you replace `/path/to/data/` with the actual path to Chrono data files if needed.
