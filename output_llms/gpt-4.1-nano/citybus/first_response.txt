```python
import pychrono.core as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# 1. Initialize the Chrono engine and create the physical system
chrono.SetChronoDataPath('path/to/data')  # Set the correct data path if needed
system = chrono.ChSystemNSC()

# 2. Create the Irrlicht application for visualization
application = irr.ChIrrApp(system, 'CityBus Simulation', irr.dimension2du(1024, 768))
application.AddTypicalSky()
application.AddTypicalCamera(irr.vector3df(0, 10, -20))
application.AddTypicalLights()

# 3. Create the terrain
terrain = veh.RigidTerrain(system)
terrain.SetTexture(vehicle.GetDataFile('terrain/textures/tile4.jpg'))
terrain.Initialize()

# 4. Create the CityBus vehicle
# Define initial position and orientation
initial_pos = chrono.ChVectorD(0, 0.5, 0)
initial_rot = chrono.Q_from_AngZ(0)

# Instantiate the vehicle with specified parameters
citybus = veh.CityBus(system)
citybus.SetPos(initial_pos)
citybus.SetRot(initial_rot)

# Set tire model to a detailed tire model (e.g., RigidTire)
for tire in citybus.GetTireList():
    tire.SetTireType(veh.TireModelType.RIGID)
    # Optional: set tire visualization type
    tire.SetVisualizationType(veh.VisualizationType.MESH)

# 5. Set visualization types for vehicle parts
# For example, mesh visualization for the chassis
citybus.GetChassisBody().GetVisualShape(0).SetMeshFilename(vehicle.GetDataFile('vehicle/CityBus/CityBus_mesh.obj'))
# Primitive visualization for other parts if needed
# (Assuming the default visualization is acceptable for other parts)

# 6. Create the driver system for interactive control
driver = veh.ChIrrGuiDriver(application)
driver.Initialize()
driver.SetInputMode(veh.ChIrrGuiDriver.InputMode.IDEAL)

# 7. Set up the simulation parameters
application.SetStepManage(true)
application.SetTimestep(1.0 / 50)  # 50 FPS

# 8. Set camera to follow the vehicle
application.GetSceneManager().AddCameraSceneNode(irr.SCameraSceneNode(
    application.GetSceneManager().GetRootSceneNode(),
    irr.SVector3(0, 5, -15),
    irr.SVector3(0, 2, 0),
    1000
))
# Alternatively, update camera position each frame to follow the vehicle

# 9. Main simulation loop
while application.GetDevice().run():
    # Begin scene rendering
    application.BeginScene()
    application.DrawAll()

    # Update driver inputs
    driver.ReadInputs()

    # Apply driver controls to the vehicle
    citybus.SetSteeringAngle(driver.GetSteering() * 0.5)  # scale steering
    citybus.SetThrottle(driver.GetThrottle() * 1.0)
    citybus.SetBraking(driver.GetBrake() * 1.0)

    # Synchronize the vehicle
    citybus.Synchronize(chrono.ChTimeStep(1.0 / 50))
    system.DoStepDynamics(chrono.ChTimeStep(1.0 / 50))

    # Update camera to follow the vehicle
    vehicle_pos = citybus.GetChassisBody().GetPos()
    camera_pos = vehicle_pos + chrono.ChVectorD(0, 5, -15)
    application.GetSceneManager().GetActiveCameraSceneNode().setPosition(
        irr.SVector3(camera_pos.x, camera_pos.y + 2, camera_pos.z)
    )
    application.GetSceneManager().GetActiveCameraSceneNode().setTarget(
        irr.SVector3(vehicle_pos.x, vehicle_pos.y + 1, vehicle_pos.z)
    )

    # End scene rendering
    application.EndScene()

# End of script
```
