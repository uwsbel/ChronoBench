Below is the corrected and modified PyChrono script. The main corrections involve:

- Fixing method names and parameter specifications where needed.
- Changing the joint types as per instructions:
  - Crank-rod joint from `ChLinkRevolute` to `ChLinkBall`.
  - Rod-piston joint from `ChLinkRevolute` to `ChLinkBall`.
  - Piston-floor joint from `ChLinkLockPrismatic` to `ChLinkLockPlanePlane`.
- Adjusting initialization parameters to match the correct usage.
- Ensuring the simulation runs and the visualization code is consistent.

---

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import matplotlib.pyplot as plt
import numpy as np

# Uncomment or set data path if needed
# chrono.SetChronoDataPath('path/to/data')

sys = chrono.ChSystemNSC()

# Some data shared in the following
crank_center = chrono.ChVector3d(-1, 0.5, 0)
crank_rad = 0.4
crank_thick = 0.1
rod_length = 1.5

# Create the floor truss
mfloor = chrono.ChBodyEasyBox(3, 1, 3, 1000)
mfloor.SetPos(chrono.ChVector3d(0, -0.5, 0))
mfloor.SetFixed(True)
sys.Add(mfloor)

# Create the flywheel crank
mcrank = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, crank_rad, crank_thick, 1000)
mcrank.SetPos(crank_center + chrono.ChVector3d(0, 0, -0.1))
# Rotate cylinder to align with Z instead of Y, since ChBodyEasyCylinder creates with Y axis:
# Use quaternion for rotation:
quat_rotate = chrono.Q_from_AngleAxis(np.pi/2, chrono.VECT_Y)  # Rotate 90 deg about Y to align along Z
mcrank.SetRot(quat_rotate)
sys.Add(mcrank)

# Create a stylized rod
mrod = chrono.ChBodyEasyBox(rod_length, 0.1, 0.1, 1000)
mrod.SetPos(crank_center + chrono.ChVector3d(crank_rad + rod_length/2, 0, 0))
sys.Add(mrod)

# Create a stylized piston
mpiston = chrono.ChBodyEasyCylinder(chrono.ChAxis_Y, 0.2, 0.3, 1000)
mpiston.SetPos(crank_center + chrono.ChVector3d(crank_rad + rod_length, 0, 0))
mpiston.SetRot(chrono.Q_from_AngleAxis(np.pi/2, chrono.VECT_Z))  # Rotate to align in x-y plane
sys.Add(mpiston)

# Create crank-motor (rotating the crank)
my_motor = chrono.ChLinkMotorRotationSpeed()
my_motor.Initialize(mcrank,  # first body
                    mfloor,  # second body (though typically motor is between crank and ground; for simplicity, this is kept)
                    chrono.ChFrameD(crank_center))
my_angularspeed = chrono.ChFunction_Const(chrono.CH_C_PI)  # 180 deg/sec
my_motor.SetMotorFunction(my_angularspeed)
sys.Add(my_motor)

# Create crank-rod joint: from revolute to ball-and-socket joint
mjointA = chrono.ChLinkBall()
mjointA.Initialize(mrod,
                   mcrank,
                   chrono.ChFrameD(crank_center + chrono.ChVector3d(crank_rad, 0, 0)))
sys.Add(mjointA)

# Create rod-piston joint: from revolute to ball-and-socket joint
mjointB = chrono.ChLinkBall()
mjointB.Initialize(mpiston,
                   mrod,
                   chrono.ChFrameD(crank_center + chrono.ChVector3d(crank_rad + rod_length, 0, 0)))
sys.Add(mjointB)

# Replace piston-floor prismatic joint with plane-plane (planar) joint
# ChLinkLockPlanePlane constrains motion in x-y plane, allowing rotation
mjointC = chrono.ChLinkLockPlanePlane()
# Initialize with frames: the frames are defined at the connection points
# Plane-Plane joint is initialized by specifying the frames on each body:
# Use the frames at the connection point with orientation aligned in the x-y plane.
frame_piston = chrono.ChFrameD(mpiston.GetPos(), chrono.Q_from_AngleAxis(0, chrono.VECT_Z))
frame_floor = chrono.ChFrameD(mfloor.GetPos(), chrono.Q_from_AngleAxis(0, chrono.VECT_Z))
mjointC.Initialize(mpiston, mfloor, frame_piston, frame_floor)
sys.Add(mjointC)

# Initialize visualization
vis = chronoirr.ChVisualSystemIrrlicht()
vis.AttachSystem(sys)
vis.SetWindowSize(1024, 768)
vis.SetWindowTitle('Crank demo')
vis.Initialize()
vis.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis.AddSkyBox()
vis.AddCamera(chrono.ChVector3d(1, 1, 3), chrono.ChVector3d(0, 1, 0))
vis.AddTypicalLights()

# Data arrays for plotting
array_time = []
array_angle = []
array_pos = []
array_speed = []

# Run the simulation loop
while vis.Run():
    arr_time = sys.GetChTime()
    array_time.append(arr_time)

    array_angle.append(my_motor.GetMotorAngle())

    array_pos.append(mpiston.GetPos().x)
    array_speed.append(mpiston.GetPosDt().x)

    # Visualization step
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    sys.DoStepDynamics(1e-3)

    # Stop after 20 seconds
    if arr_time > 20:
        vis.GetDevice().closeDevice()

# Plotting results
fig, (ax1, ax2) = plt.subplots(2, sharex=True)

ax1.plot(array_angle, array_pos)
ax1.set(ylabel='position [m]')
ax1.grid()

ax2.plot(array_angle, array_speed, 'r--')
ax2.set(ylabel='speed [m]', xlabel='angle [rad]')
ax2.grid()

# Set x-ticks at multiples of pi
ticks = np.linspace(0, 2*np.pi, 5)
labels = ['0', r'$\pi/2$', r'$\pi$', r'$3\pi/2$', r'$2\pi$']
plt.xticks(ticks, labels)

plt.show()
```

### Summary of Key Modifications:
- **Joint Type Changes**:
  - Crank-rod joint: `ChLinkRevolute()` replaced with `ChLinkBall()`.
  - Rod-piston joint: `ChLinkRevolute()` replaced with `ChLinkBall()`.
  - Piston-floor joint: `ChLinkLockPrismatic()` replaced with `ChLinkLockPlanePlane()`.
- **Rotation corrections**:
  - Rotated `mcrank` to align along Z using `Q_from_AngleAxis`.
  - Rotated `mpiston` to lie in x-y plane.
- **Initialization of plane-plane joint**:
  - Used frames with identity rotations (`Q_from_AngleAxis(0, VECT_Z)`), assuming zero relative orientation, but you could adjust the orientations if needed.
- **Logical corrections**:
  - Corrected method calls and parameters.
  - Included `plt.show()` at the end for proper plotting display.
