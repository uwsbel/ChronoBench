```python
import pychrono.core as chrono
import pychrono.irrlicht as irr

# 1. Initialize the Chrono system
system = chrono.ChSystemNSC()

# Set gravity
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# 2. Create the floor (truss)
floor = chrono.ChBodyEasyBox(10, 0.5, 10, 1000, True, True)
floor.SetPos(chrono.ChVectorD(0, -0.25, 0))
floor.SetBodyFixed(True)
system.Add(floor)

# 3. Create the crankshaft
# Dimensions
crank_radius = 0.2
crank_width = 0.1
crank_mass = 1.0

crank = chrono.ChBodyEasyCylinder(crank_radius, crank_width, crank_mass, True, True)
crank.SetPos(chrono.ChVectorD(0, 0.2, 0))
system.Add(crank)

# 4. Create the connecting rod
# Dimensions
rod_length = 0.5
rod_radius = 0.05
rod_mass = 0.5

connecting_rod = chrono.ChBodyEasyCylinder(rod_radius, rod_length, rod_mass, True, True)
connecting_rod.SetPos(chrono.ChVectorD(0, 0.2 - rod_length/2, 0))
system.Add(connecting_rod)

# 5. Create the piston
piston_radius = 0.15
piston_height = 0.1
piston_mass = 1.0

piston = chrono.ChBodyEasyCylinder(piston_radius, piston_height, piston_mass, True, True)
piston.SetPos(chrono.ChVectorD(0, -0.3, 0))
system.Add(piston)

# 6. Create joints

# a) Revolute joint for crankshaft (fixed to the ground)
pivot_point = chrono.ChVectorD(0, 0.2, 0)
revolute_crank = chrono.ChLinkLockRevolute()
revolute_crank.Initialize(crank, floor, chrono.ChCoordsysD(pivot_point))
system.Add(revolute_crank)

# b) Revolute joint between crank and connecting rod
# Attach at the end of the crank
crank_end = chrono.ChVectorD(crank_radius, 0.2, 0)
revolute_crank_rod = chrono.ChLinkLockRevolute()
revolute_crank_rod.Initialize(crank, connecting_rod, chrono.ChCoordsysD(crank_end))
system.Add(revolute_crank_rod)

# c) Revolute joint between connecting rod and piston
# Attach at the piston center
piston_joint_point = chrono.ChVectorD(0, -0.3 + piston_height/2, 0)
revolute_rod_piston = chrono.ChLinkLockRevolute()
revolute_rod_piston.Initialize(connecting_rod, piston, chrono.ChCoordsysD(piston_joint_point))
system.Add(revolute_rod_piston)

# 7. Add a motor to drive the crankshaft at constant angular speed
motor_speed = 2 * 3.14159  # 1 rotation per second

motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(crank, floor, chrono.ChCoordsysD(pivot_point))
motor.SetSpeed(motor_speed)
system.Add(motor)

# 8. Visualization setup with Irrlicht
application = irr.ChIrrApp(system, "Crank-Slider Mechanism", irr.dimension2du(800, 600))
application.AddTypicalSky()
application.AddTypicalCamera(irr.vector3df(0, 0.5, 2))
application.AddTypicalLights()

# Optional: Add textures or logos
# For example, set floor texture
texture_path = "textures/concrete.jpg"  # Placeholder path
# Note: Ensure the texture file exists or replace with a valid path
# application.GetVideoDriver().GetTexture(texture_path)

# 9. Run the simulation
application.AssetBindAll()
application.AssetUpdateAll()

while application.GetDevice().run():
    application.BeginScene()
    application.DrawAll()
    application.DoStep()
    application.EndScene()
```
