```python
import pychrono.core as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# Initialize Chrono system
chrono.SetChronoDataPath('path/to/data')  # Set the correct data path if needed
system = chrono.ChSystemNSC()

# Create the terrain
terrain = veh.RigidTerrain(system)
terrain.Initialize()

# Create the HMMWV vehicle
# Use the built-in HMMWV model
hmmwv = veh.HMMWV_Vehicle(system)
hmmwv.SetChassisCollisionType(veh.ChassisCollisionType_PRIMITIVES)
hmmwv.Initialize()

# Set vehicle position
initial_pos = chrono.ChVectorD(0, 0, 0.5)
hmmwv.GetChassisBody().SetPos(initial_pos)

# Create driver
driver = veh.HMMWV_Driver(hmmwv)
driver.SetInputMode(veh.HMMWV_Driver.InputMode_FORCE)
driver.Initialize()

# Set up Irrlicht visualization
application = irr.ChIrrApp(
    system,
    "HMMWV Simulation with Sensors",
    irr.dimension2du(1024, 768),
    irr.VerticalDir_UP
)
application.AddTypicalSky()
application.AddTypicalCamera(irr.vector3df(0, 3, -8))
application.AddTypicalLights()

# Attach the vehicle to the Irrlicht app
application.AssetBindAll()
application.AssetUpdateAll()

# Create IMU sensor attached to the vehicle's chassis
imu_sensor = veh.ChSensorManager(system)
imu_sensor.AddSensor(
    veh.ChSensor(
        veh.ChSensorType_IMU,
        hmmwv.GetChassisBody(),
        "IMU_Sensor"
    )
)

# Create GPS sensor attached to the vehicle's chassis
gps_sensor = veh.ChSensor(
    veh.ChSensorType_GPS,
    hmmwv.GetChassisBody(),
    "GPS_Sensor"
)
imu_sensor.AddSensor(gps_sensor)

# Initialize sensor manager
sensor_manager = veh.ChSensorManager(system)
sensor_manager.AddSensor(imu_sensor)
sensor_manager.Initialize()

# Function to process sensor data
def process_sensors():
    for sensor in sensor_manager.GetSensors():
        if sensor.GetType() == veh.ChSensorType_IMU:
            imu_data = sensor.GetSensorData()
            # Process IMU data (e.g., acceleration, angular velocity)
            print(f"IMU Data: Accel={imu_data.acceleration}, Gyro={imu_data.angular_velocity}")
        elif sensor.GetType() == veh.ChSensorType_GPS:
            gps_data = sensor.GetSensorData()
            # Process GPS data (e.g., position, velocity)
            print(f"GPS Position: {gps_data.position}, Velocity: {gps_data.velocity}")

# Simulation loop
time_step = 1e-3
total_time = 20  # seconds
current_time = 0

while application.GetDevice():
    # Synchronize driver inputs
    driver.Synchronize(current_time)
    driver.Advance(time_step)

    # Synchronize sensors
    sensor_manager.Synchronize(current_time)

    # Advance the vehicle system
    system.DoStepDynamics(time_step)

    # Process sensor data
    process_sensors()

    # Update visualization
    application.BeginScene()
    application.DrawAll()
    application.EndScene()

    # Advance Irrlicht application
    application.GetDevice().run()
    if not application.GetDevice().run():
        break

    # Output vehicle mass
    mass = hmmwv.GetChassisBody().GetMass()
    print(f"Vehicle mass: {mass} kg")

    current_time += time_step

# Cleanup
application.GetDevice().close()
```
