```python
import pychrono.core as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# 1. Initialize the Chrono engine and create the physical system
sys = chrono.ChSystemNSC()

# Set the contact method
sys.SetContactMethod(chrono.ChContactMethod_NSC)

# 2. Create the terrain
terrain = veh.RigidTerrain(sys)
# Define terrain dimensions
terrain_size_x = 50
terrain_size_y = 50
terrain_height = 0

# Add a flat terrain with texture
terrain.Initialize(terrain_size_x, terrain_size_y, 0.2)
terrain.SetTexture(veh.GetChronoDataFile('textures/concrete.jpg'))

# 3. Create the HMMWV vehicle
# Define initial position and orientation
initial_pos = chrono.ChVectorD(0, 0, 0.5)
initial_rot = chrono.ChQuaternionD(chrono.QUNIT)

# Create the vehicle
vehicle = veh.HMMWV(
    sys,
    veh.ChassisCollisionType_PRIMITIVES,
    veh.TireModel_TMEASY,
    False,  # use primitive visualization
    False   # use mesh visualization
)

# Set initial position and orientation
vehicle.SetChassisPosition(initial_pos)
vehicle.SetChassisRotation(initial_rot)

# Set contact method
vehicle.SetContactMethod(chrono.ChContactMethod_NSC)

# Initialize the vehicle
vehicle.Initialize()

# 4. Set up the Irrlicht visualization
application = irr.ChIrrApp(vehicle.GetSystem(), 'HMMWV Simulation', irr.dimension2du(1024, 768))
application.AddTypicalSky()
application.AddTypicalLights()
application.AddTypicalCamera(irr.vector3df(0, 5, -10))
application.AssetBindAll()
application.AssetUpdateAll()

# 5. Create an interactive driver
driver = veh.ChIrrGuiDriver(application)
driver.Initialize()

# 6. Simulation parameters
application.SetTimestep(1.0 / 50)  # 50 FPS
application.SetTryRealtime(True)

# 7. Run the simulation loop
while application.GetDevice().run():
    # Begin scene rendering
    application.BeginScene()
    application.DrawAll()

    # Get driver inputs
    driver.ReadInputs()

    # Apply driver inputs to vehicle
    vehicle.SetSteeringAngle(driver.GetSteering())
    vehicle.SetThrottle(driver.GetThrottle())
    vehicle.SetBraking(driver.GetBraking())

    # Advance the system
    sys.DoStepDynamics(application.GetTimestep())

    # End scene rendering
    application.EndScene()
```
