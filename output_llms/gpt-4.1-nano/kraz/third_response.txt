```python
import pychrono.core as chrono
import pychrono.irrlicht as irr
import pychrono.vehicle as veh
import math

# Set data paths
chrono.SetChronoDataPath(chrono.GetChronoDataPath())
veh.SetDataPath(chrono.GetChronoDataPath() + 'vehicle/')

# --- Initial vehicle locations and orientations ---

# Truck initial location and orientation
truck_initLoc = chrono.ChVector3d(0, 0, 0.5)
truck_initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# Sedan initial location and orientation (placed offset in Y for example)
sedan_initLoc = chrono.ChVector3d(5, 0, 0.5)
sedan_initRot = chrono.ChQuaterniond(1, 0, 0, 0)

# Visualization type for vehicle parts
vis_type = veh.VisualizationType_MESH

# Collision type for chassis
chassis_collision_type = veh.CollisionType_NONE

# Tire model type for truck (changed to RIGID)
tire_model_truck = veh.TireModelType_RIGID

# Terrain parameters
terrainHeight = 0
terrainLength = 100.0
terrainWidth = 100.0

# Track point for camera
trackPoint = chrono.ChVector3d(0, 0, 2.1)

# Contact method
contact_method = chrono.ChContactMethod_NSC
contact_vis = False

# Simulation step sizes
step_size = 1e-3
tire_step_size = step_size

# Render step size
render_step_size = 1.0 / 50  # FPS = 50

# --- Create the truck vehicle ---
truck = veh.Kraz()
truck.SetContactMethod(contact_method)
truck.SetChassisCollisionType(chassis_collision_type)
truck.SetChassisFixed(False)
truck.SetInitPosition(chrono.ChCoordsysd(truck_initLoc, truck_initRot))
truck.Initialize()

# Set visualization types
truck.SetChassisVisualizationType(vis_type, vis_type)
truck.SetSteeringVisualizationType(vis_type)
truck.SetSuspensionVisualizationType(vis_type, vis_type)
truck.SetWheelVisualizationType(vis_type, vis_type)
truck.SetTireVisualizationType(vis_type, vis_type)

# Set collision system type
truck.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# Set tire model for truck to RIGID
for wheel in truck.GetWheels():
    wheel.GetTire().SetTireModelType(tire_model_truck)

# --- Create the sedan vehicle ---
sedan = veh.Sedan()  # Assuming a Sedan class exists; if not, replace with appropriate vehicle class
sedan.SetContactMethod(contact_method)
sedan.SetChassisCollisionType(chassis_collision_type)
sedan.SetChassisFixed(False)
sedan.SetInitPosition(chrono.ChCoordsysd(sedan_initLoc, sedan_initRot))
sedan.Initialize()

# Set visualization types for sedan
sedan.SetChassisVisualizationType(vis_type, vis_type)
sedan.SetSteeringVisualizationType(vis_type)
sedan.SetSuspensionVisualizationType(vis_type, vis_type)
sedan.SetWheelVisualizationType(vis_type, vis_type)
sedan.SetTireVisualizationType(vis_type, vis_type)

# Set collision system type
sedan.GetSystem().SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)

# Set tire model for sedan to RIGID as well
for wheel in sedan.GetWheels():
    wheel.GetTire().SetTireModelType(veh.TireModelType_RIGID)

# --- Create terrain with highway mesh ---
patch_mat = chrono.ChContactMaterialNSC()
patch_mat.SetFriction(0.9)
patch_mat.SetRestitution(0.01)

terrain = veh.RigidTerrain(truck.GetSystem())

# Use a predefined highway mesh (assuming such a mesh exists in data files)
# For example, using a mesh file "highway.obj"
mesh_filename = 'terrain/textures/highway.obj'
patch = terrain.AddPatch(patch_mat, 
    chrono.ChCoordsysd(chrono.ChVector3d(0, 0, 0), chrono.QUNIT), 
    terrainLength, terrainWidth, mesh_filename)

patch.SetTexture(veh.GetDataFile("terrain/textures/tile4.jpg"), 200, 200)
patch.SetColor(chrono.ChColor(0.8, 0.8, 0.5))
terrain.Initialize()

# --- Create visualization for truck ---
vis_truck = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis_truck.SetWindowTitle('Kraz Truck Demo')
vis_truck.SetWindowSize(1280, 1024)
vis_truck.SetChaseCamera(trackPoint, 25.0, 1.5)
vis_truck.Initialize()
vis_truck.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis_truck.AddLightDirectional()
vis_truck.AddSkyBox()
vis_truck.AttachVehicle(truck.GetTractor())

# --- Create visualization for sedan ---
vis_sedan = veh.ChWheeledVehicleVisualSystemIrrlicht()
vis_sedan.SetWindowTitle('Sedan Demo')
vis_sedan.SetWindowSize(1280, 1024)
# Place camera differently for sedan if desired
vis_sedan.SetChaseCamera(chrono.ChVector3d(5, 0, 2.0), 20.0, 1.5)
vis_sedan.Initialize()
vis_sedan.AddLogo(chrono.GetChronoDataFile('logo_pychrono_alpha.png'))
vis_sedan.AddLightDirectional()
vis_sedan.AddSkyBox()
vis_sedan.AttachVehicle(sedan.GetTractor())

# --- Create driver systems ---
driver_truck = veh.ChInteractiveDriverIRR(vis_truck)
driver_sedan = veh.ChInteractiveDriverIRR(vis_sedan)

# Set driver input response times
steering_time = 1.0
throttle_time = 1.0
braking_time = 0.3

driver_truck.SetSteeringDelta(render_step_size / steering_time)
driver_truck.SetThrottleDelta(render_step_size / throttle_time)
driver_truck.SetBrakingDelta(render_step_size / braking_time)

driver_sedan.SetSteeringDelta(render_step_size / steering_time)
driver_sedan.SetThrottleDelta(render_step_size / throttle_time)
driver_sedan.SetBrakingDelta(render_step_size / braking_time)

driver_truck.Initialize()
driver_sedan.Initialize()

# --- Output vehicle mass ---
print("VEHICLE MASS (Truck):", truck.GetTractor().GetMass())
print("VEHICLE MASS (Sedan):", sedan.GetTractor().GetMass())

# --- Simulation parameters ---
render_steps = math.ceil(render_step_size / step_size)

# Initialize timer
realtime_timer = chrono.ChRealtimeStepTimer()

# Initialize step counter
step_number = 0
render_frame = 0

# Variables to store truck's tractor and trailer states
truck_tractor_state = None
truck_trailer_state = None

while vis_truck.Run() and vis_sedan.Run():
    time = truck.GetSystem().GetChTime()

    # Render scene at specified intervals
    if (step_number % render_steps == 0):
        vis_truck.BeginScene()
        vis_truck.Render()
        vis_truck.EndScene()

        vis_sedan.BeginScene()
        vis_sedan.Render()
        vis_sedan.EndScene()

        render_frame += 1

    # Get driver inputs
    driver_inputs_truck = driver_truck.GetInputs()
    driver_inputs_sedan = driver_sedan.GetInputs()

    # Synchronize modules
    driver_truck.Synchronize(time)
    driver_sedan.Synchronize(time)
    terrain.Synchronize(time)
    truck.Synchronize(time, driver_inputs_truck, terrain)
    sedan.Synchronize(time, driver_inputs_sedan, terrain)
    vis_truck.Synchronize(time, driver_inputs_truck)
    vis_sedan.Synchronize(time, driver_inputs_sedan)

    # Advance modules
    driver_truck.Advance(step_size)
    driver_sedan.Advance(step_size)
    terrain.Advance(step_size)
    truck.Advance(step_size)
    sedan.Advance(step_size)
    vis_truck.Advance(step_size)
    vis_sedan.Advance(step_size)

    # Store the state of the truck's tractor and trailer
    # Assuming GetTractor() returns a pointer to the vehicle node
    truck_tractor = truck.GetTractor()
    # Save position, orientation, velocity, etc.
    truck_tractor_state = {
        'pos': truck_tractor.GetPos(),
        'rot': truck_tractor.GetRot(),
        'vel': truck_tractor.GetPos_dt(),
        'ang_vel': truck_tractor.GetWvel_loc()
    }
    # For trailer, assuming a method exists or get trailer node
    # If trailer exists, replace with actual method
    # For example:
    # trailer = truck.GetTrailer()
    # if trailer:
    #     truck_trailer_state = {
    #         'pos': trailer.GetPos(),
    #         'rot': trailer.GetRot(),
    #         'vel': trailer.GetPos_dt(),
    #         'ang_vel': trailer.GetWvel_loc()
    #     }

    # Increment step
    step_number += 1

    # Spin for real-time synchronization
    realtime_timer.Spin(step_size)
```
