```python
import pychrono.core as chrono
import pychrono.vehicle as veh
import pychrono.irrlicht as irr

# 1. Initialize the Chrono engine and create the physical system
system = chrono.ChSystemNSC()

# Set gravity
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# 2. Create and configure the terrain
terrain = veh.RigidTerrain(system)
terrain.SetContactFriction(0.8)  # example friction coefficient
terrain.SetContactRestitution(0.1)  # example restitution coefficient
terrain.Initialize()

# Add a flat terrain mesh (optional, for visualization)
terrain.GetGroundBody().GetCollisionModel().ClearModel()
terrain.GetGroundBody().GetCollisionModel().AddBox(10, 0.1, 10, chrono.ChVectorD(0, -0.1, 0))
terrain.GetGroundBody().GetCollisionModel().BuildModel()
terrain.GetGroundBody().SetPos(chrono.ChVectorD(0, 0, 0))
terrain.GetGroundBody().SetBodyFixed(True)

# 3. Create the M113 vehicle
# Load the vehicle from the provided model or define parameters
m113 = veh.M113_Vehicle(system)

# Set initial position and orientation
initial_pos = chrono.ChVectorD(0, 0.5, 0)  # starting slightly above ground
initial_rot = chrono.Q_from_AngleAxis(0, chrono.ChVectorD(0, 1, 0))
m113.GetChassisBody().SetPos(initial_pos)
m113.GetChassisBody().SetRot(initial_rot)

# Initialize the vehicle
m113.Initialize()

# 4. Set up the driver system
driver = veh.ChIrrGuiDriver()
driver.Initialize(system, m113.GetChassisBody())

# 5. Set up Irrlicht visualization
vis = irr.ChVisualSystemIrrlicht()
vis.AttachSystem(system)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle('PyChrono M113 Vehicle Simulation')
vis.Initialize()

# Configure camera
cam_pos = chrono.ChVectorD(0, 2, -5)
cam_target = chrono.ChVectorD(0, 0, 0)
vis.GetActiveCamera().SetPos(cam_pos)
vis.GetActiveCamera().SetTarget(cam_target)
vis.AddLightWithShadow(chrono.ChVectorD(10, 10, 10), chrono.ChVectorD(0, 0, 0), 10, 2, 20, 40)

# 6. Simulation parameters
time_step = 1e-3  # 1 ms timestep
max_time = 20  # total simulation time in seconds
current_time = 0

# 7. Simulation loop
while vis.Run():
    # Synchronize driver inputs
    driver.Synchronize(current_time)
    driver.Advance(time_step)

    # Synchronize vehicle with driver inputs
    m113.Synchronize(driver)
    m113.Advance(time_step)

    # Synchronize terrain
    terrain.Synchronize(current_time)

    # Advance the physical system
    system.DoStepDynamics(time_step)

    # Update visualization
    vis.Render()

    # Increment time
    current_time += time_step

    # Optional: add a real-time sleep to match simulation speed
    # import time
    # time.sleep(time_step)

# End of simulation
```
