```python
import os
import math
import random
import numpy as np
import pychrono as chrono
import pychrono.robot as turtlebot
import pychrono.sensor as sens  # Added sensor module
from pychrono import irrlicht as chronoirr

# Create Chrono system
system = chrono.ChSystemNSC()
system.SetCollisionSystemType(chrono.ChCollisionSystem.Type_BULLET)
system.SetGravitationalAcceleration(chrono.ChVector3d(0, 0, -9.81))  # Set gravity in negative Z direction
chrono.ChCollisionModel.SetDefaultSuggestedEnvelope(0.0025)
chrono.ChCollisionModel.SetDefaultSuggestedMargin(0.0025)

# Create ground body as terrain plane that robot will drive on
ground_mat = chrono.ChContactMaterialNSC()
# Corrected ground position from z=-1 to z=-0.6
ground = chrono.ChBodyEasyBox(20, 20, 1, 1000, True, True, ground_mat)
ground.SetPos(chrono.ChVector3d(0, 0, -0.6))
ground.SetFixed(True)  # Fix the ground in place
ground.GetVisualShape(0).SetTexture(chrono.GetChronoDataFile("textures/concrete.jpg"))
system.Add(ground)

# Create Turtlebot Robot
init_pos = chrono.ChVector3d(0, 0.2, 0)  # Initial position of the robot
init_rot = chrono.ChQuaterniond(1, 0, 0, 0)  # Initial orientation of the robot
robot = turtlebot.TurtleBot(system, init_pos, init_rot)  # Create Turtlebot instance
robot.Initialize()  # Initialize the robot

# Add sensor manager and configure lidar sensor
sensor_manager = sens.ChSensorManager(system)
sensor_manager.Initialize()

# Create and add a lidar sensor
lidar = sens.ChLidarSensor(
    robot.GetChassisBody(),  # Attach to robot chassis
    0.5,  # offset in meters
    0,  # yaw
    270,  # scan angle in degrees
    0.2,  # max range in meters
    0.1,  # resolution in radians
    0.2,  # frequency in Hz
)
# Configure filters if needed (e.g., noise filter)
# For example, add a noise filter
# lidar.AddFilter(sens.ChFilterNoise())

sensor_manager.AddSensor(lidar)

# Add randomly placed boxes
num_boxes = 5
boxes = []
for _ in range(num_boxes):
    x = random.uniform(-5, 5)
    y = random.uniform(-5, 5)
    z = 0.5  # height of boxes
    box = chrono.ChBodyEasyBox(0.2, 0.2, 0.2, 100, True, True)
    box.SetPos(chrono.ChVector3d(x, y, z))
    box.SetMass(1)
    system.Add(box)
    boxes.append(box)

# Define motion control function for Turtlebot
def move(mode):
    LEFT_WHEEL = 0
    RIGHT_WHEEL = 1
    if mode == 'straight':
        robot.SetMotorSpeed(math.pi / 4, LEFT_WHEEL)
        robot.SetMotorSpeed(math.pi / 4, RIGHT_WHEEL)
    elif mode == 'left':
        robot.SetMotorSpeed(math.pi / 8, LEFT_WHEEL)
        robot.SetMotorSpeed(-math.pi / 8, RIGHT_WHEEL)
    elif mode == 'right':
        robot.SetMotorSpeed(-math.pi / 8, LEFT_WHEEL)
        robot.SetMotorSpeed(math.pi / 8, RIGHT_WHEEL)
    else:
        # Stop
        robot.SetMotorSpeed(0, LEFT_WHEEL)
        robot.SetMotorSpeed(0, RIGHT_WHEEL)

# Set the simulation time step
time_step = 2e-3

# Simulation loop
time = 0
while vis.Run():
    # Define WheelID enum values for motor control
    LEFT_DRIVE_WHEEL = 0
    RIGHT_DRIVE_WHEEL = 1

    # Example movement: move straight for first second, then turn left, then right
    if time < 1.0:
        move('straight')
    elif abs(time - 1.0) < 1e-4:
        move('left')
    elif abs(time - 2.0) < 1e-4:
        move('right')
    else:
        move('straight')  # Continue straight after

    # Update sensor manager (to get new data)
    sensor_manager.Update()

    # Increment time counter
    time += time_step

    # Render the scene
    vis.BeginScene()
    vis.Render()
    vis.EndScene()

    # Advance the simulation by one time step
    system.DoStepDynamics(time_step)
```
