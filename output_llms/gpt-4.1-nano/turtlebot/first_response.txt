```python
import pychrono.core as chrono
import pychrono.irrlicht as irr

# 1. Initialize the Chrono system
system = chrono.ChSystemNSC()
system.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))  # Set gravity

# 2. Create the ground (rigid terrain)
ground_body = chrono.ChBodyEasyBox(10, 0.2, 10, 1000, True, True)
ground_body.SetPos(chrono.ChVectorD(0, -0.1, 0))
ground_body.SetBodyFixed(True)
system.Add(ground_body)

# 3. Initialize the Turtlebot robot
# For simplicity, model the robot as a box with two wheels
# Create chassis
chassis = chrono.ChBodyEasyBox(0.3, 0.1, 0.2, 500, True, True)
chassis.SetPos(chrono.ChVectorD(0, 0.2, 0))
system.Add(chassis)

# Create left wheel
left_wheel = chrono.ChBodyEasyCylinder(0.05, 0.02, 200, True, True)
left_wheel.SetPos(chrono.ChVectorD(-0.1, 0.05, 0.1))
system.Add(left_wheel)

# Create right wheel
right_wheel = chrono.ChBodyEasyCylinder(0.05, 0.02, 200, True, True)
right_wheel.SetPos(chrono.ChVectorD(0.1, 0.05, 0.1))
system.Add(right_wheel)

# Create joints to connect wheels to chassis
# Left wheel joint
left_joint = chrono.ChLinkLockRevolute()
left_joint.Initialize(left_wheel, chassis, chrono.ChCoordsysD(chrono.ChVectorD(-0.1, 0.05, 0), chrono.Q_from_AngX(0)))
system.Add(left_joint)

# Right wheel joint
right_joint = chrono.ChLinkLockRevolute()
right_joint.Initialize(right_wheel, chassis, chrono.ChCoordsysD(chrono.ChVectorD(0.1, 0.05, 0), chrono.Q_from_AngX(0)))
system.Add(right_joint)

# 4. Set up visualization with Irrlicht
application = irr.ChIrrApp(system, "Turtlebot Simulation", irr.dimension2du(800, 600))
application.AddTypicalSky()
application.AddTypicalCamera(irr.vector3df(0, 1, -2))
application.AddLightWithShadow(irr.vector3df(4, 4, -4), irr.vector3df(0, 0, 0), 3, 8, 0, 0.5, 0.5)

application.AssetBindAll()
application.AssetUpdateAll()

# 5. Define motor control parameters
# For simplicity, control the wheel velocities directly
left_wheel_speed = 0.0
right_wheel_speed = 0.0

# Define turn commands at specific times
# For example: go straight for 2 seconds, turn left for 2 seconds, then straight again
def update_wheel_speeds(time):
    if time < 2:
        # Move straight
        return 1.0, 1.0
    elif 2 <= time < 4:
        # Turn left: left wheel slow, right wheel fast
        return 0.2, 1.0
    else:
        # Move straight again
        return 1.0, 1.0

# 6. Simulation loop
time_step = 1e-3
total_time = 6.0
current_time = 0.0

while application.GetDevice().run():
    # Update control commands based on current time
    left_wheel_speed, right_wheel_speed = update_wheel_speeds(current_time)

    # Apply motor torques or set wheel velocities
    # Here, we set the angular velocity of wheels directly
    left_wheel.SetWvel(chrono.ChVectorD(0, left_wheel_speed, 0))
    right_wheel.SetWvel(chrono.ChVectorD(0, right_wheel_speed, 0))

    # Advance the simulation
    system.DoStepDynamics(time_step)

    # Begin scene rendering
    application.BeginScene()
    application.DrawAll()
    application.EndScene()

    current_time += time_step
```
