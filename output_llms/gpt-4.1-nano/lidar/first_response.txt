```python
import pychrono as chrono
from pychrono.sensor import ChSensorManager, ChLidarSensor, ChSensorFilterNoiseModel, ChSensorFilterVisualizer, ChSensorSave
import math

# 1. Initialize the Chrono engine and system
chrono.SetChronoDataPath('your/data/path')  # Set path if needed
system = chrono.ChSystemNSC()

# 2. Load the triangular mesh from a Wavefront .obj file
# Replace 'path_to_mesh.obj' with your actual file path
mesh_filename = 'path_to_mesh.obj'
mesh_body = chrono.ChBodySimple()
mesh_body.SetPos(chrono.ChVectorD(0, 0, 0))
mesh_body.SetBodyFixed(True)

# Load mesh as visual shape
mesh_shape = chrono.ChTriangleMeshShape()
mesh_shape.SetMesh(chrono.ChTriangleMesh().LoadWavefront(mesh_filename))
mesh_body.AddVisualShape(mesh_shape)

# Optionally, add collision shape if needed
collision_shape = chrono.ChTriangleMeshShape()
collision_shape.SetMesh(chrono.ChTriangleMesh().LoadWavefront(mesh_filename))
mesh_body.GetCollisionModel().ClearModel()
collision_shape.AddTriangleMeshGeometry(chrono.VNULL,  # no position offset
                                       chrono.QUNIT,  # no rotation
                                       True)        # is dynamic? For fixed, set False
mesh_body.GetCollisionModel().AddTriangleMesh(collision_shape, False, False)
mesh_body.GetCollisionModel().BuildModel()
mesh_body.SetCollide(True)

system.Add(mesh_body)

# 3. Initialize the sensor manager
sensor_manager = ChSensorManager()
sensor_manager.scene = system

# 4. Add a lidar sensor to the mesh body
lidar_name = 'my_lidar'
lidar_range = 50.0  # meters
lidar_h_fov = math.radians(90)  # 90 degrees horizontal FOV
lidar_v_fov = math.radians(30)  # 30 degrees vertical FOV
lidar_h_res = 800  # horizontal resolution
lidar_v_res = 200  # vertical resolution

lidar_pose = chrono.ChFrameD(chrono.ChVectorD(0, 1, 0), chrono.QUNIT)  # Initial pose; will be updated
lidar_sensor = ChLidarSensor(
    mesh_body,  # attach to the mesh body
    lidar_pose,
    lidar_range,
    lidar_h_fov,
    lidar_v_fov,
    lidar_h_res,
    lidar_v_res
)

# Enable visualizations and noise filters
lidar_sensor.GetScanIO().EnableFilter(ChSensorFilterNoiseModel())
lidar_sensor.GetScanIO().EnableFilter(ChSensorFilterVisualizer())

# Enable saving of lidar data
lidar_sensor.GetScanIO().EnableSave(
    ChSensorSave.Format.JSON,         # Save format (other options: CSV, etc.)
    "lidar_output.json"              # Output filename
)

sensor_manager.AddSensor(lidar_sensor)

# 5. Simulation loop with dynamic lidar positioning
time_end = 10  # seconds
dt = 0.01     # time step
t = 0

import matplotlib.pyplot as plt

while t < time_end:
    # Compute a circular orbit around the mesh at a fixed height
    radius = 2.0  # meters from the center
    angular_speed = math.pi / 5  # radians per second
    angle = angular_speed * t

    # New lidar position in orbit
    x_pos = radius * math.cos(angle)
    y_pos = 1.5  # fixed height
    z_pos = radius * math.sin(angle)
    new_pos = chrono.ChVectorD(x_pos, y_pos, z_pos)
    new_rot = chrono.Q_from_AngAxis(0, chrono.ChVectorD(0, 1, 0))  # No rotation

    # Update lidar sensor pose
    lidar_sensor.Pose = chrono.ChFrameD(new_pos, new_rot)

    # Update sensor scene with new pose
    sensor_manager.Synchronize(t)
    sensor_manager.Advance(dt)

    # Retrieve and print lidar buffer data
    if lidar_sensor.GetScanIO().HasValidFrame():
        # For example, get point cloud data
        points = lidar_sensor.GetScanIO().GetPointCloud()
        print(f"Time {t:.2f}s: Lidar point cloud with {len(points)} points.")

    system.DoStepDynamics(dt)
    t += dt
```
