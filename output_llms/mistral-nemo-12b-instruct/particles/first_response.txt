Here's a Python script using PyChrono to simulate the gravitational attraction between particles with an emitter that generates particles with random shapes, positions, velocities, and orientations. The simulation is visualized using Irrlicht, and custom gravitational forces are applied to simulate particle attraction dynamically.

```python
import numpy as np
import pychrono as pc
import pychrono.visualization.irrlicht as irr

# Initialize PyChrono environment and core components
pc.set_log_level(pc.LogLevel.WARNING)
sim = pc.Simulation()
scene = pc.Scene()
collision_world = pc.CollisionWorld()
collision_world.set_scene(scene)

# Set up Irrlicht visualization
irr_app = irr.create_irrlicht_application()
irr_app.set_scene(scene)

# Define particle emitter parameters
emitter_pos = np.array([0, 0, 0])
emitter_vel = np.array([0, 0, 0])
emitter_shape = pc.Sphere(0.1)
emitter_orientation = pc.Quaternion()
emitter_spawn_rate = 50
emitter_lifetime = 10.0

# Define particle parameters
particle_mass = 1.0
particle_shape = pc.Sphere(0.05)
particle_color = pc.Color(1, 1, 0, 1)
gravity_constant = 6.67430e-11

# Create particle emitter
emitter = pc.Emitter(emitter_pos, emitter_vel, emitter_shape, emitter_orientation, emitter_spawn_rate, emitter_lifetime)
scene.add(emitter)

# Define a function to calculate gravitational force between two particles
def gravitational_force(p1, p2):
    r = p2.position - p1.position
    distance = np.linalg.norm(r)
    force_magnitude = (gravity_constant * p1.mass * p2.mass) / (distance ** 2)
    force_direction = r / distance
    return force_magnitude * force_direction

# Simulation loop
while True:
    # Update emitter and particles
    emitter.update()
    for particle in emitter.particles:
        # Apply gravitational forces from other particles
        for other_particle in emitter.particles:
            if particle != other_particle:
                force = gravitational_force(particle, other_particle)
                particle.apply_force(force)

        # Update particle position and velocity
        particle.update()

    # Update collision world and visualization
    collision_world.update()
    irr_app.update()

    # Break simulation after a certain time
    if sim.time() > 100.0:
        break

# Clean up and close Irrlicht application
irr_app.close()
```

This script creates a particle emitter that spawns particles with random shapes, positions, velocities, and orientations. It then applies custom gravitational forces between particles to simulate their attraction dynamically. The simulation is visualized using Irrlicht, and the particles are colored yellow for better visibility. The simulation runs for 100 seconds before stopping.
