```python
import pychrono as chrono
import numpy as np

# Initialize PyChrono environment
chrono.setChronoDataPath("path/to/chrono_data")

# Create a simulation and set up the scene
sim = chrono.ChSimulation()
scene = chrono.ChIrrApp(sim, 'Particle Attraction Simulation')
scene.addLight(chrono.ChLightDir())
scene.setCamera(chrono.ChVectorD(1, 1, 1), chrono.ChVectorD(0, 0, 0))

# Create a particle emitter
emitter = chrono.ChParticleEmitter()
emitter.SetEmissionRate(100)  # Emit 100 particles per second
emitter.SetEmissionDuration(chrono.GetChronoTimeStep())  # Emit particles for one time step
emitter.SetEmissionDirection(chrono.ChVectorD(0, 1, 0))  # Emit particles upwards
emitter.SetEmissionPosition(chrono.ChVectorD(0, 1, 0))  # Emit particles from the origin
sim.Add(emitter)

# Create a particle system
particle_system = chrono.ChParticleSystem()
particle_system.SetParticleRadius(0.05)  # Set particle radius to 0.05 meters
particle_system.SetMaxNumParticles(1000)  # Set maximum number of particles to 1000
particle_system.SetGravity(chrono.ChVectorD(0, -9.81, 0))  # Set gravity to -9.81 m/s^2
sim.Add(particle_system)

# Create a custom gravitational force function
def custom_gravity(particle, force):
    position = particle.GetPos()
    force[0] = 0
    force[1] = -9.81 * position.y
    force[2] = 0

# Set the custom gravitational force function
particle_system.SetCustomGravityFunction(custom_gravity)

# Run the simulation
while scene.Run():
    sim.DoStep()
    scene.Render()
```
