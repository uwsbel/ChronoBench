Here's a Python script using PyChrono and Irrlicht to simulate a CityBus vehicle on a rigid terrain according to the given instructions. Make sure you have both PyChrono and Irrlicht installed (`pip install pychrono irrlicht`).

```python
import pychrono as pc
import pychrono.irrlicht as pli
import numpy as np

# 1. Initialize the PyChrono environment and core components.
sim = pc.Simulation()
engine = pli.Engine([800, 600], 50, "CityBus Simulation")
scene = engine.getScene()
camera = scene.getCamera()

# 2. Add the required physical systems and objects.

# Vehicle
bus = pc.CityBus()
bus.set_position([0, 0, 0.5])
bus.set_orientation([0, 0, 0])
bus.set_tire_model(pc.TireModel.MagicFormula())
sim.add(bus)

# RigidTerrain
terrain = pc.RigidTerrain([-10, -10, 0], [20, 20, 1])
terrain.set_texture("path/to/terrain_texture.png")  # Replace with your texture path
sim.add(terrain)

# 3. Set necessary default parameters.

# Mesh and primitive visualization types
bus_mesh = pli.Mesh("path/to/bus_mesh.obj")  # Replace with your bus mesh path
bus_node = scene.addMeshSceneNode(bus_mesh, scene.getRootSceneNode(), -1, pc.vec3(bus.get_position()))
bus_node.setPosition(pc.vec3(bus.get_position()))
bus_node.setRotation(pc.vec3(bus.get_orientation()))

wheel_nodes = []
for i in range(4):  # Assuming 4 wheels
    wheel_node = scene.addSphereSceneNode(0.2, scene.getRootSceneNode(), -1)
    wheel_node.setPosition(pc.vec3(bus.get_wheel_positions()[i]))
    wheel_nodes.append(wheel_node)

# Camera
camera.setPosition(pc.vec3f(0, -5, 3))
camera.setTarget(pc.vec3f(0, 0, 0))

# Interactive driver system
throttle = 0
steering = 0
braking = 0

def on_key_press(event):
    global throttle, steering, braking
    if event.KeyCode == pli.KEY_KEY_W:
        throttle = 1
    elif event.KeyCode == pli.KEY_KEY_S:
        braking = 1
    elif event.KeyCode == pli.KeyCode.KEY_A:
        steering = -1
    elif event.KeyCode == pli.KeyCode.KEY_D:
        steering = 1

engine.getEventReceiver().onKeyPress = on_key_press

# Simulation loop
while engine.run():
    # Update vehicle dynamics
    bus.set_throttle(throttle)
    bus.set_steering(steering)
    bus.set_braking(braking)
    sim.step()

    # Update visualization
    bus_node.setPosition(pc.vec3(bus.get_position()))
    bus_node.setRotation(pc.vec3(bus.get_orientation()))
    for i, node in enumerate(wheel_nodes):
        node.setPosition(pc.vec3(bus.get_wheel_positions()[i]))

    engine.getEventReceiver().processEvents()
    engine.getDriver().beginScene(True, True, pc.Color(0, 0, 0))
    scene.drawAll()
    engine.getDriver().endScene()
```

This script initializes the simulation environment, adds a CityBus vehicle and a RigidTerrain with a custom texture, sets up mesh and primitive visualization types for the vehicle, and implements an interactive driver system using keyboard inputs. The simulation loop runs at 50 frames per second, updating the vehicle dynamics and rendering the scene in real-time.
