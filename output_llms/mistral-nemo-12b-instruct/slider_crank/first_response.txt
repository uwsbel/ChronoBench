Here's a Python script using PyChrono to create a crank-slider mechanism simulation as per your instructions. Note that PyChrono uses Chrono::Engine for physics simulation and Irrlicht for visualization, so you'll need to have both installed.

```python
import chrono.core as cq
import chrono.irrlicht as ci
import chrono.physics as cp
import chrono.postprocess as cpp

# 1. Initialize PyChrono environment and core components
sim = cp.ChSimulation()
irr = ci.ChIrrApp(sim, 'Crank-Slider Mechanism', cq.Vec3d(1024, 768))
irr.SetCamera(cq.Vec3d(1.5, 1.5, 1.5), cq.Vec3d(0, 0, 0))

# 2. Add physical systems and objects

# Floor (truss)
floor = cp.ChBodyEasyBox(10, 1, 10, 1000, True, True)
floor.SetPos(cq.Vec3d(0, -0.5, 0))
floor.SetBodyFixed(True)
sim.Add(floor)

# Crankshaft
crank = cp.ChBodyEasyBox(0.2, 0.2, 0.1, 1, True, True)
crank.SetPos(cq.Vec3d(0, 0.5, 0))
crank.SetRot(cq.Q_from_AngX(cq.C_DTR(45)))
sim.Add(crank)

# Connecting rod
rod = cp.ChBodyEasyBox(0.1, 0.1, 1, 0.2, True, True)
rod.SetPos(cq.Vec3d(0.2, 0.5, 0))
rod.SetRot(cq.Q_from_AngX(cq.C_DTR(45)))
sim.Add(rod)

# Piston
piston = cp.ChBodyEasyBox(0.2, 0.2, 0.1, 1, True, True)
piston.SetPos(cq.Vec3d(1, 0.5, 0))
sim.Add(piston)

# Motor for crankshaft
motor = cp.ChLinkMotorAngle()
motor.Initialize(crank, None, cp.ChFrameD(cq.Vec3d(0, 0.5, 0), cq.Q_from_AngX(cq.C_DTR(45))))
motor.SetTargetSpeed(2)  # Angular speed in rad/s
sim.Add(motor)

# Revolute joints
joint1 = cp.ChLinkRevolute()
joint1.Initialize(crank, rod, cp.ChFrameD(cq.Vec3d(0.2, 0.5, 0), cq.Q_from_AngX(cq.C_DTR(45))))
sim.Add(joint1)

joint2 = cp.ChLinkRevolute()
joint2.Initialize(rod, piston, cp.ChFrameD(cq.Vec3d(1, 0.5, 0), cq.Q_from_AngX(cq.C_DTR(45))))
sim.Add(joint2)

# 3. Set necessary default parameters

# Visualization settings
irr.SetCamera(cq.Vec3d(1.5, 1.5, 1.5), cq.Vec3d(0, 0, 0))
irr.SetLightDirection(cq.Vec3d(-1, -1, -1))
irr.SetLightColor(cq.Color(1, 1, 1))
irr.AddLogo(cq.Color(1, 1, 1))

# Simulation settings
sim.SetSolverType(cp.ChSolverType_SOR)
sim.SetSolverSORIterations(50)
sim.SetSolverSORRelTol(1e-3)
sim.SetSolverSORAbsTol(1e-4)

# Run the simulation
for i in range(1000):
    sim.DoStepDynamics(0.01)
    irr.DoStep()
```

This script creates a simple crank-slider mechanism with a floor, crankshaft, connecting rod, and piston. The crankshaft is driven by a motor at a constant angular speed of 2 rad/s. The simulation is visualized using the Irrlicht visualization system with customizable camera settings, lighting, and an added logo.
