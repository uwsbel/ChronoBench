Here's a Python script using PyChrono to simulate a Jeffcott rotor using an IGA beam. Make sure you have PyChrono installed (`pip install pychrono`).

```python
import pychrono as pc
import pychrono.beams as pb
import pychrono.visualization as pv
import numpy as np

# 1. Initialize PyChrono environment and core components
pc.init()

# 2. Add required physical systems and objects

# Create a rigid body scene
scene = pc.Scene()

# Create an IGA beam
beam = pb.Beam(beam_type=pb.BeamType.IGA,
               length=1.0,
               radius=0.05,
               young_modulus=1e7,
               density=1000,
               num_elements=10)
scene.add(beam)

# Create a flywheel at the center of the beam
flywheel = pc.RigidBody(name="flywheel")
flywheel.shape = pc.Sphere(radius=0.1)
flywheel.mass = 1.0
flywheel.inertia = pc.Inertia(mass=flywheel.mass, moment_of_inertia=flywheel.mass * 0.1 ** 2 / 2)
flywheel.pos = np.array([0.0, 0.0, 0.0])
scene.add(flywheel)

# Attach the flywheel to the beam
beam.attach_fixed(flywheel, pos=[0.0, 0.0, 0.0])

# Create a rotational motor at one end of the beam
motor = pc.Motor(name="motor")
motor.pos = np.array([-0.5, 0.0, 0.0])
motor.rot = pc.Quaternion(axis="z", angle=0.0)
motor.angular_velocity = 10.0
motor.torque = 1.0
scene.add(motor)

# Attach the motor to the beam
beam.attach_rotational(motor, pos=[-0.5, 0.0, 0.0])

# 3. Set necessary default parameters

# Set gravity
scene.gravity = np.array([0.0, -9.81, 0.0])

# Set time step and simulation duration
dt = 0.01
sim_time = 5.0

# Create a visualizer with FEM visualization and Irrlicht window
vis = pv.Vis()
vis.add_fem_visualization(beam, color=[1.0, 0.0, 0.0])
vis.add_rigid_body_visualization(flywheel, color=[0.0, 1.0, 0.0])
vis.add_rigid_body_visualization(motor, color=[0.0, 0.0, 1.0])
vis.create_irrlicht_window()

# Run the simulation
for t in np.arange(0.0, sim_time, dt):
    scene.step(dt)
    vis.update()

# Clean up
pc.quit()
```

This script creates a Jeffcott rotor system with an IGA beam, a flywheel at the center, and a rotational motor driving one end of the beam. The simulation is visualized using FEM visualization for the beam and rigid body visualization for the flywheel and motor, with an Irrlicht window to observe the dynamics.
