Here's a Python script using PyChrono to create the described simulation. Make sure you have PyChrono and its dependencies installed (`pip install pychrono`).

```python
import pychrono as pc
import numpy as np
import os

# 1. Initialize PyChrono environment and core components
pc.init()
sim = pc.Simulation()
scene = pc.Scene()
sensor_manager = pc.SensorManager()

# 2. Add physical systems and objects
# Load triangular mesh from Wavefront .obj file
mesh_path = "path/to/your/mesh.obj"  # Replace with your .obj file path
mesh = pc.Mesh(mesh_path)
mesh.set_material(pc.Material(ambient=0.5, diffuse=0.5, specular=0.5))
mesh.set_position([0, 0, 0])
scene.add(mesh)

# Create a fixed body for the mesh
body = pc.Body()
body.add_shape(mesh)
body.set_fixed(True)
scene.add(body)

# Add a lidar sensor to the body
lidar = pc.Lidar()
lidar.set_position([0, 0, 1])  # Initial position above the mesh
lidar.set_rotation([0, 0, 0])  # Initial orientation
lidar.set_range(5)  # Set lidar range
lidar.set_fov(30)  # Set field of view
body.add_shape(lidar)

# Add noise filter and visualization to the lidar sensor
noise_filter = pc.NoiseFilter(lidar)
visualization = pc.LidarVisualization(lidar)
sensor_manager.add_sensors([noise_filter, visualization])

# Set up lidar data saving options
data_dir = "lidar_data"
if not os.path.exists(data_dir):
    os.makedirs(data_dir)
saving_options = pc.LidarSavingOptions(data_dir=data_dir)

# 3. Set necessary default parameters
sim.set_scene(scene)
sim.set_sensor_manager(sensor_manager)
sim.set_lidar_saving_options(saving_options)

# Define a function to update the lidar's position in an orbit around the mesh
def update_lidar_position(t):
    angle = t * 0.5  # Angular velocity in radians per second
    lidar.set_position([np.cos(angle), np.sin(angle), 1])
    lidar.set_rotation([angle, 0, 0])

# Set up simulation parameters
sim.set_dt(0.01)  # Timestep
sim.set_max_steps(1000)  # Maximum number of steps

# Run the simulation
for t in range(sim.get_max_steps()):
    sim.step()
    update_lidar_position(t)

    # Print lidar buffer data at each simulation step
    lidar_buffer = lidar.get_buffer()
    print(f"Step {t}: Lidar buffer data - {lidar_buffer}")

    # Save lidar data
    sensor_manager.save_lidar_data()

# Clean up PyChrono
pc.exit()
```

Make sure to replace `"path/to/your/mesh.obj"` with the path to your actual Wavefront .obj file. This script loads the mesh, adds a lidar sensor to it, and simulates the system while updating the lidar's position in an orbit around the mesh. The lidar buffer data is printed at each simulation step, and the lidar data is saved using the specified saving options.
