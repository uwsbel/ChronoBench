Here's a Python script using PyChrono to create the described simulation. Make sure to have the required models and PyChrono installed.

```python
import pychrono as pc
import numpy as np

# Initialize PyChrono environment and core components
pc.Initialize()
sim = pc.Simulation()
sim.SetGravity(pc.Vec3(0, -9.81, 0))
sim.SetTimeStep(0.02)  # 50 FPS
sim.SetMaxSimTime(10.0)

# Add terrain
terrain = pc.CollisionMesh("Highway_col.obj", "Highway_vis.obj")
terrain.SetPosition(pc.Vec3(0, -1, 0))
sim.Add(terrain)

# Initialize HMMWV vehicle
hmmwv = pc.HMMWV()
hmmwv.SetPosition(pc.Vec3(0, 0.5, 0))
hmmwv.SetOrientation(pc.Quat().FromEulerAngles(0, 0, 0))
hmmwv.SetContactMethod(pc.ContactMethod.SOFT)
hmmwv.SetTireModel(pc.TireModel.TMEASY)
sim.Add(hmmwv)

# Set default parameters for vehicle components
for i in range(4):
    wheel = hmmwv.GetWheel(i)
    wheel.SetSteeringAngle(0)
    wheel.SetBrakeTorque(0)
    wheel.SetThrottle(0)

# Set up interactive driver system
class Driver:
    def __init__(self, vehicle):
        self.vehicle = vehicle
        self.steering = 0
        self.throttle = 0
        self.brake = 0

    def Update(self, dt):
        wheels = self.vehicle.GetWheels()
        for i in range(4):
            wheels[i].SetSteeringAngle(self.steering)
            wheels[i].SetThrottle(self.throttle)
            wheels[i].SetBrakeTorque(self.brake)

driver = Driver(hmmwv)

# Simulation loop
while not sim.IsDone():
    # Update driver controls (example: keyboard input)
    driver.steering = pc.GetKey(pc.Key.LEFT) - pc.GetKey(pc.Key.RIGHT)
    driver.throttle = pc.GetKey(pc.Key.UP)
    driver.brake = pc.GetKey(pc.Key.DOWN)

    # Update vehicle dynamics and render the scene
    sim.DoStep()
    driver.Update(sim.GetDeltaTime())

    # Render the scene using Irrlicht
    pc.Render()

# Clean up and finalize PyChrono
pc.Finalize()
```

This script initializes the PyChrono environment, adds a custom terrain and an HMMWV vehicle with specified parameters, sets up an interactive driver system, and runs the simulation loop in real-time at 50 frames per second. Make sure to have the required models (Highway_col.obj, Highway_vis.obj, and HMMWV components) in the same directory as the script.
