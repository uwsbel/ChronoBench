Here's a Python script using PyChrono to create a simulation of an epicyclic gear system with a fixed truss, a rotating bar, and two gears driven by a gear motor. I've also included visualization using the Irrlicht system with appropriate materials and lighting.

```python
import pychrono as chrono
import numpy as np

# Initialize PyChrono environment and core components
sim = chrono.ChSimulation()
sim.SetTimestep(0.005)
sim.SetMaxSteps(100000)

# Create a fixed truss
truss = chrono.ChBodyEasyBox(0.5, 0.5, 0.5, 1000, True)
truss.SetPos(chrono.ChVectorD(0, 0, 0))
truss.SetBodyFixed(True)
sim.Add(truss)

# Create a rotating bar
bar = chrono.ChBodyEasyBox(0.2, 0.2, 2, 100, True)
bar.SetPos(chrono.ChVectorD(0, 0, 1))
bar.SetRot(chrono.Q_from_AngAxis(chrono.CH_C_PI/2, chrono.ChVectorD(0, 0, 1)))
sim.Add(bar)

# Create a gear motor
motor = chrono.ChLinkMotor AngularMotor()
motor.Initialize(bar, chrono.ChFrameD(chrono.ChVectorD(0, 0, 1)), chrono.ChVectorD(0, 0, 1))
motor.SetTargetSpeed(5)
sim.Add(motor)

# Create two gears with one driven by the motor
gear1 = chrono.ChBodyEasyCylinder(0.1, 0.5, 100, True)
gear1.SetPos(chrono.ChVectorD(-0.25, 0, 0))
gear1.SetRot(chrono.Q_from_AngAxis(chrono.CH_C_PI/2, chrono.ChVectorD(0, 0, 1)))
sim.Add(gear1)

gear2 = chrono.ChBodyEasyCylinder(0.1, 0.5, 100, True)
gear2.SetPos(chrono.ChVectorD(0.25, 0, 0))
gear2.SetRot(chrono.Q_from_AngAxis(chrono.CH_C_PI/2, chrono.ChVectorD(0, 0, 1)))
sim.Add(gear2)

# Create a gear mesh between the two gears
gear_mesh = chrono.ChLinkMateMeshAngular()
gear_mesh.Initialize(gear1, gear2, chrono.ChVectorD(0, 0, 1), chrono.ChVectorD(0, 0, 1))
gear_mesh.SetScrewAxis1(chrono.ChVectorD(0, 0, 1))
gear_mesh.SetScrewAxis2(chrono.ChVectorD(0, 0, 1))
sim.Add(gear_mesh)

# Create a gear mesh between the motor gear and the first gear
motor_gear_mesh = chrono.ChLinkMateMeshAngular()
motor_gear_mesh.Initialize(bar, gear1, chrono.ChVectorD(0, 0, 1), chrono.ChVectorD(0, 0, 1))
motor_gear_mesh.SetScrewAxis1(chrono.ChVectorD(0, 0, 1))
motor_gear_mesh.SetScrewAxis2(chrono.ChVectorD(0, 0, 1))
sim.Add(motor_gear_mesh)

# Set up visualization
vis = chrono.ChVisualSystemIrrlicht()
vis.Attach(sim)
vis.SetWindowSize(800, 600)
vis.SetWindowTitle('Epicyclic Gears Simulation')
vis.Initialize()
vis.AddLogo(chrono.ChVectorF(0.5, 0.95, 0))

# Set materials and lighting for visualization
material = chrono.ChVisualMaterial()
material.SetTexture(chrono.GetChronoDataFile('textures/wood.jpg'))
truss.GetAssets().GetVolume().SetMaterial(material)
bar.GetAssets().GetVolume().SetMaterial(material)
gear1.GetAssets().GetVolume().SetMaterial(material)
gear2.GetAssets().GetVolume().SetMaterial(material)

light = chrono.ChLightDirectional()
light.SetDirection(chrono.ChVectorF(3, 3, -5))
vis.AddLight(light)

# Run the simulation
while vis.Run():
    sim.DoStepDynamics(0.005)
    vis.Render()
    vis.Update()
```

This script creates a simple epicyclic gear system with a fixed truss, a rotating bar driven by a gear motor, and two gears meshed together. The simulation is visualized using the Irrlicht system with a wooden texture applied to the objects and appropriate lighting.
