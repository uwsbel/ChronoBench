Here's a Python script using PyChrono to simulate a Turtlebot robot on a rigid terrain as per your instructions. Make sure you have PyChrono and its dependencies installed (`pip install pychrono`).

```python
import pychrono as pc
import numpy as np

# 1. Initialize the PyChrono environment and core components.
chrono = pc.ChSystemNSC()
assert chrono.GetSettings().GetSolverType() == pc.ChSolverType.SOR

# 2. Add the required physical systems and objects.
# Create a ground body.
ground = chrono.GetSystem().GetGround()
ground.SetBodyFixed(True)

# Create a Turtlebot robot.
turtlebot = pc.ChBodyEasyBox(0.1, 0.1, 0.1, 1000, True, True)
turtlebot.SetPos(pc.ChVectorD(0, 0, 0.05))
turtlebot.SetRot(pc.ChQuaternionD(1, 0, 0, 0))  # Set initial orientation to face positive z-axis.
chrono.GetSystem().Add(turtlebot)

# Create two wheels for the Turtlebot.
wheel1 = pc.ChBodyEasyCylinder(0.02, 0.05, 1000, True, True)
wheel1.SetPos(pc.ChVectorD(0.05, 0, 0))
wheel1.SetRot(pc.ChQuaternionD(0, 1, 0, 0))  # Set initial orientation to face positive x-axis.
turtlebot.Add(wheel1, True)

wheel2 = pc.ChBodyEasyCylinder(0.02, 0.05, 1000, True, True)
wheel2.SetPos(pc.ChVectorD(-0.05, 0, 0))
wheel2.SetRot(pc.ChQuaternionD(0, 1, 0, 0))  # Set initial orientation to face positive x-axis.
turtlebot.Add(wheel2, True)

# Create a motor for each wheel.
motor1 = pc.ChLinkMotorLinear()
motor1.Initialize(wheel1, turtlebot, pc.ChFrameD(wheel1.GetPos(), pc.ChQuaternionD(0, 1, 0, 0)), pc.ChFrameD(turtlebot.GetPos(), pc.ChQuaternionD(0, 1, 0, 0)))
motor1.SetActuatorForced(True)
motor1.SetForced(0)  # Initial force (torque) is 0.
chrono.GetSystem().Add(motor1)

motor2 = pc.ChLinkMotorLinear()
motor2.Initialize(wheel2, turtlebot, pc.ChFrameD(wheel2.GetPos(), pc.ChQuaternionD(0, 1, 0, 0)), pc.ChFrameD(turtlebot.GetPos(), pc.ChQuaternionD(0, 1, 0, 0)))
motor2.SetActuatorForced(True)
motor2.SetForced(0)  # Initial force (torque) is 0.
chrono.GetSystem().Add(motor2)

# 3. Set necessary default parameters such as positions, forces, and interactions.
chrono.Set_G(9.81)  # Set gravity.

# Set up real-time visualization using Irrlicht.
vis = pc.ChVisualSystemIrrlicht()
vis.AttachSystem(chrono.GetSystem())
vis.SetWindowSize(800, 600)
vis.Initialize()
vis.AddLight(pc.ChVectorD(0, 0, 3), pc.ChColor(1, 1, 1), 10)

# Simulation loop.
t = 0
while t < 10:
    # Update motor forces for left and right turns at specified times.
    if t > 1 and t < 3:
        motor1.SetForced(1)  # Left wheel moves forward.
        motor2.SetForced(-1)  # Right wheel moves backward.
    elif t > 5 and t < 7:
        motor1.SetForced(-1)  # Left wheel moves backward.
        motor2.SetForced(1)  # Right wheel moves forward.
    else:
        motor1.SetForced(0)  # Both wheels stop.
        motor2.SetForced(0)

    # Advance simulation and render the robot.
    chrono.DoStepDt(0.01)
    vis.Render()
    t += 0.01

vis.GetDevice().closeDevice()
```

This script creates a Turtlebot robot with two wheels and motors, sets up a real-time visualization with lighting, and implements a simulation loop where the Turtlebot's motors control left and right turns at specified times. The robot updates and renders at each timestep.
