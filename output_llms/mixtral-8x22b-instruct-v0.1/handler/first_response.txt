```python
import pychrono as chrono
import pychrono.ros as chrono_ros
import pychrono.irrlicht as chrono_irr
import rospy
from std_msgs.msg import Int32

# Initialize ROS node
rospy.init_node('pychrono_ros_node')

# Initialize Chrono system and set gravity
sys = chrono.ChSystemNSC()
sys.Set_G_acc(chrono.ChVectorD(0, -9.81, 0))

# Define physical material properties
material = chrono.ChMaterialSurfaceNSC()
material.SetFriction(0.4)
material.SetDampingF(0.4)

# Create fixed floor
floor_geometry = chrono.ChBoxGeometry(chrono.ChVectorD(5, 5, 1))
floor_body = chrono.ChBodyEasyBox(material, 5, 5, 1)
floor_body.SetBodyFixed(True)
floor_body.SetPos(chrono.ChVectorD(0, 0, -1))
sys.Add(floor_body)

# Create movable box
box_geometry = chrono.ChBoxGeometry(chrono.ChVectorD(1, 1, 1))
box_body = chrono.ChBodyEasyBox(material, 1, 1, 1)
box_body.SetPos(chrono.ChVectorD(0, 0, 2))
sys.Add(box_body)

# Create custom ROS handler
class CustomROSHandler(chrono_ros.ChROSHandler):
    def __init__(self):
        super().__init__()
        self.publisher = rospy.Publisher('custom_topic', Int32, queue_size=10)

    def OnROSMessage(self, msg):
        # Handle ROS messages here
        pass

    def OnUpdate(self):
        # Publish integer messages to the custom topic
        self.publisher.publish(Int32(10))

# Create Chrono ROS manager
ros_manager = chrono_ros.ChROSManager(sys, chrono_irr.ChIrrApp(sys, 'PyChrono ROS Simulation'))

# Configure ROS manager
ros_manager.Initialize(chrono_ros.ChROSManager.GetROSClock(),
                       chrono_ros.ChROSManager.GetROSBody(),
                       chrono_ros.ChROSManager.GetROSTransform())

# Register custom handler
custom_handler = CustomROSHandler()
ros_manager.AddHandler(custom_handler)

# Run simulation loop
while ros_manager.Run():
    sys.DoStepDynamics(0.01)
    ros_manager.Update()

```

This script initializes the PyChrono environment, sets up a Chrono system with gravity, adds a fixed floor and a movable box, creates a custom ROS handler, configures the ROS manager, and runs a simulation loop while maintaining real-time execution.
