```python
import PyChrono as chrono
import PyChrono.Irrlicht as chronoirr
import PyChrono.Vehicle as chronovehicle
import PyChrono.Postprocess as chronopost

# Initialize the PyChrono system
chrono.SetChronoDataPath("path/to/chrono/data")
chrono.SetChronoOutputMode(chrono.CHRONO_OUTPUT_ASCII)
my_system = chrono.ChSystemNSC()

# Add the Kraz vehicle
my_vehicle = chronovehicle.ChVehicle("Kraz")
my_vehicle.SetContactMethod(chrono.ChContactMethod_NSC)
my_vehicle.SetChassisCollisionType(chronovehicle.ChVehicleCollisionType_BOX)
my_vehicle.SetChassisFixed(False)
my_vehicle.SetInitPosition(chrono.ChVectorD(0, 0, 1.0))
my_vehicle.SetInitQuaternion(chrono.ChQuaternionD(1, 0, 0, 0))
my_vehicle.Initialize(my_system)

# Add the rigid terrain
my_terrain = chronovehicle.ChVehicleTerrain("terrain")
my_terrain.SetContactFriction(0.9)
my_terrain.SetContactRestitution(0.01)
my_terrain.Initialize(my_system)

# Add the driver system
my_driver = chronovehicle.ChIrrGuiDriver("driver")
my_driver.Initialize(my_system)

# Add the Irrlicht visualization
my_visual = chronoirr.ChIrrApp(my_system, "PyChrono Vehicle Simulation", chronoirr.dimension2du(800, 600))
my_visual.AddTypicalSky()
my_visual.AddTypicalLogo(chrono.GetChronoDataFile("logo_pychrono_alpha.png"))
my_visual.AddTypicalCamera(chrono.ChVectorD(0.0, 0.0, 1.5), chrono.ChVectorD(0.0, 0.0, 0.0))
my_visual.AddLightWithShadow(chrono.ChVectorD(2.0, 2.0, 2.0), chrono.ChVectorD(0.0, 0.0, 0.0), 1.0, 0.5, 50.0, 100.0, 512, 2048)
my_visual.AddLightWithShadow(chrono.ChVectorD(-2.0, 2.0, 2.0), chrono.ChVectorD(0.0, 0.0, 0.0), 1.0, 0.5, 50.0, 100.0, 512, 2048)
my_visual.AddLightWithShadow(chrono.ChVectorD(2.0, -2.0, 2.0), chrono.ChVectorD(0.0, 0.0, 0.0), 1.0, 0.5, 50.0, 100.0, 512, 2048)
my_visual.AddLightWithShadow(chrono.ChVectorD(-2.0, -2.0, 2.0), chrono.ChVectorD(0.0, 0.0, 0.0), 1.0, 0.5, 50.0, 100.0, 512, 2048)
my_visual.AddLight(chrono.ChVectorD(0.0, 0.0, 2.0), chrono.ChColor(0.5, 0.5, 0.5), 1.0, 100.0)

# Simulation loop
my_visual.SetTimestep(0.01)
my_visual.SetTryRealtime(True)

while my_visual.Run():
    my_system.DoStepDynamics(my_visual.GetTimestep())
    my_visual.BeginScene()
    my_visual.DrawAll()
    my_visual.EndScene()

    # Update the driver system
    my_driver.Synchronize(my_vehicle)

    # Advance the vehicle, terrain, and driver systems
    my_vehicle.Advance(my_visual.GetTimestep())
    my_terrain.Advance(my_visual.GetTimestep())
    my_driver.Advance(my_visual.GetTimestep())
```

This script creates a PyChrono simulation with a Kraz vehicle, a rigid terrain, a driver system, and a real-time visualization using Irrlicht. The simulation loop synchronizes and advances the vehicle, terrain, driver, and visual system at each timestep, maintaining real-time execution.
