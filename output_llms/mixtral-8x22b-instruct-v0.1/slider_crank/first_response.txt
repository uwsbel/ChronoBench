Here is a basic script to create a crank-slider mechanism simulation using PyChrono:

```python
import PyChrono as chrono
import PyChrono.Irrlicht as chronoirr
import numpy as np

# Initialize PyChrono and Irrlicht visualization system
chrono.SetChronoDataPath("path/to/your/chrono/data")
chrono.SetChronoDataPath("path/to/your/irrlicht/data")

# Create a Chrono::Engine physical system
my_system = chrono.ChSystemNSC()

# Create a floor (truss)
floor = chrono.ChBodyEasyBox(2, 0.2, 1, 1000, True, True)
floor.SetPos(chrono.ChVectorD(0, -0.1, 0))
floor.SetBodyFixed(True)
my_system.AddBody(floor)

# Create a crankshaft
crankshaft = chrono.ChBodyEasyCylinder(0.1, 0.5, 1000, True, True)
crankshaft.SetPos(chrono.ChVectorD(0, 0, 0))
my_system.AddBody(crankshaft)

# Create a connecting rod
connecting_rod = chrono.ChBodyEasyBox(0.2, 0.1, 1, 1000, True, True)
connecting_rod.SetPos(chrono.ChVectorD(0.5, 0, 0))
my_system.AddBody(connecting_rod)

# Create a piston
piston = chrono.ChBodyEasyCylinder(0.1, 0.5, 1000, True, True)
piston.SetPos(chrono.ChVectorD(1.2, 0, 0))
my_system.AddBody(piston)

# Create joints and constraints
revolute_joint = chrono.ChLinkLockRevolute()
revolute_joint.Initialize(floor, crankshaft, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngY(np.pi/2)))
my_system.AddLink(revolute_joint)

revolute_joint2 = chrono.ChLinkLockRevolute()
revolute_joint2.Initialize(crankshaft, connecting_rod, chrono.ChCoordsysD(chrono.ChVectorD(0.5, 0, 0), chrono.Q_from_AngY(0)))
my_system.AddLink(revolute_joint2)

slider_joint = chrono.ChLinkLockPrismatic()
slider_joint.Initialize(floor, piston, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngY(np.pi/2)))
my_system.AddLink(slider_joint)

# Create a motor to spin the crankshaft
motor = chrono.ChLinkMotorRotationSpeed()
motor.Initialize(floor, crankshaft, chrono.ChCoordsysD(chrono.ChVectorD(0, 0, 0), chrono.Q_from_AngY(np.pi/2)))
motor.Set_motor_speed_dt(chrono.ChFunction_Const(10 * chrono.CH_C_PI))  # 10 rad/s
my_system.AddLink(motor)

# Create an Irrlicht application
application = chronoirr.ChIrrApp(my_system, "Crank-Slider Mechanism", chronoirr.dimension2du(800, 600), False, True)

# Set custom camera settings, lighting, textures, and logos
application.AddTypicalSky()
application.AddTypicalLogo()
application.AddTypicalLights()
application.AddTypicalCamera(chrono.ChVectorD(1, 1, -1), chrono.ChVectorD(0, 0, 0))

# Run the simulation
application.Run()
```

Replace "path/to/your/chrono/data" and "path/to/your/irrlicht/data" with the actual paths to your Chrono and Irrlicht data directories. You can customize the camera settings, lighting, textures, and logos by modifying the corresponding functions in the script.

This script creates a simple crank-slider mechanism with a floor, crankshaft, connecting rod, and piston. The crankshaft is driven by a motor to spin at a constant angular speed. The simulation is visualized using the Irrlicht visualization system with customizable camera settings, lighting, and additional visual elements like textures and logos.
