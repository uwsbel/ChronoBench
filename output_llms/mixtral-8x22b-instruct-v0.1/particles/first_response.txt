Here is a basic implementation of the problem using PyChrono. Please note that you need to have PyChrono and Irrlicht installed in your Python environment.

```python
import pychrono.core as chrono
import pychrono.irrlicht as chronoirr
import numpy as np

# Initialize the PyChrono environment and core components
chrono.SetChronoDataPath('path_to_chrono_data')

# Create a Chrono system
my_system = chrono.ChSystemNSC()

# Create a particle emitter
class ParticleEmitter:
    def __init__(self, system):
        self.system = system
        self.particles = []

    def emit_particle(self, shape, position, velocity, orientation):
        particle = chrono.ChBodyEasySphere(shape, 1000, True, True, False)
        particle.SetPos(chrono.ChVectorD(*position))
        particle.SetPos_dt(chrono.ChVectorD(*velocity))
        particle.SetRot(chrono.ChQuaternionD(*orientation))
        self.system.Add(particle)
        self.particles.append(particle)

# Create an instance of the particle emitter
emitter = ParticleEmitter(my_system)

# Set necessary default parameters
for _ in range(10):
    shape = np.random.uniform(0.1, 1)
    position = np.random.uniform(-10, 10, 3)
    velocity = np.random.uniform(-1, 1, 3)
    orientation = chrono.Q_from_AngX(np.random.uniform(0, 360))
    emitter.emit_particle(shape, position, velocity, orientation)

# Apply custom gravitational forces to simulate particle attraction dynamically
for particle in emitter.particles:
    for other_particle in emitter.particles:
        if particle != other_particle:
            force = chrono.ChForceGravity()
            force.SetGravity(chrono.ChVectorD(0, 0, -9.81))
            force.SetBody(particle)
            force.SetPoint(other_particle.GetPos())
            my_system.Add(force)

# Visualize the simulation using Irrlicht
my_application = chronoirr.ChIrrApp(my_system, 'PyChrono Simulation', chronoirr.dimension2du(800, 600))
my_application.AddTypicalSky()
my_application.AddTypicalLogo()
my_application.AddTypicalCamera(chronoirr.vector3df(0, 0, -10))
my_application.AddLightWithShadow(chronoirr.vector3df(2, 5, -2), 40, 1, 2, 10, 50, 100)
my_application.AddLightWithShadow(chronoirr.vector3df(-2, 5, -2), 40, 1, 2, 10, 50, 100)

# Run the simulation
while my_application.GetDevice().run():
    my_application.BeginScene()
    my_application.DrawAll()
    my_application.DoStep()
    my_application.EndScene()
```

Please replace `'path_to_chrono_data'` with the actual path to the Chrono data directory. This script will create a PyChrono simulation with 10 particles with random shapes, positions, velocities, and orientations. The particles will be attracted to each other due to custom gravitational forces. The simulation will be visualized using Irrlicht.
